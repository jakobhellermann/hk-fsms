# Bottle Control
# 2 location(s):
#   Mines_16 / Grub Mimic Bottle
#   Deepnest_36 / Grub Mimic Bottle

fsm Bottle Control {
  start Pool

  state Idle {
    FindChild(gameObject=Self, childName="Bottle Physical", storeResult=var "Phys Box")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Shatter
  }

  state Shatter {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[crystal_break_1 (resources.assets), crystal_break_2 (resources.assets)], weights=[1, 1], pitchMin=1.3, pitchMax=1.3, volume=1, delay=0, storePlayer=<null>)
    DestroyObject(gameObject=var "Phys Box", delay=0, detachChildren=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Grub"), sendEvent="FREE", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Destroy Self
  }

  state Destroy Self {
    SetCollider(gameObject=Self, active=false)
    SpawnRandomObjects(gameObject=Bottle Lid (sharedassets33.assets), spawnPoint=var "Self", position=(0, 1.69, 0), spawnMin=1, spawnMax=1, speedMin=7, speedMax=8, angleMin=85, angleMax=95, originVariation=0)
    ActivateGameObject(gameObject=var "Debris", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObjects(containerObject=var "Debris", adjustPosition=(unset), randomisePosition=false, speedMin=10, speedMax=15, angleMin=40, angleMax=140)
    PlayParticleEmitter(gameObject=var "Pt Glass L", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Glass S", emit=0)
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  state Activate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GRUB SAVED", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    DestroyAllChildren(gameObject=var "Self", disable=false)
    SpawnRandomObjects(gameObject=Bottle Lid (sharedassets33.assets), spawnPoint=var "Self", position=(0, -0.5, 0), spawnMin=1, spawnMax=1, speedMin=0, speedMax=0, angleMin=85, angleMax=95, originVariation=0)
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Grub"), sendEvent="IN BOTTLE", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Bottle Physical", storeResult=var "Phys Box")
    ActivateGameObject(gameObject=var "Phys Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Pool {
    GetOwner(storeGameObject=var "Self")
    SetParent(gameObject=var "Pool", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Pool", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  var Grub: gameObject = Grub Mimic Top

  var Attack Type: int = 0
  var Activated: bool = false
  var Damager: gameObject = <null>
  var Debris: gameObject = Grub Mimic Top
  var Phys Box: gameObject = Grub Mimic Top
  var Pool: gameObject = <null>
  var Pt Glass L: gameObject = Grub Mimic Bottle/Pt Glass L
  var Pt Glass S: gameObject = Grub Mimic Bottle/Pt Glass S
  var Self: gameObject = <null>
}
