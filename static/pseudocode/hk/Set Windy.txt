# Set Windy
# 2 location(s):
#   Abyss_18 / Windy_Set
#   Abyss_19 / Windy_Set

fsm Set Windy {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    FindChild(gameObject=Self, childName="grasses", storeResult=var "grasses")
    on FINISHED → Got Doublejump?
  }

  state Got Doublejump? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDoubleJump", isTrue=→"YES", isFalse=→"NO")
    on YES → Unwindy
    on NO → Windy
  }

  state Unwindy {
    SendEventByName(eventTarget=GameObject(var "grasses"), sendEvent="STOP WINDY", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName="vines static", storeResult=var "vines static")
    ActivateGameObject(gameObject=var "vines static", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=0.1)
    SendMessageByTag(tag="WindyGrass", message="SetNotWindy")
  }

  state Windy {
    SendEventByName(eventTarget=GameObject(var "grasses"), sendEvent="START WINDY", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName="vines", storeResult=var "vines")
    ActivateGameObject(gameObject=var "vines", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessageByTag(tag="WindyGrass", message="SetWindy")
  }

  var grasses: gameObject = <null>
  var vines: gameObject = <null>
  var vines static: gameObject = <null>
}
