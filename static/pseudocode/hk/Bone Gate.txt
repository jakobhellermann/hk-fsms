# Bone Gate
# 1 location(s):
#   Crossroads_ShamanTemple / Bone Gate

fsm Bone Gate {
  start Pause Frame

  state Idle {
    GetOwner(storeGameObject=var "Self")
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=true)
    on OPEN → Open Audio
    on ACTIVATED → Deactivate
  }

  state Open Audio {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.7, finishEvent=(none), realTime=false)
    on FINISHED → Open Anim
  }

  state Deactivate {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Open Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Open")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Deactivate
  }

  var Shaman: int = 0
  var Activated: bool = false
  var Dust: gameObject = <null>
  var Self: gameObject = <null>
}
