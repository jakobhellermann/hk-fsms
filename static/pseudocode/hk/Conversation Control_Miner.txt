# Conversation Control
# 2 location(s):
#   Crossroads_45 / Miner
#   sharedassets471.assets / Miner

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Stop Anim  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Convo Choice {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)  // disabled
    AudioStop(gameObject=Self)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MINER", everyFrame=false)
    GetPlayerDataBool(boolName="metMiner", storeValue=var "Met")
    GetPlayerDataInt(gameObject=var "GameManager", intName="miner", storeValue=var "NPC State")
    GetPlayerDataInt(gameObject=var "GameManager", intName="fireballLevel", storeValue=var "Fireball Level")
    IntCompare(integer1=var "Fireball Level", integer2=0, equal=(none), lessThan=(none), greaterThan=→"EARLY", everyFrame=false)
    BoolTest(boolVariable=var "Met", isTrue=→"MEET REPEAT", isFalse=→"MEET", everyFrame=false)
    on MEET → Meet 1
    on MEET REPEAT → Meet Repeat
    on EARLY → Early Choice
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")  // disabled
    AudioStop(gameObject=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")  // disabled
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Meet 1 {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Miner_01_sing (sharedassets73.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataBool(boolName="metMiner", value=true)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_MEET_1", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Meet 1 B
  }

  state Stop Anim {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Play Talk End", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Meet 1 B {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Miner_05 (sharedassets73.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="metMiner", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_MEET_1_B", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Meet Repeat {
    SetPlayerDataBool(boolName="metMiner", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Miner_01_sing (sharedassets73.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_MEET_REPEAT", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Early Choice {
    BoolTest(boolVariable=var "Met", isTrue=(none), isFalse=→"MEET", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="minerEarly", storeValue=var "Early")
    IntSwitch(intVariable=var "Early", compareTo=[0, 1, 2], sendEvent=[→"EARLY 1", →"EARLY 2", →"EARLY 3"], everyFrame=false)
    on EARLY 1 → Early 1
    on EARLY 2 → Early 2
    on EARLY 3 → Early 3
    on MEET → Meet Repeat
  }

  state Early 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Miner_01 (sharedassets73.assets), Miner_02 (sharedassets73.assets), Miner_03 (sharedassets73.assets), Miner_04 (sharedassets73.assets), Miner_05 (sharedassets73.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=false, everyFrame=false)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="minerEarly")
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_EARLY_1", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Early 1 B
  }

  state Early 1 B {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_EARLY_1_B", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Early 2 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="minerEarly")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Miner_01_sing (sharedassets73.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_EARLY_2", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Early 2 B
  }

  state Early 2 B {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Miner_01 (sharedassets73.assets), Miner_02 (sharedassets73.assets), Miner_03 (sharedassets73.assets), Miner_04 (sharedassets73.assets), Miner_05 (sharedassets73.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_EARLY_2_B", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Early 3 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Miner_05 (sharedassets73.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MINER_EARLY_3", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Early: int = 0
  var Fireball Level: int = 0
  var NPC State: int = 0
  var Met: bool = false
  var Play Talk End: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
