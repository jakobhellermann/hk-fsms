# Kill Counter
# 1 location(s):
#   GG_Mighty_Zote / Battle Control

fsm Kill Counter {
  start Idle

  state Idle {
    on ZOTE WARRIOR KILLED → Increment
    on ZOTELING KILLED → Increment
  }

  state Increment {
    GetEventSender(sentByGameObject=var "Event Sender")
    GetFsmInt(gameObject=Self, fsmName="Control", variableName="Kills", storeValue=var "Kills", everyFrame=false)
    IntAdd(intVariable=var "Kills", add=1, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Kills", setValue=var "Kills", everyFrame=false)
    on FINISHED → Counter Active?
  }

  state Counter Active? {
    BoolTest(boolVariable=var "Counter Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Counter Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Counter Icon", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on ACTIVATED → Do Gold?
    on FINISHED → Set Text
  }

  state Set Text {
    ConvertIntToString(intVariable=var "Kills", stringVariable=var "Text", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Counter Text", textString=var "Text")
    on FINISHED → Idle
  }

  state Counter Tick {
    Tk2dPlayAnimation(gameObject=var "Counter Icon", animLibName="", clipName=var "Tick Anim")
    Tk2dPlayFrame(gameObject=var "Counter Icon", frame=0)
    on FINISHED → Set Text
  }

  state Do Gold? {
    BoolTest(boolVariable=var "Is Gold", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    IntCompare(integer1=var "Kills", integer2=57, equal=→"GOLD", lessThan=(none), greaterThan=→"GOLD", everyFrame=false)
    on FINISHED → Counter Tick
    on GOLD → Become Gold
  }

  state Become Gold {
    SetBoolValue(boolVariable=var "Is Gold", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Counter Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetStringValue(stringVariable=var "Tick Anim", stringValue="Counter Tick Gold", everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=boss_stun (resources.assets), pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Counter Tick
  }

  var Counter Flash: gameObject = Battle Control/Counter Flash
  var Counter Icon: gameObject = Battle Control/Counter Icon
  var Counter Text: gameObject = Battle Control/Counter Text

  var Kills: int = 0
  var Counter Activated: bool = false
  var Is Gold: bool = false
  var Text: string = ""
  var Tick Anim: string = "Counter Tick"
  var Event Sender: gameObject = <null>
}
