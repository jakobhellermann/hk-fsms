// uses template: radiant_bg_shade
fsm radiant_bg_shade {
  start Init

  state Init {
    SelectRandomString(strings=["BG Shade 1", "BG Shade 2", "BG Shade 3", "BG Shade 4"], weights=[1, 1, 1, 1], storeString=var "Anim")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim")
    RandomFloat(min=4, max=30, storeResult=var "Accel")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Accel", space=0, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Pause
  }

  state Appear {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-24, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 24, 0), time=2.3, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Drift
  }

  state Drift {
    IdleBuzzV3(gameObject=Self, waitMin=0.25, waitMax=0.75, speedMax=2, accelerationMin=5, accelerationMax=10, roamingRangeX=1, roamingRangeY=1, manualStartPos=(unset))
  }

  state Pause {
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Appear
  }

  var Offset Y: float = 0

  var Accel: float = 0
  var Hero Y: float = 0
  var Self Y: float = 0
  var Target Y: float = 0
  var Anim: string = ""
  var Self Pos: vector3 = (0, 0, 0)
}
