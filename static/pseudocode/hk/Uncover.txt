# Uncover
# 1 location(s):
#   GG_Workshop / Zote_Break_wall/Masks

fsm Uncover {
  start Uncover Instant

  state Uncover Instant {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    Wait(time=1, finishEvent=→"CANCEL", realTime=false)
    on UNCOVER → Uncover
    on CANCEL → Wait Uncover
  }

  state Uncover {
    ActivateGameObject(gameObject=var "Enable", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "First Appear", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  state Wait Uncover {
    on UNCOVER → Enable
  }

  state Reposition {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)  // disabled
    SetPosition(gameObject=var "Hero", vector=(unset), x=198.5, y=63.5, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Release
  }

  state Enable {
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Enable", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "First Appear", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Masker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Disable Mask
  }

  state Release {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
  }

  state Disable Mask {
    ActivateGameObject(gameObject=var "Masker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reposition
  }

  var Enable: gameObject = Zote_Appear_bits
  var First Appear: gameObject = Zote_Appear_bits/first_appear_sequence
  var Masker: gameObject = Zote_Break_wall/masker

  var Self: gameObject = <null>
}
