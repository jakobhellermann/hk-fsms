# Attacking
# 4 location(s):
#   Deepnest_East_10 / Warrior/Ghost Warrior Markoth
#   Deepnest_East_16 / Warrior/Ghost Warrior Markoth
#   GG_Ghost_Markoth / Warrior/Ghost Warrior Markoth
#   GG_Ghost_Markoth_V / Warrior/Ghost Warrior Markoth

fsm Attacking {
  start Init
  on ATTACK STOP → Attack Stop  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Markoth_cast_calm (sharedassets314.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on READY → Wait
  }

  state Wait {
    BoolTest(boolVariable=var "Rage", isTrue=→"RAGE", isFalse=(none), everyFrame=false)
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"ATTACK", realTime=false)
    on FINISHED → 
    on TOOK DAMAGE → 
    on ATTACK → Nail
    on RAGE → Wait rage
  }

  state Attack Stop {
    on ATTACK OK → Wait
  }

  state Nail {
    SpawnObjectFromGlobalPool(gameObject=Shot Markoth Nail (sharedassets314.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Wait
  }

  state Wait rage {
    WaitRandom(timeMin=0.75, timeMax=1.25, finishEvent=→"ATTACK", realTime=false)
    on FINISHED → Nail
  }

  var Double HP: int = 140

  var Shot Angle: float = 0
  var HP: int = 0
  var Repeats: int = 0
  var Triple HP: int = 70
  var Hero Away: bool = false
  var Rage: bool = false
  var Attack Pt: gameObject = Warrior/Ghost Warrior Markoth/Attack Pt
  var Self: gameObject = <null>
  var Shield 1: gameObject = <null>
  var Shield 2: gameObject = <null>
  var Shot: gameObject = <null>
}
