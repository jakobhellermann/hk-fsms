// component disabled: this FSM does not run until something enables it
fsm fat fly bounce {
  start Initialise
  on STOP → Stopped  // from any state
  on GO UP → Go Up  // from any state

  state Initialise {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    SetGameObject(variable=var "Camera", gameObject=var "MainCamera", everyFrame=false)
    RandomFloat(min=0, max=360, storeResult=var "Angle")  // disabled
    BoolTest(boolVariable=var "Starts Inactive", isTrue=→"STOP", isFalse=(none), everyFrame=false)
    GetDistance(gameObject=var "Self", target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=25, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Aim
  }

  state Left or Right? {
    FloatSwitch(floatVariable=var "Angle", lessThan=[90, 270, 360], sendEvent=[→"RIGHT", →"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Face Left
    on RIGHT → Face Right
  }

  state Face Left {
    SetScale(gameObject=var "Self", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    SetFloatValue(floatVariable=var "Raycast Right", floatValue=0, everyFrame=false)  // disabled
    SetFloatValue(floatVariable=var "Raycast Left", floatValue=var "Raycast X", everyFrame=false)  // disabled
    on FINISHED → Fly 2
  }

  state Face Right {
    SetScale(gameObject=var "Self", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    SetFloatValue(floatVariable=var "Raycast Right", floatValue=var "Raycast X", everyFrame=false)  // disabled
    SetFloatValue(floatVariable=var "Raycast Left", floatValue=0, everyFrame=false)  // disabled
    on FINISHED → Fly 2
  }

  state Hit Up {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Up Right
    on LEFT → Up Left
  }

  state Up Right {
    RandomFloat(min=320, max=350, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Up Left {
    RandomFloat(min=190, max=220, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Down {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Down Right
    on LEFT → Up Left 2
  }

  state Down Right {
    RandomFloat(min=10, max=40, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Up Left 2 {
    RandomFloat(min=140, max=170, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Right {
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Down Right 2
    on DOWN → Right Down
  }

  state Right Down {
    RandomFloat(min=190, max=220, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Down Right 2 {
    RandomFloat(min=140, max=170, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Left {
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Left Up
    on DOWN → Left Down
  }

  state Left Down {
    RandomFloat(min=320, max=350, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Left Up {
    RandomFloat(min=10, max=40, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Fly 2 {
    SetVelocityAsAngle(gameObject=var "Self", angle=var "Angle", speed=var "Speed", everyFrame=true)
    Collision2dEvent(collision=1, collideTag="", sendEvent=→"COLLISION STAY 2D", storeCollider=var "", storeForce=(unset))
    on COLLISION STAY 2D → Collision Check
  }

  state Collision Check {
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=(unset), contactNormal=var "Contact Normal", shapeCount=(unset), physics2dMaterialName="")
    GetVector3XYZ(vector3Variable=var "Contact Normal", storeX=var "Normal X", storeY=var "Normal Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal X", float2=-1, tolerance=0, equal=→"HIT RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Normal X", float2=1, tolerance=0, equal=→"HIT LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=-1, tolerance=0, equal=→"HIT UP", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=1, tolerance=0, equal=→"HIT DOWN", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendRandomEvent(events=[→"HIT DOWN", →"HIT LEFT", →"HIT RIGHT", →"HIT UP"], weights=[1, 1, 1, 1], delay=0)
    on HIT DOWN → Hit Down
    on HIT LEFT → Hit Left
    on HIT RIGHT → Hit Right
    on HIT UP → Hit Up
    on FINISHED → Hit Right
  }

  state Stopped {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    SetBoolValue(boolVariable=var "Starts Inactive", boolValue=false, everyFrame=false)
    on WAKE → Left or Right?
  }

  state Aim {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="START", delay=0, everyFrame=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", everyFrame=false)
    GetAngleToTarget(gameObject=var "Self", targetObject=var "Hero", targetPosition=(unset), ignoreHeight=true, storeAngle=var "Angle", everyFrame=false)  // disabled
    on FINISHED → Left or Right?
  }

  state Go Up {
    RandomFloat(min=20, max=160, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  var Speed: float = 4
  var Starts Inactive: bool = false
  var Contact Normal: vector3 = (0, 0, 0)

  var Angle: float = 0
  var Distance: float = 0
  var Normal X: float = 0
  var Normal Y: float = 0
  var Raycast Left: float = 0
  var Raycast Right: float = 0
  var Raycast X: float = 0
  var Facing Right: bool = false
  var Camera: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
