fsm Check {
  start Detect

  state Detect {
    NextFrameEvent(sendEvent=→"FINISHED")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"TERRAIN HIT", storeCollider=var "")
    on TERRAIN HIT → Send
    on FINISHED → Pause2
  }

  state Send {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="ENEMY HIT", delay=0, everyFrame=false)
  }

  state Inert {
  }

  state Pause2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Inert
  }

  var Parent: gameObject = <null>
}
