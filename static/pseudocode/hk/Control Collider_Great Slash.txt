# Control Collider
# 2 location(s):
#   Knight_Pickup / Knight/Attacks/Great Slash
#   resources.assets / Knight:1/Attacks/Great Slash

fsm Control Collider {
  start Init

  state Check Start {
    SetIntValue(intVariable=var "Frame Count", intValue=0, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false)
    IntAddV2(intVariable=var "Frame Count", add=1, everyFrame=true)
    IntCompare(integer1=var "Frame Count", integer2=7, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Enable
  }

  state Init {
    SetIntValue(intVariable=var "Frame Count", intValue=0, everyFrame=false)  // disabled
    IntAddV2(intVariable=var "Frame Count", add=1, everyFrame=true)  // disabled
    SetScale(gameObject=Self, vector=(unset), x=1.316208, y=1.316208, z=1.316208, everyFrame=false, lateUpdate=false)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NA Big Slash Effect")
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Check Start
  }

  state Enable {
    SetPolygonCollider(gameObject=Self, active=true)
    IntAddV2(intVariable=var "Frame Count", add=1, everyFrame=true)
    IntCompare(integer1=var "Frame Count", integer2=18, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Disable
  }

  state Disable {
    SetPolygonCollider(gameObject=Self, active=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetParent(gameObject=Self, storeResult=var "Parent")  // disabled
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1.316208, y=1.316208, z=1.316208, everyFrame=false, lateUpdate=false)
  }

  var Frame Count: int = 0

  var Start Pos: vector3 = (-0.68, 0, 0)
  var Parent: gameObject = <null>
}
