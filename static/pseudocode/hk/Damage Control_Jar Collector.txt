# Damage Control
# 3 location(s):
#   Ruins2_11_boss / Battle Scene/Jar Collector
#   GG_Collector / Battle Scene/Jar Collector
#   GG_Collector_V / Battle Scene/Jar Collector

fsm Damage Control {
  start Idle

  state Idle {
    on TOOK DAMAGE → Check
  }

  state Check {
    BoolTest(boolVariable=var "Phase 2", isTrue=→"DECREMENT", isFalse=(none), everyFrame=false)
    SendRandomEvent(events=[→"DECREMENT", →"FINISHED"], weights=[1, 1], delay=0)
    on FINISHED → Idle
    on DECREMENT → Dec
  }

  state Dec {
    GetFsmInt(gameObject=Self, fsmName="Control", variableName="Hops", storeValue=var "Hops", everyFrame=false)
    IntOperator(integer1=var "Hops", integer2=1, operation=Subtract, storeResult=var "Hops", everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Hops", setValue=var "Hops", everyFrame=false)
    on FINISHED → Idle
  }

  var Hops: int = 0
  var Phase 2: bool = false
}
