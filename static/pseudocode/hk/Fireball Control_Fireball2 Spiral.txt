# Fireball Control
# 1 location(s):
#   resources.assets / Fireball2 Spiral

fsm Fireball Control {
  start Set Damage

  state Idle {
    Wait(time=0.475, finishEvent=→"FINISHED", realTime=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Crt X Velocity", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Crt X Velocity", float2=0, tolerance=0.1, equal=→"WALL", lessThan=(none), greaterThan=(none), everyFrame=true)
    on FINISHED → Shrink
  }

  state Shrink {
    StopParticleEmitter(gameObject=var "Dribble")
    StopParticleEmitter(gameObject=var "Particle Emitter")
    StopParticleEmitter(gameObject=var "Particle Emitter B")
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "White Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    iTweenScaleBy(gameObject=Self, id="", vector=(0.75, 0.25, 1), time=0.075, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Off
  }

  state Init {
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")  // disabled
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=false)  // disabled
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=→"RIGHT", lessThan=→"RIGHT", greaterThan=→"LEFT", everyFrame=false)
    ActivateGameObject(gameObject=var "White Light", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on LEFT → L
    on RIGHT → R
  }

  state L {
    FindChild(gameObject=Self, childName="Dribble L", storeResult=var "Dribble")
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state R {
    FindChild(gameObject=Self, childName="Dribble R", storeResult=var "Dribble")
    on FINISHED → Idle
  }

  state Set Damage {
    SetScale(gameObject=Self, vector=(unset), x=1.8, y=1.8, z=(unset), everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=30, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_19", isTrue=(none), isFalse=→"FINISHED")
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=40, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=var "Y Scale", zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=1.3, everyFrame=false)
    FloatMultiply(floatVariable=var "Y Scale", multiplyBy=1.6, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=var "Y Scale", z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Init
  }

  state Recycle {
    RecycleSelf()
  }

  state Off {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  var Crt X Velocity: float = 0
  var Velocity: float = 0
  var X Scale: float = 0
  var Y Scale: float = 0
  var Camera: gameObject = <null>
  var Dribble: gameObject = <null>
  var Particle Emitter: gameObject = Fireball2 Spiral/Particle System
  var Particle Emitter B: gameObject = Fireball2 Spiral/Particle System B
  var Terrain Checker: gameObject = <null>
  var Wall Impact: gameObject = Fireball2 Spiral/Wall Impact
  var White Light: gameObject = Fireball2 Spiral/white_light
}
