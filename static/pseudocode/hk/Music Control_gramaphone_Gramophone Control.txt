# Music Control
# 1 location(s):
#   Fungus3_50 / Gramophone Control/gramaphone

fsm Music Control {
  start Init
  on BROKEN → Stop  // from any state

  state Init {
    on FINISHED → Start Pause
  }

  state Start Pause {
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Play
  }

  state Play {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    FadeAudio(gameObject=Self, startVolume=0, endVolume=0.85, time=0.75)
    on BREAK → Stop
  }

  state Stop {
    AudioStop(gameObject=Self)
  }
}
