# Anim
# 1 location(s):
#   Fungus3_34 / Cloth NPC QG Entrance

fsm Anim {
  start Inert

  state Inert {
    GetPosition(gameObject=Self, vector=(unset), x=var "Left X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Right X", floatValue=var "Left X", everyFrame=false)
    FloatAdd(floatVariable=var "Left X", add=-0.5, everyFrame=false, perSecond=false)
    on FINISHED → Idle
  }

  state Idle {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Left X", tolerance=0, equal=(none), lessThan=→"TURN L", greaterThan=(none), everyFrame=true)
    on TURN L → Turn L
  }

  state Turn L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="TurnL")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Left X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"TURN R", everyFrame=true)
    on TURN R → Turn R
  }

  state Turn R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="TurnR")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Left X", tolerance=0, equal=(none), lessThan=→"TURN L", greaterThan=(none), everyFrame=true)
    on TURN L → Turn L
  }

  var Hero X: float = 0
  var Left X: float = 0
  var Right X: float = 0
}
