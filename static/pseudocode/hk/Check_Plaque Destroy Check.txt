# Check
# 1 location(s):
#   Dream_Room_Believer_Shrine / Plaque Destroy Check

fsm Check {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Check
  }

  state Check {
    GetTagCount(tag="Extra Tag", storeResult=var "Plaques")
    IntCompare(integer1=var "Plaques", integer2=0, equal=→"SET", lessThan=→"SET", greaterThan=→"FINISHED", everyFrame=false)
    on SET → Set
    on FINISHED → Idle
  }

  state Set {
    SetPlayerDataBool(boolName="allBelieverTabletsDestroyed", value=true)
  }

  var Plaques: int = 0
}
