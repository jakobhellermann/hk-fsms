# FSM
# 1 location(s):
#   Crossroads_08 / Break Wall 2

// uses template: breakable_wall
fsm FSM {
  start Pause

  state Idle {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"SPELL", storeCollider=var "")
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on TAKE DAMAGE → Check If Nail
    on SPELL → Spell Destroy
  }

  state Check If Nail {
    GetEventSender(sentByGameObject=var "Damager")
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0, 2], sendEvent=[→"TRUE", →"SPELL"], everyFrame=false)
    on TRUE → Hit
    on FINISHED → Idle
    on SPELL → Spell Destroy
  }

  state Hit {
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Strike Effect")
    SetScale(gameObject=var "Strike Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Direction
  }

  state Initiate {
    FindChild(gameObject=Self, childName="Particle_rocks_large", storeResult=var "Particles Large")
    FindChild(gameObject=Self, childName="Particle_rocks_small", storeResult=var "Particles Small")
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Dust Break 1", storeResult=var "Dust Break 1")
    FindChild(gameObject=var "Self", childName="Dust Break 2", storeResult=var "Dust Break 2")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    SetFloatValue(floatVariable=var "Recoil Speed Neg", floatValue=var "Recoil Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Recoil Speed Neg", multiplyBy=-1, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    FindGameObject(objectName=var "Mask Name", withTag="Untagged", store=var "Mask")
    FindGameObject(objectName=var "CamLock Name", withTag="Untagged", store=var "CamLock")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activated
  }

  state Check Direction {
    RandomFloat(min=0.85, max=1.1, storeResult=var "Chooser")
    SetAudioPitch(gameObject=var "Self", pitch=var "Chooser", everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15)
    IntSwitch(intVariable=var "Facing", compareTo=[0, 1, 2, 3], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on RIGHT → Hit Right
    on UP → Hit Up
    on LEFT → Hit Left
    on DOWN → Hit Down
  }

  state Break {
    PlayParticleEmitter(gameObject=var "Dust Break 1", emit=0)
    PlayParticleEmitter(gameObject=var "Dust Break 2", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=breakable_wall_death (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=secret_discovered_temp (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetCollider(gameObject=var "Self", active=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=25, spawnMax=25, speedMin=5, speedMax=20, angleMin=20, angleMax=160, originVariation=1.5)  // disabled
    SpawnFromPool(pool=var "Pool S", adjustPosition=(0, 0, 0), spawnMin=30, spawnMax=30, speedMin=5, speedMax=20, angleMin=20, angleMax=160)  // disabled
    SpawnRandomObjects(gameObject=Particle Rock Large (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=10, speedMin=5, speedMax=5, angleMin=20, angleMax=160, originVariation=1)  // disabled
    SpawnFromPool(pool=var "Pool L", adjustPosition=(0, 0, 0), spawnMin=25, spawnMax=25, speedMin=5, speedMax=15, angleMin=20, angleMax=160)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Crumble")
    SendEventByName(eventTarget=GameObject(var "Mask"), sendEvent="UNCOVER", delay=0, everyFrame=false)
    CreateObject(gameObject=Dust Break Wall (sharedassets31.assets), spawnPoint=var "Self", position=(unset), rotation=(-72.5, -180, -180), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetProperty(targetProperty=BoxCollider2D on <null>, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Particles Large", emit=0)
    Tk2dWatchAnimationEvents(gameObject=var "Self", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy
  }

  state Hit Right {
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(1.12, 0, 0), spawnMin=3, spawnMax=5, speedMin=15, speedMax=20, angleMin=40, angleMax=80, originVariation=0)  // disabled
    CreateObject(gameObject=Slash Impact, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "", vector=(unset), x=-1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    SpawnFromPool(pool=var "Pool S", adjustPosition=(1.12, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=40, angleMax=80)  // disabled
    CreateObject(gameObject=Dust Hit Med (sharedassets6.assets), spawnPoint=var "Self", position=(-1, 0, 0), rotation=(0, 90, 270), storeObject=var "", networkInstantiate=false, networkGroup=0)
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed Neg", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Return Right
    on HIT → Pause Frame
  }

  state Return Right {
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit Left {
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(-1.12, 0, 0), spawnMin=3, spawnMax=5, speedMin=15, speedMax=20, angleMin=100, angleMax=140, originVariation=0)  // disabled
    CreateObject(gameObject=Slash Impact, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    SpawnFromPool(pool=var "Pool S", adjustPosition=(-1.12, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=100, angleMax=140)  // disabled
    CreateObject(gameObject=Dust Hit Med (sharedassets6.assets), spawnPoint=var "Self", position=(-1, 0, 0), rotation=(180, 90, 270), storeObject=var "", networkInstantiate=false, networkGroup=0)
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Return Left
    on HIT → Pause Frame
  }

  state Return Left {
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed Neg", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit Down {
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=5, speedMin=15, speedMax=20, angleMin=70, angleMax=110, originVariation=0)  // disabled
    SpawnFromPool(pool=var "Pool S", adjustPosition=(0, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=70, angleMax=110)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Recoil Speed Neg", everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    CreateObject(gameObject=Dust Hit Med Down, spawnPoint=var "Self", position=(unset), rotation=(-72.5, -180, -180), storeObject=var "", networkInstantiate=false, networkGroup=0)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Return Down
  }

  state Return Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Recoil Speed Neg", everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit Up {
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=5, speedMin=15, speedMax=20, angleMin=70, angleMax=110, originVariation=0)  // disabled
    SpawnFromPool(pool=var "Pool S", adjustPosition=(0, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=70, angleMax=110)  // disabled
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    CreateObject(gameObject=Dust Hit Med (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(270, 90, 270), storeObject=var "", networkInstantiate=false, networkGroup=0)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Idle
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Break
  }

  state Damage {
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
  }

  state Destroy {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DestroyObject(gameObject=var "CamLock", delay=0, detachChildren=false)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state Activated {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    DestroyObject(gameObject=var "CamLock", delay=0, detachChildren=false)
    SendEventByName(eventTarget=GameObject(var "Mask"), sendEvent="UNCOVER", delay=0, everyFrame=false)
  }

  state Spell Destroy {
    SpawnObjectFromGlobalPool(gameObject=Fireball Hit (resources.assets), spawnPoint=var "Self", position=var "Effect Origin", rotation=(unset), storeObject=var "Hit Effect")
    on FINISHED → Break
  }

  var Facing: int = 2
  var Hits: int = 4
  var CamLock Name: string = ""
  var Mask Name: string = "break_wall_masks"

  var Attack Direction: float = 0
  var Chooser: float = 0
  var Recoil Speed: float = 1
  var Recoil Speed Neg: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Attack Type: int = 0
  var Activated: bool = false
  var Audio Player: gameObject = <null>
  var CamLock: gameObject = <null>
  var Camera: gameObject = <null>
  var Damager: gameObject = <null>
  var Dust Break 1: gameObject = <null>
  var Dust Break 2: gameObject = <null>
  var Mask: gameObject = <null>
  var Particles Large: gameObject = <null>
  var Particles Small: gameObject = <null>
  var Pool L: gameObject = <null>
  var Pool S: gameObject = <null>
  var Self: gameObject = <null>
  var Strike Effect: gameObject = <null>
}
