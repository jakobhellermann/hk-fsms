# Climb
# 4 location(s):
#   Deepnest_32 / Mimic Spider
#   GG_Nosk / Mimic Spider
#   GG_Nosk_V / Mimic Spider
#   sharedassets486.assets / Mimic Spider

fsm Climb {
  start Init
  on CHARGE END → Idle  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on CHARGE START → Not Climbing
  }

  state Not Climbing {
    CheckCollisionSide(topHit=(unset), rightHit=var "Wall R", bottomHit=(unset), leftHit=var "Wall L", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=var "Wall R", bottomHit=(unset), leftHit=var "Wall L", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolTest(boolVariable=var "Wall R", isTrue=→"SWAP", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Wall L", isTrue=→"SWAP", isFalse=(none), everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    on SWAP → Climbing
  }

  state Climbing {
    CheckCollisionSide(topHit=(unset), rightHit=var "Wall R", bottomHit=(unset), leftHit=var "Wall L", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=var "Wall R", bottomHit=(unset), leftHit=var "Wall L", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolNoneTrue(boolVariables=[var "Wall L", var "Wall R"], sendEvent=→"SWAP", storeResult=(unset), everyFrame=true)
    BoolTest(boolVariable=var "Wall R", isTrue=(none), isFalse=→"SWAP", everyFrame=true)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=20, everyFrame=true)
    on SWAP → Not Climbing
  }

  var Wall L: bool = false
  var Wall R: bool = false
}
