# Control
# 1 location(s):
#   sharedassets314.assets / Markoth Shield

fsm Control {
  start Pause
  on ATTACK CCW → At Start CCW  // from any state
  on ATTACK CW → At Start CW  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FindChild(gameObject=Self, childName="Shield", storeResult=var "Shield")
    FindChild(gameObject=Self, childName="Shield 2", storeResult=var "Shield 2")
    on FINISHED → Tween Out Init
  }

  state Choose {
    SendRandomEvent(events=[→"CW", →"CCW"], weights=[0.5, 0.5], delay=0)
    on CW → Spinning CW
    on CCW → Spinning CCW
  }

  state Spinning CW {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatAdd(floatVariable=var "Rotate Speed", add=180, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=0, maxValue=180, everyFrame=true)
    WaitRandom(timeMin=5, timeMax=9, finishEvent=→"CHANGE", realTime=false)
    on CHANGE → Decelerate CW
  }

  state Pause {
    on FINISHED → Init
  }

  state Tween Out Init {
    iTweenMoveTo(gameObject=var "Shield", id="", transformPosition=var "", vectorPosition=(-4.05, 0, 0), time=1, delay=0, speed=(unset), space=1, easeType=easeOutCubic, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → Choose
  }

  state Spinning CCW {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatAdd(floatVariable=var "Rotate Speed", add=-180, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=-180, maxValue=0, everyFrame=true)
    WaitRandom(timeMin=5, timeMax=9, finishEvent=→"CHANGE", realTime=false)
    on CHANGE → Decelerate CCW
  }

  state Decelerate CW {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatAdd(floatVariable=var "Rotate Speed", add=-180, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=0, maxValue=180, everyFrame=true)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spinning CCW
  }

  state Decelerate CCW {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatAdd(floatVariable=var "Rotate Speed", add=180, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=-180, maxValue=0, everyFrame=true)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spinning CW
  }

  state At Start CW {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatAdd(floatVariable=var "Rotate Speed", add=-240, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=-400, maxValue=0, everyFrame=true)
    FloatCompare(float1=var "Rotate Speed", float2=-400, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → At Pause
  }

  state At Pause {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tween Out
  }

  state Tween Out {
    iTweenMoveTo(gameObject=var "Shield", id="", transformPosition=var "", vectorPosition=(-14.77, 0, 0), time=2, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Tween In
  }

  state At Start CCW {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatAdd(floatVariable=var "Rotate Speed", add=240, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=0, maxValue=400, everyFrame=true)
    FloatCompare(float1=var "Rotate Speed", float2=400, tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → At Pause
  }

  state Tween In {
    iTweenMoveTo(gameObject=var "Shield", id="", transformPosition=var "", vectorPosition=(-4.05, 0, 0), time=2, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Stop
  }

  state Stop {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHIELD END", delay=0, everyFrame=false)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → Choose
  }

  var Rotate Speed: float = 0
  var Parent Pos: vector3 = (0, 0, 0)
  var Parent: gameObject = <null>
  var Shield: gameObject = <null>
  var Shield 2: gameObject = <null>
}
