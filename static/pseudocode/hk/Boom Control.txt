# Boom Control
# 1 location(s):
#   Fungus3_01 / Gas Explosion M

fsm Boom Control {
  start Start

  state Explode {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=explosion_4_wet (resources.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Death Wave Infected (sharedassets32.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)  // disabled
    SpawnObjectFromGlobalPool(gameObject=Death Wave Infected R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=3, y=3, z=(unset), everyFrame=false, lateUpdate=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → destroy
  }

  state destroy {
    DestroySelf(detachChildren=true)
  }

  state Line Loop {
    GetOwner(storeGameObject=var "Self")
    IntAdd(intVariable=var "Looper", add=1, everyFrame=false)
    CreateObject(gameObject=Impact Lines (sharedassets27.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)
    Tk2dSpriteSetColor(gameObject=var "Hit Effect", color=(1, 0.51724136, 0, 1), everyframe=false)
    RandomFloat(min=2, max=3, storeResult=var "Chooser")
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=var "Chooser", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=0, max=360, storeResult=var "Chooser")
    Rotate(gameObject=var "Hit Effect", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    IntCompare(integer1=var "Looper", integer2=11, equal=(none), lessThan=(none), greaterThan=→"LOOP COMPLETE", everyFrame=false)
    on FINISHED → Loop Back
    on LOOP COMPLETE → Explode
  }

  state Loop Back {
    GotoPreviousState()
  }

  state Start {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Explode
  }

  var Chooser: float = 0
  var Looper: int = 0
  var Camera: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Self: gameObject = <null>
}
