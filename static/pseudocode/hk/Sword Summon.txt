# Sword Summon
# 3 location(s):
#   RestingGrounds_02_boss / Warrior/Ghost Warrior Xero
#   GG_Ghost_Xero / Warrior/Ghost Warrior Xero
#   GG_Ghost_Xero_V / Warrior/Ghost Warrior Xero

fsm Sword Summon {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="Sword 3", storeResult=var "Sword 3")
    FindChild(gameObject=Self, childName="Sword 4", storeResult=var "Sword 4")
    FindChild(gameObject=Self, childName="Attack Pt", storeResult=var "Attack Pt")
    GetHP(target=Self, storeValue=var "HP")
    IntOperator(integer1=var "HP", integer2=2, operation=Divide, storeResult=var "Half HP", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on TOOK DAMAGE → Check
  }

  state Check {
    GetFsmInt(gameObject=Self, fsmName="health_manager_enemy", variableName="HP", storeValue=var "HP", everyFrame=false)
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Half HP", equal=→"SUMMON", lessThan=→"SUMMON", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on SUMMON → Summon
  }

  state Summon {
    SetFsmBool(gameObject=Self, fsmName="Attacking", variableName="Rage", setValue=true, everyFrame=false)
  }

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  var HP: int = 0
  var Half HP: int = 0
  var Attack Pt: gameObject = <null>
  var Sword 3: gameObject = <null>
  var Sword 4: gameObject = <null>
}
