# Dummy Mantis
# 1 location(s):
#   Fungus3_23_boss / Battle Scene/Cloth Entry/Dummy Mantis

fsm Dummy Mantis {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Raycast X", floatValue=-2, everyFrame=false)
    SetFloatValue(floatVariable=var "Atk Speed Crt", floatValue=var "Attack Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Atk Speed Crt", multiplyBy=-1, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Scale", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatMultiply(floatVariable=var "Raycast X", multiplyBy=-1, everyFrame=false)
    FloatMultiply(floatVariable=var "Atk Speed Crt", multiplyBy=-1, everyFrame=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    FindChild(gameObject=Self, childName="Chunks", storeResult=var "Chunks")
    on FINISHED → Spawn Dust
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    BoolTest(boolVariable=var "Back Range", isTrue=(none), isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Front Range", isTrue=(none), isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Above Range", isTrue=(none), isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Walk Range", isTrue=(none), isFalse=(none), everyFrame=true)
    Wait(time=1, finishEvent=(none), realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move
  }

  state Active {
    ActivateGameObject(gameObject=var "Head Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Walk Range Obj", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Idle
  }

  state Do Spawn {
    ActivateGameObject(gameObject=var "Head Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPAWNED", delay=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    StopParticleEmitter(gameObject=var "Emerge Dust")
    PlayParticleEmitter(gameObject=var "Burst Dust", emit=0)
    on FINISHED → Spawn Recover
  }

  state Spawn Recover {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateGameObject(gameObject=var "Head Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Active
    on TOOK DAMAGE → Active
  }

  state Spawn Dust {
    PlayParticleEmitter(gameObject=var "Emerge Dust", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetCollider(gameObject=Self, active=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=stalactite_break (sharedassets6.assets))
    on FINISHED → Do Spawn
  }

  state Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack Antic")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CLOTH ENTER", delay=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on MANTIS DUMMY KILL → Die
  }

  state Die {
    InstaDeath(target=Self, direction=270)  // disabled
    ActivateAllChildren(gameObject=var "Self", activate=true)
    FlingObjects(containerObject=var "Chunks", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=0, speedMax=0, angleMin=0, angleMax=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=enemy_death_sword (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=enemy_damage (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    DestroySelf(detachChildren=true)
  }

  state Move {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-9, y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    FloatCompare(float1=var "Self X", float2=45.58, tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    on END → Antic
  }

  var Alt Spawn X: float = 0
  var Attack Speed: float = 40

  var Atk Speed Crt: float = 0
  var Raycast X: float = 0
  var Scale: float = 0
  var Self X: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Above Range: bool = false
  var Back Range: bool = false
  var Facing Right: bool = false
  var Front Range: bool = false
  var Walk Range: bool = false
  var Burst Dust: gameObject = <null>
  var Chunks: gameObject = <null>
  var Emerge Dust: gameObject = <null>
  var Head Box: gameObject = <null>
  var Self: gameObject = <null>
  var Walk Range Obj: gameObject = <null>
}
