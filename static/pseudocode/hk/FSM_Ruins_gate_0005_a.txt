// uses template: health_manager_enemy
fsm FSM {
  start Init Pause
  on KILL ALL ENEMIES → Kill All  // from any state
  on INSTA KILL → Pause  // from any state
  on INVULN → Invulnerable  // from any state
  on FORCE KILL → Unset Special Death  // from any state

  state Detecting {
    SetScale(gameObject=var "Corpse Instance", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Corpse Instance", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on TAKE DAMAGE → Get Damager Parameters
    on EXTRA KILL → Pause
  }

  state Decrement Health {
    IntOperator(integer1=var "HP", integer2=var "Attack Strength", operation=Subtract, storeResult=var "HP", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=0, equal=→"ZERO HP", lessThan=→"ZERO HP", greaterThan=(none), everyFrame=false)
    on ZERO HP → Pause
    on FINISHED → Invulnerable
  }

  state Destroy {
    SendEventByName(eventTarget=BroadcastAll, sendEvent=var "Death Broadcast Event", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=var "Snapshot On Death", transitionTime=6)
    BoolTest(boolVariable=var "Recycle", isTrue=→"RECYCLE", isFalse=(none), everyFrame=false)
    DestroySelf(detachChildren=false)
    on RECYCLE → Recycle
  }

  state Invulnerable {
    BoolTest(boolVariable=var "Ignore Invuln", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Play Animation", isTrue=→"PLAY ANIM", isFalse=(none), everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detecting
    on PLAY ANIM → Play Anim
    on CANCEL → Detecting
    on END ALL INVULN → Detecting
    on EXTRA KILL → Pause
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")

    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    SetGameObject(variable=var "Camera", gameObject=var "CameraParent", everyFrame=false)
    SetGameObject(variable=var "Hero Object", gameObject=var "Hero", everyFrame=false)
    GetScale(gameObject=var "Self", vector=(unset), xScale=var "X Scale", yScale=var "Y Scale", zScale=(unset), space=0, everyFrame=false)
    CreateObject(gameObject=var "Corpse", spawnPoint=var "Self", position=var "Corpse Spawn Point", rotation=(unset), storeObject=var "Corpse Instance", networkInstantiate=false, networkGroup=0)

    SetScale(gameObject=var "Corpse Instance", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    BoolTest(boolVariable=var "Corpse Faces Right", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Corpse Faces Right
    on FALSE → Corpse Faces Left
    on ACTIVATE → Activated
  }

  state Get Damager Parameters {
