# fling_self
# 1 location(s):
#   sharedassets149.assets / Corpse Moss Knight/Nail

fsm fling_self {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetScale(gameObject=var "Parent", vector=(unset), xScale=var "Parent X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    RandomFloat(min=95, max=105, storeResult=var "Chooser")
    FloatCompare(float1=var "Parent X Scale", float2=0, tolerance=0, equal=→"POS", lessThan=→"NEG", greaterThan=→"POS", everyFrame=false)
    on NEG → Negative
    on POS → Shoot
  }

  state Negative {
    FloatOperator(float1=180, float2=var "Chooser", operation=Subtract, storeResult=var "Chooser", everyFrame=false)
    on FINISHED → Shoot
  }

  state Shoot {
    SetVelocityAsAngle(gameObject=Self, angle=var "Chooser", speed=20, everyFrame=false)
  }

  var Chooser: float = 0
  var Parent X Scale: float = 0
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
