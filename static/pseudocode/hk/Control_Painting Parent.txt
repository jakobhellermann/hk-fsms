# Control
# 1 location(s):
#   Room_nailmaster_02 / Painting Parent

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Bone", storeResult=var "Bone")
    FindChild(gameObject=Self, childName="Nailsmith", storeResult=var "Nailsmith")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="nailsmithSheo", isTrue=→"NAILSMITH", isFalse=→"BONE")
    on BONE → Bone
    on NAILSMITH → Nailsmith
  }

  state Bone {
    ActivateGameObject(gameObject=var "Bone", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Nailsmith {
    ActivateGameObject(gameObject=var "Nailsmith", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Bone: gameObject = <null>
  var Nailsmith: gameObject = <null>
}
