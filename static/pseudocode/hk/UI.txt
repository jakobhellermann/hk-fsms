# UI
# 1 location(s):
#   GG_Workshop / GG_Summary_Board/Inspect

fsm UI {
  start Inert
  on CONVO START → Open UI  // from any state

  state Inert {
    GetHero(storeResult=var "Hero")
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
  }

  state Open UI {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    ShowBossSummaryUI(target=GG_Summary_Board, activate=true)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Listen for cancel
  }

  state Close UI {
    ShowBossSummaryUI(target=GG_Summary_Board, activate=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End convo
  }

  state Listen for cancel {
    ListenForMenuActions(eventTarget=Self, submitPressed=(none), cancelPressed=→"CANCEL", ignoreAttack=false)
    on CANCEL → Close UI
  }

  state End convo {
    SendEventByNameV2(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
  }

  var CamLock: gameObject = GG_Summary_Board/CameraLockArea

  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
