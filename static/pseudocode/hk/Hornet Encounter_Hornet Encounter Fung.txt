# Hornet Encounter
# 1 location(s):
#   Fungus2_10 / Hornet Encounter Fung

fsm Hornet Encounter {
  start Pause

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="hornetFung", storeValue=var "PlayerData")
    IntCompare(integer1=var "PlayerData", integer2=0, equal=(none), lessThan=(none), greaterThan=→"DESTROY", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasWalljump", isTrue=→"DESTROY", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hornetCityBridge_completed", isTrue=→"DESTROY", isFalse=(none))
    on FINISHED → Idle
    on DESTROY → Destroy
  }

  state Idle {
    BoolTest(boolVariable=var "Encounter Range", isTrue=→"TURN", isFalse=(none), everyFrame=true)
    on TURN → Run
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Run {
    SetPlayerDataInt(intName="hornetFung", value=1)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetVelocity2d(gameObject=Self, vector=(unset), x=12, y=(unset), everyFrame=false)
    Wait(time=5, finishEvent=→"DESTROY", realTime=false)
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    SetPlayerDataBool(boolName="hornetCityBridge_ready", value=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on DESTROY → Destroy
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Encounter
  }

  state Set Encounter {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hornetCityBridge_ready", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="hornetCityBridge_completed", value=true)
    on FINISHED → Init
  }

  var PlayerData: int = 0
  var Encounter Range: bool = false
  var Turn Range: bool = false
  var CamLock: gameObject = <null>
  var Dust: gameObject = Hornet Encounter Fung/Dust
  var Grass: gameObject = <null>
}
