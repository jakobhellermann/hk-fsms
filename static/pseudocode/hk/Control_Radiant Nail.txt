# Control
# 1 location(s):
#   sharedassets407.assets / Radiant Nail

fsm Control {
  start Appear
  on RECYCLE → Recycle  // from any state
  on FLY → Fly  // from any state
  on FAN ANTIC → Fan Antic  // from any state
  on NAIL END → End  // from any state

  state Appear {
    SetPolygonCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Nail Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Set Collider
  }

  state Recycle {
    RecycleSelf()
  }

  state Fly {
    SetPolygonCollider(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
  }

  state Set Collider {
    SetPolygonCollider(gameObject=Self, active=true)
  }

  state Fan Antic {
    SetPolygonCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Nail Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fan Out
  }

  state Fan Out {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Angle", add=90, everyFrame=false, perSecond=false)
    DecelerateV2(gameObject=Self, deceleration=0.88)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=30, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fan Ready
    on FAN ATTACK CW → Fire CW
    on FAN ATTACK CCW → Fire CCW
  }

  state Fan Ready {
    SetPolygonCollider(gameObject=Self, active=true)
    DecelerateV2(gameObject=Self, deceleration=0.88)
    on FAN ATTACK CW → Fire CW
    on FAN ATTACK CCW → Fire CCW
  }

  state Fire CW {
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=25, everyFrame=true)
    FaceAngle(gameObject=Self, angleOffset=-90, everyFrame=true)
    FloatAdd(floatVariable=var "Angle", add=-30, everyFrame=true, perSecond=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Fire CCW {
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=25, everyFrame=true)
    FaceAngle(gameObject=Self, angleOffset=-90, everyFrame=true)
    FloatAdd(floatVariable=var "Angle", add=30, everyFrame=true, perSecond=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Nail End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  var Angle: float = 0
}
