# FSM
# 5 location(s):
#   Fungus1_35 / Warrior/Ghost Warrior No Eyes
#   Fungus3_40_boss / Warrior/Ghost Warrior Marmu
#   Cliffs_02_boss / Warrior/Ghost Warrior Slug
#   RestingGrounds_02_boss / Warrior/Ghost Warrior Xero
#   sharedassets194.assets / Ghost Warrior Slug

// uses template: hp_scaler
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="nailSmithUpgrades", storeValue=var "Nail Level")
    IntAdd(intVariable=var "Nail Level", add=1, everyFrame=false)
    IntSwitch(intVariable=var "Nail Level", compareTo=[1, 2, 3, 4, 5], sendEvent=[→"1", →"2", →"3", →"4", →"5"], everyFrame=false)
    on 1 → Set 1
    on 2 → Set 2
    on 3 → Set 3
    on 4 → Set 4
    on 5 → Set 5
  }

  state Set 1 {
    SetHP(target=Self, hp=var "Level 1")
  }

  state Set 2 {
    SetHP(target=Self, hp=var "Level 2")
  }

  state Set 3 {
    SetHP(target=Self, hp=var "Level 3")
  }

  state Set 4 {
    SetHP(target=Self, hp=var "Level 4")
  }

  state Set 5 {
    SetHP(target=Self, hp=var "Level 5")
  }

  var Level 1: int = 200
  var Level 2: int = 320
  var Level 3: int = 416
  var Level 4: int = 500
  var Level 5: int = 570

  var Nail Level: int = 0
}
