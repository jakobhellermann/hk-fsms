# Mantis Throne Main
# 1 location(s):
#   GG_Mantis_Lords_V / Mantis Battle/Mantis Lord Throne 2

// uses template: mlord_throne_main_triple
fsm Mantis Throne Main {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on MLORD CHALLENGE → Music
  }

  state Stand {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Stand")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateAllChildren(gameObject=var "CamLocks", activate=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=hero_jump (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Cage
  }

  state Leave {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Throne Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_misc_jump_1 (sharedassets180.assets))
    on FINISHED → Wake
  }

  state Wake {
    SetMeshRenderer(gameObject=Self, active=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD START MAIN", delay=0, everyFrame=false)
    on MANTIS DEFEATED → Pause
  }

  state Defeated {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Wounded")
    SetMeshRenderer(gameObject=Self, active=true)
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_land_on_ground (sharedassets180.assets))
    on WAIT → Subs Stand
  }

  state Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Defeated
  }

  state Fighting {
    SetMeshRenderer(gameObject=Self, active=false)
    on RETURN → Return Pause
  }

  state Floors {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD FLOOR RETRACT", delay=0, everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Challenge
  }

  state Cage {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD CAGE DOWN", delay=0, everyFrame=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MANTIS_LORDS", everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Floors
    on GG BOSS → Leave
  }

  state Regain Control {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    on FINISHED → Leave
  }

  state End Challenge {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Challenge End")
    Tk2dWatchAnimationEvents(gameObject=var "Hero", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Flip Back
  }

  state Flip Back {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FlipSprite())
    on FINISHED → Regain Control
  }

  state Music {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    ApplyMusicCue(musicCue=BossMantisLords (sharedassets180.assets), delayTime=0, transitionTime=0)
    on FINISHED → Stand
  }

  state Subs Stand {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SUBS STAND", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"WAIT", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=hero_jump (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on WAIT → I Stand
  }

  state I Stand {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Stand")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=hero_jump (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=GG9_transition_0.5_second (sharedassets425.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)  // disabled
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SUBS LOOK", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"WAIT", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0.5)
    on WAIT → Roar 1
  }

  state Roar 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Triple Gesture 1")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SUBS ROAR 1", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=hornet_small_jump (sharedassets15.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on WAIT → Roar 2
  }

  state Roar 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Triple Gesture 2")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SUBS ROAR 2", delay=0, everyFrame=false)
    Wait(time=2.75, finishEvent=→"WAIT", realTime=false)
    ApplyMusicCue(musicCue=GG Mantis (sharedassets452.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=misc_rumble_impact (sharedassets13.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=GG_mantis_lord_3_stand, pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=GG_mantis_lord_3_stand, pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=GG_mantis_lord_3_stand, pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=mimic_spider_transormation_stage_4 (sharedassets290.assets), pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="SISTERS", everyFrame=false)
    on WAIT → Leave Again!
  }

  state Leave Again! {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Throne Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD START SUB", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    on FINISHED → Fighting
  }

  state Defeated again {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_land_on_ground (sharedassets180.assets))
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Wounded")
  }

  state Return Pause {
    Wait(time=2.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Defeated again
  }

  var Area Title: gameObject = <null>
  var CamLocks: gameObject = <null>
  var Parent: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Roar Point: gameObject = Mimic Spider/Roar Point
  var Self: gameObject = <null>
}
