# Disable
# 1 location(s):
#   Crossroads_10 / Fk Break Wall/Broken/Slope

// component disabled: this FSM does not run until something enables it
fsm Disable {
  start Idle

  state Idle {
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"TOUCH", storeCollider=var "Player", storeForce=(unset))
    Collision2dEvent(collision=1, collideTag="Player", sendEvent=→"TOUCH", storeCollider=var "Player", storeForce=(unset))
    on TOUCH → Shoot
  }

  state Disable {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    SendMessage(gameObject=var "Player", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    Collision2dEvent(collision=2, collideTag="Player", sendEvent=→"UNTOUCH", storeCollider=var "Player", storeForce=(unset))
    on UNTOUCH → Enable
  }

  state Enable {
    SendMessage(gameObject=var "Player", delivery=SendMessage, options=1, functionCall=RegainControl())
    on FINISHED → Idle
  }

  state Shoot {
    on FINISHED → Idle
  }

  var Player: gameObject = <null>
}
