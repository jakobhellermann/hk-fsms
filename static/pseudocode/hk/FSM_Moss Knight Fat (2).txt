# FSM
# 4 location(s):
#   Fungus3_39 / Moss Knight Fat (2)
#   Fungus3_39 / Moss Knight Fat
#   Fungus3_39 / Moss Knight Fat (1)
#   Fungus3_39 / Moss Cultist

// uses template: deactivate_if_playerdatabool
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var playerData bool: string = "crossroadsInfected"
}
