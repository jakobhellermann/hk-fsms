# Gate Control
# 1 location(s):
#   Waterways_09 / Waterways Gate

fsm Gate Control {
  start Pause
  on ACTIVATE → Activate  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Glow", storeResult=var "Glow")
    on FINISHED → Idle
  }

  state Idle {
    FindChild(gameObject=Self, childName="Glow", storeResult=var "Glow")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="waterwaysGate", isTrue=→"ACTIVATE", isFalse=(none))
    on OPEN → Open
  }

  state Activate {
    ActivateGameObject(gameObject=var "Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DestroySelf(detachChildren=true)
  }

  state Open {
    ActivateGameObject(gameObject=var "Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=jiji_door_open_pt_2 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataBool(boolName="waterwaysGate", value=true)
    DestroySelf(detachChildren=true)  // disabled
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    on FINISHED → Shake
  }

  state Shake {
  }

  var Glow: gameObject = <null>
}
