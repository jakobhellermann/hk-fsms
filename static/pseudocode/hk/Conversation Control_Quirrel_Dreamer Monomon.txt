# Conversation Control
# 1 location(s):
#   Fungus3_archive_02 / Dreamer Monomon/Quirrel

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Mask Anim", storeResult=var "Mask Anim")
    FindChild(gameObject=Self, childName="Glow 1", storeResult=var "Glow 1")
    FindChild(gameObject=Self, childName="Glow 2", storeResult=var "Glow 2")
    FindChild(gameObject=Self, childName="Tendrils", storeResult=var "Tendrils")
    FindChild(gameObject=Self, childName="White Flash", storeResult=var "White Flash")
    FindChild(gameObject=Self, childName="Soul", storeResult=var "Soul")
    FindChild(gameObject=Self, childName="Mask Break", storeResult=var "Mask Break")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Soul Tendrils", storeResult=var "Soul Tendrils")
    FindChild(gameObject=var "Parent", childName="Monomon", storeResult=var "Monomon")
    FindChild(gameObject=var "Parent", childName="Particle Control", storeResult=var "Particle Control")
    FindChild(gameObject=var "Particle Control", childName="Soul 1", storeResult=var "Soul 1")
    FindChild(gameObject=var "Particle Control", childName="Soul 2", storeResult=var "Soul 2")
    FindChild(gameObject=var "Parent", childName="Arrive Flash", storeResult=var "Arrive Flash")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    CreateObject(gameObject=Area Title (sharedassets9.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Area Title", networkInstantiate=false, networkGroup=0)  // disabled
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)  // disabled
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="RELICDEALER", everyFrame=false)  // disabled
    on FINISHED → Check
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")  // disabled
    on FINISHED → Box Down
  }

  state Talk Finish {
    Tk2dPlayAnimation(gameObject=var "Mask Anim", animLibName="", clipName="Wounded Talk End")
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Talk {
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Quirrel_tank_01 (sharedassets228.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEACHER_1A", "Quirrel"], storeResult=(unset var))
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on CONVO_FINISH → Talk 2
  }

  state Hero Anim {
    on FINISHED → Box Up
  }

  state Hold Anim {
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Mask Anim", active=true)
    Tk2dPlayAnimation(gameObject=var "Mask Anim", animLibName="", clipName="Mask Hold")
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Going
  }

  state Going {
    ActivateGameObject(gameObject=var "Glow 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_enter (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Glow 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Tendrils", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Soul", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tendrils Down
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Hold Anim
  }

  state Talk 2 {
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Quirrel_extra_talk_01 (sharedassets228.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEACHER_1B", "Quirrel"], storeResult=(unset var))
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Mask Anim", animLibName="", clipName="Turn R")
    on CONVO_FINISH → Talk 3
  }

  state Talk 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEACHER_1C", "Quirrel"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=var "Mask Anim", animLibName="", clipName="Turn L")
    on CONVO_FINISH → Box Down 2
  }

  state Tendrils Down {
    SendEventByName(eventTarget=GameObject(var "Soul Tendrils"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Monomon Up
  }

  state Monomon Up {
    SendEventByName(eventTarget=GameObject(var "Monomon"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Soul
  }

  state Finish {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MONOMON WAKE", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Soul 1")
    StopParticleEmitter(gameObject=var "Soul 2")
    ActivateAllChildren(gameObject=var "Arrive Flash", activate=true)
    SendMessage(gameObject=var "Monomon", delivery=SendMessage, options=1, functionCall=flashWhiteLong())
    Tk2dPlayAnimation(gameObject=var "Mask Anim", animLibName="", clipName="Mask Drop")
    PlayParticleEmitter(gameObject=var "Mask Break", emit=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=Self, fsmName="npc_control", variableName="Hero Always Right", setValue=false, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="npc_control", variableName="Move To Offset", setValue=1.3, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_ground_land (sharedassets28.assets), pitchMin=1, pitchMax=1, volume=1, delay=1, storePlayer=<null>)
    on FINISHED → Set
  }

  state End Soul {
    StopParticleEmitter(gameObject=var "Soul")
    SetMeshRenderer(gameObject=var "Tendrils", active=false)
    ActivateGameObject(gameObject=var "Glow 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_plant_emit_orbs (sharedassets228.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Finish
  }

  state Set {
    SetPlayerDataBool(boolName="summonedMonomon", value=true)
    on FINISHED → End
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="summonedMonomon", isTrue=→"REPEAT", isFalse=→"FINISHED")
    on REPEAT → Repeat
    on FINISHED → Talk
  }

  state Repeat {
    Tk2dPlayAnimation(gameObject=var "Mask Anim", animLibName="", clipName="Wounded Talk")
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Quirrel_exhausted_02 (sharedassets228.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDreamNail", isTrue=(none), isFalse=→"NO DN")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEACHER_2", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
    on NO DN → No Dream Nail
  }

  state No Dream Nail {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_NODREAMNAIL", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var History: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var mapCrossroads: bool = false
  var Area Title: gameObject = <null>
  var Arrive Flash: gameObject = <null>
  var Glow 1: gameObject = <null>
  var Glow 2: gameObject = <null>
  var Mask Anim: gameObject = <null>
  var Mask Break: gameObject = <null>
  var Monomon: gameObject = <null>
  var Parent: gameObject = <null>
  var Particle Control: gameObject = <null>
  var Self: gameObject = <null>
  var Soul: gameObject = <null>
  var Soul 1: gameObject = <null>
  var Soul 2: gameObject = <null>
  var Soul Tendrils: gameObject = <null>
  var Tendrils: gameObject = <null>
  var Voice Player: gameObject = <null>
  var White Flash: gameObject = <null>
}
