// uses template: acid_burn_particle
fsm acid_burn_particle {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Acid Steam", storeResult=var "Acid Steam")
    on FINISHED → Idle
  }

  state Idle {
    on ACID → Burn
  }

  state Burn {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=false, namedValueColor="_Color", time=0.3, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    PlayParticleEmitter(gameObject=var "Acid Steam", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Acid Steam: gameObject = <null>
}
