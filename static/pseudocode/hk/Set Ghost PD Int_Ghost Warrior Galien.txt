# Set Ghost PD Int
# 1 location(s):
#   Deepnest_40 / Warrior/Ghost Warrior Galien

fsm Set Ghost PD Int {
  start Idle

  state Idle {
    on ZERO HP → Set
  }

  state Set {
    SetPlayerDataInt(intName=var "PD Int", value=1)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST DEATH", delay=0, everyFrame=false)
  }

  var PD Int: string = "galienDefeated"
}
