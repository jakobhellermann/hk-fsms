# Control
# 1 location(s):
#   Ruins1_27 / Hornet Fountain Encounter

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hornetFountainEncounter", isTrue=→"INERT", isFalse=(none))
    FindGameObject(objectName="", withTag="Cinematic", store=var "Cinematic Player")
    FindChild(gameObject=Self, childName="Hornet", storeResult=var "Hornet")
    FindChild(gameObject=Self, childName="PanCam", storeResult=var "PanCam")
    FindChild(gameObject=var "Hornet", childName="Thread", storeResult=var "Thread")
    on INERT → Inert
    on FINISHED → Idle
  }

  state Inert {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Take Control
  }

  state Take Control {
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=3)
    ActivateGameObject(gameObject=var "PanCam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SetPlayerDataBool(boolName="disablePause", value=true)
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=5.51, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    on LAND → Fade Pause
  }

  state Fade Out {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.9, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Position
  }

  state Cinematic {
    SendMessage(gameObject=var "Cinematic Player", delivery=SendMessage, options=1, functionCall=TriggerStartVideo())
    ActivateGameObject(gameObject=var "PanCam", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CINEMATIC END → Fade In
  }

  state Position {
    SetPosition(gameObject=var "Hero", vector=(unset), x=26.91, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    Wait(time=0.25, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Hornet", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Scene Blanker
  }

  state Fade In {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER OFF", delay=0, everyFrame=false)
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Talk 1
  }

  state Talk 1 {
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["HORNET_FOUNTAIN_1", "Hornet"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=Hornet_Dialogue_Generic_02 (sharedassets105.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk 2
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Jump
  }

  state Talk 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["HORNET_FOUNTAIN_2", "Hornet"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=var "Hornet", animLibName="", clipName="TurnToIdle")
    SetScale(gameObject=var "Hornet", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=Hornet_Dialogue_Generic_03 (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down
  }

  state Jump {
    Tk2dPlayAnimation(gameObject=var "Hornet", animLibName="", clipName="Jump Full")
    iTweenMoveBy(gameObject=var "Hornet", id="", vector=(0, 6, 0), time=0.7, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=true)
    Wait(time=0.415, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=Hornet_Fight_Yell_03 (sharedassets105.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_jump (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Throw Start
  }

  state Throw Start {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hornet", clipName="Throw Side Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Throw
  }

  state Throw {
    ActivateGameObject(gameObject=var "Thread", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Thread", clipName="Thread 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=var "Hornet", animLibName="", clipName="Throw Side")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_needle_thow_spin (sharedassets105.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Yank
  }

  state Yank {
    Tk2dPlayAnimation(gameObject=var "Hornet", animLibName="", clipName="Harpoon Side")
    SetParent(gameObject=var "Thread", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    ActivateGameObject(gameObject=var "Thread", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Rotate(gameObject=var "Hornet", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=45, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetParent(gameObject=var "Thread", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    iTweenMoveBy(gameObject=Self, id="", vector=(8, 8, 0), time=0.2, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_dash (sharedassets34.assets), pitchMin=1.15, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  state End {
    SetPlayerDataBool(boolName="hornetFountainEncounter", value=true)
    SetMeshRenderer(gameObject=var "Hornet", active=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=3)
    SetPlayerDataBool(boolName="disablePause", value=false)
    on FINISHED → 
  }

  state Fade Pause {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=PlayIdle())
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Out
  }

  state Scene Blanker {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER ON", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Cinematic
  }

  var Hero Y: float = 0
  var Rotation: float = 0
  var Leave Vector: vector3 = (0, 0, 0)
  var Cinematic Player: gameObject = <null>
  var Grass: gameObject = Corpse Hornet 1/Grass (sharedassets105.assets)
  var Grass Escape: gameObject = Corpse Hornet 1/Grass Escape (sharedassets105.assets)
  var Hornet: gameObject = <null>
  var Leave Anim: gameObject = Corpse Hornet 1/Leave Anim (sharedassets105.assets)
  var PanCam: gameObject = <null>
  var Roar: gameObject = <null>
  var Self: gameObject = <null>
  var Thread: gameObject = Corpse Hornet 1/Thread (sharedassets105.assets)
}
