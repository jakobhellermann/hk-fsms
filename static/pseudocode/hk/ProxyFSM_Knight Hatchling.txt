# ProxyFSM
# 1 location(s):
#   resources.assets / Knight Hatchling

fsm ProxyFSM {
  start Idle
  on HIT LANDED → Hit Landed  // from any state
  on ALL CHARMS END → Charms End  // from any state
  on HAZARD RELOAD → Hazard Reload  // from any state
  on K HATCHLING END → Charms End  // from any state
  on BENCHREST → Bench Rest  // from any state
  on BENCHREST END → End Bench  // from any state
  on QUICK SPAWN → Quick Spawn  // from any state
  on DREAMGATE OUT → Wait  // from any state

  state Idle {
  }

  state Hit Landed {
    CallMethodProper(gameObject=Self, behaviour="KnightHatchling", methodName="FsmHitLanded", parameters=[], storeResult=(unset var))
    on FINISHED → Idle
  }

  state Charms End {
    CallMethodProper(gameObject=Self, behaviour="KnightHatchling", methodName="FsmCharmsEnd", parameters=[], storeResult=(unset var))
    on FINISHED → Idle
  }

  state Hazard Reload {
    CallMethodProper(gameObject=Self, behaviour="KnightHatchling", methodName="FsmHazardReload", parameters=[], storeResult=(unset var))
    on FINISHED → Idle
  }

  state Bench Rest {
    CallMethodProper(gameObject=Self, behaviour="KnightHatchling", methodName="FsmBenchRestStart", parameters=[], storeResult=(unset var))
    on FINISHED → Idle
  }

  state End Bench {
    CallMethodProper(gameObject=Self, behaviour="KnightHatchling", methodName="FsmBenchRestEnd", parameters=[], storeResult=(unset var))
    on FINISHED → Idle
  }

  state Quick Spawn {
    CallMethodProper(gameObject=Self, behaviour="KnightHatchling", methodName="FsmQuickSpawn", parameters=[], storeResult=(unset var))
    on FINISHED → Idle
  }

  state Teleport Out {
    CallMethodProper(gameObject=Self, behaviour="KnightHatchling", methodName="FsmDreamGateOut", parameters=[], storeResult=(unset var))
  }

  state Wait {
    WaitRandom(timeMin=0.5, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Teleport Out
  }
}
