fsm Plant Trap Control {
  start Init

  state Idle {
    on DETECT → Ready
  }

  state Ready {
    PlayParticleEmitter(gameObject=var "Ready Grass", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Snap Ready")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=plant_trap_bite_new (sharedassets128.assets))
    on FINISHED → Snap
  }

  state Init {
    FindChild(gameObject=Self, childName="Ready Grass", storeResult=var "Ready Grass")
    on FINISHED → Idle
  }

  state Snap {
    StopParticleEmitter(gameObject=var "Ready Grass")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Snap")
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on WAIT → Retract
  }

  state Retract {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Retract", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Cooldown
  }

  state Cooldown {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  var Ready Grass: gameObject = <null>
}
