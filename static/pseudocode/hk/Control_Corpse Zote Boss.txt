fsm Control {
  start Init

  state Init {
    SetPlayerDataBool(boolName="zoteDefeated", value=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC SILENT", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=Zote_battle_death, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD GASP", delay=0, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.5, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(0))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BigShake", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    FindChild(gameObject=Self, childName="Fall Sprite", storeResult=var "Fall Sprite")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD STILL", delay=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on L → L
    on R → R
  }

  state R {
    SetVelocity2d(gameObject=Self, vector=(unset), x=3, y=7.5, everyFrame=false)
    on FINISHED → In Air
  }

  state L {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-3, y=7.5, everyFrame=false)
    on FINISHED → In Air
  }

  state In Air {
    Rotate(gameObject=var "Fall Sprite", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-28, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Land {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WAVE END", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=brkn_wand_land, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=Zote_battle_defeat_end, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC STING", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["ZOTE"], storeResult=(unset var))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    ActivateGameObject(gameObject=var "Fall Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    WaitRandom(timeMin=2, timeMax=5, finishEvent=→"GRUMBLE", realTime=false)
    on GRUMBLE → Grumble
    on NAIL HIT → Whack
  }

  state Grumble {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Grumble")
    AudioPlayInState(gameObject=Self, volume=1)
    on FINISHED → Idle
    on NAIL HIT → Whack
  }

  state Whack {
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zote_battle_wound_01, Zote_battle_wound_02, Zote_battle_wound_03, Zote_battle_fall_01], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  var Fall Sprite: gameObject = <null>
  var Self: gameObject = <null>
}
