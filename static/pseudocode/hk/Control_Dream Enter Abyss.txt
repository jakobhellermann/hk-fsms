# Control
# 1 location(s):
#   Abyss_15 / Dream Enter Abyss

fsm Control {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    IntCompare(integer1=var "Royal Charm State", integer2=4, equal=→"INACTIVE", lessThan=(none), greaterThan=→"INACTIVE", everyFrame=false)
    on FINISHED → Idle
    on INACTIVE → Inactive
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"DREAM HIT", storeCollider=var "")
    on DREAM HIT → Take Control
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_damage (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_plant_hit (sharedassets43.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var))
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", setValue=true, everyFrame=false)
    on FINISHED → Impact
  }

  state Impact {
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dark_spell_get (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=roller_hit_wall (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=1.4375, storePlayer=<null>)
    SetPosition(gameObject=var "Impact", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Pt Wave", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt Wave", emit=0)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    ActivateGameObject(gameObject=var "Dummy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Dummy", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Dummy", vector=(unset), x=var "Hero Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Dummy", clipName="Knight Collapse", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Change Scene {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=TimePasses())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=ResetSemiPersistentItems())
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade In", setValue=true, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=EnterWithoutInput(true))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AcceptInput())
    SetFsmColor(gameObject=var "MainCamera", fsmName="CameraFade", variableName="Fade Colour", setValue=(1, 1, 1, 1), everyFrame=false)  // disabled
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=["Dream_Abyss", "door1", 0], storeResult=(unset var))  // disabled
    BeginSceneTransition(sceneName="Dream_Abyss", entryGateName="door1", entryDelay=0, visualization=Dream, preventCameraFadeOut=true)
  }

  state Wait {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Black
  }

  state Black {
    SetFsmBool(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Cancel Reset", setValue=true, everyFrame=false)
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.001, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=en Silent (resources.assets), transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Change Scene
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inactive {
  }

  var Hero Scale: float = 0
  var Royal Charm State: int = 0
  var Hero Pos: vector3 = (0, 0, 0)
  var Dummy: gameObject = Dream Enter Abyss/Knight Dummy
  var Impact: gameObject = Dream Enter Abyss/Impact
  var Pt Wave: gameObject = Dream Enter Abyss/Pt Wave
  var Self: gameObject = <null>
  var White Flash: gameObject = Dream Enter Abyss/White Flash
}
