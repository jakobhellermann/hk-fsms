# Get Shadow Dash
# 1 location(s):
#   Abyss_10 / higher_being/Dish Plat

fsm Get Shadow Dash {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasShadowDash", isTrue=→"GOT", isFalse=(none))
    GetHero(storeResult=var "Hero")
    FindChild(gameObject=Self, childName="Black Wave", storeResult=var "Black Wave")
    FindChild(gameObject=Self, childName="Knight Dummy", storeResult=var "Knight Dummy")
    FindChild(gameObject=Self, childName="Particle Wave", storeResult=var "Particle Wave")
    FindChild(gameObject=Self, childName="Pt Burst", storeResult=var "Pt Burst")
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
    on GOT → Got
  }

  state Idle {
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"COLLISION", storeCollider=var "", storeForce=(unset))
    on COLLISION → Check Pause
  }

  state Check Grounded {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["onGround"], storeResult=var "On Ground")
    BoolTest(boolVariable=var "On Ground", isTrue=→"ON GROUND", isFalse=→"OFF GROUND", everyFrame=false)
    on OFF GROUND → Idle
    on ON GROUND → Take Control
  }

  state Take Control {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetCState", parameters=["onGround", true], storeResult=(unset var))
    ActivateGameObject(gameObject=var "Gate Solid", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Set Respawn
  }

  state Fall {
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetScale(gameObject=var "Knight Dummy", vector=(unset), x=var "Hero Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=var "Knight Dummy", active=true)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    GetPosition(gameObject=var "Hero", vector=var "To Vector", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Knight Dummy", vector=var "To Vector", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Knight Dummy", animLibName="", clipName="Fall")
    SetVector3XYZ(vector3Variable=var "To Vector", vector3Value=(unset), x=var "Self X", y=18.52, z=(unset), everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=water_splash (sharedassets31.assets))
    iTweenMoveTo(gameObject=var "Hero", id="", transformPosition=var "", vectorPosition=var "To Vector", time=0.4, delay=0, speed=(unset), space=0, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenMoveTo(gameObject=var "Knight Dummy", id="", transformPosition=var "", vectorPosition=var "To Vector", time=0.4, delay=0, speed=(unset), space=0, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SpawnRandomObjectsV2(gameObject=Spatter Black (resources.assets), spawnPoint=var "Hero", position=(0, -0.7, 0), spawnMin=6, spawnMax=8, speedMin=7, speedMax=15, angleMin=80, angleMax=100, originVariationX=0, originVariationY=0)
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake 1
  }

  state Check Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Grounded
  }

  state Shake 1 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=water_splash (sharedassets31.assets))
    Tk2dPlayAnimation(gameObject=var "Knight Dummy", animLibName="", clipName="Shake")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake 2
  }

  state Shake 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=water_splash (sharedassets31.assets))
    Tk2dPlayAnimation(gameObject=var "Knight Dummy", animLibName="", clipName="Shake")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shaking
  }

  state Shaking {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=void_cloak_get_fill_up (sharedassets337.assets))
    Tk2dPlayAnimation(gameObject=var "Knight Dummy", animLibName="", clipName="Shaking")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emerge 1
  }

  state Emerge 1 {
    Tk2dPlayAnimationWithEvents(gameObject=var "Knight Dummy", clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Emerge 2
  }

  state Emerge 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=dark_spell_get (resources.assets))
    Tk2dPlayAnimationWithEvents(gameObject=var "Knight Dummy", clipName="Emerge 2", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Particle Wave", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Burst", emit=0)
    Wait(time=1.25, finishEvent=→"GOT", realTime=false)
    ActivateGameObject(gameObject=var "Black Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Knight Dummy", audioClip=hollow_knight_firewalls_produce (sharedassets337.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on GOT → UI Msg
  }

  state PlayerData {
    SetPlayerDataBool(boolName="hasShadowDash", value=true)
    on FINISHED → Fall
  }

  state UI Msg {
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=_0000_shadow-dash (sharedassets337.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Shadow Dash", everyFrame=false)
    on GET ITEM MSG END → Get Up
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Knight Dummy", clipName="Rise", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End
  }

  state End {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SetMeshRenderer(gameObject=var "Knight Dummy", active=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SendEventToRegister(eventName="GOT SHADOW DASH")
  }

  state Got {
  }

  state Set Respawn {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetBenchRespawn", parameters=["Death Respawn Marker", "Abyss_10", 0, false], storeResult=(unset var))
    on FINISHED → PlayerData
  }

  var Gate Solid: gameObject = higher_being/shadow_gate/solid

  var Hero Scale: float = 0
  var Self X: float = 0
  var On Ground: bool = false
  var To Vector: vector3 = (0, 0, 0)
  var Black Wave: gameObject = <null>
  var Hero: gameObject = <null>
  var Knight Dummy: gameObject = <null>
  var Particle Wave: gameObject = <null>
  var Pt Burst: gameObject = <null>
  var UI Msg: gameObject = <null>
}
