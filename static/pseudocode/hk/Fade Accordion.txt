# Fade Accordion
# 1 location(s):
#   Grimm_Main_Tent / Brum NPC

fsm Fade Accordion {
  start Idle

  state Idle {
    on ACCORDION END → Fade
  }

  state Fade {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=1)
    on ACCORDION START → Pause
  }

  state Start {
    AudioStop(gameObject=Self)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on ACCORDION END → Fade
  }

  state Pause {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start
  }
}
