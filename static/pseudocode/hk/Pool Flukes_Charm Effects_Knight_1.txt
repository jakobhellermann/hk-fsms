fsm Pool Flukes {
  start State 1
  on CHARM EQUIP CHECK → State 1  // from any state

  state State 1 {
    GetPlayerDataInt(gameObject=var "GameManager", intName="fireballLevel", storeValue=var "Fireball Level")
    IntCompare(integer1=var "Fireball Level", integer2=1, equal=→"LEVEL 1", lessThan=→"FINISHED", greaterThan=→"LEVEL 2", everyFrame=false)
    on LEVEL 1 → Pool Normal
    on LEVEL 2 → Pool Black
    on FINISHED → Idle
  }

  state Idle {
  }

  state Pool Black {
    CreateGameObjectPool(prefab=Spell Fluke Black, amount=64, useExisting=true)
  }

  state Pool Normal {
    CreateGameObjectPool(prefab=Spell Fluke, amount=36, useExisting=true)
  }

  var Fireball Level: int = 0
}
