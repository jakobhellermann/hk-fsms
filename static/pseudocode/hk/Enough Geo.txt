fsm Enough Geo {
  start Init

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="geo", storeValue=var "Geo")
    GetParent(gameObject=Self, storeResult=var "UI List")
    GetParent(gameObject=var "UI List", storeResult=var "Text YN")
    GetFsmInt(gameObject=var "Text YN", fsmName="Dialogue Page Control", variableName="Toll Cost", storeValue=var "Toll Cost", everyFrame=false)
    IntCompare(integer1=var "Toll Cost", integer2=var "Geo", equal=(none), lessThan=(none), greaterThan=→"NOT ENOUGH", everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    FindChild(gameObject=Self, childName="Not Enough", storeResult=var "Not Enough")
    ActivateGameObject(gameObject=var "Not Enough", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Unselectable", boolValue=false, everyFrame=false)
    on NOT ENOUGH → Not Enuff
  }

  state Not Enuff {
    SetMeshRenderer(gameObject=Self, active=false)
    FindChild(gameObject=Self, childName="Not Enough", storeResult=var "Not Enough")
    ActivateGameObject(gameObject=var "Not Enough", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Unselectable", boolValue=true, everyFrame=false)
  }

  var Geo: int = 0
  var Toll Cost: int = 0
  var Unselectable: bool = false
  var Not Enough: gameObject = <null>
  var Text YN: gameObject = <null>
  var UI List: gameObject = <null>
}
