# Spike Control
# 2 location(s):
#   Grimm_Main_Tent_boss / Grimm Spike Holder
#   GG_Grimm / Grimm Spike Holder

fsm Spike Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Smoke", storeResult=var "Smoke")
    on FINISHED → Idle
  }

  state Idle {
    on SPIKE ATTACK → Ready
  }

  state Ready {
    RandomFloatEither(value1=66, value2=67.125, storeResult=var "X Pos")
    SetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPIKES READY", delay=0, everyFrame=false)
    Wait(time=0.55, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
    on SPIKES DOWN → Idle
  }

  state Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPIKES UP", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=grimm_spikes_pt_2_shoot_up (sharedassets392.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Shake
    on SPIKES DOWN → Idle
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPIKES DOWN", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=grimm_spikes_pt_3_shrivel_back (sharedassets392.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Shake {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Wait(time=0.45, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Smoke", emit=0)
    on FINISHED → Down
    on SPIKES DOWN → Idle
  }

  var X Pos: float = 0
  var Smoke: gameObject = <null>
}
