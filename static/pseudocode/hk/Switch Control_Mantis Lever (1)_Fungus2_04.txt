# Switch Control
# 2 location(s):
#   Fungus2_04 / Mantis Lever (1)
#   Fungus2_15 / Mantis Lever (1)

fsm Switch Control {
  start Pause

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    FindChild(gameObject=Self, childName="Hit Origin", storeResult=var "Hit Origin")
    BoolTest(boolVariable=var "Activated", isTrue=→"TRUE", isFalse=(none), everyFrame=false)
    on TRUE → Activated
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(collideTag="", sendEvent=→"NAIL HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on NAIL HIT → Check If Nail
  }

  state Check If Nail {
    BoolTest(boolVariable=var "In Range", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0], sendEvent=[→"TRUE"], everyFrame=false)
    on TRUE → Player Data?
    on FINISHED → Idle
    on CANCEL → Idle
  }

  state Hit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_switch (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Hit Origin", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lever Hit")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=GameObject(var "Target"), sendEvent="OPEN", delay=0, everyFrame=false)
  }

  state Activated {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lever Activated")
    SendEventByName(eventTarget=GameObject(var "Target"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state Player Data? {
    BoolTest(boolVariable=var "SetPlayerData", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetPlayerDataBool(boolName=var "Player Data", value=true)
    on CANCEL → Hit
    on FINISHED → Hit
  }

  var SetPlayerData: bool = false
  var Player Data: string = ""
  var Target: gameObject = Mantis Gate (1)

  var Attack Direction: float = 0
  var Recoil Speed: float = 1
  var Recoil Speed Neg: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var TimeTest: float = 0
  var Attack Type: int = 0
  var Activated: bool = false
  var In Range: bool = false
  var Camera: gameObject = <null>
  var Damager: gameObject = <null>
  var G1: gameObject = <null>
  var G2: gameObject = <null>
  var Hit Origin: gameObject = <null>
  var Self: gameObject = <null>
}
