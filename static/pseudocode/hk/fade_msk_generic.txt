# fade
# 1 location(s):
#   Crossroads_09 / Break Floor 1/msk_generic

fsm fade {
  start Pause

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"HIT", isFalse=(none), everyFrame=false)
    on HIT → Fade
  }

  state Fade {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.4, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Activated: bool = false
}
