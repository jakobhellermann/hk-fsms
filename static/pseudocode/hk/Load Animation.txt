# Load Animation
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/HUD_frame
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/HUD_frame

fsm Load Animation {
  start Pause Frame

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    on LOAD → First Pause
    on FINISHED → First Pause
  }

  state pause {
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    SetPlayerDataBool(boolName="enteredTutorialFirstTime", value=true)
    on FINISHED → Cracked?
  }

  state Appear Normal {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim Appear")
    on SOUL LIMITER UP → Cracked
  }

  state Cracked? {
    SetMeshRenderer(gameObject=Self, active=true)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="soulLimited", isTrue=→"CRACKED", isFalse=→"UNCRACKED")
    on CRACKED → Appear Cracked
    on UNCRACKED → Appear Normal
  }

  state Appear Cracked {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim CrackAppear")
    on SOUL LIMITER DOWN → Normal
  }

  state First Pause {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bossRushMode", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="enteredTutorialFirstTime", isTrue=→"FINISHED", isFalse=(none))
    Wait(time=9.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → pause
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Anims
  }

  state Normal {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim Idle")
    on SOUL LIMITER UP → Cracked
    on LEVEL LOADED → Check
  }

  state Cracked {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim Cracked")
    on SOUL LIMITER DOWN → Normal
    on LEVEL LOADED → Check
  }

  state Set Anims {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bossRushMode", isTrue=→"GG MODE", isFalse=(none))
    SetStringValue(stringVariable=var "Anim Appear", stringValue="HUD Frame", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Idle", stringValue="HUD Frame Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Anim CrackAppear", stringValue="HUD Frame CrackAppear", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Cracked", stringValue="HUD Frame Cracked", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="permadeathMode", storeValue=var "PermaDeath Mode")
    IntCompare(integer1=var "PermaDeath Mode", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetStringValue(stringVariable=var "Anim Appear", stringValue="HUD Frame St", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Idle", stringValue="HUD Frame Idle St", everyFrame=false)
    SetStringValue(stringVariable=var "Anim CrackAppear", stringValue="HUD Frame CrackAppear St", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Cracked", stringValue="HUD Frame Cracked St", everyFrame=false)
    on FINISHED → Init
    on GG MODE → GG Anims
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="soulLimited", isTrue=→"SOUL LIMITER UP", isFalse=→"SOUL LIMITER DOWN")
    on SOUL LIMITER DOWN → Normal
    on SOUL LIMITER UP → Cracked
  }

  state GG Anims {
    SetStringValue(stringVariable=var "Anim Appear", stringValue="HUD Frame GG", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Idle", stringValue="HUD Frame Idle GG", everyFrame=false)
    SetStringValue(stringVariable=var "Anim CrackAppear", stringValue="HUD Frame CrackAppear", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Cracked", stringValue="HUD Frame Cracked", everyFrame=false)
    on FINISHED → Init
  }

  var PermaDeath Mode: int = 0
  var Anim Appear: string = ""
  var Anim CrackAppear: string = ""
  var Anim Cracked: string = ""
  var Anim Idle: string = ""
}
