  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    FindChild(gameObject=Self, childName="Plat Stream Dust", storeResult=var "Plat Stream Dust")
    FindChild(gameObject=Self, childName="Slam Dust", storeResult=var "Slam Dust")
    FindChild(gameObject=Self, childName="Whole Dust", storeResult=var "Whole Dust")
    on ACTIVATE → Antic
  }

  state Antic {
    PlayParticleEmitter(gameObject=var "Whole Dust", emit=0)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 0.2, 0), time=0.5, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=1, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)

    SetPlayerDataBool(boolName="openedGardensStagStation", value=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=jiji_door_open_pt_1 (sharedassets7.assets))
    on FINISHED → Slide
  }

  state Slide {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 5.8, 0), time=1, delay=0, speed=(unset), easeType=easeInSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Tween Out
  }

  state Tween Out {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -0.2, 0), time=0.5, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)

    PlayParticleEmitter(gameObject=var "Slam Dust", emit=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=jiji_door_open_pt_2 (sharedassets7.assets))
    on FINISHED → Tween In
  }

  state Tween In {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 0.2, 0), time=0.75, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)

    PlayParticleEmitter(gameObject=var "Plat Stream Dust", emit=0)
  }

  var Plat Stream Dust: gameObject = <null>
  var Slam Dust: gameObject = <null>
  var Whole Dust: gameObject = <null>
}
