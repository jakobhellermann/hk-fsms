# Conversation Control
# 1 location(s):
#   Cliffs_06 / Brumm Lantern NPV

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Spark", storeResult=var "Pt Spark")
    FindChild(gameObject=Self, childName="Slam Effect", storeResult=var "Slam Effect")
    FindChild(gameObject=Self, childName="Armour Smash", storeResult=var "Armour Smash")
    FindChild(gameObject=Self, childName="Flash", storeResult=var "Flash")
    FindChild(gameObject=Self, childName="Pt Spark2", storeResult=var "Pt Spark2")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Convo {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRUMM_LANTERN", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_talk_03 (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down 2
  }

  state Antic {
    SetPlayerDataBool(boolName="brummBrokeBrazier", value=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stab Antic")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hollow_knight_self_stab_prepare (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_grunt_01 (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Stab
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  state Stab {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stab")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Spark2", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_damage_armour (sharedassets46.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=nightmare_bg_eye_break_1 (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_grunt_double (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Break
  }

  state Break {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Break")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BRAZIER CRACK", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Armour Smash", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Spark2", emit=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=nightmare_bg_eye_break_2 (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_damage_armour_final (sharedassets46.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Armour Smash: gameObject = <null>
  var Flash: gameObject = <null>
  var Pt Spark: gameObject = <null>
  var Pt Spark2: gameObject = <null>
  var Self: gameObject = <null>
  var Slam Effect: gameObject = <null>
}
