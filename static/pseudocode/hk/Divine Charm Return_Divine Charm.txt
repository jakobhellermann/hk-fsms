# Divine Charm Return
# 1 location(s):
#   Town / Divine Charm

fsm Divine Charm Return {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="destroyedNightmareLantern", isTrue=(none), isFalse=→"INACTIVE")
    FindChild(gameObject=Self, childName="Shiny Item", storeResult=var "Shiny Item")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gaveFragileHeart", falseBool="gotCharm_23", isTrue=→"HEART", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gaveFragileGreed", falseBool="gotCharm_24", isTrue=→"GEO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gaveFragileStrength", falseBool="gotCharm_25", isTrue=→"STRENGTH", isFalse=(none))
    on INACTIVE → Inactive
    on HEART → Heart
    on GEO → Geo
    on STRENGTH → Strength
  }

  state Inactive {
  }

  state Heart {
    SetFsmString(gameObject=var "Shiny Item", fsmName="Shiny Control", variableName="PD Bool Name", setValue="gotCharm_23", everyFrame=false)
    SetFsmInt(gameObject=var "Shiny Item", fsmName="Shiny Control", variableName="Charm ID", setValue=23, everyFrame=false)
    on FINISHED → Activate
  }

  state Activate {
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Geo {
    SetFsmString(gameObject=var "Shiny Item", fsmName="Shiny Control", variableName="PD Bool Name", setValue="gotCharm_24", everyFrame=false)
    SetFsmInt(gameObject=var "Shiny Item", fsmName="Shiny Control", variableName="Charm ID", setValue=24, everyFrame=false)
    on FINISHED → Activate
  }

  state Strength {
    SetFsmString(gameObject=var "Shiny Item", fsmName="Shiny Control", variableName="PD Bool Name", setValue="gotCharm_25", everyFrame=false)
    SetFsmInt(gameObject=var "Shiny Item", fsmName="Shiny Control", variableName="Charm ID", setValue=25, everyFrame=false)
    on FINISHED → Activate
  }

  var Shiny Item: gameObject = <null>
}
