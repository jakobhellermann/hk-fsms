# corpse
# 1 location(s):
#   sharedassets156.assets / Corpse Jellyfish

fsm corpse {
  start State 1

  state State 1 {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Bub Cloud", storeResult=var "Acid Bubble")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_death_sword (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_damage (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FindChild(gameObject=Self, childName="Spore Cloud", storeResult=var "Acid Spore")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer="", sendEvent=→"EXPLODE", storeCollider=var "")
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Wait(time=2, finishEvent=→"END", realTime=false)
    on COLLIDE → Explode
    on ACID → Splash
    on END → End
    on EXPLODE → Instant Explosion
  }

  state Explode {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=jelly_egg_empty_pop (sharedassets140.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    StopParticleEmitter(gameObject=var "Trail")
    CreateObject(gameObject=Bubble Death, spawnPoint=var "Self", position=(0, 1.4, 0), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    CreateObject(gameObject=Lil Jellyfish, spawnPoint=var "Self", position=(0, 1.4, 0), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    DestroySelf(detachChildren=true)
  }

  state Acid {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=acid_splash (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Corpse Acid Steam (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetGravity2dScale(gameObject=Self, gravityScale=0.05)
    DecelerateV2(gameObject=var "Self", deceleration=0.65)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Acid Bubble", emit=0)
    PlayParticleEmitter(gameObject=var "Acid Spore", emit=0)
    on FINISHED → Sizzle
  }

  state Sizzle {
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    DecelerateV2(gameObject=var "Self", deceleration=0.65)
    ColorInterpolate(colors=[(1, 1, 1, 1), (1, 1, 1, 0)], time=1, storeColor=var "Colour", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  state Finish {
    StopParticleEmitter(gameObject=var "Acid Bubble")
    StopParticleEmitter(gameObject=var "Acid Spore")
  }

  state Splash {
    GetOwner(storeGameObject=var "Self")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Burn")
    SetCollider(gameObject=var "Self", active=false)
    CreateObject(gameObject=Acid Splash (resources.assets), spawnPoint=var "Self", position=(0, 0, 0), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Acid
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Instant Explosion {
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Uumuu, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    DestroySelf(detachChildren=false)
  }

  var Collider Half Height: float = 0
  var Collider Offset Y: float = 0
  var Splash Spawn Pos: vector3 = (0, 0, 0)
  var Colour: color = (1, 1, 1, 1)
  var Acid Bubble: gameObject = <null>
  var Acid Spore: gameObject = <null>
  var Self: gameObject = <null>
  var Trail: gameObject = Corpse Jellyfish/Trail
}
