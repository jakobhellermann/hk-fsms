// uses template: destroy_out_of_camera
fsm FSM {
  start Wait
  on HAZARD RELOAD → Destroy  // from any state

  state Detect {
    GetOwner(storeGameObject=var "Self")
    GameObjectIsVisible(gameObject=var "Self", trueEvent=(none), falseEvent=→"FALSE", storeResult=(unset), everyFrame=true)
    on FALSE → Destroy
  }

  state Destroy {
    DestroyObject(gameObject=var "Self", delay=0, detachChildren=false)
  }

  state Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  var Vis Check: bool = false
  var Self: gameObject = <null>
}
