# Control
# 1 location(s):
#   Dream_Final_Boss / Boss Control/Spike Control

fsm Control {
  start Init
  on SPIKES OUTER → Spikes Outer  // from any state
  on SPIKES RIGHT → Spikes Right  // from any state
  on SPIKES LEFT → Spikes Left  // from any state
  on SPIKES NONE → None  // from any state
  on SPIKE WAVES → Spike Waves  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Centre", storeResult=var "Centre")
    FindChild(gameObject=Self, childName="Far L", storeResult=var "Far L")
    FindChild(gameObject=Self, childName="Far R", storeResult=var "Far R")
    FindChild(gameObject=Self, childName="Mid L", storeResult=var "Mid L")
    FindChild(gameObject=Self, childName="Mid R", storeResult=var "Mid R")
    on FINISHED → None
  }

  state None {
    SendEventByName(eventTarget=GameObject(var "Centre"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far L"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far R"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid L"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid R"), sendEvent="DOWN", delay=0, everyFrame=false)
  }

  state Spikes Left {
    SendEventByName(eventTarget=GameObject(var "Centre"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid L"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid R"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far L"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far R"), sendEvent="DOWN", delay=0, everyFrame=false)
  }

  state Spikes Right {
    SendEventByName(eventTarget=GameObject(var "Centre"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid L"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid R"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far L"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far R"), sendEvent="UP", delay=0, everyFrame=false)
  }

  state Spikes Outer {
    SendEventByName(eventTarget=GameObject(var "Centre"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid L"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid R"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far L"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far R"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far L"), sendEvent="UP", delay=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far R"), sendEvent="UP", delay=1, everyFrame=false)
    on SPIKES DOWN → Respike Pause
  }

  state Spike Waves {
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Wave L
    on R → Wave R
  }

  state Wave L {
    SetFsmBool(gameObject=var "Beam Sweeper", fsmName="Control", variableName="Force Left", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Beam Sweeper", fsmName="Control", variableName="Force Right", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Centre"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid L"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid R"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far L"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far R"), sendEvent="UP", delay=0, everyFrame=false)
    WaitRandom(timeMin=6, timeMax=8, finishEvent=→"R", realTime=false)
    on R → Wave R
  }

  state Wave R {
    SetFsmBool(gameObject=var "Beam Sweeper", fsmName="Control", variableName="Force Left", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Beam Sweeper", fsmName="Control", variableName="Force Right", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Centre"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid L"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Mid R"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far L"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far R"), sendEvent="DOWN", delay=0, everyFrame=false)
    WaitRandom(timeMin=6, timeMax=8, finishEvent=→"L", realTime=false)
    on L → Wave L
  }

  state Respike Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spikes Outer
  }

  var Beam Sweeper: gameObject = Boss Control/Beam Sweeper
  var Marker C: gameObject = Boss Control/Spike Control/Hazard Respawn Marker A1C
  var Marker L: gameObject = Boss Control/Spike Control/Hazard Respawn Marker A1L
  var Marker R: gameObject = Boss Control/Spike Control/Hazard Respawn Marker A1R

  var Respawn Pos L: vector3 = (0, 0, 0)
  var Respawn Pos R: vector3 = (0, 0, 0)
  var Centre: gameObject = <null>
  var Far L: gameObject = <null>
  var Far R: gameObject = <null>
  var Mid L: gameObject = <null>
  var Mid R: gameObject = <null>
}
