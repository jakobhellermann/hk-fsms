    SendEventByName(eventTarget=GameObject(Self), sendEvent="EXTRA KILL", delay=0, everyFrame=false)

  }

    SendEventByName(eventTarget=GameObject(Self), sendEvent="EXTRA KILL", delay=0, everyFrame=false)

  state Dung 2 {

  state Vuln 1 {
    GetTag(gameObject=var "Self", storeResult=var "Tag", everyFrame=false)
    StringCompare(stringVariable=var "Tag", compareTo="Spell Vulnerable", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
