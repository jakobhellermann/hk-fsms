# Mossy Control
# 2 location(s):
#   Fungus1_29 / Mega Moss Charger
#   GG_Mega_Moss_Charger / Mega Moss Charger

fsm Mossy Control {
  start Start Pause
  on ZERO HP → Detach  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindAlertRange(target=Self, storeResult=var "Attack Range Obj", childName="Attack Range")
    SetGameObject(variable=var "Camera", gameObject=var "CameraParent", everyFrame=false)
    FindChild(gameObject=Self, childName="Grass Burst", storeResult=var "Grass Burst")
    FindChild(gameObject=Self, childName="Grass Explode 1", storeResult=var "G Explode 1")
    FindChild(gameObject=Self, childName="Grass Roar", storeResult=var "Grass Roar")
    FindChild(gameObject=Self, childName="Charge Grass", storeResult=var "Charge Grass")
    FindChild(gameObject=Self, childName="Slam Effect R", storeResult=var "Slam Effect R")
    FindChild(gameObject=Self, childName="Grass Dig", storeResult=var "Grass Dig")
    FindChild(gameObject=Self, childName="Grass Burst Down", storeResult=var "Grass Burst Down")
    FindChild(gameObject=Self, childName="Leap Hit", storeResult=var "Leap Hit")
    FindChild(gameObject=Self, childName="Grass Submerge", storeResult=var "Grass Submerge")
    FindChild(gameObject=Self, childName="Range Grass", storeResult=var "Range Grass")
    GetHero(storeResult=var "Hero")
    SetCollider(gameObject=Self, active=false)
    FindChild(gameObject=Self, childName="Attack Range", storeResult=var "Attack Range")
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    BoundsBoxCollider(gameObject1=var "Attack Range", scaleVector2=(unset), scaleX=var "Attack Range Length", scaleY=(unset), everyFrame=false)
    FloatDivide(floatVariable=var "Attack Range Length", divideBy=2, everyFrame=false)
    SetParent(gameObject=var "Attack Range", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Range Grass", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetFloatValue(floatVariable=var "X Min", floatValue=var "Start X", everyFrame=false)
    FloatOperator(float1=var "X Min", float2=var "Attack Range Length", operation=Subtract, storeResult=var "X Min", everyFrame=false)
    SetFloatValue(floatVariable=var "X Max", floatValue=var "Start X", everyFrame=false)
    FloatOperator(float1=var "X Max", float2=var "Attack Range Length", operation=Add, storeResult=var "X Max", everyFrame=false)
    FloatOperator(float1=var "X Min", float2=3, operation=Add, storeResult=var "X Min", everyFrame=false)
    FloatOperator(float1=var "X Max", float2=3, operation=Subtract, storeResult=var "X Max", everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="megaMossChargerEncountered", isTrue=→"SUBMERGE", isFalse=(none))
    on FINISHED → Sleep
    on SUBMERGE → Hidden
    on GG BOSS → GG Pause
  }

  state Hidden {
    CheckAlertRange(alertRange=var "Attack Range Obj", storeResult=var "In Attack Range", everyFrame=true)
    BoolTest(boolVariable=var "In Attack Range", isTrue=→"IN RANGE", isFalse=(none), everyFrame=true)
    SetMeshRenderer(gameObject=Self, active=false)
    PlayParticleEmitter(gameObject=var "Range Grass", emit=0)
    on IN RANGE → Emerge Pause
  }

  state Emerge Pause {
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero Beyond?
  }

  state Left or Right? {
    SendRandomEvent(events=[→"RIGHT", →"LEFT"], weights=[0.5, 0.5], delay=0)
    on LEFT → Emerge Left
    on RIGHT → Emerge Right
  }

  state Emerge Right {
    SetVector2XY(vector2Variable=var "RayDown X", vector2Value=(unset), x=-6.5, y=-0.5, everyFrame=false)
    SetVector2XY(vector2Variable=var "RayForward Direction", vector2Value=(unset), x=-1, y=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    RandomFloat(min=14, max=14, storeResult=var "Chooser")
    SetFloatValue(floatVariable=var "Appear X", floatValue=var "Hero X", everyFrame=false)
    FloatOperator(float1=var "Appear X", float2=var "Chooser", operation=Add, storeResult=var "Appear X", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Current Charge Speed", floatValue=var "Charge Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Current Charge Speed", multiplyBy=-1, everyFrame=false)
    FloatCompare(float1=var "Appear X", float2=var "X Max", tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"LEFT", everyFrame=false)
    on FINISHED → Music 2
    on LEFT → Pause
  }

  state Emerge Left {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "RayForward Direction", vector2Value=(unset), x=1, y=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "RayDown X", vector2Value=(unset), x=6.5, y=-0.5, everyFrame=false)
    RandomFloat(min=14, max=14, storeResult=var "Chooser")
    SetFloatValue(floatVariable=var "Appear X", floatValue=var "Hero X", everyFrame=false)
    FloatOperator(float1=var "Appear X", float2=var "Chooser", operation=Subtract, storeResult=var "Appear X", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Current Charge Speed", floatValue=var "Charge Speed", everyFrame=false)
    FloatCompare(float1=var "Appear X", float2=var "X Min", tolerance=0, equal=→"FINISHED", lessThan=→"RIGHT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Music 2
    on RIGHT → Pause 2
  }

  state Emerge {
    SendEventByName(eventTarget=GameObject(var "Camera"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_charger_emerge (sharedassets139.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFloatValue(floatVariable=var "Emerge Speed", floatValue=var "Current Charge Speed", everyFrame=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(false))
    FloatMultiply(floatVariable=var "Emerge Speed", multiplyBy=0.25, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Appear X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    StopParticleEmitter(gameObject=var "Range Grass")
    PlayParticleEmitter(gameObject=var "Grass Burst", emit=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Emerge Speed", y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Charge Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Charge
  }

  state Charge {
    AudioPlayInState(gameObject=Self, volume=1)
    PlayParticleEmitter(gameObject=var "Charge Grass", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Charge")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Current Charge Speed", y=(unset), everyFrame=false)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"SUBMERGE", storeCollider=var "", storeForce=(unset))
    on SUBMERGE → Submerge 1
    on HIT RIGHT → Hit Right
    on HIT LEFT → Hit Left
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Emerge Left
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Emerge Right
  }

  state Submerge 1 {
    StopParticleEmitter(gameObject=var "Charge Grass")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_charger_return (sharedassets139.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Decelerate(gameObject=Self, deceleration=0.7)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Disappear 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Charge Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Crouch Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Submerge 2
  }

  state Submerge CD {
    SetCollider(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Grass Submerge")
    StopParticleEmitter(gameObject=var "Grass Dig")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Play Range
  }

  state Hero Beyond? {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "X Max", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "X Min", tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Hidden
    on FINISHED → Left or Right?
  }

  state Play Range {
    PlayParticleEmitter(gameObject=var "Range Grass", emit=0)
    on FINISHED → Hidden
  }

  state Detach {
    StopParticleEmitter(gameObject=var "Range Grass")
  }

  state Submerge 2 {
    Decelerate(gameObject=Self, deceleration=0.7)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Disappear 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Crouch Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Charge Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Submerge 3
  }

  state Submerge 3 {
    Decelerate(gameObject=Self, deceleration=0.7)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Disappear 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Burrow Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Charge Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Grass Submerge", emit=0)
    on FINISHED → Submerge 4
  }

  state Submerge 4 {
    Decelerate(gameObject=Self, deceleration=0.7)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Disappear 4", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Burrow Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Submerge CD
  }

  state Hit Right {
    FloatCompare(float1=var "Current Charge Speed", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=10, y=(unset), everyFrame=false)
    on FINISHED → Charge
  }

  state Hit Left {
    FloatCompare(float1=var "Current Charge Speed", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-10, y=(unset), everyFrame=false)
    on FINISHED → Charge
  }

  state Sleep {
    on WAKE → Shake
  }

  state Shake {
    SetAudioClip(gameObject=Self, audioClip=moss_knight_rustle (sharedassets137.assets))
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetPlayerDataBool(boolName="megaMossChargerEncountered", value=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shake")
    PlayParticleEmitter(gameObject=var "Grass Shake", emit=0)
    SetFsmBool(gameObject=var "Camera", fsmName="CameraShake", variableName="RumblingSmall", setValue=true, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Title?
  }

  state Roar {
    AudioStop(gameObject=Self)
    SetAudioClip(gameObject=Self, audioClip=moss_charger_charging (sharedassets139.assets))
    StopParticleEmitter(gameObject=var "Grass Shake")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar")
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Roar Point", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SetFsmBool(gameObject=var "Camera", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Grass Roar", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mawlek_scream (sharedassets45.assets), pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_charger_emerge (sharedassets139.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar End
  }

  state Roar End {
    StopParticleEmitter(gameObject=var "Grass Roar")
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    on FINISHED → Submerge 1
  }

  state Attack Choice {
    SendRandomEventV2(events=[→"CHARGE", →"LEAP"], weights=[1, 1], trackingInts=[var "Ct Charge", var "Ct Leap"], eventMax=[3, 2])
    on CHARGE → Emerge
    on LEAP → Leap Start
  }

  state Leap Start {
    SendEventByName(eventTarget=GameObject(var "Camera"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_charger_emerge (sharedassets139.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Leap Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=var "Appear X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    StopParticleEmitter(gameObject=var "Range Grass")
    PlayParticleEmitter(gameObject=var "Grass Burst", emit=0)
    on FINISHED → Leap Launch
  }

  state Leap Launch {
    ActivateGameObject(gameObject=var "Leap Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Leap")
    FloatMultiply(floatVariable=var "Current Charge Speed", multiplyBy=1.2, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Current Charge Speed", y=30, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1.25)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Air
  }

  state In Air {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Start Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    on LAND → Land
  }

  state Land {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Leap Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Camera"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mimic_spider_land (sharedassets156.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Submerge 1
  }

  state Start Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
    on CANCEL → Dormant
  }

  state Dormant {
  }

  state GG Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  state Title? {
    GGCheckIfBossScene(bossSceneEvent=(none), regularSceneEvent=→"FINISHED")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MEGA_MOSS", everyFrame=false)
    on FINISHED → Music
  }

  state Music {
    GGCheckIfBossSequence(trueEvent=→"FINISHED", falseEvent=(none))
    TransitionToAudioSnapshot(snapshot=Action and Sub (resources.assets), transitionTime=0.5)
    GGCheckIfBossScene(bossSceneEvent=(none), regularSceneEvent=→"FINISHED")
    ApplyMusicCue(musicCue=Greenpath (sharedassets36.assets), delayTime=0, transitionTime=0)
    on FINISHED → Roar
  }

  state Music 2 {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Action and Sub (resources.assets), transitionTime=0.5)
    on FINISHED → Attack Choice
  }

  var Accel: float = 0.13
  var Appear X: float = 0
  var Attack Direction: float = 0
  var Attack Range Length: float = 0
  var Charge Speed: float = 18
  var Chooser: float = 0
  var Current Charge Speed: float = 0
  var Emerge Speed: float = 0
  var Hero X: float = 0
  var Max Speed L: float = -6
  var Max Speed R: float = 6
  var Recoil Speed: float = 0
  var Self Y: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Velocity: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var Ct Charge: int = 0
  var Ct Leap: int = 0
  var Looper: int = 0
  var Ms Charge: int = 0
  var Ms Leap: int = 0
  var Dig Check: bool = false
  var In Attack Range: bool = false
  var RayDown1: bool = false
  var left: bool = false
  var right: bool = false
  var RayDown X: vector2 = (0, 0)
  var RayForward Direction: vector2 = (0, 0)
  var Area Title: gameObject = <null>
  var Attack Range: gameObject = <null>
  var Burrow Hit: gameObject = Mega Moss Charger/Burrow Hit
  var Camera: gameObject = <null>
  var Charge Grass: gameObject = <null>
  var Charge Hit: gameObject = Mega Moss Charger/Charge Hit
  var Crouch Hit: gameObject = Mega Moss Charger/Crouch Hit
  var G Explode 1: gameObject = <null>
  var Grass Burst: gameObject = <null>
  var Grass Burst Down: gameObject = Mega Moss Charger/Grass Burst Down
  var Grass Dig: gameObject = Mega Moss Charger/Grass Dig
  var Grass Roar: gameObject = Mega Moss Charger/Grass Shake
  var Grass Shake: gameObject = Mega Moss Charger/Grass Shake
  var Grass Submerge: gameObject = Mega Moss Charger/Grass Submerge
  var Hero: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Leap Hit: gameObject = <null>
  var Range Grass: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Roar Point: gameObject = Mega Moss Charger/Roar Point
  var Self: gameObject = <null>
  var Slam Effect R: gameObject = <null>
  var Tent Box: gameObject = Jellyfish/Tentacle Box (sharedassets156.assets)
  var Attack Range Obj: object = <null>
}
