// uses template: npc_dream_dialogue
fsm npc_dream_dialogue {
  start Init
  on CONVO CANCEL → Idle  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Active Pt", storeResult=var "Active Pt")
    FindChild(gameObject=Self, childName="Impact Pt", storeResult=var "Impact Pt")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Take Control
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Convo", setValue=true, everyFrame=false)
    on FINISHED → Impact
  }

  state Impact {
    CreateObject(gameObject=dream_area_effect, spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Impact Pt", emit=0)
    SetParticleEmission(gameObject=var "Active Pt", emission=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM DIALOGUE START", delay=0, everyFrame=false)
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP DREAM", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo
  }

  state Convo {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Convo Name", var "Sheet Name"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN DREAM", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM AREA DISABLE", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Active Pt", emission=false)
    on FINISHED → Relinquish Control
  }

  state Relinquish Control {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM DIALOGUE END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Convo Name: string = "GRUB_BOTTLE_DREAM"
  var Sheet Name: string = "Lore Tablets"

  var Active Pt: gameObject = Dream Dialogue/Active Pt
  var Impact Pt: gameObject = Dream Dialogue/Impact Pt
  var Self: gameObject = <null>
}
