fsm Heal HP {
  start Idle

  state Idle {
    SetFsmBool(gameObject=Self, fsmName="Sound", variableName="Health", setValue=false, everyFrame=false)
    on START HEAL → Set Healing
  }

  state Heal Pause {
    Wait(time=0.7, finishEvent=→"WAIT", realTime=false)
    on STOP HEAL → Idle
    on WAIT → Heal
  }

  state Heal {
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxHealth", storeValue=var "Max HP")
    IntCompare(integer1=var "HP", integer2=var "Max HP", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddHealth", parameters=[1], storeResult=(unset var))
    CreateObject(gameObject=Spa Heal HP Burst (sharedassets31.assets), spawnPoint=var "Hero", position=(0, 0, -0.1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on STOP HEAL → Idle
    on FINISHED → Not Healing?
  }

  state Set Healing {
    SetFsmBool(gameObject=Self, fsmName="Sound", variableName="Health", setValue=true, everyFrame=false)
    on FINISHED → Heal
  }

  state Not Healing? {
    IntCompare(integer1=var "HP", integer2=var "Max HP", equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Sound", variableName="Health", setValue=false, everyFrame=false)
    on FINISHED → Heal Pause
  }

  var HP: int = 0
  var Max HP: int = 0
}
