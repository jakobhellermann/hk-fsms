# Battle End
# 2 location(s):
#   Deepnest_40 / Warrior/Galien Hammer
#   GG_Ghost_Galien / Warrior/Galien Hammer

fsm Battle End {
  start Idle

  state Idle {
    on GHOST DEATH → Slow
  }

  state Slow {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    ActivateGameObject(gameObject=var "Hit End", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Spin", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Weapon Idle")
    AudioStop(gameObject=Self)
    on FINISHED → Fall
  }

  state Fall {
    SetGravity2dScale(gameObject=Self, gravityScale=1)
  }

  var Hit End: gameObject = Warrior/Galien Hammer/Hit End
  var Hit Idle: gameObject = Warrior/Galien Hammer/Hit Idle
  var Hit Spin: gameObject = Warrior/Galien Hammer/Hit Spin
}
