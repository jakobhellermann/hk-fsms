# Control
# 1 location(s):
#   sharedassets32.assets / Zote Boss

fsm Control {
  start Init
  on HIT DOWN → Hit Down  // from any state
  on HIT RIGHT → Hit Right  // from any state
  on HIT LEFT → Hit Left  // from any state
  on HIT UP → Hit Up  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Land", storeResult=var "Pt Land")
    FindChild(gameObject=Self, childName="Pt Run", storeResult=var "Pt Run")
    FindChild(gameObject=Self, childName="Bonker", storeResult=var "Bonker")
    FindChild(gameObject=Self, childName="Attack Voice", storeResult=var "Attack Voice")
    on FINISHED → Tumble Out
  }

  state Tumble Out {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll")
    RandomFloat(min=70, max=110, storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=20, everyFrame=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    on FINISHED → GG Tumble
  }

  state In Air {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Prev X Speed", y=(unset), space=0, everyFrame=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=→"LAND", leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=→"LAND", leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetFloatValue(floatVariable=var "Prev X Speed", floatValue=var "X Speed", everyFrame=true)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=true)
    on WALL → Reflect
    on LAND → Trip?
  }

  state Reflect {
    FloatMultiply(floatVariable=var "Prev X Speed", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Prev X Speed", y=(unset), everyFrame=false)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → In Air
  }

  state Fall Over {
    SetBoolValue(boolVariable=var "Will Fall", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    DecelerateXY(gameObject=Self, decelerationX=0.85, decelerationY=(unset))
    WaitRandom(timeMin=0.75, timeMax=1.25, finishEvent=→"FINISHED", realTime=false)
    StopParticleEmitter(gameObject=var "Pt Run")
    ActivateGameObject(gameObject=var "Bonker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_land, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zote_battle_fall_01, Zote_battle_fall_02, Zote_battle_wound_02, Zote_battle_wound_04], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Get Up
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Get Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Roar?
  }

  state Roar? {
    BoolTest(boolVariable=var "Roared", isTrue=→"ROARED", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on ROARED → Idle
    on FINISHED → Roar Antic
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="TurnToIdle", resetFrame=true, everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move Choice
  }

  state Roar {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD LAUGH", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=Zote_battle_roar, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC 6", delay=0, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar")
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ZOTE", everyFrame=false)
    on FINISHED → Roar End
  }

  state Roar Antic {
    SetBoolValue(boolVariable=var "Roared", boolValue=true, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on FINISHED → Roar
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Roar Recover
  }

  state Roar Recover {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    CallMethodProper(gameObject=Self, behaviour="NonBouncer", methodName="SetActive", parameters=[false], storeResult=(unset var))
    MakeEnemyDreamnailReactionReady(target=Self)
    on FINISHED → Idle
  }

  state Move Choice {
    SendRandomEventV2(events=[→"ATTACK", →"JUMP"], weights=[0.35, 0.65], trackingInts=[var "Ct Attack", var "Ct Jump"], eventMax=[2, 3])
    on ATTACK → Attack Antic
    on JUMP → Jump Aim
  }

  state Jump Aim {
    RandomFloat(min=-18, max=18, storeResult=var "Jump X")
    RandomFloat(min=20, max=30, storeResult=var "Jump Y")
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatCompare(float1=var "Jump X", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Jump Antic
  }

  state Jump Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_jump (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Jump Launch
  }

  state Jump Launch {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll")
    NextFrameEvent(sendEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zote_battle_leap_01, Zote_battle_leap_02, Zote_battle_leap_03], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → In Air
  }

  state Trip? {
    PlayParticleEmitter(gameObject=var "Pt Land", emit=0)
    BoolTest(boolVariable=var "Roared", isTrue=(none), isFalse=→"TRIP", everyFrame=false)
    BoolTest(boolVariable=var "Will Fall", isTrue=→"TRIP", isFalse=(none), everyFrame=false)
    SendRandomEvent(events=[→"TRIP", →"LAND"], weights=[1, 1], delay=0)
    on TRIP → Fall Over
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_ground_land (sharedassets28.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Attack Antic {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Attack Chase
  }

  state Attack Chase {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Prev X Speed", y=(unset), space=0, everyFrame=false)
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=9, acceleration=0.2, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=true)
    ActivateGameObject(gameObject=var "Bonker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetFloatValue(floatVariable=var "Prev X Speed", floatValue=var "X Speed", everyFrame=true)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=true)
    PlayParticleEmitter(gameObject=var "Pt Run", emit=0)
    WaitRandom(timeMin=3, timeMax=6, finishEvent=→"END", realTime=false)
    AudioPlayInState(gameObject=var "Attack Voice", volume=1)
    on WALL → Hit Wall
    on END → Fall Over
  }

    ActivateGameObject(gameObject=var "Bonker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)

    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Prev X Speed", y=15, everyFrame=false)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    StopParticleEmitter(gameObject=var "Pt Run")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zote_battle_wound_01, Zote_battle_wound_02, Zote_battle_wound_03, Zote_battle_fall_01], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → In Air
  }

  }
