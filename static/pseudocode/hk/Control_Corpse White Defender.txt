# Control
# 1 location(s):
#   sharedassets398.assets / Corpse White Defender

fsm Control {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Self Y", minValue=8.54, maxValue=99999, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="JUST BURST", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DUNGBALL END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DUNG DEFENDER DEFEAT", delay=0, everyFrame=false)
    GetHero(storeResult=var "Hero")
    SetGameObject(variable=var "Camera", gameObject=var "CameraParent", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X", float2=75, tolerance=0, equal=(none), lessThan=→"FACE R", greaterThan=→"FACE L", everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X", float2=75, tolerance=0, equal=(none), lessThan=→"FACE R", greaterThan=→"FACE L", everyFrame=false)
    on FACE L → Face L 2
    on FACE R → Face R 2
  }

  state Face L {
    SetScale(gameObject=Self, vector=(unset), x=-1.1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Land
  }

  state In Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land 3
  }

  state Face R {
    SetScale(gameObject=Self, vector=(unset), x=1.1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")
    Wait(time=2.2, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Arm Up
  }

  state Dissipate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="FRIENDS OUT", delay=0, everyFrame=false)
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=dream_plant_emit_orbs (sharedassets228.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Slump
  }

  state Take Control {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    on FINISHED → Set Outro Msg
  }

  state Slump {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death End")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=WD_hand_raise_02, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Fade Out
  }

  state Fade Out {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WHITE PALACE END", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_enter (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=5.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Take Control
  }

  state New Scene {
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    SetTextMeshProColor(gameObject=var "DialogueText", color=(1, 1, 1, 1), everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=["Waterways_15", "door_dreamReturn", 0], storeResult=(unset var))  // disabled
    BeginSceneTransition(sceneName="Waterways_15", entryGateName="door_dreamReturn", entryDelay=0, visualization=Dream, preventCameraFadeOut=true)
  }

  state Outro Msg Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Convo", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Wake up?
  }

  state Land 3 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dung_defender_lands_on_distant_mound (sharedassets355.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Steam 2
  }

  state Steam 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_gushing (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Death Puff Boss Dung (sharedassets355.assets), spawnPoint=var "Self", position=(0, -1, -5), rotation=(unset), storeObject=var "Smoker", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Camera", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    FloatAdd(floatVariable=var "emissionRate", add=5, everyFrame=true, perSecond=false)
    FloatAdd(floatVariable=var "emissionSpeed", add=1, everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "emissionSpeed", minValue=0, maxValue=110, everyFrame=true)
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterInState(gameObject=var "Pt Death")
    on FINISHED → Ready 2
  }

  state Ready 2 {
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    EaseFloat(fromValue=var "emissionRate", toValue=0, floatVariable=var "emissionRate", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "emissionSpeed", toValue=0, floatVariable=var "emissionSpeed", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Blow 2
  }

  state Blow 2 {
    SetFsmBool(gameObject=var "Camera", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="StopRumble", delay=0, everyFrame=false)
    CreateObject(gameObject=Death Explode Dung (sharedassets355.assets), spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Smoker", emission=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    on FINISHED → Facing
    on GG BOSS → State 1
  }

  state Pause {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    SetPlayerDataBool(boolName="whiteDefenderDefeated", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=Dung_def_death_part_02 (sharedassets355.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["WHITE_DEFENDER"], storeResult=(unset var))
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="whiteDefenderDefeats")
    on FINISHED → Music
  }

  state Set Outro Msg {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=WD_outro, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    GetPlayerDataInt(gameObject=var "GameManager", intName="whiteDefenderDefeats", storeValue=var "Convo Int")
    IntClamp(intVariable=var "Convo Int", minValue=1, maxValue=5, everyFrame=false)
    ConvertIntToString(intVariable=var "Convo Int", stringVariable=var "Convo Num", format="", everyFrame=false)
    BuildString(stringParts=["WHITE_DEFENDER_OUTRO_", var "Convo Num"], separator="", addToEnd=true, storeResult=var "Convo", everyFrame=false)
    SetTextMeshProColor(gameObject=var "DialogueText", color=(0, 0, 0, 1), everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=false, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=true, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    IntCompare(integer1=var "Convo Int", integer2=1, equal=→"FIRST", lessThan=→"FIRST", greaterThan=→"REPEAT", everyFrame=false)
    on FIRST → Outro Msg 1a
    on REPEAT → Outro Msg Repeat
  }

  state Outro Msg 1a {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["WHITE_DEFENDER_OUTRO_1a", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Outro Msg 1b
  }

  state Outro Msg 1b {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["WHITE_DEFENDER_OUTRO_1b", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Wake up?
  }

  state Arm Up {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=WD_hand_raise, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=1.3, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Dissipate
  }

  state Wake up? {
    IntCompare(integer1=var "Convo Int", integer2=5, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetPlayerDataBool(boolName="dungDefenderAwoken", value=true)
    on FINISHED → New Scene
  }

  state Facing {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X", float2=75, tolerance=0, equal=(none), lessThan=→"FACE R", greaterThan=→"FACE L", everyFrame=false)
    on FACE L → Face L
    on FACE R → Face R
  }

  state Face L 2 {
    SetScale(gameObject=Self, vector=(unset), x=-1.1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → In Air
  }

  state Face R 2 {
    SetScale(gameObject=Self, vector=(unset), x=1.1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → In Air
  }

  state State 1 {
  }

  state Music {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))  // disabled
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=2)
    on FINISHED → Init
  }

  var Self X: float = 0
  var Self Y: float = 0
  var emissionRate: float = 50
  var emissionSpeed: float = 5
  var Convo Int: int = 0
  var Convo: string = ""
  var Convo Num: string = ""
  var Camera: gameObject = <null>
  var Hero: gameObject = <null>
  var Pt Death: gameObject = Corpse White Defender/Pt Death
  var Self: gameObject = <null>
  var Smoker: gameObject = <null>
}
