# Reposition
# 1 location(s):
#   Fungus2_05 / Hollow_Shade Marker

fsm Reposition {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Dir
  }

  state Check Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on LEFT → Pos L
    on RIGHT → Pos R
  }

  state Pos L {
    SetPosition(gameObject=Self, vector=(unset), x=11.09, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
  }

  state Pos R {
    SetPosition(gameObject=Self, vector=(unset), x=49.15, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
  }
}
