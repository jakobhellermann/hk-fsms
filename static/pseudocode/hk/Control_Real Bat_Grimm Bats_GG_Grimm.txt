# Control
# 1 location(s):
#   GG_Grimm / Grimm Bats/Real Bat

fsm Control {
  start Init
  on BATS IN → Return  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    on BOSS AWAKE → Dormant
  }

  state Go Up {
    RandomFloat(min=1, max=4, storeResult=var "X Accel")
    RandomFloat(min=2, max=3, storeResult=var "Y Accel")
    RandomlyFlipFloat(storeResult=var "X Accel")
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Velocity", multiplyBy=0.3, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", everyFrame=false)
    on FINISHED → Set Max X
  }

  state Fly {
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Bat TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    AccelerateVelocity(gameObject=Self, xAccel=var "X Accel", yAccel=var "Y Accel", xMaxSpeed=var "Max Speed X", yMaxSpeed=var "Max Speed Y")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=var "Y Pos", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=84, tolerance=0, equal=(none), lessThan=→"GO RIGHT", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=89, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"GO LEFT", everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=10, tolerance=0, equal=(none), lessThan=→"GO UP", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=13, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"GO DOWN", everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=1.5, finishEvent=→"CHOOSE", realTime=false)
    on GO UP → Go Up
    on GO DOWN → Go Down
    on GO LEFT → Go Left
    on GO RIGHT → Go Right
    on CHOOSE → Choose
  }

  state Go Down {
    RandomFloat(min=1, max=4, storeResult=var "X Accel")
    RandomFloat(min=-2, max=-3, storeResult=var "Y Accel")
    RandomlyFlipFloat(storeResult=var "X Accel")
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Velocity", multiplyBy=0.3, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", everyFrame=false)
    on FINISHED → Set Max X
  }

  state Fly Start {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Bat TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    AccelerateVelocity(gameObject=Self, xAccel=var "X Accel", yAccel=var "Y Accel", xMaxSpeed=15, yMaxSpeed=10)
    on FINISHED → Fly
  }

  state Go Left {
    RandomFloat(min=-3, max=-5, storeResult=var "X Accel")
    RandomFloat(min=0.5, max=2, storeResult=var "Y Accel")
    RandomlyFlipFloat(storeResult=var "Y Accel")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Velocity", multiplyBy=0.5, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=false)
    on FINISHED → Set Max X
  }

  state Go Right {
    RandomFloat(min=3, max=5, storeResult=var "X Accel")
    RandomFloat(min=0.5, max=2, storeResult=var "Y Accel")
    RandomlyFlipFloat(storeResult=var "Y Accel")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Velocity", multiplyBy=0.5, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=false)
    on FINISHED → Set Max X
  }

  state Choose {
    SetCollider(gameObject=Self, active=true)
    SendRandomEvent(events=[→"GO UP", →"GO DOWN", →"GO LEFT", →"GO RIGHT"], weights=[1, 1, 1, 1], delay=0)
    on GO UP → Go Up
    on GO DOWN → Go Down
    on GO LEFT → Go Left
    on GO RIGHT → Go Right
  }

  state Dormant {
    on BATS OUT → Position
  }

  state Set Max X {
    FloatOperator(float1=var "X Accel", float2=0.25, operation=Multiply, storeResult=var "X Accel", everyFrame=false)
    FloatOperator(float1=var "X Accel", float2=10, operation=Multiply, storeResult=var "Max Speed X", everyFrame=false)
    FloatCompare(float1=var "Max Speed X", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    FloatMultiply(floatVariable=var "Max Speed X", multiplyBy=-1, everyFrame=false)
    on FINISHED → Set Max Y
  }

  state Set Max Y {
    FloatOperator(float1=var "Y Accel", float2=0.25, operation=Multiply, storeResult=var "Y Accel", everyFrame=false)
    FloatOperator(float1=var "Y Accel", float2=10, operation=Multiply, storeResult=var "Max Speed Y", everyFrame=false)
    FloatCompare(float1=var "Max Speed Y", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    FloatMultiply(floatVariable=var "Max Speed Y", multiplyBy=-1, everyFrame=false)
    on FINISHED → Fly Start
  }

  state Position {
    GetPosition(gameObject=var "Grimm", vector=var "Grimm Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Grimm Pos", x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bat Fly")
    SetCollider(gameObject=Self, active=false)
    SetHP(target=Self, hp=0)
    on FINISHED → Face Middle
  }

  state Return {
    FaceObject(objectA=var "Self", objectB=var "Grimm", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Grimm Pos", time=0, delay=0, speed=25, space=0, easeType=easeInSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → End
  }

  state End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bat End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Inactive
  }

  state Inactive {
    SetMeshRenderer(gameObject=Self, active=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=-50, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Dormant
  }

  state Get To Middle {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=(87.2, 12.6, 0.00510025), time=(unset), delay=0, speed=20, space=0, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Choose
  }

  state Face Middle {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Pos", float2=87.2, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Get To Middle
  }

  var Grimm: gameObject = Grimm Scene/Grimm Boss

  var Max Speed X: float = 0
  var Max Speed Y: float = 0
  var Velocity: float = 0
  var X Accel: float = 0
  var X Pos: float = 0
  var Y Accel: float = 0
  var Y Pos: float = 0
  var Z Pos: float = 0
  var Grimm Pos: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
}
