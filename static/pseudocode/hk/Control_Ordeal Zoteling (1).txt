# Control
# 2 location(s):
#   GG_Mighty_Zote / Battle Control/Zotelings/Ordeal Zoteling (1)
#   GG_Mighty_Zote / Battle Control/Zotelings/Ordeal Zoteling

fsm Control {
  start Pause
  on ZERO HP → Die  // from any state
  on ZOTE KILLED → Die  // from any state
  on SPELL DAMAGE → Die  // from any state

  state Init {
    SetDamageHeroAmount(target=Self, damageDealt=var "Damage")
    FindChild(gameObject=Self, childName="Hitbox Buzzer", storeResult=var "Hitbox Buzzer")
    FindChild(gameObject=Self, childName="Hitbox Hopper", storeResult=var "Hitbox Hopper")
    FindChild(gameObject=Self, childName="Burst", storeResult=var "Burst")
    GetOwner(storeGameObject=var "Self")
    GetHP(target=Self, storeValue=var "Max HP")
    on FINISHED → Dormant
  }

  state Reset {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetHP(target=Self, hp=var "Max HP")
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=60, y=-5, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetTag(gameObject=Self, tag="Boss")
    ActivateGameObject(gameObject=var "Hitbox Buzzer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hitbox Hopper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Respawn Pause
  }

  state Dormant {
    on SPAWN → Ball
  }

  state Ball {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RandomFloat(min=91.54, max=114.56, storeResult=var "X Pos")
    SetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=20.11, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    FlingObjectsFromGlobalPoolTime(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), frequency=0.02, spawnMin=1, spawnMax=1, speedMin=0, speedMax=1, angleMin=90, angleMax=90, originVariationX=0.5, originVariationY=0.5)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zoteling Spit")
    WaitRandom(timeMin=0.7, timeMax=0.7, finishEvent=→"FINISHED", realTime=false)
    SetTag(gameObject=Self, tag="Boss Attack")
    RandomFloat(min=0.0051, max=0.0059, storeResult=var "zPos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "zPos", space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=0.85, max=1.15, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    SetAudioClip(gameObject=Self, audioClip=mantis_heavy_flyer_projectile_fly_loop (sharedassets34.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetCollider(gameObject=Self, active=true)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0)
    on FINISHED → Choice
  }

  state Buzzer Start {
    DecelerateV2(gameObject=Self, deceleration=0.7)
    FlingObjectsFromGlobalPoolTime(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), frequency=0.3, spawnMin=1, spawnMax=1, speedMin=0, speedMax=1, angleMin=90, angleMax=90, originVariationX=0.5, originVariationY=0.5)
    SetAudioClip(gameObject=Self, audioClip=Zoteling_loop (sharedassets399.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Buzzer Idle")
    WaitRandom(timeMin=0.1, timeMax=0.1, finishEvent=→"FINISHED", realTime=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    ActivateGameObject(gameObject=var "Hitbox Buzzer", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FaceDirection(gameObject=var "Self", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Buzzer TurnToIdle", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    IdleBuzzV3(gameObject=Self, waitMin=0.5, waitMax=2, speedMax=10, accelerationMin=5, accelerationMax=10, roamingRangeX=5, roamingRangeY=5, manualStartPos=(unset))
    SetCollider(gameObject=Self, active=true)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    on FINISHED → Buzzer Chase
  }

  state Buzzer Chase {
    FaceDirection(gameObject=var "Self", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Buzzer TurnToIdle", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    ChaseObject(gameObject=var "Self", target=var "Hero", speedMax=9, acceleration=0.23, targetSpread=2, spreadResetTimeMin=5, spreadResetTimeMax=5)
  }

  state Die {
    FlingObjectsFromGlobalPool(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=15, speedMin=15, speedMax=20, angleMin=0, angleMax=360, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    SendEventToRegister(eventName="ZOTE WARRIOR KILLED")
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_death_sword (resources.assets), pitchMin=0.75, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_damage (resources.assets), pitchMin=0.75, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    on FINISHED → Reset
  }

  state Hopper Jump {
    SendRandomEvent(events=[→"TOWARD", (none)], weights=[0.5, 0.5], delay=0)
    RandomFloat(min=-10, max=10, storeResult=var "Jump X")
    RandomFloat(min=25, max=27, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hopper Jump")
    NextFrameEvent(sendEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zoteling_yelp_01 (sharedassets399.assets), Zoteling_yelp_04 (sharedassets399.assets)], weights=[1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Extra Frame
    on TOWARD → Dir
  }

  state Hopper Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hopper Jump")
    SetGravity2dScale(gameObject=Self, gravityScale=0.5)
    ActivateGameObject(gameObject=var "Hitbox Hopper", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    on FINISHED → Extra Frame
  }

  state Hopper Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Hopper Land
  }

  state Hopper Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hopper Land")
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zoteling_yelp_02 (sharedassets399.assets), Zoteling_yelp_03 (sharedassets399.assets)], weights=[1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Hopper Antic
  }

  state Hopper Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hopper Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hopper Jump
  }

  state Choice {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mini_mawlek_spit (sharedassets34.assets), pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FlingObjectsFromGlobalPool(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=8, spawnMax=10, speedMin=10, speedMax=15, angleMin=0, angleMax=360, originVariationX=0.5, originVariationY=0.5, FSM="", FSMEvent="")
    AudioStop(gameObject=Self)
    SendRandomEvent(events=[→"BUZZER", →"HOPPER"], weights=[1, 1], delay=0)
    on BUZZER → Buzzer Start
    on HOPPER → Hopper Start
  }

  state Pause {
    on FINISHED → Init
  }

  state Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"L", delay=0, everyFrame=false)
    on L → L
    on R → R
  }

  state Toward Jump {
    RandomFloat(min=25, max=27, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hopper Jump")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Extra Frame
  }

  state L {
    RandomFloat(min=-15, max=-5, storeResult=var "Jump X")
    on FINISHED → Toward Jump
  }

  state R {
    RandomFloat(min=5, max=15, storeResult=var "Jump X")
    on FINISHED → Toward Jump
  }

  state Respawn Pause {
    WaitRandom(timeMin=3, timeMax=4, finishEvent=→"SPAWN", realTime=false)
    on SPAWN → Ball
  }

  state Extra Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Hopper Air
  }

  var Jump X: float = 0
  var Jump Y: float = 0
  var Pitch: float = 0
  var X Pos: float = 0
  var zPos: float = 0
  var Damage: int = 1
  var Level: int = 0
  var Max HP: int = 0
  var Is Buzzer: bool = false
  var Burst: gameObject = <null>
  var Hitbox Buzzer: gameObject = <null>
  var Hitbox Hopper: gameObject = <null>
  var Self: gameObject = <null>
}
