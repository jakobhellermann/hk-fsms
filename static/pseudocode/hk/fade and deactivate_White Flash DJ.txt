# fade and deactivate
# 2 location(s):
#   Knight_Pickup / Knight/Effects/White Flash DJ
#   resources.assets / Knight:1/Effects/White Flash DJ

fsm fade and deactivate {
  start Pause

  state Idle {
    EaseColor(fromValue=(1, 1, 1, 0.603), toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=0.4, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RecycleSelf()  // disabled
  }

  state Pause {
    SetSpriteRenderer(gameObject=Self, active=false)
    Wait(time=0.125, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Colour: color = (1, 1, 1, 0.5176471)
}
