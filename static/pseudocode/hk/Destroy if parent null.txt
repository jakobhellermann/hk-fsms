  start Check

  state Check {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    GameObjectIsNull(gameObject=var "Parent", isNull=→"DESTROY", isNotNull=(none), storeResult=(unset), everyFrame=true)
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Parent: gameObject = <null>
  var Self: gameObject = <null>
