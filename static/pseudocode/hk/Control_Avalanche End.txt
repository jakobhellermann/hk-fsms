# Control
# 1 location(s):
#   Room_Wyrm / Avalanche End

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on SHINY PICKED UP → Ready
  }

  state Ready {
    SetCollider(gameObject=Self, active=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Take Control
  }

  state Take Control {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="SLOPE CANCEL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=true, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", setValue=true, everyFrame=false)
    on FINISHED → Hero Fall
  }

  state Change Scene {
    SendEventByName(eventTarget=GameObject(var "MainCamera"), sendEvent="JUST FADE", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="blizzardEnded", value=true)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=MaxHealth())
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade In", setValue=true, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=EnterWithoutInput(true))
    SetMeshRenderer(gameObject=var "Hero", active=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=["Deepnest_East_12", "door_cutsceneReturn", 0], storeResult=(unset var))
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=false, everyFrame=false)
  }

  state Hero Fall {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=blizzard_start (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=breakable_wall_death (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=cave_in_rock_hit_2 (sharedassets173.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=var "Hero", frame=9)
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=-20, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=7.65, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Dreamer Land")
    on FINISHED → Fade Pause
  }

  state Fade {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "MainCamera"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Change Scene
  }

  state Fade Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade
  }

  var Hero Y: float = 0
  var Self: gameObject = <null>
}
