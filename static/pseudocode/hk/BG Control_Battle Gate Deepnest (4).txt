# BG Control
# 1 location(s):
#   Deepnest_32 / Battle Gate Deepnest (4)

fsm BG Control {
  start Opened
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="BG Opened")
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → Close 2
    on BG QUICK CLOSE → Quick Close
  }

  state Close 2 {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="BG Close")
    SetCollider(gameObject=Self, active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mimic_spider_transormation_stage_4 (sharedassets290.assets), pitchMin=0.8, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on BG OPEN → Open
  }

  state Open {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="BG Open")
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mimic_spider_transormation_stage_4 (sharedassets290.assets), pitchMin=0.8, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on BG CLOSE → 
    on BG QUICK CLOSE → Quick Close
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="BG Closed")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on BG OPEN → Open
    on FINISHED → Double Close
  }

  state Double Close {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="BG Closed")
    on BG OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="BG Opened")
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → 
    on BG QUICK CLOSE → Quick Close
  }

  var Start Closed: bool = false

  var Self: gameObject = <null>
  var Sprite: gameObject = Battle Gate Deepnest (4)/Sprite
}
