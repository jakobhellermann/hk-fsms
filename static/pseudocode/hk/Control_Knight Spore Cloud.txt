# Control
# 1 location(s):
#   resources.assets / Knight Spore Cloud

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=true)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_34", isTrue=→"DEEP", isFalse=→"NORMAL")
    on NORMAL → Normal
    on DEEP → Deep
  }

  state Wait {
    Wait(time=4.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    RecycleSelf()
  }

  state Normal {
    ActivateGameObject(gameObject=var "Pt Deep", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Wait
  }

  state Deep {
    ActivateGameObject(gameObject=var "Pt Normal", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1.35, y=1.35, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Wait
  }

  var Pt Deep: gameObject = Knight Spore Cloud/Pt Deep
  var Pt Normal: gameObject = Knight Spore Cloud/Pt Normal
  var Self: gameObject = <null>
}
