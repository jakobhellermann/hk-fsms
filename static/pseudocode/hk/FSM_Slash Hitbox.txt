// uses template: nail_clash_tink
fsm FSM {
  start Initiate

  state Detecting {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Tinker(16), sendEvent=→"TAKE DAMAGE", storeCollider=var "Collider")
    on TAKE DAMAGE → Blocked Hit
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Detecting
  }

  state Blocked Hit {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(3))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=NailParry())
    GetParent(gameObject=var "Collider", storeResult=var "Slash")
    GetFsmFloat(gameObject=var "Slash", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClips=[hero_parry], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on RIGHT → No Box Right
    on LEFT → No Box Left
    on UP → No Box Up
    on DOWN → No Box Down
  }

  state No Box Down {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="Bounce", parameters=[], storeResult=(unset var))
    SpawnObjectFromGlobalPool(gameObject=Nail Clash Effect, spawnPoint=var "Hero", position=(0, -1.5, 0), rotation=(unset), storeObject=var "")
    on FINISHED → Pause Frame
  }

  state No Box Up {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RecoilDown", parameters=[], storeResult=(unset var))
    SpawnObjectFromGlobalPool(gameObject=Nail Clash Effect, spawnPoint=var "Hero", position=(0, 1.5, 0), rotation=(unset), storeObject=var "")
    on FINISHED → Pause Frame
  }

  state No Box Left {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RecoilRight", parameters=[], storeResult=(unset var))
    SpawnObjectFromGlobalPool(gameObject=Nail Clash Effect, spawnPoint=var "Hero", position=(-1.5, 0, 0), rotation=(unset), storeObject=var "")
    on FINISHED → Pause Frame
  }

  state No Box Right {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RecoilLeft", parameters=[], storeResult=(unset var))
    SpawnObjectFromGlobalPool(gameObject=Nail Clash Effect, spawnPoint=var "Hero", position=(1.5, 0, 0), rotation=(unset), storeObject=var "")
    on FINISHED → Pause Frame
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Detecting
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → NailParryRecover
  }

  state NailParryRecover {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=NailParryRecover())
    NextFrameEvent(sendEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="PARRIED", delay=0, everyFrame=false)
    on FINISHED → Pause
  }

  var Attack Direction: float = 0
  var Chooser: float = 0
  var Camera: gameObject = <null>
  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Slash: gameObject = <null>
}
