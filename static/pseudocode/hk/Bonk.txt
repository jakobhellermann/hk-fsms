# Bonk
# 1 location(s):
#   sharedassets32.assets / Zote Boss/Bonker

fsm Bonk {
  start Init

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Direction
  }

  state Direction {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[Zote_Tonk, Zote_Tonk_big], weights=[1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Block Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on L → Hit L
    on R → Hit R
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Block Hit", storeResult=var "Block Hit")
    on FINISHED → Idle
  }

  state Hit L {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilLeft())
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=10, y=(unset), everyFrame=false)
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit R {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilRight())
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=-10, y=(unset), everyFrame=false)
    on FINISHED → Pause
  }

  var Block Hit: gameObject = <null>
  var Parent: gameObject = <null>
}
