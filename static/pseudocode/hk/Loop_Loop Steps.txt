# Loop
# 2 location(s):
#   Dream_Mighty_Zote / Grey Prince/Loop Steps
#   GG_Grey_Prince_Zote / Grey Prince/Loop Steps

fsm Loop {
  start Init

  state Idle {
    AudioStop(gameObject=Self)
    on CHARGE START → Play
    on RUN START → Play
  }

  state Play {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on CHARGE END → Idle
    on RUN END → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
}
