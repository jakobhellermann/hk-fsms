fsm Control {
  start Init

  state Chase {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Bee TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.4)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=10, accelerationForce=6.5, offsetX=0, offsetY=0)
    SetIsDead(target=Self, setValue=false)
    on CENTIPEDE DEATH → Death 3
    on CHECK → Check
  }

  state Inert {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    on SPAWN → Set Collider
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Cage")
    FindChild(gameObject=Self, childName="Pt Death", storeResult=var "Pt Death")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Inert
  }

  state Death {
    SetHP(target=Self, hp=5)
    SetParent(gameObject=Self, parent=var "Cage", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    SetIsDead(target=Self, setValue=false)
    GetFsmInt(gameObject=var "Owner", fsmName="Zombie Hive", variableName="Spawned", storeValue=var "Spawned", everyFrame=false)
    IntAdd(intVariable=var "Spawned", add=-1, everyFrame=false)
    SetFsmInt(gameObject=var "Owner", fsmName="Zombie Hive", variableName="Spawned", setValue=var "Spawned", everyFrame=false)
    on FINISHED → Inert
  }

  state Check {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=55, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CENTIPEDE DEATH", everyFrame=false)
    on FINISHED → Chase
    on CENTIPEDE DEATH → Death 3
  }

  state Set Collider {
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Chase
  }

  state Death 3 {
    PlayParticleEmitter(gameObject=var "Pt Death", emit=0)
    SetVelocity(gameObject=Self, vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_death_sword (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Death 2
  }

  state Death 2 {
    Wait(time=1.2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Death
  }

  var Distance: float = 0
  var Spawned: int = 0
  var Cage: gameObject = <null>
  var Owner: gameObject = <null>
  var Pt Death: gameObject = <null>
  var Self: gameObject = <null>
}
