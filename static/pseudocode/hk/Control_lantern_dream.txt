# Control
# 1 location(s):
#   Cliffs_06 / Nightmare Lantern/lantern_dream

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Lantern Lit Objects", storeResult=var "Lantern Lit Objects")
    FindChild(gameObject=Self, childName="Lantern First Light", storeResult=var "Lantern First Light")
    on FINISHED → Idle
  }

  state Idle {
    on LANTERN LIT → Light
  }

  state Light {
    ActivateGameObject(gameObject=var "Lantern Lit Objects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lantern First Light", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Lantern First Light: gameObject = <null>
  var Lantern Lit Objects: gameObject = <null>
}
