# Menu Control
# 2 location(s):
#   Fungus3_35 / Bank Menu/Deposit Menu
#   sharedassets218.assets / Bank Menu/Deposit Menu

fsm Menu Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Menu 1", storeResult=var "Menu 1")
    FindChild(gameObject=var "Parent", childName="Txt Balance Amount", storeResult=var "Balance Txt")
    FindChild(gameObject=Self, childName="Arrow D", storeResult=var "Arrow D")
    FindChild(gameObject=Self, childName="Arrow U", storeResult=var "Arrow U")
    FindChild(gameObject=Self, childName="Txt Deposit Amount", storeResult=var "Deposit Txt")
    FindChild(gameObject=Self, childName="Max Up 1", storeResult=var "Max Up 1")
    FindChild(gameObject=Self, childName="Max Up 2", storeResult=var "Max Up 2")
    GetLanguageString(sheetName="Prices", convName="BANK_TRANSACTION_AMOUNT", storeValue=var "Tran String")
    ConvertStringToInt(stringVariable=var "Tran String", intVariable=var "Transaction Amount", everyFrame=false)
    SetIntValue(intVariable=var "Transaction Amount Rpt", intValue=var "Transaction Amount", everyFrame=false)
    SetIntValue(intVariable=var "Transaction Amount Rpt2", intValue=var "Transaction Amount", everyFrame=false)
    IntOperator(integer1=var "Transaction Amount Rpt", integer2=5, operation=Multiply, storeResult=var "Transaction Amount Rpt", everyFrame=false)
    IntOperator(integer1=var "Transaction Amount Rpt2", integer2=10, operation=Multiply, storeResult=var "Transaction Amount Rpt2", everyFrame=false)
    SetIntValue(intVariable=var "Deposit Amount", intValue=0, everyFrame=false)
    ConvertIntToString(intVariable=var "Deposit Amount", stringVariable=var "Deposit String", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Deposit Txt", textString=var "Deposit String")
    ActivateGameObject(gameObject=var "Max Up 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Max Up 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetLanguageString(sheetName="Prices", convName="BANK_LIMIT", storeValue=var "Balance Limit String")
    ConvertStringToInt(stringVariable=var "Balance Limit String", intVariable=var "Balance Limit", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerBalance", storeValue=var "Balance")
    on FINISHED → Pause
  }

  state Menu Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  state Idle {
    on UI UP → Repeats?
    on UI DOWN → Repeats? D
    on UI CONFIRM → Confirm
    on UI CANCEL → Cancel amount
    on UI RS UP → Up Pressed Rpt 2
    on UI RS DOWN → Down Pressed 3
  }

  state Up Pressed {
    SendEventByName(eventTarget=GameObject(var "Arrow U"), sendEvent="MOVE", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Prev Deposit", intValue=var "Deposit Amount", everyFrame=false)
    IntAdd(intVariable=var "Deposit Amount", add=var "Transaction Amount", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="geo", storeValue=var "Player Geo")
    IntCompare(integer1=var "Deposit Amount", integer2=var "Player Geo", equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetIntValue(intVariable=var "Deposit Amount", intValue=var "Prev Deposit", everyFrame=false)
    on FINISHED → Check Balance Limit
  }

  state Down Pressed {
    SendEventByName(eventTarget=GameObject(var "Arrow D"), sendEvent="MOVE", delay=0, everyFrame=false)
    IntOperator(integer1=var "Deposit Amount", integer2=var "Transaction Amount", operation=Subtract, storeResult=var "Deposit Amount", everyFrame=false)
    on FINISHED → Stop at 0
  }

  state Refresh Amount {
    ConvertIntToString(intVariable=var "Deposit Amount", stringVariable=var "Deposit String", format="", everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=ui_option_click (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetTextMeshProText(gameObject=var "Deposit Txt", textString=var "Deposit String")
    on FINISHED → Idle
  }

  state Repeats? {
    GetFsmInt(gameObject=Self, fsmName="ui_list_getinput", variableName="Repeats", storeValue=var "Repeats", everyFrame=false)
    IntCompare(integer1=var "Repeats", integer2=5, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Repeats", integer2=10, equal=→"REPEAT", lessThan=→"REPEAT", greaterThan=→"REPEAT 2", everyFrame=false)
    on FINISHED → Up Pressed
    on REPEAT → Up Pressed Rpt
    on REPEAT 2 → Up Pressed Rpt 2
  }

  state Up Pressed Rpt {
    SendEventByName(eventTarget=GameObject(var "Arrow U"), sendEvent="MOVE", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Prev Deposit", intValue=var "Deposit Amount", everyFrame=false)
    IntAdd(intVariable=var "Deposit Amount", add=var "Transaction Amount Rpt", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="geo", storeValue=var "Player Geo")
    IntCompare(integer1=var "Deposit Amount", integer2=var "Player Geo", equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetIntValue(intVariable=var "Deposit Amount", intValue=var "Prev Deposit", everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"CANCEL", delay=0, everyFrame=false)
    on FINISHED → Check Balance Limit
    on CANCEL → Up Pressed
  }

  state Down Pressed 2 {
    SendEventByName(eventTarget=GameObject(var "Arrow D"), sendEvent="MOVE", delay=0, everyFrame=false)
    IntOperator(integer1=var "Deposit Amount", integer2=var "Transaction Amount Rpt", operation=Subtract, storeResult=var "Deposit Amount", everyFrame=false)
    on FINISHED → Stop at 0
  }

  state Repeats? D {
    GetFsmInt(gameObject=Self, fsmName="ui_list_getinput", variableName="Repeats", storeValue=var "Repeats", everyFrame=false)
    IntCompare(integer1=var "Repeats", integer2=5, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Repeats", integer2=10, equal=→"REPEAT", lessThan=→"REPEAT", greaterThan=→"REPEAT 2", everyFrame=false)
    on FINISHED → Down Pressed
    on REPEAT → Down Pressed 2
    on REPEAT 2 → Down Pressed 3
  }

  state Stop at 0 {
    IntCompare(integer1=var "Deposit Amount", integer2=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetIntValue(intVariable=var "Deposit Amount", intValue=0, everyFrame=false)
    on FINISHED → Refresh Amount
  }

  state Up Pressed Rpt 2 {
    SendEventByName(eventTarget=GameObject(var "Arrow U"), sendEvent="MOVE", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Prev Deposit", intValue=var "Deposit Amount", everyFrame=false)
    IntAdd(intVariable=var "Deposit Amount", add=var "Transaction Amount Rpt2", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="geo", storeValue=var "Player Geo")
    IntCompare(integer1=var "Deposit Amount", integer2=var "Player Geo", equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetIntValue(intVariable=var "Deposit Amount", intValue=var "Prev Deposit", everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"CANCEL", delay=0, everyFrame=false)
    on FINISHED → Check Balance Limit
    on CANCEL → Up Pressed Rpt
  }

  state Down Pressed 3 {
    SendEventByName(eventTarget=GameObject(var "Arrow D"), sendEvent="MOVE", delay=0, everyFrame=false)
    IntOperator(integer1=var "Deposit Amount", integer2=var "Transaction Amount Rpt2", operation=Subtract, storeResult=var "Deposit Amount", everyFrame=false)
    on FINISHED → Stop at 0
  }

  state Confirm {
    ActivateGameObject(gameObject=var "Max Up 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=ui_button_confirm (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Max Up 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Not Zero?
  }

  state Cancel amount {
    SetIntValue(intVariable=var "Deposit Amount", intValue=0, everyFrame=false)
    ConvertIntToString(intVariable=var "Deposit Amount", stringVariable=var "Deposit String", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Deposit Txt", textString=var "Deposit String")
    on FINISHED → Confirm
  }

  state Not Zero? {
    IntCompare(integer1=var "Deposit Amount", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    on END → Menu Down
    on FINISHED → Take Geo
  }

  state Menu Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Menu 1 Up
  }

  state Menu 1 Up {
    ActivateGameObject(gameObject=var "Menu 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    ConvertIntToString(intVariable=var "Deposit Amount", stringVariable=var "Deposit String", format="", everyFrame=false)
    on FINISHED → Menu Up
  }

  state Take Geo {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=TakeGeo(var "Deposit Amount"))
    on FINISHED → Confirm Down
  }

  state Confirm Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Increment Balance
  }

  state Increment Balance {
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerBalance", storeValue=var "Balance")
    SetIntValue(intVariable=var "Balance Display", intValue=var "Balance", everyFrame=false)
    IntAdd(intVariable=var "Balance", add=var "Deposit Amount", everyFrame=false)
    SetPlayerDataInt(intName="bankerBalance", value=var "Balance")
    on FINISHED → Set Roller
  }

  state Set Roller {
    ConvertIntToFloat(intVariable=var "Deposit Amount", floatVariable=var "Roller Float", everyFrame=false)
    FloatMultiply(floatVariable=var "Roller Float", multiplyBy=0.025, everyFrame=false)
    FloatMultiply(floatVariable=var "Roller Float", multiplyBy=1.75, everyFrame=false)
    ConvertFloatToInt(floatVariable=var "Roller Float", intVariable=var "Roller Int", rounding=RoundUp, everyFrame=false)
    on FINISHED → Roll Time
  }

  state Roll Time {
    Wait(time=0.025, finishEvent=→"FINISHED", realTime=false)
    IntCompare(integer1=var "Balance Display", integer2=var "Balance", equal=→"END", lessThan=(none), greaterThan=→"END", everyFrame=false)
    on FINISHED → Roll Down
    on END → Set Correct
  }

  state Roll Down {
    IntAdd(intVariable=var "Balance Display", add=var "Roller Int", everyFrame=false)
    ConvertIntToString(intVariable=var "Balance Display", stringVariable=var "Balance String", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Balance Txt", textString=var "Balance String")
    on FINISHED → Roll Time
  }

  state Set Correct {
    SetIntValue(intVariable=var "Balance Display", intValue=var "Balance", everyFrame=false)
    ConvertIntToString(intVariable=var "Balance Display", stringVariable=var "Balance String", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Balance Txt", textString=var "Balance String")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Menu 1 Up
  }

  state Check Balance Limit {
    SetIntValue(intVariable=var "New Balance", intValue=var "Balance", everyFrame=false)
    IntAdd(intVariable=var "New Balance", add=var "Deposit Amount", everyFrame=false)
    IntCompare(integer1=var "New Balance", integer2=var "Balance Limit", equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    IntOperator(integer1=var "New Balance", integer2=var "Balance Limit", operation=Subtract, storeResult=var "Balance Over", everyFrame=false)
    IntOperator(integer1=var "Deposit Amount", integer2=var "Balance Over", operation=Subtract, storeResult=var "Deposit Amount", everyFrame=false)
    on FINISHED → Refresh Amount
  }

  var Roller Float: float = 0
  var Balance: int = 0
  var Balance Display: int = 0
  var Balance Limit: int = 0
  var Balance Over: int = 0
  var Deposit Amount: int = 0
  var New Balance: int = 0
  var Player Geo: int = 0
  var Prev Deposit: int = 0
  var Repeats: int = 0
  var Roller Int: int = 0
  var Transaction Amount: int = 0
  var Transaction Amount Rpt: int = 0
  var Transaction Amount Rpt2: int = 0
  var Balance Limit String: string = ""
  var Balance String: string = ""
  var Deposit String: string = ""
  var Tran String: string = ""
  var Arrow D: gameObject = <null>
  var Arrow U: gameObject = <null>
  var Balance Txt: gameObject = <null>
  var Deposit Txt: gameObject = <null>
  var Max Up 1: gameObject = <null>
  var Max Up 2: gameObject = <null>
  var Menu 1: gameObject = <null>
  var Parent: gameObject = <null>
}
