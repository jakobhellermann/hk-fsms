# FSM
# 1 location(s):
#   RestingGrounds_08 / Ghost Battle Revek/A Dash Effect

// uses template: deactivate_after_anim
fsm FSM {
  start Init

  state Wait {
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dWatchAnimationEvents(gameObject=var "Self", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reparent?
  }

  state Destroy {
    ActivateGameObject(gameObject=var "Self", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Reposition? {
    SetMeshRenderer(gameObject=var "Self", active=true)
    BoolTest(boolVariable=var "Reposition", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetPosition(gameObject=var "Self", vector=var "Init Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Set Rotation?
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on FINISHED → Pause?
  }

  state Unparent? {
    BoolTest(boolVariable=var "Unparent", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    GetScale(gameObject=var "Self", vector=(unset), xScale=var "X Scale", yScale=var "Y Scale", zScale=(unset), space=1, everyFrame=false)
    GetScale(gameObject=var "Self", vector=(unset), xScale=var "World X Scale", yScale=var "World Y Scale", zScale=(unset), space=0, everyFrame=false)  // disabled
    SetParent(gameObject=var "Self", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetScale(gameObject=var "Self", vector=(unset), x=var "World X Scale", y=var "World Y Scale", z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    BoolTest(boolVariable=var "Reset Scale", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)  // disabled
    SetScale(gameObject=Self, vector=var "Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    on FINISHED → Wait
  }

  state Reparent? {
    BoolTest(boolVariable=var "Unparent", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetParent(gameObject=var "Self", parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SetScale(gameObject=var "Self", vector=(unset), x=var "X Scale", y=var "Y Scale", z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Destroy
  }

  state Pause? {
    FloatCompare(float1=var "Pause", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetMeshRenderer(gameObject=var "Self", active=false)
    Wait(time=var "Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reposition?
  }

  state Set Rotation? {
    BoolTest(boolVariable=var "Reset Rotation", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Unparent?
  }

  var Pause: float = 0
  var Rotation: float = 0
  var Reposition: bool = true
  var Reset Rotation: bool = true
  var Reset Scale: bool = true
  var Unparent: bool = true
  var Init Pos: vector3 = (3.146759, 0.5900002, -0.0015)
  var Scale: vector3 = (1, 1, 1)

  var World X Scale: float = 0
  var World Y Scale: float = 0
  var X Scale: float = 0
  var Y Scale: float = 0
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
