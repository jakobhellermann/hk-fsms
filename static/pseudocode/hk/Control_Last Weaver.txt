# Control
# 1 location(s):
#   Deepnest_45_v02 / Last Weaver

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Weaver_Escape", storeResult=var "Weaver_Escape")
    FindChild(gameObject=Self, childName="Weaver_Idle", storeResult=var "Weaver_Idle")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activate
    on APPEAR → Appear
  }

  state Activate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Appear {
    ActivateGameObject(gameObject=var "Weaver_Idle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"LEAVE", storeCollider=var "")
    on LEAVE → Leave
  }

  state Leave {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Weaver_Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Weaver_Escape", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Weaver_Escape: gameObject = <null>
  var Weaver_Idle: gameObject = <null>
}
