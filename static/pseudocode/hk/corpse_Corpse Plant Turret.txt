# corpse
# 1 location(s):
#   sharedassets140.assets / Corpse Plant Turret

fsm corpse {
  start Wait

  state Fart {
    PlayParticleEmitter(gameObject=var "Spurt", emit=0)
    WaitRandom(timeMin=1.5, timeMax=2, finishEvent=→"DONE", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=fungus_balloon_death_3, pitchMin=1, pitchMax=1, volume=0.8, delay=0, storePlayer=var "Audio Player")
    on DONE → Done
    on ACID → Splash
  }

  state Wait {
    AddTorque2d(gameObject=Self, forceMode=0, torque=400, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Fart
    on ACID → Splash
  }

  state Done {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death")
    StopParticleEmitter(gameObject=var "Spurt")
    PlayParticleEmitter(gameObject=var "Start", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=fungus_balloon_death_final_pop, pitchMin=1, pitchMax=1, volume=0.8, delay=0, storePlayer=<null>)
    AudioStop(gameObject=var "Audio Player")
    on ACID → Splash
  }

  state Acid {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Start", audioClip=acid_splash (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Corpse Acid Steam (resources.assets), spawnPoint=var "Start", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetGravity2dScale(gameObject=Self, gravityScale=0.05)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")  // disabled
    DecelerateV2(gameObject=var "Self", deceleration=0.65)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Acid Spore", emit=0)
    PlayParticleEmitter(gameObject=var "Acid Bubble", emit=0)
    StopParticleEmitter(gameObject=var "Spurt")
    StopParticleEmitter(gameObject=var "Start")
    on FINISHED → Sizzle
  }

  state Sizzle {
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    DecelerateV2(gameObject=var "Self", deceleration=0.65)
    ColorInterpolate(colors=[(1, 1, 1, 1), (1, 1, 1, 0)], time=1, storeColor=var "Colour", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  state Finish {
    StopParticleEmitter(gameObject=var "Acid Spore")
    StopParticleEmitter(gameObject=var "Acid Bubble")
  }

  state Splash {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopBounce())
    AudioStop(gameObject=var "Audio Player")
    GetOwner(storeGameObject=var "Self")
    SetCollider(gameObject=var "Start", active=false)
    BoxColliderOffset(gameObject1=var "Self", offsetVector2=(unset), offsetX=(unset), offsetY=var "Collider Offset Y", everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Splash Spawn Pos", addX=0, addY=var "Collider Offset Y", addZ=0, everyFrame=false, perSecond=false)
    BoundsBoxCollider(gameObject1=var "Self", scaleVector2=(unset), scaleX=(unset), scaleY=var "Collider Half Height", everyFrame=false)
    FloatMultiply(floatVariable=var "Collider Half Height", multiplyBy=0.5, everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Splash Spawn Pos", addX=0, addY=var "Collider Half Height", addZ=-0.1, everyFrame=false, perSecond=false)
    CreateObject(gameObject=Acid Splash (resources.assets), spawnPoint=var "Self", position=var "Splash Spawn Pos", rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Acid
  }

  var Angle: float = 0
  var Chooser: float = 0
  var Collider Half Height: float = 0
  var Collider Offset Y: float = 0
  var Counter: float = 0
  var Speed: float = 0
  var Splash Spawn Pos: vector3 = (0, 0, 0)
  var Colour: color = (1, 1, 1, 1)
  var Acid Bubble: gameObject = Corpse Plant Turret/Bub Cloud
  var Acid Spore: gameObject = Corpse Plant Turret/Spore Cloud
  var Audio Player: gameObject = <null>
  var Corpse Flame: gameObject = <null>
  var Corpse Steam: gameObject = Corpse Zombie Fung A/Corpse Steam (sharedassets138.assets)
  var Initial: gameObject = Corpse Plant Turret/Initial Steam
  var Self: gameObject = <null>
  var Spurt: gameObject = Corpse Plant Turret/Corpse Steam
  var Start: gameObject = Corpse Plant Turret/Spurt Start
}
