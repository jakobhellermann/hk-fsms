# Control
# 1 location(s):
#   Room_Colosseum_02 / Shade Present Particles

fsm Control {
  start Idle

  state Idle {
    on SHADE PRESENT → Play
  }

  state Play {
    PlayParticleEmitter(gameObject=Self, emit=0)
    on SHADE KILLED → Stop
  }

  state Stop {
    StopParticleEmitter(gameObject=Self)
  }
}
