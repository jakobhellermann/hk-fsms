fsm Control {
  start Init

  state Chase {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.4)
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=5, acceleration=0.1, targetSpread=1.5, spreadResetTimeMin=1, spreadResetTimeMax=2)
    on CENTIPEDE DEATH → Death
  }

  state Inert {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on SPAWN → Chase
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Cage")
    GetHero(storeResult=var "Hero")
    SetHealthManagerReset(target=Self, reset=true)
    on FINISHED → Inert
  }

  state Death {
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=4, speedMin=5, speedMax=10, angleMin=0, angleMax=360, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetHP(target=Self, hp=5)
    SetIsDead(target=Self, setValue=false)
    SetParent(gameObject=Self, parent=var "Cage", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Inert
  }

  var Angle: float = 0
  var Speed: float = 0
  var Cage: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Shot: gameObject = <null>
