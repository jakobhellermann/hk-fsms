# Spell Destroy
# 2 location(s):
#   Crossroads_ShamanTemple / _Props/Plank Solid Bunch 2
#   Crossroads_ShamanTemple / _Props/Plank Solid Bunch 1

fsm Spell Destroy {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"SPELL", storeCollider=var "")
    on SPELL → Destroy
  }

  state Destroy {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="INSTA BREAK", delay=0, everyFrame=false)
  }
}
