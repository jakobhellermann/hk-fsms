# Spawn
# 1 location(s):
#   sharedassets481.assets / Shockwave Wave Short

fsm Spawn {
  start Init
  on SPAWN END → Idle  // from any state

  state Idle {
    on SPAWN START → Spawn
  }

  state Spawn {
    SpawnObjectFromGlobalPool(gameObject=var "Shockwave Object", spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Spurt")
    SetScale(gameObject=var "Spurt", vector=(unset), x=var "Scale X", y=var "Scale X", z=(unset), everyFrame=false, lateUpdate=false)
    SetDamageHeroAmount(target=var "Spurt", damageDealt=var "Damage")
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.005, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Scale X: float = 0
  var Damage: int = 0
  var Self: gameObject = <null>
  var Shockwave Object: gameObject = <null>
  var Spurt: gameObject = <null>
}
