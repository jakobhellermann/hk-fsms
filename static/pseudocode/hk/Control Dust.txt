fsm Control Dust {
  start Init
  on BIG BUZZER KILLED → Destroy Self  // from any state

  state Init {
    SetParticleEmissionRate(gameObject=Self, emissionRate=0, everyFrame=false)
    on FINISHED → Out
  }

  state Out {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer="", sendEvent=→"ENTER", storeCollider=var "Buzzer")
    SetParticleEmissionRate(gameObject=Self, emissionRate=0, everyFrame=false)
    on ENTER → Swooping?
  }

  state In {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Boss", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SetParticleEmissionRate(gameObject=Self, emissionRate=250, everyFrame=false)
    on EXIT → Out
  }

  state Swooping? {
    GetFsmBool(gameObject=var "Buzzer", fsmName="Big Buzzer", variableName="Swooping", storeValue=var "Swooping", everyFrame=false)
    BoolTest(boolVariable=var "Swooping", isTrue=→"SWOOP", isFalse=→"NOT SWOOP", everyFrame=false)
    on SWOOP → In
    on NOT SWOOP → Out
  }

  state Destroy Self {
    DestroySelf(detachChildren=false)
  }

  var Swooping: bool = false
  var Buzzer: gameObject = <null>
}
