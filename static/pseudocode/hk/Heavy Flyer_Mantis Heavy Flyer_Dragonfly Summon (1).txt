# Heavy Flyer
# 2 location(s):
#   Fungus3_05 / Battle Scene/Wave 2/Dragonfly Summon (1)/Mantis Heavy Flyer
#   Fungus3_10 / Battle Scene/Wave 2/Dragonfly Summon (1)/Mantis Heavy Flyer

fsm Heavy Flyer {
  start Init
  on UNALERT → Idle  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    IdleBuzz(gameObject=Self, waitMin=0.25, waitMax=0.75, speedMax=5, accelerationMax=200, roamingRange=3)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    SetFloatValue(floatVariable=var "Range Out Timer", floatValue=0, everyFrame=false)
    BoolTest(boolVariable=var "startAlert", isTrue=→"ALERT", isFalse=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="GetCurrentMapZone", parameters=[], storeResult=var "Map Zone")
    SendEventByName(eventTarget=Self, sendEvent=var "Map Zone", delay=0, everyFrame=false)
    on ALERT → Alert
    on COLOSSEUM → Alert
  }

  state Alert {
    RandomFloat(min=6, max=8, storeResult=var "Distance")
    RandomFloat(min=4, max=5.5, storeResult=var "Height")
    DistanceFly(gameObject=Self, target=var "Hero", distance=var "Distance", speedMax=4.5, acceleration=0.5, targetsHeight=true, height=var "Height")  // disabled
    DistanceFlyV2(gameObject=Self, target=var "Hero", distance=var "Distance", speedMax=4.5, acceleration=0.5, targetsHeight=true, height=var "Height", stayLeft=var "Stay Left", stayRight=var "Stay Right")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    WaitRandom(timeMin=1.75, timeMax=2.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetBoolValue(boolVariable=var "Out of Range", boolValue=var "Shoot Range", everyFrame=false)
    BoolFlipEveryFrame(boolVariable=var "Out of Range", everyFrame=true)
    FloatAddV2(floatVariable=var "Range Out Timer", add=1, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=var "Out of Range")
    SetFloatValueV2(floatVariable=var "Range Out Timer", floatValue=0, everyFrame=true, activeBool=var "Shoot Range")
    FloatCompare(float1=var "Range Out Timer", float2=8, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=true)
    on FINISHED → Shoot Antic
  }

  state Shoot Antic {
    BoolTest(boolVariable=var "Shoot Range", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_flying_sentry_prepare (sharedassets34.assets))
    on FINISHED → Shoot
    on CANCEL → Alert
  }

  state Shoot {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_heavy_flyer_projectile_throw (sharedassets34.assets))
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Shot Mantis (sharedassets34.assets), spawnPoint=var "Shot Point", position=(unset), rotation=(unset), storeObject=var "Shot")
    GetAngleToTarget2D(gameObject=var "Shot Point", target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", everyFrame=false)
    SetVelocityAsAngle(gameObject=var "Shot", angle=var "Angle", speed=17, everyFrame=false)
    on FINISHED → Recover
  }

  state Recover {
    DistanceFly(gameObject=Self, target=var "Hero", distance=8, speedMax=5.5, acceleration=0.65, targetsHeight=true, height=5)  // disabled
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    DistanceFlyV2(gameObject=Self, target=var "Hero", distance=var "Distance", speedMax=4.5, acceleration=0.5, targetsHeight=true, height=var "Height", stayLeft=var "Stay Left", stayRight=var "Stay Right")
    SetBoolValue(boolVariable=var "Out of Range", boolValue=var "Shoot Range", everyFrame=false)
    BoolFlipEveryFrame(boolVariable=var "Out of Range", everyFrame=true)
    FloatAddV2(floatVariable=var "Range Out Timer", add=1, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=var "Out of Range")
    SetFloatValueV2(floatVariable=var "Range Out Timer", floatValue=0, everyFrame=true, activeBool=var "Shoot Range")
    FloatCompare(float1=var "Range Out Timer", float2=8, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=true)
    on FINISHED → Alert
  }

  state Shot Position {
    DistanceFly(gameObject=Self, target=var "Hero", distance=9.5, speedMax=5.5, acceleration=0.65, targetsHeight=true, height=5)  // disabled
    DistanceFlyV2(gameObject=Self, target=var "Hero", distance=var "Distance", speedMax=4.5, acceleration=0.5, targetsHeight=true, height=var "Height", stayLeft=var "Stay Left", stayRight=var "Stay Right")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shoot Antic
  }

  var Stay Left: bool = false
  var Stay Right: bool = false

  var Angle: float = 0
  var Distance: float = 0
  var Height: float = 0
  var Range Out Timer: float = 0
  var Alert Range: bool = false
  var Out of Range: bool = false
  var Shoot Range: bool = false
  var startAlert: bool = false
  var Map Zone: string = ""
  var Self: gameObject = <null>
  var Shot: gameObject = <null>
  var Shot Point: gameObject = Battle Scene/Wave 2/Dragonfly Summon (1)/Mantis Heavy Flyer/Shot Point
}
