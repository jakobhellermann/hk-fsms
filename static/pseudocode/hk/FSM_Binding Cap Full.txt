# FSM
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Binding Cap Full
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Binding Cap Full

fsm FSM {
  start Play Full
  on NOT FULL → Play Not Full  // from any state

  state Play Full {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Soul Cap Full")
  }

  state Play Not Full {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Soul Cap Down")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Disable
  }

  state Disable {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
