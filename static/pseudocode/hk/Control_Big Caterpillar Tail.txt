# Control
# 1 location(s):
#   Deepnest_East_04 / Big Caterpillar Tail

fsm Control {
  start Init

  state Idle {
    GetOwner(storeGameObject=var "Self")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Tail Idle")
    on NAIL HIT → Hit
  }

  state Hit {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Tail Hit")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetPlayerDataBool(boolName="bigCatHitTail", value=true)
    AudioStop(gameObject=var "Audio")
    AudioPlayRandom(gameObject=var "Audio", audioClips=[Caterpillar_tickle_01 (sharedassets309.assets), Caterpillar_tickle_02 (sharedassets309.assets), Caterpillar_tickle_03 (sharedassets309.assets), Caterpillar_tickle_04 (sharedassets309.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bounce_shroom (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Shake {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MANTIS FLOOR FINISH", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Tail Hit")
    on FINISHED → Idle
  }

  state Init {
    FindChild(gameObject=Self, childName="Audio", storeResult=var "Audio")
    on FINISHED → Idle
  }

  var Audio: gameObject = <null>
  var Self: gameObject = <null>
}
