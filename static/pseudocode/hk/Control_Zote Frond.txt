# Control
# 1 location(s):
#   Dream_Mighty_Zote / Audience/Zote Frond

fsm Control {
  start Idle
  on ZOTE EXPLODE → Explode  // from any state
  on ZOTE KILLED → Zote Killed  // from any state

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Frond Idle")
    on CROWD SHOCK → Shock
    on ZOTE STUNNED → Shock
  }

  state Shock {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Frond Shock")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Zote Killed {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Frond Shock")
  }

  state Explode {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zote Collapse")
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var X Scale: float = 0
}
