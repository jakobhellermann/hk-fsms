# Tele Effects
# 3 location(s):
#   sharedassets192.assets / Flamebearer Large/Teleport Effects
#   sharedassets6.assets / Flamebearer Small/Teleport Effects
#   sharedassets6.assets / Flamebearer Med/Teleport Effects

fsm Tele Effects {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Flame Pillar", storeResult=var "Flame Pillar")
    FindChild(gameObject=Self, childName="Pt Tele", storeResult=var "Pt Tele")
    on FINISHED → Idle
  }

  state Idle {
    on TELEPORT → Effects
  }

  state Effects {
    ActivateGameObject(gameObject=var "Flame Pillar", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Tele", emit=0)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    on FINISHED → Idle
  }

  state Pillar {
    ActivateGameObject(gameObject=var "Flame Pillar", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Flame Pillar: gameObject = <null>
  var Pt Tele: gameObject = <null>
}
