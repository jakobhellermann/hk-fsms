# Prompt
# 5 location(s):
#   Ruins1_31 / Toll Machine Bench/RestBench/Prompt Marker
#   Fungus3_50 / Toll Machine Bench/RestBench/Prompt Marker
#   Deepnest_Spider_Town / RestBench Spider/Prompt Marker
#   Abyss_18 / Toll Machine Bench/RestBench/Prompt Marker
#   White_Palace_09 / White King Corpse/Throne Sit/Prompt Marker

fsm Prompt {
  start Init

  state Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Blank")
    on UP → Go Up
  }

  state Go Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Up
    on DOWN → Down
  }

  state Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest Idle")
    on DOWN → Down
  }

  state Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest Down")
    on UP → Go Up
  }
}
