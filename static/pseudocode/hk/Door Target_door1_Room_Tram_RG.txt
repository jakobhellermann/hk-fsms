# Door Target
# 1 location(s):
#   Room_Tram_RG / door1

fsm Door Target {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get Station
  }

  state Get Station {
    GetPlayerDataInt(gameObject=var "GameManager", intName="tramRestingGroundsPosition", storeValue=var "Tram Position")
    IntSwitch(intVariable=var "Tram Position", compareTo=[0, 1], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Left
    on RIGHT → Right
  }

  state Left {
    SetFsmString(gameObject=Self, fsmName="Door Control", variableName="New Scene", setValue="Crossroads_46", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on TRAM ARRIVE → Get Station
  }

  state Right {
    SetFsmString(gameObject=Self, fsmName="Door Control", variableName="New Scene", setValue="Crossroads_46b", everyFrame=false)
    on FINISHED → Idle
  }

  var Tram Position: int = 0
}
