fsm Worm Control Inert {
  start Initialise

  state Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    WaitRandom(timeMin=1.5, timeMax=3, finishEvent=→"WAIT", realTime=false)
    on WAIT → Retract
  }

  state Initialise {
    BoolTest(boolVariable=var "Start Down", isTrue=→"DOWN", isFalse=(none), everyFrame=false)
    on FINISHED → Up
    on DOWN → Down
  }

  state Retract {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Retract")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Down
  }

  state Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Down")
    WaitRandom(timeMin=1.5, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst
  }

  state Burst {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Burst")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Up
  }

  var Start Down: bool = false

  var Angle: float = 0
}
