// uses template: hero_detect_region
fsm FSM {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Send to Parent?
  }

  state Send to Parent? {
    BoolTest(boolVariable=var "Send to Parent", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    on FINISHED → Deparent?
  }

  state Detect {
    SendTrigger2DEventByName(eventTarget=GameObject(var "Parent"), trigger=OnTriggerEnter2D, collideTag="", collideLayer=(unset), sendEvent=var "Enter Event", storeCollider=var "")
    SendTrigger2DEventByName(eventTarget=GameObject(var "Parent"), trigger=OnTriggerStay2D, collideTag="", collideLayer=(unset), sendEvent=var "Stay Event", storeCollider=var "")
    SendTrigger2DEventByName(eventTarget=GameObject(var "Parent"), trigger=OnTriggerExit2D, collideTag="", collideLayer=(unset), sendEvent=var "Exit Event", storeCollider=var "")
  }

  state On Stay? {
    StringCompare(stringVariable=var "Stay Event", compareTo="", equalEvent=→"NO STAY", notEqualEvent=→"STAY", storeResult=(unset), everyFrame=false)
    on STAY → Detect
    on NO STAY → Detect No Stay
  }

  state Detect No Stay {
    SendTrigger2DEventByName(eventTarget=GameObject(var "Parent"), trigger=OnTriggerEnter2D, collideTag="", collideLayer=(unset), sendEvent=var "Enter Event", storeCollider=var "")
    SendTrigger2DEventByName(eventTarget=GameObject(var "Parent"), trigger=OnTriggerExit2D, collideTag="", collideLayer=(unset), sendEvent=var "Exit Event", storeCollider=var "")
  }

  state Deparent? {
    BoolTest(boolVariable=var "Deparent", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → On Stay?
  }

  var Deparent: bool = false
  var Send to Parent: bool = true
  var Enter Event: string = "BOW"
  var Exit Event: string = ""
  var Stay Event: string = ""

  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
