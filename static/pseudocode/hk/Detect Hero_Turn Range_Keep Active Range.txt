# Detect Hero
# 1 location(s):
#   Room_shop / Basement Closed/Sly Shop/Keep Active Range/Turn Range

fsm Detect Hero {
  start Init

  state Out {
    Tk2dPlayAnimation(gameObject=var "NPC", animLibName="", clipName="Turn L")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Tk2dPlayAnimation(gameObject=var "NPC", animLibName="", clipName="Turn R")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetParent(gameObject=var "Parent", storeResult=var "NPC")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
    on EXIT → Out
  }

  var NPC: gameObject = <null>
  var Parent: gameObject = <null>
}
