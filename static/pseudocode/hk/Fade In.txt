# Fade In
# 1 location(s):
#   GG_End_Sequence / Fade In

fsm Fade In {
  start Fade Instant

  state Fade Instant {
    GetStaticVariable(variableName="skipEndWhiteFader", storeValue=var "Skip Fader")
    BoolTest(boolVariable=var "Skip Fader", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fade Out
    on CANCEL → Wait
  }

  state Fade Out {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    on FINISHED → 
  }

  state Inert {
    SetStaticVariable(variableName="skipEndWhiteFader", setValue=false)
    SendEventByName(eventTarget=GameObject(var "2dtk Blanker"), sendEvent="FADE OUT INSTANT", delay=0, everyFrame=false)
  }

  state Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Inert
  }

  var Skip Fader: bool = false
}
