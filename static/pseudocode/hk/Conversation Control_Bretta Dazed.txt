# Conversation Control
# 1 location(s):
#   Fungus2_23 / Bretta Dazed

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Box Up
  }

  state Convo Choice {
    GetPlayerDataBool(boolName="brettaRescued", storeValue=var "Met")
    BoolTest(boolVariable=var "Met", isTrue=→"MEET 2", isFalse=→"MEET", everyFrame=false)
    on MEET → Meet
    on MEET 2 → Repeat
  }

  state Init {
    FindChild(gameObject=Self, childName="Steam", storeResult=var "Steam")
    FindChild(gameObject=Self, childName="Haze", storeResult=var "Haze")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Active?
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Box Up {
    AudioStop(gameObject=var "Voice Player")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    AudioStop(gameObject=Self)
    on FINISHED → Convo Choice
  }

  state Meet {
    SetPlayerDataBool(boolName="brettaRescued", value=true)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_RESCUE_1", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Bretta_rescue_01 (sharedassets187.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Meet 2
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Active? {
    GetPlayerDataBool(boolName="brettaRescued", storeValue=var "Met")
    BoolTest(boolVariable=var "Met", isTrue=→"DESTROY", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Scared Loop
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Meet 2 {
    PlayParticleEmitter(gameObject=var "Steam", emit=0)
    StopParticleEmitter(gameObject=var "Haze")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wake")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_RESCUE_2", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Pause
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_RESCUE_5", "Minor NPC"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="BRETTA", everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Bretta_rescue_01 (sharedassets187.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Meet 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_RESCUE_3", "Minor NPC"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="BRETTA", everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Bretta_rescue_02 (sharedassets187.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Pause 2
  }

  state Pause {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 2
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Meet 3
  }

  state Meet 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_RESCUE_4", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Pause 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 3
  }

  state Box Up 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Meet 4
  }

  state Scared Loop {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Idle
  }

  var Met: bool = false
  var Area Title: gameObject = <null>
  var Haze: gameObject = <null>
  var Self: gameObject = <null>
  var Steam: gameObject = <null>
  var Voice Player: gameObject = <null>
}
