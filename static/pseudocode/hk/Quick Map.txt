# Quick Map
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Quick Map
#   resources.assets / _GameCameras/HudCamera/Quick Map

fsm Quick Map {
  start Globalise
  on NO MAP → No Map  // from any state
  on LEVEL LOADED → Reset Override  // from any state

  state Init {
    FindChild(gameObject=Self, childName="No Map", storeResult=var "No Map Msg")
    FindChild(gameObject=Self, childName="BG", storeResult=var "BG")
    FindChild(gameObject=Self, childName="Area Name", storeResult=var "Area Name Txt")
    ActivateGameObject(gameObject=var "No Map Msg", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Area Name Txt", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on OPEN QUICK MAP → Open
  }

  state Check Area {
    SetScale(gameObject=var "Game Map", vector=(unset), x=1.55, y=1.55, z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=Self, sendEvent=var "Map Zone", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="NONE", delay=0, everyFrame=false)
    on CROSSROADS → Crossroads
    on FOG_CANYON → Fog Canyon
    on GREEN_PATH → Greenpath
    on WASTES → Fungal Wastes
    on CITY → City
    on WATERWAYS → Waterways
    on MINES → Mines
    on DEEPNEST → Deepnest
    on TOWN → Dirtmouth
    on CLIFFS → Cliffs
    on NONE → No Map
    on TEST_AREA → No Map
    on KINGS_PASS → Dirtmouth
    on ROYAL_GARDENS → Royal Gardens
    on SHAMAN_TEMPLE → Crossroads
    on QUEENS_STATION → Fungal Wastes
    on RESTING_GROUNDS → Resting Grounds
    on ABYSS → Abyss
    on KINGS_STATION → City
    on OUTSKIRTS → Outskirts
    on HIVE → Outskirts
    on COLOSSEUM → Outskirts
    on SOUL_SOCIETY → City
    on MONOMON_ARCHIVE → Fog Canyon
    on BEASTS_DEN → Deepnest
    on LURIENS_TOWER → City
    on FINISHED → No Map
    on GODSEEKER_WASTE → Waterways
  }

  state Crossroads {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="mapCrossroads", isTrue=(none), isFalse=→"NO MAP")
    GetLanguageString(sheetName="Map Zones", convName="CROSSROADS", storeValue=var "Area Name")
    SetTextMeshProText(gameObject=var "Area Name Txt", textString=var "Area Name")
    FadeGroupUp(target=var "Area Name Txt")
    SendEventByName(eventTarget=GameObject(var "Area Name Txt"), sendEvent="UP", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Game Map", delivery=SendMessage, options=1, functionCall=QuickMapCrossroads())
    SetPosition(gameObject=var "Game Map", vector=(unset), x=-0.04, y=-7.88, z=18, space=1, everyFrame=false, lateUpdate=false)
    on CLOSE QUICK MAP → Check State
  }

  state No Map {
    FadeGroupUp(target=var "No Map Msg")
    on CLOSE QUICK MAP → Check State
  }

  state Close {
    SendMessage(gameObject=var "Game Map", delivery=SendMessage, options=1, functionCall=CloseQuickMap())
    FadeGroupDown(target=var "Area Name Txt", fast=false)
    FadeGroupDown(target=var "BG", fast=false)
    FadeGroupDown(target=var "No Map Msg", fast=false)
    on OPEN QUICK MAP → Open
  }

  state Greenpath {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="mapGreenpath", isTrue=(none), isFalse=→"NO MAP")
    GetLanguageString(sheetName="Map Zones", convName="GREEN_PATH", storeValue=var "Area Name")

    on CLOSE QUICK MAP → Check State
  }

  state Fungal Wastes {

    on CLOSE QUICK MAP → Check State
  }

  state Open {
    SetFsmBool(gameObject=var "Game Map", fsmName="Map FSM State", variableName="Display Next Area", setValue=true, everyFrame=false)
    SetCameraCullingMask(gameObject=var "HUD Camera", cullingMask=[UI(5), Map Pin(30)], invertMask=false, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    FadeGroupUp(target=var "BG")
    on FINISHED → Get Map Zone
    on CLOSE QUICK MAP → Close
  }

  state Dirtmouth {
    GetLanguageString(sheetName="Map Zones", convName="TOWN", storeValue=var "Area Name")
    SetTextMeshProText(gameObject=var "Area Name Txt", textString=var "Area Name")

    SendMessage(gameObject=var "Game Map", delivery=SendMessage, options=1, functionCall=QuickMapDirtmouth())
    SetPosition(gameObject=var "Game Map", vector=(unset), x=4.07, y=-11.62, z=18, space=1, everyFrame=false, lateUpdate=false)
    on CLOSE QUICK MAP → Check State
  }

  state Reset Override {
    SetBoolValue(boolVariable=var "In Room", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state In Room? {
    BoolTest(boolVariable=var "In Room", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "Game Map", behaviour="GameMap", methodName="GetDoorMapZone", parameters=[], storeResult=var "Map Zone")
    on FINISHED → Check Area
  }

  state Get Map Zone {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="GetCurrentMapZone", parameters=[], storeResult=var "Map Zone")
    on FINISHED → In Room?
  }

  state Globalise {
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "Quick Map", gameObject=var "Self", everyFrame=false)
    on FINISHED → Init
  }

  state Cliffs {

  }

  state Check State {
    GetPlayerDataInt(gameObject=var "GameManager", intName="mapKeyPref", storeValue=var "Map Key Pref")
    IntSwitch(intVariable=var "Map Key Pref", compareTo=[0, 1, 2], sendEvent=[→"KEY AND PIN", →"PIN", →"NONE"], everyFrame=false)
    on KEY AND PIN → Pin On
    on PIN → Pin On
    on NONE → Pin Off
  }

  state Pin On {
    SetCameraCullingMask(gameObject=var "HUD Camera", cullingMask=[UI(5), Map Pin(30)], invertMask=false, everyFrame=false)
    on FINISHED → Close
  }

  state Pin Off {
    SetCameraCullingMask(gameObject=var "HUD Camera", cullingMask=[UI(5)], invertMask=false, everyFrame=false)
    on FINISHED → Close
  }

  var Map Key Pref: int = 0
  var In Room: bool = false
  var Area Name: string = ""
  var Map Zone: string = ""
  var Area Name Txt: gameObject = <null>
  var BG: gameObject = <null>
  var No Map Msg: gameObject = <null>
  var Self: gameObject = <null>
