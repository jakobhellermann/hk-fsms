# Lil Jelly
# 1 location(s):
#   sharedassets156.assets / Lil Jellyfish

fsm Lil Jelly {
  start Startle
  on ZERO HP → Die  // from any state

  state Startle {
    GetOwner(storeGameObject=var "Self")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lil Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Chase
  }

  state Die {
    GGCheckIfBossScene(bossSceneEvent=→"GG SCENE", regularSceneEvent=(none))
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Recycle L (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    StopParticleEmitter(gameObject=var "Steam")
    DestroySelf(detachChildren=true)
    on GG SCENE → Uumuu Explosion
  }

  state Chase {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lil Fly")
    FaceAngle(gameObject=Self, angleOffset=270, everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"ZERO HP", storeCollider=var "")
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"ZERO HP", storeCollider=var "", storeForce=(unset))
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=80, accelerationForce=90, offsetX=0, offsetY=0)
  }

  state Uumuu Explosion {
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Uumuu, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    StopParticleEmitter(gameObject=var "Steam")
    DestroySelf(detachChildren=true)
  }

  var Self: gameObject = <null>
  var Steam: gameObject = Lil Jellyfish/Corpse Steam
}
