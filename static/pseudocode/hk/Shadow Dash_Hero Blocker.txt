// uses template: shadow_dash_solid
fsm Shadow Dash {
  start Idle

  state Idle {
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Collision2dEvent(collision=1, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    SetPolygonCollider(gameObject=Self, active=true)
    on COLLIDE → Check
  }

  state Check {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["shadowDashing"], storeResult=var "Shadow Dashing")
    BoolTest(boolVariable=var "Shadow Dashing", isTrue=→"SHADOW DASH", isFalse=→"FINISHED", everyFrame=false)
    on SHADOW DASH → Shadow Dashed
    on FINISHED → Return Pause
  }

  state Shadow Dashed {
    SetPolygonCollider(gameObject=Self, active=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Return Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Shadow Dashing: bool = false
}
