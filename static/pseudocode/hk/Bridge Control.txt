# Bridge Control
# 1 location(s):
#   Mines_31 / mine_break_bridge

fsm Bridge Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasSuperDash", isTrue=→"BROKEN", isFalse=(none))
    on BROKEN → Destroy
    on SUPERDASH BLAST → Remove Collider
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Remove Collider {
    SetCollider(gameObject=Self, active=false)
  }
}
