# Animation
# 1 location(s):
#   Town / Nymm NPC

fsm Animation {
  start Pause

  state Init {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on L → Hero L
    on R → Hero R
  }

  state Pause {
    on FINISHED → Init
  }

  state Hero L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look L")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on R → Hero R
  }

  state Hero R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look R")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on L → Hero L
  }
}
