# Destroy
# 3 location(s):
#   Ruins1_27 / Destroyer
#   Ruins1_03 / Destroyer
#   Ruins1_04 / Destroyer

fsm Destroy {
  start Detect

  state Detect {
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"DESTROY", storeCollider=var "Object", storeForce=(unset))
    on DESTROY → Player Test
  }

  state Destroy {
    DestroyObject(gameObject=var "Object", delay=0, detachChildren=false)
    on FINISHED → Detect
  }

  state Player Test {
    GameObjectCompareTag(gameObject=var "Object", tag="Player", trueEvent=→"FINISHED", falseEvent=→"DESTROY", storeResult=(unset), everyFrame=false)
    on DESTROY → Destroy
    on FINISHED → Detect
  }

  var Object: gameObject = <null>
}
