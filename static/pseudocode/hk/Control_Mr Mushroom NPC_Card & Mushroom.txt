# Control
# 1 location(s):
#   Fungus2_18 / Card & Mushroom/Mr Mushroom NPC

fsm Control {
  start Pause
  on LEAVE → Leave  // from any state
  on ROCKET → Rocket Pause  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="mrMushroomState", storeValue=var "Current Mush State")
    IntCompare(integer1=var "Current Mush State", integer2=var "Mushroom State", equal=→"FINISHED", lessThan=→"DESTROY", greaterThan=→"DESTROY", everyFrame=false)
    on DESTROY → Destroy
    on FINISHED → Init
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dust", storeResult=var "Dust")
    FindChild(gameObject=Self, childName="Pt Blastoff 1", storeResult=var "Pt Blastoff 1")
    FindChild(gameObject=Self, childName="Pt Trail", storeResult=var "Pt Trail")
    FindChild(gameObject=Self, childName="Pt Blastoff 2", storeResult=var "Pt Blastoff 2")
    ActivateGameObject(gameObject=var "Cornifer Card", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Rest
  }

  state Rest {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest")
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on NAIL HIT → Rest Hit
  }

  state Rest Hit {
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(0, -0.5, 0), rotation=(unset), storeObject=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest Hit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bounce_shroom (resources.assets), pitchMin=0.9, pitchMax=0.7, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mr_Mush_hit_01 (sharedassets6.assets), Mr_Mush_hit_02 (sharedassets6.assets), Mr_Mush_hit_03 (sharedassets6.assets), Mr_Mush_hit_04 (sharedassets6.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SendMessage(gameObject=var "Hit Detect", delivery=SendMessage, options=1, functionCall=SetActive(true))
    on FINISHED → Bounce On
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetCollider(gameObject=Self, active=true)
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"TURN BACK", everyFrame=true)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on NAIL HIT → Idle Hit
    on TURN BACK → Turn Back
  }

  state Idle Hit {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle Hit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(0, -0.5, 0), rotation=(unset), storeObject=var "")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bounce_shroom (resources.assets), pitchMin=0.9, pitchMax=0.7, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mr_Mush_hit_01 (sharedassets6.assets), Mr_Mush_hit_02 (sharedassets6.assets), Mr_Mush_hit_03 (sharedassets6.assets), Mr_Mush_hit_04 (sharedassets6.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioStop(gameObject=Self)
    on FINISHED → Idle
  }

  state Turn Back {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn Back", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    on FINISHED → Rest
    on NAIL HIT → Rest Hit
  }

  state Leave {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mr_mushroom_burrow (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Left
  }

  state Left {
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Rocket Start {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rocket Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Self", audioClip=mr_mush_takeoff (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Liftoff
  }

  state Liftoff {
    PlayParticleEmitter(gameObject=var "Pt Blastoff 1", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Blastoff 2", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Trail", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="RumblingMed", variableName="", setValue=true, everyFrame=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 3.5, 0), time=1, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"NEXT", realTime=false, stopOnExit=true, loopDontFinish=true)
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=true)
    on NEXT → Fly Away
  }

  state Fly Away {
    iTweenMoveBy(gameObject=Self, id="", vector=(0.5, 15, 0), time=1, delay=0, speed=(unset), easeType=easeInSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=1.25, finishEvent=→"NEXT", realTime=false)
    on NEXT → Talk Pause
  }

  state Talk Pause {
    SetMeshRenderer(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Pt Trail")
    SetFsmBool(gameObject=var "CameraParent", fsmName="RumblingMed", variableName="", setValue=false, everyFrame=false)
    Wait(time=1.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up
  }

  state Rocket Pause {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Silent")
    on FINISHED → Rocket Start
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Final Convo
  }

  state Final Convo {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MR_MUSHROOM_FINAL2", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mr_Mush_talk_03 (sharedassets6.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → End
  }

  state End {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Bounce On {
    SendMessage(gameObject=var "Hit Detect", delivery=SendMessage, options=1, functionCall=SetActive(false))
    on FINISHED → Idle
  }

  var Mushroom State: int = 1
  var Cornifer Card: gameObject = Card & Mushroom/Cornifer Card

  var Current Mush State: int = 0
  var Hero Range: bool = false
  var Dust: gameObject = <null>
  var Hit Detect: gameObject = Card & Mushroom/Mr Mushroom NPC/Hit Detect
  var Pt Blastoff 1: gameObject = <null>
  var Pt Blastoff 2: gameObject = <null>
  var Pt Trail: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
