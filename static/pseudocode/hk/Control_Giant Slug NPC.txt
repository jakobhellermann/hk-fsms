# Control
# 1 location(s):
#   Fungus1_Slug / Giant Slug NPC

fsm Control {
  start Pause

  state Pause {
    FindChild(gameObject=Self, childName="Shiny Item", storeResult=var "Shiny Item")
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    ActivateGameObject(gameObject=var "Shiny Item", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="slugEncounterComplete", isTrue=→"SEEN", isFalse=(none))
    FindChild(gameObject=Self, childName="Pt Bub", storeResult=var "Pt Bub")
    FindChild(gameObject=Self, childName="Pt Spore", storeResult=var "Pt Spore")
    FindChild(gameObject=Self, childName="Pt Steam", storeResult=var "Pt Steam")
    FindChild(gameObject=Self, childName="Spatter Point", storeResult=var "Spatter Point")
    FindChild(gameObject=Self, childName="Pt Steam In", storeResult=var "Pt Steam In")
    FindChild(gameObject=Self, childName="Pt Spore In", storeResult=var "Pt Spore In")
    on SEEN → Seen
    on FINISHED → Idle
  }

  state Seen {
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter Pause
  }

  state Enter Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rumble
  }

  state Rumble {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Spore", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Steam", emit=0)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Emerge
  }

  state Emerge {
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Spatter Point", audioClip=uumuu_appear (sharedassets164.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Spore")
    StopParticleEmitter(gameObject=var "Pt Steam")
    PlayParticleEmitter(gameObject=var "Pt Bub", emit=0)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Emerge")
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Spatter Point", position=(unset), spawnMin=50, spawnMax=50, speedMin=20, speedMax=25, angleMin=80, angleMax=100, originVariationX=4, originVariationY=0, FSM="", FSMEvent="")  // disabled
    SpawnRandomObjectsV2(gameObject=var "", spawnPoint=var "Hero", position=(0.2, 0, 0), spawnMin=6, spawnMax=8, speedMin=7, speedMax=15, angleMin=35, angleMax=60, originVariationX=0, originVariationY=0)  // disabled
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spit 1
  }

  state Spit 1 {
    SetPlayerDataBool(boolName="slugEncounterComplete", value=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Spit 2
  }

  state Spit 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Spatter Point", audioClip=mawlek_spit (sharedassets33.assets), pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Shiny Item", fsmName="Shiny Control", variableName="Slug Fling", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Shiny Item", position=(unset), spawnMin=10, spawnMax=10, speedMin=10, speedMax=15, angleMin=80, angleMax=30, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    on FINISHED → Leave Pause
  }

  state Leave Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Submerge
  }

  state Submerge {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Submerge", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Spatter Point", audioClip=uumuu_appear (sharedassets164.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Enter
  }

  state Gone {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Enter {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Spore In", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Steam In", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Bub", emit=0)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Spatter Point", audioClip=uumuu_appear (sharedassets164.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Spatter Point", audioClip=acid_splash (resources.assets), pitchMin=0.75, pitchMax=0.75, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Spatter Point", audioClip=acid_splash (resources.assets), pitchMin=0.75, pitchMax=0.75, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Gone
  }

  var Pt Bub: gameObject = <null>
  var Pt Spore: gameObject = <null>
  var Pt Spore In: gameObject = <null>
  var Pt Steam: gameObject = <null>
  var Pt Steam In: gameObject = <null>
  var Shiny Item: gameObject = <null>
  var Spatter Point: gameObject = <null>
}
