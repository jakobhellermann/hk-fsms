# Conversation Control
# 1 location(s):
#   Dream_Nailcollection / Witch Control/Dream Nail Get

fsm Conversation Control {
  start Pause
  on RESET CONVO → Idle  // from any state

  state Idle {
    on CONVO START → PD Get
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Knight Dummy", storeResult=var "Knight Dummy")
    FindChild(gameObject=Self, childName="Dream Nail", storeResult=var "Dream Nail")
    FindChild(gameObject=Self, childName="Get Flash", storeResult=var "Get Flash")
    FindChild(gameObject=var "Dream Nail", childName="Sprite", storeResult=var "DN Sprite")
    FindChild(gameObject=var "Dream Nail", childName="Idle Pt", storeResult=var "DN Pt")
    FindChild(gameObject=Self, childName="Summon Pt", storeResult=var "Summon Pt")
    FindChild(gameObject=Self, childName="Summon Pt 2", storeResult=var "Summon Pt 2")
    FindChild(gameObject=Self, childName="Pickup Glow", storeResult=var "Pickup Glow")
    on FINISHED → Set Convos
  }

  state Set Convos {
    BuildString(stringParts=[var "Ghost Name", "_TALK"], separator="", addToEnd=true, storeResult=var "Talk Convo", everyFrame=false)
    BuildString(stringParts=[var "Ghost Name", "_REPEAT"], separator="", addToEnd=true, storeResult=var "Repeat Convo", everyFrame=false)
    BuildString(stringParts=["GH_", var "Ghost Name", "_C"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    BuildString(stringParts=["GH_", var "Ghost Name", "_NC"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state PD Get {
    SetPlayerDataBool(boolName="hasDreamNail", value=true)
    on FINISHED → Anim 1
  }

  state Anim 1 {
    SetMeshRenderer(gameObject=var "Hero", active=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Knight Dummy", vector=(unset), x=var "Hero Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Summon Pt", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Summon Pt 2", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Summon Pt 2", vector=(unset), x=0, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Knight Dummy", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Knight Dummy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Knight Dummy", animLibName="", clipName="DN Get 1")
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    TransitionToAudioSnapshot(snapshot=en Arena (resources.assets), transitionTime=1)
    on FINISHED → Anim 2
  }

  state Anim 2 {
    StopParticleEmitter(gameObject=var "DN Pt")
    Tk2dPlayAnimation(gameObject=var "Knight Dummy", animLibName="", clipName="DN Get 2")
    ActivateGameObject(gameObject=var "DN Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Get Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pickup Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_orb_pickup (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Anim 3
  }

  state Anim 3 {
    Tk2dPlayAnimation(gameObject=var "Knight Dummy", animLibName="", clipName="DN Get 3")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Particle
  }

  state Fade Out {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=4.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → UI Msg
  }

  state New Scene {
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", setValue=false, everyFrame=false)
    SetTextMeshProColor(gameObject=var "DialogueText", color=(1, 1, 1, 1), everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Return Long", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    SetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", value="Dream_NailCollection")
    BeginSceneTransition(sceneName="RestingGrounds_07", entryGateName="door_dreamReturn", entryDelay=0, visualization=Custom, preventCameraFadeOut=true)
  }

  state Particle {
    PlayParticleEmitter(gameObject=var "Summon Pt", emit=0)
    PlayParticleEmitter(gameObject=var "Summon Pt 2", emit=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Fade Out
  }

  state UI Msg {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=false, everyFrame=false)
    CreateUIMsgGetItem(gameObject=UI Msg Get Item White (sharedassets394.assets), storeObject=var "UI Msg", sprite=_0008_dream_nail (sharedassets242.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Dream Nail", everyFrame=false)
    on GET ITEM MSG END → New Scene
  }

  var Ghost Name: string = ""

  var Hero Scale: float = 0
  var Repeat Convo: string = ""
  var Talk Convo: string = ""
  var Title Convo: string = ""
  var Hero Pos: vector3 = (0, 0, 0)
  var DN Pt: gameObject = <null>
  var DN Sprite: gameObject = <null>
  var Dream Nail: gameObject = <null>
  var Get Flash: gameObject = <null>
  var Knight Dummy: gameObject = <null>
  var Pickup Glow: gameObject = <null>
  var Self: gameObject = <null>
  var Summon Pt: gameObject = <null>
  var Summon Pt 2: gameObject = <null>
  var UI Msg: gameObject = <null>
}
