# Control2
# 1 location(s):
#   Dream_02_Mage_Lord / Dream Gate Phase 2 (2)

fsm Control2 {
  start Init

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "CamLocks", activate=false)
    on PHASE 2 → Close
  }

  state Close {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    ActivateAllChildren(gameObject=var "CamLocks", activate=true)
    on FINISHED → Closed
    on DREAM GATE OPEN → Open
  }

  state Closed {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on DREAM GATE OPEN → Open
  }

  state Open {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "CamLocks", activate=false)
    on FINISHED → Opened
    on PHASE 2 → Close
  }

  state Opened {
    SetMeshRenderer(gameObject=Self, active=false)
    on PHASE 2 → Close
  }

  var CamLocks: gameObject = Dream Gate Phase 2 (2)/CamLocks
}
