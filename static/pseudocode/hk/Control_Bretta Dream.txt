# Control
# 1 location(s):
#   Dream_Mighty_Zote / Audience/Bretta Dream

fsm Control {
  start Init
  on ZOTE EXPLODE → Pause  // from any state
  on ZOTE KILLED → Zote Killed  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bretta Idle")
    on ZOTE ROAR START → Clapping
    on ZOTE STUNNED → Shock
    on HERO DAMAGED → Damage Clap
  }

  state Clapping {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bretta Clap")
    on ZOTE ROAR END → Idle
  }

  state Shock {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bretta Shock")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[<null>, <null>], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Damage Clap {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bretta Clap")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Zote Killed {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bretta Shock")
  }

  state Pause {
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[queen_bretta_collapse_long_01 (sharedassets399.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Collapse
  }

  state Collapse {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bretta Collapse", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[brkn_wand_land (sharedassets32.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0.5, storePlayer=<null>)
    on FINISHED → Dream End
  }

  state Dream End {
    CreateObject(gameObject=Dream Orb Get Long (sharedassets399.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=ghost_absorb_final_impact (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  var X Scale: float = 0
  var Self: gameObject = <null>
}
