# Extra Anim
# 1 location(s):
#   Room_Charm_Shop / Charm Slug

fsm Extra Anim {
  start Idle

  state Idle {
    on SHOP START → Shop Start
    on SHOP STOP → Shop Stop
  }

  state Shop Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shop Start")
    AudioStop(gameObject=Self)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[Salubra_Shop_Greet (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Idle
  }

  state Shop Stop {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shop End")
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))  // disabled
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
