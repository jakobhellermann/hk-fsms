# ghost_npc_death
# 1 location(s):
#   RestingGrounds_08 / Ghost revek

// uses template: ghost_npc_death
fsm ghost_npc_death {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Character Sprite", storeResult=var "Character Sprite")
    BoolTest(boolVariable=var "Activated", isTrue=→"DESTROYED", isFalse=(none), everyFrame=false)
    on DESTROYED → Revek
    on FINISHED → Spirit Glade
  }

  state Destroy {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    on DREAMNAIL → Revek?
    on GHOST NPC KILLED → Remove?
  }

  state Impact {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="NPC CONTROL OFF", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST NPC KILLED", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    CreateObject(gameObject=Dream Burst Pt, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    CreateObject(gameObject=Dream Orb Get S, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayerDataIntAdd(gameObject=var "GameManager", intName="dreamOrbs", amount=var "Orbs")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM ORB COLLECT", delay=0, everyFrame=false)  // disabled
    on FINISHED → Kill
  }

  state Kill {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SetCollider(gameObject=Self, active=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioStop(gameObject=Self)
  }

  state Revek? {
    BoolTest(boolVariable=var "z_Revek", isTrue=→"REVEK", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Impact
    on REVEK → Warp Out
  }

  state Warp Out {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST NPC KILLED", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="NPC CONTROL OFF", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    on FINISHED → Remove
  }

  state Remove? {
    BoolTest(boolVariable=var "z_Revek", isTrue=→"REVEK", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on REVEK → Remove
  }

  state Spirit Glade {
    BoolTest(boolVariable=var "z_Spirit Glade", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="gladeGhostsKilled")  // disabled
    GetFsmInt(gameObject=var "Death Counting Object", fsmName="Control", variableName="Ghosts Alive", storeValue=var "Ghosts Remaining", everyFrame=false)
    IntAdd(intVariable=var "Ghosts Remaining", add=1, everyFrame=false)
    SetFsmInt(gameObject=var "Death Counting Object", fsmName="Control", variableName="Ghosts Alive", setValue=var "Ghosts Remaining", everyFrame=false)
    on FINISHED → Idle
  }

  state Remove {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SetCollider(gameObject=Self, active=false)
    AudioStop(gameObject=Self)
  }

  state Revek {
    BoolTest(boolVariable=var "z_Revek", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REVEK KILLED", delay=0, everyFrame=false)
    on FINISHED → Destroy
  }

  var z_Revek: bool = true
  var z_Spirit Glade: bool = false
  var Death Counting Object: gameObject = <null>

  var Ghosts Remaining: int = 0
  var Orbs: int = 1
  var Activated: bool = false
  var Character Sprite: gameObject = <null>
  var Self: gameObject = <null>
}
