# proj control
# 1 location(s):
#   sharedassets138.assets / Shot Orange LG 0.7

fsm proj control {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=1.4, stretchMinX=0.5, stretchMaxY=2, scaleModifier=0.7, everyFrame=true)
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"COLLISION ENTER 2D", storeCollider=var "", storeForce=(unset))
    SetCollider(gameObject=Self, active=true)
    on COLLISION ENTER 2D → Check Normal
  }

  state Impact {
    SetCollider(gameObject=Self, active=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=5, spawnMax=8, speedMin=10, speedMax=15, angleMin=var "Angle Min", angleMax=var "Angle Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    CreateObject(gameObject=Audio Player Actor (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Audio Player", networkInstantiate=false, networkGroup=0)
    RandomFloat(min=0.8, max=1.2, storeResult=var "Chooser")
    SetAudioPitch(gameObject=var "Audio Player", pitch=var "Chooser", everyFrame=false)
    AudioPlay(gameObject=var "Audio Player", volume=1, oneShotClip=orange_shot_impact (sharedassets32.assets), finishedEvent=(none))
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Destroy
  }

  state Destroy {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RecycleSelf()
  }

  state Check Normal {
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=(unset), contactNormal=var "Contact Normal", shapeCount=(unset), physics2dMaterialName="")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Impact")
    GetVector3XYZ(vector3Variable=var "Contact Normal", storeX=var "X Normal", storeY=var "Y Normal", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Y Normal", float2=1, tolerance=0, equal=→"DOWN", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Y Normal", float2=-1, tolerance=0, equal=→"UP", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "X Normal", float2=1, tolerance=0, equal=→"LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "X Normal", float2=-1, tolerance=0, equal=→"RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    Wait(time=0, finishEvent=→"WAIT", realTime=false)
    on DOWN → Hit Down
    on RIGHT → Hit Right
    on WAIT → Dunno
    on UP → Hit Up
    on LEFT → Hit Left
  }

  state Hit Up {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    ReflectAngle(angle=var "Angle", reflectHorizontally=false, reflectVertically=true, storeResult=var "Angle")
    FloatClamp(floatVariable=var "Angle Max", minValue=40, maxValue=140, everyFrame=false)  // disabled
    SetFloatValue(floatVariable=var "Angle Min", floatValue=var "Angle", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=var "Angle", everyFrame=false)
    FloatAdd(floatVariable=var "Angle Max", add=20, everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Angle Min", subtract=20, everyFrame=false, perSecond=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=180, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  state Hit Right {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    ReflectAngle(angle=var "Angle", reflectHorizontally=true, reflectVertically=false, storeResult=var "Angle")
    SetFloatValue(floatVariable=var "Angle Min", floatValue=var "Angle", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=var "Angle", everyFrame=false)
    FloatAdd(floatVariable=var "Angle Max", add=20, everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Angle Min", subtract=20, everyFrame=false, perSecond=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=90, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  state Hit Left {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    ReflectAngle(angle=var "Angle", reflectHorizontally=true, reflectVertically=false, storeResult=var "Angle")
    SetFloatValue(floatVariable=var "Angle Min", floatValue=var "Angle", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=var "Angle", everyFrame=false)
    FloatAdd(floatVariable=var "Angle Max", add=20, everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Angle Min", subtract=20, everyFrame=false, perSecond=false)
    SpawnRandomObjects(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=5, spawnMax=8, speedMin=10, speedMax=15, angleMin=var "Angle Min", angleMax=var "Angle Max", originVariation=0)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=270, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  state Dunno {
    on FINISHED → Hit Down
  }

  state Hit Down {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    ReflectAngle(angle=var "Angle", reflectHorizontally=false, reflectVertically=true, storeResult=var "Angle")
    SetFloatValue(floatVariable=var "Angle Min", floatValue=var "Angle", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=var "Angle", everyFrame=false)
    FloatAdd(floatVariable=var "Angle Max", add=20, everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Angle Min", subtract=20, everyFrame=false, perSecond=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  var Angle: float = 0
  var Angle Max: float = 0
  var Angle Min: float = 0
  var Chooser: float = 0
  var X Normal: float = 0
  var Y Normal: float = 0
  var Contact Normal: vector3 = (0, 0, 0)
  var Audio Player: gameObject = <null>
  var Self: gameObject = <null>
}
