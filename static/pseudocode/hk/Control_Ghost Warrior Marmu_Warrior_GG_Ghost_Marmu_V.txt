    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[Marmu_long_01 (sharedassets221.assets), Marmu_long_02 (sharedassets221.assets), Marmu_long_03 (sharedassets221.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll Start")
    ActivateGameObject(gameObject=var "Hit Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=true)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0.5, storeAngle=var "Angle To Hero", everyFrame=false)

    SetVelocityAsAngle(gameObject=Self, angle=var "Angle To Hero", speed=40, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.85)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fire
  }

  state Chase {

    FloatAdd(floatVariable=var "Roll Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Roll Timer", float2=var "Roll Time", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UNROLL", everyFrame=true)
    on HIT DOWN → Hit Down
    on HIT LEFT → Hit Left
    on HIT RIGHT → Hit Right
    on HIT UP → Hit Up
    on UNROLL → Roll Warp?
  }

  state Hit Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-40, everyFrame=false)
    on FINISHED → Hit Voice
  }

  state Hit Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-40, y=15, everyFrame=false)
    on FINISHED → Hit Voice
  }

  state Hit Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=40, y=15, everyFrame=false)

  }

  state Hit Up {

    on FINISHED → Hit Voice
  }

    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Warp?
  }

  state Fire {
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=-0.5, storeAngle=var "Angle To Hero", everyFrame=false)

    SetVelocityAsAngle(gameObject=Self, angle=var "Angle To Hero", speed=20, everyFrame=false)
    on FINISHED → Chase
  }

  state Warp? {

    RandomFloat(min=var "Tele Y Min", max=var "Tele Y Max", storeResult=var "Tele Y")

  }

    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[mage_knight_teleport (sharedassets6.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

  }

  state Warp Out {
    ActivateGameObject(gameObject=var "Warp Out", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Warp
  }
