# grub_rewards_shinyitem
# 1 location(s):
#   Crossroads_38 / Grub King/Rewards Parent/Reward 31

// uses template: grub_reward_shinyitem
fsm grub_rewards_shinyitem {
  start Pause

  state Init {
    ActivateGameObject(gameObject=var "Item", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindGameObject(objectName=var "Item Name", withTag="Heart Piece", store=var "Item")  // disabled
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Item", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → Idle
    on ACTIVATED → Activated
  }

  state Idle {
    on GIVE → Give
  }

  state Give {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="Loose Rewards", withTag="Extra Tag", store=var "Loose Rewards")
    FindChild(gameObject=var "Loose Rewards", childName=var "Item Name", storeResult=var "Item")
    ActivateGameObject(gameObject=var "Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activated {
    ActivateGameObject(gameObject=var "Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Item Name: string = "Shiny Item Ore"

  var Activated: bool = false
  var Item: gameObject = <null>
  var Loose Rewards: gameObject = <null>
  var Self: gameObject = <null>
}
