# Appear
# 7 location(s):
#   Fungus1_35 / Ghost Warrior NPC
#   Fungus2_32 / Ghost Warrior NPC
#   Fungus3_40_boss / Ghost Warrior NPC
#   Cliffs_02_boss / Ghost Warrior NPC
#   RestingGrounds_02_boss / Ghost Warrior NPC
#   Deepnest_40 / Ghost Warrior NPC
#   Deepnest_East_10 / Ghost Warrior NPC

fsm Appear {
  start Pause
  on FIGHT START → Fighting  // from any state
  on GHOST DEFEAT → Reappear  // from any state

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDreamNail", isTrue=(none), isFalse=→"NO NAIL")
    on FINISHED → Away
    on NO NAIL → Inert
  }

  state Away {
    BoolTest(boolVariable=var "Appear Range", isTrue=→"APPEAR", isFalse=(none), everyFrame=true)
    SetParticleEmission(gameObject=var "Idle Pt", emission=false)
    SetParticleEmission(gameObject=var "Away Pt", emission=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    on APPEAR → Close
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inert {
    AudioStop(gameObject=Self)
  }

  state Close {
    BoolTest(boolVariable=var "Unappear Range", isTrue=(none), isFalse=→"DISAPPEAR", everyFrame=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=dream_ghost_appear (resources.assets))
    SetParticleEmission(gameObject=var "Idle Pt", emission=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Away Pt", emission=false)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Burst", animLibName="", clipName="Burst")
    Tk2dPlayFrame(gameObject=var "Burst", frame=0)
    on DISAPPEAR → Burst Away
  }

  state Burst Away {
    Tk2dPlayAnimation(gameObject=var "Burst", animLibName="", clipName="Burst")
    Tk2dPlayFrame(gameObject=var "Burst", frame=0)
    on FINISHED → Away
  }

  state Fighting {
  }

  state Reappear {
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wounded", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Sprite", delivery=SendMessage, options=1, functionCall=FlashingGhostWounded())
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Attack Pt", emit=0)
    PlayParticleEmitter(gameObject=var "Idle Pt", emit=0)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=true)
  }

  var Appear Range: bool = false
  var Unappear Range: bool = false
  var Attack Pt: gameObject = Ghost Warrior NPC/Attack Pt
  var Away Pt: gameObject = Ghost Warrior NPC/Away Pt
  var Burst: gameObject = Ghost Warrior NPC/Burst
  var Idle Pt: gameObject = Ghost Warrior NPC/Idle Pt
  var Sprite: gameObject = Ghost Warrior NPC/Character Sprite
  var Warp: gameObject = Ghost Warrior NPC/Warp
  var White Flash: gameObject = Ghost Warrior NPC/White Flash
  var Wounded: gameObject = Ghost Warrior NPC/Wounded
}
