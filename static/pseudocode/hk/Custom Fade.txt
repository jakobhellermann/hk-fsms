# Custom Fade
# 6 location(s):
#   GG_Spa / right1
#   GG_Engine / right1
#   GG_Engine_Prime / right1
#   GG_Engine_Root / right1
#   GG_Wyrm / right1
#   GG_Unn / right1

// uses template: gg_custom_exit_fade
fsm Custom Fade {
  start Idle

  state Idle {
    on FADE → Fade Out
  }

  state Fade Out {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0.33, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "MainCamera", behaviour="CameraController", methodName="FreezeInPlace", parameters=[true], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
  }
}
