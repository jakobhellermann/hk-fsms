// uses template: deactivate_if_pd_int_under
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName=var "PD Int Name", storeValue=var "PD Value")
    IntCompare(integer1=var "PD Value", integer2=var "Value", equal=(none), lessThan=→"INACTIVE", greaterThan=(none), everyFrame=false)
    on INACTIVE → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Value: int = 1
  var PD Int Name: string = "greyPrinceDefeats"

  var PD Value: int = 0
}
