# Open Stag
# 1 location(s):
#   Menu_Title / _GameCameras/HudCamera/Menus

fsm Open Stag {
  start Init
  on SPAWN → Spawn Stag Map  // from any state
  on DESPAWN → Destroy Map  // from any state

  state Idle {
    on OPEN STAG MENU → Open
  }

  state Open {
    ActivateGameObject(gameObject=var "Menu", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    CreateObject(gameObject=Stag Map (resources.assets), spawnPoint=var "Self", position=(0.03, 0.03, 35.35), rotation=(unset), storeObject=var "Menu", networkInstantiate=false, networkGroup=0)  // disabled
    ActivateGameObject(gameObject=var "Menu", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Menu", vector=(unset), x=0.3, y=0.3, z=35.35, space=1, everyFrame=false, lateUpdate=false)
    SetFsmGameObject(gameObject=var "Menu", fsmName="Control", variableName="Requester", setValue=var "Requester", everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    FindChild(gameObject=Self, childName="Stag Map", storeResult=var "Menu")  // disabled
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Inert
  }

  state Spawn Stag Map {
    CreateObject(gameObject=Stag Map (resources.assets), spawnPoint=var "Self", position=(0.03, 0.03, 35.35), rotation=(unset), storeObject=var "Menu", networkInstantiate=false, networkGroup=0)
    SetTransformParent(gameObject=var "Menu", parent=var "Self", worldPositionStays=true)
    ActivateGameObject(gameObject=var "Menu", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Inert {
  }

  state Destroy Map {
    DestroyObject(gameObject=var "Menu", delay=0, detachChildren=false)
    on FINISHED → Inert
  }

  var Menu: gameObject = <null>
  var Requester: gameObject = <null>
  var Self: gameObject = <null>
}
