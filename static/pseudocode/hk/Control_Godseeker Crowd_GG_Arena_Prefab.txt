// uses template: godseeker_crowd
fsm Control {
  start Init Target
  on LOOK UP → Delay  // from any state

  state Front {
    GetPosition(gameObject=var "Target", vector=(unset), x=var "Target X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Right X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"R", everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Left X", tolerance=0, equal=(none), lessThan=→"L", greaterThan=(none), everyFrame=true)
    on R → Delay 1
    on L → Delay 4
  }

  state Init {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Right X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Right X", add=4, everyFrame=false, perSecond=false)
    FloatOperator(float1=var "Right X", float2=8, operation=Subtract, storeResult=var "Left X", everyFrame=false)
    BoolTest(boolVariable=var "Start Looking Up", isTrue=→"LOOK UP", isFalse=(none), everyFrame=false)
    on FINISHED → Front
  }

  state LookR {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookR")
    GetPosition(gameObject=var "Target", vector=(unset), x=var "Target X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Right X", tolerance=0, equal=(none), lessThan=→"L", greaterThan=(none), everyFrame=true)
    on L → Delay 2
  }

  state Return R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ReturnR")
    GetPosition(gameObject=var "Target", vector=(unset), x=var "Target X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Right X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"R", everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Left X", tolerance=0, equal=(none), lessThan=→"L", greaterThan=(none), everyFrame=true)
    on R → Delay 1
    on L → Delay 4
  }

  state LookL {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookL")
    GetPosition(gameObject=var "Target", vector=(unset), x=var "Target X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Left X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"R", everyFrame=true)
    on R → Delay 3
  }

  state Return L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ReturnL")
    GetPosition(gameObject=var "Target", vector=(unset), x=var "Target X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Right X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"R", everyFrame=true)
    FloatCompare(float1=var "Target X", float2=var "Left X", tolerance=0, equal=(none), lessThan=→"L", greaterThan=(none), everyFrame=true)
    on R → Delay 1
    on L → Delay 4
  }

  state Delay 1 {
    WaitRandom(timeMin=0.05, timeMax=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → LookR
  }

  state Delay 2 {
    WaitRandom(timeMin=0.05, timeMax=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return R
  }

  state Delay 3 {
    WaitRandom(timeMin=0.05, timeMax=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return L
  }

  state Delay 4 {
    WaitRandom(timeMin=0.05, timeMax=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → LookL
  }

  state Look Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookUp")
    on RESET → Look Up Return
  }

  state Look Up Return {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="LookUpReturn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Front
  }

  state Init Target {
    AddEventRegister(target=Self, eventName="LOOK UP")
    GameObjectIsNull(gameObject=var "Target", isNull=(none), isNotNull=→"FINISHED", storeResult=(unset), everyFrame=false)
    GetHero(storeResult=var "Target")
    on FINISHED → Init
  }

  state Delay {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Look Up
  }

  var Start Looking Up: bool = false
  var Target: gameObject = <null>

  var Left X: float = 0
  var Right X: float = 0
  var Target X: float = 0
}
