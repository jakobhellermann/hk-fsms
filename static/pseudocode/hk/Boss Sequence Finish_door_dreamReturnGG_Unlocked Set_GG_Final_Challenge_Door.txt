# Boss Sequence Finish
# 1 location(s):
#   GG_Atrium_Roof / GG_Final_Challenge_Door/Unlocked Set/door_dreamReturnGG

fsm Boss Sequence Finish {
  start Pause

  state Take Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    NextFrameEvent(sendEvent=→"FINISHED")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=MaxHealth())
    SendEventByName(eventTarget=BroadcastAll, sendEvent="UPDATE BLUE HEALTH", delay=0, everyFrame=false)
    on FINISHED → Facing
  }

  state Return Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AcceptInput())
    StopParticleEmittersInChildren(gameObject=var "Behind Particles")
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=false, everyFrame=false)
    SetStaticVariable(variableName="finishedBossReturning", setValue=false)
    on FINISHED → Enable Inspect
  }

  state Facing {
    BoolTest(boolVariable=var "Hero Faces left", isTrue=(none), isFalse=(none), everyFrame=false)
    on L → Left
    on R → Right
  }

  state Left {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Fade In
  }

  state Right {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    on FINISHED → Fade In
  }

  state Pause {
    ActivateGameObject(gameObject=var "Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Door Entry
  }

  state Door Entry {
    GetName(gameObject=GG_Final_Challenge_Door/Unlocked Set/door_dreamReturnGG, storeName=var "Door Entry", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetEntryGateName", parameters=[], storeResult=var "Door Entered")
    StringCompare(stringVariable=var "Door Entry", compareTo=var "Door Entered", equalEvent=(none), notEqualEvent=→"INACTIVE", storeResult=(unset), everyFrame=false)
    GetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", storeValue=var "Dream Returning", everyFrame=false)
    BoolTest(boolVariable=var "Dream Returning", isTrue=→"WAKE", isFalse=(none), everyFrame=false)
    on INACTIVE → Not Returning
    on FINISHED → Door Open
    on WAKE → Wait for dream return
  }

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → TP In
  }

  state Not Returning {
    ActivateGameObject(gameObject=GG_Final_Challenge_Door/Unlocked Set/Inspect, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Door Open {
    SendEventByName(eventTarget=GameObject(var "MainCamera"), sendEvent="FADE OUT INSTANT", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Take Control
  }

  state Fade In {
    SendEventByName(eventTarget=GameObject(var "MainCamera"), sendEvent="FADE SCENE IN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on FINISHED → Wait
  }

  state Dream Return {
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=false, everyFrame=false)  // disabled
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="", setValue=1, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    on DREAM WAKE → Enable Inspect
  }

  state Enable Inspect {
    ActivateGameObject(gameObject=var "Inspect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Reset
  }

  state Wait for dream return {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dream Return
  }

  state Reset {
    SetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", value="GG_Waterways")
    SetPlayerDataString(gameObject=var "GameManager", stringName="bossReturnEntryGate", value="")
    GGResetBossSequenceController()
    on FINISHED → Save
  }

  state Save {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=TimePasses())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=SaveGame())
    on FINISHED → 
  }

  state TP In {
    SetMeshRenderer(gameObject=var "Hero", active=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="DG Warp In", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → TP In 2
  }

  state TP In 2 {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Super Hard Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Return Control
  }

  var Hero Faces left: bool = true

  var Dream Returning: bool = false
  var Door Entered: string = ""
  var Door Entry: string = ""
  var Behind Particles: gameObject = <null>
  var Inspect: gameObject = GG_Final_Challenge_Door/Unlocked Set/Inspect
}
