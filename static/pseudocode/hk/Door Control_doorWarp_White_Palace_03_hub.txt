# Door Control
# 1 location(s):
#   White_Palace_03_hub / doorWarp

fsm Door Control {
  start Init
  on SHINY ITEM ENTER → On Item  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    SetGameObject(variable=var "Main Camera", gameObject=var "MainCamera", everyFrame=false)
    FindChild(gameObject=var "Self", childName="Prompt Marker", storeResult=var "Prompt Marker")
    CreateObject(gameObject=Arrow Prompt (resources.assets), spawnPoint=var "Prompt Marker", position=(unset), rotation=(unset), storeObject=var "Prompt", networkInstantiate=false, networkGroup=0)
    SetFsmString(gameObject=var "Prompt", fsmName="Prompt Control", variableName="Prompt Name", setValue=var "Prompt Name", everyFrame=false)
    DestroyObject(gameObject=var "Prompt Marker", delay=0, detachChildren=false)
    FindChild(gameObject=Self, childName="Pt Warp", storeResult=var "Pt Warp")
    on FINISHED → Crossroads?
  }

  state Idle {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"IN RANGE", storeCollider=var "")
    on IN RANGE → In Range
  }

  state In Range {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="UP", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"OUT OF RANGE", storeCollider=var "")
    GetAxis(axisName="Vertical", multiplier=1, store=var "Axis Vert", everyFrame=true)  // disabled
    FloatCompare(float1=var "Axis Vert", float2=0.6, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UP PRESSED", everyFrame=true)  // disabled
    ListenForUp(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=false, stateEntryOnly=false)
    ListenForDown(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=false, stateEntryOnly=false)
    on OUT OF RANGE → Idle
    on UP PRESSED → Can Enter?
  }

  state Can Enter? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="CanInput", parameters=[], storeResult=var "Can Input")
    BoolTest(boolVariable=var "Can Input", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["onGround"], storeResult=var "On Ground")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["attacking"], storeResult=var "Attacking")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["upAttacking"], storeResult=var "Attacking Up")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["downAttacking"], storeResult=var "Attacking Down")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["dashing"], storeResult=var "Dashing")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["backDashing"], storeResult=var "Backdashing")
    BoolTest(boolVariable=var "On Ground", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    BoolAnyTrue(boolVariables=[var "Attacking", var "Attacking Down", var "Attacking Up", var "Backdashing", var "Dashing"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Set PD Bool
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Range
  }

  state Enter {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOOR ENTER", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Main Camera"), sendEvent="JUST FADE", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Change Scene
  }

  state Change Scene {
    SetPlayerDataInt(intName="currentArea", value=0)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=[var "New Scene", var "Entry Gate", var "Entry Pause"], storeResult=(unset var))  // disabled
    BeginSceneTransition(sceneName=var "New Scene", entryGateName=var "Entry Gate", entryDelay=var "Entry Pause", visualization=Default, preventCameraFadeOut=false)
  }

  state Audio Snapshots {
    TransitionToAudioSnapshot(snapshot=var "Atmos Snapshot", transitionTime=var "Audio Transition Time")
    TransitionToAudioSnapshot(snapshot=var "Enviro Snapshot", transitionTime=var "Audio Transition Time")
    TransitionToAudioSnapshot(snapshot=var "Music Snapshot", transitionTime=var "Audio Transition Time")
    TransitionToAudioSnapshot(snapshot=Off (resources.assets), transitionTime=0.3)  // disabled
    on FINISHED → Anim
  }

  state Crossroads? {
    BoolTest(boolVariable=var "Crossroads Ascent", isTrue=→"WAIT", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Normal Pause
    on WAIT → Visited Check
  }

  state Visited Check {
    GetPlayerDataBool(boolName="visitedCrossroads", storeValue=var "Visited Crossroads")
    BoolTest(boolVariable=var "Visited Crossroads", isTrue=→"VISITED", isFalse=→"UNVISITED", everyFrame=false)
    on VISITED → Visited
    on UNVISITED → Unvisited
  }

  state Visited {
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Unvisited {
    Wait(time=10, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Normal Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state On Item {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)
    on SHINY ITEM EXIT → Idle
  }

  state Anim {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 1")
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    SetPlayerDataBool(boolName="disablePause", value=true)
    on FINISHED → Poof
  }

  state Poof {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Hero", position=(0, -1, -2), rotation=(unset), storeObject=var "")
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Warp", emit=0)
    on FINISHED → Enter
  }

  state Set PD Bool {
    StringCompare(stringVariable=var "Set PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "Set PD Bool", value=true)
    on FINISHED → Audio Snapshots
  }

  var Audio Transition Time: float = 0.3
  var Crossroads Ascent: bool = false
  var Dream Return: bool = false
  var Over Hero: bool = false
  var Entry Gate: string = "door_dreamReturn_reality"
  var New Scene: string = "Abyss_05"
  var Prompt Name: string = "Exit"
  var Set PD Bool: string = "whitePalaceMidWarp"
  var Atmos Snapshot: object = <null>
  var Enviro Snapshot: object = <null>
  var Music Snapshot: object = <null>

  var Axis Vert: float = 0
  var Entry Pause: float = 0
  var Attacking: bool = false
  var Attacking Down: bool = false
  var Attacking Up: bool = false
  var Backdashing: bool = false
  var Can Input: bool = false
  var Dashing: bool = false
  var On Ground: bool = false
  var Visited Crossroads: bool = false
  var Hero: gameObject = <null>
  var Main Camera: gameObject = <null>
  var Prompt: gameObject = <null>
  var Prompt Marker: gameObject = <null>
  var Pt Warp: gameObject = <null>
  var Self: gameObject = <null>
}
