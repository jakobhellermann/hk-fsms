# Destroy
# 1 location(s):
#   Crossroads_11_alt / _Enemies/Battle Scene

fsm Destroy {
  start Pause

  state Detect {
    FindChild(gameObject=Self, childName="Blocker", storeResult=var "Blocker")
    FindChild(gameObject=Self, childName="CameraLockArea C", storeResult=var "CamLock C")
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "CamLock")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock C", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on KILLED → Activate
    on ACTIVATED → Activate
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Activate {
    ActivateGameObject(gameObject=var "CamLock C", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DestroyObject(gameObject=var "CamLock", delay=0, detachChildren=false)
    ActivateGameObject(gameObject=var "Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Blocker: gameObject = <null>
  var CamLock: gameObject = <null>
  var CamLock C: gameObject = <null>
  var Roller B: gameObject = <null>
}
