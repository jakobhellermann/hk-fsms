# Reactivate HUD
# 1 location(s):
#   GG_Atrium_Roof / Land of Storms Doors

fsm Reactivate HUD {
  start Wait

  state Wait {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Door Entry
  }

  state Door Entry {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetEntryGateName", parameters=[], storeResult=var "Door Entered")
    StringContains(stringVariable=var "Door Entered", containsString=var "Door Name Contains", trueEvent=(none), falseEvent=→"INACTIVE", storeResult=(unset), everyFrame=false)
    on INACTIVE → Inert
    on FINISHED → Show HUD
  }

  state Inert {
  }

  state Show HUD {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
  }

  var Door Name Contains: string = "door_Land_of_Storms"

  var Dream Returning: bool = false
  var Door Entered: string = ""
  var Door Entry: string = ""
}
