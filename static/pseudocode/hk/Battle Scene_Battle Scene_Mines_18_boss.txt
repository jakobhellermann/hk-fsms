# Battle Scene
# 1 location(s):
#   Mines_18_boss / Battle Scene

fsm Battle Scene {
  start Init

  state Init {
    FindGameObject(objectName="Dummy Bench", withTag="Untagged", store=var "Dummy Bench")
    FindGameObject(objectName="RestBench", withTag="RespawnPoint", store=var "Rest Bench")
    on FINISHED → Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Detect {
    ActivateGameObject(gameObject=var "Rest Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dummy Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedMegaBeamMiner", isTrue=→"ACTIVATED", isFalse=(none))
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on ACTIVATED → Activate
    on BATTLE END → Wait
  }

  state Activate {
    ActivateGameObject(gameObject=var "Rest Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dummy Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="StopRumble", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Open
  }

  state Open {
    ActivateGameObject(gameObject=var "Rest Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dummy Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    FindGameObject(objectName="Cam Lock", withTag="Untagged", store=var "Cam Lock")
    DestroyObject(gameObject=var "Cam Lock", delay=0, detachChildren=false)
  }

  var Activated: bool = false
  var Cam Lock: gameObject = <null>
  var Dummy Bench: gameObject = <null>
  var Rest Bench: gameObject = <null>
}
