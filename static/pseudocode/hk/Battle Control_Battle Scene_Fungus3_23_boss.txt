# Battle Control
# 1 location(s):
#   Fungus3_23_boss / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Completed", storeResult=var "Completed")
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Cloth Entry", storeResult=var "Cloth Entry")
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Wave 2", storeResult=var "Wave 2")
    FindChild(gameObject=Self, childName="Wave 3", storeResult=var "Wave 3")
    FindChild(gameObject=Self, childName="Wave 4", storeResult=var "Wave 4")
    FindChild(gameObject=Self, childName="Floor Saver", storeResult=var "Floor Saver")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on ACTIVATE → Activated
    on HIT → Pause 3
  }

  state Activated {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Completed", activate=true)
    DestroyObject(gameObject=var "Wave 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 2", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 3", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 4", delay=0, detachChildren=false)
  }

  state Wave 1 {
    TransitionToAudioSnapshot(snapshot=Action (resources.assets), transitionTime=1)
    ActivateAllChildren(gameObject=var "Wave 1", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=1, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    on END → Pause 1
  }

  state Wave 2 {
    ActivateAllChildren(gameObject=var "Wave 2", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPAWN", delay=0, everyFrame=false)
    on END → Pause 2
  }

  state Wave 3 {
    ActivateAllChildren(gameObject=var "Wave 3", activate=true)
    on BOSS DEATH → End Pause
  }

  state End Pause {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="savedCloth", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="clothKilled", value=true)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Saver", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=6)
  }

  state Pause 1 {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Cloth Entry?
  }

  state Pause 2 {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 3
  }

  state Cloth Entry? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="savedCloth", isTrue=→"ENTER", isFalse=→"FINISHED")
    on FINISHED → Wave 2
    on ENTER → Cloth Activate
  }

  state Cloth Activate {
    ActivateGameObject(gameObject=var "Cloth Entry", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on CLOTH READY → Wave 2
  }

  state Pause 3 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Saver", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wave 1
  }

  var Battle Enemies: int = 0
  var Activated: bool = false
  var CamLock: gameObject = <null>
  var Cloth Entry: gameObject = <null>
  var Completed: gameObject = <null>
  var Floor Saver: gameObject = <null>
  var Wave 1: gameObject = <null>
  var Wave 2: gameObject = <null>
  var Wave 3: gameObject = <null>
  var Wave 4: gameObject = <null>
}
