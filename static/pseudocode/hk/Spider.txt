fsm Spider {
  start Init
  on MOVESTOP → Stop  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Ground Detect", storeResult=var "Ground Detect Obj")
    FindChild(gameObject=Self, childName="Wall Detect", storeResult=var "Wall Detect Obj")
    on FINISHED → Detect Angle
  }

  state Detect Angle {
    FloatCompare(float1=var "Angle", float2=0, tolerance=1, equal=→"FLOOR", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Angle", float2=90, tolerance=1, equal=→"WALL RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Angle", float2=180, tolerance=1, equal=→"ROOF", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Angle", float2=270, tolerance=1, equal=→"WALL LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FLOOR → Floor
    on ROOF → Roof
    on WALL LEFT → Wall Left
    on WALL RIGHT → Wall Right
  }

  state Roof {
    SetFloatValue(floatVariable=var "xSpeed", floatValue=5, everyFrame=false)
    SetFloatValue(floatVariable=var "ySpeed", floatValue=0, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Shoot", variableName="Orientation", setValue=2, everyFrame=false)
    on FINISHED → Sleep
  }

  state Floor {
    SetFloatValue(floatVariable=var "xSpeed", floatValue=-5, everyFrame=false)
    SetFloatValue(floatVariable=var "ySpeed", floatValue=0, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Shoot", variableName="Orientation", setValue=0, everyFrame=false)
    on FINISHED → Sleep
  }

  state Wall Left {
    SetFloatValue(floatVariable=var "xSpeed", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "ySpeed", floatValue=5, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Shoot", variableName="Orientation", setValue=1, everyFrame=false)
    on FINISHED → Sleep
  }

  state Wall Right {
    SetFloatValue(floatVariable=var "xSpeed", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "ySpeed", floatValue=-5, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Shoot", variableName="Orientation", setValue=3, everyFrame=false)
    on FINISHED → Sleep
  }

  state Walk {
    AudioPlayInState(gameObject=Self, volume=1)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "xSpeed", y=var "ySpeed", everyFrame=true)
    WaitRandom(timeMin=1, timeMax=4, finishEvent=→"CHANCE", realTime=false)
    SetBoolValue(boolVariable=var "Walking", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "GROUND DETECT", boolValue=true, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Ground Detect Obj", fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "GROUND DETECT", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=3, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "GROUND DETECT", isTrue=(none), isFalse=→"FLIP DIRECTION", everyFrame=true)
    ConstrainMovement(gameObject=Self, xConstrain=0.5, yConstrain=0.5)
    SetBoolValue(boolVariable=var "WALL DETECT", boolValue=false, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Wall Detect Obj", fromPosition=(unset), direction=(-1, 0), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "WALL DETECT", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=3, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "WALL DETECT", isTrue=→"FLIP DIRECTION", isFalse=(none), everyFrame=true)
    on FLIP DIRECTION → Turn Around
    on CHANCE → Chance To Flip
  }

  state Turn Around {
    SetBoolValue(boolVariable=var "Walking", boolValue=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FloatMultiplyV2(floatVariable=var "ySpeed", multiplyBy=-1, everyFrame=false, fixedUpdate=false)
    FloatMultiplyV2(floatVariable=var "xSpeed", multiplyBy=-1, everyFrame=false, fixedUpdate=false)
    on FINISHED → Flip Scale
  }

  state Choose Move {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=30, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SLEEP", everyFrame=false)
    SendRandomEvent(events=[→"RUN", →"RETREAT"], weights=[1, 1], delay=0)
    on RUN → Walk Start
    on RETREAT → Set Speed
    on SLEEP → Sleep
  }

  state Chance To Flip {
    SetBoolValue(boolVariable=var "Walking", boolValue=false, everyFrame=false)
    SendRandomEvent(events=[→"FLIP DIRECTION", →"RUN"], weights=[1, 1], delay=0)
    on FLIP DIRECTION → Turn Around
    on RUN → Choose Move
  }

  state Flip Scale {
    FlipScale(gameObject=var "Self", flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Choose Move
  }

  state Walk Start {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Run", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "xSpeed", y=var "ySpeed", everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Walk
  }

  state Retreat2 {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "RxSpeed", y=var "RySpeed", everyFrame=true)
    AudioPlayInState(gameObject=Self, volume=1)
    WaitRandom(timeMin=0.5, timeMax=1.5, finishEvent=→"CHANCE", realTime=false)
    SetBoolValue(boolVariable=var "Walking", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "GROUND DETECT", boolValue=true, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Ground Detect Obj", fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "GROUND DETECT", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=3, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "GROUND DETECT", isTrue=(none), isFalse=→"FLIP DIRECTION", everyFrame=true)
    ConstrainMovement(gameObject=Self, xConstrain=0.5, yConstrain=0.5)
    SetBoolValue(boolVariable=var "WALL DETECT", boolValue=false, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Wall Detect Obj", fromPosition=(unset), direction=(-1, 0), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "WALL DETECT", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=3, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "WALL DETECT", isTrue=→"FLIP DIRECTION", isFalse=(none), everyFrame=true)
    on FLIP DIRECTION → Turn Around
    on CHANCE → Chance To Flip
  }

  state Set Speed {
    SetFloatValue(floatVariable=var "RxSpeed", floatValue=var "xSpeed", everyFrame=false)
    SetFloatValue(floatVariable=var "RySpeed", floatValue=var "ySpeed", everyFrame=false)
    FloatMultiplyV2(floatVariable=var "RySpeed", multiplyBy=1.2, everyFrame=false, fixedUpdate=false)
    FloatMultiplyV2(floatVariable=var "RxSpeed", multiplyBy=1.2, everyFrame=false, fixedUpdate=false)
    on FINISHED → Retreat Start
  }

  state Retreat Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Retreat")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "RxSpeed", y=var "RySpeed", everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Retreat2
  }

  state Stop {
    on MOVESTART → Choose Move
  }

  state Sleep {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=30, tolerance=0, equal=(none), lessThan=→"WAKE", greaterThan=(none), everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on WAKE → Choose Move
  }

  var Angle: float = 0
  var Distance: float = 0
  var RxSpeed: float = 0
  var RySpeed: float = 0
  var xSpeed: float = 0
  var ySpeed: float = 0
  var GROUND DETECT: bool = false
  var WALL DETECT: bool = false
  var Walking: bool = false
  var Ground Detect Obj: gameObject = <null>
  var Self: gameObject = <null>
  var Wall Detect Obj: gameObject = <null>
}
