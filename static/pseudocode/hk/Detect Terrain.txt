fsm Detect Terrain {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"EXIT", storeCollider=var "")
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=true, everyFrame=false)
    on FINISHED → Detect
  }

  state Exit {
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=false, everyFrame=false)
    on FINISHED → Detect
  }

  var Bool Name: string = "Wall Range"
  var FSM Name: string = "Acid Walker"

  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
