# Heart Container Control
# 2 location(s):
#   Room_Mansion / Heart Piece Folder/Heart Piece
#   Crossroads_09 / Heart Piece

fsm Heart Container Control {
  start Pause

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    FindChild(gameObject=Self, childName="Sprite 2", storeResult=var "Sprite 2")
    FindChild(gameObject=Self, childName="Get Cloud", storeResult=var "Get Cloud")
    FindChild(gameObject=Self, childName="Sprite 1", storeResult=var "Sprite 1")
    FindChild(gameObject=Self, childName="Get Anim", storeResult=var "Get Anim")
    FindChild(gameObject=Self, childName="Sprite 3", storeResult=var "Sprite 3")
    FindChild(gameObject=Self, childName="Idle Particles", storeResult=var "Idle Particles")
    FindChild(gameObject=Self, childName="Appear Cloud", storeResult=var "Appear Cloud")
    FindChild(gameObject=Self, childName="Appear Trail", storeResult=var "Appear Trail")
    FindChild(gameObject=Self, childName="Plink", storeResult=var "Plink")
    IntSwitch(intVariable=var "Sprite", compareTo=[1, 2, 3], sendEvent=[→"ONE", →"TWO", →"THREE"], everyFrame=false)
    on ONE → Sprite 1
    on TWO → Sprite 2
    on THREE → Sprite 3
    on ACTIVATED → Activated
  }

  state Sprite 1 {
    ActivateGameObject(gameObject=var "Sprite 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Appear Check
  }

  state Sprite 2 {
    ActivateGameObject(gameObject=var "Sprite 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Appear Check
  }

  state Sprite 3 {
    ActivateGameObject(gameObject=var "Sprite 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Appear Check
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=(unset), sendEvent=→"GET", storeCollider=var "")
    on GET → Pickup Disabled?
  }

  state Get {
    AudioStop(gameObject=Self)
    SetPlayerDataBool(boolName="isInvincible", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=heartpiece_collect (sharedassets10.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataBool(boolName="heartPieceCollected", value=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Heart Piece")
    SetGravity2dScale(gameObject=var "Hero", gravityScale=0)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="GET", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Idle Particles")
    ActivateGameObject(gameObject=var "Sprite 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Get Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SpawnRandomObjects(gameObject=Heart Piece Orb (sharedassets10.assets), spawnPoint=var "Self", position=(unset), spawnMin=20, spawnMax=20, speedMin=18, speedMax=25, angleMin=0, angleMax=360, originVariation=1)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → UI
  }

  state Activated {
    DestroySelf(detachChildren=false)
  }

  state UI {
    GetPlayerDataInt(gameObject=var "GameManager", intName="heartPieces", storeValue=var "Pieces")
    IntAdd(intVariable=var "Pieces", add=1, everyFrame=false)
    SetPlayerDataInt(intName="heartPieces", value=var "Pieces")
    CreateObject(gameObject=Heart Container UI (sharedassets10.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "UI", networkInstantiate=false, networkGroup=0)
    SetFsmInt(gameObject=var "UI", fsmName="", variableName="Pieces", setValue=var "Pieces", everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Appear Check {
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    BoolTest(boolVariable=var "Appears", isTrue=→"APPEAR", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on APPEAR → Appear Cloud
  }

  state Appear Cloud {
    GetPosition(gameObject=var "Appear Source", vector=var "Appear Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[heartpiece_collect (sharedassets10.assets)], weights=[1], pitchMin=1.25, pitchMax=1.25, volume=1, delay=2, storePlayer=<null>)
    SetPosition(gameObject=var "Appear Cloud", vector=var "Appear Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Appear Trail", vector=var "Appear Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Plink", vector=var "Appear Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Plink", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="STOP", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Idle Particles")
    SetSpriteRenderer(gameObject=var "Sprite 1", active=false)
    SetSpriteRenderer(gameObject=var "Sprite 2", active=false)
    SetSpriteRenderer(gameObject=var "Sprite 3", active=false)
    PlayParticleEmitter(gameObject=var "Appear Cloud", emit=0)
    iTweenScaleTo(gameObject=var "Appear Cloud", id="", transformScale=var "", vectorScale=(0.1, 0.1, 0.1), time=1.5, delay=1, speed=(unset), easeType=easeInSine, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Trail
  }

  state Trail {
    StopParticleEmitter(gameObject=var "Appear Cloud")
    PlayParticleEmitter(gameObject=var "Appear Trail", emit=0)
    iTweenMoveTo(gameObject=var "Appear Trail", id="", transformPosition=var "Self", vectorPosition=(unset), time=(unset), delay=0, speed=12, space=0, easeType=linear, loopType=none, orientToPath=true, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=true, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Appear
  }

  state Appear {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[new_heartpiece_puzzle_bit (resources.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Get Cloud", emit=0)
    PlayParticleEmitter(gameObject=var "Idle Particles", emit=0)
    StopParticleEmitter(gameObject=var "Appear Trail")
    SetSpriteRenderer(gameObject=var "Sprite 1", active=true)
    SetSpriteRenderer(gameObject=var "Sprite 2", active=true)
    SetSpriteRenderer(gameObject=var "Sprite 3", active=true)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="START", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Idle Particles", emit=0)
    CreateObject(gameObject=White Wave Default (sharedassets22.assets), spawnPoint=var "Self", position=(0, 0, 0.5), rotation=(unset), storeObject=var "Wave", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Wave", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=var "Wave", color=(1, 1, 1, 0.74509805), everyframe=false)
    NextFrameEvent(sendEvent=→"APPEAR")
    on APPEAR → Appear Pause
  }

  state Appear Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Pickup Disabled? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetCState", parameters=["superDashing"], storeResult=var "Is Superdashing")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetCState", parameters=["recoiling"], storeResult=var "IsRecoiling")
    BoolTest(boolVariable=var "Is Superdashing", isTrue=→"TRUE", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "IsRecoiling", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Idle
    on FALSE → Get
    on FINISHED → Get
  }

  var Sprite: int = 1
  var Appears: bool = true

  var Pieces: int = 0
  var Activated: bool = false
  var Is Superdashing: bool = false
  var IsRecoiling: bool = false
  var Appear Pos: vector3 = (0, 0, 0)
  var Appear Cloud: gameObject = <null>
  var Appear Source: gameObject = <null>
  var Appear Trail: gameObject = <null>
  var Camera: gameObject = <null>
  var Get Anim: gameObject = <null>
  var Get Cloud: gameObject = <null>
  var Hero: gameObject = <null>
  var Idle Particles: gameObject = <null>
  var Plink: gameObject = <null>
  var Self: gameObject = <null>
  var Sprite 1: gameObject = <null>
  var Sprite 2: gameObject = <null>
  var Sprite 3: gameObject = <null>
  var UI: gameObject = <null>
  var Wave: gameObject = <null>
}
