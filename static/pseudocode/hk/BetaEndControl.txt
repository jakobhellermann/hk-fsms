# BetaEndControl
# 1 location(s):
#   Crossroads_11_alt / _Areas/BetaEnder

fsm BetaEndControl {
  start Wait For Trigger

  state Wait For Trigger {
    FindGameObject(objectName="", withTag="Beta End", store=var "BetaEndSprite")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"LEAVE_BETA", storeCollider=var "Hero")
    on LEAVE_BETA → Beta Leave
  }

  state Beta Leave {
    SetPlayerDataBool(boolName="disablePause", value=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Run", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=StopAnimationControl())
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=-8.3, y=(unset), everyFrame=true)
    Wait(time=1.25, finishEvent=→"HOLD", realTime=false)
    on HOLD → Fade In
  }

  state Fade In {
    SendMessage(gameObject=var "BetaEndSprite", delivery=SendMessage, options=0, functionCall=FadeIn())
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=(unset), everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=AffectedByGravity(true))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=FaceRight())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=SetBackOnGround())
    Wait(time=1, finishEvent=→"HOLD_DONE", realTime=false)
    on HOLD_DONE → Wait For Return
  }

  state Wait For Return {
    AnyKey(sendEvent=→"RETURN_BETA")
    on RETURN_BETA → Fade Out
  }

  state Beta Return {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=StartAnimationControl())
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=(unset), everyFrame=true)
    SetPlayerDataBool(boolName="disablePause", value=false)
    Wait(time=0.25, finishEvent=→"HOLD_DONE", realTime=false)
    on HOLD_DONE → Wait For Trigger
  }

  state Fade Out {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Run", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendMessage(gameObject=var "BetaEndSprite", delivery=SendMessage, options=0, functionCall=FadeOut())
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=8.3, y=(unset), everyFrame=true)
    Wait(time=1.4, finishEvent=→"ENTER_DONE", realTime=false)
    on ENTER_DONE → Beta Return
  }

  var BetaEndSprite: gameObject = <null>
  var Hero: gameObject = <null>
}
