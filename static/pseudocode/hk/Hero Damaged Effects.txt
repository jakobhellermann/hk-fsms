# Hero Damaged Effects
# 2 location(s):
#   Menu_Title / _GameManager/AudioManager
#   resources.assets / _GameManager/AudioManager

fsm Hero Damaged Effects {
  start Idle
  on DAMAGE LOWPASS → To Lowpass  // from any state

  state Idle {
  }

  state To Lowpass {
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Damaged (sharedassets0.assets), transitionTime=0.1)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Back
    on CANCEL → Idle
  }

  state Fade Back {
    TransitionToAudioSnapshot(snapshot=Normal (sharedassets0.assets), transitionTime=2)
  }

  var HP: int = 0
}
