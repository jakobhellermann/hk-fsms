// uses template: spider_spawn
fsm Spider Spawn {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Can Spawn", isTrue=(none), isFalse=→"NO SPAWN", everyFrame=false)
    BoolTest(boolVariable=var "Always Spawn", isTrue=→"SPAWN", isFalse=(none), everyFrame=false)
    SendRandomEvent(events=[→"NO SPAWN", →"SPAWN"], weights=[0.8, 0.2], delay=0)
    on NO SPAWN → No Spawn
    on SPAWN → Spawn Init
  }

  state No Spawn {
  }

  state Spawn Init {
    FindChild(gameObject=Self, childName="Spiders", storeResult=var "Spiders")
    CreateObject(gameObject=Tiny Spider:1, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Spider", networkInstantiate=false, networkGroup=0)
    SetFsmBool(gameObject=var "Spider", fsmName="Spawn", variableName="Spawns", setValue=true, everyFrame=false)
    SetParent(gameObject=var "Spider", parent=var "Spiders", resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Spider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on BREAK → Hanging?
  }

  state Hanging? {
    ActivateAllChildren(gameObject=var "Spiders", activate=true)
    BoolTest(boolVariable=var "Hanging", isTrue=→"HANGING", isFalse=→"FINISHED", everyFrame=false)
    on HANGING → Drop
    on FINISHED → Spit
  }

  state Drop {
    FlingObjects(containerObject=var "Spiders", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=5, speedMax=10, angleMin=240, angleMax=300)
  }

  state Spit {
    FlingObjects(containerObject=var "Spiders", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=10, speedMax=15, angleMin=40, angleMax=140)
  }

  var Always Spawn: bool = true
  var Can Spawn: bool = false
  var Hanging: bool = false

  var Self: gameObject = <null>
  var Spider: gameObject = <null>
  var Spiders: gameObject = <null>
}
