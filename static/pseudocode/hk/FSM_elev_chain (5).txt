// uses template: mask_if_highpos
fsm FSM {
  start Init

  state Init {
    on FINISHED → Visible
  }

  state Visible {
    SetSpriteRenderer(gameObject=Self, active=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Mask Y", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"HIDE", everyFrame=true)
    on HIDE → Invisible
  }

  state Invisible {
    SetSpriteRenderer(gameObject=Self, active=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Mask Y", tolerance=0, equal=→"UNHIDE", lessThan=→"UNHIDE", greaterThan=(none), everyFrame=true)
    on UNHIDE → Visible
  }

  var Mask Y: float = 53

  var Self Y: float = 0
}
