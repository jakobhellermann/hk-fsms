// component disabled: this FSM does not run until something enables it
fsm Invincibility Control {
  start Init

  state Init {
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"TURN LEFT", negativeEvent=→"TURN RIGHT", space=0)
    on TURN LEFT → Left
    on TURN RIGHT → Right
  }

  state Left {
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=11)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=(none), lessThan=→"TURN RIGHT", greaterThan=(none), everyFrame=true)
    on TURN RIGHT → Right
  }

  state Right {
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=10)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"TURN LEFT", everyFrame=true)
    on TURN LEFT → Left
  }

  var X Scale: float = 0
}
