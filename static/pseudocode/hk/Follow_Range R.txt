# Follow
# 4 location(s):
#   GG_Lurker / Lurker Control/Pale Lurker/Range R
#   GG_Lurker / Lurker Control/Pale Lurker/Range L
#   GG_Sly / Battle Scene/Sly Boss/Spin Tink
#   GG_Sly / Battle Scene/Sly Boss/Cyclone Tink

// uses template: deparent_and_follow
fsm Follow {
  start Pause Frame

  state Follow {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetPosition(gameObject=var "Parent", vector=var "Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=var "Y Scale", zScale=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=var "Y Scale", z=(unset), everyFrame=true, lateUpdate=false)
  }

  state Pause Frame {
    BoolTest(boolVariable=var "Pause Frame", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Follow 2
  }

  state Follow 2 {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetPosition(gameObject=var "Parent", vector=var "Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
  }

  var Pause Frame: bool = true

  var X Scale: float = 0
  var Y Scale: float = 0
  var Pos: vector3 = (0, 0, 0)
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
