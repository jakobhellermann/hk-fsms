fsm Acid Walker {
  start Init

  state Init {
    SetFloatValue(floatVariable=var "Walk Speed Crt", floatValue=var "Walk Speed", everyFrame=false)
    on FINISHED → Check Dir
  }

  state Walk {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    BoolTest(boolVariable=var "Enemy Range", isTrue=→"TURN", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Wall Range", isTrue=→"TURN", isFalse=(none), everyFrame=true)
    on TURN → Turn
    on BLOCKED DOWN → Blocked Down
  }

  state Check Dir {
    SendRandomEvent(events=[→"LEFT", →"RIGHT"], weights=[1, 1], delay=0)
    on LEFT → Left
    on RIGHT → Right
  }

  state Left {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatMultiply(floatVariable=var "Walk Speed Crt", multiplyBy=-1, everyFrame=false)
    on FINISHED → Start Walk
  }

  state Right {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Start Walk
  }

  state Turn {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Flip
  }

  state Flip {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    FloatMultiply(floatVariable=var "Walk Speed Crt", multiplyBy=-1, everyFrame=false)
    on FINISHED → Start Walk
  }

  state Start Walk {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Walk Speed Crt", y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Walk
    on BLOCKED DOWN → Blocked Down
  }

  state Blocked Down {
    AudioStop(gameObject=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BounceToWalk")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Walk
  }

  var Walk Speed: float = 3.5

  var Walk Speed Crt: float = 0
  var Enemy Range: bool = false
  var Wall Range: bool = false
}
