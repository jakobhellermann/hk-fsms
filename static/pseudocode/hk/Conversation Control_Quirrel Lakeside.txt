# Conversation Control
# 1 location(s):
#   Crossroads_50 / Quirrel Lakeside

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Sit Region", storeResult=var "Sit Region")
    on FINISHED → Idle
  }

  state End {
    ActivateGameObject(gameObject=var "Sit Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="QUIRREL", everyFrame=false)
    on FINISHED → Talk
  }

  state Anim End {
    on FINISHED → Box Down
  }

  state Talk Finish {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["QUIRREL_EPILOGUE"], storeResult=(unset var))
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Talk {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Quirrel_blue_lake_03 (sharedassets81.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataBool(boolName="quirrelEpilogueCompleted", value=true)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_EPILOGUE_A", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk 2
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lake Talk")
    on FINISHED → Box Up
  }

  state Convo Choice {
  }

  state Talk 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_EPILOGUE_B", "Quirrel"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Quirrel_blue_lake_02 (sharedassets81.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lake Talk End")
    on CONVO_FINISH → Talk Finish
  }

  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Sit Region: gameObject = <null>
  var Voice Player: gameObject = <null>
}
