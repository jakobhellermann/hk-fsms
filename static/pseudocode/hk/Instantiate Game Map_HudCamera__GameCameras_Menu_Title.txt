# Instantiate Game Map
# 1 location(s):
#   Menu_Title / _GameCameras/HudCamera

fsm Instantiate Game Map {
  start Pause

  state Init Setup {
    SendMessage(gameObject=var "Map", delivery=SendMessage, options=1, functionCall=SetupMap(false))
    SendMessage(gameObject=var "Map", delivery=SendMessage, options=1, functionCall=GetDoorValues(false))
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="SetGameMap", parameters=[var "Map"], storeResult=(unset var))
    CallMethodProper(gameObject=var "Map", behaviour="GameMap", methodName="PositionDreamGateMarker", parameters=[], storeResult=(unset var))  // disabled
    on FINISHED → Set Pan Arrows
  }

  state Instantiate {
    GetOwner(storeGameObject=var "Self")
    CreateObject(gameObject=Game_Map (resources.assets), spawnPoint=var "Self", position=(0, 0, 38.1), rotation=(unset), storeObject=var "Map", networkInstantiate=false, networkGroup=0)
    SetParent(gameObject=var "Map", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetGameObject(variable=var "Game Map", gameObject=var "Map", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init Setup
  }

  state Set Pan Arrows {
    FindChild(gameObject=Self, childName="Inventory", storeResult=var "Inventory")
    FindChild(gameObject=var "Inventory", childName="Map", storeResult=var "HUD Map")
    FindChild(gameObject=var "HUD Map", childName="World Map", storeResult=var "World Map")
    FindChild(gameObject=var "World Map", childName="Pan Arrows", storeResult=var "Pan Arrows")
    FindChild(gameObject=var "Pan Arrows", childName="Arrow U", storeResult=var "Arrow U")
    FindChild(gameObject=var "Pan Arrows", childName="Arrow D", storeResult=var "Arrow D")
    FindChild(gameObject=var "Pan Arrows", childName="Arrow L", storeResult=var "Arrow L")
    FindChild(gameObject=var "Pan Arrows", childName="Arrow R", storeResult=var "Arrow R")
    CallMethodProper(gameObject=var "Game Map", behaviour="GameMap", methodName="SetPanArrows", parameters=[var "Arrow U", var "Arrow D", var "Arrow L", var "Arrow R"], storeResult=(unset var))
    on FINISHED → Check Map Key Pref
  }

  state Key And Pin {
    SetCameraCullingMask(gameObject=Self, cullingMask=[UI(5), uGUI(27), Map Pin(30)], invertMask=false, everyFrame=false)
  }

  state Check Map Key Pref {
    GetPlayerDataInt(gameObject=var "", intName="mapKeyPref", storeValue=var "Map Key Pref")
    IntSwitch(intVariable=var "Map Key Pref", compareTo=[0, 1, 2], sendEvent=[→"KEY AND PIN", →"PIN", →"NONE"], everyFrame=false)
    on KEY AND PIN → Key And Pin
    on PIN → Pin
    on NONE → None
  }

  state Pin {
    SetCameraCullingMask(gameObject=Self, cullingMask=[UI(5), uGUI(27), Map Pin(30)], invertMask=false, everyFrame=false)
  }

  state None {
    SetCameraCullingMask(gameObject=Self, cullingMask=[UI(5), uGUI(27)], invertMask=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Instantiate
  }

  var Map Key Pref: int = 0
  var Str Hide Key: string = ""
  var Str Hide Pins: string = ""
  var Str Show Both: string = ""
  var Action: gameObject = _GameCameras/HudCamera/Inventory/Map Key/Action
  var Action Text: gameObject = _GameCameras/HudCamera/Inventory/Map Key/Action/Text
  var Arrow D: gameObject = <null>
  var Arrow L: gameObject = <null>
  var Arrow R: gameObject = <null>
  var Arrow U: gameObject = <null>
  var Backboard Key: gameObject = _GameCameras/HudCamera/Inventory/Map Key/Backboard Key
  var HUD Map: gameObject = <null>
  var Inventory: gameObject = <null>
  var Keys: gameObject = _GameCameras/HudCamera/Inventory/Map Key/Keys
  var Map: gameObject = <null>
  var Pan Arrows: gameObject = <null>
  var Self: gameObject = <null>
  var World Map: gameObject = <null>
}
