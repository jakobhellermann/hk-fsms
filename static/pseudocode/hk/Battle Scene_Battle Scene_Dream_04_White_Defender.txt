  start Pause

  state Init {
    FindChild(gameObject=Self, childName="CamLocks", storeResult=var "CamLocks")
    FindChild(gameObject=Self, childName="Wall Saver", storeResult=var "Wall Saver")
    on FINISHED → Idle
  }

  state Pause {
    on FINISHED → Init
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Battle Start
  }

  state Battle Start {
    ActivateGameObject(gameObject=var "CamLocks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall Saver", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)

    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Dung Start
  }

  state Dung Start {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DUNG START", delay=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    on BATTLE END → End
  }

  state End {
    ActivateGameObject(gameObject=var "CamLocks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall Saver", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var CamLocks: gameObject = <null>
  var Self: gameObject = White Defender:0/Slam Pillars/Dung Pillar (6)/Chunks (sharedassets398.assets)

}
