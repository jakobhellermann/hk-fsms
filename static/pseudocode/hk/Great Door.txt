# Great Door
# 1 location(s):
#   Tutorial_01 / _Props/Hallownest_Main_Gate/Door

fsm Great Door {
  start Pause

  state Idle {
    ReceivedDamage(collideTag="", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on HIT → Check If Nail
  }

  state Check If Nail {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0], sendEvent=[→"TRUE"], everyFrame=false)
    on TRUE → Hit
    on FINISHED → Idle
  }

  state Hit {
    CreateObject(gameObject=Door_dust (sharedassets6.assets), spawnPoint=var "Self", position=(-0.5, 0, -1), rotation=(1, 0, -180), storeObject=var "", networkInstantiate=false, networkGroup=0)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Damager", position=(1, 0, 0), rotation=(unset), storeObject=var "Strike Effect", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Strike Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=0.85, max=1.1, storeResult=var "Chooser")
    SetAudioPitch(gameObject=var "Self", pitch=var "Chooser", everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15)
    CreateObject(gameObject=Dust Hit Med (sharedassets6.assets), spawnPoint=var "Damager", position=(1, 0, 0), rotation=(180, 90, 270), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Damager", position=(1, 0, 0), spawnMin=3, spawnMax=5, speedMin=15, speedMax=20, angleMin=100, angleMax=140, originVariation=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnRandomObjectsV2(gameObject=Particle Rock Small Transient (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=12, speedMin=15, speedMax=25, angleMin=100, angleMax=140, originVariationX=0.5, originVariationY=4.5)
    on FINISHED → Check Hits
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "")
    SetFloatValue(floatVariable=var "Recoil Speed Neg", floatValue=var "Recoil Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Recoil Speed Neg", multiplyBy=-1, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Check Hits {
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    IntSwitch(intVariable=var "Hits", compareTo=[4, 8, 13], sendEvent=[→"BREAK1", →"BREAK2", →"BREAK3"], everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on BREAK1 → Break 1
    on BREAK2 → Break 2
    on BREAK3 → Break
    on FINISHED → Idle
  }

  state Break {
    FindChild(gameObject=var "GameManager", childName="Atmos", storeResult=var "Atmos Folder")
    FindChild(gameObject=var "Atmos Folder", childName="Atmos Cave Wind", storeResult=var "Atmos Cave Wind")
    SendEventByName(eventTarget=GameObjectFSM(var "Atmos Cave Wind", fsm="Atmos Control"), sendEvent="STOP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "MainCamera"), sendEvent="FADE OUT INSTANT", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioPlay(gameObject=var "Self", volume=1, oneShotClip=breakable_wall_death (sharedassets6.assets), finishedEvent=(none))
    NextFrameEvent(sendEvent=→"RIGHT")
    on RIGHT → Extra Frame
  }

  state Break 1 {
    Tk2dSpriteSetId(gameObject=Self, spriteID=(unset), ORSpriteName="door_v02", spriteCollection=var "")
    CreateObject(gameObject=Rubble Fall Short (sharedassets6.assets), spawnPoint=var "", position=(185, 75, 0.1), rotation=(90, -180, 0), storeObject=var "", networkInstantiate=false, networkGroup=0)
    AudioPlay(gameObject=var "Self", volume=1, oneShotClip=breakable_wall_death (sharedassets6.assets), finishedEvent=(none))
    SpawnRandomObjectsV2(gameObject=Particle Rock Small Transient (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=12, spawnMax=12, speedMin=25, speedMax=30, angleMin=100, angleMax=140, originVariationX=0.5, originVariationY=4.5)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Break 2 {
    Tk2dSpriteSetId(gameObject=Self, spriteID=(unset), ORSpriteName="door_v03", spriteCollection=var "")
    CreateObject(gameObject=Rubble Fall Short (sharedassets6.assets), spawnPoint=var "", position=(185, 75, 0.1), rotation=(90, -180, 0), storeObject=var "", networkInstantiate=false, networkGroup=0)
    AudioPlay(gameObject=var "Self", volume=1, oneShotClip=breakable_wall_death (sharedassets6.assets), finishedEvent=(none))
    SpawnRandomObjectsV2(gameObject=Tute Door Piece A (sharedassets6.assets), spawnPoint=var "Self", position=(-1.2, 5.3, -0.1), spawnMin=1, spawnMax=1, speedMin=12, speedMax=12, angleMin=180, angleMax=180, originVariationX=0, originVariationY=0)
    SpawnRandomObjectsV2(gameObject=Tute Door Piece B (sharedassets6.assets), spawnPoint=var "Self", position=(-1.2, 4.65, -0.1), spawnMin=1, spawnMax=1, speedMin=8, speedMax=8, angleMin=180, angleMax=180, originVariationX=0, originVariationY=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state Activate {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Move {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=EnterWithoutInput(false))
    BeginSceneTransition(sceneName="Town", entryGateName="left1", entryDelay=2.5, visualization=Default, preventCameraFadeOut=false, 0, 0, 0, 0, 0, 0, visualization=Default, preventCameraFadeOut=false)
  }

  state Extra Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=0)
    on FINISHED → Move
  }

  var Hits: int = 0

  var Attack Direction: float = 0
  var Chooser: float = 0
  var Recoil Speed: float = 1
  var Recoil Speed Neg: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Attack Type: int = 0
  var Activated: bool = false
  var Atmos Cave Wind: gameObject = <null>
  var Atmos Folder: gameObject = <null>
  var Damager: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Strike Effect: gameObject = <null>
}
