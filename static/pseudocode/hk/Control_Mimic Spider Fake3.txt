# Control
# 1 location(s):
#   Deepnest_31 / Mimic Spider Fake3

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Detect {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ACTIVATE → Activate
    on ENTER → Run
  }

  state Activate {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Run {
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-10, y=(unset), everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Drop X", tolerance=0, equal=→"DROP", lessThan=→"DROP", greaterThan=(none), everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Hero Disappear X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"DISAPPEAR", everyFrame=true)
    on DROP → Drop
    on DISAPPEAR → Disappear
  }

  state Drop {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=Self, frame=6)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=-12, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Disappear
  }

  state Disappear {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var Drop X: float = 70.59
  var Hero Disappear X: float = 106

  var Hero X: float = 0
  var Self X: float = 0
  var Activated: bool = false
}
