# Battle Control
# 1 location(s):
#   Fungus2_05 / Battle Scene v2

fsm Battle Control {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Start
  }

  state Init {
    ActivateGameObject(gameObject=var "Shiny Item", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Completed", storeResult=var "Completed")
    FindGameObject(objectName="CameraLockArea A", withTag="Untagged", store=var "Camera Lock A")
    FindGameObject(objectName="CameraLockArea B", withTag="Untagged", store=var "Camera Lock B")
    FindGameObject(objectName="Mushroom Brawler 1", withTag="Untagged", store=var "Brawler 1")
    GetHP(target=var "Brawler 1", storeValue=var "Brawler HP")
    FindGameObject(objectName="Mushroom Brawler 2", withTag="Untagged", store=var "Brawler 2")
    ActivateGameObject(gameObject=var "Camera Lock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock A", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    SetFsmGameObject(gameObject=var "Brawler 2", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Self", everyFrame=false)
    SetFsmGameObject(gameObject=var "Brawler 1", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Self", everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activate
  }

  state Start {
    TransitionToAudioSnapshot(snapshot=Action Only (resources.assets), transitionTime=0)
    ApplyMusicCue(musicCue=RoyalGardens (sharedassets151.assets), delayTime=0, transitionTime=0)
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Brawler 1"), sendEvent="WAKE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"CHECK", lessThan=→"CHECK", greaterThan=(none), everyFrame=true)
    IntCompare(integer1=var "Battle Enemies", integer2=1, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    ActivateGameObject(gameObject=var "Camera Lock A", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock B", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=30, finishEvent=→"NEXT", realTime=false)
    GetHPEveryFrame(target=var "Brawler 1", storeValue=var "Brawler HP")
    IntCompare(integer1=var "Brawler HP", integer2=56, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    on CHECK → End Wait
    on NEXT → Wave 2
  }

  state End Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=2)
    on FINISHED → Drop Item
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock A", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    FindChild(gameObject=Self, childName="Uncompleted", storeResult=var "Uncompleted")
    FindChild(gameObject=var "Uncompleted", childName="Shiny Item", storeResult=var "Shiny Item")
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    ActivateGameObject(gameObject=var "Camera Lock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock A", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Uncompleted", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Brawler 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Brawler 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Completed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wave 2 {
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"CHECK", lessThan=→"CHECK", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Brawler 2"), sendEvent="WAKE", delay=0, everyFrame=false)
    on CHECK → End Wait
  }

  state Drop Item {
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  var Battle Enemies: int = 0
  var Brawler HP: int = 0
  var Activated: bool = false
  var Brawler 1: gameObject = <null>
  var Brawler 2: gameObject = <null>
  var Camera Lock A: gameObject = <null>
  var Camera Lock B: gameObject = <null>
  var Completed: gameObject = <null>
  var Moss Knight: gameObject = <null>
  var Self: gameObject = <null>
  var Shiny Item: gameObject = <null>
  var Uncompleted: gameObject = <null>
}
