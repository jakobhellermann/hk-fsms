// uses template: soul_totem_infinite
fsm soul_totem {
  start Pause Frame
  on RESET → Reset  // from any state

  state Init {
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    FindChild(gameObject=var "Self", childName="Glower", storeResult=var "Glower")
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    SetMaterialColor(gameObject=var "Glower", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"DEPLETED", isFalse=(none), everyFrame=false)  // disabled
    IntCompare(integer1=var "Value", integer2=0, equal=→"DEPLETED", lessThan=→"DEPLETED", greaterThan=(none), everyFrame=false)
    on FINISHED → Far
  }

  state Far {
    SetParticleEmission(gameObject=var "Emitter", emission=false)
    GetDistance(gameObject=var "Self", target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=var "Start Distance", tolerance=0, equal=(none), lessThan=→"MOVE", greaterThan=(none), everyFrame=true)
    GetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=var "Current Colour", fail=(none))
    EaseColor(fromValue=var "Current Colour", toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    ReceivedDamage(collideTag="", sendEvent=→"DAMAGED", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)  // disabled
    on MOVE → Close
  }

  state Close {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=soul_totem_awake (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetParticleEmission(gameObject=var "Emitter", emission=true)
    SetMaterialColor(gameObject=var "Glower", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    GetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=var "Current Colour", fail=(none))
    EaseColor(fromValue=var "Current Colour", toValue=(1, 1, 1, 1), colorVariable=var "Colour", time=1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    GetDistance(gameObject=var "Self", target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=var "Stop Distance", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"MOVE", everyFrame=true)
    on MOVE → Far
    on TAKE DAMAGE → Check if Nail
  }

  state Check if Nail {
    GetEventSender(sentByGameObject=var "Damager")
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    IntCompare(integer1=var "Attack Type", integer2=0, equal=→"DAMAGED", lessThan=(none), greaterThan=(none), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Close
    on DAMAGED → Hit
  }

  state Hit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=soul_totem_slash (sharedassets56.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetMaterialColor(gameObject=var "Glower", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetProperty(targetProperty=SpriteRenderer on <null>, everyFrame=false)
    IntOperator(integer1=var "Value", integer2=1, operation=Subtract, storeResult=var "Value", everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Strike")
    SpawnObjectFromGlobalPool(gameObject=White Flash R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Strike")
    FlingObjectsFromGlobalPool(gameObject=Soul Orb R (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=8, spawnMax=9, speedMin=10, speedMax=20, angleMin=0, angleMax=360, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    IntCompare(integer1=var "Value", integer2=0, equal=→"DEPLETED", lessThan=→"DEPLETED", greaterThan=(none), everyFrame=false)
    on FINISHED → Close
  }

  state Pause Frame {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Soul Particles", storeResult=var "Emitter")
    SetParticleEmission(gameObject=var "Emitter", emission=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset?
  }

  state Reset? {
    IntCompare(integer1=var "Value", integer2=-1, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    SetIntValue(intVariable=var "Value", intValue=5, everyFrame=false)
    on FINISHED → Init
  }

  state Reset {
    SetIntValue(intVariable=var "Value", intValue=5, everyFrame=false)
    ActivateGameObject(gameObject=var "Glower", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Self", active=true)
    on FINISHED → Reset?
  }

  var Distance: float = 0
  var Start Distance: float = 6
  var Stop Distance: float = 7
  var Attack Type: int = 0
  var Value: int = 5
  var Activated: bool = false
  var Colour: color = (0, 0, 0, 1)
  var Current Colour: color = (0, 0, 0, 1)
  var Camera: gameObject = <null>
  var Damager: gameObject = <null>
  var Emitter: gameObject = <null>
  var Glower: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Strike: gameObject = <null>
}
