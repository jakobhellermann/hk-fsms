# Blanker Control
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/2dtk Blanker
#   resources.assets / _GameCameras/HudCamera/2dtk Blanker

fsm Blanker Control {
  start Pause
  on FADE IN INSTANT → Fade In Instant  // from any state
  on FADE OUT INSTANT → Fade Out Instant  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetSpriteRenderer(gameObject=var "Self", active=false)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    SetFsmGameObject(gameObject=var "MainCamera", fsmName="CameraFade", variableName="Blanker", setValue=var "Self", everyFrame=false)
    on FINISHED → Idle
  }

  state Fade In {
    SetColorValue(colorVariable=var "Colour", color=var "Start Colour", everyFrame=false)
    EaseColor(fromValue=var "Start Colour", toValue=var "End Colour", colorVariable=var "Colour", time=var "Fade Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=true)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    GetEventSender(sentByGameObject=var "asdf")
    on FADE OUT → Fade Out
  }

  state Fade Out {
    SetColorValue(colorVariable=var "Colour", color=var "End Colour", everyFrame=false)
    EaseColor(fromValue=var "End Colour", toValue=var "Start Colour", colorVariable=var "Colour", time=var "Fade Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=true)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FADE IN → Fade In
    on FINISHED → Faded Out
  }

  state Idle {
    SetSpriteRenderer(gameObject=Self, active=false)  // disabled
    on FADE IN → Fade In
  }

  state Fade In Instant {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(0, 0, 0, 1), everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FADE OUT → Fade Out
  }

  state Fade Out Instant {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(0, 0, 0, 0), everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    on FADE IN → Fade In
  }

  state Pause {
    SetGameObjectSelf(variable=var "2dtk Blanker", gameObject=Self, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Faded Out {
    SetSpriteRenderer(gameObject=Self, active=false)
    on FADE IN → Fade In
  }

  var Fade Time: float = 1
  var Cancel Reset: bool = false
  var Colour: color = (0, 0, 0, 1)
  var End Colour: color = (0, 0, 0, 1)
  var Start Colour: color = (0, 0, 0, 0)
  var Self: gameObject = <null>
  var asdf: gameObject = <null>
}
