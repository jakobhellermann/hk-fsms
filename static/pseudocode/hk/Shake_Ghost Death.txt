  start Shake

  state Shake {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST FIGHT END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM GATE OPEN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM AREA DISABLE", delay=0, everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tween
    on GG BOSS → Disable particles
  }

  state Tween {
    FindGameObject(objectName="", withTag="Ghost Warrior NPC", store=var "Target")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=soul_totem_awake (sharedassets6.assets))
    GetPosition(gameObject=var "Target", vector=var "Target Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)

    on FINISHED → End
  }

  state End {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST DEFEAT", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)
  }

  state Disable particles {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=soul_totem_awake (sharedassets6.assets))
    ActivateGameObject(gameObject=Ghost Death/Orb Pt, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Ghost Death/Trail Pt, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Target Pos: vector3 = (0, 0, 0)
  var Target: gameObject = <null>
}
