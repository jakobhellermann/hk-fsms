# BG Control
# 3 location(s):
#   Fungus1_32 / Battle Gate Bone (1)
#   Fungus3_05 / Battle Gate Bone (1)
#   Fungus3_10 / Battle Gate Bone (1)

fsm BG Control {
  start Opened
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Opened")
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=gate_slam (sharedassets27.assets), pitchMin=0.8, pitchMax=1.2, volume=0.5, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bone Gate Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Close 2
  }

  state Close 2 {
    Tk2dPlayFrame(gameObject=var "Slam Effect", frame=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Closed")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    on BG OPEN → Open
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=gate_open (sharedassets27.assets), pitchMin=0.85, pitchMax=1.15, volume=0.85, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Open")
    PlayParticleEmitter(gameObject=var "Raise Dust", emit=0)
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Closed")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on BG OPEN → Open
    on FINISHED → Double Close
  }

  state Double Close {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Closed")
    on BG OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Opened")
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  var Start Closed: bool = false

  var Camera: gameObject = <null>
  var Dust: gameObject = Battle Gate Bone (1)/Dust
  var Raise Dust: gameObject = Battle Gate Bone (1)/Raise Dust
  var Self: gameObject = <null>
  var Slam Effect: gameObject = Battle Gate Bone (1)/Close Effect
}
