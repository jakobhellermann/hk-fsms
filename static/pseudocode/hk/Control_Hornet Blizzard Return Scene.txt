# Control
# 1 location(s):
#   Deepnest_East_12 / Hornet Blizzard Return Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="savedByHornet", isTrue=→"INERT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="blizzardEnded", isTrue=→"PLAY", isFalse=(none))
    on PLAY → Fade Pause
    on INERT → Inert
  }

  state Fade Pause {
    FindChild(gameObject=Self, childName="Exit Puff", storeResult=var "Exit Puff")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    FindChild(gameObject=Self, childName="Land Puff", storeResult=var "Land Puff")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Fade In
  }

  state Fade In {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    on FINISHED → Burst
  }

  state Burst {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Anim 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Exit Puff", emit=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=jump_out_of_snow (sharedassets316.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Land
  }

  state Land {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Anim 2")
    PlayParticleEmitter(gameObject=var "Land Puff", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_ground_land (sharedassets28.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Leave Antic
  }

  state Leave Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Anim 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_needle_thow_spin (sharedassets105.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Leave
  }

  state Leave {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Anim 4", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Prostrate")
    SetMeshRenderer(gameObject=var "Hero", active=true)
    iTweenMoveBy(gameObject=Self, id="", vector=(-25, 6, 0), time=0.25, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_dash (sharedassets34.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Left
  }

  state Left {
    SetMeshRenderer(gameObject=Self, active=false)
    ListenForLeft(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForDown(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForUp(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForRight(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForAttack(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on GET UP → Get Up
  }

  state Get Up {
    SetPlayerDataBool(boolName="savedByHornet", value=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Prostrate Rise", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Return Control
  }

  state Inert {
  }

  state Return Control {
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AcceptInput())
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetBenchRespawn", parameters=["Death Respawn Marker", "Deepnest_East_12", 0, true], storeResult=(unset var))
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SaveGame())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=TimePasses())
  }

  var Exit Puff: gameObject = <null>
  var Land Puff: gameObject = <null>
}
