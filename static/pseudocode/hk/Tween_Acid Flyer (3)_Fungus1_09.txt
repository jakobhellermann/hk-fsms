# Tween
# 1 location(s):
#   Fungus1_09 / Acid Flyer (3)

fsm Tween {
  start Init

  state Tween Up {
    iTweenMoveBy(gameObject=Self, id="", vector=var "Move Vector", time=(unset), delay=0, speed=var "Speed", easeType=easeInOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Tween Down
  }

  state Init {
    SetVector3Value(vector3Variable=var "Inverse Vector", vector3Value=var "Move Vector", everyFrame=false)
    Vector3Multiply(vector3Variable=var "Inverse Vector", multiplyBy=-1, everyFrame=false)
    GetPosition(gameObject=Self, vector=var "Initial Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tween Up
  }

  state Reset Pos {
    SetPosition(gameObject=Self, vector=var "Initial Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Tween Up
  }

  state Tween Down {
    iTweenMoveBy(gameObject=Self, id="", vector=var "Inverse Vector", time=(unset), delay=0, speed=var "Speed", easeType=easeInOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Reset Pos
  }

  var Speed: float = 4
  var Move Vector: vector3 = (0, -8.5, 0)

  var Initial Pos: vector3 = (0, 0, 0)
  var Inverse Vector: vector3 = (0, 0, 0)
}
