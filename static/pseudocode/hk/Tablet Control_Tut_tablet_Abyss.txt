# Tablet Control
# 1 location(s):
#   Abyss_06_Core / Tut_tablet_Abyss

fsm Tablet Control {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="lit_tablet", storeResult=var "Lit")
    FindChild(gameObject=Self, childName="Glows", storeResult=var "Glows")
    FindChild(gameObject=Self, childName="Rim", storeResult=var "Rim")
    SetParticleEmissionRate(gameObject=var "Rim", emissionRate=0, everyFrame=false)
    on FINISHED → Away
  }

  state Away {
    AudioStop(gameObject=Self)
    SetParticleEmission(gameObject=var "Glows", emission=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Hero Distance", everyFrame=true)
    FloatCompare(float1=var "Hero Distance", float2=10, tolerance=0, equal=(none), lessThan=→"ENTER", greaterThan=(none), everyFrame=true)
    on ENTER → Close
  }

  state Close {
    AudioPlay(gameObject=Self, volume=0.874, oneShotClip=<null>, finishedEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=soul_totem_awake (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetParticleEmissionRate(gameObject=var "Rim", emissionRate=500, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Rim", emit=0)
    SetParticleEmission(gameObject=var "Glows", emission=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Hero Distance", everyFrame=true)
    FloatCompare(float1=var "Hero Distance", float2=12, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"EXIT", everyFrame=true)
    on EXIT → Away
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Move To Offset: float = 1.15

  var Axis Vert: float = 0
  var Hero Distance: float = 0
  var Hero X: float = 0
  var Hero X Scale: float = 0
  var Left Prox: float = 0
  var Right Prox: float = 0
  var Self X: float = 0
  var Attacking: bool = false
  var Attacking Down: bool = false
  var Attacking Up: bool = false
  var Backdashing: bool = false
  var Can Input: bool = false
  var Dashing: bool = false
  var Hero Facing Left: bool = false
  var Hero Facing Right: bool = false
  var Hero On Left: bool = false
  var Hero On Right: bool = false
  var On Ground: bool = false
  var Glows: gameObject = <null>
  var Lit: gameObject = <null>
  var Prompt: gameObject = <null>
  var Rim: gameObject = <null>
  var Self: gameObject = <null>
}
