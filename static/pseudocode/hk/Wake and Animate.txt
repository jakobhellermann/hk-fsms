# Wake and Animate
# 1 location(s):
#   Room_Jinn / Jinn NPC

fsm Wake and Animate {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Talk NPC", storeResult=var "Talk NPC")
    on FINISHED → Sleep
  }

  state Sleep {
    on TAKE DAMAGE → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Talk NPC", audioClips=[Jinn_Wake (sharedassets29.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Activate NPC
  }

  state Activate NPC {
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Talk NPC?
  }

  state Idle L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    BoolTest(boolVariable=var "Behind Range", isTrue=(none), isFalse=(none), everyFrame=true)
    SetCollider(gameObject=Self, active=true)
    on R → Turn R
    on TAKE DAMAGE → Hit
    on TALK → Talk L
  }

  state Turn R {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle R
    on TAKE DAMAGE → Hit
    on TALK → Talk R
  }

  state Idle R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look R")
    BoolTest(boolVariable=var "Behind Range", isTrue=(none), isFalse=(none), everyFrame=true)
    on L → Turn L
    on TAKE DAMAGE → Hit
    on TALK → Talk R
  }

  state Turn L {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle L
    on TAKE DAMAGE → Hit
    on TALK → Talk L
  }

  state Hit {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Talk NPC", audioClips=[Jinn_head_bounce_01 (sharedassets29.assets), Jinn_head_bounce_02 (sharedassets29.assets), Jinn_head_bounce_03 (sharedassets29.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle L
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk R")
    on TALK END → Idle R
  }

  state Talk L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    on TALK END → Idle L
  }

  state Talk NPC? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metJinn", isTrue=(none), isFalse=→"FINISHED")
    ActivateGameObject(gameObject=var "Talk NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle L
  }

  var Behind Range: bool = false
  var Talk NPC: gameObject = <null>
}
