# Control
# 1 location(s):
#   Abyss_06_Core / Blue Door

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Collider", storeResult=var "Collider")
    FindChild(gameObject=Self, childName="Vines", storeResult=var "Vines")
    FindChild(gameObject=Self, childName="Vines Grey", storeResult=var "Vines Grey")
    FindChild(gameObject=Self, childName="Pt Open", storeResult=var "Pt Open")
    FindChild(gameObject=Self, childName="White Flash", storeResult=var "White Flash")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotCharm_9", isTrue=→"GOT CHARM", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="blueVineDoor", isTrue=→"OPENED", isFalse=(none))
    on OPENED → Opened
    on OPEN VINE DOOR → Open
    on GOT CHARM → Got Charm
  }

  state Opened {
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vines", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Open {
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vines", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Open", emit=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Vines Grey", audioClip=spell_pickup_final_boom (sharedassets10.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  state Got Charm {
    ActivateGameObject(gameObject=var "Vines Grey", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vines", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Collider: gameObject = <null>
  var Pt Open: gameObject = <null>
  var Vines: gameObject = <null>
  var Vines Grey: gameObject = <null>
  var White Flash: gameObject = <null>
}
