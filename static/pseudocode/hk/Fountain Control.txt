# Fountain Control
# 1 location(s):
#   Abyss_04 / Wishing_Well_anims

fsm Fountain Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    FindChild(gameObject=Self, childName="Wishing_Well_normal", storeResult=var "Wishing_Well_normal")
    FindChild(gameObject=Self, childName="Mask_Appear", storeResult=var "Mask_Appear")
    FindChild(gameObject=Self, childName="Vessel Fragment", storeResult=var "Vessel Fragment")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="fountainVesselSummoned", isTrue=→"ALREADY PAID", isFalse=(none))
    on FOUNTAIN PAID → Appear Anim
    on ALREADY PAID → Already Paid
  }

  state Appear Anim {
    ActivateGameObject(gameObject=var "Mask_Appear", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Appear
  }

  state Appear {
    ActivateGameObject(gameObject=var "Wishing_Well_normal", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vessel Fragment", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
  }

  state Already Paid {
    ActivateGameObject(gameObject=var "Wishing_Well_normal", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vessel Fragment", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Mask_Appear: gameObject = <null>
  var Vessel Fragment: gameObject = <null>
  var Wishing_Well_normal: gameObject = <null>
}
