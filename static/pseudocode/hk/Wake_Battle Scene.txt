# Wake
# 2 location(s):
#   Ruins2_01 / Battle Scene
#   Ruins2_01_b / Battle Scene

fsm Wake {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Wake
  }

  state Wake {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WAKE", delay=0, everyFrame=false)
  }
}
