# Control Audio
# 2 location(s):
#   Knight_Pickup / Knight/Charm Effects/Fury
#   resources.assets / Knight:1/Charm Effects/Fury

fsm Control Audio {
  start Init

  state Init {
    on PLAY → Play
  }

  state Play {
    SetFloatValue(floatVariable=var "Volume", floatValue=1, everyFrame=false)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetAudioVolume(gameObject=Self, volume=var "Volume", everyFrame=true)
    EaseFloat(fromValue=1, toValue=0, floatVariable=var "Volume", time=5, speed=(unset), delay=4, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    on STOP → Stop
  }

  state Stop {
    AudioStop(gameObject=Self)
    on PLAY → Play
  }

  var Volume: float = 0
}
