# Control
# 1 location(s):
#   Room_Queen / Queen Item

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotQueenFragment", isTrue=→"INACTIVE", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metQueen", isTrue=→"CHARM DROP", isFalse=(none))
    on CHARM DROP → Drop
    on INACTIVE → Inactive
  }

  state Drop {
    FindGameObject(objectName="Queen", withTag="Untagged", store=var "Queen")
    SetCollider(gameObject=var "Queen", active=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  state Inactive {
  }

  var Queen: gameObject = <null>
  var Self: gameObject = <null>
}
