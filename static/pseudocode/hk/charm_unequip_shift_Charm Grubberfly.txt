// uses template: charm_unequip_shift
fsm charm_unequip_shift {
  start Init

  state Idle {
    on UNEQUIP SHIFT → Shift?
  }

  state Shift? {
    BoolTest(boolVariable=var "Always Shift", isTrue=→"NOTCH", isFalse=(none), everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=1, everyFrame=false)
    GetFsmFloat(gameObject=var "Charms Menu", fsmName="UI Charms", variableName="Target Charm X", storeValue=var "Target Charm X", everyFrame=false)
    FloatCompare(float1=var "Self X", float2=var "Target Charm X", tolerance=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Idle
    on FINISHED → Shift
    on NOTCH → Notch Shift
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Charms Pane", store=var "Charms Menu")
    FindChild(gameObject=Self, childName="Sprite Next", storeResult=var "Sprite Next")
    on FINISHED → Idle
  }

  state Shift {
    iTweenMoveBy(gameObject=Self, id="", vector=(-1.76, 0, 0), time=0.25, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=1, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Idle
  }

  state Notch Shift {
    Translate(gameObject=Self, vector=(unset), x=-1.76, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    GetFsmBool(gameObject=var "Charms Menu", fsmName="UI Charms", variableName="Open Slot", storeValue=var "Open Slot", everyFrame=false)
    BoolTest(boolVariable=var "Open Slot", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetSpriteRenderer(gameObject=var "Sprite Next", active=true)
    SetScale(gameObject=var "Sprite Next", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=var "Sprite Next", id="", transformScale=var "", vectorScale=(0.1, 0.1, 0.1), time=0.15, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Sprite Next Off
    on CANCEL → Idle
  }

  state Sprite Next Off {
    SetSpriteRenderer(gameObject=var "Sprite Next", active=false)
    on FINISHED → Idle
  }

  var Always Shift: bool = false

  var Self X: float = 0
  var Target Charm X: float = 0
  var Open Slot: bool = false
  var Charms Menu: gameObject = <null>
  var Self: gameObject = <null>
  var Sprite Next: gameObject = <null>
}
