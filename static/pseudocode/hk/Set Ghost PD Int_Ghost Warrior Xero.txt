# Set Ghost PD Int
# 1 location(s):
#   RestingGrounds_02_boss / Warrior/Ghost Warrior Xero

fsm Set Ghost PD Int {
  start Idle

  state Idle {
    on ZERO HP → Set
  }

  state Set {
    SetPlayerDataInt(intName=var "PD Int", value=1)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST DEAD", delay=0, everyFrame=false)
  }

  var PD Int: string = "xeroDefeated"
}
