# Hit
# 1 location(s):
#   sharedassets361.assets / Corpse Fluke Mother/Remains

fsm Hit {
  start Pause

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    SetCollider(gameObject=Self, active=true)
    ReceivedDamage(collideTag="", sendEvent=→"NAIL HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on NAIL HIT → Check
  }

  state Check {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage Dealt", everyFrame=false)
    IntCompare(integer1=var "Damage Dealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Thwack
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Thwack {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Remains Hit")
    CreateObject(gameObject=Audio Player Actor (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Audio Player", networkInstantiate=false, networkGroup=0)  // disabled
    AudioPlay(gameObject=var "Audio Player", volume=1, oneShotClip=enemy_damage (resources.assets), finishedEvent=(none))  // disabled
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_damage (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Cancel Frame
  }

  var Damage Dealt: int = 0
  var Audio Player: gameObject = <null>
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
