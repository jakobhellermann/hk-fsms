# Conversation Control
# 1 location(s):
#   GG_Boss_Door_Entrance / Conversation

fsm Conversation Control {
  start Wait

  state Init {
    GetOwner(storeGameObject=var "Self")
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)  // disabled
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "GG_Entrance_cutscene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set Appearance
  }

  state Talk Finish {
    GetStaticVariable(variableName="firstBossScene", storeValue=var "To Scene")
    BeginSceneTransition(sceneName=var "To Scene", entryGateName="door_dreamEnter", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  state Greet 0 {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[GS_standard_05 (sharedassets475.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["PANTHEON_ENTER_0", "CP3"], storeResult=(unset var))
    on CONVO_FINISH → Out Pause
  }

  state Set Appearance {
    SetTextMeshProColor(gameObject=var "DialogueText", color=(0, 0, 0, 1), everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=false, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=true, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    on FINISHED → Text.
  }

  state Reset Appearance {
    SetTextMeshProColor(gameObject=var "DialogueText", color=(1, 1, 1, 1), everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Show HUD
  }

  state Text? {
    GGCheckIsBossRushMode(trueEvent=→"CANCEL", falseEvent=(none))
    GetStaticVariable(variableName="currentBossDoorPD", storeValue=var "Boss Door PD")
    GGGetBossDoorCompletion(playerDataVariable=var "Boss Door PD", unlocked=(unset), completed=var "Door Completed", allBindings=(unset), noHits=(unset), boundNail=(unset), boundShell=(unset), boundCharms=(unset), boundSoul=(unset))
    BoolTest(boolVariable=var "Door Completed", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="bossDoorEntranceTextSeen", storeValue=var "Seen Text")
    GGCountCompletedBossDoors(storeValue=var "Completed Doors")
    IntCompare(integer1=var "Seen Text", integer2=var "Completed Doors", equal=→"CANCEL", lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    IntCompare(integer1=var "Completed Doors", integer2=4, equal=→"CANCEL", lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    SetPlayerDataInt(intName="bossDoorEntranceTextSeen", value=var "Completed Doors")
    on FINISHED → Pause
    on CANCEL → Talk Finish
  }

  state Text. {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[GS_standard_01 (sharedassets480.assets), GS_standard_02 (sharedassets475.assets), GS_standard_05 (sharedassets475.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)  // disabled
    ConvertIntToString(intVariable=var "Completed Doors", stringVariable=var "Text Event", format="", everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent=var "Text Event", delay=0, everyFrame=false)
    on 0 → Greet 0
    on 1 → Greet 1
    on 2 → Greet 2
    on 3 → Greet 3
    on 4 → Greet 4
  }

  state Greet 1 {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[GS_standard_01 (sharedassets480.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["PANTHEON_ENTER_1", "CP3"], storeResult=(unset var))
    on CONVO_FINISH → Out Pause
  }

  state Greet 2 {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[GS_standard_02 (sharedassets475.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["PANTHEON_ENTER_2", "CP3"], storeResult=(unset var))
    on CONVO_FINISH → Out Pause
  }

  state Greet 3 {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClips=[GS_engine_room_04 (sharedassets475.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["PANTHEON_ENTER_3", "CP3"], storeResult=(unset var))
    on CONVO_FINISH → Out Pause
  }

  state Greet 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["PANTHEON_ENTER_4", "CP3"], storeResult=(unset var))
    on CONVO_FINISH → Out Pause
  }

  state Wait {
    WaitForSceneLoadFinish(sendEvent=→"FINISHED")
    on FINISHED → Wait 1
  }

  state Wait 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Text?
  }

  state Show HUD {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on FINISHED → Talk Finish
  }

  state Pause {
    FindChild(gameObject=Self, childName="GG_Entrance_cutscene", storeResult=var "GG_Entrance_cutscene")
    FindChild(gameObject=Self, childName="Audio", storeResult=var "Audio")
    TransitionToAudioSnapshot(snapshot=Off (resources.assets), transitionTime=0.001)  // disabled
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=0.001)  // disabled
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Atmos
  }

  state Atmos {
    TransitionToAudioSnapshot(snapshot=On (resources.assets), transitionTime=4)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Audio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    on FINISHED → Init
  }

  state Out Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    TransitionToAudioSnapshot(snapshot=Off (resources.assets), transitionTime=1)
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=1)
    on FINISHED → Reset Appearance
  }

  var Completed Doors: int = 0
  var Seen Text: int = 0
  var Door Completed: bool = false
  var Boss Door PD: string = ""
  var Text Event: string = ""
  var To Scene: string = ""
  var Audio: gameObject = <null>
  var GG_Entrance_cutscene: gameObject = <null>
  var Self: gameObject = <null>
}
