# Chain Move
# 4 location(s):
#   Cliffs_03 / elev_chain (6)
#   Cliffs_03 / elev_chain (7)
#   Cliffs_03 / elev_chain (5)
#   Room_Town_Stag_Station / elev_chain

fsm Chain Move {
  start Init

  state Init {
    FindGameObject(objectName="Stag Lift", withTag="Untagged", store=var "Lift")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=0, everyFrame=false)
    on FINISHED → Move
  }

  state Move {
    GetPosition(gameObject=var "Lift", vector=(unset), x=(unset), y=var "Move To Y", z=(unset), space=0, everyFrame=true)
    FloatSubtract(floatVariable=var "Move To Y", subtract=var "Lift Bot Y", everyFrame=true, perSecond=false)
    FloatMultiply(floatVariable=var "Move To Y", multiplyBy=0.6, everyFrame=true)
    FloatAdd(floatVariable=var "Move To Y", add=var "Start Y", everyFrame=true, perSecond=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Move To Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
  }

  var Lift Bot Y: float = 4.68
  var Lift Y: float = 0
  var Move To Y: float = 0
  var Start Y: float = 0
  var Lift: gameObject = <null>
}
