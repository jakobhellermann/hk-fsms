# CameraFade
# 2 location(s):
#   Menu_Title / _GameCameras/CameraParent/tk2dCamera
#   resources.assets / _GameCameras/CameraParent/tk2dCamera

// uses template: camera_fade
fsm CameraFade {
  start Pause
  on FADE SCENE IN MENU → Pre Fade Menu  // from any state
  on FADE OUT → FadingOut  // from any state
  on FADE SCENE IN SLOWLY → Slow Fade In  // from any state
  on FADE SCENE IN INSTANT → Instant Fade In  // from any state
  on FADE SCENE IN → Pre Fade  // from any state
  on FADE OUT INSTANT → Fade Out Instant  // from any state

  state Normal {
    SetColorValue(colorVariable=var "Fade Colour", color=(0, 0, 0, 1), everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "White Fade In", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "No Fade", boolValue=false, everyFrame=false)  // disabled
    on FADE OUT → FadingOut
    on RESPAWN FADE → RespawnFadingOut
    on HAZARD FADE → HazardFadingOut
    on JUST FADE → Just Fading
    on FADE INSTANT → Instant Fade
    on START FADE → Start Fade
  }

  sequence state FadingOut {
    CameraFadeOut(color=var "Fade Colour", time=0.33, finishEvent=(none), realTime=false)  // disabled
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=FadeToBlack(0.25))  // disabled
    Wait(time=0.25, finishEvent=→"FADE OUT COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.33, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    on FADE OUT COMPLETE → White Fade In? 2
  }

  state FadeIn {
    CameraFadeInWithDelay(color=var "Fade Colour", delay=0, time=0.5, finishEvent=→"FADE IN COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE OUT", delay=0.1, everyFrame=false)
    Wait(time=0.6, finishEvent=→"FADE IN COMPLETE", realTime=false)
    on FADE IN COMPLETE → Normal
    on JUST FADE → Just Fading
  }

  state Done {
    CameraFadeIn(color=(0, 0, 0, 1), time=50, finishEvent=(none), realTime=false)  // disabled
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=LeftScene())  // disabled
    CameraFadeOut(color=(0, 0, 0, 1), time=0, finishEvent=→"FADE OUT COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    on LEVEL LOADED → Wait For Fade In
  }

  state RespawnFadingOut {
    CameraFadeOut(color=(0, 0, 0, 1), time=1, finishEvent=→"RESPAWN FADE OUT COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.75, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"RESPAWN FADE OUT COMPLETE", realTime=false)
    on RESPAWN FADE OUT COMPLETE → WaitForRespawn
  }

  state WaitForRespawn {
    CameraFadeIn(color=(0, 0, 0, 1), time=50, finishEvent=(none), realTime=false)  // disabled
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=ReadyForRespawn())  // disabled
    on RESPAWN → FadeIn
  }

  state HazardFadingOut {
    CameraFadeOut(color=(0, 0, 0, 1), time=1, finishEvent=→"HAZARD FADE OUT COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.75, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"HAZARD FADE OUT COMPLETE", realTime=false)
    on HAZARD FADE OUT COMPLETE → WaitForHazardRespawn
  }

  state WaitForHazardRespawn {
    CameraFadeIn(color=(0, 0, 0, 1), time=50, finishEvent=(none), realTime=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=HazardRespawn())  // disabled
    on RESPAWN → FadeIn
  }

  sequence state Just Fading {
    CameraFadeOut(color=(0, 0, 0, 1), time=0.25, finishEvent=→"FADE OUT COMPLETE", realTime=true)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.25, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
  }

  state Instant Fade {
    CameraFadeOut(color=(0, 0, 0, 1), time=0, finishEvent=→"FADE OUT COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN INSTANT", delay=0, everyFrame=false)
    SetMaterialColor(gameObject=var "Blanker", materialIndex=0, material=<null>, namedColor="_Color", color=(0, 0, 0, 1), everyFrame=false)
    SetSpriteRenderer(gameObject=var "Blanker", active=true)  // disabled
    on FADE BACK → Fade Back
    on LEVEL LOADED → Wait For Fade In
  }

  state Start Fade {
    CameraFadeOut(color=(0, 0, 0, 1), time=2.3, finishEvent=→"FADE OUT COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Start Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=2.3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Start Blanker"), sendEvent="START FADE", delay=0, everyFrame=false)
  }

  state Fade Back {
    CameraFadeInWithDelay(color=(0, 0, 0, 1), delay=0, time=3, finishEvent=→"FADE IN COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FADE IN COMPLETE", realTime=false)
    on FADE IN COMPLETE → Normal
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "MainCamera", gameObject=var "Self", everyFrame=false)
    SetGameObject(variable=var "Blanker", gameObject=var "MainCamera", everyFrame=false)  // disabled
    FindChild(gameObject=Self, childName="Start Blanker", storeResult=var "Start Blanker")
    on FINISHED → Wait For Fade In
  }

  state Wait For Fade In {
    CameraFadeOut(color=var "Fade Colour", time=0, finishEvent=→"FADE OUT COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    CheckSceneName(sceneName="Menu_Title", equalEvent=→"FADE SCENE IN", notEqualEvent=(none))
    Wait(time=5, finishEvent=→"FADE SCENE IN", realTime=false)
    on FADE SCENE IN → White Fade In?
  }

  state Slow Fade In {
    CameraFadeInWithDelay(color=(0, 0, 0, 1), delay=0, time=3, finishEvent=→"FADE IN COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE OUT", delay=0.1, everyFrame=false)
    Wait(time=3.1, finishEvent=→"FADE IN COMPLETE", realTime=false)
    on FADE IN COMPLETE → Normal
    on JUST FADE → Just Fading
  }

  state White Fade In? {
    SendEventByName(eventTarget=GameObject(var "Start Blanker"), sendEvent="OUT", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "No Fade", isTrue=→"NO FADE", isFalse=(none), everyFrame=false)  // disabled
    BoolTest(boolVariable=var "White Fade In", isTrue=→"WHITE", isFalse=→"BLACK", everyFrame=false)
    on WHITE → Set White
    on BLACK → Set Black
    on NO FADE → Insta Fade Out
  }

  state Set Black {
    SetColorValue(colorVariable=var "Fade Colour", color=(0, 0, 0, 1), everyFrame=false)  // disabled
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Start Colour", setValue=(0, 0, 0, 0), everyFrame=false)
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="End Colour", setValue=(0, 0, 0, 1), everyFrame=false)
    on FINISHED → FadeIn
  }

  state Set White {
    SetColorValue(colorVariable=var "Fade Colour", color=(1, 1, 1, 1), everyFrame=false)  // disabled
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Start Colour", setValue=(1, 1, 1, 0), everyFrame=false)
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="End Colour", setValue=(1, 1, 1, 1), everyFrame=false)
    SetMaterialColor(gameObject=var "Blanker", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → FadeIn
  }

  state White Fade In? 2 {
    BoolTest(boolVariable=var "White Fade In", isTrue=→"WHITE", isFalse=→"BLACK", everyFrame=false)
    on WHITE → Set White 2
    on BLACK → Set Black 2
  }

  state Set White 2 {
    SetColorValue(colorVariable=var "Fade Colour", color=(1, 1, 1, 1), everyFrame=false)  // disabled
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Start Colour", setValue=(1, 1, 1, 0), everyFrame=false)
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="End Colour", setValue=(1, 1, 1, 1), everyFrame=false)
    on FINISHED → Done
  }

  state Set Black 2 {
    SetColorValue(colorVariable=var "Fade Colour", color=(0, 0, 0, 1), everyFrame=false)  // disabled
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Start Colour", setValue=(0, 0, 0, 0), everyFrame=false)
    SetFsmColor(gameObject=var "Blanker", fsmName="Blanker Control", variableName="End Colour", setValue=(0, 0, 0, 1), everyFrame=false)
    on FINISHED → Done
  }

  state Instant Fade In {
    CameraFadeInWithDelay(color=(0, 0, 0, 1), delay=0, time=0.001, finishEvent=→"FADE IN COMPLETE", realTime=false)  // disabled
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=0, finishEvent=→"FADE IN COMPLETE", realTime=false)
    on FADE IN COMPLETE → Normal
    on JUST FADE → Just Fading
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "MainCamera", gameObject=var "Self", everyFrame=false)
    on FINISHED → Init
  }

  state Pre Fade {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN INSTANT", delay=0, everyFrame=false)
    on FINISHED → White Fade In?
  }

  state Insta Fade Out {
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE OUT INSTANT", delay=0, everyFrame=false)
    on FINISHED → Normal
  }

  state Fade Out Instant {
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN INSTANT", delay=0, everyFrame=false)
    on FINISHED → White Fade In? 2
  }

  state Pre Fade Menu {
    SetFsmFloat(gameObject=var "Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Blanker"), sendEvent="FADE IN INSTANT", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → White Fade In?
  }

  var Start Blanker: gameObject = _GameCameras/CameraParent/tk2dCamera/Start Blanker

  var No Fade: bool = false
  var White Fade In: bool = false
  var Fade Colour: color = (0, 0, 0, 1)
  var Blanker: gameObject = <null>
  var Scene Manager: gameObject = <null>
  var Self: gameObject = <null>
}
