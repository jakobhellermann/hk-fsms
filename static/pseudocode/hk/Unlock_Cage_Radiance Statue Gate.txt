# Unlock
# 1 location(s):
#   GG_Workshop / Radiance Statue Gate/Cage

fsm Unlock {
  start Wait

  state Idle {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    GGCheckBossSceneUnlocked(bossScene=Radiance Boss Scene (resources.assets), checkSource=Statue, trueEvent=→"CAN OPEN", falseEvent=(none))
    on CAN OPEN → Activated?
  }

  state Can Open {
    TriggerEnterEventSubscribe(trigger=Radiance Statue Gate/Cage/Hero Detector@TriggerEnterEvent, triggerEnteredEvent=→"HERO ENTERED", triggerExitedEvent=(none), triggerStayedEvent=(none))
    on HERO ENTERED → Do Open
  }

  state Do Open {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    AudioPlayInState(gameObject=Self, volume=1)
    ActivateGameObject(gameObject=Radiance Statue Gate/Fall Dust, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlayStateWait(target=Self, stateName="Open", finishEvent=→"FINISHED")
    on FINISHED → Restore Movement
  }

  state Opened {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Radiance Statue Gate/Cage/solid collider, active=false)
  }

  state Activated? {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Can Open
    on ACTIVATE → Opened
  }

  state Wait {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Restore Movement {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    SetCollider(gameObject=Radiance Statue Gate/Cage/solid collider, active=false)
    AudioPlayV2(gameObject=Self, volume=1, oneShotClip=misc_rumble_impact (sharedassets13.assets), finishedEvent=(none))
    StopParticleEmitter(gameObject=Radiance Statue Gate/Fall Dust)
    ActivateGameObject(gameObject=Radiance Statue Gate/Rise Dust, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Self: gameObject = <null>
}
