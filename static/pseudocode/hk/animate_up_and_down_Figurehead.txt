# animate_up_and_down
# 2 location(s):
#   Fungus3_35 / Bank Menu/Window/Figurehead
#   sharedassets218.assets / Bank Menu/Window/Figurehead

// uses template: animate_up_and_down
fsm animate_up_and_down {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetMeshRenderer(gameObject=var "Self", active=false)
    on UP → Delay
    on DOWN → Down
  }

  state Up {
    SetMeshRenderer(gameObject=var "Self", active=true)
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName=var "Up Anim")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ANIM UP", delay=0, everyFrame=false)
    on DOWN → Down
    on REPEAT UP ANIM → repeat
  }

  state Down {
    Tk2dPlayAnimationWithEvents(gameObject=var "Self", clipName=var "Down Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on UP → Delay
    on FINISHED → Renderer off
  }

  state Delay {
    Wait(time=var "Up Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
    on DOWN → Down
  }

  state Renderer off {
    SetMeshRenderer(gameObject=var "Self", active=false)
    on UP → Delay
  }

  state repeat {
    on FINISHED → Up
  }

  var Up Delay: float = 0.35
  var Down Anim: string = "Shop_Figurehead_Banker Down"
  var Up Anim: string = "Shop_Figurehead_Banker Up"

  var Self: gameObject = <null>
}
