# Control
# 1 location(s):
#   Ruins_House_03 / Open Door

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="city2_sewerDoor", isTrue=→"OPENED", isFalse=(none))
    on OPENED → Opened
    on FINISHED → Closed
  }

  state Opened {
  }

  state Init {
    FindChild(gameObject=Self, childName="Closed", storeResult=var "Closed")
    FindChild(gameObject=Self, childName="Open", storeResult=var "Open")
    FindChild(gameObject=Self, childName="Glow", storeResult=var "Glow")
    on FINISHED → Check
  }

  state Closed {
    ActivateGameObject(gameObject=var "Closed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on OPEN → Open
  }

  state Open {
    ActivateGameObject(gameObject=var "Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
  }

  var Closed: gameObject = <null>
  var Glow: gameObject = <null>
  var Open: gameObject = <null>
}
