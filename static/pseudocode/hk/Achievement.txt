# Achievement
# 1 location(s):
#   Cliffs_03 / Area Title Controller

fsm Achievement {
  start State 1

  state State 1 {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → State 3
  }

  state State 2 {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["STAG_STATION_ALL"], storeResult=(unset var))
  }

  state State 3 {
    GetPlayerDataInt(gameObject=var "GameManager", intName="stagPosition", storeValue=var "Current Position")
    IntCompare(integer1=var "Current Position", integer2=9, equal=→"ENTER", lessThan=(none), greaterThan=(none), everyFrame=false)
    on ENTER → State 2
    on FINISHED → State 1
  }

  var Current Position: int = 0
}
