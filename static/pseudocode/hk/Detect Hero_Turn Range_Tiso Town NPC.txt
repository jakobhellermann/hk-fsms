# Detect Hero
# 1 location(s):
#   Town / _NPCs/Tiso Town NPC/Turn Range

fsm Detect Hero {
  start Init

  state Out {
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="LookLeft")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="LookRight")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
    on EXIT → Out
  }

  var Parent: gameObject = <null>
}
