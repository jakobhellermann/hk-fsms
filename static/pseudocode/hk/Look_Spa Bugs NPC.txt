# Look
# 1 location(s):
#   Crossroads_30 / Spa Bugs NPC

fsm Look {
  start Pause
  on SPLASH → Splash Anim  // from any state

  state Idle {
    SetPlayerDataBool(boolName="spaBugsEncountered", value=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    AudioPlayInState(gameObject=Self, volume=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on ENTER → Look
  }

  state Look {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look")
    AudioPlayRandom(gameObject=var "Self", audioClips=[Spa_Bugs_Startle_01 (sharedassets62.assets), Spa_Bugs_Startle_02 (sharedassets62.assets), Spa_Bugs_Startle_03 (sharedassets62.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on EXIT → Unlook
  }

  state Unlook {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look End")
    AudioPlayInState(gameObject=Self, volume=1)
    on ENTER → Look
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="spaBugsEncountered", isTrue=→"DESTROY", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasWalljump", isTrue=(none), isFalse=→"DESTROY")
    on DESTROY → Destroy
    on FINISHED → Idle
  }

  state Destroy {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Splash Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Splashed", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[<null>, <null>, <null>], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Reset Check
  }

  state Reset Check {
    SendTrigger2DEvent(eventTarget=Self, trigger=OnTriggerStay2D, collideTag="", collideLayer=(unset), sendEvent=→"ENTER", storeCollider=var "")
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on ENTER → Look
  }

  var Hero Pos: float = 0
  var Hero Scale: float = 0
  var Self: gameObject = <null>
}
