# Display Title
# 1 location(s):
#   GG_Ghost_Hu / GG Ghost Boss Title

fsm Display Title {
  start Wait

  state Display Title {
    ActivateGameObject(gameObject=var "AreaTitle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "AreaTitle", fsmName="", variableName="NPC Title", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "AreaTitle", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "AreaTitle", fsmName="", variableName="Area Event", setValue=var "Area Event", everyFrame=false)
  }

  state Wait {
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Build String
  }

  state Build String {
    BuildString(stringParts=["GH_", var "Ghost Name", "_C"], separator="", addToEnd=true, storeResult=var "Area Event", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    BuildString(stringParts=["GH_", var "Ghost Name", "_NC"], separator="", addToEnd=true, storeResult=var "Area Event", everyFrame=false)
    on FINISHED → Display Title
  }

  var Delay: float = 2
  var Ghost Name: string = "HU"

  var Area Event: string = ""
}
