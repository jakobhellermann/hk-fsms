# Damage timing
# 1 location(s):
#   sharedassets399.assets / Shockwave Spurt Z

fsm Damage timing {
  start Wait

  state Wait {
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Activate
  }

  state Activate {
    SetFsmInt(gameObject=Self, fsmName="damages_hero", variableName="damageDealt", setValue=1, everyFrame=false)  // disabled
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Deactivate
  }

  state Deactivate {
    SetFsmInt(gameObject=Self, fsmName="damages_hero", variableName="damageDealt", setValue=0, everyFrame=false)  // disabled
    SetCollider(gameObject=Self, active=false)
  }
}
