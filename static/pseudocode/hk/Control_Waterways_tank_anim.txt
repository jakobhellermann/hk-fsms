# Control
# 1 location(s):
#   Waterways_05 / Waterways_tank_anim

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="waterwaysAcidDrained", isTrue=→"FILLED", isFalse=(none))
    on FILLED → Filled
    on OPEN → Fill
  }

  state Filled {
    ActivateGameObject(gameObject=var "Empty", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }

  state Fill {
    ActivateGameObject(gameObject=var "Empty", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Switch", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Empty: gameObject = Waterways_tank_anim/tank_empty
  var Return: gameObject = Waterways_tank_anim/tank_return
  var Switch: gameObject = Waterways_tank_anim/tank_switch
}
