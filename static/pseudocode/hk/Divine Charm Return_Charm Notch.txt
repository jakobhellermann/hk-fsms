# 1 location(s):

fsm Divine Charm Return {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="destroyedNightmareLantern", isTrue=(none), isFalse=→"INACTIVE")
    FindChild(gameObject=Self, childName="Shiny Item", storeResult=var "Shiny Item")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotGrimmNotch", isTrue=(none), isFalse=→"NOTCH")
    on INACTIVE → Inactive
    on NOTCH → Activate
  }

  state Inactive {
  }

  state Activate {
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Shiny Item: gameObject = <null>
}
