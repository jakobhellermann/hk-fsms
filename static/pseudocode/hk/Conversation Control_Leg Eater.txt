# Conversation Control
# 1 location(s):
#   Fungus2_26 / Leg Eater

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetLanguageString(sheetName="Prices", convName="LEGEATER_SOMETHING", storeValue=var "Cost String")
    ConvertStringToInt(stringVariable=var "Cost String", intVariable=var "Something Cost", everyFrame=false)
    FindGameObject(objectName="Shop Region", withTag="Untagged", store=var "Shop Region")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → All Gold?
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Self Anim
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    on FINISHED → Box Down
  }

  state Self Anim {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    on FINISHED → Box Up
  }

  state Convo Choice {
    AudioStop(gameObject=var "Voice Player")
    BoolTest(boolVariable=var "Ready To Leave", isTrue=→"READY TO LEAVE", isFalse=(none), everyFrame=false)
    PlayerDataBoolAllTrue(gameObject=var "GameManager", stringVariables=["fragileHealth_unbreakable", "fragileGreed_unbreakable", "fragileStrength_unbreakable"], trueEvent=→"ALL GOLD", falseEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="fragileHealth_unbreakable", falseBool="legEaterGoldConvo", isTrue=→"GOLD CONVO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="fragileGreed_unbreakable", falseBool="legEaterGoldConvo", isTrue=→"GOLD CONVO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="fragileStrength_unbreakable", falseBool="legEaterGoldConvo", isTrue=→"GOLD CONVO", isFalse=(none))
    GetPlayerDataBool(boolName="gotCharm_23", storeValue=var "Glass Charm1")
    GetPlayerDataBool(boolName="gotCharm_24", storeValue=var "Glass Charm2")
    GetPlayerDataBool(boolName="gotCharm_25", storeValue=var "Glass Charm3")
    GetPlayerDataBool(boolName="legEaterBoughtConvo", storeValue=var "Bought Convo")
    BoolAnyTrue(boolVariables=[var "Glass Charm1", var "Glass Charm2", var "Glass Charm3"], sendEvent=(none), storeResult=var "Bought Charm", everyFrame=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="LEGEATER", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metLegEater", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="paidLegEater", isTrue=(none), isFalse=→"REMEET")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_10", falseBool="legEaterDungConvo", isTrue=→"DUNG", isFalse=(none))
    BoolTestMulti(boolVariables=[var "Bought Charm", var "Bought Convo"], boolStates=[true, false], trueEvent=→"BOUGHT CHARM", falseEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="crossroadsInfected", falseBool="legEaterInfectedCrossroadConvo", isTrue=→"INFECTED CROSSROAD", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="legEaterConvo1", isTrue=(none), isFalse=→"CONVO 1")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="legEaterConvo2", isTrue=(none), isFalse=→"CONVO 2")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="legEaterConvo3", isTrue=(none), isFalse=→"CONVO 3")
    on MEET → Meet
    on REMEET → Remeet
    on FINISHED → Repeat
    on DUNG → Dung
    on CONVO 1 → Convo 1
    on CONVO 2 → Convo 2
    on CONVO 3 → Convo 3
    on INFECTED CROSSROAD → Infected Crossroad
    on BOUGHT CHARM → Bought Charm
    on GOLD CONVO → Gold Convo
    on ALL GOLD → All Gold
    on READY TO LEAVE → Ready To Leave
  }

  state Meet {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Leg_Eater_generic_01 (sharedassets189.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataBool(boolName="metLegEater", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_MEET", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Dial Box Down
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=var "Something Cost", everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_SOMETHING", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=3.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 3
  }

  state Dial Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up YN
  }

  state Remeet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_REMEET", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Leg_Eater_generic_01 (sharedassets189.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Dial Box Down
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enough Geo?
  }

  state First Decline {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_REFUSE", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="refusedLegEater", value=true)
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_angry_01 (sharedassets10.assets), Leg_Eater_angry_02 (sharedassets189.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 3
  }

  state Check Declined {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="refusedLegEater", isTrue=→"DECLINE 2", isFalse=→"DECLINE 1")
    on DECLINE 1 → First Decline
    on DECLINE 2 → Decline Again
  }

  state Talk Finish 3 {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Decline Again {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_REFUSE_AGAIN", "Minor NPC"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_angry_01 (sharedassets10.assets), Leg_Eater_angry_02 (sharedassets189.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 3
  }

  state Decline Pause {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Box Up 2
  }

  state Enough Geo? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="geo", storeValue=var "Player Geo")
    IntCompare(integer1=var "Player Geo", integer2=var "Something Cost", equal=→"ENOUGH", lessThan=→"NOT ENOUGH", greaterThan=→"ENOUGH", everyFrame=false)
    on ENOUGH → Check Declined
    on NOT ENOUGH → Not Enough Geo
  }

  state Not Enough Geo {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_NOTENOUGHGEO", "Minor NPC"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_angry_01 (sharedassets10.assets), Leg_Eater_angry_02 (sharedassets189.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 3
  }

  state Talk Finish 2 {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Something Bought {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_PAID", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="paidLegEater", value=true)
    SetBoxCollider2DSize(gameObject1=Self, width=2.43, height=0.24, offsetX=2.32, offsetY=-1.96)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_alert_02 (sharedassets189.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Dial Box Down 2
  }

  state Box Up 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Something Bought
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_GREET", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Leg_Eater_generic_05 (sharedassets189.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state Shop Opened? {
    ActivateGameObject(gameObject=var "Shop Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Shop Region", active=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="paidLegEater", isTrue=(none), isFalse=→"FINISHED")
    SetBoxCollider2DSize(gameObject1=Self, width=2.43, height=0.24, offsetX=2.32, offsetY=-1.96)
    ActivateGameObject(gameObject=var "Shop Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Shop Region", active=true)
    on FINISHED → Idle
  }

  state Open Shop {
    ActivateGameObject(gameObject=var "Shop Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Shop Region", active=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO CANCEL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Shop Region"), sendEvent="DO SHOP OPEN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Dial Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    on FINISHED → Open Shop
  }

  state Dung {
    SetPlayerDataBool(boolName="legEaterDungConvo", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_DUNG", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_generic_01 (sharedassets189.assets), Leg_Eater_generic_02 (sharedassets189.assets), Leg_Eater_generic_04 (sharedassets10.assets), Leg_Eater_generic_05 (sharedassets189.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state Convo 1 {
    SetPlayerDataBool(boolName="legEaterConvo1", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_generic_01 (sharedassets189.assets), Leg_Eater_generic_02 (sharedassets189.assets), Leg_Eater_generic_04 (sharedassets10.assets), Leg_Eater_generic_05 (sharedassets189.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_CONVO1", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish 2
  }

  state Convo 2 {
    SetPlayerDataBool(boolName="legEaterConvo2", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_CONVO2", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_generic_01 (sharedassets189.assets), Leg_Eater_generic_02 (sharedassets189.assets), Leg_Eater_generic_04 (sharedassets10.assets), Leg_Eater_generic_05 (sharedassets189.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state Convo 3 {
    SetPlayerDataBool(boolName="legEaterConvo3", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_CONVO3", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_generic_01 (sharedassets189.assets), Leg_Eater_generic_02 (sharedassets189.assets), Leg_Eater_generic_04 (sharedassets10.assets), Leg_Eater_generic_05 (sharedassets189.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state Bought Charm {
    SetPlayerDataBool(boolName="legEaterBoughtConvo", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_CONVO5", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_generic_01 (sharedassets189.assets), Leg_Eater_generic_02 (sharedassets189.assets), Leg_Eater_generic_04 (sharedassets10.assets), Leg_Eater_generic_05 (sharedassets189.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state Infected Crossroad {
    SetPlayerDataBool(boolName="legEaterInfectedCrossroadConvo", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEGEATER_CONVO4", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Leg_Eater_generic_01 (sharedassets189.assets), Leg_Eater_generic_02 (sharedassets189.assets), Leg_Eater_generic_04 (sharedassets10.assets), Leg_Eater_generic_05 (sharedassets189.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state Gold Convo {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEG_EATER_GOLD1", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Leg_Eater_generic_05 (sharedassets189.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataBool(boolName="legEaterGoldConvo", value=true)
    on CONVO_FINISH → Talk Finish 2
  }

  state All Gold {
    SetBoolValue(boolVariable=var "Ready To Leave", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="legEaterLeft", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEG_EATER_GOLD3", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Leg_Eater_generic_05 (sharedassets189.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state Ready To Leave {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LEG_EATER_GOLD_REPEAT", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Leg_Eater_generic_05 (sharedassets189.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish 2
  }

  state All Gold? {
    PlayerDataBoolAllTrue(gameObject=var "GameManager", stringVariables=["fragileHealth_unbreakable", "fragileGreed_unbreakable", "fragileStrength_unbreakable"], trueEvent=→"ALL GOLD", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on ALL GOLD → No Shop
    on FINISHED → Shop Opened?
  }

  state No Shop {
    SetCollider(gameObject=var "Shop Region", active=false)
    on FINISHED → Idle
  }

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var Player Geo: int = 0
  var Something Cost: int = 0
  var Bought Charm: bool = false
  var Bought Convo: bool = false
  var Glass Charm1: bool = false
  var Glass Charm2: bool = false
  var Glass Charm3: bool = false
  var Has Glass1: bool = false
  var Ready To Leave: bool = false
  var Cost String: string = ""
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Shop Region: gameObject = <null>
  var Voice Player: gameObject = <null>
}
