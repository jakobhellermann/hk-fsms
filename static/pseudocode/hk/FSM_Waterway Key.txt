// uses template: equipment_overmax_shift
fsm FSM {
  start Init

  state Init {
    on EQUIPMENT OVERMAX SHIFT → Shift
  }

  state Shift {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.7, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pause 2
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Shift
  }
}
