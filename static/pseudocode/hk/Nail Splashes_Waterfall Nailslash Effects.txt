# Nail Splashes
# 1 location(s):
#   Fungus2_06 / Waterfall Nailslash Effects

fsm Nail Splashes {
  start Init

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=(none), storeCollider=var "")  // disabled
    ReceivedDamage(collideTag="Nail Attack", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on HIT → Splosh
  }

  state Check Direction {
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Direction", everyFrame=false)
    FloatSwitch(floatVariable=var "Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on UP → Splosh
    on DOWN → Splosh
    on LEFT → Splosh
    on RIGHT → Splosh
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Slash Left {
    SpawnRandomObjects(gameObject=Spatter White (sharedassets170.assets), spawnPoint=var "Hero", position=(-2, -0.5, -0.01), spawnMin=6, spawnMax=8, speedMin=10, speedMax=20, angleMin=150, angleMax=120, originVariation=0)
    SpawnRandomObjects(gameObject=Spatter White (sharedassets170.assets), spawnPoint=var "Hero", position=(0.5, -0.25, -0.01), spawnMin=2, spawnMax=6, speedMin=9, speedMax=15, angleMin=20, angleMax=55, originVariation=0)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[water_cut_01 (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Water Cut (resources.assets), spawnPoint=var "Hero", position=(-3.3, -0.25, -0.01), rotation=(unset), storeObject=var "Effect", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Effect", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state Slash Right {
    SpawnRandomObjects(gameObject=Spatter White (sharedassets170.assets), spawnPoint=var "Hero", position=(2, -0.25, -0.01), spawnMin=6, spawnMax=8, speedMin=10, speedMax=20, angleMin=30, angleMax=60, originVariation=0)
    SpawnRandomObjects(gameObject=Spatter White (sharedassets170.assets), spawnPoint=var "Hero", position=(-0.5, -0.5, -0.01), spawnMin=2, spawnMax=6, speedMin=9, speedMax=15, angleMin=125, angleMax=160, originVariation=0)
    CreateObject(gameObject=Water Cut (resources.assets), spawnPoint=var "Hero", position=(3.3, -0.25, -0.01), rotation=(unset), storeObject=var "Effect", networkInstantiate=false, networkGroup=0)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[water_cut_01 (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetScale(gameObject=var "Effect", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state Slash Up {
    SpawnRandomObjectsV2(gameObject=Spatter White (sharedassets170.assets), spawnPoint=var "Hero", position=(0, -0.5, -0.01), spawnMin=18, spawnMax=20, speedMin=5, speedMax=25, angleMin=85, angleMax=95, originVariationX=0.75, originVariationY=0)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[water_cut_01 (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Splash Out Small (resources.assets), spawnPoint=var "Hero", position=(0, -1, 0), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Pause
  }

  state Splosh {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Splash Point", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[water_cut_01 (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Splash Point", position=(unset), spawnMin=15, spawnMax=20, speedMin=13, speedMax=19, angleMin=0, angleMax=360, originVariationX=0.4, originVariationY=0.4, FSM="", FSMEvent="")
    on FINISHED → Pause
  }

  state Init {
    FindChild(gameObject=Self, childName="Splash Point", storeResult=var "Splash Point")
    on FINISHED → Detect
  }

  var Direction: float = 0
  var Hero Y: float = 0
  var Damager: gameObject = <null>
  var Effect: gameObject = <null>
  var Splash Point: gameObject = <null>
}
