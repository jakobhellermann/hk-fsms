    AudioStop(gameObject=Self)

    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Get", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get 2
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Inert
  }

  state Get 2 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetPosition(gameObject=Self, vector=var "World Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)  // disabled
    on FINISHED → Inert
    on GEO GET → Get 1
  }

  var World Pos: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
}
