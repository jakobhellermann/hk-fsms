# Battle Control
# 1 location(s):
#   Ruins1_05 / Battle Scene v2

fsm Battle Control {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Wave 1
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="CameraLockArea B", withTag="Untagged", store=var "Camera Lock")
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Wave 2", storeResult=var "Wave 2")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activate
  }

  state Wave 1 {
    TransitionToAudioSnapshot(snapshot=Action and Sub (resources.assets), transitionTime=1)
    SetIntValue(intVariable=var "Battle Enemies", intValue=1, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    ActivateGameObject(gameObject=var "Camera Lock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "W1 SentryFat", vector=(unset), x=16.15, y=152.68, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "W1 SentryFat"), sendEvent="BATTLE SPAWN", delay=0, everyFrame=false)
    on END → W2 Pause
  }

  state End Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Sub Area (resources.assets), transitionTime=5)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wave 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wave 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Activated", storeResult=var "Activated Folder")
    ActivateAllChildren(gameObject=var "Activated Folder", activate=true)
  }

  state Wave 2 {
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    ActivateAllChildren(gameObject=var "Wave 2", activate=true)  // disabled
    SetPosition(gameObject=var "W2 Sentry", vector=(unset), x=32.32, y=151.88, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "W2 Sentry"), sendEvent="BATTLE SPAWN", delay=0, everyFrame=false)
    SetPosition(gameObject=var "W2 SentryFly", vector=(unset), x=18.93, y=160.8, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on END → End Wait
  }

  state W2 Pause {
    Wait(time=1.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 2
  }

  var W1 SentryFat: gameObject = Battle Scene v2/Wave 1/Ruins Sentry FatB
  var W2 Sentry: gameObject = Battle Scene v2/Wave 2/Ruins SentryB 1
  var W2 SentryFly: gameObject = Battle Scene v2/Wave 2/Ruins Flying SentryB

  var Battle Enemies: int = 0
  var Activated: bool = false
  var Activated Folder: gameObject = <null>
  var Camera Lock: gameObject = <null>
  var Moss Knight: gameObject = <null>
  var Self: gameObject = <null>
  var Wave 1: gameObject = <null>
  var Wave 2: gameObject = <null>
}
