# Control
# 1 location(s):
#   resources.assets / Orbit Shield

fsm Control {
  start Init
  on ALL CHARMS END → Disappear  // from any state

  state Init {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FindChild(gameObject=Self, childName="Shield", storeResult=var "Shield")
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "Orbit Shield", gameObject=var "Self", everyFrame=false)
    on FINISHED → Follow
  }

  state Follow {
    GetPosition2D(gameObject=var "", vector=(unset), x=(unset), y=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    Vector3AddXYZ(vector3Variable=var "Hero Pos", addX=0, addY=-0.5, addZ=0, everyFrame=false, perSecond=false)
    SetPosition(gameObject=Self, vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=true)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Speed", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on CHARM EQUIP CHECK → Still equipped?
    on SLASH → Send Event
  }

  state Still equipped? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_38", isTrue=→"YES", isFalse=→"NO")
    on YES → Check Combo
    on NO → Disappear
  }

  state Disappear {
    SendEventByName(eventTarget=GameObject(var "Shield"), sendEvent="DISAPPEAR", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(var "Shield"), sendEvent="SLASH", delay=0, everyFrame=false)
    on FINISHED → Follow
  }

  state Check Combo {
    SendEventByName(eventTarget=GameObject(var "Shield"), sendEvent="CHECK COMBO", delay=0, everyFrame=false)
    on FINISHED → Follow
  }

  var Speed: float = 110
  var Hero Pos: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
  var Shield: gameObject = <null>
}
