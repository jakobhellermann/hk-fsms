# Beam Miner
# 3 location(s):
#   Mines_23 / Zombie Beam Miner
#   Mines_24 / Zombie Beam Miner
#   Mines_25 / Zombie Beam Miner

fsm Beam Miner {
  start Pause Frame
  on TOOK DAMAGE → Cancel  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Beam Impact", childName="Impact Particle", storeResult=var "Impact Particle")
    FindChild(gameObject=Self, childName="Beam", storeResult=var "Beam")
    FindChild(gameObject=Self, childName="Beam Ball", storeResult=var "Beam Ball")
    FindChild(gameObject=Self, childName="Beam Impact", storeResult=var "Beam Impact")
    SetScale(gameObject=var "Beam Ball", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Beam Impact", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FindChild(gameObject=var "Beam Impact", childName="Init Particle", storeResult=var "Init Particle")
    FindChild(gameObject=var "Beam Ball", childName="End Particle", storeResult=var "End Particle")
    FindChild(gameObject=var "Beam", childName="Beam Glow", storeResult=var "Beam Glow")
    FindChild(gameObject=var "Beam Ball", childName="End Particle", storeResult=var "End Particle")
    FindChild(gameObject=var "Beam", childName="Beam Glow", storeResult=var "Beam Glow")
    FindChild(gameObject=var "Beam Impact", childName="Init Particle", storeResult=var "Init Particle")
    FindChild(gameObject=var "Beam Impact", childName="Impact Particle", storeResult=var "Impact Particle")
    FindChild(gameObject=var "Beam Impact", childName="Burn Particle", storeResult=var "Burn Particle")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Beam Range", isTrue=→"ATTACK", isFalse=(none), everyFrame=true)
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on ATTACK → Face 2
  }

  state Aim {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetAngleToTarget2D(gameObject=var "Laser Origin", target=var "Hero", offsetX=0, offsetY=-0.5, storeAngle=var "Angle", everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)
    ReflectAngle(angle=var "Angle", reflectHorizontally=true, reflectVertically=true, storeResult=var "Angle")
    on FINISHED → Rotate Beam
  }

  state CD {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Facing
  }

  state Cancel {
    AudioStop(gameObject=Self)
    AudioStop(gameObject=var "Beam Ball")
    SetMeshRenderer(gameObject=var "Beam Impact", active=false)
    StopParticleEmitter(gameObject=var "Impact Particle")
    SetMeshRenderer(gameObject=var "Beam Ball", active=false)
    SetMeshRenderer(gameObject=var "Beam", active=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Beam", clipName="Beam End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Beam Glow"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Check Facing {
    SetWalkerFacing(walkRight=false, randomStartDir=false, target=Self, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignTest(floatValue=var "Scale", isPositive=→"FINISHED", isNegative=(none), everyFrame=false)
    SetWalkerFacing(walkRight=true, randomStartDir=false, target=Self, everyFrame=false)
    on FINISHED → Run?
  }

  state Face 2 {
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "X Distance", everyFrame=false)
    FloatCompare(float1=var "X Distance", float2=3, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on FINISHED → Aim
    on CANCEL → Attack Cancel
  }

  state Attack Cancel {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    WaitRandom(timeMin=0.3, timeMax=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Run? {
    SendRandomEvent(events=[→"FINISHED", →"RUN"], weights=[1, 1], delay=0)
    on FINISHED → Idle
    on RUN → Attack Cancel
  }

  state Beam Antic {
    SendEventByName(eventTarget=GameObject(var "Beam"), sendEvent="START", delay=0, everyFrame=false)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mines_pink_laser_prepare (sharedassets252.assets))
    PlayParticleEmitter(gameObject=var "Init Particle", emit=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Loop", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=var "Beam Ball", active=true)
    SetMeshRenderer(gameObject=var "Beam", active=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Beam Ball", clipName="Ball Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=var "Beam", animLibName="", clipName="Beam Antic")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    GetPosition(gameObject=var "Beam", vector=(unset), x=var "Ray Origin X", y=var "Ray Origin Y", z=(unset), space=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "Ray Origin", vector2Value=(unset), x=var "Ray Origin X", y=var "Ray Origin Y", everyFrame=false)
    RayCast2d(fromGameObject=Self, fromPosition=var "Ray Origin", direction=(unset), space=0, distance=100, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Beam
  }

  state Beam {
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mega_laser_burst (sharedassets261.assets))
    Tk2dPlayAnimation(gameObject=var "Beam Ball", animLibName="", clipName="Ball Shoot")
    Tk2dPlayAnimation(gameObject=var "Beam", animLibName="", clipName="Beam Shoot")
    SendEventByName(eventTarget=GameObject(var "Beam Glow"), sendEvent="UP", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Impact Particle", emit=0)
    StopParticleEmitter(gameObject=var "Init Particle")
    SetMeshRenderer(gameObject=var "Beam Impact", active=true)
    AudioPlaySimple(gameObject=var "Beam Ball", volume=1, oneShotClip=<null>)
    SetAudioVolume(gameObject=var "Beam Ball", volume=1, everyFrame=false)
    on FINISHED → Beam End
  }

  state Beam End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Beam Ball", clipName="Ball End", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=var "Beam", animLibName="", clipName="Beam End")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "End Particle", emit=0)
    SendEventByName(eventTarget=GameObject(var "Beam Glow"), sendEvent="DOWN", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Impact Particle")
    StopParticleEmitter(gameObject=var "Burn Particle")  // disabled
    SetMeshRenderer(gameObject=var "Beam Impact", active=false)
    FadeAudio(gameObject=var "Beam Ball", startVolume=1, endVolume=0, time=0.15)
    on FINISHED → Beam Off
  }

  state Beam Off {
    SetMeshRenderer(gameObject=var "Beam Ball", active=false)
    SetMeshRenderer(gameObject=var "Beam", active=false)
    AudioStop(gameObject=var "Beam Ball")
    SendEventByName(eventTarget=GameObject(var "Beam"), sendEvent="STOP", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → CD
  }

  state Rotate Beam {
    SetRotation(gameObject=var "Beam", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Beam Ball", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Beam Impact", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Beam Antic
  }

  var Angle: float = 0
  var Laser Origin X: float = 0
  var Laser Origin Y: float = 0
  var Ray Distance: float = 0
  var Ray Hit X: float = 0
  var Ray Hit Y: float = 0
  var Ray Origin X: float = 0
  var Ray Origin Y: float = 0
  var Ray Vector X: float = 0
  var Ray Vector Y: float = 0
  var Scale: float = 0
  var Self X: float = 0
  var Self X Scale: float = 0
  var Self Y: float = 0
  var X Distance: float = 0
  var Beam Range: bool = false
  var Ray Hit Point: vector2 = (0, 0)
  var Ray Origin: vector2 = (0, 0)
  var Ray Vector: vector2 = (0, 0)
  var Hero Pos: vector3 = (0, 0, 0)
  var Impact Pos: vector3 = (0, 0, 0)
  var Laser Origin Pos: vector3 = (0, 0, 0)
  var Beam: gameObject = <null>
  var Beam Ball: gameObject = <null>
  var Beam Glow: gameObject = <null>
  var Beam Impact: gameObject = <null>
  var Burn Particle: gameObject = <null>
  var End Particle: gameObject = <null>
  var Impact Particle: gameObject = <null>
  var Init Particle: gameObject = <null>
  var Laser Origin: gameObject = Zombie Beam Miner/Laser Origin
  var Self: gameObject = <null>
}
