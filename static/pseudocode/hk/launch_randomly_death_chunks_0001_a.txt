// uses template: launch_randomly
fsm launch_randomly {
  start Launch

  state Launch {
    GetOwner(storeGameObject=var "Self")
    RandomFloat(min=var "Angle Min", max=var "Angle Max", storeResult=var "Angle")
    RandomFloat(min=var "Speed Min", max=var "Speed Max", storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=var "Self", angle=var "Angle", speed=var "Speed", everyFrame=false)
  }

  var Angle Max: float = 130
  var Angle Min: float = 110
  var Speed Max: float = 15
  var Speed Min: float = 15

  var Angle: float = 0
  var Speed: float = 0
  var Self: gameObject = <null>
}
