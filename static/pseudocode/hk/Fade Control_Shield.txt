# Fade Control
# 3 location(s):
#   Ruins2_Watcher_Room / Dreamer Lurien/Shield
#   Fungus3_archive_02 / Dreamer Monomon/Shield
#   Deepnest_Spider_Town / Dreamer Hegemol/Shield

fsm Fade Control {
  start Init

  state Init {
    BoolTest(boolVariable=var "Hero Range", isTrue=→"ENTER", isFalse=(none), everyFrame=true)
    on ENTER → Enter
  }

  state Enter {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"EXIT", everyFrame=true)
    on EXIT → Exit
  }

  state Exit {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Hero Range", isTrue=→"ENTER", isFalse=(none), everyFrame=true)
    on ENTER → Enter
  }

  var Hero Range: bool = false
}
