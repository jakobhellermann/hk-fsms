# Set Rage
# 1 location(s):
#   GG_Oblobbles / Mega Fat Bee

fsm Set Rage {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    on OBLOBBLE RAGE → Set
  }

  state Set {
    SetFsmBool(gameObject=Self, fsmName="fat fly bounce", variableName="Rage", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Fatty Fly Attack", variableName="Rage", setValue=true, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="fat fly bounce", variableName="Speed", setValue=12, everyFrame=false)
    GetHP(target=Self, storeValue=var "HP")
    IntAdd(intVariable=var "HP", add=var "HP Add", everyFrame=false)
    IntClamp(intVariable=var "HP", minValue=10, maxValue=var "HP Max", everyFrame=false)
    SetHP(target=Self, hp=var "HP")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="RAGING", delay=0, everyFrame=false)
    on FINISHED → Roar Antic
  }

  state Roar Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.85)
    on FINISHED → Roar
  }

  state Roar {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dung_defender_call (sharedassets33.assets), pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shoot")
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar End
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="RAGE END", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
  }

  var HP Add: int = 200
  var HP Max: int = 750

  var HP: int = 0
  var Roar Emitter: gameObject = <null>
  var Self: gameObject = <null>
}
