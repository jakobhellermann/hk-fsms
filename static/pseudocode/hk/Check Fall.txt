# Check Fall
# 1 location(s):
#   Crossroads_ShamanTemple / _Props/Knight Get Fireball/Knight Cutscene Animator

fsm Check Fall {
  start Fall

  state Fall {
    GetHero(storeResult=var "Hero")
    FindChild(gameObject=Self, childName="Land", storeResult=var "Land Efefct")
    FindChild(gameObject=Self, childName="Corpse Flame", storeResult=var "Flame")
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"FINISHED", storeCollider=var "", storeForce=(unset))
    Collision2dEvent(collision=0, collideTag="TileMap", sendEvent=→"FINISHED", storeCollider=var "", storeForce=(unset))
    on FINISHED → Land
  }

  state Land {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="MainCamera", store=var "MainCamera")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hero_land_hard (sharedassets76.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Collect Magical Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Black
  }

  state Fade Back {
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Fade Colour", time=3, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=var "HUD Blanker", materialIndex=0, material=<null>, namedColor="_Color", color=var "Fade Colour", everyFrame=true)
    CreateObject(gameObject=Shaman Trapped (sharedassets76.assets), spawnPoint=<null>, position=(39.15, 8.84, 0), rotation=(unset), storeObject=var "Shaman NPC", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Shaman NPC", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FindGameObject(objectName="Shaman Meeting", withTag="Untagged", store=var "Prev Shaman")
    DestroyObject(gameObject=var "Prev Shaman", delay=0, detachChildren=false)
    Wait(time=3.5, finishEvent=→"WAIT", realTime=false)
    on WAIT → Wait For Input
  }

  state Wait For Input {
    SetSpriteRenderer(gameObject=var "HUD Blanker", active=false)
    ListenForRight(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForLeft(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForUp(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on GET UP → Get Up
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Prostrate Rise", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Regain Control
  }

  state Regain Control {
    SetPlayerDataInt(intName="shaman", value=2)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    SetPlayerDataBool(boolName="disablePause", value=false)
    DestroySelf(detachChildren=false)
  }

  state Black {
    SetMaterialColor(gameObject=var "HUD Blanker", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    SetSpriteRenderer(gameObject=var "HUD Blanker", active=true)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SetScale(gameObject=var "Hero", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Prostrate")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="FaceRight", parameters=[], storeResult=(unset var))
    SetGravity2dScale(gameObject=var "Hero", gravityScale=0.79)
    SetPosition(gameObject=var "Hero", vector=(unset), x=46, y=8.2, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="FindGroundPoint", parameters=[(46, 8.2, 0, 0), false], storeResult=var "Hero New Pos")
    SetPosition(gameObject=var "Hero", vector=var "Hero New Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=SetBackOnGround())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=AddMPCharge(100))
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Respawns
  }

  state Set Respawns {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetBenchRespawn", parameters=["Death Respawn Marker", "Crossroads_ShamanTemple", 0, true], storeResult=(unset var))
    SetPlayerDataBool(boolName="hasSpell", value=true)
    SetPlayerDataInt(intName="fireballLevel", value=1)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetHazardRespawn", parameters=[(47.44, 8.58, 0, 0), true], storeResult=(unset var))
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SaveGame())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=TimePasses())
    on FINISHED → Spawn Msg
  }

  state Spawn Msg {
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=Fireball_prompt (sharedassets76.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Fireball", everyFrame=false)
    on GET ITEM MSG END → Fade Back
  }

  var Axis Hoz: float = 0
  var Axis Vert: float = 0
  var Hero Ground Y: float = 0
  var Hero New Pos: vector3 = (0, 0, 0)
  var Fade Colour: color = (1, 1, 1, 1)
  var Flame: gameObject = <null>
  var Hero: gameObject = <null>
  var Land Efefct: gameObject = <null>
  var MainCamera: gameObject = <null>
  var New Shaman: gameObject = <null>
  var Prev Shaman: gameObject = <null>
  var Self: gameObject = <null>
  var Shaman NPC: gameObject = <null>
  var UI Msg: gameObject = <null>
}
