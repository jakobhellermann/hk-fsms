# Spore Cooldown
# 2 location(s):
#   Knight_Pickup / Knight
#   resources.assets / Knight:1

fsm Spore Cooldown {
  start Init

  state Idle {
    on SPORE ATTACK → Cooldown
  }

  state Cooldown {
    Wait(time=4.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset
  }

  state Reset {
    SetFsmBool(gameObject=Self, fsmName="Spell Control", variableName="Spore Cooldown", setValue=false, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_10", isTrue=→"DUNG", isFalse=(none))
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashSporeQuick())
    PlayParticleEmitter(gameObject=var "Spore Recharge", emit=0)
    on FINISHED → Idle
    on DUNG → Dung
  }

  state Init {
    FindChild(gameObject=Self, childName="Charm Effects", storeResult=var "Charm Effects")
    FindChild(gameObject=var "Charm Effects", childName="Spore Recharge", storeResult=var "Spore Recharge")
    FindChild(gameObject=var "Charm Effects", childName="Dung Recharge", storeResult=var "Dung Recharge")
    on FINISHED → Idle
  }

  state Dung {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashDungQuick())
    PlayParticleEmitter(gameObject=var "Dung Recharge", emit=0)
    on FINISHED → Init
  }

  var Charm Effects: gameObject = <null>
  var Dung Recharge: gameObject = <null>
  var Spore Recharge: gameObject = <null>
}
