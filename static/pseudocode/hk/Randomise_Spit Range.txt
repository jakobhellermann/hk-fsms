# Randomise
# 6 location(s):
#   Fungus1_21 / Moss Knight/Spit Range
#   Fungus1_21 / Battle Scene/Moss Knight/Spit Range
#   Fungus1_21 / Moss Knight (1)/Spit Range
#   Fungus1_26 / Moss Knight/Spit Range
#   Fungus1_32 / Moss Knight B/Spit Range
#   Fungus1_32 / Moss Knight C/Spit Range

fsm Randomise {
  start On
  on OFF → To Off  // from any state

  state On {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Off
  }

  state Off {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Range Detect", variableName="In Alert Range", setValue=false, everyFrame=false)
    on FINISHED → On
  }

  state To Off {
    SetCollider(gameObject=Self, active=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Range Detect", variableName="In Alert Range", setValue=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Off
  }
}
