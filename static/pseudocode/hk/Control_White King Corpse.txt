# Control
# 1 location(s):
#   White_Palace_09 / White King Corpse

fsm Control {
  start Initiate

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Throne Sit", storeResult=var "Throne Sit")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Hits
  }

  state Hits {
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=3, equal=→"BREAK", lessThan=→"HIT", greaterThan=→"BREAK", everyFrame=false)
    on HIT → Hit
    on BREAK → Break
  }

  state Hit {
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Hit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SpawnRandomObjects(gameObject=Zote Chip 2 (sharedassets148.assets), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=3, speedMin=15, speedMax=20, angleMin=0, angleMax=306, originVariation=0)
    SpawnRandomObjects(gameObject=Zote Chip 1 (sharedassets148.assets), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=2, speedMin=15, speedMax=20, angleMin=0, angleMax=306, originVariation=0)
    on FINISHED → Idle
  }

  state Break {
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Break", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SpawnRandomObjects(gameObject=Zote Chip 2 (sharedassets148.assets), spawnPoint=var "Self", position=(unset), spawnMin=4, spawnMax=4, speedMin=15, speedMax=20, angleMin=0, angleMax=306, originVariation=0)
    SpawnRandomObjects(gameObject=Zote Chip 1 (sharedassets148.assets), spawnPoint=var "Self", position=(unset), spawnMin=4, spawnMax=4, speedMin=15, speedMax=20, angleMin=0, angleMax=306, originVariation=0)
    on FINISHED → Spawn Charm
  }

  state Spawn Charm {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WHITE CHARM SPAWN", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"THRONE", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_wall_land (sharedassets15.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on THRONE → Activate Throne
  }

  state Activate Throne {
    ActivateGameObject(gameObject=var "Throne Sit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Hits: int = 0
  var Dream Dialogue: gameObject = <null>
  var Self: gameObject = <null>
  var Sprite: gameObject = White King Corpse/Sprite
  var Throne Sit: gameObject = <null>
}
