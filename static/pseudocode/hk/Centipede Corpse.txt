# Centipede Corpse
# 1 location(s):
#   sharedassets278.assets / Corpse Centipede Baby

fsm Centipede Corpse {
  start Wait 1

  state Wait 1 {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → New Birth
  }

  state New Birth {
    SpawnObjectFromGlobalPool(gameObject=Baby Centipede, spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "New Centipede")
    SetHP(target=var "New Centipede", hp=20)
    SetFsmBool(gameObject=var "New Centipede", fsmName="Centipede", variableName="Warps", setValue=true, everyFrame=false)
    WaitRandom(timeMin=20, timeMax=25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Steam
  }

  state Shrink {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(0, 0, 0, 1), duration=0.5)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=0.8, everyFrame=false)  // disabled
    SetVector3XYZ(vector3Variable=var "Shrink Scale", vector3Value=(unset), x=var "X Scale", y=0.1, z=1, everyFrame=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Shrink Scale", time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Disappear
  }

  state Disappear {
    StopParticleEmitter(gameObject=var "Shrink Steam")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Start Steam {
    PlayParticleEmitter(gameObject=var "Shrink Steam", emit=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shrink
  }

  var X Scale: float = 0
  var Shrink Scale: vector3 = (0, 0, 0)
  var New Centipede: gameObject = <null>
  var Shrink Steam: gameObject = Corpse Centipede Baby/Shrink Steam
}
