# Spawn Drops
# 1 location(s):
#   sharedassets50.assets / Dripper Black

fsm Spawn Drops {
  start Init

  state Spawn {
    SpawnRandomObjectsV2(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=12, spawnMax=15, speedMin=5, speedMax=12, angleMin=80, angleMax=110, originVariationX=1, originVariationY=0.5)  // disabled
    SpawnRandomObjectsOverTimeV2(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), frequency=0.025, spawnMin=1, spawnMax=1, speedMin=0, speedMax=1, angleMin=90, angleMax=90, originVariationX=0.25, originVariationY=0.5, scaleMin=1, scaleMax=1)  // disabled
    FlingObjectsFromGlobalPoolTime(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), frequency=0.025, spawnMin=1, spawnMax=1, speedMin=0, speedMax=1, angleMin=90, angleMax=90, originVariationX=0.5, originVariationY=0.5)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    SetParent(gameObject=Self, parent=var "OriginalParent", resetLocalPosition=true, resetLocalRotation=true)  // disabled
    on FINISHED → Recycle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    SetParent(gameObject=Self, parent=var "Hero", resetLocalPosition=true, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=-0.5, z=0.01, space=1, everyFrame=false, lateUpdate=false)
    Wait(time=0.04, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Recycle {
    RecycleSelf()
  }

  var Emission Rate: float = 0
  var Frequency: float = 0
  var Hero Y Velocity: float = 0
  var Hero: gameObject = <null>
  var OriginalParent: gameObject = <null>
  var Self: gameObject = <null>
}
