# Detect Hero
# 5 location(s):
#   Deepnest_33 / Zote Deepnest/Faller/NPC/Ranges/Look Range
#   Room_Colosseum_02 / Zote Colosseum/Ranges/Look Range
#   Town / _NPCs/Zote Town/Ranges/Look Range
#   Ruins1_06 / Zote Ruins/Ranges/Look Range
#   sharedassets148.assets / Zote Buzzer Convo/Ranges/Look Range

fsm Detect Hero {
  start Init

  state Out {
    Tk2dPlayAnimation(gameObject=var "Parent NPC", animLibName="", clipName="Idle")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on RIGHT → Right
    on LEFT → Left
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetParent(gameObject=var "Parent", storeResult=var "Parent NPC")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
    on EXIT → Out
  }

  state Right {
    Tk2dPlayAnimation(gameObject=var "Parent NPC", animLibName="", clipName="Look Right")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
    on LEFT → Left
  }

  state Left {
    Tk2dPlayAnimation(gameObject=var "Parent NPC", animLibName="", clipName="Look Left")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
    on RIGHT → Right
  }

  var Hero In Range: bool = false
  var Talking: bool = false
  var Bench Colour: color = (0, 0, 0, 1)
  var Bench Prev Colour: color = (0, 0, 0, 1)
  var Colour: color = (0, 0, 0, 1)
  var Prev Colour: color = (0, 0, 0, 1)
  var Light: gameObject = <null>
  var Lit: gameObject = <null>
  var Parent: gameObject = <null>
  var Parent NPC: gameObject = <null>
  var Particle B: gameObject = <null>
  var Particle F: gameObject = <null>
}
