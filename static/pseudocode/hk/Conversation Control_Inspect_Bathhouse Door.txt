# Conversation Control
# 1 location(s):
#   Ruins2_04 / Bathhouse Door/Inspect

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Got Key?
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Anim End {
    on FINISHED → Box Down
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["BATH_HOUSE_KEY", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Decline Pause {
    Wait(time=0.3, finishEvent=→"TALK FINISH", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → 
  }

  state Open {
    SetPlayerDataBool(boolName="bathHouseOpened", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bathhouse_door_open (sharedassets9.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    IntOperator(integer1=var "Simple Keys", integer2=1, operation=Subtract, storeResult=var "Simple Keys", everyFrame=false)
    SetPlayerDataInt(intName="simpleKeys", value=var "Simple Keys")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Closed", storeResult=var "Door Closed")
    FindChild(gameObject=var "Parent", childName="Open", storeResult=var "Door Open")
    FindChild(gameObject=var "Parent", childName="door_Ruin_Elevator", storeResult=var "Door Object")
    ActivateGameObject(gameObject=var "Door Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Door Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  state Finish {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=Self, active=false)
    SetCollider(gameObject=var "Door Object", active=true)
    on FINISHED → Anim End
  }

  state Got Key? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="simpleKeys", storeValue=var "Simple Keys")
    IntCompare(integer1=var "Simple Keys", integer2=0, equal=→"NO", lessThan=→"NO", greaterThan=→"YES", everyFrame=false)
    on YES → Box Up YN
    on NO → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → No Key
  }

  state No Key {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BATH_HOUSE_NOKEY", "Prompts"], storeResult=(unset var))
    on CONVO_FINISH → Anim End
  }

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var History: int = 0
  var Simple Keys: int = 0
  var Something Cost: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var mapCrossroads: bool = false
  var Area Title: gameObject = <null>
  var Door Closed: gameObject = <null>
  var Door Object: gameObject = <null>
  var Door Open: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
