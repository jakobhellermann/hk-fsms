# Conversation Control
# 1 location(s):
#   Ruins1_05b / Relic Dealer

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → NPC Anim
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="RELICDEALER", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Shop Intro {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["RELICDEALER_SHOP_INTRO", "Relic Dealer"], storeResult=(unset var))
    SetPlayerDataBool(boolName="metRelicDealerShop", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_02 (sharedassets93.assets), Relic_Dealer_01 (sharedassets10.assets), Relic_Dealer_03 (sharedassets10.assets), Relic_Dealer_05 (sharedassets93.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Inside R")
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state NPC Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Inside TalkR")
    on FINISHED → Box Up
  }

  state Convo Choice {
    GetPlayerDataBool(boolName="honedNail", storeValue=var "Has Upgraded")
    GetPlayerDataBool(boolName="marmConvoNailsmith", storeValue=var "Nail Convo")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_10", isTrue=→"DUNG", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metRelicDealerShop", isTrue=(none), isFalse=→"MEET")
    BoolTestMulti(boolVariables=[var "Has Upgraded", var "Nail Convo"], boolStates=[false, false], trueEvent=→"NAIL", falseEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="marmConvo1", isTrue=(none), isFalse=→"CONVO1")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="marmConvo2", isTrue=(none), isFalse=→"CONVO2")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasSuperDash", falseBool="marmConvo3", isTrue=→"CONVO3", isFalse=(none))
    on MEET → Shop Intro
    on CONVO1 → Convo1
    on CONVO2 → Convo2
    on CONVO3 → Convo3
    on FINISHED → Repeat
    on DUNG → Dung
    on NAIL → Nailsmith
  }

  state Convo1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["RELICDEALER_CONVO_1", "Relic Dealer"], storeResult=(unset var))
    SetPlayerDataBool(boolName="marmConvo1", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_02 (sharedassets93.assets), Relic_Dealer_01 (sharedassets10.assets), Relic_Dealer_03 (sharedassets10.assets), Relic_Dealer_05 (sharedassets93.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Convo2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["RELICDEALER_TALK", "Relic Dealer"], storeResult=(unset var))
    SetPlayerDataBool(boolName="marmConvo2", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_02 (sharedassets93.assets), Relic_Dealer_01 (sharedassets10.assets), Relic_Dealer_03 (sharedassets10.assets), Relic_Dealer_05 (sharedassets93.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Convo3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["RELICDEALER_TALK_2", "Relic Dealer"], storeResult=(unset var))
    SetPlayerDataBool(boolName="marmConvo3", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_02 (sharedassets93.assets), Relic_Dealer_01 (sharedassets10.assets), Relic_Dealer_03 (sharedassets10.assets), Relic_Dealer_05 (sharedassets93.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["RELICDEALER_TALK_REPEAT", "Relic Dealer"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_06 (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Dung {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["RELICDEALER_DUNG", "Relic Dealer"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_04 (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Nailsmith {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["RELICDEALER_NAILSMITH", "Relic Dealer"], storeResult=(unset var))
    SetPlayerDataBool(boolName="marmConvoNailsmith", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_02 (sharedassets93.assets), Relic_Dealer_01 (sharedassets10.assets), Relic_Dealer_03 (sharedassets10.assets), Relic_Dealer_05 (sharedassets93.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  var Has Upgraded: bool = false
  var Nail Convo: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
