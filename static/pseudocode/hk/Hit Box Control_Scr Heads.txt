# Hit Box Control
# 2 location(s):
#   Knight_Pickup / Knight/Spells/Scr Heads
#   resources.assets / Knight:1/Spells/Scr Heads

fsm Hit Box Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Hit L", storeResult=var "Hit L")
    FindChild(gameObject=Self, childName="Hit R", storeResult=var "Hit R")
    FindChild(gameObject=Self, childName="Hit R", storeResult=var "Hit U")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Scream1", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero X scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Neg Scale", vector3Value=var "Init Scale", everyFrame=false)
    GetVector3XYZ(vector3Variable=var "Neg Scale", storeX=var "X Scale", storeY=(unset), storeZ=(unset), everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Neg Scale", vector3Value=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false)
    on FINISHED → Check Scale
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Deactivate Hit
  }

  state Deactivate Hit {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Deactivate Self
  }

  state Check Scale {
    FloatCompare(float1=var "Hero X scale", float2=0, tolerance=0, equal=(none), lessThan=→"NEG", greaterThan=→"POS", everyFrame=false)
    on NEG → Neg
    on POS → Pos
  }

  state Neg {
    SetScale(gameObject=Self, vector=var "Neg Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Activate
  }

  state Pos {
    SetScale(gameObject=Self, vector=var "Init Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Activate
  }

  state Deactivate Self {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
  }

  state Start Anim {
    Wait(time=0.06, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  var Hero X scale: float = 0
  var X Scale: float = 0
  var Init Scale: vector3 = (1.3, 1.3, 1.3)
  var Neg Scale: vector3 = (-1.664587, 1.664587, 1.664587)
  var Hit L: gameObject = <null>
  var Hit R: gameObject = <null>
  var Hit U: gameObject = <null>
  var Self: gameObject = <null>
}
