# Control
# 1 location(s):
#   Fungus3_archive_02 / Dreamer Monomon/Quirrel

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Mask Anim", storeResult=var "Mask Anim")
    FindChild(gameObject=Self, childName="Dust Land Small", storeResult=var "Land Dust")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    on FINISHED → Leap Ready
  }

  state Leap Ready {
    on TANK INSPECT → Leap Pause
  }

  state Leap Pause {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Leap In
  }

  state Leap In {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Leap In", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_jump (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetMeshRenderer(gameObject=Self, active=true)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-15, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 15, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Leaped In Pause
  }

  state Leaped In Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Mask Anim", active=true)
    Tk2dPlayAnimation(gameObject=var "Mask Anim", animLibName="", clipName="Stand")
    PlayParticleEmitter(gameObject=var "Land Dust", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_ground_land (sharedassets28.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Talk Active
  }

  state Talk Active {
    SetCollider(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetPosition(gameObject=var "Dream Dialogue", vector=(unset), x=0, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  var Dream Dialogue: gameObject = <null>
  var Land Dust: gameObject = <null>
  var Mask Anim: gameObject = <null>
  var Self: gameObject = <null>
}
