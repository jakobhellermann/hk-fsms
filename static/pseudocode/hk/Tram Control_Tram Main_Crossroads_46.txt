# Tram Control
# 1 location(s):
#   Crossroads_46 / Tram Main

fsm Tram Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName=var "PD Position Int", storeValue=var "Tram Position")
    GetPlayerDataBool(boolName=var "PD Opened Bool", storeValue=var "Tram Opened")
    IntCompare(integer1=var "Tram Position", integer2=var "Station Position", equal=→"HERE", lessThan=→"AWAY LEFT", greaterThan=→"AWAY RIGHT", everyFrame=false)
    on AWAY LEFT → Set Left
    on AWAY RIGHT → Set Right
    on HERE → Check Opened
  }

  state Check Opened {
    BoolTest(boolVariable=var "Tram Opened", isTrue=→"OPEN", isFalse=→"CLOSED", everyFrame=false)
    on OPEN → Opened
    on CLOSED → Closed
  }

  state Set Left {
    Translate(gameObject=Self, vector=(unset), x=-40, y=40, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetVector3XYZ(vector3Variable=var "Tween Vector", vector3Value=(unset), x=40, y=(unset), z=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Door Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Away
  }

  state Set Right {
    Translate(gameObject=Self, vector=(unset), x=40, y=40, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetVector3XYZ(vector3Variable=var "Tween Vector", vector3Value=(unset), x=-40, y=(unset), z=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Door Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Away
  }

  state Away {
    ActivateGameObject(gameObject=var "Door Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CALL TRAM → Tween In
  }

  state Closed {
    ActivateGameObject(gameObject=var "Door Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Opened {
    SendEventByName(eventTarget=GameObject(var "Interior Lights"), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    Translate(gameObject=var "Door", vector=(unset), x=-3.47, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    ActivateGameObject(gameObject=var "Door Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Tween In {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-40, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    iTweenMoveBy(gameObject=Self, id="", vector=var "Tween Vector", time=5, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Dust", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Rocks", emit=0)
    on FINISHED → Tween End
  }

  state Door Open {
    AudioStop(gameObject=Self)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=toll_bench_open (sharedassets23.assets))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=ruins_big_lift_gate_door_thingy_open (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.15, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Interior Lights"), sendEvent="UP", delay=0.5, everyFrame=false)
    iTweenMoveBy(gameObject=var "Door", id="", vector=(-3.47, 0, 0), time=1, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    ActivateGameObject(gameObject=var "Door Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Door Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="TRAM ARRIVE", delay=0, everyFrame=false)
    on FINISHED → Arrived
  }

  state Arrived {
    ActivateGameObject(gameObject=var "Door Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Tween End {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=1)
    Wait(time=1.4, finishEvent=→"FINISHED", realTime=false)
    StopParticleEmitter(gameObject=var "Pt Dust")
    StopParticleEmitter(gameObject=var "Pt Rocks")
    on FINISHED → Door Open
  }

  var Station Position: int = 0
  var PD Opened Bool: string = "openedTramRestingGrounds"
  var PD Position Int: string = "tramRestingGroundsPosition"

  var Tram Position: int = 0
  var Tram Opened: bool = false
  var Start Pos: vector3 = (0, 0, 0)
  var Tween Vector: vector3 = (0, 0, 0)
  var Door: gameObject = Tram Main/Door
  var Door Inspect: gameObject = Tram Main/Door Inspect
  var Door Region: gameObject = Tram Main/door_tram
  var Interior Lights: gameObject = Tram Main/Interior Lights
  var Pt Dust: gameObject = Tram Main/Pt Dust
  var Pt Rocks: gameObject = Tram Main/Pt Rocks
}
