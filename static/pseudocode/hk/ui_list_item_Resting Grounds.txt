# ui_list_item
# 3 location(s):
#   Ruins1_29 / Stag Map/UI List Stag/Resting Grounds
#   RestingGrounds_09 / Stag Map/UI List Stag/Resting Grounds
#   resources.assets / Stag Map/UI List Stag/Resting Grounds

// uses template: ui_list_item
fsm ui_list_item {
  start Pause
  on SELECTION MADE → Selection Made  // from any state
  on SELECTION MADE CANCEL → Selection Made Cancel  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Chosen", storeResult=var "Chosen Effects")
    GetParent(gameObject=var "Self", storeResult=var "List")
    GetFsmInt(gameObject=var "List", fsmName="ui_list", variableName="Current Item", storeValue=var "Current Item", everyFrame=false)
    IntCompare(integer1=var "Current Item", integer2=var "Item Number", equal=→"GET SELECTED", lessThan=→"GET UNSELECTED", greaterThan=→"GET UNSELECTED", everyFrame=false)
    on GET UNSELECTED → Unselected
    on GET SELECTED → Selected
  }

  state Idle {
    on SELECT UPDATE → Check
  }

  state Selected {
    SetFsmString(gameObject=var "List", fsmName="ui_list", variableName="Current Selection", setValue=var "Selection Name", everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent=var "Selection Name", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Selected", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="SELECTED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Unselected {
    SetBoolValue(boolVariable=var "Selected", boolValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UNSELECTED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Check {
    GetFsmInt(gameObject=var "List", fsmName="ui_list", variableName="Current Item", storeValue=var "Current Item", everyFrame=false)
    IntCompare(integer1=var "Current Item", integer2=var "Item Number", equal=→"GET SELECTED", lessThan=→"GET UNSELECTED", greaterThan=→"GET UNSELECTED", everyFrame=false)
    on GET SELECTED → Selected
    on GET UNSELECTED → Unselected
  }

  state Selection Made {
    GetFsmInt(gameObject=var "List", fsmName="ui_list", variableName="Current Item", storeValue=var "Current Item", everyFrame=false)
    IntCompare(integer1=var "Current Item", integer2=var "Item Number", equal=→"GET SELECTED", lessThan=→"GET UNSELECTED", greaterThan=→"GET UNSELECTED", everyFrame=false)
    on GET SELECTED → Selected?
    on GET UNSELECTED → Unselected
  }

  state Chosen {
    SetFsmBool(gameObject=var "List", fsmName="ui_list", variableName="Selected Unselectable", setValue=false, everyFrame=false)
    SetFsmString(gameObject=var "List", fsmName="ui_list", variableName="Current Selection", setValue=var "Selection Name", everyFrame=false)
    ActivateGameObject(gameObject=var "Chosen Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="CHOSEN", delay=0, everyFrame=false)
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Remove Selection
  }

  state Selected? {
    BoolTest(boolVariable=var "Selected", isTrue=→"FINISHED", isFalse=→"GET SELECTED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="SELECTED", delay=0, everyFrame=false)
    on FINISHED → Unselectable?
  }

  state Unselectable? {
    GetFsmBool(gameObject=var "Self", fsmName="Enough Geo", variableName="Unselectable", storeValue=var "Unselectable", everyFrame=false)
    BoolTest(boolVariable=var "Unselectable", isTrue=→"GET UNSELECTED", isFalse=→"GET SELECTED", everyFrame=false)
    on GET SELECTED → Chosen
    on GET UNSELECTED → Chosen but Unselectable
  }

  state Chosen but Unselectable {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Self", audioClip=sword_hit_reject, pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Self", audioClip=sword_hit_reject, pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    SetFsmBool(gameObject=var "List", fsmName="ui_list", variableName="Selected Unselectable", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="CHOSEN UNSELECTABLE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Pause {
    ActivateGameObject(gameObject=var "Chosen Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Remove Selection {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UNSELECTED", delay=0, everyFrame=false)
  }

  state Selection Made Cancel {
    GetFsmInt(gameObject=var "List", fsmName="ui_list", variableName="Current Item", storeValue=var "Current Item", everyFrame=false)
    IntCompare(integer1=var "Current Item", integer2=var "Item Number", equal=→"GET SELECTED", lessThan=→"GET UNSELECTED", greaterThan=→"GET UNSELECTED", everyFrame=false)
    on GET UNSELECTED → Unselected
    on GET SELECTED → Chosen 2
  }

  state Chosen 2 {
    SetFsmBool(gameObject=var "List", fsmName="ui_list", variableName="Selected Unselectable", setValue=false, everyFrame=false)
    SetFsmString(gameObject=var "List", fsmName="ui_list", variableName="Current Selection", setValue=var "Selection Name", everyFrame=false)
    ActivateGameObject(gameObject=var "Chosen Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="CHOSEN", delay=0, everyFrame=false)  // disabled
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UNSELECTED", delay=0, everyFrame=false)
    on FINISHED → 
  }

  var Item Number: int = 8
  var Selection Name: string = "Resting Grounds"

  var Current Item: int = 0
  var Selected: bool = false
  var Unselectable: bool = false
  var Chosen Effects: gameObject = <null>
  var Glow: gameObject = <null>
  var List: gameObject = <null>
  var Pointer L: gameObject = <null>
  var Pointer R: gameObject = <null>
  var Self: gameObject = <null>
}
