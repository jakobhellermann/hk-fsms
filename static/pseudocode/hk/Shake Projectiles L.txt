# Shake Projectiles L
# 3 location(s):
#   Abyss_19 / Infected Knight
#   Dream_03_Infected_Knight / Infected Knight Dream
#   GG_Broken_Vessel / Infected Knight

fsm Shake Projectiles L {
  start Idle
  on SHAKE END → Shake End  // from any state

  state Idle {
    GetOwner(storeGameObject=var "Self")
    on SHAKE START → Choose Speed
  }

  state Spawn {
    SetBoolValue(boolVariable=var "Prev Adjacent", boolValue=var "Was Adjacent", everyFrame=false)
    SetFloatValue(floatVariable=var "Prev Speed", floatValue=var "X Velocity", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=IK Projectile SH (sharedassets343.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Projectile")
    RandomFloat(min=6, max=25, storeResult=var "X Velocity")  // disabled
    SetVelocity2d(gameObject=var "Projectile", vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    on FINISHED → Pause
    on REDO → 
  }

  state Pause {
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choose Speed
  }

  state Choose Speed {
    SendRandomEvent(events=[(none), (none), (none), (none), (none)], weights=[1, 1, 1, 1, 1], delay=0)
    on 1 → Speed 1
    on 2 → Speed 2
    on 3 → Speed 3
    on 4 → Speed 4
    on 5 → Speed 5
  }

  state Speed 1 {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-5, everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-8, tolerance=0, equalBool=var "Adjacent 1", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Adjacent 2", boolValue=false, everyFrame=false)
    on FINISHED → Prev?
  }

  state Speed 2 {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-8, everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-5, tolerance=0, equalBool=var "Adjacent 1", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-11, tolerance=0, equalBool=var "Adjacent 2", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    on FINISHED → Prev?
  }

  state Speed 3 {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-11, everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-8, tolerance=0, equalBool=var "Adjacent 1", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-14, tolerance=0, equalBool=var "Adjacent 2", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    on FINISHED → Prev?
  }

  state Speed 4 {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-14, everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-11, tolerance=0, equalBool=var "Adjacent 1", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-17, tolerance=0, equalBool=var "Adjacent 2", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    on FINISHED → Prev?
  }

  state Speed 5 {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-17, everyFrame=false)
    FloatTestToBool(float1=var "Prev Speed", float2=-14, tolerance=0, equalBool=var "Adjacent 1", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Adjacent 2", boolValue=false, everyFrame=false)
    on FINISHED → Prev?
  }

  state Prev? {
    FloatCompare(float1=var "Prev Speed", float2=var "X Velocity", tolerance=0, equal=→"REDO", lessThan=(none), greaterThan=(none), everyFrame=false)
    on REDO → Choose Speed
    on FINISHED → Was Adjacent?
  }

  state Was Adjacent? {
    SetBoolValue(boolVariable=var "Was Adjacent", boolValue=false, everyFrame=false)
    BoolAnyTrue(boolVariables=[var "Adjacent 1", var "Adjacent 2"], sendEvent=(none), storeResult=var "Was Adjacent", everyFrame=false)
    BoolAllTrue(boolVariables=[var "Prev Adjacent", var "Was Adjacent"], sendEvent=→"REDO", storeResult=(unset), everyFrame=false)
    on REDO → Choose Speed
    on FINISHED → Spawn
  }

  state Shake End {
    SetFloatValue(floatVariable=var "Pause Time", floatValue=0.5, everyFrame=false)
    on FINISHED → Idle
  }

  var Multiplier: float = 1
  var Pause Time: float = 1
  var Prev Speed: float = 0
  var X Velocity: float = 0
  var Adjacent 1: bool = false
  var Adjacent 2: bool = false
  var Prev Adjacent: bool = false
  var Was Adjacent: bool = false
  var Projectile: gameObject = <null>
  var Self: gameObject = <null>
}
