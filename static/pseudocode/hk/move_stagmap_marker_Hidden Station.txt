# move_stagmap_marker
# 3 location(s):
#   Ruins1_29 / Stag Map/UI List Stag/Hidden Station
#   RestingGrounds_09 / Stag Map/UI List Stag/Hidden Station
#   resources.assets / Stag Map/UI List Stag/Hidden Station

// uses template: move_stagmap_marker
fsm move_stagmap_marker {
  start Init
  on SELECTED → Have Map Marker?  // from any state

  state Init {
    FindGameObject(objectName="", withTag="StagMapMarker", store=var "Marker")
    on FINISHED → Idle
  }

  state Idle {
    on SELECTED → Have Map Marker?
  }

  state Move {
    SetPosition(gameObject=var "Marker", vector=var "Marker Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Marker"), sendEvent="MARKER UP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Have Map Marker? {
    GameObjectIsNull(gameObject=var "Marker", isNull=(none), isNotNull=→"FINISHED", storeResult=(unset), everyFrame=false)
    FindGameObject(objectName="", withTag="StagMapMarker", store=var "Marker")
    on FINISHED → Move
  }

  var Marker Pos: vector3 = (3.19, -2.1, -1)

  var Marker: gameObject = <null>
}
