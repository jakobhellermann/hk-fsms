# Control
# 1 location(s):
#   sharedassets197.assets / Corpse Jellyfish Baby

fsm Control {
  start Audio
  on ACID → Burn  // from any state

  state Air {
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=1.4, stretchMinX=0.6, stretchMaxY=1.75, scaleModifier=1, everyFrame=true)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)  // disabled
    CheckCollisionSide(topHit=var "Top", rightHit=var "Right", bottomHit=var "Bot", leftHit=var "Left", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=true)
    BoolAnyTrue(boolVariables=[var "Bot", var "Left", var "Right", var "Top"], sendEvent=→"GROUND", storeResult=(unset), everyFrame=true)
    on GROUND → Ground
  }

  state Burn {
    SetSpriteRenderer(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    FindChild(gameObject=Self, childName="Acid Steam", storeResult=var "Steam")
    PlayParticleEmitter(gameObject=var "Steam", emit=0)
  }

  state Ground {
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Bot", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Left", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Right", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Top", boolValue=true, everyFrame=false)
    CheckCollisionSide(topHit=var "Top", rightHit=var "Right", bottomHit=var "Bot", leftHit=var "Left", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=true)
    BoolNoneTrue(boolVariables=[var "Bot", var "Left", var "Right", var "Top"], sendEvent=→"AIR", storeResult=(unset), everyFrame=true)
    on AIR → Air
  }

  state Audio {
    GetOwner(storeGameObject=var "Self")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=jellyfish_mini_death, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetParent(gameObject=var "Spit", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Splat", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Air
  }

  var Bot: bool = false
  var Left: bool = false
  var Right: bool = false
  var Top: bool = false
  var Self: gameObject = <null>
  var Spit: gameObject = Corpse Jellyfish Baby/Spit Effect
  var Splat: gameObject = Corpse Jellyfish Baby/Splat Effect
  var Steam: gameObject = <null>
}
