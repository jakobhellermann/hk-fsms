// uses template: vessel_size
fsm vessel_size {
  start Init
  on BIND VESSEL ORB → Bound  // from any state
  on UNBIND VESSEL ORB → Check  // from any state

  state Init {
    on START → Max Inclusive?
  }

  state Init Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserve", storeValue=var "MP Reserve")
    IntCompare(integer1=var "MP Reserve", integer2=var "Empty Amount", equal=(none), lessThan=→"SMALL", greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "MP Reserve", integer2=var "Full Amount", equal=→"SMALL", lessThan=→"LARGE", greaterThan=→"SMALL", everyFrame=false)
    on SMALL → Start Small
    on LARGE → Start Large
  }

  state Start Small {
    SetScale(gameObject=Self, vector=var "Small Size", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Small", boolValue=true, everyFrame=false)
    on SIZE CHECK → Check
  }

  state Start Large {
    SetScale(gameObject=Self, vector=var "Full Size", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Small", boolValue=false, everyFrame=false)
    on SIZE CHECK → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserve", storeValue=var "MP Reserve")
    IntCompare(integer1=var "MP Reserve", integer2=var "Empty Amount", equal=(none), lessThan=→"SMALL", greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "MP Reserve", integer2=var "Full Amount", equal=→"SMALL", lessThan=→"LARGE", greaterThan=→"SMALL", everyFrame=false)
    on SMALL → Small
    on LARGE → Large
  }

  state Small {
    BoolTest(boolVariable=var "Small", isTrue=→"NO CHANGE", isFalse=(none), everyFrame=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Small Size", time=var "Tween Time", delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Small", boolValue=true, everyFrame=false)
    on SIZE CHECK → Check
    on NO CHANGE → No Change
  }

  state Large {
    BoolTest(boolVariable=var "Small", isTrue=(none), isFalse=→"NO CHANGE", everyFrame=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Full Size", time=var "Tween Time", delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Small", boolValue=false, everyFrame=false)
    on SIZE CHECK → Check
    on NO CHANGE → No Change
  }

  state No Change {
    on SIZE CHECK → Check
  }

  state Max Inclusive? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserveMax", storeValue=var "MP Reserve Max")
    IntCompare(integer1=var "MP Reserve Max", integer2=var "Full Amount", equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    IntAdd(intVariable=var "Full Amount", add=1, everyFrame=false)
    on FINISHED → Init Check
  }

  state Bound {
    SetScale(gameObject=Self, vector=var "Small Size", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
  }

  var Tween Time: float = 0.75
  var Empty Amount: int = 0
  var Full Amount: int = 0
  var MP Reserve: int = 0
  var MP Reserve Max: int = 0
  var Small: bool = false
  var Full Size: vector3 = (1, 1, 1)
  var Small Size: vector3 = (0.75, 0.75, 0.75)
}
