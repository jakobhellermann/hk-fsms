# Initial Shade Dreamgate Position
# 1 location(s):
#   resources.assets / Game_Map

fsm Initial Shade Dreamgate Position {
  start Pause

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Position
  }

  state Position {
    FindChild(gameObject=Self, childName="Shade Pos", storeResult=var "Shade Pos Obj")
    FindChild(gameObject=Self, childName="Dream_Gate_Pin", storeResult=var "Dream_Gate_Pin")
    GetPlayerDataVector3(gameObject=var "GameManager", vector3Name="shadeMapPos", storeValue=var "Shade Pos")
    GetPlayerDataVector3(gameObject=var "GameManager", vector3Name="dreamgateMapPos", storeValue=var "Dreamgate Pos")
    SetPosition(gameObject=var "Shade Pos Obj", vector=var "Shade Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Dream_Gate_Pin", vector=var "Dreamgate Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  var Dreamgate Pos: vector3 = (0, 0, 0)
  var Shade Pos: vector3 = (0, 0, 0)
  var Dream_Gate_Pin: gameObject = <null>
  var Shade Pos Obj: gameObject = <null>
}
