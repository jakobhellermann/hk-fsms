fsm Set Battle Scene {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set
  }

  state Set {
    SetFsmString(gameObject=Self, fsmName="health_manager_enemy", variableName="Send KILLED to", setValue="Battle Scene", everyFrame=false)
  }
}
