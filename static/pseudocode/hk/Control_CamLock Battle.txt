# Control
# 2 location(s):
#   Dream_Mighty_Zote / Audience/CamLock Battle
#   GG_Grey_Prince_Zote / CamLock Battle

fsm Control {
  start Inactive

  state Active {
    SetCollider(gameObject=Self, active=true)
  }

  state Inactive {
    SetCollider(gameObject=Self, active=false)
    on ZOTE ROAR END → Pause Frame
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Active
  }
}
