# FSM
# 1 location(s):
#   resources.assets / Game_Map/Town_Tutorial/Area Name

// uses template: map_worldmaponly
fsm FSM {
  start Check

  state Check {
    GetFsmBool(gameObject=var "Game Map", fsmName="Map FSM State", variableName="Display Next Area", storeValue=var "Display", everyFrame=false)
    BoolTest(boolVariable=var "Display", isTrue=→"INACTIVE", isFalse=→"ACTIVE", everyFrame=false)
    on INACTIVE → Inactive
    on ACTIVE → Active
  }

  state Inactive {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Active {
    SetMeshRenderer(gameObject=Self, active=true)
  }

  var Display: bool = false
}
