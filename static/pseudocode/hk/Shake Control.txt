# Shake Control
# 1 location(s):
#   GG_Land_of_Storms / Shake Region

fsm Shake Control {
  start Wait

  state Wait {
    WaitRandom(timeMin=2, timeMax=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  state Shake {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=193.4, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Wait
  }

  var Hero X: float = 0
}
