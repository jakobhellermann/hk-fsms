# Shift
# 1 location(s):
#   Fungus1_20_v02 / GradeMarker

fsm Shift {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Centre X", tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=true)
    on L → L
    on R → R
  }

  state L {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Centre X", tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=2, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on R → R
  }

  state R {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Centre X", tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=162.5, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on L → L
  }

  var Centre X: float = 84
  var Hero X: float = 0
}
