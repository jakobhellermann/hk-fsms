# Force Hard landing
# 1 location(s):
#   Crossroads_36 / Force Hard Landing

fsm Force Hard landing {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "Hero")
    on ACTIVATE → Activate
    on ENTER → Enter
  }

  state Activate {
    DestroySelf(detachChildren=false)
  }

  state Enter {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=ForceHardLanding())
  }

  var Activated: bool = false
  var Hero: gameObject = <null>
}
