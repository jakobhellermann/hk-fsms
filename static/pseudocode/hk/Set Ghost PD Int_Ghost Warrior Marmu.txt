# Set Ghost PD Int
# 1 location(s):
#   Fungus3_40_boss / Warrior/Ghost Warrior Marmu

fsm Set Ghost PD Int {
  start Idle

  state Idle {
    on ZERO HP → Set
  }

  state Set {
    SetPlayerDataInt(intName=var "PD Int", value=1)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST DEAD", delay=0, everyFrame=false)
  }

  var PD Int: string = "mumCaterpillarDefeated"
}
