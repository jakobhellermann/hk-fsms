// uses template: shopitem_shiftpos
fsm Shift_pos {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=var "Self", vector=var "Pos Original", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    CallMethodProper(gameObject=var "Self", behaviour="ShopItemStats", methodName="GetItemNumber", parameters=[], storeResult=var "Item Num")
    SetVector3Value(vector3Variable=var "Pos Down", vector3Value=var "Pos Original", everyFrame=false)
    SetVector3Value(vector3Variable=var "Pos Up", vector3Value=var "Pos Original", everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Pos Down", addX=0, addY=-0.8, addZ=0, everyFrame=false, perSecond=false)
    Vector3AddXYZ(vector3Variable=var "Pos Up", addX=0, addY=0.8, addZ=0, everyFrame=false, perSecond=false)
    GetParent(gameObject=var "Self", storeResult=var "Enemy List")
    FindChild(gameObject=Self, childName="Geo Sprite", storeResult=var "Geo Sprite")
    FindChild(gameObject=Self, childName="Item cost", storeResult=var "Item cost")
    FindChild(gameObject=Self, childName="Item Sprite", storeResult=var "Item Sprite")
    on FINISHED → Idle
  }

  state Idle {
    SetBoolValue(boolVariable=var "Instant", boolValue=false, everyFrame=false)
    on SHIFT → Check Num
    on SHIFT INSTANT → Instant
  }

  state Check Num {
    GetFsmInt(gameObject=var "Enemy List", fsmName="Item List Control", variableName="Current Item", storeValue=var "Current Num", everyFrame=false)
    IntCompare(integer1=var "Item Num", integer2=var "Current Num", equal=→"CURRENT", lessThan=→"UP", greaterThan=→"DOWN", everyFrame=false)
    on CURRENT → Current
    on UP → Up
    on DOWN → Down
  }

  state Tween {
    iTweenMoveTo(gameObject=var "Self", id="", transformPosition=var "", vectorPosition=var "Tween Vector", time=0.25, delay=0, speed=(unset), space=1, easeType=easeOutSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Idle
    on SHIFT → Check Num
    on SHIFT INSTANT → Instant
  }

  state Current {
    SetVector3Value(vector3Variable=var "Tween Vector", vector3Value=var "Pos Original", everyFrame=false)
    BoolTest(boolVariable=var "Instant", isTrue=→"SHIFT INSTANT", isFalse=(none), everyFrame=false)
    on FINISHED → Tween
    on SHIFT INSTANT → Instant Shift
  }

  state Up {
    SetVector3Value(vector3Variable=var "Tween Vector", vector3Value=var "Pos Up", everyFrame=false)
    BoolTest(boolVariable=var "Instant", isTrue=→"SHIFT INSTANT", isFalse=(none), everyFrame=false)
    on FINISHED → Tween
    on SHIFT INSTANT → Instant Shift
  }

  state Down {
    SetVector3Value(vector3Variable=var "Tween Vector", vector3Value=var "Pos Down", everyFrame=false)
    BoolTest(boolVariable=var "Instant", isTrue=→"SHIFT INSTANT", isFalse=(none), everyFrame=false)
    on FINISHED → Tween
    on SHIFT INSTANT → Instant Shift
  }

  state Instant {
    SetBoolValue(boolVariable=var "Instant", boolValue=true, everyFrame=false)
    on FINISHED → Check Num
  }

  state Instant Shift {
    SetPosition(gameObject=var "Self", vector=var "Tween Vector", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  var Current Num: int = 0
  var Item Num: int = 0
  var Num Diff: int = 0
  var Instant: bool = false
  var Pos Down: vector3 = (0, 0, 0)
  var Pos Original: vector3 = (0, 0, 0)
  var Pos Up: vector3 = (0, 0, 0)
  var Tween Vector: vector3 = (0, 0, 0)
  var Half Color: color = (1, 1, 1, 0.528)
  var Enemy List: gameObject = <null>
  var Geo Sprite: gameObject = <null>
  var Item Sprite: gameObject = <null>
  var Item cost: gameObject = <null>
  var Self: gameObject = <null>
}
