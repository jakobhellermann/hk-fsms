# Control
# 1 location(s):
#   Ruins2_03 / chandelier

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForBossLoad(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Slash Point", storeResult=var "Slash Point")
    FindChild(gameObject=Self, childName="chandelier_top", storeResult=var "Chand Top")
    FindChild(gameObject=var "Chand Top", childName="rope_top", storeResult=var "Top Rope")
    FindChild(gameObject=Self, childName="chandelier_full", storeResult=var "Chandelier")
    FindChild(gameObject=Self, childName="chandelier_broken", storeResult=var "Broken")
    FindChild(gameObject=Self, childName="Dust Hit Plume", storeResult=var "Dust Plume")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="watcherChandelier", isTrue=→"ACTIVATE", isFalse=(none))
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Hit
  }

  state Hit {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Slash Point", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=rope_cut (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Slash Point", position=(unset), rotation=(unset), storeObject=var "")
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"DESTROY", lessThan=→"DESTROY", greaterThan=(none), everyFrame=false)
    Wait(time=0.17, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on DESTROY → Drop
  }

  state Drop {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=chandelier_fall_and_perish (sharedassets113.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=rope_cut_final (sharedassets113.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataBool(boolName="watcherChandelier", value=true)
    SendEventByName(eventTarget=GameObject(var "Top Rope"), sendEvent="CUT", delay=0, everyFrame=false)
    iTweenMoveBy(gameObject=var "Chandelier", id="", vector=(0, -11, 0), time=0.2, delay=0.2, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Slam
  }

  state Slam {
    ActivateGameObject(gameObject=var "Chandelier", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust Plume", emit=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHANDELIER CRASH", delay=0, everyFrame=false)
  }

  state Activate {
    ActivateGameObject(gameObject=var "Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Chandelier", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Top Rope", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHANDELIER CRASH", delay=0, everyFrame=false)
  }

  var Chooser: float = 0
  var Hits: int = 3
  var Broken: gameObject = <null>
  var Chand Top: gameObject = <null>
  var Chandelier: gameObject = <null>
  var Dust Plume: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Self: gameObject = <null>
  var Slash Point: gameObject = <null>
  var Top Rope: gameObject = <null>
}
