# Control
# 2 location(s):
#   Room_Colosseum_Gold / Colosseum Manager/Waves/Lobster Lancer/Entry Object/Lancer
#   GG_God_Tamer / Entry Object/Lancer

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Slash 1", storeResult=var "Slash 1")
    FindChild(gameObject=Self, childName="Slash 2", storeResult=var "Slash 2")
    FindChild(gameObject=Self, childName="Damage Box", storeResult=var "Damage Box")
    FindChild(gameObject=Self, childName="Lance", storeResult=var "Lance")
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Lobster", storeResult=var "Lobster")
    on FINISHED → First Aim
  }

  state Launch {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lancer Jump")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=30, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_shield_sentry_shield_move (sharedassets34.assets))
    on FINISHED → In Air
  }

  state First Aim {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Jump X", floatValue=var "Hero X", everyFrame=false)
    FloatSubtract(floatVariable=var "Jump X", subtract=var "Self X", everyFrame=false, perSecond=false)
    FloatMultiply(floatVariable=var "Jump X", multiplyBy=0.5, everyFrame=false)
    FloatClamp(floatVariable=var "Jump X", minValue=-15, maxValue=-3, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[God_Tamer_Roar_01 (sharedassets34.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Launch
  }

  state In Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Slash 1
  }

  state Slash 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lancer Slash1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[ruin_shield_sentry_sword_1 (sharedassets34.assets), ruin_shield_sentry_sword_2 (sharedassets34.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_ground_land (sharedassets28.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[God_Tamer_attack_01 (sharedassets34.assets), God_Tamer_attack_02 (sharedassets34.assets), God_Tamer_attack_03 (sharedassets34.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Slash 2
  }

  state Slash 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lancer Slash2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slash 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Slash Recover
  }

  state Slash Recover {
    ActivateGameObject(gameObject=var "Slash 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lancer Slash Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lancer Idle")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Lancer TurnToIdle", resetFrame=true, everyFrame=true)
    WaitRandom(timeMin=2, timeMax=3.5, finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "Death", isTrue=→"DEFEAT", isFalse=(none), everyFrame=true)
    on FINISHED → Aim
    on DEFEAT → Defeat
  }

  state Aim {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_shield_sentry_prepare (sharedassets34.assets))
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Jump X", floatValue=var "Hero X", everyFrame=false)
    FloatSubtract(floatVariable=var "Jump X", subtract=var "Self X", everyFrame=false, perSecond=false)
    FloatMultiply(floatVariable=var "Jump X", multiplyBy=0.55, everyFrame=false)
    FloatClamp(floatVariable=var "Jump X", minValue=-15, maxValue=15, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lancer Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on FINISHED → Jump Sfx
  }

  state Defeat {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    ActivateGameObject(gameObject=var "Slash 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slash 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[God_Tamer_Defeat (sharedassets34.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    ActivateGameObject(gameObject=var "Damage Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lance", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Lance", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lancer Defeat")
    GameObjectIsNull(gameObject=var "Lobster", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Lobster", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=7.61, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Done
  }

  state Done {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=7.61, z=(unset), space=0, everyFrame=true, lateUpdate=false)
  }

  state Jump Sfx {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[God_Tamer_Jump_01 (sharedassets34.assets), God_Tamer_Jump_02 (sharedassets34.assets), God_Tamer_Jump_03 (sharedassets34.assets), God_Tamer_Jump_04 (sharedassets34.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Launch
  }

  var Hero X: float = 0
  var Jump X: float = 0
  var Self X: float = 0
  var Death: bool = false
  var Damage Box: gameObject = <null>
  var Lance: gameObject = <null>
  var Lobster: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Slash 1: gameObject = <null>
  var Slash 2: gameObject = <null>
  var Voice Player: gameObject = <null>
}
