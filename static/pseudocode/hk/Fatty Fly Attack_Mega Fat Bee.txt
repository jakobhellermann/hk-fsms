    BoolTest(boolVariable=var "Rage", isTrue=→"RAGE", isFalse=(none), everyFrame=false)
    on FINISHED → Attack Antic
    on RAGE → Rage Wait
  }

  state Attack Antic {
    Decelerate(gameObject=Self, deceleration=0.1)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled

    on FINISHED → Attack Antic 2
  }

    Wait(time=var "Shot Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack 2
  }

  state CD {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shoot CD")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait
  }

  state Attack Antic 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Decelerate(gameObject=Self, deceleration=0.1)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=centipede_prepare (sharedassets33.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

    on START → Wait

    Wait(time=var "Shot Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack Check
  }

  state Attack Check {
    FloatAdd(floatVariable=var "Angle Mod", add=20, everyFrame=false, perSecond=false)
    IntOperator(integer1=var "Attacks", integer2=1, operation=Subtract, storeResult=var "Attacks", everyFrame=false)
    IntCompare(integer1=var "Attacks", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"REPEAT", everyFrame=false)
    on END → CD
    on REPEAT → Attack
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on START → Wait
  }

  state Shoot Anim {
    RandomFloat(min=0, max=360, storeResult=var "Angle Mod")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shoot")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Attack
  }

  state Rage Wait {
    SetFloatValue(floatVariable=var "Shot Speed", floatValue=22, everyFrame=false)
    SetFloatValue(floatVariable=var "Shot Pause", floatValue=0.2, everyFrame=false)
    WaitRandom(timeMin=1, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → R Antic
  }

  state R Antic {
    Decelerate(gameObject=Self, deceleration=0.1)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetIntValue(intVariable=var "Attacks", intValue=4, everyFrame=false)
    on FINISHED → Attack Antic 2
  }

  state Raging {
    on RAGE END → Wait
  }

  var Long Wait: bool = false

  var Angle Max: float = 0
  var Angle Min: float = 0
  var Angle Mod: float = 0
  var Shot Pause: float = 0.4
  var Shot Speed: float = 15
  var Attacks: int = 0
  var Rage: bool = false
  var Started: bool = false
  var Spawn Position: vector3 = (0, 0, 0.019)
  var Self: gameObject = <null>
}
