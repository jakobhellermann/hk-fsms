# Emerge
# 1 location(s):
#   Fungus3_39 / Moss Cultist

fsm Emerge {
  start Preaching

  state Preaching {
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    AudioPlayInState(gameObject=Self, volume=1)
    GetOwner(storeGameObject=var "Self")
    on MOSS FAT WAKE → Hide
  }

  state Hide {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hide")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Moss_Cultist_Hide (sharedassets219.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Hidden
  }

  state Hidden {
    Wait(time=1, finishEvent=→"CHECK", realTime=false)
    on CHECK → Check
  }

  state Check {
    GetTagCount(tag="Boss", storeResult=var "Acolytes")
    IntCompare(integer1=var "Acolytes", integer2=0, equal=→"EMERGE", lessThan=→"EMERGE", greaterThan=(none), everyFrame=false)
    on FINISHED → Hidden
    on EMERGE → Emerge Pause
  }

  state Emerge Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emerge
  }

  state Emerge {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayInState(gameObject=Self, volume=1)
  }

  var Acolytes: int = 0
  var Dream Dialogue: gameObject = <null>
  var Self: gameObject = <null>
}
