# Fling
# 1 location(s):
#   sharedassets350.assets / Corpse Flukeman

fsm Fling {
  start Pause

  state Fling {
    GetVelocity2d(gameObject=Self, vector=var "Velocity", x=(unset), y=(unset), space=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Bot", vector=var "Velocity", x=(unset), y=(unset), everyFrame=false)
    SetVelocity2d(gameObject=var "Top", vector=var "Velocity", x=(unset), y=(unset), everyFrame=false)
    DestroySelf(detachChildren=true)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fling
  }

  var Velocity: vector2 = (0, 0)
  var Bot: gameObject = Corpse Flukeman/Flukeman Bot
  var Top: gameObject = Corpse Flukeman/Flukeman Top
}
