# Control
# 1 location(s):
#   Ruins2_11_boss / Battle Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Jar Collector", storeResult=var "Boss")
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "CamLock")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Start
  }

  state Start {
    ActivateGameObject(gameObject=var "Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WAKE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="PARTICLE CLEANUP", delay=0, everyFrame=false)
    on BATTLE END → End
  }

  state End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Boss: gameObject = <null>
  var CamLock: gameObject = <null>
}
