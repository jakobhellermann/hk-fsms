  start Init

  state Out {
    Tk2dPlayAnimation(gameObject=var "Elderbug", animLibName="", clipName="Turn Right")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {

    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetParent(gameObject=var "Parent", storeResult=var "Elderbug")

    FindChild(gameObject=var "Elderbug", childName="Light", storeResult=var "Light")

    on ENTER → First Call?
    on EXIT → Out
  }

  state First Call? {

    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → In
  }

  var Hero In Range: bool = false
  var Talking: bool = false

  var Bench Prev Colour: color = (0, 0, 0, 1)
