# Stop Stompers
# 1 location(s):
#   Mines_19 / stomper_door_contraption

fsm Stop Stompers {
  start Inert
  on OPEN → Wait  // from any state
  on ACTIVATE → Open Door Quick  // from any state

  state Inert {
  }

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Shake
  }

  state Stop Stompers Quick {
    SendEventToRegister(eventName="END STOMPERS INSTANT")
    ActivateGameObject(gameObject=var "Shake Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Open Door {
    SendEventByName(eventTarget=GameObject(var "Door"), sendEvent="OPEN", delay=0, everyFrame=false)
  }

  state Open Door Quick {
    SendEventByName(eventTarget=GameObject(var "Door"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    on FINISHED → Stop Stompers Quick
  }

  state Start Shake {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=true, everyFrame=false)
    on FINISHED → Stop Stompers
  }

  state End Shake {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=stomper_door_contraption/Finish Dust, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=stomper_door_contraption/rumble audio, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=stomper_door_contraption/stop audio, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimSpeedLerp(target=Self, duration=0.5, toSpeed=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open Door
  }

  state Turn Gears {
    ActivateGameObject(gameObject=stomper_door_contraption/Antic Dust, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=stomper_door_contraption/rumble audio, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=Self, stateName="gear turn", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Shake
  }

  state Stop Stompers {
    SendEventToRegister(eventName="END STOMPERS")
    ActivateGameObject(gameObject=var "Shake Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Turn Gears
  }

  var Door: gameObject = Metal Gate v2
  var Shake Region: gameObject = Shake Region
}
