# Conversation Control
# 1 location(s):
#   Town / Jiji Door

fsm Conversation Control {
  start Pause
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    SetCollider(gameObject=var "door_jiji", active=false)
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Antic Dust", storeResult=var "Antic Dust")
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite")
    FindChild(gameObject=Self, childName="Plat Stream Dust", storeResult=var "Stream Dust")
    FindChild(gameObject=Self, childName="Slam Dust", storeResult=var "Slam Dust")
    FindChild(gameObject=Self, childName="door_jiji", storeResult=var "door_jiji")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="jijiDoorUnlocked", isTrue=→"INACTIVE", isFalse=(none))
    on FINISHED → Idle
    on INACTIVE → Inactive
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → No Key
  }

  state Anim End {
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state No Key {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["JIJI_DOOR_NOKEY", "Prompts"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Hero Anim {
    on FINISHED → Key?
  }

  state Key? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="simpleKeys", storeValue=var "Simple Keys")
    IntCompare(integer1=var "Simple Keys", integer2=0, equal=→"NOKEY", lessThan=→"NOKEY", greaterThan=→"KEY", everyFrame=false)
    on KEY → Box Up YN
    on NOKEY → Box Up
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["JIJI_DOOR_KEY", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  state Decline Pause {
    Wait(time=0.3, finishEvent=→"TALK FINISH", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → 
  }

  state Activate {
    PlayerDataIntAdd(gameObject=var "GameManager", intName="simpleKeys", amount=var "Decrement")
    SetPlayerDataBool(boolName="jijiDoorUnlocked", value=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=manhole_key_turn (sharedassets7.assets))
    on FINISHED → Tween Antic
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inactive {
    SetSpriteRenderer(gameObject=var "Sprite", active=false)
    SetCollider(gameObject=Self, active=false)
  }

  state Tween Antic {
    iTweenMoveBy(gameObject=var "Sprite", id="", vector=(0, -1, 0), time=1, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Wait(time=1.2, finishEvent=→"FINISHED", realTime=false)
    ActivateAllChildren(gameObject=var "Antic Dust", activate=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=jiji_door_open_pt_1 (sharedassets7.assets))
    on FINISHED → Slam
  }

  state Slam {
    iTweenMoveBy(gameObject=var "Sprite", id="", vector=(0, -3.1, 0), time=0.5, delay=0, speed=(unset), easeType=easeInSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Slam End
  }

  state Activate Door {
    SetCollider(gameObject=var "door_jiji", active=true)
  }

  state Slam End {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Slam Dust", emit=0)
    PlayParticleEmitter(gameObject=var "Stream Dust", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=jiji_door_open_pt_2 (sharedassets7.assets))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[platform_dirt_fall_1 (sharedassets6.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0.65, storePlayer=var "Voice Player")
    on FINISHED → Activate Door
  }

  var Decrement: int = -1
  var Simple Keys: int = 0
  var Antic Dust: gameObject = <null>
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Slam Dust: gameObject = <null>
  var Sprite: gameObject = <null>
  var Stream Dust: gameObject = <null>
  var Voice Player: gameObject = <null>
  var door_jiji: gameObject = <null>
}
