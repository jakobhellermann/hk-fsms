# Control
# 2 location(s):
#   Room_Tram / Tram Console R/Inspect Region
#   Room_Tram_RG / Tram Console R/Inspect Region

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetChild(gameObject=var "Parent", childName="Flash", withTag="Untagged", storeResult=var "Flash")
    on FINISHED → Idle
  }

  state Idle {
    on CONVO START → Box Up YN
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Prompt
  }

  state Prompt {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["TRAM_CONSOLE", "Prompts"], storeResult=(unset var))
    on NO → N Box Down
    on YES → Y Box Down
  }

  state Y Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=white_seal_1 (sharedassets19.assets))
    on FINISHED → Send Event
  }

  state N Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo End
  }

  state Convo End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Send Event {
    SendEventByName(eventTarget=BroadcastAll, sendEvent=var "Event", delay=0, everyFrame=false)
    on TRAM ARRIVE → Convo End
  }

  var Event: string = "TRAM MOVE R"

  var Flash: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
