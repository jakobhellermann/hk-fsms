fsm Award Orbs {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on DREAM WAKE → Check
  }

  state Check {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="whiteDefenderDefeated", falseBool="whiteDefenderOrbsCollected", isTrue=→"AWARD", isFalse=(none))
    on AWARD → Award
  }

  state Award {
    SetPlayerDataBool(boolName="whiteDefenderOrbsCollected", value=true)
    PlayerDataIntAdd(gameObject=var "GameManager", intName="dreamOrbs", amount=var "Orbs")
    CreateObject(gameObject=Dream Orb Get S (sharedassets7.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
  }

  var Orbs: int = 300
  var Self: gameObject = <null>
}
