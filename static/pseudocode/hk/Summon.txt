# Summon
# 1 location(s):
#   Deepnest_East_16 / Giant Hopper Summon

fsm Summon {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on SUMMON → Antic
  }

  state Summon {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=cave_in_rock_hit_2 (sharedassets173.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioStop(gameObject=var "Audio Player")
  }

  state Antic {
    FindChild(gameObject=Self, childName="Antic Dust 1", storeResult=var "Antic Dust 1")
    FindChild(gameObject=Self, childName="Antic Dust 2", storeResult=var "Antic Dust 2")
    FindChild(gameObject=Self, childName="Antic Rocks 1", storeResult=var "Antic Rocks 1")
    FindChild(gameObject=Self, childName="Antic Rocks 2", storeResult=var "Antic Rocks 2")
    PlayParticleEmitter(gameObject=var "Antic Dust 1", emit=0)
    PlayParticleEmitter(gameObject=var "Antic Dust 2", emit=0)
    PlayParticleEmitter(gameObject=var "Antic Rocks 1", emit=0)
    PlayParticleEmitter(gameObject=var "Antic Rocks 2", emit=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=misc_rumble_loop (sharedassets101.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Audio Player")
    on FINISHED → Summon
  }

  var Antic Dust 1: gameObject = <null>
  var Antic Dust 2: gameObject = <null>
  var Antic Rocks 1: gameObject = <null>
  var Antic Rocks 2: gameObject = <null>
  var Audio Player: gameObject = <null>
  var Self: gameObject = <null>
}
