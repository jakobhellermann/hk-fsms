# Spawn
# 1 location(s):
#   Knight_Pickup / Knight/Effects/SD Crystal Gen W2

fsm Spawn {
  start Init
  on SUPERDASH BLAST → Idle  // from any state
  on SUPERDASH CANCEL → Idle  // from any state
  on SUPERDASH READY → Idle  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on SUPERDASH CHARGING W → Gen Init
  }

  state Gen Pause {
    Wait(time=0.09, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gen
  }

  state Gen Init {
    SetFloatValue(floatVariable=var "Scale", floatValue=0.5, everyFrame=false)
    SetFloatValue(floatVariable=var "Tilt Crt", floatValue=var "Tilt", everyFrame=false)
    SendEventByScale(gameObject=var "Hero", eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on R → Wall R
    on L → Wall L
  }

  state Gen {
    SpawnObjectFromGlobalPool(gameObject=SD Crystal (resources.assets), spawnPoint=var "Self", position=var "Spawn Pos", rotation=(unset), storeObject=var "Crystal")
    SetRotation(gameObject=var "Crystal", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tilt Crt", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Crystal", vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    FloatAdd(floatVariable=var "Scale", add=0.12, everyFrame=false, perSecond=false)
    on FINISHED → Gen Pause
  }

  state Wall R {
    FloatAdd(floatVariable=var "Tilt Crt", add=-90, everyFrame=false, perSecond=false)
    SetVector3Value(vector3Variable=var "Spawn Pos", vector3Value=(-0.5, 0, 0), everyFrame=false)
    on FINISHED → Gen Pause
  }

  state Wall L {
    FloatAdd(floatVariable=var "Tilt Crt", add=90, everyFrame=false, perSecond=false)
    SetVector3Value(vector3Variable=var "Spawn Pos", vector3Value=(0.5, 0, 0), everyFrame=false)
    on FINISHED → Gen Pause
  }

  var Tilt: float = -10

  var Hero Scale: float = 0
  var Scale: float = 0
  var Tilt Crt: float = 0
  var Init Pos: vector3 = (0, -1.42, -0.001)
  var Spawn Pos: vector3 = (0, 0, 0)
  var Crystal: gameObject = <null>
  var Self: gameObject = <null>
}
