# Battle Control
# 1 location(s):
#   Fungus3_10 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Completed", storeResult=var "Completed")
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Wave 2", storeResult=var "Wave 2")
    FindChild(gameObject=Self, childName="Wave 3", storeResult=var "Wave 3")
    FindChild(gameObject=Self, childName="Wave 4", storeResult=var "Wave 4")
    FindChild(gameObject=Self, childName="Wave 5", storeResult=var "Wave 5")
    FindChild(gameObject=Self, childName="Wave 6", storeResult=var "Wave 6")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on ACTIVATE → Activated
    on HIT → Wave 1
  }

  state Activated {
    ActivateAllChildren(gameObject=var "Completed", activate=true)
    DestroyObject(gameObject=var "Wave 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 2", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 3", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 4", delay=0, detachChildren=false)
  }

  state Wave 1 {
    TransitionToAudioSnapshot(snapshot=Action (resources.assets), transitionTime=1)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Wave 1", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=1, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Wave 1"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on END → Pause 1
  }

  state Wave 2 {
    ActivateAllChildren(gameObject=var "Wave 2", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=3, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Wave 2"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Wave 2"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on END → Pause 2
  }

  state Wave 3 {
    ActivateAllChildren(gameObject=var "Wave 3", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Wave 3"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on END → Lil Pause
  }

  state Wave 4 {
    ActivateAllChildren(gameObject=var "Wave 4", activate=true)
    IntAdd(intVariable=var "Battle Enemies", add=1, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Wave 4"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on END → Pause 3
  }

  state End Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=2)
  }

  state Pause 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 2
  }

  state Pause 2 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 3
  }

  state Wave 5 {
    ActivateAllChildren(gameObject=var "Wave 5", activate=true)
    SendEventByName(eventTarget=GameObject(var "Wave 5"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Battle Enemies", intValue=3, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=1, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Wave 5"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on END → Pause 4
  }

  state Pause 3 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 5
  }

  state Wave 6 {
    ActivateAllChildren(gameObject=var "Wave 6", activate=true)
    IntAdd(intVariable=var "Battle Enemies", add=1, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Wave 6"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on END → End Pause
  }

  state Pause 4 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 6
  }

  state Lil Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 4
  }

  var Battle Enemies: int = 0
  var Activated: bool = false
  var CamLock: gameObject = <null>
  var Completed: gameObject = <null>
  var Wave 1: gameObject = <null>
  var Wave 2: gameObject = <null>
  var Wave 3: gameObject = <null>
  var Wave 4: gameObject = <null>
  var Wave 5: gameObject = <null>
  var Wave 6: gameObject = <null>
}
