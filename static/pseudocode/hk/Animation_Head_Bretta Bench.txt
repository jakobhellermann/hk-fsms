fsm Animation {
  start Init

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Idle")
    on LOOK R → Look R
    on LOOK L → Look L
  }

  state Look R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head LookRight")
    on TO IDLE → To Idle
  }

  state Look L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head LookLeft")
    on TO IDLE → To Idle
  }

  state To Idle {
    SetPlayerDataBool(boolName="brettaSeenBench", value=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Head LookEnd", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Bretta_startle_01 (sharedassets7.assets), Bretta_startle_02 (sharedassets7.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
    on LOOK L → Look L
    on LOOK R → Look R
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
}
