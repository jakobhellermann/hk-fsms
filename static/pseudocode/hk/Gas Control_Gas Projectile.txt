# Gas Control
# 1 location(s):
#   sharedassets167.assets / Gas Projectile

fsm Gas Control {
  start Reset

  state Reset {
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Particle", emit=0)
    SetFloatValue(floatVariable=var "Scaler", floatValue=1, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    FloatAddV2(floatVariable=var "Scaler", add=2, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=(unset))
    SetScale(gameObject=Self, vector=(unset), x=var "Scaler", y=var "Scaler", z=var "Scaler", everyFrame=true, lateUpdate=false)
    Wait(time=0.75, finishEvent=→"END", realTime=false)
    on END → Stop
    on ORBIT SHIELD → Stop Velocity
  }

  state Recycle {
    RecycleSelf()
  }

  state Stop {
    StopParticleEmitter(gameObject=var "Particle")
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Stop Velocity {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Stop
  }

  var Scaler: float = 0
  var Wait Time: float = 0
  var Particle: gameObject = Gas Projectile/Gas Attack
}
