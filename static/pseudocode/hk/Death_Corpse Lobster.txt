# Death
# 1 location(s):
#   sharedassets34.assets / Corpse Lobster

fsm Death {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD STILL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC SILENT", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Chunks", storeResult=var "Chunks")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_final_hit (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    CreateObject(gameObject=Death Wave Infected (sharedassets32.assets), spawnPoint=var "Self", position=(0, 0, -1), rotation=(unset), storeObject=var "Wave", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Wave", vector=(unset), x=3, y=3, z=(unset), everyFrame=false, lateUpdate=false)
    SpawnBloodTime(delay=0.1, spawnPoint=var "Self", position=(0, 1.8, 0.002), spawnMin=1, spawnMax=2, speedMin=20, speedMax=30, angleMin=75, angleMax=105, colorOverride=(unset))
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Steam
  }

  state Steam {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CORPSE STEAM", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="LOBSTER KILLED", delay=0, everyFrame=false)
    SpawnBloodTime(delay=0.1, spawnPoint=var "Self", position=(0, 0, 0.002), spawnMin=12, spawnMax=15, speedMin=24, speedMax=30, angleMin=0, angleMax=360, colorOverride=(unset))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_gushing (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Death Puff Boss (sharedassets32.assets), spawnPoint=var "Self", position=(0, 0, -5), rotation=(unset), storeObject=var "Smoker", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Camera", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    FloatAdd(floatVariable=var "emissionRate", add=5, everyFrame=true, perSecond=false)
    FloatAdd(floatVariable=var "emissionSpeed", add=1, everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "emissionSpeed", minValue=0, maxValue=110, everyFrame=true)
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Ready
  }

  state Blow {
    ActivateGameObject(gameObject=var "Chunks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CORPSE BLOW", delay=0, everyFrame=false)
    EndGGBossScene()
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BOSS DEATH", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Camera", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_explode (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="StopRumble", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Smoker", emission=false)
    SpawnBlood(spawnPoint=var "Self", position=(0, 0, 0), spawnMin=90, spawnMax=90, speedMin=15, speedMax=40, angleMin=60, angleMax=120, colorOverride=(unset))
    CreateObject(gameObject=Death Explode Boss (sharedassets32.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Ready {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CORPSE READY", delay=0, everyFrame=false)
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    EaseFloat(fromValue=var "emissionRate", toValue=0, floatVariable=var "emissionRate", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "emissionSpeed", toValue=0, floatVariable=var "emissionSpeed", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Blow
  }

  var emissionRate: float = 50
  var emissionSpeed: float = 5
  var Camera: gameObject = <null>
  var Chunks: gameObject = <null>
  var Self: gameObject = <null>
  var Smoker: gameObject = <null>
  var Wave: gameObject = <null>
}
