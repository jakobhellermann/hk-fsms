fsm Dream Return {
  start Idle

  state Fade Out {
    SendEventByName(eventTarget=GameObject(var "MainCamera"), sendEvent="FADE OUT INSTANT", delay=0, everyFrame=false)  // disabled
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    on FINISHED → Heal
  }

  state New Scene {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    BeginSceneTransition(sceneName=var "Return Scene", entryGateName=var "Entry Gate", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  state Idle {
    on DREAM EXIT → Cinematic?
    on DREAM RETURN → Statue
  }

  state Get PlayerData {
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    GetPlayerDataString(gameObject=var "GameManager", stringName="bossReturnEntryGate", storeValue=var "Entry Gate")
    SetStaticVariable(variableName="finishedBossReturning", setValue=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → New Scene
  }

  state Statue {
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    GetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", storeValue=var "Return Scene")
    on FINISHED → Fade Out
  }

  state Door {
    GetStaticVariable(variableName="ggEndScene", storeValue=var "Return Scene")
    on FINISHED → Get PlayerData
  }

  state Heal {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="MaxHealth", parameters=[], storeResult=(unset var))
    on FINISHED → Get PlayerData
  }

  state Cinematic? {
    CheckStaticBool(variableName="ggCinematicEnding", trueEvent=→"CINEMATIC", falseEvent=→"FINISHED")
    on CINEMATIC → Cinematic
    on FINISHED → Door
  }

  state Cinematic {
    SetStaticVariable(variableName="ggCinematicEnding", setValue=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="givenGodseekerFlower", isTrue=→"ENDING E", isFalse=→"ENDING D")
    on ENDING D → Get Scene D
    on ENDING E → Get Scene E
  }

  state Change Scene {
    SetStaticVariable(variableName="skipEndWhiteFader", setValue=true)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=[var "Cinematic Scene", "door1", 0], storeResult=(unset var))
  }

  state Get Scene D {
    GetConstantsValue(variableName="ENDING_D_CINEMATIC", storeValue=var "Cinematic Scene")
    on FINISHED → Change Scene
  }

  state Get Scene E {
    GetConstantsValue(variableName="ENDING_E_CINEMATIC", storeValue=var "Cinematic Scene")
    on FINISHED → Change Scene
  }

  var Cinematic Scene: string = ""
  var Entry Gate: string = ""
  var Return Scene: string = "Crossroads_10"
  var Dream Fall Catcher: gameObject = <null>
}
