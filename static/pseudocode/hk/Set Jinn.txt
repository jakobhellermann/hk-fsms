# Set Jinn
# 1 location(s):
#   Town / Jiji Door/door_jiji

fsm Set Jinn {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="permadeathMode", storeValue=var "PermaDeath Mode")
    IntCompare(integer1=var "PermaDeath Mode", integer2=0, equal=(none), lessThan=(none), greaterThan=→"JINN", everyFrame=false)
    on JINN → Change
  }

  state Change {
    SetFsmString(gameObject=Self, fsmName="Door Control", variableName="New Scene", setValue="Room_Jinn", everyFrame=false)
  }

  var PermaDeath Mode: int = 0
}
