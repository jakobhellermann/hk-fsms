fsm Control Lever {
  start Init
  on MOVE UP → Move Up  // from any state
  on LIFT STOP → Return Anim  // from any state
  on MOVE DOWN → Move Down  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    SetStringValue(stringVariable=var "Return Anim", stringValue="Lift Switch Idle Return", everyFrame=false)
    on NAIL HIT → Hit Effect
  }

  state Check Direction {
    BoolTest(boolVariable=var "Going Down", isTrue=→"DOWN", isFalse=→"UP", everyFrame=false)
    on DOWN → Down
    on UP → Up
  }

  state Hit Effect {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_switch (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))  // disabled
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Check Direction
  }

  state Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lift Switch Right")
    SetBoolValue(boolVariable=var "Going Down", boolValue=true, everyFrame=false)
    SetStringValue(stringVariable=var "Return Anim", stringValue="Lift Switch Right Back", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="CONTROL DOWN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Wait For Lift Move
  }

  state Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lift Switch Left")
    SetBoolValue(boolVariable=var "Going Down", boolValue=false, everyFrame=false)
    SetStringValue(stringVariable=var "Return Anim", stringValue="Lift Switch Left Back", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="CONTROL UP", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Wait For Lift Move
  }

  state Return Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Return Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle Anim
  }

  state Flip Direction {
    BoolFlip(boolVariable=var "Going Down")
    on FINISHED → Hit Effect
  }

  state Move Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lift Switch Right")
    SetBoolValue(boolVariable=var "Going Down", boolValue=true, everyFrame=false)
    SetStringValue(stringVariable=var "Return Anim", stringValue="Lift Switch Right Back", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lift Switch Idle")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pause Frame
  }

  state Move Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lift Switch Left")
    SetBoolValue(boolVariable=var "Going Down", boolValue=false, everyFrame=false)
    SetStringValue(stringVariable=var "Return Anim", stringValue="Lift Switch Left Back", everyFrame=false)
    on FINISHED → Idle
  }

  state Hero Check {
    BoolTest(boolVariable=var "Hero Check", isTrue=→"IN RANGE", isFalse=(none), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on IN RANGE → Hit Effect
  }

  state Wait For Hit {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Flip Direction
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Wait 1 {
  }

  state Wait For Lift Move {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    on CAN HIT → Wait For Hit
  }

  var Going Down: bool = false
  var Hero Check: bool = false
  var Return Anim: string = ""
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
