# proxy_event_handler
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Prompts/First Map/Button
#   resources.assets / _GameCameras/HudCamera/Prompts/First Map/Button

fsm proxy_event_handler {
  start Init

  state Init {
    on DOWN → Send Down
    on UP → Send Up
    on DOWN INSTANT → 
    on UP INSTANT → 
    on STOP → 
    on PULSE DOWN → 
  }

  state Send Down {
    SendEventByName(eventTarget=GameObject(_GameCameras/HudCamera/Prompts/First Map/Button/Label), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Init
  }

  state Send Up {
    SendEventByName(eventTarget=GameObject(_GameCameras/HudCamera/Prompts/First Map/Button/Label), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Init
  }
}
