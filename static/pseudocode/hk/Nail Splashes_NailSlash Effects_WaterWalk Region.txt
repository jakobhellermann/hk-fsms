# Nail Splashes
# 2 location(s):
#   RestingGrounds_08 / WaterWalk Region/NailSlash Effects
#   RestingGrounds_08 / WaterWalk Region/NailSlash Effects

fsm Nail Splashes {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=(none), storeCollider=var "")  // disabled
    ReceivedDamage(collideTag="Nail Attack", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on HIT → Check Direction
  }

  state Check Direction {
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Direction", everyFrame=false)
    FloatSwitch(floatVariable=var "Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on UP → Slash Up
    on DOWN → Pause
    on LEFT → Slash Left
    on RIGHT → Slash Right
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Slash Left {
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(-2, -0.5, -0.01), spawnMin=6, spawnMax=8, speedMin=10, speedMax=20, angleMin=120, angleMax=150, originVariationX=0.75, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(0.5, -0.25, -0.01), spawnMin=2, spawnMax=6, speedMin=9, speedMax=15, angleMin=20, angleMax=55, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[water_cut_01 (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Water Cut (resources.assets), spawnPoint=var "Hero", position=(-3.3, -0.25, -0.01), rotation=(unset), storeObject=var "Effect")
    SetScale(gameObject=var "Effect", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state Slash Right {
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(2, -0.25, -0.01), spawnMin=6, spawnMax=8, speedMin=10, speedMax=20, angleMin=30, angleMax=60, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(-0.5, -0.5, -0.01), spawnMin=2, spawnMax=6, speedMin=9, speedMax=15, angleMin=125, angleMax=160, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    SpawnObjectFromGlobalPool(gameObject=Water Cut (resources.assets), spawnPoint=var "Hero", position=(3.3, -0.25, -0.01), rotation=(unset), storeObject=var "Effect")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[water_cut_01 (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetScale(gameObject=var "Effect", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state Slash Up {
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(0, -0.5, -0.01), spawnMin=18, spawnMax=20, speedMin=5, speedMax=25, angleMin=85, angleMax=95, originVariationX=0.75, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[water_cut_01 (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Splash Out Small (resources.assets), spawnPoint=var "Hero", position=(0, -1, 0), rotation=(unset), storeObject=var "")
    on FINISHED → Pause
  }

  var Direction: float = 0
  var Damager: gameObject = <null>
  var Effect: gameObject = <null>
}
