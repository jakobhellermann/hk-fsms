fsm Movement {
  start Start
  on ATTACK START → Attacking  // from any state

  state Init {
    SetParent(gameObject=var "Target", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Warp In
  }

  state Warp In {
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Attack Pt", emit=0)
    on FINISHED → Init Choice
  }

  state Choose Target {
    SendRandomEventV3(events=[(none), (none), (none), (none), (none), (none), (none)], weights=[1, 1, 1, 1, 1, 1, 1], trackingInts=[var "Ct 1", var "Ct 2", var "Ct 3", var "Ct 4", var "Ct 5", var "Ct 6", var "Ct 7"], eventMax=[1, 1, 1, 1, 1, 1, 1], trackingIntsMissed=[var "Ms 1", var "Ms 2", var "Ms 3", var "Ms 4", var "Ms 5", var "Ms 6", var "Ms 7"], missedMax=[7, 7, 7, 7, 7, 7, 7])
    on 1 → Set 1
    on 2 → Set 2
    on 3 → Set 3
    on 4 → Set 4
    on 5 → Set 5
    on 6 → Set 6
    on 7 → Set 7
  }

  state Set 1 {
    SetPosition(gameObject=var "Target", vector=var "P1", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 2 {
    SetPosition(gameObject=var "Target", vector=var "P2", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 3 {
    SetPosition(gameObject=var "Target", vector=var "P3", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 4 {
    SetPosition(gameObject=var "Target", vector=var "P4", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 5 {
    SetPosition(gameObject=var "Target", vector=var "P5", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 6 {
    SetPosition(gameObject=var "Target", vector=var "P6", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 7 {
    SetPosition(gameObject=var "Target", vector=var "P7", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Hover {
    ChaseObject(gameObject=Self, target=var "Target", speedMax=5, acceleration=0.3, targetSpread=(unset), spreadResetTimeMin=(unset), spreadResetTimeMax=(unset))
    WaitRandom(timeMin=4, timeMax=5, finishEvent=→"FINISHED", realTime=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToIdle", resetFrame=true, everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=36, tolerance=0, equal=(none), lessThan=→"RETURN", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Self X", float2=76, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"RETURN", everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=30, tolerance=0, equal=(none), lessThan=→"RETURN", greaterThan=(none), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Hero Left", var "Self Left"], boolStates=[true, true], trueEvent=→"WARP R", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hero Left", var "Self Left"], boolStates=[false, false], trueEvent=→"WARP L", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Choose Target
    on RETURN → Warp Out
    on TOOK DAMAGE → Warp Check
    on WARP → Set Warp
  }

  state Attacking {
    DecelerateV2(gameObject=Self, deceleration=0.9)
    on ATTACK END → Choose Target
  }

  state Init Choice {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="READY", delay=0, everyFrame=false)
    SendRandomEvent(events=[(none), (none), (none)], weights=[1, 1, 1], delay=0)
    on 5 → Set 5
    on 6 → Set 6
    on 7 → Set 7
  }

  state Warp Out {
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK STOP", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Return
  }

  state Return {
    SetPosition(gameObject=Self, vector=(unset), x=56, y=41, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK OK", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Choose Target
  }

  state Warp Check {
    SendRandomEvent(events=[→"CANCEL", (none)], weights=[0.75, 0.25], delay=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on CANCEL → Hover
    on FINISHED → Set Warp
  }

  state Choose R {
    SelectRandomVector3(vector3Array=[var "P4", var "P6"], weights=[1, 1], storeVector3=var "Warp Pos")
    on FINISHED → Warp Out Fx
  }

  state Choose L {
    SelectRandomVector3(vector3Array=[var "P3", var "P7"], weights=[1, 1], storeVector3=var "Warp Pos")
    on FINISHED → Warp Out Fx
  }

  state Warp {
    GetPosition(gameObject=Self, vector=var "Prev Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Warp Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Warp Out", vector=var "Prev Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_knight_teleport (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Hover
  }

  state Set Warp {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Self X", float2=56, tolerance=0, equalBool=(unset), lessThanBool=var "Self Left", greaterThanBool=(unset), everyFrame=false)
    FloatTestToBool(float1=var "Hero X", float2=56, tolerance=0, equalBool=(unset), lessThanBool=var "Hero Left", greaterThanBool=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hero Left", var "Self Left"], boolStates=[true, true], trueEvent=→"WARP R", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hero Left", var "Self Left"], boolStates=[false, false], trueEvent=→"WARP L", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Hover
    on WARP L → Choose L
    on WARP R → Choose R
  }

  state Warp Out Fx {
    ActivateGameObject(gameObject=var "Warp Out", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Warp
  }

  state Start {
    BoolTest(boolVariable=var "Init", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Init", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  var P1: vector3 = (56.06, 41, 0.006)
  var P2: vector3 = (56.06, 36, 0.006)
  var P3: vector3 = (46.41, 36, 0.006)
  var P4: vector3 = (65.78, 36, 0.006)
  var P5: vector3 = (56.06, 37.7, 0.006)
  var P6: vector3 = (62.42, 37.7, 0.006)
  var P7: vector3 = (49.8, 37.7, 0)

  var Hero X: float = 0
  var Self X: float = 0
  var Self Y: float = 0
  var Ct 1: int = 0
  var Ct 2: int = 0
  var Ct 3: int = 0
  var Ct 4: int = 0
  var Ct 5: int = 0
  var Ct 6: int = 0
  var Ct 7: int = 0
  var Ms 1: int = 0
  var Ms 2: int = 0
  var Ms 3: int = 0
  var Ms 4: int = 0
  var Ms 5: int = 0
  var Ms 6: int = 0
  var Ms 7: int = 0
  var Hero Left: bool = false

  var Attack Pt: gameObject = Warrior/Ghost Warrior Slug/Attack Pt
  var Self: gameObject = <null>
  var Target: gameObject = Warrior/Ghost Warrior Slug/Target

  var White Flash: gameObject = Warrior/Ghost Warrior Slug/White Flash
}
