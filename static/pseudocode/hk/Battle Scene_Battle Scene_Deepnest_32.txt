# Battle Scene
# 1 location(s):
#   Deepnest_32 / Battle Scene

fsm Battle Scene {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="CameraLockArea", storeResult=var "Cam Lock")
    FindGameObject(objectName="Mimic Spider", withTag="Untagged", store=var "Mimic Spider")
    FindChild(gameObject=Self, childName="Gate Closers", storeResult=var "Gate Closers")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="encounteredMimicSpider", isTrue=→"ENCOUNTERED", isFalse=(none))
    on BATTLE START → Battle
    on ACTIVATE → Activate
    on ENCOUNTERED → Encountered
  }

  state Idle {
    on KILLED → Set Activation
  }

  state Wait {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Cam Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set Activation {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on BOSS DEATH → Wait
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Battle {
    ActivateGameObject(gameObject=var "Cam Lock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="encounteredMimicSpider", value=true)
    on FINISHED → Idle
  }

  state Activate {
    ActivateGameObject(gameObject=var "Mimic Spider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate Closers", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
  }

  state Encountered {
    SendEventByName(eventTarget=GameObject(var "Mimic Spider"), sendEvent="ENCOUNTERED", delay=0, everyFrame=false)
    on BATTLE START → Battle
  }

  var Activated: bool = false
  var Cam Lock: gameObject = <null>
  var Gate Closers: gameObject = <null>
  var Mimic Spider: gameObject = <null>
}
