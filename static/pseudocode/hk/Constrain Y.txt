fsm Constrain Y {
  start Check

  state Check {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=6.3, tolerance=0, equal=(none), lessThan=→"CONSTRAIN", greaterThan=(none), everyFrame=true)
    on CONSTRAIN → Constrain
  }

  state Constrain {
    SetFloatValue(floatVariable=var "Self Y", floatValue=6.3, everyFrame=false)  // disabled
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=6.4, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Check
  }

  var Self Y: float = 0
}
