# Control
# 1 location(s):
#   Dream_Abyss / End Cutscene

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="CameraLockArea", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Dummy", storeResult=var "Dummy")
    FindChild(gameObject=Self, childName="Audio", storeResult=var "Audio")
    FindChild(gameObject=Self, childName="Dummy Over", storeResult=var "Dummy Over")
    FindChild(gameObject=Self, childName="Hollow Knight", storeResult=var "Hollow Knight")
    FindChild(gameObject=Self, childName="King", storeResult=var "King")
    FindChild(gameObject=Self, childName="Flasher", storeResult=var "Flasher")
    FindChild(gameObject=Self, childName="FallCam", storeResult=var "FallCam")
    FindChild(gameObject=Self, childName="Plat", storeResult=var "Plat")
    FindChild(gameObject=Self, childName="Mask", storeResult=var "Mask")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Start
  }

  state Start {
    SetPlayerDataBool(boolName="disablePause", value=true)
    AudioStop(gameObject=var "Audio")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=EndTakeNoDamage())
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CORPSE END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE START", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=true)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dummy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flasher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=(unset), y=(unset), everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Plat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → King Look
  }

  state King Look {
    Tk2dPlayAnimation(gameObject=var "King", animLibName="", clipName="King Turn")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → King Leave
  }

  state King Leave {
    Tk2dPlayAnimation(gameObject=var "King", animLibName="", clipName="King Walk")
    SetVelocity2d(gameObject=var "King", vector=(unset), x=5, y=(unset), everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → HK Step
  }

  state HK Step {
    AudioPlaySimple(gameObject=var "Hollow Knight", volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=var "Hollow Knight", animLibName="", clipName="HK Walk")
    SetVelocity2d(gameObject=var "Hollow Knight", vector=(unset), x=3, y=(unset), everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Clamber1
  }

  state Clamber Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hollow Knight", clipName="HK Stop", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=var "Hollow Knight", vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioStop(gameObject=var "Hollow Knight")
    on FINISHED → Look
  }

  state HK Look End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hollow Knight", clipName="HK Look End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Leave Pause
  }

  state Look {
    Tk2dPlayAnimation(gameObject=var "Hollow Knight", animLibName="", clipName="HK Look")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → HK Look End
  }

  state Leave Pause {
    Tk2dPlayAnimation(gameObject=var "Hollow Knight", animLibName="", clipName="HK Idle")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → HK Leave
  }

  state HK Leave {
    Tk2dPlayAnimation(gameObject=var "Hollow Knight", animLibName="", clipName="HK Walk")
    SetVelocity2d(gameObject=var "Hollow Knight", vector=(unset), x=4.5, y=(unset), everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=var "Hollow Knight", volume=1, oneShotClip=<null>)
    on FINISHED → Falter
  }

  state Falter {
    Tk2dPlayAnimation(gameObject=var "Dummy", animLibName="", clipName="Knight Falter")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  state Clamber1 {
    Tk2dPlayAnimation(gameObject=var "Dummy", animLibName="", clipName="Knight Grab")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Clamber Up
  }

  state Fall {
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=misc_rumble_loop (sharedassets101.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=1)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "FallCam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dummy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dummy Over", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Dummy Over", animLibName="", clipName="Knight Fall")
    iTweenMoveBy(gameObject=var "Dummy Over", id="", vector=(0, -13, 0), time=1, delay=0, speed=(unset), easeType=easeInCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    ActivateGameObject(gameObject=var "Mask", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    iTweenMoveBy(gameObject=var "Mask", id="", vector=(0, 25, 0), time=1, delay=0, speed=(unset), easeType=easeInCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=3.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Black
  }

  state Shake {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=misc_rumble_impact (sharedassets13.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=break_wall_after_tutorial_area (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=var "Hollow Knight")
    on FINISHED → Fall
  }

  state Black {
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="NONE", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Cancel Reset", setValue=true, everyFrame=false)  // disabled
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.001, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Master Silence (sharedassets0.assets), transitionTime=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Text 1
  }

  state Text 1 {
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=false, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=true, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    TransitionToAudioSnapshot(snapshot=Master On (sharedassets0.assets), transitionTime=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=Lore_Tablet_activate_temp (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["KING_ABYSS_FINAL_A", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Text 2
  }

  state Charm Pause {
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Charm Get
  }

  state Charm Get {
    SetPlayerDataInt(intName="royalCharmState", value=4)
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmCost_36", storeValue=var "Notch Decrement")
    IntOperator(integer1=var "Notch Decrement", integer2=-1, operation=Multiply, storeResult=var "Notch Decrement", everyFrame=false)
    PlayerDataIntAdd(gameObject=var "GameManager", intName="charmSlotsFilled", amount=var "Notch Decrement")
    SetPlayerDataInt(intName="charmCost_36", value=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHARM INDICATOR CHECK", delay=0, everyFrame=false)
    on FINISHED → Remove Overcharm
  }

  state Text 2 {
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["KING_ABYSS_FINAL_B", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Text 3
  }

  state Text 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["KING_ABYSS_FINAL_C", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Charm Pause
  }

  state Return Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return
  }

  state Return {
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=TimePasses())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=ResetSemiPersistentItems())
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    GetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", storeValue="")
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=["Abyss_15", "door_dreamReturn", 0], storeResult=(unset var))  // disabled
    BeginSceneTransition(sceneName="Abyss_15", entryGateName="door_dreamReturn", entryDelay=0, visualization=Custom, preventCameraFadeOut=true)
  }

  state Get Msg {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    CreateObject(gameObject=UI Msg Get WhiteCharm (sharedassets137.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on GET ITEM MSG END → Return Pause
  }

  state Remove Overcharm {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="overcharmed", isTrue=(none), isFalse=→"FINISHED")
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlots", storeValue=var "Notches")
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlotsFilled", storeValue=var "Notches Filled")
    IntCompare(integer1=var "Notches Filled", integer2=var "Notches", equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetPlayerDataBool(boolName="overcharmed", value=false)
    on FINISHED → Get Msg
  }

  var Notch Decrement: int = 0
  var Notches: int = 0
  var Notches Filled: int = 0
  var Audio: gameObject = <null>
  var CamLock: gameObject = <null>
  var Dummy: gameObject = <null>
  var Dummy Over: gameObject = <null>
  var FallCam: gameObject = <null>
  var Flasher: gameObject = <null>
  var Hollow Knight: gameObject = <null>
  var King: gameObject = <null>
  var Mask: gameObject = <null>
  var Plat: gameObject = <null>
}
