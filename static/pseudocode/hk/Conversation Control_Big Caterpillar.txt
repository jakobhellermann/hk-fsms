# Conversation Control
# 1 location(s):
#   Deepnest_East_04 / Big Caterpillar

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="BIG_CATERPILLAR", everyFrame=false)
    on FINISHED → Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Meet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BIGCAT_INTRO", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="bigCatMeet", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Caterpillar_talk_01 (sharedassets309.assets), Caterpillar_talk_02 (sharedassets309.assets), Caterpillar_talk_03 (sharedassets309.assets), Caterpillar_talk_04 (sharedassets309.assets), Caterpillar_talk_05 (sharedassets309.assets), Caterpillar_talk_06 (sharedassets309.assets), Caterpillar_talk_07 (sharedassets309.assets)], weights=[1, 1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk Finish
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Choice {
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    GetPlayerDataBool(boolName="bigCatShadeConvo", storeValue=var "Shade Convo")
    IntTestToBool(int1=var "Royal Charm State", int2=4, equalBool=var "Has Shade Charm", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bigCatMeet", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="bigCatHitTail", falseBool="bigCatHitTailConvo", isTrue=→"HIT TAIL", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasKingsBrand", falseBool="bigCatKingsBrandConvo", isTrue=→"KINGS BRAND", isFalse=(none))
    BoolTestMulti(boolVariables=[var "Has Shade Charm", var "Shade Convo"], boolStates=[true, false], trueEvent=→"SHADE CHARM", falseEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bigCatTalk1", isTrue=(none), isFalse=→"CONVO 1")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bigCatTalk2", isTrue=(none), isFalse=→"CONVO 2")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bigCatTalk3", isTrue=(none), isFalse=→"CONVO 3")
    on MEET → Meet
    on CONVO 1 → Convo 1
    on CONVO 2 → Convo 2
    on CONVO 3 → Convo 3
    on HIT TAIL → Hit Tail
    on KINGS BRAND → Kings Brand
    on SHADE CHARM → Shade Charm
    on FINISHED → Repeat
  }

  state Convo 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BIGCAT_TALK_01", "Minor NPC"], storeResult=(unset var))

    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BIGCAT_REPEAT", "Minor NPC"], storeResult=(unset var))
