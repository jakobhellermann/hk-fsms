# Parry Damage
# 1 location(s):
#   RestingGrounds_08 / Ghost Battle Revek

fsm Parry Damage {
  start Idle

  state Idle {
    on PARRIED → Pause
  }

  state Pause {
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TAKE DAMAGE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }
}
