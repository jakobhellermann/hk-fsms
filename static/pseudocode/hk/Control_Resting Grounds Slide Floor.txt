# Control
# 1 location(s):
#   RestingGrounds_06 / Resting Grounds Slide Floor

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"DESTROY", isFalse=(none), everyFrame=false)
    on DESTROY → Destroy
    on ACTIVATE → Antic
  }

  state Destroy {
    DestroySelf(detachChildren=false)  // disabled
    SetCollider(gameObject=Self, active=false)
    Translate(gameObject=Self, vector=(unset), x=4.9, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
  }

  state Antic {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Whole Dust", emit=0)
    PlayParticleEmitter(gameObject=var "Plat Stream Dust", emit=0)
    iTweenMoveBy(gameObject=var "Sprite", id="", vector=(0.2, 0, 0), time=0.5, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=jiji_door_open_pt_1 (sharedassets7.assets))
    on FINISHED → Slide
  }

  state Slide {
    iTweenMoveBy(gameObject=Self, id="", vector=(3.7, 0, 0), time=0.5, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Tween Out
  }

  state Tween Out {
    iTweenMoveBy(gameObject=Self, id="", vector=(-0.2, 0, 0), time=0.5, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Slam Dust", emit=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=jiji_door_open_pt_2 (sharedassets7.assets))
    on FINISHED → Tween In
  }

  state Tween In {
    iTweenMoveBy(gameObject=Self, id="", vector=(0.2, 0, 0), time=0.75, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetCollider(gameObject=Self, active=false)
  }

  var Activated: bool = false
  var Plat Stream Dust: gameObject = Resting Grounds Slide Floor/Plat Stream Dust
  var Slam Dust: gameObject = Resting Grounds Slide Floor/Slam Dust
  var Sprite: gameObject = Resting Grounds Slide Floor/Sprite
  var Whole Dust: gameObject = Resting Grounds Slide Floor/Whole Dust
}
