# Fall
# 1 location(s):
#   GG_Dung_Defender / GG_Arena_Prefab/Dung Corpse BG

fsm Fall {
  start Idle

  state Idle {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z", space=0, everyFrame=false, lateUpdate=false)
    on DUNG CORPSE FALL → Fall
  }

  state Fall {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Fly")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-15, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Smoke", emit=0)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=12.97, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    on LAND → Keep Falling?
  }

  state Land {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Hero", audioClip=Boss Defeat (sharedassets40.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Hero", audioClip=Dung_def_death_part_03 (sharedassets354.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Hero", audioClip=dung_defender_dive (sharedassets354.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHARM DROP", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Slam Chunks", emit=0)
    StopParticleEmitter(gameObject=var "Smoke")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Splat")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=12.97, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sink
  }

  state Sink {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-0.5, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Sink")
    on FINISHED → End Battle
  }

  state End {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
  }

  state End Battle {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Keep Falling? {
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=→"FINISHED")
    on GG BOSS → End boss scene
    on FINISHED → Land
  }

  state End boss scene {
    EndGGBossScene()
  }

  var Z: float = 35.38

  var Y Pos: float = 0
  var Self: gameObject = <null>
  var Slam Chunks: gameObject = GG_Arena_Prefab/Dung Corpse BG/Slam Chunks
  var Smoke: gameObject = GG_Arena_Prefab/Dung Corpse BG/Smoke
  var Thunk Effect: gameObject = GG_Arena_Prefab/Dung Corpse BG/Thunk Effect
}
