# Fluke Fly
# 2 location(s):
#   Waterways_12_boss / Hatcher Cage (2)/Fluke Fly Spawner (2)
#   GG_Flukemarm / Hatcher Cage (2)/Fluke Fly Spawner (2)

fsm Fluke Fly {
  start Init
  on CENTIPEDE DEATH → Death  // from any state

  state Spawn {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=fluke_fairy_call (sharedassets27.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Chase
  }

  state Chase {
    SetAudioClip(gameObject=Self, audioClip=fluke_fairy_charge_loop (sharedassets350.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    FaceAngle(gameObject=Self, angleOffset=180, everyFrame=true)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=15, accelerationForce=35, offsetX=0, offsetY=0)
    on FINISHED → 
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Cage")
    on FINISHED → Inert
  }

  state Inert {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on SPAWN → Spawn
  }

  state Death {
    SetIsDead(target=Self, setValue=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    PlayParticleEmitter(gameObject=var "Pt Death", emit=0)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset
  }

  state Reset {
    SetHP(target=Self, hp=13)
    SetParent(gameObject=Self, parent=var "Cage", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Inert
  }

  var Cage: gameObject = <null>
  var Pt Death: gameObject = Hatcher Cage (2)/Fluke Fly Spawner (2)/Pt Death
  var Self: gameObject = <null>
}
