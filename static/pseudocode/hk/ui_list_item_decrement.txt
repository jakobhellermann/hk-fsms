// uses template: ui_list_item_decrement
fsm ui_list_item_decrement {
  start Idle

  state Idle {
    on DECREMENT LIST NUMBER → Decrement
  }

  state Decrement {
    GetFsmInt(gameObject=Self, fsmName="ui_list_item", variableName="Item Number", storeValue=var "Item Number", everyFrame=false)
    IntOperator(integer1=var "Item Number", integer2=1, operation=Subtract, storeResult=var "Item Number", everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="ui_list_item", variableName="Item Number", setValue=var "Item Number", everyFrame=false)
    on FINISHED → Idle
  }

  var Item Number: int = 0
}
