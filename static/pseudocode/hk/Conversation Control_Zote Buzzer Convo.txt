# Conversation Control
# 1 location(s):
#   sharedassets148.assets / Zote Buzzer Convo

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Check Direction
  }

  state Convo Choice {
    GetPlayerDataInt(gameObject=var "GameManager", intName="zote", storeValue=var "NPC State")
    IntSwitch(intVariable=var "NPC State", compareTo=[1, 2, 0], sendEvent=[→"MEET 1", →"MEET 2", →"MEET 1"], everyFrame=false)
    on MEET 1 → Buzzer 1
    on MEET 2 → Buzzer 2
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → Check Direction 2
  }

  state Check Direction {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Talk R
    on LEFT → Talk Back
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk L")
    on FINISHED → Box Up
  }

  state Talk Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk R")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Check Direction 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Face Right
    on LEFT → Face Left
  }

  state Face Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Right")
    on FINISHED → End
  }

  state Face Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Left")
    on FINISHED → End
  }

  state Buzzer 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_02 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataInt(intName="zote", value=2)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_BUZZER_1", "Zote"], storeResult=(unset var))
    on CONVO_FINISH → Buzzer 1 B
  }

  state Buzzer 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_01 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ZOTE", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_BUZZER_2", "Zote"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Buzzer 1 B {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ZOTE", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_BUZZER_1_B", "Zote"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var History: int = 0
  var NPC State: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
