fsm collapse small {
  start Pause

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dust Collapse Small", storeResult=var "Dust")
    FindChild(gameObject=Self, childName="Strike Nail R", storeResult=var "Strike")
    FindChild(gameObject=Self, childName="Dust Break Floor", storeResult=var "Dust Break Floor")
    FindChild(gameObject=Self, childName="floor1", storeResult=var "Floor 1")
    FindChild(gameObject=Self, childName="floor2", storeResult=var "Floor 2")
    FindChild(gameObject=Self, childName="Pt Rocks", storeResult=var "Pt Rocks")
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on BREAK → Split
    on ACTIVATE → Activate
  }

  state Break {
    SetParent(gameObject=var "", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Pt Rocks", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Strike", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=breakable_wall_death (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(2.5, 0, 0), spawnMin=20, spawnMax=20, speedMin=2, speedMax=15, angleMin=0, angleMax=360, originVariationX=1.5, originVariationY=0, FSM="", FSMEvent="")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    PlayParticleEmitter(gameObject=var "Pt Rocks", emit=0)
    DestroyAllChildren(gameObject=var "Self", disable=false)
    ActivateGameObject(gameObject=var "Strike", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Split {
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    SetParent(gameObject=var "Dust", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    Rotate(gameObject=var "Floor 1", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-7, space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Rotate(gameObject=var "Floor 2", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=4, space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Floor 1", vector=(unset), x=(unset), y=-0.25, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Floor 2", vector=(unset), x=(unset), y=-0.2, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Activate {
    SetCollider(gameObject=Self, active=false)
    DestroyAllChildren(gameObject=var "Self", disable=false)
  }

  var Activated: bool = false
  var Dust: gameObject = <null>
  var Dust Break Floor: gameObject = <null>
  var Floor 1: gameObject = <null>
  var Floor 2: gameObject = <null>
  var Pt Rocks: gameObject = <null>
  var Self: gameObject = <null>
  var Strike: gameObject = <null>
}
