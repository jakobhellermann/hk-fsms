// component disabled: this FSM does not run until something enables it
fsm Atmos Control {
  start Init

  state Init {
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    on START → On
  }

  state Off {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=var "Fade Time")
    on START → On
  }

  state On {
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=var "Fade Time")
    on STOP → Off
  }

  var Fade Time: float = 0.3
}
