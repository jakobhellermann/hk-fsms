# Confirm Control
# 2 location(s):
#   Fungus3_35 / Bank Menu/Menu 1
#   sharedassets218.assets / Bank Menu/Menu 1

fsm Confirm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=Self, childName="Deposit", storeResult=var "Opt Deposit")
    FindChild(gameObject=Self, childName="Exit", storeResult=var "Opt Exit")
    FindChild(gameObject=Self, childName="Withdraw", storeResult=var "Opt Withdraw")
    FindChild(gameObject=var "Parent", childName="Deposit Menu", storeResult=var "Deposit Menu")
    FindChild(gameObject=var "Parent", childName="Withdraw Menu", storeResult=var "Withdraw Menu")
    GetLanguageString(sheetName="Prices", convName="BANK_TRANSACTION_AMOUNT", storeValue=var "Transaction String")
    ConvertStringToInt(stringVariable=var "Transaction String", intVariable=var "Transaction Amount", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Can Withdraw?
  }

  state Listen {
    SetFsmBool(gameObject=var "Parent", fsmName="shop_control", variableName="In Second Menu", setValue=false, everyFrame=false)
    on UI SELECTION MADE → Check Selection
  }

  state Check Selection {
    SendEventByName(eventTarget=Self, sendEvent="UI INACTIVE", delay=0, everyFrame=false)
    GetFsmInt(gameObject=Self, fsmName="ui_list", variableName="Current Item", storeValue=var "Current Item", everyFrame=false)
    IntSwitch(intVariable=var "Current Item", compareTo=[1, 2, 3], sendEvent=[→"DEPOSIT", →"WITHDRAW", →"EXIT"], everyFrame=false)
    on WITHDRAW → Start Withdraw
    on DEPOSIT → Start Deposit
    on EXIT → Close Pause
  }

  state Reset {
    ActivateGameObject(gameObject=var "Thankyou", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="RESET SHOP WINDOW", delay=0, everyFrame=false)
  }

  state Bob {
    GetParent(gameObject=var "Parent", storeResult=var "Shop Window")
    IntAdd(intVariable=var "Current Item", add=-1, everyFrame=false)
    SetFsmInt(gameObject=var "Shop Window", fsmName="shop_control", variableName="Current Item", setValue=var "Current Item", everyFrame=false)
    FindChild(gameObject=var "Shop Window", childName="Window", storeResult=var "Window")
    FindChild(gameObject=var "Window", childName="Figureheads", storeResult=var "Figurehead")
    SendEventByName(eventTarget=GameObject(var "Figurehead"), sendEvent="REPEAT UP ANIM", delay=0, everyFrame=false)
    iTweenMoveBy(gameObject=var "Shop Window", id="", vector=(0, 0.5, 0), time=0.13, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    iTweenMoveBy(gameObject=var "Shop Window", id="", vector=(0, -0.5, 0), time=0.2, delay=0.15, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Thankyou
  }

  state Thankyou {
    FindChild(gameObject=var "Shop Window", childName="Thankyou", storeResult=var "Thankyou")
    FindChild(gameObject=var "Shop Window", childName="Purchase Prt 1", storeResult=var "Prt 1")
    FindChild(gameObject=var "Shop Window", childName="Purchase Prt 2", storeResult=var "Prt 2")
    ActivateGameObject(gameObject=var "Thankyou", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Thankyou", vector=(unset), x=-3.41, y=-1.38, z=0, space=1, everyFrame=false, lateUpdate=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    iTweenMoveBy(gameObject=var "Thankyou", id="", vector=(0, 1, 0), time=1, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    on FINISHED → Particles
  }

  state Thank Fade {
    SendEventByName(eventTarget=GameObject(var "Figurehead"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Special Type?
  }

  state Deduct Geo and set PD {
    SetPlayerDataBool(boolName=var "PD Bool Name", value=true)
    CallMethodProper(gameObject=var "Prt 2", behaviour="HeroController", methodName="TakeGeo", parameters=[var "Cost"], storeResult=(unset var))
    on FINISHED → Bob
  }

  state Particles {
    PlayParticleEmitter(gameObject=var "Prt 1", emit=0)
    PlayParticleEmitter(gameObject=var "Prt 2", emit=0)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Thank Fade
  }

  state Special Type? {
    IntSwitch(intVariable=var "Special Type", compareTo=[0, 1, 2, 3], sendEvent=[→"FINISHED", →"HEART PIECE", →"CHARM", →"VESSEL FRAGMENT"], everyFrame=false)
    on FINISHED → Reset
    on HEART PIECE → Heart Piece
    on CHARM → Charm
    on VESSEL FRAGMENT → Vessel Fragment
  }

  state Heart Piece {
    CreateObject(gameObject=Heart Piece Instant (sharedassets10.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Close Shop Window
  }

  state Close Shop Window {
    ActivateGameObject(gameObject=var "Thankyou", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CLOSE SHOP WINDOW", delay=0, everyFrame=false)
  }

  state Charm {
    PlayerDataBoolTest(gameObject=var "Thankyou", boolName="hasCharm", isTrue=→"GOT", isFalse=(none))
    SetPlayerDataBool(boolName="hasCharm", value=true)
    on GOT → Reset
    on FINISHED → Charm Msg
  }

  state Charm Msg {
    CreateObject(gameObject=Charm Tutorial Msg (sharedassets6.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Charm Msg", networkInstantiate=false, networkGroup=0)
    SetFsmBool(gameObject=var "Charm Msg", fsmName="Charm Tute Msg", variableName="Input", setValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Thankyou", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CLOSE SHOP WINDOW", delay=0, everyFrame=false)
    on FINISHED → 
  }

  state Vessel Fragment {
    CreateObject(gameObject=Vessel Fragment Instant (sharedassets10.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Close Shop Window
  }

  state Close {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="CLOSE", delay=0, everyFrame=false)
  }

  state Close Pause {
    Wait(time=0.3, finishEvent=→"EXIT", realTime=false)
    on EXIT → Close
  }

  state Can Withdraw? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerBalance", storeValue=var "Balance")
    SetFsmBool(gameObject=var "Opt Withdraw", fsmName="ui_list_item", variableName="Unselectable", setValue=false, everyFrame=false)
    IntCompare(integer1=var "Balance", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetTextMeshProColor(gameObject=var "Opt Withdraw", color=(0.49264705, 0.49264705, 0.49264705, 1), everyFrame=false)
    SetFsmBool(gameObject=var "Opt Withdraw", fsmName="ui_list_item", variableName="Unselectable", setValue=true, everyFrame=false)
    on FINISHED → Can Deposit?
  }

  state Start Withdraw {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "Parent", fsmName="shop_control", variableName="In Second Menu", setValue=true, everyFrame=false)
    on FINISHED → Withdraw
  }

  state Can Deposit? {
    SetFsmBool(gameObject=var "Opt Deposit", fsmName="ui_list_item", variableName="Unselectable", setValue=false, everyFrame=false)
    GetLanguageString(sheetName="Prices", convName="BANK_LIMIT", storeValue=var "Deposit Limit Str")
    ConvertStringToInt(stringVariable=var "Deposit Limit Str", intVariable=var "Deposit Limit", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerBalance", storeValue=var "Balance")
    IntCompare(integer1=var "Balance", integer2=var "Deposit Limit", equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetTextMeshProColor(gameObject=var "Opt Deposit", color=(0.49264705, 0.49264705, 0.49264705, 1), everyFrame=false)
    SetFsmBool(gameObject=var "Opt Deposit", fsmName="ui_list_item", variableName="Unselectable", setValue=true, everyFrame=false)
    on FINISHED → Listen
  }

  state Start Deposit {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "Parent", fsmName="shop_control", variableName="In Second Menu", setValue=true, everyFrame=false)
    on FINISHED → Deposit
  }

  state Deposit {
    ActivateGameObject(gameObject=var "Deposit Menu", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Withdraw {
    ActivateGameObject(gameObject=var "Withdraw Menu", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Balance: int = 0
  var Cost: int = 0
  var Current Item: int = 0
  var Deposit Limit: int = 0
  var Player Geo: int = 0
  var Special Type: int = 0
  var Transaction Amount: int = 0
  var Get Input: bool = false
  var Deposit Limit Str: string = ""
  var PD Bool Name: string = ""
  var Transaction String: string = ""
  var Charm Msg: gameObject = <null>
  var Deposit Menu: gameObject = <null>
  var Figurehead: gameObject = <null>
  var Opt Deposit: gameObject = <null>
  var Opt Exit: gameObject = <null>
  var Opt Withdraw: gameObject = <null>
  var Parent: gameObject = <null>
  var Prt 1: gameObject = <null>
  var Prt 2: gameObject = <null>
  var Shop Window: gameObject = <null>
  var Thankyou: gameObject = <null>
  var Window: gameObject = <null>
  var Withdraw Menu: gameObject = <null>
}
