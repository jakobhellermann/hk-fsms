  start State 1

  state State 1 {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Knight Unform")
    on FINISHED → State 2
  }

  state State 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Knight Unform")
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -20, 0), time=1, delay=0, speed=10, easeType=easeInSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
  }
}
