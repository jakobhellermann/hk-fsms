# Spawn
# 1 location(s):
#   Room_Colosseum_Silver / Colosseum Manager/Waves/Wave 29 Obble/Colosseum Cage Small (6)

fsm Spawn {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Colosseum Manager", store=var "Colosseum Manager")
    SetSpriteRenderer(gameObject=Self, active=false)
    CreateObject(gameObject=var "Enemy Type", spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Enemy", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=var "Enemy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBattleScene(target=var "Enemy", battleScene=var "Colosseum Manager")
    SetGeoDrop(target=var "Enemy", smallGeo=0, mediumGeo=0, largeGeo=0)
    on SPAWN → Spawn Pause
    on FINISHED → Summon Idle
  }

  state Anim {
    ActivateGameObject(gameObject=var "Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    WaitRandom(timeMin=1.25, timeMax=1.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="AUDIO CAGE UP", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="AUDIO CAGE UP")
    on FINISHED → Spawn
  }

  state Spawn {
    ActivateGameObject(gameObject=var "Enemy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Strike", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    SendEventToRegister(eventName="AUDIO CAGE OPEN")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="AUDIO CAGE OPEN", delay=0, everyFrame=false)  // disabled
    BoolTest(boolVariable=var "Stunner", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="KILLED GameObject", setValue=var "Self", everyFrame=false)  // disabled
    on FINISHED → Pause
  }

  state End {
    ActivateGameObject(gameObject=var "Anim", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Spawn Pause {
    Wait(time=var "Spawn Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Anim
  }

  state End 2 {
    ActivateGameObject(gameObject=var "Anim", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on KILLED → Stun
  }

  state Pause {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End 2
    on KILLED → Stun
  }

  state Stun {
    CreateObject(gameObject=Stun Effect (resources.assets), spawnPoint=var "Enemy", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
  }

  state Create Corpse {
    GetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="Corpse", storeValue=var "z Corpse to Instantiate", everyFrame=false)
    SetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="Corpse", setValue=<null>, everyFrame=false)
    CreateObject(gameObject=var "z Corpse to Instantiate", spawnPoint=var "Enemy", position=(unset), rotation=(unset), storeObject=var "z Corpse Instance", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=var "z Corpse Instance", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "z Corpse Instance", parent=var "Enemy", resetLocalPosition=false, resetLocalRotation=false)
    SetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="Corpse Instance", setValue=var "z Corpse Instance", everyFrame=false)
    on FINISHED → Summon Idle
  }

  state Summon Idle {
    on SPAWN → Spawn Pause
  }

  var Spawn Pause: float = 0
  var Stunner: bool = false
  var Enemy Type: gameObject = Blobble (sharedassets33.assets)

  var Anim: gameObject = Colosseum Manager/Waves/Wave 29 Obble/Colosseum Cage Small (6)/Anim
  var Colosseum Manager: gameObject = <null>
  var Enemy: gameObject = Colosseum_Armoured_Roller (sharedassets32.assets)
  var Self: gameObject = <null>
  var Strike: gameObject = Colosseum Manager/Waves/Wave 29 Obble/Colosseum Cage Small (6)/Strike Nail
  var z Corpse Instance: gameObject = <null>
  var z Corpse to Instantiate: gameObject = <null>
}
