fsm Black Charm {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    IntCompare(integer1=var "Royal Charm State", integer2=4, equal=→"INACTIVE", lessThan=(none), greaterThan=(none), everyFrame=false)
    on INACTIVE → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Royal Charm State: int = 0
}
