# Shift Based On Hero Pos
# 1 location(s):
#   Ruins1_05c / Ruins Sentry Fat

fsm Shift Based On Hero Pos {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check
  }

  state Check {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO RIGHT", leftEvent=→"HERO LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on HERO LEFT → Hero Left
    on HERO RIGHT → Hero Right
  }

  state Hero Left {
    SetPosition(gameObject=Self, vector=(unset), x=24.85, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
  }

  state Hero Right {
    SetPosition(gameObject=Self, vector=(unset), x=5.21, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }
}
