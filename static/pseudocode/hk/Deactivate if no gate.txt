fsm Deactivate if no gate {
  start Check

  state Check {
    GetPlayerDataString(gameObject=var "GameManager", stringName="dreamGateScene", storeValue=var "Scene")
    StringCompare(stringVariable=var "Scene", compareTo="", equalEvent=→"DEACTIVATE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Scene", compareTo=" ", equalEvent=→"DEACTIVATE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on DEACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Scene: string = ""
}
