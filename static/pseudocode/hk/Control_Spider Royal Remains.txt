# Control
# 1 location(s):
#   Deepnest_Spider_Town / Spider Royal Remains

fsm Control {
  start Pause

  state Pause {
    on FINISHED → Check
  }

  state Check {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="spiderCapture", isTrue=(none), isFalse=→"HIDE")
    on HIDE → Hide
  }

  state Hide {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on CAPTURE END → Show
  }

  state Show {
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  var Self: gameObject = <null>
}
