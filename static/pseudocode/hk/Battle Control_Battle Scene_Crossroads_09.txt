# Battle Control
# 1 location(s):
#   Crossroads_09 / Battle Scene

fsm Battle Control {
  start PrePause

  state Detect {
    on START → Start
  }

  state Init {
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "Boss Camera Lock")
    FindChild(gameObject=Self, childName="Music Region B", storeResult=var "Tension Region")
    FindChild(gameObject=Self, childName="CameraLockArea NB", storeResult=var "CamLock NB")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Boss Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Detect
    on ACTIVATED → Activate
  }

  state Start {
    ActivateGameObject(gameObject=var "Boss Camera Lock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DestroyObject(gameObject=var "Tension Region", delay=0, detachChildren=false)
    ActivateGameObject(gameObject=var "CamLock NB", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIntValue(intVariable=var "Battle Enemies", intValue=1, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on BATTLE END → Blow Wait
  }

  state End Wait {
    FindGameObject(objectName="", withTag="Boss Corpse", store=var "Boss Corpse")
    ActivateGameObject(gameObject=var "Heart Piece", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmGameObject(gameObject=var "Heart Piece", fsmName="", variableName="Appear Source", setValue=var "Boss Corpse", everyFrame=false)
    SetFsmBool(gameObject=var "Heart Piece", fsmName="", variableName="Appears", setValue=true, everyFrame=false)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=var "Boss Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock NB", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    DestroyObject(gameObject=var "Tension Region", delay=0, detachChildren=false)
    SetFsmBool(gameObject=var "Heart Piece", fsmName="Heart Container Control", variableName="Appears", setValue=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Heart Piece", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DestroyObject(gameObject=var "Boss Camera Lock", delay=0, detachChildren=false)
  }

  state PrePause {
    FindGameObject(objectName="", withTag="Heart Piece", store=var "Heart Piece")
    ActivateGameObject(gameObject=var "Heart Piece", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Pause
  }

  state Blow Wait {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Wait(time=5.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Wait
  }

  var Delay: float = 0
  var Track Event: string = ""

  var Fade Time: float = 0
  var Battle Enemies: int = 0
  var Activated: bool = false
  var Boss: gameObject = <null>
  var Boss Camera Lock: gameObject = <null>
  var Boss Corpse: gameObject = <null>
  var CamLock B: gameObject = <null>
  var CamLock NB: gameObject = <null>
  var Heart Piece: gameObject = <null>
  var Mawlek: gameObject = <null>
  var Music Folder: gameObject = <null>
  var Tension Region: gameObject = <null>
  var Track 1: gameObject = <null>
}
