# Quake
# 1 location(s):
#   Ruins1_24 / Quake Real Parent/Quake Pickup/Quake Item

fsm Quake {
  start Pause

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Idle Particles", childName="", storeResult=var "")
    SetGameObject(variable=var "Hero Object", gameObject=var "Hero", everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"GOT", storeCollider=var "")
    on GOT → Check Superdashing
  }

  state Get {
    AudioStop(gameObject=Self)
    SpawnObjectFromGlobalPool(gameObject=White Flash R (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    SetPlayerDataBool(boolName="disablePause", value=true)
    SetPlayerDataBool(boolName="corn_cityLeft", value=true)
    SetPlayerDataInt(intName="quakeLevel", value=1)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="brettaState")
    SendMessage(gameObject=var "Hero Object", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero Object", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SendMessage(gameObject=var "Hero Object", delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    DestroyObject(gameObject=var "Idle Particles", delay=0, detachChildren=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    GetGrandparent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Knight Get Quake", storeResult=var "Fireball Getter")
    SetPositionToObject(gameObject=var "Fireball Getter", targetObject=var "Hero Object", xOffset=0, yOffset=-0.5, zOffset=0)
    ActivateGameObject(gameObject=var "Fireball Getter", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetScale(gameObject=var "Hero Object", vector=(unset), xScale=var "Hero X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Fireball Getter", vector=(unset), x=var "Hero X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    DestroyAllChildren(gameObject=var "Self", disable=false)
    SetMeshRenderer(gameObject=var "Hero Object", active=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetHero(storeResult=var "Hero")
    on FINISHED → Idle
  }

  state Set Respawns {
    CallMethodProper(gameObject=var "Hero Object", behaviour="HeroController", methodName="SetBenchRespawn", parameters=["Death Respawn Marker", "Ruins1_24", 0, true], storeResult=(unset var))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="END", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero Object", behaviour="HeroController", methodName="SetHazardRespawn", parameters=[(30.16, 10.66, 0, 0), true], storeResult=(unset var))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ACTIVATE BATTLE ROOF", delay=0, everyFrame=false)
    on FINISHED → State 2
  }

  state Check Superdashing {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetCState", parameters=["superDashing"], storeResult=var "Is Superdashing")
    BoolTest(boolVariable=var "Is Superdashing", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Idle
    on FALSE → Set Respawns
  }

  state State 2 {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Get
  }

  var Hero X Scale: float = 0
  var Is Superdashing: bool = false
  var Fireball Getter: gameObject = <null>
  var Hero: gameObject = <null>
  var Hero Object: gameObject = <null>
  var Idle Particles: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
