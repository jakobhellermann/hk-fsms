# Rotate By Speed
# 1 location(s):
#   sharedassets167.assets / Spore Bomb

fsm Rotate By Speed {
  start Rotate

  state Rotate {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=true)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-50, everyFrame=true)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
  }

  var Speed: float = 0
}
