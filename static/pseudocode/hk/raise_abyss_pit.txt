fsm raise_abyss_pit {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetParent(gameObject=var "Parent", storeResult=var "Boss Control")
    FindChild(gameObject=var "Boss Control", childName="Abyss Pit", storeResult=var "Abyss Pit")
    FindChild(gameObject=Self, childName="Hazard Respawn Marker", storeResult=var "Hazard Respawn Marker")
    GetPosition(gameObject=var "Hazard Respawn Marker", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Set
  }

  state Set {
    SetFsmFloat(gameObject=var "Abyss Pit", fsmName="Ascend", variableName="Hero Y", setValue=var "Hero Y", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Abyss Pit"), sendEvent="ASCEND", delay=0, everyFrame=false)
  }

  var Hero Y: float = 0
  var Abyss Pit: gameObject = <null>
  var Boss Control: gameObject = <null>
  var Hazard Respawn Marker: gameObject = <null>
  var Parent: gameObject = <null>
}
