    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on ATTACK → Shield Start
    on FINISHED → Reset
  }

  state Evade {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk R")
    RandomFloat(min=4, max=6, storeResult=var "Evade Speed")
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Evade Speed", multiplyBy=var "X Scale", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Evade Speed", y=(unset), everyFrame=false)
    WaitRandom(timeMin=0.4, timeMax=0.8, finishEvent=→"FINISHED", realTime=false)
    on TOOK DAMAGE → Lunge Speed Set 2
    on FINISHED → Lunge Speed Set
  }

  state Evade End {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SendRandomEvent(events=[→"ATTACK1", →"ATTACK2", (none)], weights=[0.25, 0.25, 0.5], delay=0)
    CheckCanSeeHero(storeResult=var "Can See Hero", everyFrame=true)
    CheckAlertRangeByName(storeResult=var "In Attack Range", childName="Attack Range", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Attack Range"], sendEvent=→"ATTACK", storeResult=(unset), everyFrame=true)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on ATTACK → Shield Start
    on FINISHED → Recover Idle
    on ATTACK1 → A1 Switch?
    on ATTACK2 → A3 Switch?
  }

  state Lunge Speed Set {
    SetFloatValue(floatVariable=var "Lunge3 Speed", floatValue=-16, everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge1 Speed", floatValue=-18, everyFrame=false)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge3 Speed", floatValue=16, everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge1 Speed", floatValue=18, everyFrame=false)
    on FINISHED → Evade End
  }

  state Lunge Speed Set 2 {
    SetFloatValue(floatVariable=var "Lunge3 Speed", floatValue=-16, everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge1 Speed", floatValue=-18, everyFrame=false)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge3 Speed", floatValue=16, everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge1 Speed", floatValue=18, everyFrame=false)
    on FINISHED → A3 Switch?
  }
