# Cut Down
# 1 location(s):
#   Deepnest_33 / Zote Deepnest

fsm Cut Down {
  start Pause

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"CUT", storeCollider=var "")
    on CUT → Hero Range
  }

  state Cut {
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Struggler", audioClip=grass_cut_1 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataBool(boolName="zoteRescuedDeepnest", value=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Faller", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Struggler", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Particle 1", emit=0)
    PlayParticleEmitter(gameObject=var "Particle 2", emit=0)
    ActivateGameObject(gameObject=var "Slash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Zote Trapped?
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Zote Trapped? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="zoteRescuedDeepnest", isTrue=→"DESTROY", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="zoteDead", isTrue=→"DESTROY", isFalse=(none))
    SetPlayerDataBool(boolName="zoteTrappedDeepnest", value=true)
    on DESTROY → Destroy
    on FINISHED → Init
  }

  state Hero Range {
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Cut
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Zote Death: int = 0
  var Hero Range: bool = false
  var Faller: gameObject = Zote Deepnest/Faller
  var Particle 1: gameObject = Zote Deepnest/Particle 1
  var Particle 2: gameObject = Zote Deepnest/Particle 2
  var Slash: gameObject = Zote Deepnest/Slash Impact
  var Struggler: gameObject = Zote Deepnest/Struggler
}
