fsm Display {
  start Init

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="monomonDefeated", isTrue=→"DISABLE", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasPinGuardian", isTrue=→"ENABLE", isFalse=→"DISABLE")
    on ENABLE → Enable
    on DISABLE → Disable
  }

  state Enable {
    SetSpriteRenderer(gameObject=Self, active=true)
  }

  state Disable {
    SetSpriteRenderer(gameObject=Self, active=false)
  }
}
