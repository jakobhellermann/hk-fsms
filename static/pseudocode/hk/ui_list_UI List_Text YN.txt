# ui_list

// uses template: ui_list
fsm ui_list {
  start Init
  on SELECTION MADE → Selection Made  // from any state
  on UI INACTIVE → Inactive  // from any state
  on SELECTION MADE CANCEL → Selection Made Cancel  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Inactive", isTrue=(none), isFalse=→"UI ACTIVE", everyFrame=false)
    on UI ACTIVE → Activate
  }

  state Idle {
    SetBoolValue(boolVariable=var "Repeating", boolValue=false, everyFrame=false)
    ListenForRight(eventTarget=Self, wasPressed=→"RIGHT", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForLeft(eventTarget=Self, wasPressed=→"LEFT", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForDown(eventTarget=Self, wasPressed=→"DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForUp(eventTarget=Self, wasPressed=→"UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    on UP → Select Up
    on DOWN → Select Down
    on LEFT → Select Left
    on RIGHT → Select Right
  }

  state Select Up {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Self", audioClip=ui_change_selection (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=0.65, delay=0, storePlayer=<null>)
    BoolTest(boolVariable=var "Allow Vertical Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetIntValue(intVariable=var "Direction Pressed", intValue=2, everyFrame=false)
    IntOperator(integer1=var "Current Item", integer2=1, operation=Subtract, storeResult=var "Current Item", everyFrame=false)
    on CANCEL → CancelToIdle
    on FINISHED → Wrap Back?
  }

  state Wrap Back? {
    IntCompare(integer1=var "Current Item", integer2=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    SetIntValue(intVariable=var "Current Item", intValue=var "Items", everyFrame=false)
    on CANCEL → Update
    on FINISHED → Update
  }

  state Select Left {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Self", audioClip=ui_change_selection (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=0.65, delay=0, storePlayer=<null>)
    BoolTest(boolVariable=var "Allow Horizontal Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetIntValue(intVariable=var "Direction Pressed", intValue=3, everyFrame=false)
    IntOperator(integer1=var "Current Item", integer2=1, operation=Subtract, storeResult=var "Current Item", everyFrame=false)
    on CANCEL → CancelToIdle
    on FINISHED → Wrap Back?
  }

  state Select Down {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Self", audioClip=ui_change_selection (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=0.65, delay=0, storePlayer=<null>)
    BoolTest(boolVariable=var "Allow Vertical Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetIntValue(intVariable=var "Direction Pressed", intValue=4, everyFrame=false)
    IntOperator(integer1=var "Current Item", integer2=1, operation=Add, storeResult=var "Current Item", everyFrame=false)
    on CANCEL → CancelToIdle
    on FINISHED → Wrap Forward?
  }

  state Wrap Forward? {
    IntCompare(integer1=var "Current Item", integer2=var "Items", equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    SetIntValue(intVariable=var "Current Item", intValue=1, everyFrame=false)
    on CANCEL → Update
    on FINISHED → Update
  }

  state Select Right {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Self", audioClip=ui_change_selection (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=0.65, delay=0, storePlayer=<null>)
    SetIntValue(intVariable=var "Direction Pressed", intValue=1, everyFrame=false)
    BoolTest(boolVariable=var "Allow Horizontal Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    IntOperator(integer1=var "Current Item", integer2=1, operation=Add, storeResult=var "Current Item", everyFrame=false)
    on CANCEL → CancelToIdle
    on FINISHED → Wrap Forward?
  }

  state Update {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SELECT UPDATE", delay=0, everyFrame=false)
    IntSwitch(intVariable=var "Direction Pressed", compareTo=[1, 2, 3, 4], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on UP → Still Up?
    on DOWN → Still Down?
    on RIGHT → Still Right?
    on LEFT → Still Left?
  }

  sequence state Selection Made {
    NextFrameEvent(sendEvent=(none))
    BoolTest(boolVariable=var "Selected Unselectable", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Self", audioClip=ui_button_confirm (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=0.75, delay=0, storePlayer=<null>)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Notify
    on CANCEL → Will Cancel
  }

  state Notify {
    BoolTest(boolVariable=var "Selected Unselectable", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="UI SELECTION MADE", delay=0, everyFrame=false)
    on CANCEL → Idle
  }

  state Activate {
    SetIntValue(intVariable=var "Current Item", intValue=var "Initial Item", everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SELECT UPDATE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Still Up? {
    ListenForUp(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, stateEntryOnly=false)
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Repeat Up
  }

  state Repeat Up {
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    ListenForUp(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, stateEntryOnly=false)
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Select Up
  }

  state Still Down? {
    ListenForDown(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, stateEntryOnly=false)
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Repeat Down
  }

  state Repeat Down {
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    ListenForDown(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, stateEntryOnly=false)
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Select Down
  }

  state Still Right? {
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL")
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Repeat Right
  }

  state Repeat Right {
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL")
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Select Right
  }

  state Still Left? {
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL")
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Repeat Left
  }

  state Repeat Left {
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL")
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=false)
    on CANCEL → Idle
    on REPEAT → Select Left
  }

  state Inactive {
    on UI ACTIVE → Activate
  }

  state Selection Made Cancel {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Notify
  }

  state CancelToIdle {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Will Cancel {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Selection Pause: float = 0.5
  var Cancel Item: int = 2
  var Current Item: int = 0
  var Initial Item: int = 1
  var Items: int = 2
  var Allow Horizontal Selection: bool = true
  var Allow Vertical Selection: bool = false
  var Start Inactive: bool = false
  var Current Selection: string = ""

  var Axis Horizontal: float = 0
  var Axis Vertical: float = 0
  var Direction Pressed: int = 0
  var Repeating: bool = false
  var Selected Unselectable: bool = false
  var Self: gameObject = <null>
}
