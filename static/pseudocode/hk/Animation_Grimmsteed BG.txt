# Animation
# 1 location(s):
#   Town / Grimmsteed BG

fsm Animation {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → LookLeft
  }

  state LookingLeft {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO R", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on HERO R → LookRight
  }

  state LookLeft {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG LookLeft")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → LookingLeft
  }

  state LookingRight {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"HERO L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on HERO L → LookLeft
  }

  state LookRight {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG LookRight")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → LookingRight
  }
}
