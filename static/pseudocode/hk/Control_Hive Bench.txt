# Control
# 1 location(s):
#   Hive_01 / Hive Bench

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Spr Idle", storeResult=var "Spr Idle")
    FindChild(gameObject=Self, childName="Spr Cracked", storeResult=var "Spr Cracked")
    FindChild(gameObject=Self, childName="Spr Base", storeResult=var "Spr Base")
    FindChild(gameObject=Self, childName="Terrain Broken", storeResult=var "Terrain Broken")
    FindChild(gameObject=Self, childName="Terrain Unbroken", storeResult=var "Terrain Unbroken")
    FindChild(gameObject=Self, childName="Effect Origin", storeResult=var "Effect Origin")
    FindChild(gameObject=Self, childName="Chunks", storeResult=var "Chunks")
    FindChild(gameObject=Self, childName="RestBench", storeResult=var "RestBench")
    FindChild(gameObject=Self, childName="Bench Fall", storeResult=var "Bench Fall")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "RestBench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activated
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Hit
  }

  state Hit {
    AudioPlayRandom(gameObject=var "Self", audioClips=[honey_pillar_hit_1 (sharedassets384.assets), honey_pillar_hit_2 (sharedassets384.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Effect Origin", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Effect Origin", position=(2, 0, 0), spawnMin=8, spawnMax=10, speedMin=5, speedMax=15, angleMin=0, angleMax=30, originVariationX=0, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Effect Origin", position=(-2, 0, 0), spawnMin=8, spawnMax=5, speedMin=10, speedMax=15, angleMin=150, angleMax=180, originVariationX=0, originVariationY=1, FSM="", FSMEvent="")
    IntCompare(integer1=var "Hits", integer2=3, equal=→"BREAK 1", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=6, equal=→"BREAK 2", lessThan=(none), greaterThan=→"BREAK 2", everyFrame=false)
    on BREAK 1 → Break 1
    on FINISHED → Idle
    on BREAK 2 → Break 2
  }

  state Break 1 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=honey_orb_break (sharedassets384.assets))
    ActivateGameObject(gameObject=var "Spr Cracked", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spr Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Break 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=honey_pillar_final_break (sharedassets384.assets))
    ActivateGameObject(gameObject=var "Spr Base", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Unbroken", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Chunks", activate=true)
    FlingObjects(containerObject=var "Chunks", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=5, speedMax=15, angleMin=0, angleMax=360)
    ActivateGameObject(gameObject=var "Spr Cracked", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Effect Origin", position=(0, 0, 0), spawnMin=100, spawnMax=100, speedMin=5, speedMax=15, angleMin=0, angleMax=360, originVariationX=2, originVariationY=2, FSM="", FSMEvent="")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Bench Fall 1
  }

  state Activated {
    ActivateGameObject(gameObject=var "Spr Base", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spr Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Unbroken", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
  }

  state Bench Fall 1 {
    ActivateGameObject(gameObject=var "Bench Fall", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    iTweenMoveBy(gameObject=var "Bench Fall", id="", vector=(0, -7, 0), time=(unset), delay=0, speed=15, easeType=easeInSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Bench Active
  }

  state Bench Fall 2 {
    iTweenMoveBy(gameObject=var "Bench Fall", id="", vector=(0, 1, 0), time=(unset), delay=0, speed=15, easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Bench Fall 3
  }

  state Bench Fall 3 {
    iTweenMoveBy(gameObject=var "Bench Fall", id="", vector=(0, -1, 0), time=(unset), delay=0, speed=15, easeType=easeInCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Bench Active
  }

  state Bench Active {
    ActivateGameObject(gameObject=var "RestBench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bench Fall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Bench Fall", audioClip=hero_land_soft (resources.assets), pitchMin=0.8, pitchMax=0.8, volume=1, delay=0, storePlayer=<null>)
  }

  var Hits: int = 0
  var Activated: bool = false
  var Bench Fall: gameObject = <null>
  var Chunks: gameObject = <null>
  var Effect Origin: gameObject = <null>
  var RestBench: gameObject = <null>
  var Self: gameObject = <null>
  var Spr Base: gameObject = <null>
  var Spr Cracked: gameObject = <null>
  var Spr Idle: gameObject = <null>
  var Terrain Broken: gameObject = <null>
  var Terrain Unbroken: gameObject = <null>
}
