# Add
# 1 location(s):
#   Abyss_05 / Add White Palace Journal Entries

fsm Add {
  start Pause

  state Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Completed WP?
  }

  state Completed WP? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotKingFragment", isTrue=→"CHECK", isFalse=(none))
    on CHECK → Palace Fly?
  }

  state Palace Fly? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedPalaceFly", isTrue=(none), isFalse=→"ADD")
    GetPlayerDataInt(gameObject=var "GameManager", intName="killsPalaceFly", storeValue=var "Kills Remaining")
    IntCompare(integer1=var "Kills Remaining", integer2=0, equal=(none), lessThan=(none), greaterThan=→"ADD", everyFrame=false)
    on ADD → Add Palace Fly
    on FINISHED → Royal Guard?
  }

  state Add Palace Fly {
    SetPlayerDataBool(boolName="killedPalaceFly", value=true)
    SetPlayerDataBool(boolName="newDataPalaceFly", value=true)
    SetPlayerDataInt(intName="killsPalaceFly", value=0)
    SetBoolValue(boolVariable=var "Updated Journal", boolValue=true, everyFrame=false)
    on FINISHED → Royal Guard?
  }

  state Royal Guard? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedRoyalGuard", isTrue=(none), isFalse=→"ADD")
    GetPlayerDataInt(gameObject=var "GameManager", intName="killsRoyalGuard", storeValue=var "Kills Remaining")
    IntCompare(integer1=var "Kills Remaining", integer2=0, equal=(none), lessThan=(none), greaterThan=→"ADD", everyFrame=false)
    on ADD → Add RoyalGuard
    on FINISHED → White Royal?
  }

  state Add RoyalGuard {
    SetPlayerDataBool(boolName="killedRoyalGuard", value=true)
    SetPlayerDataBool(boolName="newDataRoyalGuard", value=true)
    SetPlayerDataInt(intName="killsRoyalGuard", value=0)
    SetBoolValue(boolVariable=var "Updated Journal", boolValue=true, everyFrame=false)
    on FINISHED → White Royal?
  }

  state White Royal? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedWhiteRoyal", isTrue=(none), isFalse=→"ADD")
    GetPlayerDataInt(gameObject=var "GameManager", intName="killsWhiteRoyal", storeValue=var "Kills Remaining")
    IntCompare(integer1=var "Kills Remaining", integer2=0, equal=(none), lessThan=(none), greaterThan=→"ADD", everyFrame=false)
    on ADD → Add WhiteRoyal
    on FINISHED → Updated?
  }

  state Add WhiteRoyal {
    SetPlayerDataBool(boolName="killedWhiteRoyal", value=true)
    SetPlayerDataBool(boolName="newDataWhiteRoyal", value=true)
    SetPlayerDataInt(intName="killsWhiteRoyal", value=0)
    SetBoolValue(boolVariable=var "Updated Journal", boolValue=true, everyFrame=false)
    on FINISHED → Updated?
  }

  state Updated? {
    BoolTest(boolVariable=var "Updated Journal", isTrue=→"MSG", isFalse=(none), everyFrame=false)
    on MSG → Msg
  }

  state Msg {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasJournal", isTrue=(none), isFalse=→"CANCEL")
    CreateObject(gameObject=Journal Update Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on CANCEL → Cancel
  }

  state Cancel {
  }

  var Kills Remaining: int = 0
  var Updated Journal: bool = false
}
