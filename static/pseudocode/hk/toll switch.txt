# toll switch
# 1 location(s):
#   Crossroads_03 / _Props/Toll Gate Switch

fsm toll switch {
  start Pause

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    FindGameObject(objectName="Toll Gate", withTag="Untagged", store=var "G1")
    FindGameObject(objectName="Toll Gate 1", withTag="Untagged", store=var "G2")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATED → Activate
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Hit
  }

  state Hit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_switch (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Switch Hit")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=GameObject(var "G1"), sendEvent="OPEN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "G2"), sendEvent="OPEN", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state Activate {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Switch Hit")
    Tk2dPlayFrame(gameObject=Self, frame=10)
    DestroyObject(gameObject=var "G1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "G2", delay=0, detachChildren=false)
  }

  var Attack Direction: float = 0
  var Recoil Speed: float = 1
  var Recoil Speed Neg: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Attack Type: int = 0
  var Activated: bool = false
  var Camera: gameObject = <null>
  var Damager: gameObject = <null>
  var G1: gameObject = <null>
  var G2: gameObject = <null>
  var Self: gameObject = <null>
}
