# Activate
# 2 location(s):
#   Room_Colosseum_Bronze / Exit Blocker
#   Room_Colosseum_Gold / Exit Blocker:1

fsm Activate {
  start Pause

  sequence state Pause {
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=true)
    WaitForFinishedEnteringScene(sendEvent=(none))
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
    on GATE OPEN → Deactivate
  }

  state Deactivate {
    SetCollider(gameObject=Self, active=false)
  }
}
