# Control
# 1 location(s):
#   Fungus1_35 / Cam Lock Control

fsm Control {
  start Init

  state Inactive {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on DREAM GATE CLOSE → Active
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Inactive
  }

  state Active {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on DREAM GATE OPEN → Inactive
  }

  var Self: gameObject = <null>
}
