# Deactivate in darkness without lantern
# 1 location(s):
#   Fungus1_35 / Area Title Controller

fsm Deactivate in darkness without lantern {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Lantern?
  }

  state Lantern? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasLantern", isTrue=(none), isFalse=→"NO LANTERN")
    on NO LANTERN → Dark?
  }

  state Dark? {
    FindGameObject(objectName="", withTag="SceneManager", store=var "SceneManager")
    CallMethodProper(gameObject=var "SceneManager", behaviour="SceneManager", methodName="GetDarknessLevel", parameters=[], storeResult=var "Darkness Level")
    IntCompare(integer1=var "Darkness Level", integer2=2, equal=→"DARK", lessThan=(none), greaterThan=→"DARK", everyFrame=false)
    on DARK → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Darkness Level: int = 0
  var SceneManager: gameObject = <null>
}
