# Spin
# 1 location(s):
#   sharedassets366.assets / Shot Kings Guard

fsm Spin {
  start Init

  state Init {
    on FINISHED → Launch
  }

  state Caught {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="CAUGHT", delay=0, everyFrame=false)
    RecycleSelf()
  }

  state Launch {
    FireAtTarget(gameObject=Self, target=var "Hero", speed=35, position=(0, -0.5, 0), spread=0, everyFrame=false)  // disabled
    SetVelocityAsAngle(gameObject=Self, angle=var "Fire Angle", speed=35, everyFrame=false)
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    FloatInRange(floatVariable=var "Angle", lowerValue=185, upperValue=270, boolVariable=(unset), trueEvent=→"CLAMP BL", falseEvent=(none), everyFrame=false)
    FloatInRange(floatVariable=var "Angle", lowerValue=270, upperValue=355, boolVariable=(unset), trueEvent=→"CLAMP BR", falseEvent=(none), everyFrame=false)
    FloatInRange(floatVariable=var "Angle", lowerValue=90, upperValue=135, boolVariable=(unset), trueEvent=→"CLAMP TL", falseEvent=(none), everyFrame=false)
    FloatInRange(floatVariable=var "Angle", lowerValue=45, upperValue=90, boolVariable=(unset), trueEvent=→"CLAMP TR", falseEvent=(none), everyFrame=false)
    on CLAMP BL → Clamp BL
    on CLAMP TL → Clamp TL
    on CLAMP TR → Clamp TR
    on CLAMP BR → Clamp BR
    on FINISHED → Get Vel
  }

  state In Air {
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=var "Spd Vector", x=(unset), y=(unset), vector3=(unset), everyFrame=true)
    GetDistance(gameObject=Self, target=var "Parent", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=2, tolerance=0, equal=→"RETURN", lessThan=→"RETURN", greaterThan=(none), everyFrame=true)
    Wait(time=2, finishEvent=→"RETURN", realTime=false)
    on RETURN → Caught
  }

  state Clamp BL {
    SetVelocityAsAngle(gameObject=Self, angle=185, speed=35, everyFrame=false)
    on FINISHED → Get Vel
  }

  state Get Vel {
    GetVelocity2d(gameObject=Self, vector=var "Spd Vector", x=(unset), y=(unset), space=0, everyFrame=false)
    Vector2Multiply(vector2Variable=var "Spd Vector", multiplyBy=-7, everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → In Air
  }

  state Clamp TL {
    SetVelocityAsAngle(gameObject=Self, angle=135, speed=35, everyFrame=false)
    on FINISHED → Get Vel
  }

  state Clamp TR {
    SetVelocityAsAngle(gameObject=Self, angle=45, speed=35, everyFrame=false)
    on FINISHED → Get Vel
  }

  state Clamp BR {
    SetVelocityAsAngle(gameObject=Self, angle=355, speed=35, everyFrame=false)
    on FINISHED → Get Vel
  }

  var Angle: float = 0
  var Distance: float = 0
  var Fire Angle: float = 0
  var Spd Vector: vector2 = (0, 0)
  var Parent: gameObject = <null>
}
