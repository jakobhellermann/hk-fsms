# Gate
# 1 location(s):
#   Fungus1_22 / Metal Gate

fsm Gate {
  start Pause
  on ACTIVATE AGAIN → Activate  // from any state

  state Idle {
    on ACTIVATE → Open Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Activate {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Gate Opened")
  }

  state Open {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=gate_open (sharedassets27.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Opened
  }

  state Open 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Door Open 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Opened
  }

  state Opened {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Open Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Dust {
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  var Dust: gameObject = <null>
  var Self: gameObject = <null>
}
