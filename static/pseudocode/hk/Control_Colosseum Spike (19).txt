fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Dust", storeResult=var "Dust")
    on FINISHED → Idle
  }

  state Idle {
    on EXPAND → Antic
  }

  state Antic {
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Antic")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Expand
  }

  state Expand {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Expand")
    StopParticleEmitter(gameObject=var "Dust")
    on RETRACT → Retract
  }

  state Retract {
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Retract", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Dust: gameObject = <null>
}
