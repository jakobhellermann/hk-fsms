# Head Control
# 1 location(s):
#   GG_Brooding_Mawlek_V / Battle Scene/Tiso Boss/Shield

fsm Head Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=var "Self", vector=(unset), x=(unset), y=(unset), z=var "z", space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on TAKE DAMAGE → Hit Effects
  }

  state Check Direction {
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="magnitudeMult", storeValue=var "Attack Magnitude", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Left
    on RIGHT → Right
    on UP → Up
    on DOWN → Down
  }

  state Left {
    RandomFloat(min=130, max=140, storeResult=var "Angle")
    RandomFloat(min=15, max=20, storeResult=var "Speed")
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "Attack Magnitude", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Right {
    RandomFloat(min=40, max=50, storeResult=var "Angle")
    RandomFloat(min=15, max=20, storeResult=var "Speed")
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "Attack Magnitude", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Spin {
    GetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=10, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Speed", everyFrame=false)
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Up {
    RandomFloat(min=85, max=95, storeResult=var "Angle")
    RandomFloat(min=20, max=25, storeResult=var "Speed")
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "Attack Magnitude", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Down {
    RandomFloat(min=250, max=290, storeResult=var "Angle")
    RandomFloat(min=15, max=20, storeResult=var "Speed")
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "Attack Magnitude", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Hit Effects {
    GetEventSender(sentByGameObject=var "Damager")
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage Dealt", everyFrame=false)
    IntCompare(integer1=var "Damage Dealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Check Direction
    on CANCEL → Idle
  }

  var Type: int = 0
  var Death Anim: string = ""
  var Idle Anim: string = ""
  var Push Anim: string = ""

  var Angle: float = 0
  var Attack Direction: float = 0
  var Attack Magnitude: float = 0
  var Chooser: float = 0
  var Speed: float = 0
  var Spin Factor: float = 0
  var z: float = 0
  var Damage Dealt: int = 0
  var Chooser Bool: bool = false
  var Damager: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Pusher: gameObject = <null>
  var Self: gameObject = <null>
}
