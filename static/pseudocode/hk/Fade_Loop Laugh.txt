fsm Fade {
  start Idle

  state Idle {
    on START → Start
  }

  state Start {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on OUT → Out
  }

  state Out {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.15)
    on IN → In
  }

  state In {
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=0.5)
    on OUT → Out
  }
}
