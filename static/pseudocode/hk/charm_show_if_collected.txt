// uses template: charm_show_if_collected
fsm charm_show_if_collected {
  start Pause
  on UPDATE → Check  // from any state
  on EQUIP → Equipped  // from any state

  state Check {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Glow", storeResult=var "Glow")
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Charm Sprite")
    GetName(gameObject=var "Self", storeName=var "Name", everyFrame=false)
    ConvertStringToInt(stringVariable=var "Name", intVariable=var "ID", everyFrame=false)
    BuildString(stringParts=["gotCharm_", var "Name"], separator="", addToEnd=true, storeResult=var "PD Bool Name", everyFrame=false)
    CallMethodProper(gameObject=var "Charm Icons", behaviour="CharmIconList", methodName="GetSprite", parameters=[var "ID"], storeResult=var "Sprite")
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=var "Sprite")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool Name", isTrue=→"GOT", isFalse=→"NOT")
    on GOT → Glass HP
    on NOT → Not
  }

  state Got {
    ActivateGameObject(gameObject=var "Charm Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetName(gameObject=var "Self", storeName=var "Name", everyFrame=false)
    BuildString(stringParts=["equippedCharm_", var "Name"], separator="", addToEnd=true, storeResult=var "PD Bool Name", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool Name", isTrue=→"EQUIPPED", isFalse=(none))
    SetMaterialColor(gameObject=var "Charm Sprite", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    ActivateGameObject(gameObject=var "Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmColor(gameObject=var "Charm Sprite", fsmName="color_fader", variableName="Up Colour", setValue=(1, 1, 1, 1), everyFrame=false)  // disabled
    on EQUIPPED → Equipped
  }

  state Not {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  state Equipped {
    ActivateGameObject(gameObject=var "Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMaterialColor(gameObject=var "Charm Sprite", materialIndex=0, material=<null>, namedColor="_Color", color=(0.4705882, 0.4705882, 0.4705882, 1), everyFrame=false)
    SetFsmColor(gameObject=var "Charm Sprite", fsmName="color_fader", variableName="Up Colour", setValue=(0.4705882, 0.4705882, 0.4705882, 1), everyFrame=false)  // disabled
  }

  state Glass HP {
    IntCompare(integer1=var "ID", integer2=23, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brokenCharm_23", isTrue=(none), isFalse=→"FINISHED")
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=_0002_charm_glass_heal_broken)
    on FINISHED → Glass Geo
  }

  state Glass Geo {
    IntCompare(integer1=var "ID", integer2=24, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brokenCharm_24", isTrue=(none), isFalse=→"FINISHED")
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=_0003_charm_glass_geo_broken)
    on FINISHED → Glass Attack
  }

  state Glass Attack {
    IntCompare(integer1=var "ID", integer2=25, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brokenCharm_25", isTrue=(none), isFalse=→"FINISHED")
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=_0002_charm_glass_attack_up_broken)
    on FINISHED → Royal Charm
  }

  state Royal Charm {
    IntCompare(integer1=var "ID", integer2=36, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    IntSwitch(intVariable=var "Royal Charm State", compareTo=[1, 2, 3, 4], sendEvent=[→"R QUEEN", →"R KING", →"R FINAL", →"R SHADE"], everyFrame=false)
    on FINISHED → Got
    on R QUEEN → R Queen
    on R KING → R King
    on R FINAL → R Final
    on R SHADE → R Shade
  }

  state R Queen {
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=charm_white_left)
    on FINISHED → Got
  }

  state R King {
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=charm_white_right)
    on FINISHED → Got
  }

  state R Final {
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=charm_white_full)
    on FINISHED → Got
  }

  state R Shade {
    SetSpriteRendererSprite(gameObject=var "Charm Sprite", sprite=charm_black)
    on FINISHED → Got
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  var ID: int = 0
  var Royal Charm State: int = 0
  var Name: string = ""
  var PD Bool Name: string = ""
  var Charm Sprite: gameObject = <null>
  var Glow: gameObject = <null>
  var Self: gameObject = <null>
  var Sprite: object = <null>
}
