# Conversation Control
# 1 location(s):
#   Room_shop / Basement Closed/Sly Shop

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Convo Choice {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="Area Title Control", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="Area Title Control", variableName="Area Event", setValue="SLY", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metSlyShop", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasSlykey", isTrue=→"STORE KEY", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="troupeInTown", falseBool="slyConvoGrimm", isTrue=→"GRIMM", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="nymmInTown", falseBool="slyNymmConvo", isTrue=→"NYMM", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasNailArt", falseBool="slyConvoNailArt", isTrue=→"NAIL ART", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="honedNail", falseBool="slyConvoNailHoned", isTrue=→"NAIL HONED", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="openedMapperShop", falseBool="slyConvoMapper", isTrue=→"MAPPER", isFalse=(none))
    on STORE KEY → Store Key
    on MEET → Meet
    on FINISHED → Generic
    on NAIL ART → Nail Art
    on NAIL HONED → Nail Honed
    on MAPPER → Mapper
    on GRIMM → Grimm
    on NYMM → Nymm
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Active?
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookDown")  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")  // disabled
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Active? {
    GetPlayerDataBool(boolName="slyRescued", storeValue=var "Met")
    BoolTest(boolVariable=var "Met", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
  }

  state Store Key {
    FindGameObject(objectName="Shop Menu", withTag="Untagged", store=var "Shop Menu")
    SetPlayerDataBool(boolName="hasSlykey", value=false)
    SetPlayerDataBool(boolName="gaveSlykey", value=true)
    SendMessage(gameObject=var "Shop Menu", delivery=SendMessage, options=1, functionCall=UpdateStock())
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_KEY_2", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Meet {
    SetPlayerDataBool(boolName="metSlyShop", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_SHOP_INTRO", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → To Shop
  }

  state Generic {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotCharm_26", isTrue=→"GOT NM CHARM", isFalse=(none))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_GENERIC", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
    on GOT NM CHARM → Generic NM
  }

  state Nail Art {
    SetPlayerDataBool(boolName="slyConvoNailArt", value=true)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotCharm_26", isTrue=→"GOT NM CHARM", isFalse=(none))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_NAILART", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
    on GOT NM CHARM → Generic NM
  }

  state Nail Honed {
    SetPlayerDataBool(boolName="slyConvoNailHoned", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_NAIL_HONED", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Mapper {
    SetPlayerDataBool(boolName="slyConvoMapper", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_MAPPER", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Generic NM {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_GENERIC_NM", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state To Shop {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Turn L")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DO SHOP OPEN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO CANCEL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Grimm {
    SetPlayerDataBool(boolName="slyConvoGrimm", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_GRIMM", "CP2"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Nymm {
    SetPlayerDataBool(boolName="slyNymmConvo", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk (sharedassets10.assets), Sly_talk_02 (sharedassets10.assets), Sly_talk_03 (sharedassets10.assets), Sly_talk_04 (sharedassets17.assets), Sly_talk_05 (sharedassets17.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_NYMM", "CP2"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var Egg Temple: int = 0
  var Elderbug State: int = 0
  var Beta: bool = false
  var Met: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Shop Menu: gameObject = <null>
  var Voice Player: gameObject = <null>
}
