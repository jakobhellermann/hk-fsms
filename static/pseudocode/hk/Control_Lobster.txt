# Control
# 2 location(s):
#   Room_Colosseum_Gold / Colosseum Manager/Waves/Lobster Lancer/Entry Object/Lobster
#   GG_God_Tamer / Entry Object/Lobster

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Hit Ball", storeResult=var "Hit Ball")
    FindChild(gameObject=Self, childName="Hit Idle", storeResult=var "Hit Idle")
    FindChild(gameObject=Self, childName="Hit Spit Antic", storeResult=var "Hit Spit Antic")
    FindChild(gameObject=Self, childName="Hit Spit", storeResult=var "Hit Spit")
    FindChild(gameObject=Self, childName="Spit Effect", storeResult=var "Spit Effect")
    FindChild(gameObject=Self, childName="Spit Point", storeResult=var "Spit Point")
    FindChild(gameObject=Self, childName="Roll Effect", storeResult=var "Roll Effect")
    on FINISHED → Dormant
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lobster Idle")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Lobster TurnToIdle", resetFrame=true, everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move Choice
  }

  state Move Choice {
    SendRandomEvent(events=[→"ATTACK", →"BACK", →"FORWARD"], weights=[0.5, 0.25, 0.25], delay=0)
    on FORWARD → Forward
    on BACK → Back
    on ATTACK → Attack Choice
  }

  state Forward {
    SetFloatValue(floatVariable=var "Speed Crt", floatValue=var "Scuttle Speed", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=var "Scale", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lobster Walk")
    on FINISHED → Scuttle
  }

  state Back {
    SetFloatValue(floatVariable=var "Speed Crt", floatValue=var "Scuttle Speed", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=var "Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=-1, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lobster Reverse")
    on FINISHED → Scuttle
  }

  state Scuttle {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed Crt", y=(unset), everyFrame=false)
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on FINISHED → Scuttle End
  }

  state Scuttle End {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lobster Idle")
    on FINISHED → Attack Choice
  }

  state Attack Choice {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendRandomEvent(events=[→"ROLL CHARGE", →"IDLE", →"SPIT"], weights=[0.375, 0.25, 0.375], delay=0)
    SendRandomEventV3(events=[→"ROLL CHARGE", →"IDLE", →"SPIT"], weights=[0.375, 0.25, 0.375], trackingInts=[var "Ct Roll", var "Ct Idle", var "Ct Spit"], eventMax=[3, 2, 1], trackingIntsMissed=[var "Ms Roll", var "Ms Idle", var "Ms Spit"], missedMax=[4, 8, 5])
    on FINISHED → Idle
    on ROLL CHARGE → RC Antic
    on SPIT → Spit Turn?
  }

  state RC Antic {
    ActivateGameObject(gameObject=var "Hit Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Ball", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lobster Roll Up")
    SetFloatValue(floatVariable=var "Speed Crt", floatValue=var "Antic Speed", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=var "Scale", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed Crt", y=(unset), everyFrame=false)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0)
    Wait(time=0.85, finishEvent=→"FINISHED", realTime=false)
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=blocker_close (resources.assets))
    ActivateGameObject(gameObject=var "Roll Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → RC Charge
  }

  state RC Charge {
    SetFloatValue(floatVariable=var "Speed Crt", floatValue=var "Charge Speed", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=var "Scale", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed Crt", y=(unset), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → RC Charging
  }

  state RC Charging {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on WALL → RC Launch
  }

  state RC Launch {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Jump X", floatValue=var "Hero X", everyFrame=false)
    FloatSubtract(floatVariable=var "Jump X", subtract=var "Self X", everyFrame=false, perSecond=false)
    FloatMultiply(floatVariable=var "Jump X", multiplyBy=1.25, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=48, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=zombie_guard_club (sharedassets34.assets))
    ActivateGameObject(gameObject=var "Roll Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FORWARD", realTime=false)
    on FORWARD → RC Air
  }

  state RC Air {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → RC Land
  }

  state RC Land {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lobster Roll End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=blocker_open (resources.assets))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_land_1st_time (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    on FINISHED → Idle
  }

  state Spit Turn? {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Hero Is Right", leftBool=(unset), everyFrame=false)
    FloatTestToBool(float1=var "X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Facing Right", greaterThanBool=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Is Right"], boolStates=[true, true], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Is Right"], boolStates=[false, false], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lobster TurnToIdle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spit Retreat?
  }

  state Spit Retreat? {
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "X Distance", everyFrame=false)
    AudioPlayInState(gameObject=Self, volume=1)
    FloatCompare(float1=var "X Distance", float2=6, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Speed Crt", floatValue=var "Scuttle Speed", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=var "Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=-1, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lobster Reverse")
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=1.5, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed Crt", y=(unset), everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spit Antic
  }

  state Spit Antic {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Spit Antic", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lobster Spit Antic")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mawlek_big_spit (sharedassets34.assets))
    on FINISHED → Fire Globs
  }

  state Spit {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lobster Spit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Spit Antic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Spit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Spit Recover
  }

  state Spit Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lobster Spit Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Hit Spit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Idle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Fire Globs {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    FloatOperator(float1=2, float2=var "X Scale", operation=Multiply, storeResult=var "Spit Speed", everyFrame=false)
    FlingObjectsFromGlobalPoolVel(gameObject=Vomit Glob:0 (sharedassets34.assets), spawnPoint=var "Spit Point", position=(unset), spawnMin=4, spawnMax=4, speedMinX=var "Spit Speed", speedMaxX=var "Spit Speed", speedMinY=0, speedMaxY=4, originVariationX=0.25, originVariationY=0.25)
    FloatOperator(float1=10, float2=var "X Scale", operation=Multiply, storeResult=var "Spit Speed", everyFrame=false)
    FlingObjectsFromGlobalPoolVel(gameObject=Vomit Glob:0 (sharedassets34.assets), spawnPoint=var "Spit Point", position=(unset), spawnMin=4, spawnMax=4, speedMinX=var "Spit Speed", speedMaxX=var "Spit Speed", speedMinY=14, speedMaxY=19, originVariationX=0.25, originVariationY=0.25)
    FloatOperator(float1=20, float2=var "X Scale", operation=Multiply, storeResult=var "Spit Speed", everyFrame=false)
    FlingObjectsFromGlobalPoolVel(gameObject=Vomit Glob:0 (sharedassets34.assets), spawnPoint=var "Spit Point", position=(unset), spawnMin=4, spawnMax=4, speedMinX=var "Spit Speed", speedMaxX=var "Spit Speed", speedMinY=14, speedMaxY=19, originVariationX=0.25, originVariationY=0.25)
    on FINISHED → Spit
  }

  state Dormant {
    on WAKE → Wake
  }

  state Wake {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Idle
  }

  var Antic Speed: float = 15
  var Charge Speed: float = -25
  var Hero X: float = 0
  var Jump X: float = 0
  var Scale: float = 0
  var Scuttle Speed: float = -6
  var Self X: float = 0
  var Speed Crt: float = 0
  var Spit Speed: float = 0
  var X Distance: float = 0
  var X Scale: float = 0
  var Ct Idle: int = 0
  var Ct Roll: int = 0
  var Ct Spit: int = 0
  var Ms Idle: int = 0
  var Ms Roll: int = 0
  var Ms Spit: int = 0
  var Facing Right: bool = false
  var Hero Is Right: bool = false
  var Hit Ball: gameObject = <null>
  var Hit Idle: gameObject = <null>
  var Hit Spit: gameObject = <null>
  var Hit Spit Antic: gameObject = <null>
  var Roll Effect: gameObject = <null>
  var Self: gameObject = <null>
  var Spit Effect: gameObject = <null>
  var Spit Point: gameObject = <null>
}
