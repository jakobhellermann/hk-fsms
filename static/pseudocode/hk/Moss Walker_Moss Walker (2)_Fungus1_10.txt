# Moss Walker
# 1 location(s):
#   Fungus1_10 / Moss Walker (2)

fsm Moss Walker {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Edge Range", storeResult=var "Edge Range Obj")
    FindChild(gameObject=Self, childName="Wall Range", storeResult=var "Wall Range Obj")
    FindChild(gameObject=Self, childName="Ground Range", storeResult=var "Ground Range Obj")
    on FINISHED → Check Type
  }

  state Rest {
    SetBoolValue(boolVariable=var "Encountered Hero", boolValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Wake Range", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    on WAKE → Wake Pause
  }

  state Wake Pause {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  state Shake {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hide Grass", audioClip=moss_flyer_walker_emerge (sharedassets128.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shake")
    SetParticleEmissionRate(gameObject=var "Shake Grass", emissionRate=35, everyFrame=false)
    Wait(time=1.2, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Shake Grass", audioClips=[ceiling_dropper_look_2 (sharedassets48.assets), ceiling_dropper_look_4 (sharedassets48.assets), ceiling_dropper_look_5 (sharedassets48.assets)], weights=[1, 1, 1], pitchMin=0.65, pitchMax=0.7, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Wake
  }

  state Wake {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetParticleEmissionRate(gameObject=var "Shake Grass", emissionRate=0, everyFrame=false)
    RandomFloat(min=3, max=5, storeResult=var "Hide Timer")
    on FINISHED → Activate
  }

  state Activate {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(false))
    on FINISHED → Check Dir
  }

  state Check Dir {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=→"LEFT DOWN", lessThan=→"UP RIGHT", greaterThan=→"LEFT DOWN", everyFrame=false)
    on UP RIGHT → Up Right
    on LEFT DOWN → Left Down
  }

  state Up Right {
    SetVector2XY(vector2Variable=var "Current Velocity", vector2Value=var "UR Velocity", x=(unset), y=(unset), everyFrame=false)
    GetVector2XY(vector2Variable=var "UR Velocity", storeX=(unset), storeY=var "Velocity Y", everyFrame=false)
    on FINISHED → Walk Start
  }

  state Check Type {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=(unset), yScale=var "Y Scale", zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Angle", float2=90, tolerance=1, equal=→"WALL", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Y Scale", float2=-1, tolerance=0.1, equal=→"ROOF", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Angle", float2=0, tolerance=1, equal=→"FLOOR", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FLOOR → Floor
    on ROOF → Roof
    on WALL → Wall
    on FINISHED → Floor
  }

  state Floor {
    SetVector2XY(vector2Variable=var "LD Velocity", vector2Value=(unset), x=-3, y=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "UR Velocity", vector2Value=(unset), x=3, y=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SetRecoilSpeed(target=Self, newRecoilSpeed=15)
    SetRecoilFreeze(target=Self, freeze=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Flip Particles?
  }

  state Roof {
    SetVector2XY(vector2Variable=var "LD Velocity", vector2Value=(unset), x=-3, y=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "UR Velocity", vector2Value=(unset), x=3, y=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetRecoilSpeed(target=Self, newRecoilSpeed=0)
    SetRecoilFreeze(target=Self, freeze=true)
    on FINISHED → Flip Particles?
  }

  state Wall {
    SetVector2XY(vector2Variable=var "LD Velocity", vector2Value=(unset), x=0, y=-3, everyFrame=false)
    SetVector2XY(vector2Variable=var "UR Velocity", vector2Value=(unset), x=0, y=3, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetRecoilSpeed(target=Self, newRecoilSpeed=0)
    SetRecoilFreeze(target=Self, freeze=true)
    on FINISHED → Flip Particles?
  }

  state Left Down {
    SetVector2XY(vector2Variable=var "Current Velocity", vector2Value=var "LD Velocity", x=(unset), y=(unset), everyFrame=false)
    GetVector2XY(vector2Variable=var "LD Velocity", storeX=var "Velocity X", storeY=(unset), everyFrame=false)
    on FINISHED → Walk Start
  }

  state Walking {
    BoolTest(boolVariable=var "Edge Range", isTrue=(none), isFalse=→"TURN", everyFrame=true)  // disabled
    ConstrainMovement(gameObject=Self, xConstrain=0.25, yConstrain=0.25)
    BoolTest(boolVariable=var "Wall Range", isTrue=→"TURN", isFalse=(none), everyFrame=true)  // disabled
    SetVelocity2d(gameObject=Self, vector=var "Current Velocity", x=(unset), y=(unset), everyFrame=true)  // disabled
    FloatSubtract(floatVariable=var "Hide Timer", subtract=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Hide Timer", float2=0, tolerance=0, equal=→"CHECK HIDE", lessThan=→"CHECK HIDE", greaterThan=(none), everyFrame=true)
    SetBoolValue(boolVariable=var "Wall Range", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Edge Range", boolValue=true, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Wall Range Obj", fromPosition=(unset), direction=(-1, 0), space=1, distance=0.5, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Wall Range", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=3, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    RayCast2dV2(fromGameObject=var "Edge Range Obj", fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Edge Range", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=3, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "Edge Range", isTrue=(none), isFalse=→"TURN", everyFrame=true)
    BoolTest(boolVariable=var "Wall Range", isTrue=→"TURN", isFalse=(none), everyFrame=true)
    on TURN → Turn Check
    on CHECK HIDE → Set Encountered
  }

  state Turn Check {
    RayCast2dV2(fromGameObject=var "Ground Range Obj", fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Ground Range", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=3, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "Ground Range", isTrue=→"FINISHED", isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Turn
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Walking
  }

  state Walk Start {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    ConstrainMovement(gameObject=Self, xConstrain=0.25, yConstrain=0.25)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    SetVelocity2d(gameObject=Self, vector=var "Current Velocity", x=(unset), y=(unset), everyFrame=false)
    FloatSubtract(floatVariable=var "Hide Timer", subtract=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Hide Timer", float2=0, tolerance=0, equal=→"CHECK HIDE", lessThan=→"CHECK HIDE", greaterThan=(none), everyFrame=true)
    BoolTest(boolVariable=var "Edge Range", isTrue=(none), isFalse=→"TURN", everyFrame=true)  // disabled
    BoolTest(boolVariable=var "Wall Range", isTrue=→"TURN", isFalse=(none), everyFrame=true)  // disabled
    on FINISHED → Walking
    on CHECK HIDE → Set Encountered
    on TURN → Turn Check
  }

  state Turn {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Flip
  }

  state Flip {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Check Dir
  }

  state Check Hide {
    SetFloatValue(floatVariable=var "Hide Timer", floatValue=1, everyFrame=false)
    BoolTest(boolVariable=var "Encountered Hero", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    BoolTest(boolVariable=var "Wake Range", isTrue=→"CANCEL", isFalse=→"HIDE", everyFrame=false)
    on CANCEL → Restart Velocity
    on HIDE → Hide
  }

  state Hide {
    AudioStop(gameObject=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bury", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHeroAmount(target=Self, damageDealt=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Hide Grass", emit=0)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(true))
    on FINISHED → Rest
  }

  state Flip Particles? {
    FloatCompare(float1=var "Y Scale", float2=1, tolerance=0.1, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetRotation(gameObject=var "Hide Grass", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=180, space=1, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Shake Grass", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=180, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Roams?
  }

  state Roams? {
    BoolTest(boolVariable=var "Roams", isTrue=→"YES", isFalse=→"NO", everyFrame=false)
    on YES → Activate
    on NO → Rest
  }

  state Set Encountered {
    BoolTest(boolVariable=var "Encountered Hero", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Wake Range", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Encountered Hero", boolValue=true, everyFrame=false)
    on FINISHED → Check Hide
  }

  state Restart Velocity {
    SetVelocity2d(gameObject=Self, vector=var "Current Velocity", x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Walking
  }

  var Roams: bool = true

  var Angle: float = 0
  var Hide Timer: float = 0
  var Velocity X: float = 0
  var Velocity Y: float = 0
  var X Scale: float = 0
  var Y Scale: float = 0
  var Edge Range: bool = false
  var Encountered Hero: bool = false
  var Ground Range: bool = false
  var Wake Range: bool = false
  var Wall Range: bool = false
  var Current Velocity: vector2 = (0, 0)
  var LD Velocity: vector2 = (0, 0)
  var UR Velocity: vector2 = (0, 0)
  var Edge Range Obj: gameObject = <null>
  var Ground Range Obj: gameObject = <null>
  var Hide Grass: gameObject = Moss Walker (2)/Hide Grass
  var Self: gameObject = <null>
  var Shake Grass: gameObject = Moss Walker (2)/Shake Grass
  var Wall Range Obj: gameObject = <null>
}
