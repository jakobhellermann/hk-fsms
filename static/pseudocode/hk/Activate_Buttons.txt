# Activate
# 3 location(s):
#   Ruins1_29 / Stag Map/Buttons
#   RestingGrounds_09 / Stag Map/Buttons
#   resources.assets / Stag Map/Buttons

fsm Activate {
  start Wait

  state Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  state Activate {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on STAG MENU DOWN → Deactivate
  }

  state Deactivate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Self: gameObject = <null>
}
