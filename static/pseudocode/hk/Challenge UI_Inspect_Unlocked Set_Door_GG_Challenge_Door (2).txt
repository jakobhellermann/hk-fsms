# Challenge UI
# 1 location(s):
#   GG_Atrium / GG_Challenge_Door (2)/Door/Unlocked Set/Inspect

fsm Challenge UI {
  start Inert
  on CONVO START → Open UI  // from any state

  state Inert {
    GetHero(storeResult=var "Hero")
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
  }

  state Open UI {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    ShowBossDoorChallengeUI(targetDoor=var "Door", prefab=GG_Challenge_Door_Canvas (sharedassets424.assets), cancelEvent=→"FINISHED", challengeEvent=→"DREAM")
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Close UI
    on DREAM → Wait
  }

  state Close UI {
    SendEventByNameV2(eventTarget=GameObject(var "Self"), sendEvent="CONVO END", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=<null>, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Convo", setValue=true, everyFrame=false)
    on FINISHED → Impact
  }

  state Impact {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DREAM ENTER", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", value=var "Return Scene")
    CreateObject(gameObject=dream_area_effect (sharedassets6.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=1.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Dream Box Down
  }

  state Change Scene {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=TimePasses())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=ResetSemiPersistentItems())
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=EnterWithoutInput(true))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AcceptInput())
    SetStaticVariable(variableName="firstBossScene", setValue=var "To Scene")
    GetConstantsValue(variableName="GG_DOOR_ENTRANCE_SCENE", storeValue=var "To Scene")
    BeginSceneTransition(sceneName=var "To Scene", entryGateName="door_dreamEnter", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=false)
  }

  state Dream Box Down {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BOX DOWN DREAM", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CONVO CANCEL", delay=0, everyFrame=false)  // disabled
    ApplyMusicCue(musicCue=None (resources.assets), delayTime=0, transitionTime=1.5)
    Wait(time=1.75, finishEvent=→"DONE", realTime=false)
    on DONE → Change Scene
  }

  state Door Anim {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="TurnToBG")  // disabled
    AnimatorPlay(gameObject=var "Door Animator", stateName="Open", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero Anim
  }

  state Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Door Anim
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Enter")
    on FINISHED → Take Control
  }

  var To Scene: string = ""

  var Return Scene: string = "GG_Atrium"
  var CamLock: gameObject = GG_Challenge_Door (2)/CameraLockArea
  var Door: gameObject = GG_Challenge_Door (2)
  var Door Animator: gameObject = GG_Challenge_Door (2)/Door
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
