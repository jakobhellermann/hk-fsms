# gg_music_control
# 1 location(s):
#   GG_Ghost_Hu / _SceneManager

// uses template: gg_music_control
fsm gg_music_control {
  start Init

  state Boss Sequence {
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "Wait For Event", isTrue=→"EVENT", isFalse=(none), everyFrame=false)
    on FINISHED → Transition Down
    on EVENT → Wait for Event
  }

  state Transition Down {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=GG9_transition_0.5_second, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0.5)
    Wait(time=0.5, finishEvent=→"END", realTime=false)
    on END → Transition Up
  }

  state Transition Up {
    ApplyMusicCue(musicCue=var "MusicCue", delayTime=0.1, transitionTime=0.1)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Frame
  }

  state Wait for Event {
    SetBoolValue(boolVariable=var "Wait For Event", boolValue=false, everyFrame=false)
    on GG MUSIC → Boss Sequence
  }

  state Init {
    GGCheckIfBossSequence(trueEvent=→"FINISHED", falseEvent=→"CANCEL")
    on CANCEL → Wait
    on FINISHED → Skip?
  }

  state Boss Statue {
    ApplyMusicCue(musicCue=var "StatueRegularMusicCue", delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=var "StatueRegularSnapshot", transitionTime=0)
  }

  state Wait {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Statue Level
  }

  state Statue Level {
    CheckGGBossLevel(notGG=(none), level1=→"1", level2=→"2", level3=→"3")
    on 1 → Boss Statue
    on 2 → Boss Statue Alt
    on 3 → Boss Statue Alt
  }

  state Boss Statue Alt {
    BoolTest(boolVariable=var "Has Remix", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    ApplyMusicCue(musicCue=var "StatueRemixMusicCue", delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=var "StatueRemixSnapshot", transitionTime=0)
    on CANCEL → Boss Statue
  }

  state Skip? {
    BoolTest(boolVariable=var "SkipIfNotStatue", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    on CANCEL → Inert
    on FINISHED → Boss Sequence
  }

  state Inert {
  }

  state Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State 1
  }

  state State 1 {
    TransitionToAudioSnapshot(snapshot=var "Audio Snapshot", transitionTime=0.1)
  }

  var Delay: float = 1
  var Has Remix: bool = false
  var SkipIfNotStatue: bool = true
  var Wait For Event: bool = false
  var Audio Snapshot: object = Normal (resources.assets)
  var MusicCue: object = GG Normal (sharedassets428.assets)
  var StatueRegularMusicCue: object = DreamFight (sharedassets161.assets)
  var StatueRegularSnapshot: object = Normal (resources.assets)
  var StatueRemixMusicCue: object = <null>
  var StatueRemixSnapshot: object = Normal (resources.assets)
}
