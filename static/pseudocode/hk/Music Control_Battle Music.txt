# Music Control
# 1 location(s):
#   Fungus1_28 / Battle Music

fsm Music Control {
  start Pause

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedDoubleBlockers", isTrue=→"CANCEL", isFalse=(none))
    on FINISHED → Idle
    on CANCEL → Inert
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Children?
  }

  state Children? {
    GetChildCount(gameObject=Self, storeResult=var "Children")
    IntCompare(integer1=var "Children", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"FINISHED", everyFrame=false)
    on CANCEL → Inert
    on FINISHED → In Battle
  }

  state Inert {
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    SetPlayerDataBool(boolName="defeatedDoubleBlockers", value=true)
  }

  state In Battle {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    ApplyMusicCue(musicCue=Crossroads (sharedassets19.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Action Only (resources.assets), transitionTime=1)
    Wait(time=1, finishEvent=→"KILLED", realTime=false)
    on EXIT → End Music
    on KILLED → Children?
  }

  state End Music {
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Blocker 1: gameObject = Battle Music/Blocker 1
  var Blocker 2: gameObject = Battle Music/Blocker 2

  var Children: int = 0
}
