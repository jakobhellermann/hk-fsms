# Floor Control
# 1 location(s):
#   Fungus2_15_boss / Mantis Battle/mantis_lord_opening_floors (1)

fsm Floor Control {
  start Extended

  state Extended {
    on MLORD FLOOR RETRACT → Retract
  }

  state Retract {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_spike_trap_reveal_3 (sharedassets180.assets))
    Tk2dPlayAnimationWithEvents(gameObject=var "Floor", clipName="Floor Retract", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Retracted
  }

  state Retracted {
    SetMeshRenderer(gameObject=var "Floor", active=false)
    on MLORD FLOOR EXTEND → Extend
  }

  state Extend {
    BoolTest(boolVariable=var "Hero Check", isTrue=→"HERO CHECK", isFalse=(none), everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_spike_trap_reveal_2 (sharedassets180.assets))
    Tk2dPlayAnimationWithEvents(gameObject=var "Floor", clipName="Floor Extend", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Floor", active=true)
    on FINISHED → Extended
    on HERO CHECK → Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Extend
  }

  var Hero Check: bool = false
  var Floor: gameObject = Mantis Battle/mantis_lord_opening_floors (1)/Floor
}
