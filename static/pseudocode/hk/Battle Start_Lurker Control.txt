# Battle Start
# 1 location(s):
#   GG_Lurker / Lurker Control

fsm Battle Start {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    FindChild(gameObject=Self, childName="Battle Gate", storeResult=var "Gate")
    FindChild(gameObject=Self, childName="Lurker Gate", storeResult=var "Colosseum Gate")
    on ENTER → Start
  }

  state Start {
    SendEventByName(eventTarget=GameObject(var "Gate"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Pale Lurker"), sendEvent="START", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Pale Lurker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ApplyMusicCue(musicCue=City (sharedassets83.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Action Only (resources.assets), transitionTime=4)  // disabled
    on SHINY PICKED UP → End Pause
  }

  state End {
    SendEventByName(eventTarget=GameObject(var "Gate"), sendEvent="BG OPEN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Colosseum Gate"), sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  var Pale Lurker: gameObject = Lurker Control/Pale Lurker

  var Colosseum Gate: gameObject = <null>
  var Gate: gameObject = <null>
}
