# Control
# 3 location(s):
#   Room_Colosseum_Bronze / Colosseum Manager/Music Control
#   Room_Colosseum_Silver / Colosseum Manager/Music Control
#   Room_Colosseum_Gold / Colosseum Manager/Music Control

fsm Control {
  start Idle
  on MUSIC 6 → Set Music 6  // from any state
  on MUSIC 5 → Set Music 5  // from any state
  on MUSIC 4 → Set Music 4  // from any state
  on MUSIC 3 → Set Music 3  // from any state
  on MUSIC 2 → Set Music 2  // from any state
  on MUSIC 1 → Set Music 1  // from any state
  on MUSIC SILENT → None  // from any state
  on MUSIC STING DELAY → Sting Delay Long  // from any state
  on MUSIC STING → Sting Pause  // from any state
  on HERO LEAVE → Hero Death  // from any state

  state Idle {
  }

  state Sting {
    AudioStop(gameObject=Self)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=S57 COLOSSEUM STING (sharedassets32.assets))
  }

  state Set Music 1 {
    SetObjectValue(objectVariable=var "New Track", objectValue=S57 COLOSSEUM INTENSITY 1 (sharedassets32.assets), everyFrame=false)
    on FINISHED → Music Fade
  }

  state Music Fade {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.25)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → New Track
  }

  state New Track {
    AudioStop(gameObject=Self)
    SetAudioClip(gameObject=Self, audioClip=var "New Track")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }

  state Set Music 2 {
    SetObjectValue(objectVariable=var "New Track", objectValue=S57 COLOSSEUM INTENSITY 2 (sharedassets32.assets), everyFrame=false)
    on FINISHED → Music Fade
  }

  state None {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.25)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 1
  }

  state Set Music 3 {
    SetObjectValue(objectVariable=var "New Track", objectValue=S57 COLOSSEUM INTENSITY 3 (sharedassets32.assets), everyFrame=false)
    on FINISHED → Music Fade
  }

  state Set Music 4 {
    SetObjectValue(objectVariable=var "New Track", objectValue=S57 COLOSSEUM INTENSITY 4 (sharedassets32.assets), everyFrame=false)
    on FINISHED → Music Fade
  }

  state Set Music 5 {
    SetObjectValue(objectVariable=var "New Track", objectValue=S57 COLOSSEUM INTENSITY 5 (sharedassets32.assets), everyFrame=false)
    on FINISHED → Music Fade
  }

  state Set Music 6 {
    SetObjectValue(objectVariable=var "New Track", objectValue=S57 COLOSSEUM INTENSITY 6 (sharedassets32.assets), everyFrame=false)
    on FINISHED → Music Fade
  }

  state Sting Delay Long {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sting
  }

  state State 1 {
    AudioStop(gameObject=Self)
  }

  state Sting Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sting
  }

  state Hero Death {
    FinishFSM()
  }

  var New Track: object = <null>
}
