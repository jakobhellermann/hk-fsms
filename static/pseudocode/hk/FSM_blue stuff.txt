# FSM
# 1 location(s):
#   GG_Spa / blue stuff

fsm FSM {
  start Get Bindings

  state Get Bindings {
    GGGetCompletedBindings(storeValue=var "Completed Bindings")
    IntCompare(integer1=var "Completed Bindings", integer2=16, equal=→"LEVEL 3", lessThan=(none), greaterThan=→"LEVEL 3", everyFrame=false)
    IntCompare(integer1=var "Completed Bindings", integer2=12, equal=→"LEVEL 2", lessThan=(none), greaterThan=→"LEVEL 2", everyFrame=false)
    IntCompare(integer1=var "Completed Bindings", integer2=8, equal=→"LEVEL 1", lessThan=(none), greaterThan=→"LEVEL 1", everyFrame=false)
    on FINISHED → Inert
    on LEVEL 1 → Level 1
    on LEVEL 2 → Level 2
    on LEVEL 3 → Level 3
  }

  state Inert {
  }

  state Set Amount {
    CallMethodProper(gameObject=blue stuff/Health Cocoon (1), behaviour="HealthCocoon", methodName="SetScuttlerAmount", parameters=[var "Scuttler Amount"], storeResult=(unset var))
  }

  state Level 1 {
    SetIntValue(intVariable=var "Scuttler Amount", intValue=3, everyFrame=false)
    on FINISHED → Set Amount
  }

  state Level 2 {
    SetIntValue(intVariable=var "Scuttler Amount", intValue=4, everyFrame=false)
    on FINISHED → Set Amount
  }

  state Level 3 {
    SetIntValue(intVariable=var "Scuttler Amount", intValue=5, everyFrame=false)
    on FINISHED → Set Amount
  }

  var Completed Bindings: int = 0
  var Scuttler Amount: int = 0
}
