# Spawn Barbs
# 2 location(s):
#   Deepnest_East_Hornet / Barb Region
#   GG_Hornet_2 / Barb Region

fsm Spawn Barbs {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=false)
    BoundsBoxCollider(gameObject1=Self, scaleVector2=(unset), scaleX=var "Spawn Max X", scaleY=var "Spawn Max Y", everyFrame=false)
    FloatMultiply(floatVariable=var "Spawn Max X", multiplyBy=0.5, everyFrame=false)
    FloatMultiply(floatVariable=var "Spawn Max Y", multiplyBy=0.5, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Min X", floatValue=var "Spawn Max X", everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Min Y", floatValue=var "Spawn Max Y", everyFrame=false)
    FloatMultiply(floatVariable=var "Spawn Min X", multiplyBy=-1, everyFrame=false)
    FloatMultiply(floatVariable=var "Spawn Min Y", multiplyBy=-1, everyFrame=false)
    FloatAdd(floatVariable=var "Spawn Min X", add=var "Self X", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Spawn Max X", add=var "Self X", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Spawn Min Y", add=var "Self Y", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Spawn Max Y", add=var "Self Y", everyFrame=false, perSecond=false)
    on FINISHED → Idle
  }

  state Idle {
    SetFloatValue(floatVariable=var "Spawn X", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Y", floatValue=0, everyFrame=false)
    on SPAWN3 → Spawn 1
  }

  state Spawn 1 {
    RandomFloat(min=var "Spawn Min X", max=var "Spawn Max X", storeResult=var "Spawn X")
    RandomFloat(min=var "Spawn Min Y", max=var "Spawn Max Y", storeResult=var "Spawn Y")
    SetVector3XYZ(vector3Variable=var "Spawn Pos", vector3Value=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Hornet Barb (sharedassets324.assets), spawnPoint=var "", position=var "Spawn Pos", rotation=(unset), storeObject=var "Barb Obj")
    SetIntValue(intVariable=var "Redos", intValue=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Prev Spawn Pos", vector3Value=var "Spawn Pos", everyFrame=false)
    Wait(time=0.12, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn 2
  }

  state Spawn 2 {
    RandomFloat(min=var "Spawn Min X", max=var "Spawn Max X", storeResult=var "Spawn X")
    RandomFloat(min=var "Spawn Min Y", max=var "Spawn Max Y", storeResult=var "Spawn Y")
    SetVector3XYZ(vector3Variable=var "Spawn Pos", vector3Value=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Spawn Pos", point2=var "Prev Spawn Pos", ignoreZ=false, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=var "Min Distance", tolerance=0, equal=(none), lessThan=→"REDO", greaterThan=(none), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Hornet Barb (sharedassets324.assets), spawnPoint=var "", position=var "Spawn Pos", rotation=(unset), storeObject=var "Barb Obj")
    SetIntValue(intVariable=var "Redos", intValue=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Prev Spawn Pos", vector3Value=var "Spawn Pos", everyFrame=false)
    Wait(time=0.12, finishEvent=→"FINISHED", realTime=false)
    on REDO → Redo 2
    on FINISHED → Spawn 3
  }

  state Redo 2 {
    IntAdd(intVariable=var "Redos", add=1, everyFrame=false)
    IntCompare(integer1=var "Redos", integer2=500, equal=(none), lessThan=(none), greaterThan=→"FRAME", everyFrame=false)
    on FINISHED → Spawn 2
    on FRAME → Frame 2
  }

  state Frame 2 {
    SetIntValue(intVariable=var "Redos", intValue=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Spawn 2
  }

  state Spawn 3 {
    RandomFloat(min=var "Spawn Min X", max=var "Spawn Max X", storeResult=var "Spawn X")
    RandomFloat(min=var "Spawn Min Y", max=var "Spawn Max Y", storeResult=var "Spawn Y")
    SetVector3XYZ(vector3Variable=var "Spawn Pos", vector3Value=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Spawn Pos", point2=var "Prev Spawn Pos", ignoreZ=false, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=var "Min Distance", tolerance=0, equal=(none), lessThan=→"REDO", greaterThan=(none), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Hornet Barb (sharedassets324.assets), spawnPoint=var "", position=var "Spawn Pos", rotation=(unset), storeObject=var "Barb Obj")
    SetIntValue(intVariable=var "Redos", intValue=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Prev Spawn Pos", vector3Value=var "Spawn Pos", everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on REDO → Redo 3
    on FINISHED → Spike
  }

  state Redo 3 {
    IntAdd(intVariable=var "Redos", add=1, everyFrame=false)
    IntCompare(integer1=var "Redos", integer2=500, equal=(none), lessThan=(none), greaterThan=→"FRAME", everyFrame=false)
    on FINISHED → Spawn 3
    on FRAME → Frame 3
  }

  state Frame 3 {
    SetIntValue(intVariable=var "Redos", intValue=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Spawn 3
  }

  state Spike {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BARB READY", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Distance: float = 0
  var Min Distance: float = 3
  var Self X: float = 0
  var Self Y: float = 0
  var Spawn Max X: float = 0
  var Spawn Max Y: float = 0
  var Spawn Min X: float = 0
  var Spawn Min Y: float = 0
  var Spawn X: float = 0
  var Spawn Y: float = 0
  var Redos: int = 0
  var Prev Spawn Pos: vector3 = (0, 0, 0)
  var Spawn Pos: vector3 = (0, 0, 0)
  var Barb Obj: gameObject = <null>
}
