# FSM
# 2 location(s):
#   Deepnest_33 / Zote Deepnest/Slash Impact
#   sharedassets39.assets / Slash Impact

fsm FSM {
  start Choose Anim

  state Choose Anim {
    BoolTest(boolVariable=var "Chooser", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    RandomBool(storeResult=var "Chooser")
    on TRUE → Anim 1
    on FALSE → Anim 2
  }

  state Anim 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Slash Impact 1")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Watch
  }

  state Anim 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Slash Impact 2")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Watch
  }

  state Watch {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Wait(time=0.17, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Kill
  }

  state Kill {
    DestroySelf(detachChildren=false)
  }

  var Chooser: bool = false
}
