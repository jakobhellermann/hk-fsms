fsm Arrow Anim {
  start Init

  state Init {
    SetVector3Value(vector3Variable=var "Move Pos", vector3Value=var "Init Pos", everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Move Pos", addX=0, addY=-0.15, addZ=0, everyFrame=false, perSecond=false)
    on FINISHED → Idle
  }

  state Idle {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Init Pos", time=0.1, delay=0, speed=(unset), space=1, easeType=easeOutSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=false)
    on MOVE → Up
  }

  state Up {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Move Pos", time=0.05, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Idle
    on MOVE → Repeat
  }

  state Repeat {
    on FINISHED → Up
  }

  var Init Pos: vector3 = (-5.68, -8.25, -1)
  var Move Pos: vector3 = (-5.68, -8.25, -1)
}
