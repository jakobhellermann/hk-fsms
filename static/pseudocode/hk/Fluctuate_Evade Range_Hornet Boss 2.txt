# Fluctuate
# 2 location(s):
#   Deepnest_East_Hornet_boss / Hornet Boss 2/Evade Range
#   GG_Hornet_2 / Boss Holder/Hornet Boss 2/Evade Range

fsm Fluctuate {
  start Off

  state Off {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → On
  }

  state On {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Off
    on EVADED → Off
  }
}
