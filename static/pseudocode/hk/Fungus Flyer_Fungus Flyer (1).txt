# Fungus Flyer
# 5 location(s):
#   Fungus2_03 / Fungus Flyer (1)
#   Fungus2_06 / Fungus Flyer (1)
#   Fungus2_29 / Fungus Flyer (1)
#   Fungus2_30 / Fungus Flyer (1)
#   Fungus2_33 / Fungus Flyer (1)

fsm Fungus Flyer {
  start Init
  on SLEEP → Sleep  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Antic Audio", storeResult=var "Antic Audio")
    on FINISHED → Sleep
  }

  state Idle {
    AudioPlayInState(gameObject=Self, volume=1)
    IdleBuzzV2(gameObject=Self, waitMin=0.5, waitMax=0.75, speedMax=4, accelerationMax=20, roamingRangeX=3, roamingRangeY=1, manualStartPos=var "Start Pos")
    on HERO ENTER R → Attack Right
    on HERO ENTER L → Attack Left
  }

  state Spit Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandom(gameObject=var "Antic Audio", audioClips=[Fungoon_antic_05 (sharedassets167.assets), Fungoon_antic_02 (sharedassets167.assets), Fungoon_antic_04 (sharedassets167.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on FINISHED → Spit
  }

  state Spit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Spit Effect", audioClip=plant_turret_spit (sharedassets140.assets), pitchMin=0.75, pitchMax=0.75, volume=1, delay=0, storePlayer=<null>)
    SetAudioPitch(gameObject=Self, pitch=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=fungus_balloon_spit (sharedassets167.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=(unset), y=1, vector3=(unset), everyFrame=true)
    FlingObjectsFromGlobalPoolTime(gameObject=Gas Projectile (sharedassets167.assets), spawnPoint=var "Spawn Point", position=(unset), frequency=0.05, spawnMin=1, spawnMax=1, speedMin=10, speedMax=11, angleMin=var "Angle Min", angleMax=var "Angle Max", originVariationX=0, originVariationY=0)
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Spit Recover
  }

  state Spit Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Spit Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Spit CD
  }

  state Spit CD {
    IdleBuzzV2(gameObject=Self, waitMin=0.75, waitMax=1, speedMax=2.5, accelerationMax=20, roamingRangeX=3, roamingRangeY=1, manualStartPos=var "Start Pos")
    SetAudioPitch(gameObject=Self, pitch=0.75, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Float", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Attack Right {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=278, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=292, everyFrame=false)
    SetFloatValue(floatVariable=var "Horizontal Force", floatValue=-1.5, everyFrame=false)
    on FINISHED → Spit Antic
  }

  state Attack Left {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=248, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=262, everyFrame=false)
    SetFloatValue(floatVariable=var "Horizontal Force", floatValue=1.5, everyFrame=false)
    on FINISHED → Spit Antic
  }

  state Sleep {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    AudioStop(gameObject=Self)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Spit Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on WAKE → Wake
  }

  state Wake {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Float")
    on FINISHED → Idle
  }

  var Angle Max: float = 0
  var Angle Min: float = 0
  var Horizontal Force: float = 0
  var Start Pos: vector3 = (0, 0, 0)
  var Antic Audio: gameObject = <null>
  var Spawn Point: gameObject = Fungus Flyer (1)/Spawn Point
  var Spit Effect: gameObject = Fungus Flyer (1)/Spit Effect
  var Voice Player: gameObject = <null>
}
