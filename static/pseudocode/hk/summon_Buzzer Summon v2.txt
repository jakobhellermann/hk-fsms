fsm summon {
  start Instantiate
  on SPAWN → Init  // from any state

  state Init {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=8, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    GetOwner(storeGameObject=var "Self")
    RandomFloat(min=0, max=100, storeResult=var "Chooser")
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)  // disabled
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Enter
  }

  state Enter {
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=1)  // disabled
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    ColorInterpolate(colors=[(0, 0, 0, 1), (1, 1, 1, 1)], time=1, storeColor=var "Colour", finishEvent=(none), realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    RandomFloat(min=0.75, max=1, storeResult=var "Tween Time")
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -8, -16.98), time=var "Tween Time", delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Summon
  }

  state Summon {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Buzzer", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPosition(gameObject=Self, vector=var "Position", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Buzzer", vector=var "Position", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Buzzer", vector=(unset), x=var "Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFsmBool(gameObject=var "Buzzer", fsmName="chaser", variableName="Start Alert", setValue=true, everyFrame=false)
    DestroySelf(detachChildren=false)
  }

  state Instantiate {
    CreateObject(gameObject=Buzzer (sharedassets32.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Buzzer", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=var "Buzzer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var Chooser: float = 0
  var Scale: float = 0
  var Tween Time: float = 0
  var Position: vector3 = (0, 0, 0)
  var Colour: color = (0, 0, 0, 1)
  var Buzzer: gameObject = <null>
  var Self: gameObject = <null>
}
