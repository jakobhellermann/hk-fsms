# Control
# 1 location(s):
#   Deepnest_East_12 / Hornet Encounter Outskirts

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ACTIVATE → Deactivate
    on ENTER → Turn
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Run {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetVelocity2d(gameObject=Self, vector=(unset), x=14, y=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }

  state Turn {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Run
  }

  var Activated: bool = false
}
