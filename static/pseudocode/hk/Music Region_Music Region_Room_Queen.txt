# Music Region
# 1 location(s):
#   Room_Queen / Music Region

// uses template: music_regionv3
fsm Music Region {
  start Pause

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Dirtmouth Check
  }

  state Enter {
    TransitionToAudioSnapshot(snapshot=var "Enter Music Snapshot", transitionTime=var "Enter Transition Time")
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent=var "Enter Track Event", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Exit
  }

  state Exit {
    TransitionToAudioSnapshot(snapshot=var "Exit Music Snapshot", transitionTime=var "Exit Transition Time")
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent=var "Exit Track Event", delay=0, everyFrame=false)  // disabled
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    ApplyMusicCue(musicCue=var "Exit Music Cue", delayTime=0, transitionTime=0)
    on ENTER → Mines Delay
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Mines Delay {
    BoolTest(boolVariable=(unset), isTrue=(none), isFalse=→"FINISHED", everyFrame=false)  // disabled
    Wait(time=var "Enter Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter Init
  }

  state Dirtmouth Check {
    BoolTest(boolVariable=var "Dirtmouth", isTrue=(none), isFalse=→"NON", everyFrame=false)
    GetFsmInt(gameObject=var "AudioManager", fsmName="Load Music Into Groups", variableName="Current Track", storeValue=var "Current Track", everyFrame=false)
    IntCompare(integer1=var "Current Track", integer2=1, equal=→"NON", lessThan=(none), greaterThan=(none), everyFrame=false)
    TransitionToAudioSnapshot(snapshot=var "Enter Music Snapshot", transitionTime=0)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent=var "Enter Track Event", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on NON → Mines Delay
    on EXIT → Exit
  }

  state Enter Init {
    BoolTest(boolVariable=var "Init", isTrue=→"INIT", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Init", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=var "Enter Music Snapshot", transitionTime=var "Enter Transition Time Init")
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent=var "Enter Track Event", delay=0, everyFrame=false)  // disabled
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    ApplyMusicCue(musicCue=var "Enter Music Cue", delayTime=0, transitionTime=0)
    on INIT → Enter
    on EXIT → Exit
  }

  var Enter Delay: float = 1
  var Enter Transition Time: float = 2
  var Enter Transition Time Init: float = 0
  var Exit Transition Time: float = 5
  var Dirtmouth: bool = false
  var Enter Track Event: string = "QUEEN"
  var Exit Track Event: string = ""
  var Enter Music Cue: object = Queen (sharedassets348.assets)
  var Enter Music Snapshot: object = Normal (resources.assets)
  var Exit Music Cue: object = <null>
  var Exit Music Snapshot: object = Silent (resources.assets)

  var Current Track: int = 0
  var Init: bool = false
}
