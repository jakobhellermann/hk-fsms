# Creep
# 1 location(s):
#   Deepnest_Spider_Town / Spider Royal Tall (1)

fsm Creep {
  start Idle
  on CAPTURE END → Capture End  // from any state

  state Idle {
    on FIRST STRUGGLE → Pause
  }

  state Pause {
    Wait(time=var "Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Creep
  }

  state Creep {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Tall Shuffle")
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "X Scale", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    RandomFloat(min=0.85, max=1.15, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlayInState(gameObject=Self, volume=1)
  }

  state Capture End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Pause: float = 5.3
  var Speed: float = 0.5

  var Pitch: float = 0.5
  var X Scale: float = 0
}
