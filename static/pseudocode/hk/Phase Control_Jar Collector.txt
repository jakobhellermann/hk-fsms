# Phase Control
# 3 location(s):
#   Ruins2_11_boss / Battle Scene/Jar Collector
#   GG_Collector / Battle Scene/Jar Collector
#   GG_Collector_V / Battle Scene/Jar Collector

fsm Phase Control {
  start Init

  state Init {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set
  }

  state Idle {
    on TOOK DAMAGE → Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Phase 2 HP", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on NEXT → Phase 2
  }

  state Phase 2 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Spawn Min", setValue=2, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Spawn Max", setValue=3, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Resummon Pause", setValue=0.35, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Spawn Recover Time", setValue=0.75, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Damage Control", variableName="Phase 2", setValue=true, everyFrame=false)
  }

  state Set {
    GetHP(target=Self, storeValue=var "Phase 2 HP")
    IntAdd(intVariable=var "Phase 2 HP", add=-350, everyFrame=false)
    on FINISHED → Idle
  }

  var HP: int = 0
  var Phase 2 HP: int = 300
}
