fsm Destroy {
  start Check

  state Check {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Inert
    on ENTER → Deactivate
  }

  state Inert {
  }

  state Deactivate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Self: gameObject = <null>
}
