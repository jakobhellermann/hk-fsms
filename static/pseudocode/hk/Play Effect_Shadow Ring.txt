# Play Effect
# 1 location(s):
#   resources.assets / Shadow Ring

fsm Play Effect {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetScale(gameObject=Self, vector=(unset), x=0.5, y=0.5, z=0.5, everyFrame=false, lateUpdate=false)
    on FINISHED → Grow
  }

  state Grow {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(1.4, 1.4, 1.4), time=0.4, delay=0, speed=(unset), easeType=easeOutQuad, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    ScaleTo(gameObject=Self, target=(1.4, 1.4, 1.4), duration=0.4, delay=0, curve=QuadraticOut)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=0.4, speed=(unset), delay=(unset), easeType=1, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Pos: vector3 = (0, -0.53, 0)

  var Colour: color = (0, 0, 0, 1)
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
