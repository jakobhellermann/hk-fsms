  start Destroy After Delay

  state Destroy After Delay {
    GetOwner(storeGameObject=var "Self")
    DestroyObject(gameObject=var "Self", delay=var "Delay Time", detachChildren=false)
  }

  var Delay Time: float = 2

  var Self: gameObject = <null>
}
