# FSM
# 4 location(s):
#   Room_Colosseum_Bronze / colosseum gold trophy/Debris/Particle Rock Large Pool 3
#   Room_Colosseum_Silver / colosseum gold trophy/Debris/Particle Rock Large Pool 3
#   Room_Colosseum_Gold / colosseum gold trophy/Debris/Particle Rock Large Pool 3
#   Room_Colosseum_Spectate / colosseum gold trophy/Debris/Particle Rock Large Pool 3

fsm FSM {
  start Initiate

  state Initiate {
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    RandomInt(min=1, max=4, storeResult=var "Chooser Int", inclusiveMax=true)
    IntSwitch(intVariable=var "Chooser Int", compareTo=[1, 2, 3, 4], sendEvent=[(none), (none), (none), (none)], everyFrame=false)
    on 1 → Sprite 1
    on 2 → Sprite 2
    on 3 → Sprite 3
    on 4 → Sprite 4
  }

  state Sprite 1 {
    Tk2dSpriteSetId(gameObject=Self, spriteID=0, ORSpriteName="rock_01_b", spriteCollection=var "")
    on FINISHED → Scale
  }

  state Sprite 2 {
    Tk2dSpriteSetId(gameObject=Self, spriteID=0, ORSpriteName="rock_02_b", spriteCollection=var "")
    on FINISHED → Scale
  }

  state Sprite 3 {
    Tk2dSpriteSetId(gameObject=Self, spriteID=0, ORSpriteName="rock_03_b", spriteCollection=var "")
    on FINISHED → Scale
  }

  state Sprite 4 {
    Tk2dSpriteSetId(gameObject=Self, spriteID=0, ORSpriteName="rock_04_b", spriteCollection=var "")
    on FINISHED → Scale
  }

  state Scale {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.019, space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=1, max=2, storeResult=var "Chooser Float")
    SetScale(gameObject=Self, vector=(unset), x=var "Chooser Float", y=var "Chooser Float", z=(unset), everyFrame=false, lateUpdate=false)
    RandomInt(min=1, max=3, storeResult=var "Chooser Int", inclusiveMax=true)
    IntSwitch(intVariable=var "Chooser Int", compareTo=[1, 2, 3, 4], sendEvent=[(none), (none), (none), (none)], everyFrame=false)
    on 1 → Wait
    on 3 → Black
  }

  state Wait {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Spin
  }

  state Spin {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "X Speed", everyFrame=false)
  }

  state Black {
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=-0.05, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Wait
  }

  var Chooser Float: float = 0
  var X Speed: float = 0
  var Chooser Int: int = 0
}
