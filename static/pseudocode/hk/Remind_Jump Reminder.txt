# Remind
# 1 location(s):
#   Tutorial_01 / _Props/Tute Door 5/Active:1/Jump Reminder

fsm Remind {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Check
  }

  state State 1 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REMINDER JUMP", delay=0, everyFrame=false)
  }

  state Check {
    Wait(time=4, finishEvent=→"DISPLAY", realTime=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Idle
    on DISPLAY → State 1
  }
}
