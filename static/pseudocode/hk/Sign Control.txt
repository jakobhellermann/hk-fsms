# Sign Control
# 1 location(s):
#   Fungus3_35 / Bank Sign

fsm Sign Control {
  start Init
  on BANK FALLEN → Fallen  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on BANK FALL → Fall Ready
  }

  state Fall Ready {
    SendTrigger2DEvent(eventTarget=Self, trigger=OnTriggerEnter2D, collideTag="", collideLayer=(unset), sendEvent=→"CLOSE", storeCollider=var "")
    on CLOSE → Fall
  }

  state Fall {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sign Fall", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=banker_sign_move_initial (sharedassets218.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Fall 2
  }

  state Land {
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=banker_sign_impact (sharedassets218.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  state Fallen {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sign Fallen")
  }

  state Fall 2 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=banker_sign_snap_off (sharedassets218.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Land
  }

  var Dust: gameObject = Bank Sign/Dust
  var Self: gameObject = <null>
}
