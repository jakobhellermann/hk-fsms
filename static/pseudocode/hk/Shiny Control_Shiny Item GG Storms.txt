# Shiny Control
# 1 location(s):
#   GG_Land_of_Storms / Shiny Item GG Storms

fsm Shiny Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetHero(storeResult=var "Hero")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="Inspect Region", storeResult=var "Inspect Region")
    FindChild(gameObject=Self, childName="Appear Trail", storeResult=var "Trail")
    on FINISHED → PD Bool?
    on ACTIVATE → Destroy
  }

  state PD Bool? {
    StringCompare(stringVariable=var "PD Bool Name", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool Name", isTrue=→"COLLECTED", isFalse=(none))
    on FINISHED → Idle
    on COLLECTED → Destroy
  }

  state Idle {
    SetFsmBool(gameObject=var "Inspect Region", fsmName="inspect", variableName="Inspectable", setValue=true, everyFrame=false)
    on START INSPECT → Hero Down
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Hero Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 1")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Big Get Flash
  }

  state Hero Up {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="CheckCharmAchievements", parameters=[], storeResult=(unset var))
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    on FINISHED → Fade Out
  }

  state Journal {
    SetPlayerDataBool(boolName="killedGodseekerMask", value=true)
    SetPlayerDataBool(boolName="newDataGodseekerMask", value=true)
    SetPlayerDataInt(intName="killsGodseekerMask", value=0)
    GGCheckIsBossRushMode(trueEvent=→"FINISHED", falseEvent=(none))
    CreateObject(gameObject=Journal Update Msg (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "UI Msg", networkInstantiate=false, networkGroup=0)
    on FINISHED → Hero Up
  }

  state Big Get Flash {
    CreateObject(gameObject=Item Get Effect (sharedassets7.assets), spawnPoint=var "Hero", position=(0, -0.76, -1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=shiny_item_pickup (resources.assets))
    SetSpriteRenderer(gameObject=var "Self", active=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 2")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Journal
  }

  state Fade Out {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=dream_enter (resources.assets))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=5.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → New Scene
  }

  state New Scene {
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    BeginSceneTransition(sceneName="GG_Atrium_Roof", entryGateName="door_Land_of_Storms_return", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  var PD Bool Name: string = ""

  var Activated: bool = false
  var Hero: gameObject = <null>
  var Inspect Region: gameObject = <null>
  var Self: gameObject = <null>
  var Trail: gameObject = Shiny Item GG Storms/Appear Trail
  var UI Msg: gameObject = <null>
}
