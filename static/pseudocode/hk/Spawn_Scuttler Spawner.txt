# Spawn
# 3 location(s):
#   Abyss_19 / Scuttler Spawner
#   Dream_03_Infected_Knight / Scuttler Spawner
#   GG_Broken_Vessel / Scuttler Spawner

fsm Spawn {
  start Init
  on SCUTTLER SPAWN END → Idle  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on SCUTTLER SPAWN START → Spawn
  }

  state Spawn {
    RandomFloat(min=15.3, max=36.4, storeResult=var "Pos X")
    SetVector3XYZ(vector3Variable=var "Pos Vector", vector3Value=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Orange Scuttler IK (sharedassets343.assets), spawnPoint=<null>, position=var "Pos Vector", rotation=(unset), storeObject=var "Scuttler")
    RandomFloat(min=2, max=8, storeResult=var "Jump")
    SetVelocity2d(gameObject=var "Scuttler", vector=(unset), x=(unset), y=var "Jump", everyFrame=false)
    on FINISHED → Pause
  }

  state Pause {
    WaitRandom(timeMin=0.05, timeMax=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  var Jump: float = 0
  var Pos X: float = 0
  var Pos Y: float = 27.7
  var Pos Vector: vector3 = (0, 0, 0)
  var Scuttler: gameObject = <null>
}
