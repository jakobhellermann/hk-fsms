# Inflater
# 1 location(s):
#   Waterways_09 / Battle Scene/Inflater (7)

fsm Inflater {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    IdleBuzz(gameObject=Self, waitMin=0.5, waitMax=1, speedMax=2, accelerationMax=10, roamingRange=3)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Small TurnToIdle", everyFrame=true, pauseBetweenTurns=false, pauseTime=1)
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    ReceivedDamage(collideTag="", sendEvent=→"DAMAGED", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"INFLATE", storeCollider=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Small Idle")
    on ALERT → Alert
    on INFLATE → Inflate
    on DAMAGED → Causes damage?
  }

  state Alert {
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=3, accelerationForce=3, offsetX=0, offsetY=0)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Small TurnToIdle", everyFrame=true, pauseBetweenTurns=false, pauseTime=0.5)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"INFLATE", storeCollider=var "")
    ReceivedDamage(collideTag="", sendEvent=→"DAMAGED", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    BoolTest(boolVariable=var "Alert Range", isTrue=(none), isFalse=→"UNALERT", everyFrame=true)
    on DAMAGED → Causes damage?
    on INFLATE → Inflate
    on UNALERT → Idle
  }

  state Causes damage? {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage Dealt", everyFrame=false)
    IntCompare(integer1=var "Damage Dealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on FINISHED → Check Direction
    on CANCEL → Idle
  }

  state Check Direction {
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on UP → Hit Up
    on DOWN → Hit Down
    on LEFT → Hit Left
    on RIGHT → Hit Right
  }

  state Hit Up {
    AddForce2dAsAngle(gameObject=Self, atPosition=(unset), angle=90, speed=var "Hit Force", maxSpeed=0, maxSpeedX=0, maxSpeedY=0, everyFrame=false)
    BoolTest(boolVariable=var "Inflated", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    RandomFloat(min=-50, max=50, storeResult=var "Torque")
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Torque", everyFrame=false)
    on FINISHED → Inflated?
  }

  state Hit Down {
    AddForce2dAsAngle(gameObject=Self, atPosition=(unset), angle=270, speed=var "Hit Force", maxSpeed=0, maxSpeedX=0, maxSpeedY=0, everyFrame=false)
    BoolTest(boolVariable=var "Inflated", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    RandomFloat(min=-80, max=80, storeResult=var "Torque")
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Torque", everyFrame=false)
    on FINISHED → Inflated?
  }

  state Hit Left {
    AddForce2dAsAngle(gameObject=Self, atPosition=(unset), angle=180, speed=var "Hit Force", maxSpeed=0, maxSpeedX=0, maxSpeedY=0, everyFrame=false)
    BoolTest(boolVariable=var "Inflated", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=100, everyFrame=false)
    on FINISHED → Inflated?
  }

  state Hit Right {
    AddForce2dAsAngle(gameObject=Self, atPosition=(unset), angle=0, speed=var "Hit Force", maxSpeed=0, maxSpeedX=0, maxSpeedY=0, everyFrame=false)
    BoolTest(boolVariable=var "Inflated", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=-100, everyFrame=false)
    on FINISHED → Inflated?
  }

  state Inflated? {
    BoolTest(boolVariable=var "Inflated", isTrue=→"FINISHED", isFalse=→"INFLATE", everyFrame=false)
    on FINISHED → Inflated
    on INFLATE → Inflate
  }

  state Inflate {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=inflater_inflate (sharedassets349.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetAudioClip(gameObject=Self, audioClip=inflator_big_moving_loop (sharedassets349.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Enemy Inflater", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFloatValue(floatVariable=var "Unalert Timer", floatValue=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Inflate", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Inflated", boolValue=true, everyFrame=false)
    EaseFloat(fromValue=0.43, toValue=2.2, floatVariable=var "Collider Radius", time=0.25, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)
    SetCircleColliderRadius(gameObject=Self, radius=var "Collider Radius", everyFrame=true)
    SetCircleColliderRadius(gameObject=var "Alert range Obj", radius=10.32, everyFrame=false)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=8, accelerationForce=4, offsetX=0, offsetY=0)
    on FINISHED → Increase Collider
  }

  state Increase Collider {
    SetCircleColliderRadius(gameObject=Self, radius=2.2, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Inflate Idle")
    ActivateGameObject(gameObject=var "Bouncer", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetBounceFactor(1))
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetBounceAnimation(true))
    SendEventByName(eventTarget=Self, sendEvent="FINISHED", delay=0, everyFrame=false)
    on FINISHED → Inflated
  }

  state Inflated {
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=8, accelerationForce=4, offsetX=0, offsetY=0)
    ReceivedDamage(collideTag="", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Inflate TurnToIdle", everyFrame=true, pauseBetweenTurns=true, pauseTime=1)
    RotateTo(gameObject=Self, targetAngle=0, speed=var "Rotate Back Speed")
    SetBoolValue(boolVariable=var "Unalert", boolValue=var "Alert Range", everyFrame=true)
    BoolFlipEveryFrame(boolVariable=var "Unalert", everyFrame=true)
    FloatAddV2(floatVariable=var "Unalert Timer", add=1, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=var "Unalert")
    FloatCompare(float1=var "Unalert Timer", float2=8, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=true)
    on TAKE DAMAGE → Causes damage? 2
    on UNALERT → Deflate
  }

  state Causes damage? 2 {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage Dealt", everyFrame=false)
    IntCompare(integer1=var "Damage Dealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on FINISHED → Check Direction
    on CANCEL → Inflated
  }

  state Deflate {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Deflate", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetAudioClip(gameObject=Self, audioClip=inflater_idle_loop (sharedassets349.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Enemy Inflater", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleColliderRadius(gameObject=Self, radius=0.43, everyFrame=false)
    ActivateGameObject(gameObject=var "Bouncer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetBounceFactor(0.4))
    SetCircleColliderRadius(gameObject=var "Alert range Obj", radius=3.59, everyFrame=false)
    SetBoolValue(boolVariable=var "Inflated", boolValue=false, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetBounceAnimation(false))
    on FINISHED → Idle
  }

  var Rotate Back Speed: float = 40

  var Attack Direction: float = 0
  var Collider Radius: float = 0.43
  var Hit Force: float = 1500
  var Torque: float = 0
  var Unalert Timer: float = 0
  var Damage Dealt: int = 0
  var Alert Range: bool = false
  var Inflated: bool = false
  var Unalert: bool = false
  var Alert range Obj: gameObject = Battle Scene/Inflater (7)/Alert Range
  var Bouncer: gameObject = Battle Scene/Inflater (7)/Bouncer
  var Damager: gameObject = <null>
  var Enemy Inflater: gameObject = Battle Scene/Inflater (7)/Enemy Inflater
  var Self: gameObject = <null>
}
