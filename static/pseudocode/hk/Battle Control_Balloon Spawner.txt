# Battle Control
# 1 location(s):
#   GG_Mage_Knight_V / Balloon Spawner

fsm Battle Control {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Balloons", storeResult=var "Balloons")
    FindChild(gameObject=Self, childName="Mage Knight", storeResult=var "Mage Knight")
    on FINISHED → Battling
  }

  state End {
    SendEventToRegister(eventName="KILL ALL ENEMIES")
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Battling {
    RandomWait(min=4, max=4, finishEvent=→"SPAWN", realTime=false)
    GameObjectIsNull(gameObject=var "Mage Knight", isNull=→"END", isNotNull=(none), storeResult=(unset), everyFrame=true)
    on END → End
    on SPAWN → Spawn
  }

  state Spawn {
    GetRandomChild(gameObject=var "Balloons", storeResult=var "Balloon")
    ActivateGameObject(gameObject=var "Balloon", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Battling
  }

  var Battle Enemies: int = 0
  var Activated: bool = false
  var Balloon: gameObject = <null>
  var Balloons: gameObject = <null>
  var Camera Lock: gameObject = <null>
  var Mage Knight: gameObject = <null>
  var Self: gameObject = <null>
}
