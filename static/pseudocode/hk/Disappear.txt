# Disappear
# 1 location(s):
#   Abyss_06_Core / floor_closed

fsm Disappear {
  start State 3

  state State 1 {
    GetPlayerDataBool(boolName="equippedCharm_36", storeValue=var "Equipped CHarm")
    GetPlayerDataBool(boolName="openedBlackEggPath", storeValue=var "Gate Opened")
    BoolAllTrue(boolVariables=[var "Equipped CHarm", var "Gate Opened"], sendEvent=→"BLACK FLOOR OPEN", storeResult=(unset), everyFrame=false)
    on BLACK FLOOR OPEN → State 2
  }

  state State 2 {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state State 3 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State 1
  }

  var Equipped CHarm: bool = false
  var Gate Opened: bool = false
}
