# Set
# 1 location(s):
#   Deepnest_30 / Quirrel Spa/Set Encounter

fsm Set {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Set
  }

  state Set {
    SetPlayerDataBool(boolName="quirrelSpaEncountered", value=true)
  }

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Detect
  }
}
