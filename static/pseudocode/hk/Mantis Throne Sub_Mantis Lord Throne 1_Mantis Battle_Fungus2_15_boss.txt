# Mantis Throne Sub
# 6 location(s):
#   Fungus2_15_boss / Mantis Battle/Mantis Lord Throne 1
#   Fungus2_15_boss / Mantis Battle/Mantis Lord Throne 3
#   Fungus2_15_boss_defeated / Mantis Lord Throne 1
#   Fungus2_15_boss_defeated / Mantis Lord Throne 3
#   GG_Mantis_Lords / Mantis Battle/Mantis Lord Throne 1
#   GG_Mantis_Lords / Mantis Battle/Mantis Lord Throne 3

// uses template: mlord_throne_sub
fsm Mantis Throne Sub {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on MLORD CHALLENGE → Stand
  }

  state Stand {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Stand")
    on MLORD START SUB → Leave
  }

  state Leave {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Throne Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_misc_jump_2)
    on FINISHED → Fighting
  }

  state Fighting {
    SetMeshRenderer(gameObject=Self, active=false)
    on RETURN → Return Pause
  }

  state Return Pause {
    Wait(time=2.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return
  }

  state Return {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_land_on_ground)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Wounded")
  }
}
