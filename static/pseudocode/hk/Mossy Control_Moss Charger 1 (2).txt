# Mossy Control
# 8 location(s):
#   Fungus1_10 / Moss Charger 1 (2)
#   Fungus1_10 / Moss Charger
#   Fungus1_10 / Moss Charger (1)
#   Fungus1_10 / Moss Charger 1 (1)
#   Fungus1_17 / Moss Charger
#   Fungus1_21 / Moss Charger (1)
#   Fungus1_21 / Moss Charger (2)
#   Fungus1_21 / Moss Charger

fsm Mossy Control {
  start Init Pause
  on ZERO HP → Detach  // from any state
  on BLOCKED HIT → Line Loop  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "Camera", gameObject=var "CameraParent", everyFrame=false)
    FindChild(gameObject=Self, childName="Grass Burst", storeResult=var "Grass Burst")
    FindChild(gameObject=Self, childName="Grass Explode 1", storeResult=var "G Explode 1")
    FindChild(gameObject=Self, childName="Charge Grass", storeResult=var "Charge Grass")
    FindChild(gameObject=Self, childName="Grass Dig", storeResult=var "Grass Dig")
    FindChild(gameObject=Self, childName="Grass Burst Down", storeResult=var "Grass Burst Down")
    FindChild(gameObject=Self, childName="Grass Submerge", storeResult=var "Grass Submerge")
    FindChild(gameObject=Self, childName="Range Grass", storeResult=var "Range Grass")
    GetHero(storeResult=var "Hero")
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    FindChild(gameObject=Self, childName="Attack Range", storeResult=var "Attack Range")
    FindAlertRange(target=Self, storeResult=var "Attack Range Obj", childName="Attack Range")
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    BoundsBoxCollider(gameObject1=var "Attack Range", scaleVector2=(unset), scaleX=var "Attack Range Length", scaleY=(unset), everyFrame=false)
    FloatDivide(floatVariable=var "Attack Range Length", divideBy=2, everyFrame=false)
    SetParent(gameObject=var "Attack Range", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Range Grass", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetFloatValue(floatVariable=var "X Min", floatValue=var "Start X", everyFrame=false)
    FloatOperator(float1=var "X Min", float2=var "Attack Range Length", operation=Subtract, storeResult=var "X Min", everyFrame=false)
    SetFloatValue(floatVariable=var "X Max", floatValue=var "Start X", everyFrame=false)
    FloatOperator(float1=var "X Max", float2=var "Attack Range Length", operation=Add, storeResult=var "X Max", everyFrame=false)
    FloatOperator(float1=var "X Min", float2=2, operation=Add, storeResult=var "X Min", everyFrame=false)
    FloatOperator(float1=var "X Max", float2=2, operation=Subtract, storeResult=var "X Max", everyFrame=false)
    on FINISHED → Hidden
  }

  state Hidden {
    GetFsmBool(gameObject=var "Attack Range", fsmName="Attack Range Detect", variableName="In Attack Range", storeValue=var "In Attack Range", everyFrame=true)  // disabled
    CheckAlertRange(alertRange=var "Attack Range Obj", storeResult=var "In Attack Range", everyFrame=true)
    BoolTest(boolVariable=var "In Attack Range", isTrue=→"IN RANGE", isFalse=(none), everyFrame=true)
    on IN RANGE → Emerge Pause
  }

  state Emerge Pause {
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero Beyond?
  }

  state Left or Right? {
    SendRandomEvent(events=[→"RIGHT", →"LEFT"], weights=[1, 1], delay=0)
    on LEFT → Emerge Left
    on RIGHT → Emerge Right
  }

  state Emerge Right {
    SetVector2XY(vector2Variable=var "RayDown X", vector2Value=(unset), x=-6.5, y=-0.5, everyFrame=false)
    SetVector2XY(vector2Variable=var "RayForward Direction", vector2Value=(unset), x=-1, y=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    RandomFloat(min=14, max=14, storeResult=var "Chooser")
    SetFloatValue(floatVariable=var "Appear X", floatValue=var "Hero X", everyFrame=false)
    FloatOperator(float1=var "Appear X", float2=var "Chooser", operation=Add, storeResult=var "Appear X", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Current Charge Speed", floatValue=var "Charge Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Current Charge Speed", multiplyBy=-1, everyFrame=false)
    FloatCompare(float1=var "Appear X", float2=var "X Max", tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"LEFT", everyFrame=false)
    on FINISHED → Emerge
    on LEFT → Pause
  }

  state Emerge Left {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "RayForward Direction", vector2Value=(unset), x=1, y=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "RayDown X", vector2Value=(unset), x=6.5, y=-0.5, everyFrame=false)
    RandomFloat(min=14, max=14, storeResult=var "Chooser")
    SetFloatValue(floatVariable=var "Appear X", floatValue=var "Hero X", everyFrame=false)
    FloatOperator(float1=var "Appear X", float2=var "Chooser", operation=Subtract, storeResult=var "Appear X", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Current Charge Speed", floatValue=var "Charge Speed", everyFrame=false)
    FloatCompare(float1=var "Appear X", float2=var "X Min", tolerance=0, equal=→"FINISHED", lessThan=→"RIGHT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Emerge
    on RIGHT → Pause 2
  }

  state Emerge {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_charger_emerge (sharedassets139.assets), pitchMin=0.85, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Camera"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Emerge Speed", floatValue=var "Current Charge Speed", everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(false))
    FloatMultiply(floatVariable=var "Emerge Speed", multiplyBy=0.25, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Appear X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    StopParticleEmitter(gameObject=var "Range Grass")
    PlayParticleEmitter(gameObject=var "Grass Burst", emit=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Emerge Speed", y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Charge
  }

  state Charge {
    SetAudioClip(gameObject=Self, audioClip=moss_charger_charging (sharedassets139.assets))
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    PlayParticleEmitter(gameObject=var "Charge Grass", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Charge")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Current Charge Speed", y=(unset), everyFrame=false)
    RayCast2d(fromGameObject=Self, fromPosition=(0, -0.5), direction=var "RayForward Direction", space=1, distance=5.5, minDepth=(unset), maxDepth=(unset), hitEvent=→"SUBMERGE", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=2, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=Self, fromPosition=var "RayDown X", direction=(0, -1), space=1, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "RayDown1", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=2, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    BoolTest(boolVariable=var "RayDown1", isTrue=(none), isFalse=→"SUBMERGE", everyFrame=true)
    BoolTest(boolVariable=var "RayDown1", isTrue=(none), isFalse=→"SUBMERGE", everyFrame=true)
    on SUBMERGE → Submerge
    on TAKE DAMAGE → Line Loop
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Emerge Left
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Emerge Right
  }

  state Submerge {
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_charger_return (sharedassets139.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    StopParticleEmitter(gameObject=var "Charge Grass")
    Decelerate(gameObject=Self, deceleration=0.7)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Disappear", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Submerge Grass effect
  }

  state Submerge CD {
    SetCollider(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Grass Submerge")
    StopParticleEmitter(gameObject=var "Grass Dig")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Play Range
  }

  state Burst {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_charger_leafy_explode (sharedassets139.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="INVULN", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "G Explode 1", emit=0)
    StopParticleEmitter(gameObject=var "Charge Grass")
    StopParticleEmitter(gameObject=var "Charge Grass")
    SetGravity2dScale(gameObject=Self, gravityScale=1.5)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun")
    GetFsmFloat(gameObject=Self, fsmName="health_manager_enemy", variableName="Attack Direction", storeValue=var "Attack Direction", everyFrame=false)  // disabled
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)  // disabled
    CallMethodProper(gameObject=Self, behaviour="HealthManager", methodName="GetAttackDirection", parameters=[], storeResult=var "Cardinal Direction")
    IntSwitch(intVariable=var "Cardinal Direction", compareTo=[0, 1, 2, 3], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Fly Left
    on RIGHT → Fly Right
    on UP → FlyUp
    on DOWN → Fly Down
  }

  state Hero Beyond? {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "X Max", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "X Min", tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Hidden
    on FINISHED → Left or Right?
  }

  state Fly Left {
    SetVelocityAsAngle(gameObject=Self, angle=110, speed=18, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Air
  }

  state Fly Right {
    SetVelocityAsAngle(gameObject=Self, angle=70, speed=18, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Air
  }

  state FlyUp {
    SetVelocityAsAngle(gameObject=Self, angle=90, speed=20, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=-100, max=-80, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Air
  }

  state Fly Down {
    SetVelocityAsAngle(gameObject=Self, angle=270, speed=10, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=100, max=80, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → In Air
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Get Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Direction
  }

  state Run L {
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AccelerateVelocity(gameObject=Self, xAccel=-0.5, yAccel=(unset), xMaxSpeed=10, yMaxSpeed=(unset))
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    RayCast2d(fromGameObject=Self, fromPosition=(unset), direction=(-1, 0), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"SUBMERGE", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=Self, fromPosition=(-3, 0), direction=(0, -1), space=1, distance=1.3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Ground Ahead", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    BoolTest(boolVariable=var "Ground Ahead", isTrue=(none), isFalse=→"SUBMERGE", everyFrame=true)
    Wait(time=1, finishEvent=→"SUBMERGE", realTime=false)
    on LEFT → Run R
    on SUBMERGE → On Ground?
  }

  state Run R {
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AccelerateVelocity(gameObject=Self, xAccel=0.5, yAccel=(unset), xMaxSpeed=10, yMaxSpeed=(unset))
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    RayCast2d(fromGameObject=Self, fromPosition=(unset), direction=(1, 0), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"SUBMERGE", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=Self, fromPosition=(3, 0), direction=(0, -1), space=1, distance=1.3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Ground Ahead", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    BoolTest(boolVariable=var "Ground Ahead", isTrue=(none), isFalse=→"SUBMERGE", everyFrame=true)
    Wait(time=1, finishEvent=→"SUBMERGE", realTime=false)
    on RIGHT → Run L
    on SUBMERGE → On Ground?
  }

  state Direction {
    SetAudioClip(gameObject=Self, audioClip=moss_charger_small_run (sharedassets139.assets))
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="TurnRun")
    Tk2dPlayFrame(gameObject=Self, frame=2)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on LEFT → Run R
    on RIGHT → Run L
  }

  state In Air {
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=(unset))
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"DOWN", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"DOWN", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetCollider(gameObject=Self, active=true)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetGravity2dScale(gameObject=Self, gravityScale=1.5)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun")
    StopParticleEmitter(gameObject=var "Grass Dig")
    on DOWN → Land
  }

  state Land {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Get Up
  }

  state Dig Start {
    SetBoolValue(boolVariable=var "Dig Check", boolValue=true, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Dig Check Obj", fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Dig Check", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CANCEL RECOIL", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    Decelerate(gameObject=Self, deceleration=0.4)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Escape", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetFsmBool(gameObject=Self, fsmName="health_manager_enemy", variableName="Invincible", setValue=false, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(true))
    BoolTest(boolVariable=var "Dig Check", isTrue=(none), isFalse=→"FALL", everyFrame=true)
    on FINISHED → Dig
    on FALL → In Air
  }

  state On Ground? {
    AudioStop(gameObject=Self)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=var "Grounded2", leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=var "Grounded", leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
    BoolAnyTrue(boolVariables=[var "Grounded", var "Grounded2"], sendEvent=→"DOWN", storeResult=(unset), everyFrame=true)  // disabled
    on DOWN → Dig Start
    on FINISHED → Dig Start
  }

  state Dig {
    SetBoolValue(boolVariable=var "Dig Check", boolValue=true, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Dig Check Obj", fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Dig Check", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    PlayParticleEmitter(gameObject=var "Grass Dig", emit=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[grub_burrow (sharedassets39.assets)], weights=[1], pitchMin=1.1, pitchMax=1)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    BoolTest(boolVariable=var "Dig Check", isTrue=(none), isFalse=→"FALL", everyFrame=true)
    on FINISHED → Submerge CD
    on FALL → In Air
  }

  state Play Range {
    PlayParticleEmitter(gameObject=var "Range Grass", emit=0)
    on FINISHED → Hidden
  }

  state Detach {
    StopParticleEmitter(gameObject=var "Range Grass")
    StopParticleEmitter(gameObject=var "Grass Dig")
    DetachChildren(gameObject=Self)
  }

  state Submerge Grass effect {
    PlayParticleEmitter(gameObject=var "Grass Submerge", emit=0)
    PlayParticleEmitter(gameObject=var "Grass Burst Down", emit=0)  // disabled
    Decelerate(gameObject=Self, deceleration=0.7)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Submerge CD
  }

  state Line Loop {
    GetOwner(storeGameObject=var "Self")
    IntAdd(intVariable=var "Looper", add=1, everyFrame=false)
    CreateObject(gameObject=Impact Lines (sharedassets27.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)
    Tk2dSpriteSetColor(gameObject=var "Hit Effect", color=(0.098039225, 0.20000002, 0.2627451, 1), everyframe=false)
    RandomFloat(min=2, max=3, storeResult=var "Chooser")
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=var "Chooser", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=0, max=360, storeResult=var "Chooser")
    Rotate(gameObject=var "Hit Effect", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    IntCompare(integer1=var "Looper", integer2=11, equal=(none), lessThan=(none), greaterThan=→"LOOP COMPLETE", everyFrame=false)
    on FINISHED → State 2
