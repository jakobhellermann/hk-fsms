# Droplet Control
# 1 location(s):
#   resources.assets / Spatter Brown

fsm Droplet Control {
  start Idle

  state Idle {
    RandomFloat(min=0.75, max=1.5, storeResult=var "Randomiser")
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=1.4, stretchMinX=0.6, stretchMaxY=1.75, scaleModifier=var "Randomiser", everyFrame=true)
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"COLLISION ENTER 2D", storeCollider=var "", storeForce=(unset))
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetSpriteRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Splash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on COLLISION ENTER 2D → Check Normal
  }

  state Impact {
    RandomFloat(min=1.5, max=2, storeResult=var "Randomiser")  // disabled
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")  // disabled
    FindChild(gameObject=Self, childName="Splash", storeResult=var "Splash")
    ActivateGameObject(gameObject=var "Splash", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    DestroySelf(detachChildren=false)  // disabled
    RecycleSelf()
  }

  state Check Normal {
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=(unset), contactNormal=var "Contact Normal", shapeCount=(unset), physics2dMaterialName="")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    GetVector3XYZ(vector3Variable=var "Contact Normal", storeX=var "X Normal", storeY=var "Y Normal", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Y Normal", float2=1, tolerance=0.2, equal=→"DOWN", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Y Normal", float2=-1, tolerance=0.2, equal=→"UP", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "X Normal", float2=1, tolerance=0.2, equal=→"LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "X Normal", float2=-1, tolerance=0.2, equal=→"RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    Wait(time=0, finishEvent=→"WAIT", realTime=false)
    on DOWN → Hit Down
    on RIGHT → Hit Right
    on WAIT → Impact
    on UP → Hit Up
    on LEFT → Hit Left
  }

  state Hit Down {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=270, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  state Hit Up {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=90, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  state Hit Right {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  state Hit Left {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=180, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Impact
  }

  var Randomiser: float = 0
  var X Normal: float = 0
  var Y Normal: float = 0
  var Contact Normal: vector3 = (0, 0, 0)
  var Splash: gameObject = <null>
}
