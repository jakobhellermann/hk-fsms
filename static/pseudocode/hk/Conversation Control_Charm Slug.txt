# Conversation Control
# 1 location(s):
#   Room_Charm_Shop / Charm Slug

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=CountCharms())
    on FINISHED → Met?
  }

  state End {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Slug Anim
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choice
  }

  state Meet {
    SetPlayerDataBool(boolName="metCharmSlug", value=true)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_MEET_1", "Charm Slug"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Salubra_Talk (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Meet2
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"FINISHED", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shop End")
    on FINISHED → Box Down
  }

  state To Shop {
    SetBoxCollider2DSize(gameObject1=Self, width=2.339492, height=0.2408066, offsetX=2.17705, offsetY=-2.8)
    ActivateGameObject(gameObject=var "Shop Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DO SHOP OPEN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO CANCEL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Slug Anim {
    AudioStop(gameObject=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk Start")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"FINISHED", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shop Start")
    on FINISHED → Title
  }

  state Met? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metCharmSlug", isTrue=→"FINISHED", isFalse=(none))
    SetBoxCollider2DSize(gameObject1=Self, width=6.6, height=0.25, offsetX=0, offsetY=-2.8)
    FindGameObject(objectName="Shop Region", withTag="Untagged", store=var "Shop Region")
    ActivateGameObject(gameObject=var "Shop Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Choice {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metCharmSlug", isTrue=(none), isFalse=→"MEET")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=CountCharms())
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmsOwned", storeValue=var "Charms Owned")
    IntCompare(integer1=var "Charms Owned", integer2=1, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="salubraConvoCombo", isTrue=(none), isFalse=→"COMBO")
    IntCompare(integer1=var "Charms Owned", integer2=6, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="salubraConvoOvercharm", isTrue=(none), isFalse=→"OVERCHARM")
    IntCompare(integer1=var "Charms Owned", integer2=12, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="salubraConvoTruth", isTrue=→"FINISHED", isFalse=→"TRUTH")
    on MEET → Meet
    on FINISHED → Generic
    on COMBO → Combo
    on OVERCHARM → Overcharm
    on TRUTH → Truth
  }

  state Meet2 {
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_MEET_2", "Charm Slug"], storeResult=(unset var))
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="CHARM_SLUG", everyFrame=false)
    on CONVO_FINISH → Got Charm?
  }

  state Title {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metCharmSlug", isTrue=(none), isFalse=→"FINISHED")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="CHARM_SLUG", everyFrame=false)
    on FINISHED → Box Up
  }

  state Got Charm? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasCharm", isTrue=→"HAS CHARM", isFalse=→"NO CHARM")
    on HAS CHARM → Meet 3A
    on NO CHARM → Meet 3B
  }

  state Meet 3A {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_MEET_3A", "Charm Slug"], storeResult=(unset var))
    on CONVO_FINISH → To Shop
  }

  state Meet 3B {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_MEET_3B", "Charm Slug"], storeResult=(unset var))
    on CONVO_FINISH → To Shop
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Generic {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_REPEAT", "Charm Slug"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Salubra_Talk_04 (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Combo {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_COMBO", "Charm Slug"], storeResult=(unset var))
    SetPlayerDataBool(boolName="salubraConvoCombo", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Salubra_Talk (sharedassets10.assets), Salubra_Talk_02 (sharedassets10.assets), Salubra_Talk_03 (sharedassets10.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Overcharm {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_OVERCHARM", "Charm Slug"], storeResult=(unset var))
    SetPlayerDataBool(boolName="salubraConvoOvercharm", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Salubra_Talk (sharedassets10.assets), Salubra_Talk_02 (sharedassets10.assets), Salubra_Talk_03 (sharedassets10.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Truth {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_TRUTH", "Charm Slug"], storeResult=(unset var))
    SetPlayerDataBool(boolName="salubraConvoTruth", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Salubra_Talk (sharedassets10.assets), Salubra_Talk_02 (sharedassets10.assets), Salubra_Talk_03 (sharedassets10.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var Charms Owned: int = 0
  var Elderbug State: int = 0
  var History: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var mapCrossroads: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Shop Region: gameObject = <null>
  var Voice Player: gameObject = <null>
}
