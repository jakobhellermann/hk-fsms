# Detect
# 1 location(s):
#   sharedassets170.assets / Spatter White/Acid Detector

fsm Detect {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Acid", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Send Msg
  }

  state Send Msg {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="ACID", delay=0, everyFrame=false)
  }

  var Parent: gameObject = <null>
}
