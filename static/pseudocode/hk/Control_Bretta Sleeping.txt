# Control
# 1 location(s):
#   Room_Bretta / Bretta Sleeping

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brettaRescued", isTrue=(none), isFalse=→"DESTROY")
    GetPlayerDataInt(gameObject=var "GameManager", intName="brettaPosition", storeValue=var "Bretta Position")
    IntCompare(integer1=var "Bretta Position", integer2=1, equal=(none), lessThan=→"DESTROY", greaterThan=→"DESTROY", everyFrame=false)
    on DESTROY → Destroy
    on FINISHED → Set Seen
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Set Seen {
    SetPlayerDataBool(boolName="brettaSeenBed", value=true)
  }

  var Bretta Position: int = 0
}
