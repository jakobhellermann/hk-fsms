# Run Effects
# 1 location(s):
#   resources.assets / Run Effects

fsm Run Effects {
  start Check Enviro

  state Check Enviro {
    SetParticleEmission(gameObject=var "Bone", emission=false)
    SetParticleEmission(gameObject=var "Dust", emission=false)
    SetParticleEmission(gameObject=var "Wet", emission=false)
    SetParticleEmission(gameObject=var "Grass", emission=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="environmentType", storeValue=var "Environment Type")
    IntSwitch(intVariable=var "Environment Type", compareTo=[0, 1, 2, 3, 4, 6], sendEvent=[→"DUST", →"GRASS", →"BONE", →"SPA", →"METAL", →"WET"], everyFrame=false)
    on DUST → Dust
    on GRASS → Grass
    on BONE → Bone
    on SPA → Wet
    on METAL → Dust
    on RUN STOP → End
    on WET → Wet
    on FINISHED → Dust
  }

  state Dust {
    FindChild(gameObject=Self, childName="Dust", storeResult=var "Particle")
    SetGameObject(variable=var "Particle 2", gameObject=<null>, everyFrame=false)
    on FINISHED → Play
  }

  state Play {
    PlayParticleEmitter(gameObject=var "Particle", emit=0)
    PlayParticleEmitter(gameObject=var "Particle 2", emit=0)
    SetParticleEmission(gameObject=var "Particle", emission=true)
    SetParticleEmission(gameObject=var "Particle 2", emission=true)
    on RUN STOP → End
    on ENVIRO UPDATE → Check Enviro
  }

  state End {
    StopParticleEmitter(gameObject=var "Particle")
    StopParticleEmitter(gameObject=var "Particle 2")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Grass {
    FindChild(gameObject=Self, childName="Grass", storeResult=var "Particle")
    SetGameObject(variable=var "Particle 2", gameObject=<null>, everyFrame=false)
    on FINISHED → Play
  }

  state Bone {
    FindChild(gameObject=Self, childName="Dust", storeResult=var "Particle")
    FindChild(gameObject=Self, childName="Bone", storeResult=var "Particle 2")
    on FINISHED → Play
  }

  state Recycle {
    RecycleSelf()
  }

  state Wet {
    FindChild(gameObject=Self, childName="Wet", storeResult=var "Particle")
    SetGameObject(variable=var "Particle 2", gameObject=<null>, everyFrame=false)
    on FINISHED → Play
  }

  var Environment Type: int = 0
  var Bone: gameObject = Run Effects/Bone
  var Dust: gameObject = Run Effects/Dust
  var Grass: gameObject = Run Effects/Grass
  var Particle: gameObject = <null>
  var Particle 2: gameObject = <null>
  var Wet: gameObject = Run Effects/Wet
}
