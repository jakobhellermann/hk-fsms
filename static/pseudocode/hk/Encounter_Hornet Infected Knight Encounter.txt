# Encounter
# 1 location(s):
#   Fungus1_04_boss / Hornet Infected Knight Encounter

fsm Encounter {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetParent(gameObject=var "Grass Escape", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="hornetGreenpath", storeValue=var "Hornet Encounter")
    IntCompare(integer1=var "Hornet Encounter", integer2=4, equal=→"DESTROY", lessThan=(none), greaterThan=→"DESTROY", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hornet1Defeated", isTrue=→"DESTROY", isFalse=(none))
    on FINISHED → Idle
    on DESTROY → Destroy
  }

  state Idle {
    FindGameObject(objectName="Hornet Boss 1", withTag="Untagged", store=var "Boss")
    FindGameObject(objectName="Hornet Saver", withTag="Untagged", store=var "Hornet Saver")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Point
    on ATTACKED → Attacked
  }

  state Point {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Point")
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Idle")  // disabled
    SetPlayerDataInt(intName="hornetGreenpath", value=4)
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dialogue
  }

  state Dialogue {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["HORNET_GREENPATH", "Hornet"], storeResult=(unset var))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=Hornet_Greenpath_01 (sharedassets133.assets))
    on CONVO_FINISH → Box Down
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Hero Active
  }

  state Set Hero Active {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    on FINISHED → Leap Antic
  }

  state Jump {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump Full")
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 6, 0), time=0.7, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=true)
    Wait(time=0.415, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Grass", emit=0)
    on FINISHED → Throw Start
  }

  state Throw Start {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Throw Side Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Throw
  }

  state Throw {
    ActivateGameObject(gameObject=var "Thread", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Thread", clipName="Thread 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throw Side")
    on FINISHED → Yank
  }

  state Yank {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Harpoon Side")
    SetParent(gameObject=var "Thread", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-45, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetParent(gameObject=var "Thread", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(-12, 12, 0), time=0.5, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)  // disabled
    PlayParticleEmitter(gameObject=var "Grass Escape", emit=0)
    on FINISHED → 
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Leap Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Evade Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=Hornet_Fight_Yell_03 (sharedassets105.assets))
    on FINISHED → Leap Back
  }

  state Leap Back {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Evade")
    iTweenMoveBy(gameObject=Self, id="", vector=(-7, 0, 0), time=0.25, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetPlayerDataInt(intName="hornetGreenpath", value=4)
    on FINISHED → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Soft Land")
    on FINISHED → Start Fight
  }

  state Start Fight {
    ActivateAllChildren(gameObject=var "Hornet Saver", activate=true)
    SendEventByName(eventTarget=GameObject(var "Boss"), sendEvent="WAKE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="BOSS HORNET", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
  }

  state Attacked {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on FINISHED → Leap Antic
  }

  var Hornet Encounter: int = 0
  var Area Title: gameObject = <null>
  var Boss: gameObject = <null>
  var Grass: gameObject = Hornet Infected Knight Encounter/Grass
  var Grass Escape: gameObject = Hornet Infected Knight Encounter/Grass Escape
  var Hornet Saver: gameObject = <null>
  var Self: gameObject = <null>
  var Thread: gameObject = Hornet Infected Knight Encounter/Thread
}
