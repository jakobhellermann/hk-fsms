  start Pause

  state Pause {
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Set
  }

  state Set {
    CallMethodProper(gameObject=var "Game Map", behaviour="GameMap", methodName="SetCustomCompassPos", parameters=[var "Compass X", var "Compass Y", var "Scene Name", var "Map Zone", var "Offset X", var "Offset Y", var "Tilemap Width", var "Tilemap Height"], storeResult=(unset var))
    SetFsmBool(gameObject=var "Quick Map", fsmName="Quick Map", variableName="In Room", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Room Override", setValue=var "z_Allow Dreamgate Set", everyFrame=false)
  }

  var Compass X: float = 173.86
  var Compass Y: float = 12.75
  var Offset X: float = 0
  var Offset Y: float = 0
  var Tilemap Height: float = 50
  var Tilemap Width: float = 200
  var z_Allow Dreamgate Set: bool = false
  var Map Zone: string = "OUTSKIRTS"
  var Scene Name: string = "Deepnest_East_09"
}
