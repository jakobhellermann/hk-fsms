# Chest Control
# 2 location(s):
#   Fungus2_31 / Mantis Chest (1)
#   Fungus2_31 / Mantis Chest

fsm Chest Control {
  start Pause Frame

  state Pause Frame {
    FindChild(gameObject=Self, childName="Item", storeResult=var "Item Holder")
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    ActivateAllChildren(gameObject=var "Item Holder", activate=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Opened", storeResult=var "Opened")
    FindChild(gameObject=Self, childName="Idle Glow", storeResult=var "Idle Glow")
    FindChild(gameObject=Self, childName="Open Effects", storeResult=var "Open Effects")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Geo Chest", isTrue=→"GEO", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activated
    on GEO → Pool Geo
    on FINISHED → Idle
  }

  state Activated {
    SetCollider(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    StopParticleEmitter(gameObject=var "Idle Glow")
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Open
  }

  state Pool Geo {
    FindChild(gameObject=Self, childName="Geo Pool", storeResult=var "Geo Pool")
    CreatePoolObjects(gameObject=Geo Small (resources.assets), pool=var "Geo Pool", position=(unset), amount=var "Geo Small", originVariationX=1, originVariationY=0, deactivate=true)
    CreatePoolObjects(gameObject=Geo Med (resources.assets), pool=var "Geo Pool", position=(unset), amount=var "Geo Med", originVariationX=1, originVariationY=0, deactivate=true)
    CreatePoolObjects(gameObject=Geo Large (resources.assets), pool=var "Geo Pool", position=(unset), amount=var "Geo Large", originVariationX=1, originVariationY=0, deactivate=true)
    on FINISHED → Idle
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=chest_open (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=barrel_death_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    StopParticleEmitter(gameObject=var "Idle Glow")
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Open", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Spawn Items
  }

  state Spawn Items {
    ActivateGameObject(gameObject=var "Open Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    SpawnFromPool(pool=var "Geo Pool", adjustPosition=(0, 0, 0), spawnMin=9999, spawnMax=9999, speedMin=25, speedMax=38, angleMin=78, angleMax=102)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Opened
  }

  state Opened {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Geo Large: int = 0
  var Geo Med: int = 0
  var Geo Small: int = 0
  var Geo Chest: bool = false

  var Activated: bool = false
  var Camera: gameObject = <null>
  var Geo Pool: gameObject = <null>
  var Idle Glow: gameObject = <null>
  var Item Holder: gameObject = <null>
  var Open Effects: gameObject = <null>
  var Opened: gameObject = <null>
  var Self: gameObject = <null>
}
