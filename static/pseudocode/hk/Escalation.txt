# Escalation
# 3 location(s):
#   Fungus1_35 / Warrior/Ghost Warrior No Eyes
#   GG_Ghost_No_Eyes / Warrior/Ghost Warrior No Eyes
#   GG_Ghost_No_Eyes_V / Warrior/Ghost Warrior No Eyes

fsm Escalation {
  start Idle

  state Idle {
    on TOOK DAMAGE → Check
  }

  state Check {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Esc 1", equal=(none), lessThan=→"ESCALATE", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on ESCALATE → Escalate
  }

  state Escalate {
    SetFsmFloat(gameObject=Self, fsmName="Shot Spawn", variableName="Spawn Pause", setValue=1.75, everyFrame=false)
    on FINISHED → Idle 2
  }

  state Idle 2 {
    on TOOK DAMAGE → Check 2
  }

  state Check 2 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Esc 2", equal=(none), lessThan=→"ESCALATE", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle 2
    on ESCALATE → Escalate 2
  }

  state Escalate 2 {
    SetFsmFloat(gameObject=Self, fsmName="Shot Spawn", variableName="Spawn Pause", setValue=1.25, everyFrame=false)
    on FINISHED → 
  }

  var Esc 1: int = 150
  var Esc 2: int = 90

  var HP: int = 0
}
