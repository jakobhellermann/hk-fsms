# Control
# 1 location(s):
#   White_Palace_09 / White Palace Lift

fsm Control {
  start Init

  state Init {
    CallMethodProper(gameObject=Self, behaviour="HeroPlatformStick", methodName="Deactivate", parameters=[], storeResult=(unset var))
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Init Y", z=(unset), space=0, everyFrame=false)
    SetParent(gameObject=var "Bot Beam", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "White Flash", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Bot
  }

  state Bot {
    CallMethodProper(gameObject=Self, behaviour="HeroPlatformStick", methodName="Deactivate", parameters=[], storeResult=(unset var))
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"COLLISION ENTER 2D", storeCollider=var "Collider", storeForce=(unset))
    on COLLISION ENTER 2D → Check
  }

  state Check {
    GetCollision2dInfo(gameObjectHit=var "Collider", relativeVelocity=(unset), relativeSpeed=var "Force", contactPoint=(unset), contactNormal=var "Normal", shapeCount=(unset), physics2dMaterialName="")
    GetTag(gameObject=var "Collider", storeResult=var "Tag", everyFrame=false)
    StringCompare(stringVariable=var "Tag", compareTo="Player", equalEvent=(none), notEqualEvent=→"FINISHED", storeResult=(unset), everyFrame=false)
    GetVector3XYZ(vector3Variable=var "Normal", storeX=(unset), storeY=var "Normal Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=-1, tolerance=0.1, equal=→"HIT", lessThan=(none), greaterThan=(none), everyFrame=false)
    on HIT → Antic Pause
    on FINISHED → Bot
  }

  state Rise Antic {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Sprite", audioClip=white_seal_1 (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    ObjectJitterLocal(gameObject=var "Sprite", x=0.1, y=(unset), z=(unset))
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bot Beam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Rise Beam"), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Rise
  }

  state Antic Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rise Antic
  }

  state Rise {
    CallMethodProper(gameObject=Self, behaviour="HeroPlatformStick", methodName="Activate", parameters=[], storeResult=(unset var))
    AudioPlayInState(gameObject=Self, volume=1)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Sprite", audioClip=misc_rumble_impact (sharedassets13.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPosition(gameObject=var "Sprite", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=30, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 20), x=(unset), y=(unset), everyFrame=false)  // disabled
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Top Y", tolerance=0, equal=→"TOP", lessThan=(none), greaterThan=→"TOP", everyFrame=true)
    PlayParticleEmitter(gameObject=var "Pt", emit=0)
    on TOP → Hit Top
  }

  state Hit Top {
    CallMethodProper(gameObject=Self, behaviour="HeroPlatformStick", methodName="Deactivate", parameters=[], storeResult=(unset var))
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Top Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Rise Beam"), sendEvent="DOWN", delay=0.5, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Sprite", audioClip=focus_health_heal (resources.assets), pitchMin=0.75, pitchMax=0.75, volume=1, delay=0, storePlayer=<null>)
    on RETURN → Return
  }

  state Return {
    CallMethodProper(gameObject=Self, behaviour="HeroPlatformStick", methodName="Activate", parameters=[], storeResult=(unset var))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Init Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bot Beam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Sprite", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bot
  }

  var Top Y: float = 91.9

  var Force: float = 0
  var Init Y: float = 0
  var Normal Y: float = 0
  var Self Y: float = 0
  var Tag: string = ""
  var Normal: vector3 = (0, 0, 0)
  var Bot Beam: gameObject = White Palace Lift/Bot Beam
  var Collider: gameObject = <null>
  var Pt: gameObject = White Palace Lift/Pt
  var Rise Beam: gameObject = White Palace Lift/Rise Beam
  var Sprite: gameObject = White Palace Lift/Sprite
  var White Flash: gameObject = White Palace Lift/White Flash
}
