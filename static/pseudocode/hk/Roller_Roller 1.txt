# Roller
# 1 location(s):
#   Crossroads_ShamanTemple / _Enemies/Roller 1

fsm Roller {
  start Initiate
  on GO LEFT → Go Left  // from any state
  on GO RIGHT → Go Right  // from any state

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    SetParticleEmission(gameObject=var "Dust Emitter", emission=false)
    FindChild(gameObject=Self, childName="Alert Range New", storeResult=var "Alerter")
    GetHero(storeResult=var "Hero")
    SetFloatValue(floatVariable=var "Max Speed Neg", floatValue=var "Max Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Max Speed Neg", multiplyBy=-1, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=true)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=true, everyFrame=true)
    CheckAlertRangeByName(storeResult=var "In Alert Range", childName="Alert Range New", everyFrame=true)
    CheckCanSeeHero(storeResult=var "Can See Hero", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Alert Range"], sendEvent=→"ALERT", storeResult=(unset), everyFrame=true)
    on ALERT → Facing Check
  }

  state Facing Check {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "Self X", tolerance=0, equal=→"RIGHT", lessThan=→"LEFT", greaterThan=→"RIGHT", everyFrame=false)
    on LEFT → Start Left
    on RIGHT → Start Right
  }

  state Start Right {
    SetBoolValue(boolVariable=var "Moving Right", boolValue=true, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Start
  }

  state Start Left {
    SetBoolValue(boolVariable=var "Moving Right", boolValue=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Start
  }

  state Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Start")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    RandomFloat(min=var "Roll time Min", max=var "Roll time Max", storeResult=var "Roll time")
    AudioPlay(gameObject=var "Self", volume=1, oneShotClip=roller_curl (sharedassets32.assets), finishedEvent=(none))
    on WAIT → Left or right?
  }

  state Roll R {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Velocity X", add=var "Acceleration", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Velocity X", minValue=var "Max Speed Neg", maxValue=var "Max Speed", everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=true, otherLayerNumber=25, ignoreTriggers=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=(none), otherLayer=true, otherLayerNumber=11, ignoreTriggers=true)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=(none), otherLayer=true, otherLayerNumber=11, ignoreTriggers=true)
    FloatSubtract(floatVariable=var "Roll time", subtract=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Roll time", float2=0, tolerance=0, equal=→"STOP", lessThan=→"STOP", greaterThan=(none), everyFrame=true)
    on WALL → Collide Right
    on STOP → Stop
    on RECOIL HORIZONTAL → Recoil Decel R
  }

  state Left or right? {
    SetParticleEmission(gameObject=var "Dust Emitter", emission=true)
    SetAudioPitch(gameObject=Self, pitch=1, everyFrame=false)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    BoolTest(boolVariable=var "Moving Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Roll R
    on LEFT → Roll L
  }

  state Collide Right {
    SetParticleEmission(gameObject=var "Dust Emitter", emission=false)
    AudioStop(gameObject=Self)
    SetBoolValue(boolVariable=var "Moving Right", boolValue=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=115, speed=12, everyFrame=false)
    AudioPlay(gameObject=var "Self", volume=1, oneShotClip=roller_hit_wall (sharedassets32.assets), finishedEvent=(none))
    NextFrameEvent(sendEvent=→"WAIT")
    on WAIT → In Air
  }

  state In Air {
    FloatSubtract(floatVariable=var "Roll time", subtract=1, everyFrame=true, perSecond=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"GROUND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on GROUND → Land
  }

  state Roll L {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=true)
    FloatSubtract(floatVariable=var "Velocity X", subtract=var "Acceleration", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Velocity X", minValue=var "Max Speed Neg", maxValue=var "Max Speed", everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=true, otherLayerNumber=11, ignoreTriggers=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=true, otherLayerNumber=25, ignoreTriggers=true)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=true, otherLayerNumber=11, ignoreTriggers=true)
    FloatSubtract(floatVariable=var "Roll time", subtract=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Roll time", float2=0, tolerance=0, equal=→"STOP", lessThan=→"STOP", greaterThan=(none), everyFrame=true)
    on WALL → Collide Left
    on STOP → Stop
    on RECOIL HORIZONTAL → Recoil Decel L
  }

  state Collide Left {
    SetParticleEmission(gameObject=var "Dust Emitter", emission=false)
    AudioStop(gameObject=Self)
    SetBoolValue(boolVariable=var "Moving Right", boolValue=true, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=65, speed=12, everyFrame=false)
    AudioPlay(gameObject=var "Self", volume=1, oneShotClip=roller_hit_wall (sharedassets32.assets), finishedEvent=(none))
    NextFrameEvent(sendEvent=→"WAIT")
    on WAIT → In Air
  }

  state Stop {
    SetParticleEmission(gameObject=var "Dust Emitter", emission=false)
    AudioStop(gameObject=Self)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stop")
    AudioPlay(gameObject=var "Self", volume=1, oneShotClip=roller_curl (sharedassets32.assets), finishedEvent=(none))
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Rest
  }

  state Rest {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=var "Stop Time", finishEvent=→"WAIT", realTime=false)
    on WAIT → Idle
  }

  state Recoil Decel R {
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Roll R
  }

  state Recoil Decel L {
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Roll L
  }

  state Land {
    AudioPlayRandom(gameObject=var "Self", audioClips=[geo_hit_ground_1 (resources.assets), geo_hit_ground_2 (resources.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    SpawnObjectFromGlobalPool(gameObject=Dust Land Small (sharedassets32.assets), spawnPoint=var "Self", position=(0, -0.4, 0), rotation=(-72, -180, -180), storeObject=var "")
    on FINISHED → Left or right?
  }

  state Go Right {
    Translate(gameObject=Self, vector=(unset), x=0.1, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Stop
  }

  state Go Left {
    Translate(gameObject=Self, vector=(unset), x=-0.1, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Stop
  }

  var Acceleration: float = 0.45
  var Max Speed: float = 11
  var Roll time Max: float = 3
  var Roll time Min: float = 2
  var Stop Time: float = 0.5

  var Hero X: float = 0
  var Max Speed Neg: float = 0
  var Raycast Length: float = 0
  var Roll time: float = 0
  var Self X: float = 0
  var Velocity X: float = 0
  var Can See Hero: bool = false
  var In Alert Range: bool = false
  var Moving Right: bool = false
  var Box Origin: vector2 = (0, 0)
  var Dimensions: vector2 = (0, 0)
  var Alerter: gameObject = <null>
  var Dust Emitter: gameObject = _Enemies/Roller 1/Roll Dust
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
