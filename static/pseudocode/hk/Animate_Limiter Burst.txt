# Animate
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Limiter Burst
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Limiter Burst

fsm Animate {
  start Idle

  state Idle {
    SetMeshRenderer(gameObject=Self, active=false)
    on SOUL LIMITER DOWN → Animate
    on SOUL LIMITER UP → Animate
  }

  state Animate {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Soul Burst", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }
}
