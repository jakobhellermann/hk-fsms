# Hit Count
# 1 location(s):
#   GG_Lurker / Lurker Control/Pale Lurker

fsm Hit Count {
  start Idle

  state Idle {
    on TOOK DAMAGE → Hit
  }

  state Hit {
    GetFsmInt(gameObject=Self, fsmName="Lurker Control", variableName="Hit Count", storeValue=var "Hit Count", everyFrame=false)
    IntAdd(intVariable=var "Hit Count", add=-1, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="Lurker Control", variableName="Hit Count", setValue=var "Hit Count", everyFrame=false)
    on FINISHED → Idle
  }

  var Hit Count: int = 0
}
