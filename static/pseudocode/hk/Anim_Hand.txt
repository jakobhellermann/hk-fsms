# Anim
# 1 location(s):
#   Room_GG_Shortcut / Fluke Hermit/Hand

fsm Anim {
  start Idle

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stick Idle")
    WaitRandom(timeMin=1, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Poke
  }

  state Poke {
    SelectRandomString(strings=["Poke 1", "Poke 2"], weights=[1, 0.6], storeString=var "Anim")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Anim: string = ""
}
