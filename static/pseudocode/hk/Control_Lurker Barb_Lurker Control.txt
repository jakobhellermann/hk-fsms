# Control
# 1 location(s):
#   GG_Lurker / Lurker Control/Lurker Barb

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barb Ball")
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[ruin_sentry_sword_1 (sharedassets33.assets), ruin_sentry_sword_2 (sharedassets33.assets), ruin_sentry_sword_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Air
  }

  state Air {
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    on COLLIDE → Stick
  }

  state Stick {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=ruin_flying_javelin_sentry_projectile_impact (sharedassets92.assets), pitchMin=0.9, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Antic
  }

  state Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barb Antic")
    Wait(time=0.45, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spike Up
  }

  state Spike Up {
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barb Spike")
    Wait(time=8, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[white_spikes_appear_01 (sharedassets451.assets), white_spikes_appear_02 (sharedassets451.assets), white_spikes_appear_03 (sharedassets371.assets)], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Down
  }

  state Down {
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Retract", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[white_spikes_dissapear_01 (sharedassets371.assets), white_spikes_dissapear_02 (sharedassets451.assets), white_spikes_dissapear_03 (sharedassets451.assets)], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Burst: gameObject = Lurker Control/Lurker Barb/Burst
  var Damager: gameObject = Lurker Control/Lurker Barb/Damager
  var Self: gameObject = <null>
}
