# Switch Control
# 1 location(s):
#   Crossroads_06 / _Props/Gate Switch

fsm Switch Control {
  start Pause

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Hit Origin", storeResult=var "Hit Origin")
    BoolTest(boolVariable=var "Activated", isTrue=→"TRUE", isFalse=(none), everyFrame=false)
    on TRUE → Activated
    on FINISHED → PD On?
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Player Data?
  }

  state Hit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_switch (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Hit Origin", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Switch Hit")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=GameObject(var "Target"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
  }

  state Activated {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Switch Activated")
    SendEventByName(eventTarget=GameObject(var "Target"), sendEvent="ACTIVATE AGAIN", delay=0, everyFrame=false)
    on FINISHED → Open
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state Player Data? {
    BoolTest(boolVariable=var "SetPlayerData", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetPlayerDataBool(boolName=var "Player Data", value=true)
    on CANCEL → Hit
    on FINISHED → Hit
  }

  state PD On? {
    BoolTest(boolVariable=var "SetPlayerData", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "Player Data", isTrue=→"TRUE", isFalse=→"FINISHED")
    on FINISHED → Idle
    on TRUE → Activated
  }

  var SetPlayerData: bool = true
  var Player Data: string = "shamanPillar"
  var Target: gameObject = _Scenery/Raising Pillar

  var Attack Direction: float = 0
  var Attack Type: int = 0
  var Activated: bool = false
  var Camera: gameObject = <null>
  var Damager: gameObject = <null>
  var Hit Origin: gameObject = <null>
  var Self: gameObject = <null>
}
