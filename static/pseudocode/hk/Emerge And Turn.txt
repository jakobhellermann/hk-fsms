# Emerge And Turn
# 1 location(s):
#   Waterways_05 / Dung Defender NPC

fsm Emerge And Turn {
  start Sleep

  state Sleep {
    BoolTest(boolVariable=var "Emerge Range", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    on WAKE → Wake Pause
  }

  state Wake Pause {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emerge
  }

  state Emerge {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="NPC Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Emerge Chunks", emit=0)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dung_defender_emerge (sharedassets354.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Dung_def_burstout_03 (sharedassets354.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Ready
  }

  state Ready {
    SetCollider(gameObject=Self, active=true)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on L → Hero R
    on R → Hero R
  }

  state Hero R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NPC TurnRight")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on L → Hero L
  }

  state Hero L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NPC TurnLeft")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on R → Hero R
  }

  var Emerge Range: bool = false
  var Dream Dialogue: gameObject = <null>
  var Emerge Chunks: gameObject = Dung Defender NPC/Erupt Chunks
  var Self: gameObject = <null>
}
