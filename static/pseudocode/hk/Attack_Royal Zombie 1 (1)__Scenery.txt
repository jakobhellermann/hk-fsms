# Attack
# 1 location(s):
#   Ruins1_27 / _Scenery/Royal Zombie 1 (1)

fsm Attack {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Attack_Range", isTrue=→"Hero in range", isFalse=(none), everyFrame=true)
    on Hero in range → Prepare Charge
  }

  state Prepare Charge {
    StopWalker(target=Self, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=royal_pointy_prepare (sharedassets84.assets))
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Attack Start", doNotResetCurrentClip=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Get Scale
  }

  state Get Scale {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetWalkerFacing(walkRight=false, randomStartDir=false, target=Self, everyFrame=false)
    SetFloatValue(floatVariable=var "Dash Speed", floatValue=-8, everyFrame=false)
    FloatSignTest(floatValue=var "Scale", isPositive=→"FINISHED", isNegative=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Dash Speed", floatValue=8, everyFrame=false)
    SetWalkerFacing(walkRight=true, randomStartDir=false, target=Self, everyFrame=false)
    on FINISHED → Dash
  }

  state Dash {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Dash Speed", y=(unset), everyFrame=true)
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Attack Dash", doNotResetCurrentClip=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Royal_Zombie_Point_attack_01 (sharedassets84.assets), Royal_Zombie_Point_attack_02 (sharedassets84.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Swipe
  }

  state Swipe {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=royal_pointy_punching_loop (sharedassets84.assets))
    Decelerate(gameObject=Self, deceleration=0.8)
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Attack Loop", doNotResetCurrentClip=false)
    WaitRandom(timeMin=1.5, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Swipe", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Attack Recover
  }

  state Movestart {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on FINISHED → Idle
  }

  state Attack Recover {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Attack Recover", doNotResetCurrentClip=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Swipe", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Return to idle
  }

  state Return to idle {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Idle", doNotResetCurrentClip=false)
    WaitRandom(timeMin=0.4, timeMax=0.65, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Movestart
  }

  var Dash Speed: float = 0
  var Scale: float = 0
  var Attack_Range: bool = false
  var Self: gameObject = <null>
  var Swipe: gameObject = _Scenery/Royal Zombie 1 (1)/Swipe
}
