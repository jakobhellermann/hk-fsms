# Conversation Control
# 1 location(s):
#   Crossroads_ShamanTemple / _Props/Shaman Meeting

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state

  state Idle {
    on CONVO START → Convo Choice
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    on FINISHED → Turn Hero?
  }

  state Convo Choice {
    GetPlayerDataBool(boolName="shamanFireball2Convo", storeValue=var "Fireball2 Convo")
    GetPlayerDataBool(boolName="shamanScreamConvo", storeValue=var "Scream Convo")
    GetPlayerDataBool(boolName="shamanScream2Convo", storeValue=var "Scream2 Convo")
    GetPlayerDataBool(boolName="shamanQuakeConvo", storeValue=var "Quake Convo")
    GetPlayerDataBool(boolName="shamanQuake2Convo", storeValue=var "Quake2 Convo")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="SHAMAN", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    BoolTestMulti(boolVariables=[var "Fireball2 Convo", var "Has Fireball2"], boolStates=[false, true], trueEvent=→"FIREBALL 2", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Quake Convo", var "Has Quake"], boolStates=[false, true], trueEvent=→"QUAKE", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Quake2 Convo", var "Has Quake 2"], boolStates=[false, true], trueEvent=→"QUAKE 2", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Scream Convo", var "Has Scream"], boolStates=[false, true], trueEvent=→"SCREAM", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Scream2 Convo", var "Has Scream2"], boolStates=[false, true], trueEvent=→"SCREAM 2", falseEvent=(none), storeResult=(unset), everyFrame=false)
    IntSwitch(intVariable=var "Shaman State", compareTo=[0, 1, 2, 6], sendEvent=[→"MEET", →"SUMMONED 1", →"SUMMONED 2", →"RETURNED"], everyFrame=false)
    on MEET → Turn Wait
    on SUMMONED 1 → Turn Wait 2
    on SUMMONED 2 → Turn Wait 3
    on RETURNED → Turn Wait 4
    on FIREBALL 2 → Turn Wait 5
    on SCREAM → Turn Wait 6
    on SCREAM 2 → Turn Wait 7
    on QUAKE → Turn Wait 8
    on QUAKE 2 → Turn Wait 9
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    GetHero(storeResult=var "Hero")
    on FINISHED → Get Spell Levels
  }

  state Meet {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Shaman_01 (sharedassets76.assets), Shaman_02 (sharedassets76.assets), Shaman_03 (sharedassets76.assets), Shaman_04 (sharedassets76.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_MEET", "Shaman"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Turn Wait {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)
    IntCompare(integer1=var "Frame", integer2=9, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Meet
  }

  state Turn? {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Self X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Turn")
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)
    SetFsmBool(gameObject=Self, fsmName="npc_control", variableName="Turned", setValue=false, everyFrame=false)
    IntCompare(integer1=var "Frame", integer2=9, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Summon Anim
  }

  state Summon Anim {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=shaman_charging_spell (sharedassets76.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Summon")
    FindChild(gameObject=Self, childName="Summon 1", storeResult=var "Summon Pt 1")
    EaseFloat(fromValue=0, toValue=40, floatVariable=var "Emission Rate", time=2, speed=(unset), delay=0.5, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetParticleEmissionRate(gameObject=var "Summon Pt 1", emissionRate=var "Emission Rate", everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spell Appear
  }

  state Spell Appear {
    SetParticleEmissionRate(gameObject=var "Summon Pt 1", emissionRate=0, everyFrame=true)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=shaman_spell (sharedassets43.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataInt(intName="shaman", value=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateGameObject(gameObject=var "Summon Pt 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="VS Summon Fx", storeResult=var "Summon FX")
    ActivateGameObject(gameObject=var "Summon FX", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vengeful Spirit", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Turn Wait 2 {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)
    IntCompare(integer1=var "Frame", integer2=9, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)

  }

  state Summoned 1 {

    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Shaman_04 (sharedassets76.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)

  }

  state Box Down 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")

    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Check Summoned {
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=6, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    FindChild(gameObject=Self, childName="Vengeful Spirit", storeResult=var "Vengeful Spirit")
    SetParent(gameObject=var "Vengeful Spirit", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=1, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetPlayerDataInt(intName="shaman", value=2)
    IntCompare(integer1=var "Shaman State", integer2=2, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    ActivateGameObject(gameObject=var "Vengeful Spirit", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Turn Wait 3 {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)

    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_SUMMONED_2", "Shaman"], storeResult=(unset var))

  }

  state Turn Hero? {
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"LOOK END", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="TurnToIdle")
    Tk2dPlayFrame(gameObject=var "Hero", frame=0)
    SetScale(gameObject=var "Hero", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="FaceLeft", parameters=[], storeResult=(unset var))
    on FINISHED → Turn?
    on LOOK END → Hero Lookend
  }

  state Check Active {
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=6, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    IntCompare(integer1=var "Fireball Level", integer2=1, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    DestroySelf(detachChildren=false)
    on FINISHED → Check Summoned
  }

  state Turn Wait 4 {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)

  }

  }

  state Hero Lookend {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")

  }

  state Get Spell Levels {
    GetPlayerDataInt(gameObject=var "GameManager", intName="fireballLevel", storeValue=var "Fireball Level")
    GetPlayerDataInt(gameObject=var "GameManager", intName="quakeLevel", storeValue=var "Quake Level")
    GetPlayerDataInt(gameObject=var "GameManager", intName="screamLevel", storeValue=var "Scream Level")
    IntTestToBool(int1=var "Fireball Level", int2=1, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Has Fireball2", everyFrame=false)
    IntTestToBool(int1=var "Quake Level", int2=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Has Quake", everyFrame=false)
    IntTestToBool(int1=var "Quake Level", int2=1, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Has Quake 2", everyFrame=false)
    IntTestToBool(int1=var "Scream Level", int2=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Has Scream", everyFrame=false)
    IntTestToBool(int1=var "Scream Level", int2=1, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Has Scream2", everyFrame=false)
    on FINISHED → Check Active
  }

  state Turn Wait 5 {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)
    IntCompare(integer1=var "Frame", integer2=9, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
