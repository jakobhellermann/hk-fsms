# Ceiling Dropper
# 1 location(s):
#   RestingGrounds_10 / Ceiling Dropper (1)

fsm Ceiling Dropper {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Flare", storeResult=var "Flare")
    PreBuildTK2DSprites(gameObject=var "Flare", useChildren=false)
    GetOwner(storeGameObject=var "Self")
    SendRandomEvent(events=[(none), (none)], weights=[0.5, 0.5], delay=0)
    on 1 → Anim 1
    on 2 → Anim 2
  }

  state Idle {
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    WaitRandom(timeMin=2, timeMax=6, finishEvent=→"CRY", realTime=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on ALERT → Startle
    on CRY → Cry
    on COLLIDE → Explode
  }

  state Startle {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ceiling_dropper_emerge (sharedassets33.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle To Dive", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Flare", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on FINISHED → Dive
    on COLLIDE → Explode
  }

  state Dive {
    AudioPlayInState(gameObject=Self, volume=1)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=ceiling_dropper_charge (sharedassets88.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-20, everyFrame=true)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=25, accelerationForce=60, offsetX=0, offsetY=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dive")
    FaceAngle(gameObject=Self, angleOffset=90, everyFrame=true)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Explode
    on TOOK DAMAGE → Explode
  }

  state Explode {
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Recycle M (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Explosion")  // disabled
    SetScale(gameObject=var "Explosion", vector=(unset), x=1.2, y=1.2, z=1.2, everyFrame=false, lateUpdate=false)  // disabled
    ActivateGameObject(gameObject=var "Death Skull", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Death Skull", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FlingObject(flungObject=var "Death Skull", speedMin=22, speedMax=30, angleMin=60, angleMax=120)
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=15, spawnMax=20, speedMin=20, speedMax=25, angleMin=40, angleMax=140, originVariationX=0.3, originVariationY=0.3, FSM="", FSMEvent="")  // disabled
    InstaDeath(target=Self, direction=0)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
  }

  state Anim 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    RandomInt(min=1, max=58, storeResult=var "Start Frame", inclusiveMax=true)
    Tk2dPlayFrame(gameObject=Self, frame=var "Start Frame")
    on FINISHED → Idle
  }

  state Anim 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 2")
    RandomInt(min=1, max=70, storeResult=var "Start Frame", inclusiveMax=true)
    Tk2dPlayFrame(gameObject=Self, frame=var "Start Frame")
    on FINISHED → Idle
  }

  state Cry {
    AudioPlayRandom(gameObject=var "Self", audioClips=[ceiling_dropper_look_1 (sharedassets48.assets), ceiling_dropper_look_2 (sharedassets48.assets), ceiling_dropper_look_3 (sharedassets48.assets), ceiling_dropper_look_4 (sharedassets48.assets), ceiling_dropper_look_5 (sharedassets48.assets)], weights=[1, 1, 1, 0, 0], pitchMin=1, pitchMax=1)
    on FINISHED → Idle
    on COLLIDE → Explode
  }

  var Accel: float = 100
  var Speed Max: float = 20

  var Start Frame: int = 0
  var Alert Range: bool = false
  var Death Skull: gameObject = Ceiling Dropper (1)/Death Skull
  var Explosion: gameObject = <null>
  var Flare: gameObject = <null>
  var Self: gameObject = <null>
}
