fsm Movement {
  start Start
  on ATTACK START → Attacking  // from any state

  state Init {
    SetParent(gameObject=var "Target", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Warp In
  }

  state Warp In {
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Attack Pt", emit=0)
    on FINISHED → Ready
  }

  state Choose Target {
    BoolTest(boolVariable=var "Left", isTrue=(none), isFalse=(none), everyFrame=false)
    on 1 → Set 1
    on 2 → Set 2
    on 3 → Set 3
    on 4 → Set 4
    on 5 → Set 5
    on 6 → Set 6
    on 7 → Set 7
  }

  state Set 1 {
    SetBoolValue(boolVariable=var "Left", boolValue=true, everyFrame=false)
    SetPosition(gameObject=var "Target", vector=var "P1", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 2 {
    SetBoolValue(boolVariable=var "Left", boolValue=false, everyFrame=false)
    SetPosition(gameObject=var "Target", vector=var "P2", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 3 {
    SetPosition(gameObject=var "Target", vector=var "P3", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)

  state Set 4 {
    SetPosition(gameObject=var "Target", vector=var "P4", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover

  state Set 5 {
    SetPosition(gameObject=var "Target", vector=var "P5", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover

  state Set 6 {
    SetPosition(gameObject=var "Target", vector=var "P6", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Set 7 {
    SetPosition(gameObject=var "Target", vector=var "P7", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Hover
  }

  state Hover {
    ChaseObjectV2(gameObject=Self, target=var "Target", speedMax=9, accelerationForce=50, offsetX=0, offsetY=0)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)
    BoolTestMulti(boolVariables=[var "Hero Left", var "Self Left"], boolStates=[true, true], trueEvent=→"WARP R", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hero Left", var "Self Left"], boolStates=[false, false], trueEvent=→"WARP L", falseEvent=(none), storeResult=(unset), everyFrame=false)
    FloatTestToBool(float1=var "Self X", float2=85, tolerance=0, equalBool=(unset), lessThanBool=var "Below Min", greaterThanBool=(unset), everyFrame=true)
    FloatTestToBool(float1=var "Self X", float2=111, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Over Max", everyFrame=true)
    BoolTestMulti(boolVariables=[var "Below Min", var "Left"], boolStates=[true, true], trueEvent=→"CHANGE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Over Max", var "Left"], boolStates=[true, false], trueEvent=→"CHANGE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    WaitRandom(timeMin=1, timeMax=6, finishEvent=→"CHANGE", realTime=false)
    on FINISHED → Choose Target
    on WARP → Set Warp
    on CHANGE → Choose Target
  }

  state Attacking {
    DecelerateV2(gameObject=Self, deceleration=0.9)
    on ATTACK END → Hover
  }

  state Warp Out {
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK STOP", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Return
  }

  state Return {
    SetPosition(gameObject=Self, vector=(unset), x=56, y=41, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)

    SetCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Choose Target
  }

  state Warp Check {
    SendRandomEvent(events=[→"CANCEL", (none)], weights=[0.75, 0.25], delay=0)
    on CANCEL → Hover
    on FINISHED → Set Warp
  }

  state Choose R {
    FloatCompare(float1=var "Hero X", float2=28.11, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)

    on CANCEL → Hover
  }

  state Choose L {
    FloatCompare(float1=var "Hero X", float2=8.99, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Warp Pos", vector3Value=(unset), x=7, y=9.91, z=0.006, everyFrame=false)
    on FINISHED → Warp
    on CANCEL → Hover
  }

  state Warp {
    GetPosition(gameObject=Self, vector=var "Prev Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Warp Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Warp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Warp Out", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Warp Out", vector=var "Prev Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Hover
  }

  state Set Warp {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X", float2=20, tolerance=0, equal=→"WARP L", lessThan=→"WARP R", greaterThan=→"WARP L", everyFrame=false)
    on FINISHED → Hover
    on WARP L → Choose L
    on WARP R → Choose R
  }

  state Ready {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="READY", delay=0, everyFrame=false)
    on FINISHED → Choose Target
  }

  state Start {
    BoolTest(boolVariable=var "Init", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Init", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  var White Flash: gameObject = Warrior/Ghost Warrior Xero/White Flash
}
