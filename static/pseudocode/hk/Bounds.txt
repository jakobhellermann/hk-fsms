# Bounds
# 4 location(s):
#   Fungus3_archive_02_boss / Mega Jellyfish
#   GG_Uumuu / Mega Jellyfish GG
#   GG_Uumuu_V / Mega Jellyfish GG
#   sharedassets466.assets / Mega Jellyfish

fsm Bounds {
  start Init

  state Idle {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "X Min", tolerance=0, equal=(none), lessThan=→"UNDER X", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "X Max", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"OVER X", everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Y Min", tolerance=0, equal=(none), lessThan=→"UNDER Y", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Y Max", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"OVER Y", everyFrame=true)
    on UNDER X → Under X
    on UNDER Y → Under Y
    on OVER X → Over X
    on OVER Y → Over Y
  }

  state Under X {
    SetPosition(gameObject=Self, vector=(unset), x=var "X Min", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=5, y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Under Y {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Min", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=5, everyFrame=false)
    on FINISHED → Idle
  }

  state Over X {
    SetPosition(gameObject=Self, vector=(unset), x=var "X Max", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-5, y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Over Y {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Max", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-5, everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    on AWAKE → Idle
  }

  var Self X: float = 0
  var Self Y: float = 0
  var X Max: float = 69.18
  var X Min: float = 36.46
  var Y Max: float = 131.21
  var Y Min: float = 105.73
}
