# Miner FX
# 2 location(s):
#   Crossroads_45 / Miner
#   sharedassets471.assets / Miner

fsm Miner FX {
  start Init

  state Idle {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame ID", everyframe=true)
    IntCompare(integer1=var "Frame ID", integer2=20, equal=→"PLAY", lessThan=(none), greaterThan=(none), everyFrame=true)
    on PLAY → Choice
  }

  state Play {
    SetParticleEmissionRate(gameObject=var "Dust", emissionRate=80, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Rocks", emit=0)
    on FINISHED → Pause
  }

  state Pause {
    SetParticleEmissionRate(gameObject=var "Dust", emissionRate=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dust", storeResult=var "Dust")
    on FINISHED → Idle
  }

  state Choice {
    RandomFloat(min=0.85, max=1.15, storeResult=var "Chooser")
    SendRandomEvent(events=[(none), (none)], weights=[1, 1], delay=0)
    on 1 → 1
    on 2 → 2
  }

  state 2 {
    AudioPlay(gameObject=var "Self", volume=var "Chooser", oneShotClip=myla_pickaxe_2 (sharedassets73.assets), finishedEvent=(none))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Play
  }

  state 1 {
    AudioPlay(gameObject=var "Self", volume=var "Chooser", oneShotClip=myla_pickaxe_1 (sharedassets73.assets), finishedEvent=(none))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Play
  }

  var Chooser: float = 0
  var Frame ID: int = 0
  var Dust: gameObject = <null>
  var Pt Rocks: gameObject = Miner/Pt Rocks
  var Self: gameObject = <null>
}
