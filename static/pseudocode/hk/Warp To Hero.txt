# Warp To Hero
# 1 location(s):
#   resources.assets / Weaverling

fsm Warp To Hero {
  start Init
  on BG CLOSE → Warp  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    WaitRandom(timeMin=1, timeMax=1.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Check {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=15, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"WARP", everyFrame=false)
    on WARP → Warp
    on FINISHED → Idle
  }

  state Warp {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WARP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Distance: float = 0
}
