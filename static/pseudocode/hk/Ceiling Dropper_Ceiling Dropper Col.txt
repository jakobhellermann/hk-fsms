# Ceiling Dropper
# 1 location(s):

fsm Ceiling Dropper {
  start Startle

  state Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle To Dive", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ceiling_dropper_emerge)
    on FINISHED → Dive
    on TOOK DAMAGE → Explode
  }

  state Dive {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-20, everyFrame=true)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=25, accelerationForce=60, offsetX=0, offsetY=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dive")
    FaceAngle(gameObject=Self, angleOffset=90, everyFrame=true)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Explode
    on TOOK DAMAGE → Explode
  }

  state Explode {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Explosion", vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    InstaDeath(target=Self, direction=0)
  }

  var Self X: float = 0
  var Self Y: float = 0
  var Death Skull: gameObject = <null>
  var Explosion: gameObject = <null>
}
