# Control
# 1 location(s):
#   White_Palace_09 / Nursery Music Player

fsm Control {
  start Init

  state Init {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Start Pause
  }

  state Start Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Init
    on FINISHED → Play
  }

  state Play {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Fade Out
  }

  state Fade Out {
    GetAudioVolume(gameObject=Self, storeVolume=var "Volume", everyFrame=false)
    FadeAudio(gameObject=Self, startVolume=var "Volume", endVolume=0, time=5)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Fade In
  }

  state Fade In {
    GetAudioVolume(gameObject=Self, storeVolume=var "Volume", everyFrame=false)
    FadeAudio(gameObject=Self, startVolume=var "Volume", endVolume=1, time=2)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Fade Out
  }

  var Volume: float = 0
}
