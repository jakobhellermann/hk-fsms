# Control
# 1 location(s):
#   sharedassets34.assets / Electro Zap

fsm Control {
  start Init

  state Init {
    SetCircleCollider(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Pt Attack")
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Antic
  }

  state Antic {
    Wait(time=var "Antic Time", finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=zap_bug_pt1, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Damage
  }

  state Damage {
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Pt Attack", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=zap_bug_pt2, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Antic")
    StopParticleEmitter(gameObject=var "Pt Attack")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Antic Time: float = 1
  var Pt Antic: gameObject = Electro Zap/Pt Antic
  var Pt Attack: gameObject = Electro Zap/Pt Attack
  var Self: gameObject = Electro Zap
}
