# Battle Control
# 1 location(s):
#   Ruins2_09 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Wave 2", storeResult=var "Wave 2")
    FindChild(gameObject=Self, childName="Wave 3", storeResult=var "Wave 3")
    FindChild(gameObject=Self, childName="Wave 4", storeResult=var "Wave 4")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on ACTIVATE → Activated
    on HIT → Wave 1
  }

  state Activated {
  }

  state Wave 1 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Action and Sub (resources.assets), transitionTime=1)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Wave 1", activate=true)  // disabled
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=var "W1 Sentry", vector=(unset), x=31, y=3.81, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "W1 SentryFat", vector=(unset), x=51.68, y=4.3, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Wave 1"), sendEvent="BATTLE SPAWN", delay=0, everyFrame=false)
    on END → W2 Pause
  }

  state Wave 2 {
    ActivateAllChildren(gameObject=var "Wave 2", activate=true)  // disabled
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=var "W2 SentryFly", vector=(unset), x=38.86, y=16.86, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "W2 SentryJav", vector=(unset), x=42.1, y=16.1, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on END → Pause W 3
  }

  state Wave 3 {
    ActivateAllChildren(gameObject=var "Wave 3", activate=true)  // disabled
    SetIntValue(intVariable=var "Battle Enemies", intValue=3, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=var "W3 SentryFly", vector=(unset), x=38.86, y=16.86, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "W3 Sentry", vector=(unset), x=31, y=3.81, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "W3 SentryFat", vector=(unset), x=51.68, y=4.3, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Wave 3"), sendEvent="BATTLE SPAWN", delay=0, everyFrame=false)
    on END → Pause W 4
  }

  state Wave 4 {
    ActivateAllChildren(gameObject=var "Wave 4", activate=true)  // disabled
    SetIntValue(intVariable=var "Battle Enemies", intValue=3, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)

    SetPosition(gameObject=var "W4 SentryFat", vector=(unset), x=51.68, y=4.3, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Wave 3"), sendEvent="BATTLE SPAWN", delay=0, everyFrame=false)
    on END → End Pause
  }

  state End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

    on FINISHED → Wave 3
  }

    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 4
  }

  var W1 Sentry: gameObject = Battle Scene/Wave 1/Ruins Sentry 1
  var W1 SentryFat: gameObject = Battle Scene/Wave 1/Ruins Sentry Fat

}
