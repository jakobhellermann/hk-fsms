# Control
# 1 location(s):
#   GG_Hollow_Knight / Battle Scene/Focus Blasts/HK Prime Blast

fsm Control {
  start Init

  state Idle {
    on BLAST → Wait
  }

  state Wait {
    WaitRandom(timeMin=0.25, timeMax=1.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Pos X
  }

  state Set Pos X {
    RandomFloat(min=var "X Min", max=var "X Max", storeResult=var "Spawn X")
    BoolTest(boolVariable=var "Low", isTrue=→"LOW", isFalse=→"HIGH", everyFrame=false)
    on LOW → Pos Low
    on HIGH → Pos High
  }

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Start X", float2=-1.5, operation=Add, storeResult=var "X Min", everyFrame=false)
    FloatOperator(float1=var "Start X", float2=1.5, operation=Add, storeResult=var "X Max", everyFrame=false)
    on FINISHED → Idle
  }

  state Blast {
    ActivateGameObject(gameObject=var "Blast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Sound
  }

  state End {
    ActivateGameObject(gameObject=var "Blast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolFlip(boolVariable=var "Low")
    on FINISHED → Idle
  }

  state Sound {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Blast", audioClip=explosion_1 (sharedassets236.assets), pitchMin=1, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  state Pos Low {
    RandomFloat(min=7.88, max=10.08, storeResult=var "Spawn Y")
    SetPosition(gameObject=Self, vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Blast
  }

  state Pos High {
    RandomFloat(min=11.88, max=14.08, storeResult=var "Spawn Y")
    SetPosition(gameObject=Self, vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Blast
  }

  var Low: bool = true

  var Spawn X: float = 0
  var Spawn Y: float = 0
  var Start X: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var Blast: gameObject = Battle Scene/Focus Blasts/HK Prime Blast/Blast
}
