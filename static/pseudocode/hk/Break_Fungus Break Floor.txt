# Break
# 1 location(s):
#   Deepnest_01 / _Scenery/Fungus Break Floor

fsm Break {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"BREAK", storeCollider=var "")
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))  // disabled
    on BREAK → Check
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activate
  }

  state Activate {
    DestroySelf(detachChildren=false)
  }

  state Break {
    SetPlayerDataBool(boolName="deepnestBridgeCollapsed", value=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Rubble", activate=true)
    FlingObjects(containerObject=var "Rubble", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=0, speedMax=10, angleMin=0, angleMax=360)
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=break_wall_after_tutorial_area (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_ceiling_break (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    SetCollider(gameObject=Self, active=false)
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="steppedBeyondBridge", isTrue=(none), isFalse=→"CANCEL")
    on FINISHED → Break
    on CANCEL → Detect
  }

  var Activated: bool = false
  var Dust: gameObject = _Scenery/Fungus Break Floor/Dust
  var Rubble: gameObject = _Scenery/Fungus Break Floor/Rubble
  var Self: gameObject = <null>
  var Terrain: gameObject = _Scenery/Fungus Break Floor/Terrain
}
