# Deactivate
# 2 location(s):
#   Fungus1_04 / Camera Locks Boss
#   Abyss_19 / Camera Locks Boss

fsm Deactivate {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    GetOwner(storeGameObject=var "Self")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ACTIVATE", storeCollider=var "")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activate
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  var Activated: bool = false
  var Self: gameObject = <null>
}
