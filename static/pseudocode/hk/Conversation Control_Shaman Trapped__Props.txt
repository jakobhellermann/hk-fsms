# Conversation Control
# 1 location(s):
#   Crossroads_ShamanTemple / _Props/Shaman Trapped

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on UPDATE SHAMAN → Check Active  // from any state

  state Idle {
    on CONVO START → Convo Choice
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sit Talk End")
    on FINISHED → End
  }

  state Convo Choice {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="SHAMAN", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=3, equal=(none), lessThan=→"TRAPPED 1", greaterThan=(none), everyFrame=false)
    IntSwitch(intVariable=var "Shaman State", compareTo=[3], sendEvent=[→"TRAPPED 2"], everyFrame=false)
    on TRAPPED 1 → Trapped 1
    on TRAPPED 2 → Trapped 2
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    on FINISHED → Check Active
  }

  state Trapped 1 {
    SetPlayerDataInt(intName="shaman", value=3)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Shaman_01 (sharedassets76.assets), Shaman_02 (sharedassets76.assets), Shaman_03 (sharedassets76.assets), Shaman_04 (sharedassets76.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sit Talk")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_TRAPPED_1", "Shaman"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Check Summoned {
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=1, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    SetPlayerDataInt(intName="shaman", value=2)
    on FINISHED → Idle
  }

  state Check Active {
    GetLastEvent(storeEvent=var "Event Sent")
    GetPlayerDataInt(gameObject=var "GameManager", intName="fireballLevel", storeValue=var "Fireball Level")
    IntCompare(integer1=var "Fireball Level", integer2=0, equal=→"DESTROY", lessThan=(none), greaterThan=(none), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=4, equal=→"DESTROY", lessThan=(none), greaterThan=→"DESTROY", everyFrame=false)
    DestroySelf(detachChildren=false)  // disabled
    on FINISHED → Check Summoned
    on DESTROY → Destroy Self
  }

  state Trapped 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sit Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Shaman_01 (sharedassets76.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_TRAPPED_2", "Shaman"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state Destroy Self {
    DestroySelf(detachChildren=false)
  }

  var Emission Rate: float = 0
  var Hero X Scale: float = 0
  var Self X: float = 0
  var Self X Scale: float = 0
  var Fireball Level: int = 0
  var Frame: int = 0
  var Shaman State: int = 0
  var Event Sent: string = ""
  var Area Title: gameObject = <null>
  var CameraParent: gameObject = <null>
  var Hero: gameObject = <null>
  var MainCamera: gameObject = <null>
  var Self: gameObject = <null>
  var Sent By: gameObject = <null>
  var Summon FX: gameObject = <null>
  var Summon Pt 1: gameObject = <null>
  var Vengeful Spirit: gameObject = <null>
}
