fsm Lost Hero Check {
  start Out

  state Out {
    BoolTest(boolVariable=var "Lost Region", isTrue=→"IN", isFalse=(none), everyFrame=true)
    FloatAdd(floatVariable=var "Lost Counter", add=1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Lost Counter", minValue=0, maxValue=1000, everyFrame=true)
    on IN → In
  }

  state In {
    BoolTest(boolVariable=var "Lost Region", isTrue=(none), isFalse=→"OUT", everyFrame=true)
    SetFloatValue(floatVariable=var "Lost Counter", floatValue=0, everyFrame=false)
    on OUT → Out
  }

  var Lost Counter: float = 0
  var Lost Region: bool = false
}
