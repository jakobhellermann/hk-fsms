# FSM
# 1 location(s):
#   Crossroads_47 / _Transition Gates/right1

// uses template: gate_control_atmos
fsm FSM {
  start Pause Frame

  state Init {
    BoolTest(boolVariable=var "_Is Door", isTrue=→"DOOR", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Wait
    on DOOR → Door Wait
  }

  state Cave Wind {
    BoolTest(boolVariable=var "Cave Wind", isTrue=(none), isFalse=→"OFF", everyFrame=false)
    FindChild(gameObject=var "Atmos", childName="Atmos Cave Wind", storeResult=var "Atmos Cave Wind")
    SendEventByName(eventTarget=GameObjectFSM(var "Atmos Cave Wind", fsm="Atmos Control"), sendEvent="STOP", delay=0, everyFrame=false)
    on FINISHED → Surface Wind
    on OFF → Surface Wind
  }

  state Wait {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer=(unset), sendEvent=→"TRIGGER", storeCollider=var "")
    on TRIGGER → Music
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Surface Wind {
    BoolTest(boolVariable=var "Surface Wind", isTrue=(none), isFalse=→"OFF", everyFrame=false)
    FindChild(gameObject=var "Atmos", childName="Atmos Surface Wind", storeResult=var "Atmos Surface Wind")
    SendEventByName(eventTarget=GameObjectFSM(var "Atmos Surface Wind", fsm="Atmos Control"), sendEvent="STOP", delay=0, everyFrame=false)
    on FINISHED → Surface Wind C
    on OFF → Surface Wind C
  }

  state Surface Wind C {
    BoolTest(boolVariable=var "Surface Wind C", isTrue=(none), isFalse=→"OFF", everyFrame=false)
    FindChild(gameObject=var "Atmos", childName="Atmos Surface Wind C", storeResult=var "Atmos Surface Wind C")
    SendEventByName(eventTarget=GameObjectFSM(var "Atmos Surface Wind C", fsm="Atmos Control"), sendEvent="STOP", delay=0, everyFrame=false)
    on FINISHED → 
    on OFF → 
  }

  state Music {
    BoolTest(boolVariable=var "Fade Music", isTrue=(none), isFalse=→"OFF", everyFrame=false)
    FindChild(gameObject=var "GameManager", childName="Music", storeResult=var "Music Folder")  // disabled
    FindChild(gameObject=var "Music Folder", childName="Track 1", storeResult=var "Track 1")  // disabled
    SetFsmFloat(gameObject=var "Track 1", fsmName="Music Player", variableName="Fade Vol End", setValue=0, everyFrame=false)
    SetFsmFloat(gameObject=var "Track 1", fsmName="Music Player", variableName="Fade Time", setValue=0.3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Track 1"), sendEvent="FADE", delay=0, everyFrame=false)
    on FINISHED → Cave Wind
    on OFF → Cave Wind
  }

  state Door Wait {
    on DOOR ENTERED → Music
  }

  var Cave Wind: bool = false
  var Fade Music: bool = true
  var Surface Wind: bool = false
  var Surface Wind C: bool = true
  var _Is Door: bool = false

  var Atmos Cave Wind: gameObject = <null>
  var Atmos Folder: gameObject = <null>
  var Atmos Surface Wind: gameObject = <null>
  var Atmos Surface Wind C: gameObject = <null>
  var Music Folder: gameObject = <null>
  var Track 1: gameObject = <null>
}
