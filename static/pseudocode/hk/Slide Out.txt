# Slide Out
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas

fsm Slide Out {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    SetVector3Value(vector3Variable=var "Off position", vector3Value=var "Start Position", everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Off position", addX=0, addY=50, addZ=0, everyFrame=false, perSecond=false)
    FindChild(gameObject=Self, childName="Health", storeResult=var "Health")
    FindChild(gameObject=var "OC Backboard", childName="OC Backboard", storeResult=var "OC Backboard")
    FindChild(gameObject=Self, childName="Soul Orb", storeResult=var "Soul Orb")
    FindChild(gameObject=var "Soul Orb", childName="SoulOrb_fill", storeResult=var "Soul Fill")
    FindChild(gameObject=var "Soul Fill", childName="Liquid", storeResult=var "Liquid")
    on FINISHED → Idle
  }

  state Idle {
    SetBoolValue(boolVariable=var "Is HUD Out", boolValue=false, everyFrame=false)
    on OUT → Go Out
    on OUT INSTANT → Go Out Instant
  }

  state Go Out {
    SetBoolValue(boolVariable=var "Is HUD Out", boolValue=true, everyFrame=false)
    SetMeshRenderer(gameObject=var "Liquid", active=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Health"), sendEvent="HUD OUT", delay=0, everyFrame=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.75, 0.75, 0.75), time=var "Scale Time", delay=0, speed=(unset), easeType=easeInSine, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)  // disabled
    on IN → Come In
    on FINISHED → Out
  }

  state Come In {
    SetBoolValue(boolVariable=var "Is HUD Out", boolValue=false, everyFrame=false)
    SetFsmFloat(gameObject=_GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health, fsmName="Low Health FX", variableName="Wait Time", setValue=var "Scale Time", everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Start Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)  // disabled
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(1, 1, 1), time=var "Scale Time", delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)  // disabled
    on OUT → Go Out
    on FINISHED → In
    on OUT INSTANT → Go Out Instant
  }

  state Out {
    SetPosition(gameObject=Self, vector=var "Off position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on IN → Come In
  }

  state In {
    SendEventByName(eventTarget=GameObject(var "Health"), sendEvent="HUD IN", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Liquid", active=true)
    on OUT → Go Out
    on OUT INSTANT → Go Out Instant
  }

  state Go Out Instant {
    SetBoolValue(boolVariable=var "Is HUD Out", boolValue=true, everyFrame=false)
    SetMeshRenderer(gameObject=var "Liquid", active=false)
    SendEventByName(eventTarget=GameObject(var "Health"), sendEvent="HUD OUT", delay=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=0.75, y=0.75, z=0.75, everyFrame=false, lateUpdate=false)
    on FINISHED → Out
  }

  var Scale Time: float = 0.15
  var Shrink Vector: vector3 = (0.85, 0.85, 0.85)

  var Off position: vector3 = (0, 0, 0)
  var Start Position: vector3 = (0, 0, 0)
  var Health: gameObject = <null>
  var Liquid: gameObject = <null>
  var OC Backboard: gameObject = <null>
  var Soul Fill: gameObject = <null>
  var Soul Orb: gameObject = <null>
}
