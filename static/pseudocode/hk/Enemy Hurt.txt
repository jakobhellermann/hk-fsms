# Enemy Hurt
# 3 location(s):
#   Room_Colosseum_Bronze / Colosseum Manager/Ground Spikes/Colosseum Spike (19)
#   Room_Colosseum_Silver / Colosseum Manager/Ground Spikes/Colosseum Spike (19)
#   Room_Colosseum_Gold / Colosseum Manager/Ground Spikes/Colosseum Spike (19)

fsm Enemy Hurt {
  start Hurt

  state Hurt {
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=9999, everyFrame=false)
    on UNHURT ENEMY → Unhurt
  }

  state Unhurt {
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=0, everyFrame=false)
    on HURT ENEMY → Hurt
  }
}
