# Kill
# 1 location(s):
#   Ruins1_04 / Nailsmith Cliff NPC

fsm Kill {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    FindChild(gameObject=Self, childName="Hit Spawn", storeResult=var "Hit Spawn")
    FindChild(gameObject=Self, childName="Splash Point", storeResult=var "Splash Point")
    FindGameObject(objectName="Nailsmith Death", withTag="Untagged", store=var "Nailsmith Death")
    on NAIL KILL → Nail
    on SPELL KILL → Spell
  }

  state Nail {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Hit Spawn", position=(unset), rotation=(unset), storeObject=var "Slash")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Hit Spawn", position=(unset), rotation=(unset), storeObject=var "Slash")
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Slash", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Slash", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashFocusHeal())
    CreateObject(gameObject=Uninfected Hit Pt (sharedassets6.assets), spawnPoint=var "Slash", position=(unset), rotation=(0, 0, -45), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SpawnRandomObjectsV2(gameObject=Slash Effect Ghost1 (resources.assets), spawnPoint=var "Hit Spawn", position=(unset), spawnMin=2, spawnMax=3, speedMin=20, speedMax=35, angleMin=-40, angleMax=40, originVariationX=0, originVariationY=0)
    SpawnRandomObjectsV2(gameObject=Slash Effect Ghost2 (resources.assets), spawnPoint=var "Hit Spawn", position=(unset), spawnMin=2, spawnMax=3, speedMin=10, speedMax=35, angleMin=-40, angleMax=40, originVariationX=0, originVariationY=0)
    on FINISHED → Hit
  }

  state Spell {
    SetBoolValue(boolVariable=var "Spell Kill", boolValue=true, everyFrame=false)
    on FINISHED → Hit
  }

  state Hit {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["NAILSMITH_KILL"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailsmith_Death (sharedassets91.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[enemy_death_sword (resources.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[enemy_damage (resources.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="NPC CONTROL OFF", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="nailsmithKilled", value=true)
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Outside Hit")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SaveGame())
    on FINISHED → Fall
  }

  state Fall {
    ActivateAllChildren(gameObject=var "Nailsmith Death", activate=true)
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=1.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Splash
  }

  state Splash {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Splash Point", audioClips=[water_splash (sharedassets31.assets)], weights=[1], pitchMin=0.8, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Splash Out Black (resources.assets), spawnPoint=var "Splash Point", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SpawnRandomObjectsV2(gameObject=Spatter Black (resources.assets), spawnPoint=var "Splash Point", position=(0, 1, 0), spawnMin=12, spawnMax=16, speedMin=12, speedMax=20, angleMin=70, angleMax=110, originVariationX=0.5, originVariationY=0.5)
  }

  var Chooser: float = 0
  var Spell Kill: bool = false
  var Arrow Prompt(Clone): gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var Hit Spawn: gameObject = <null>
  var Nailsmith Death: gameObject = <null>
  var Self: gameObject = <null>
  var Slash: gameObject = <null>
  var Splash Point: gameObject = <null>
}
