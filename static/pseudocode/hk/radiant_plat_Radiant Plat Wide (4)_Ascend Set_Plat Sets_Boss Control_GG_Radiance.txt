// uses template: radiant_plat
fsm radiant_plat {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Colliders", storeResult=var "Colliders")
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Colliders", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Out
  }

  state Out {
    on APPEAR → Appear 1
    on IDLE → Idle
  }

  state Appear 1 {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Appear", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Appear 2
  }

  state Appear 2 {
    ActivateGameObject(gameObject=var "Colliders", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on DISAPPEAR → Disappear
    on SLOW VANISH → Pre Vanish
  }

  state Disappear {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Colliders", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on IDLE → Idle
    on APPEAR → Appear 1
    on FINISHED → Disappeared
  }

  state Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Idle", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Colliders", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on IDLE → 
    on DISAPPEAR → Disappear
    on SLOW VANISH → Pre Vanish
  }

  state Disappeared {
    SetMeshRenderer(gameObject=Self, active=false)
    on IDLE → Idle
    on APPEAR → Appear 1
  }

  state Pre Vanish {
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Vanish Antic
  }

  state Vanish Antic {
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=FlashingGhostWounded())
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Disappear
  }

  var Anim Appear: string = "R Plat Wide Appear"
  var Anim Disappear: string = "R Plat Wide Disappear"
  var Anim Idle: string = "R Plat Wide Idle"

  var Colliders: gameObject = <null>
}
