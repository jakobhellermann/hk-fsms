# Control
# 2 location(s):
#   Deepnest_30 / Mimic Spider Fake4
#   Deepnest_31 / Mimic Spider Fake1

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Detect {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on ACTIVATE → Activate
    on HIT → Disappear
  }

  state Activate {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Disappear {
    SetMeshRenderer(gameObject=Self, active=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  var Activated: bool = false
}
