# Movement
# 1 location(s):
#   sharedassets209.assets / Shot Traitor Lord

fsm Movement {
  start Tween

  state Tween {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shot")
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 9.5, 0), time=0.75, delay=0, speed=(unset), easeType=easeInOutSine, loopType=pingPong, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Steam", emit=0)
    on FINISHED → End
  }

  state Recycle {
    RecycleSelf()
  }

  state End {
    StopParticleEmitter(gameObject=var "Pt Steam")
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shot End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  var Pt Steam: gameObject = Shot Traitor Lord/Pt Steam
}
