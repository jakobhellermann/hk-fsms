# FSM
# 1 location(s):
#   sharedassets447.assets / HK Plume Prime

// uses template: hk_plume_prime
fsm FSM {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Outside Arena?
  }

  state Plume1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Plume Burst", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Plume 2
  }

  state Antic {
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=1, z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Plume Antic")
    Wait(time=1, finishEvent=→"PLUMES UP", realTime=false)
    on PLUMES UP → Plume1
    on PLUME CANCEL → Auto?
    on HK STUN → Auto?
  }

  state Plume 2 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetPolygonCollider(gameObject=Self, active=true)
    on FINISHED → Plume 3
  }

  state Plume 3 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false)
    on FINISHED → End
  }

  state Auto? {
    BoolTest(boolVariable=var "Auto", isTrue=→"AUTO", isFalse=→"FINISHED", everyFrame=false)
    on AUTO → Deactivate
    on FINISHED → Recycle
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Recycle {
    RecycleSelf()
  }

  state End {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Auto?
  }

  state Outside Arena? {
    SetMeshRenderer(gameObject=Self, active=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Pos", float2=61.08, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"OUTSIDE", everyFrame=false)
    FloatCompare(float1=var "X Pos", float2=29.86, tolerance=0, equal=(none), lessThan=→"OUTSIDE", greaterThan=(none), everyFrame=false)
    on FINISHED → Antic
    on OUTSIDE → Shift Y
  }

  state Shift Y {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=2, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Antic
  }

  var Auto: bool = false
  var Embers: gameObject = HK Plume Prime/Embers
  var Embers B: gameObject = HK Plume Prime/Embers B

  var X Pos: float = 0
}
