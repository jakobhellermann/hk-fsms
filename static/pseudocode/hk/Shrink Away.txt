# Shrink Away
# 1 location(s):
#   sharedassets34.assets / Corpse Mage Knight/Sword

fsm Shrink Away {
  start Wait

  state Wait {
    Wait(time=10, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shrink
  }

  state Shrink {
    EaseFloat(fromValue=1, toValue=0.2, floatVariable=var "Scale", time=0.25, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=var "Scale", everyFrame=false, lateUpdate=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Scale: float = 0
}
