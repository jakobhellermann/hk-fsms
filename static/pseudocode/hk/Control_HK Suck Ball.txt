# Control
# 1 location(s):
#   sharedassets409.assets / HK Suck Ball

fsm Control {
  start Init

  state Init {
    PlayParticleEmitter(gameObject=var "Pt", emit=0)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Hero Pos", addX=0, addY=-0.5, addZ=0, everyFrame=false, perSecond=false)
    RandomFloat(min=-0.5, max=0.5, storeResult=var "Translate X")
    RandomFloat(min=-0.5, max=0.5, storeResult=var "Translate Y")
    Translate(gameObject=Self, vector=(unset), x=var "Translate X", y=var "Translate Y", z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Chase
  }

  state Chase {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Hero Pos", time=0.25, delay=0, speed=(unset), space=0, easeType=easeInSine, loopType=none, orientToPath=true, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=true, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → End pt
  }

  state End pt {
    StopParticleEmitter(gameObject=var "Pt")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Translate X: float = 0
  var Translate Y: float = 0
  var Hero Pos: vector3 = (0, 0, 0)
  var Pt: gameObject = HK Suck Ball/Pt
}
