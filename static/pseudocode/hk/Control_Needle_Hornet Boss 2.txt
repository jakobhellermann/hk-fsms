# Control
# 2 location(s):
#   Deepnest_East_Hornet_boss / Hornet Boss 2/Needle
#   GG_Hornet_2 / Boss Holder/Hornet Boss 2/Needle

fsm Control {
  start Init
  on HORNET KILLED → Destroy  // from any state

  state Out {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=(unset), space=0, everyFrame=false, lateUpdate=false)
    FaceAngle(gameObject=Self, angleOffset=180, everyFrame=true)  // disabled
    FaceAngleV2(gameObject=Self, angleOffset=180, worldSpace=true, everyFrame=true)
    on FINISHED → Decel
  }

  state Decel {
    DecelerateV2(gameObject=Self, deceleration=0.8)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=true)
    FloatCompare(float1=var "Speed", float2=0, tolerance=0.5, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Return
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")  // disabled
    FindChild(gameObject=Self, childName="Thread", storeResult=var "Thread")
    GetPosition(gameObject=Self, vector=var "Return Vector", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Out
  }

  state Return {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Return Vector", time=1, delay=0, speed=30, space=1, easeType=easeInSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    DecelerateV2(gameObject=Self, deceleration=0.8)
    ActivateGameObject(gameObject=var "Thread", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Notify
  }

  state Notify {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NEEDLE RETURN", delay=0, everyFrame=false)
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Speed: float = 0
  var Return Vector: vector3 = (0, 0, 0)
  var Parent: gameObject = <null>
  var Thread: gameObject = <null>
}
