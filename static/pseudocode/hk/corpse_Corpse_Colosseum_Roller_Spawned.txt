# corpse
# 1 location(s):
#   sharedassets32.assets / Corpse_Colosseum_Roller_Spawned

fsm corpse {
  start Initiate

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    SetProperty(targetProperty=ParticleSystem.enableEmission on Corpse_Colosseum_Roller_Spawned/Corpse Flame@ParticleSystem, everyFrame=false)
    Wait(time=0.1, finishEvent=(none), realTime=false)
    on FINISHED → Flame Check
  }

  state In Air {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LANDED", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LANDED → Landed
  }

  state Landed {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")
    FloatCompare(float1=var "X Speed", float2=0, tolerance=2, equal=→"LANDED", lessThan=(none), greaterThan=(none), everyFrame=true)
    on LANDED → Unfurl
  }

  state Unfurl {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=0.2, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Scale Vector", vector3Value=(unset), x=var "X Scale", y=0.2, z=0.2, everyFrame=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Scale Vector", time=0.2, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetProperty(targetProperty=ParticleSystem.enableEmission on Corpse_Colosseum_Roller_Spawned/Corpse Flame@ParticleSystem, everyFrame=false)
    SetProperty(targetProperty=ParticleSystem.enableEmission on Corpse_Colosseum_Roller_Spawned/Corpse Steam@ParticleSystem, everyFrame=false)
    on FINISHED → Destroy
  }

  state Flame Check {
    BoolTest(boolVariable=var "spellBurn", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Start Flame
    on FALSE → In Air
  }

  state Start Flame {
    Tk2dSpriteSetColor(gameObject=Self, color=(0.19607843, 0.19607843, 0.19607843, 1), everyframe=false)
    SetProperty(targetProperty=ParticleSystem.enableEmission on Corpse_Colosseum_Roller_Spawned/Corpse Flame@ParticleSystem, everyFrame=false)
    on FINISHED → In Air
  }

  state Destroy {
    CreateObject(gameObject=Death Puff Mini, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    DestroySelf(detachChildren=true)
  }

  var Scale: float = 0
  var X Scale: float = 0
  var X Speed: float = 0
  var spellBurn: bool = false
  var Velocity: vector2 = (0, 0)
  var Scale Vector: vector3 = (0, 0, 0)
  var Colour: color = (0, 0, 0, 1)
  var Self: gameObject = <null>
  var Unfurler: gameObject = <null>
}
