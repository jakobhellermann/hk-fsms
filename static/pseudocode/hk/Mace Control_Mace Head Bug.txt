# Mace Control
# 1 location(s):
#   sharedassets48.assets / Mace Head Bug

fsm Mace Control {
  start Init

  state Emerge {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mace Emerge")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emerge Hit
    on COLLIDE → Check Damage
  }

  state Leave R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mace Leave")
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=5, y=(unset), everyFrame=false)
    ReceivedDamage(collideTag="", sendEvent=→"COLLIDE", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on COLLIDE → Check Damage 2
  }

  state Check Direction {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[sword_hit_reject (resources.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[ceiling_dropper_look_1, ceiling_dropper_look_2, ceiling_dropper_look_3, ceiling_dropper_look_4, ceiling_dropper_look_5], weights=[1, 1, 1, 1, 1], pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Hit L
    on RIGHT → Hit R
    on UP → Hit D
    on DOWN → Hit D
  }

  state Hit L {
    CreateObject(gameObject=Block Hit (resources.assets), spawnPoint=var "Self", position=(1, 0, 0), rotation=(0, 0, 180), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=1050, everyFrame=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=-40, y=(unset), everyFrame=false)
    on FINISHED → Rolling
  }

  state Rolling {
    DecelerateV2(gameObject=Self, deceleration=0.885)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=700, everyFrame=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mace Roll")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(collideTag="", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)  // disabled
    on FINISHED → Rolling Hit
  }

  state Stop {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)  // disabled
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    ReceivedDamage(collideTag="", sendEvent=→"COLLIDE", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on WAIT → Flip Back
    on COLLIDE → Check Damage 5
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.0055, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Emerge
  }

  state Direction? {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Leave R
    on LEFT → Leave L
  }

  state Leave L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mace Leave")
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-5, y=(unset), everyFrame=false)
    ReceivedDamage(collideTag="", sendEvent=→"COLLIDE", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on COLLIDE → Check Damage 3
  }

  state Hit R {
    CreateObject(gameObject=Block Hit (resources.assets), spawnPoint=var "Self", position=(1, 0, 0), rotation=(0, 0, 0), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=-1050, everyFrame=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=40, y=(unset), everyFrame=false)
    on FINISHED → Rolling
  }

  state Hit D {
    CreateObject(gameObject=Block Hit (resources.assets), spawnPoint=var "Self", position=(0, 1, 0), rotation=(0, 0, 90), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Emerge
  }

  state Flip Back {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Emerge
  }

  state Check Damage {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Check Direction
    on CANCEL → Cancel Pause
  }

  state Cancel Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Emerge Hit
  }

  state Check Damage 2 {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Check Direction
    on CANCEL → Cancel Pause 2
  }

  state Cancel Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Leave R
  }

  state Check Damage 3 {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Check Direction
    on CANCEL → Cancel Pause 3
  }

  state Cancel Pause 3 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Leave L
  }

  state Check Damage 4 {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Check Direction
    on CANCEL → Cancel Pause 4
  }

  state Cancel Pause 4 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Rolling
  }

  state Check Damage 5 {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Check Direction
    on CANCEL → Cancel Pause 5
  }

  state Cancel Pause 5 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Stop
  }

  state Rolling Hit {
    DecelerateV2(gameObject=Self, deceleration=0.885)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(collideTag="", sendEvent=→"COLLIDE", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on FINISHED → Stop
    on COLLIDE → Check Damage 4
  }

  state Emerge Hit {
    ReceivedDamage(collideTag="", sendEvent=→"COLLIDE", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on FINISHED → Direction?
    on COLLIDE → Check Damage
  }

  var Attack Direction: float = 0
  var Damage: int = 0
  var Facing Right: bool = false
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
