# Spa Region
# 7 location(s):
#   Deepnest_30 / Spa Region
#   GG_Atrium / Spa Region
#   GG_Spa / Spa Region
#   GG_Atrium_Roof / Spa Region
#   Crossroads_30 / Spa Region
#   Ruins_Bathhouse / Spa Region
#   Ruins1_04 / Spa Region

fsm Spa Region {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"ENTER", storeCollider=var "Hero")
    on ENTER → Big Splash?
  }

  state Enter {
    SetPlayerDataInt(intName="environmentType", value=3)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(0.2, -0.7, 0), spawnMin=6, spawnMax=8, speedMin=5, speedMax=15, angleMin=45, angleMax=70, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(-0.2, -0.7, 0), spawnMin=6, spawnMax=8, speedMin=5, speedMax=15, angleMin=115, angleMax=135, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=0.8, pitchMax=0.95, volume=1, delay=0, storePlayer=<null>)
    SetParent(gameObject=var "Glows", parent=var "Hero", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Steam", parent=var "Hero", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetParent(gameObject=var "Glow Front", parent=var "Hero", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Glows", vector=(unset), x=0, y=-0.68, z=0.1, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Glow Front", vector=(unset), x=0, y=-1, z=-0.01, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Steam", vector=(unset), x=0, y=-0.04, z=0.106, space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Spa Ripple", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParticleEmission(gameObject=var "Glows", emission=true)
    SetParticleEmission(gameObject=var "Glow Front", emission=true)
    SetParticleEmission(gameObject=var "Steam", emission=true)
    SpawnObjectFromGlobalPool(gameObject=Splash In Small (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Splash")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Self Y", add=0, everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Splash", vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Stay
  }

  state Stay {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Y", add=0.31, everyFrame=true, perSecond=false)
    SetPosition(gameObject=var "Spa Ripple", vector=(unset), x=var "Hero X", y=var "Self Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Player", collideLayer="", sendEvent=→"LEAVE", storeCollider=var "Hero")
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on LEAVE → Leave
    on WAIT → Heal Pause
  }

  state Leave {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=1, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataInt(intName="environmentType", value=var "Scene Enviro Type")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP HEAL", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Glows", emission=false)
    SetParticleEmission(gameObject=var "Glow Front", emission=false)
    SetParticleEmission(gameObject=var "Steam", emission=false)
    SetParent(gameObject=var "Glows", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Glow Front", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Steam", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    ActivateGameObject(gameObject=var "Spa Ripple", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Splash Out Small (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Splash")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Self Y", add=0, everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Splash", vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Detect
  }

  state Heal Pause {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Y", add=0.31, everyFrame=true, perSecond=false)
    SetPosition(gameObject=var "Spa Ripple", vector=(unset), x=var "Hero X", y=var "Self Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Player", collideLayer="", sendEvent=→"LEAVE", storeCollider=var "Hero")
    SendMessageV2(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddMPCharge(1), everyFrame=true)  // disabled
    Wait(time=0.5, finishEvent=→"WAIT", realTime=false)
    on WAIT → Healing
    on LEAVE → Leave
  }

  state Heal {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=spa_heal (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Plink effect (sharedassets31.assets), spawnPoint=var "Hero", position=(0, -0.4, 0), rotation=(unset), storeObject=var "Plink", networkInstantiate=false, networkGroup=0)
    SendMessageV2(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddMPCharge(100), everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddHealth(1))
    on FINISHED → Heal Pause
  }

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="environmentType", storeValue=var "Scene Enviro Type")
    FindChild(gameObject=Self, childName="Ripple", storeResult=var "Spa Ripple")
    on FINISHED → Detect
  }

  state Max Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxHealth", storeValue=var "Max HP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPCharge", storeValue=var "MP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxMP", storeValue=var "Max MP")
    IntTestToBool(int1=var "HP", int2=var "Max HP", equalBool=var "Maxed HP", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    IntTestToBool(int1=var "MP", int2=var "Max MP", equalBool=var "Maxed MP", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    BoolAllTrue(boolVariables=[var "Maxed HP", var "Maxed MP"], sendEvent=→"MAXED", storeResult=(unset), everyFrame=false)
    on MAXED → Heal Pause
    on FINISHED → Heal
  }

  state Healing {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="START HEAL", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Player", collideLayer="", sendEvent=→"LEAVE", storeCollider=var "Hero")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Y", add=0.31, everyFrame=true, perSecond=false)
    SetPosition(gameObject=var "Spa Ripple", vector=(unset), x=var "Hero X", y=var "Self Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on LEAVE → Leave
    on FINISHED → Heal Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Big Splash? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["willHardLand"], storeResult=var "Will Hard Land")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["spellQuake"], storeResult=var "Quaking")
    BoolTest(boolVariable=var "Quaking", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Will Hard Land", isTrue=→"BIG", isFalse=→"NORMAL", everyFrame=false)
    on NORMAL → Enter
    on BIG → Splash In Big
  }

  state Splash In Big {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(0, -0.7, 0), spawnMin=60, spawnMax=60, speedMin=10, speedMax=30, angleMin=80, angleMax=100, originVariationX=1, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=0.8, pitchMax=0.95, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Splash In Black (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Splash")
    SetScale(gameObject=var "Splash", vector=(unset), x=2.5, y=2.5, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=var "Splash", color=var "Colour", everyframe=false)
    SpawnObjectFromGlobalPool(gameObject=var "Splash Out Obj", spawnPoint=var "Hero", position=(0, 1, 0), rotation=(unset), storeObject=var "Splash")
    SetScale(gameObject=var "Splash", vector=(unset), x=2, y=2, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Enter
  }

  var Hero Surface Y: float = 0
  var Hero X: float = 0
  var Self Y: float = 0
  var HP: int = 0
  var MP: int = 0
  var Max HP: int = 0
  var Max MP: int = 0
  var Scene Enviro Type: int = 0
  var Maxed HP: bool = false
  var Maxed MP: bool = false
  var Quaking: bool = false
  var Will Hard Land: bool = false
  var Colour: color = (0, 0, 0, 1)
  var Glow Front: gameObject = Spa Region/Spa Glows Front
  var Glows: gameObject = Spa Region/Spa Glows
  var Hero: gameObject = <null>
  var Plink: gameObject = <null>
  var Spa Ripple: gameObject = <null>
  var Spatter Obj: gameObject = <null>
  var Splash: gameObject = <null>
  var Splash Out Obj: gameObject = <null>
  var Steam: gameObject = Spa Region/Spa Steam
}
