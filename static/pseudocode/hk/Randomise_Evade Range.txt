#   Fungus1_26 / Moss Knight/Evade Range
#   Fungus1_32 / Moss Knight B/Evade Range
#   Fungus1_32 / Moss Knight C/Evade Range

fsm Randomise {
  start On

  state On {
    WaitRandom(timeMin=2, timeMax=2.5, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Off
  }

  state Off {
    WaitRandom(timeMin=2.5, timeMax=3.5, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Range Detect", variableName="In Attack Range", setValue=false, everyFrame=false)
    on FINISHED → On
  }
}
