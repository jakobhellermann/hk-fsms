# Map Msg
# 1 location(s):
#   sharedassets7.assets / Map Update Msg

fsm Map Msg {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Animation", storeResult=var "Animation")
    FindChild(gameObject=Self, childName="Text", storeResult=var "Text")
    SetPosition(gameObject=Self, vector=var "Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    BoolTest(boolVariable=var "Map Shop", isTrue=→"MAP SHOP", isFalse=(none), everyFrame=false)
    on FINISHED → Set
    on MAP SHOP → Map Shop Pos
  }

  state Set {
    GetLanguageString(sheetName="UI", convName="MAP_UPDATED", storeValue=var "Text Str")
    Tk2dPlayAnimation(gameObject=var "Animation", animLibName="", clipName="Map Writing")
    SetTextMeshProText(gameObject=var "Text", textString=var "Text Str")
    SetStringValue(stringVariable=var "Anim Down", stringValue="Entry Full Down", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Unseen
  }

  state Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Map Complete
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroyObject(gameObject=var "Self", delay=0, detachChildren=false)
  }

  state Seen {
    ActivateGameObject(gameObject=var "Backboard Small", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=0.8, y=0.8, z=0.8, everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Wait Time", floatValue=3, everyFrame=false)
    on FINISHED → Up
  }

  state Unseen {
    ActivateGameObject(gameObject=var "Backboard Large", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetFloatValue(floatVariable=var "Wait Time", floatValue=3, everyFrame=false)
    on FINISHED → Up
  }

  state Map Complete {
    Tk2dPlayAnimation(gameObject=var "Animation", animLibName="", clipName="Map Complete")
    Wait(time=0, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  state Map Shop Pos {
    SetPosition(gameObject=Self, vector=(unset), x=-10.1, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Set
  }

  var Position: vector3 = (-1.29, -6.83, 0)

  var Wait Time: float = 0
  var Full: bool = false
  var Map Shop: bool = false
  var Anim Down: string = ""
  var Text Str: string = ""
  var Animation: gameObject = <null>
  var Backboard Large: gameObject = Map Update Msg/Backboard Large
  var Backboard Small: gameObject = Map Update Msg/Backboard Small
  var Self: gameObject = <null>
  var Text: gameObject = <null>
}
