# Check State
# 1 location(s):
#   Ruins2_04 / Bathhouse Door

fsm Check State {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Opened?
  }

  state Opened? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bathHouseOpened", isTrue=→"OPENED", isFalse=(none))
    on OPENED → Opened
    on FINISHED → Have Key?
  }

  state Opened {
    ActivateGameObject(gameObject=var "Door Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Door Open", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Door Object", active=true)
  }

  state Have Key? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasSpaKey", isTrue=→"KEY", isFalse=→"NO KEY")  // disabled
    GetPlayerDataInt(gameObject=var "GameManager", intName="simpleKeys", storeValue=var "Simple Keys")
    IntCompare(integer1=var "Simple Keys", integer2=0, equal=→"NO KEY", lessThan=→"NO KEY", greaterThan=→"KEY", everyFrame=false)
    on KEY → Key
    on NO KEY → No Key
  }

  state Key {
    SetCollider(gameObject=var "Door Object", active=false)  // disabled
  }

  state No Key {
    ActivateGameObject(gameObject=var "Door Object", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
  }

  var Simple Keys: int = 0
  var Door Closed: gameObject = Bathhouse Door/Closed
  var Door Object: gameObject = Bathhouse Door/door_Ruin_Elevator
  var Door Open: gameObject = Bathhouse Door/Open
  var Inspect: gameObject = Bathhouse Door/Inspect
}
