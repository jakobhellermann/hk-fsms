# Jellyfish
# 1 location(s):
#   sharedassets466.assets / Jellyfish GG

fsm Jellyfish {
  start Init
  on ZERO HP → Detach  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    SetVelocity2d(gameObject=Self, vector=(0, 1.9), x=(unset), y=(unset), everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=141.7, tolerance=0, equal=→"END", lessThan=(none), greaterThan=(none), everyFrame=true)
    on END → End
  }

  state Detach {
    DestroyObject(gameObject=var "Tent Box", delay=0, detachChildren=false)
    DetachChildren(gameObject=Self)
  }

  state End {
    DestroySelf(detachChildren=false)
  }

  var Chooser: float = 0
  var Self Y: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var X Roam: float = 0.5
  var Tent Box: gameObject = Jellyfish GG/Tentacle Box
}
