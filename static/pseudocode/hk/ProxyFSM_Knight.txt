# ProxyFSM
# 1 location(s):
#   Knight_Pickup / Knight

fsm ProxyFSM {
  start Init

  state Idle {
    on HeroCtrl-HeroDamaged → Flower?
    on HeroCtrl-LeftGround → Left Ground
    on HeroCtrl-FocusCompleted → Focus Completed
    on HeroCtrl-LeavingScene → Leaving Scene
    on HeroCtrl-Respawned → Respawn
    on HeroCtrl-Landed → Landed
    on HeroCtrl-Healed → Healed
    on HeroCtrl-MaxHealth → Healed Max
    on HeroCtrl-TookBlockerHit → Blocker Hit
    on HeroCtrl-EnterSuperDash → Enter Super Dash
    on HeroCtrl-ShadowDashEnd → 
    on HeroCtrl-DashEnd → End Dash
    on HeroCtrl-EnterQuake → Enter Quake
  }

  state Damaged {
    SendEventByName(eventTarget=GameObject(var "Charm Effects"), sendEvent="HERO DAMAGED", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Charm Effects", sendEvent="HERO DAMAGED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="HERO DAMAGED", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=Self, sendEvent="HERO DAMAGED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="HERO DAMAGED", delay=0, everyFrame=false)
    SendEventToRegister(eventName="HERO DAMAGED")
    SendEventByName(eventTarget=GameObject(var "Inventory"), sendEvent="HERO DAMAGED", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Inventory", sendEvent="HERO DAMAGED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="disablePause", isTrue=→"BROADCAST", isFalse=(none))
    on FINISHED → Idle
    on BROADCAST → Broadcast Damaged
  }

  state Left Ground {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LEFT GROUND", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=Self, sendEvent="LEFT GROUND", everyFrame=false)
    on FINISHED → Idle
  }

  state Focus Completed {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FOCUS COMPLETED", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=Self, sendEvent="FOCUS COMPLETED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="HERO HEALED", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Charm Effects"), sendEvent="HERO HEALED", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Charm Effects", sendEvent="HERO HEALED", everyFrame=false)
    on FINISHED → Idle
  }

  state Leaving Scene {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LEAVING SCENE", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=Self, sendEvent="LEAVING SCENE", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Charm Effects"), sendEvent="LEAVING SCENE", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Charm Effects", sendEvent="LEAVING SCENE", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Effects"), sendEvent="LEAVING SCENE", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Effects", sendEvent="LEAVING SCENE", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Inventory"), sendEvent="INVENTORY CANCEL", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Inventory", sendEvent="LEAVING SCENE", everyFrame=false)
    on FINISHED → Idle
  }

  state Respawn {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="HERO RESPAWNED", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHARM INDICATOR CHECK", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Landed {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="HERO LANDED", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=Self, sendEvent="HERO LANDED", everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    FindChild(gameObject=Self, childName="White_Flower_Break", storeResult=var "White_Flower_Break")
    FindChild(gameObject=var "HUD Camera", childName="Inventory", storeResult=var "Inventory")
    FindChild(gameObject=Self, childName="Charm Effects", storeResult=var "Charm Effects")
    FindChild(gameObject=Self, childName="Effects", storeResult=var "Effects")
    FindChild(gameObject=var "Effects", childName="Acid Armour", storeResult=var "Acid Armour")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Healed Max {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="HERO HEALED FULL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Charm Effects"), sendEvent="HERO HEALED FULL", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Charm Effects", sendEvent="HERO HEALED FULL", everyFrame=false)
    on FINISHED → Idle
  }

  state Healed {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="HERO HEALED", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Charm Effects"), sendEvent="HERO HEALED", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=var "Charm Effects", sendEvent="HERO HEALED", everyFrame=false)
    on FINISHED → Idle
  }

  state Blocker Hit {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BLOCKER HIT", delay=0, everyFrame=false)
    on FINISHED → Damaged
  }

  state Enter Super Dash {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ENTER SUPER DASH", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state End Dash {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DASH END", delay=0, everyFrame=false)  // disabled
    SendEventToGameObjectOptimized(target=Self, sendEvent="DASH END", everyFrame=false)
    on FINISHED → Idle
  }

  state Flower? {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasXunFlower", falseBool="xunFlowerBroken", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="xunFlowerBroken", value=true)
    ActivateGameObject(gameObject=var "White_Flower_Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="xunFlowerBrokeTimes")
    SpawnObjectFromGlobalPool(gameObject=Relic Get Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg")
    FindChild(gameObject=var "Msg", childName="Text", storeResult=var "Msg Text")
    FindChild(gameObject=var "Msg", childName="Icon", storeResult=var "Msg Item")
    SetSpriteRendererSprite(gameObject=var "Msg Item", sprite=White_Flower_Half (resources.assets))
    GetLanguageString(sheetName="UI", convName="NOTIFICATION_FLOWER_BREAK", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SaveGame())
    on FINISHED → Damaged
  }

  state Enter Quake {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ENTER QUAKE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Broadcast Damaged {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="HERO DAMAGED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Faced Nightmare: bool = false
  var Faced Radiance: bool = false
  var Faced Zote: bool = false
  var No Charms: bool = false
  var Item Name: string = ""
  var Acid Armour: gameObject = <null>
  var Charm Effects: gameObject = <null>
  var Effects: gameObject = <null>
  var Inventory: gameObject = <null>
  var Msg: gameObject = <null>
  var Msg Item: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Self: gameObject = <null>
  var White_Flower_Break: gameObject = <null>
}
