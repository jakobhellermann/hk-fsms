# Update Cursor
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Inv
#   resources.assets / _GameCameras/HudCamera/Inventory/Inv

fsm Update Cursor {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Cursor", storeResult=var "Cursor")
    on FINISHED → Idle
  }

  state Update {
    GetPosition(gameObject=var "Item", vector=var "Item Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    BoundsBoxCollider(gameObject1=var "Item", scaleVector2=var "Box Bounds", scaleX=(unset), scaleY=(unset), everyFrame=false)
    BoxColliderOffset(gameObject1=var "Item", offsetVector2=(unset), offsetX=var "Box Offset X", offsetY=var "Box Offset Y", everyFrame=false)
    SetFsmVector3(gameObject=var "Cursor", fsmName="Cursor Movement", variableName="MoveToPos", setValue=var "Item Pos", everyFrame=false)
    SetFsmVector2(gameObject=var "Cursor", fsmName="Cursor Movement", variableName="ColliderBounds", setValue=var "Box Bounds", everyFrame=false)
    SetFsmFloat(gameObject=var "Cursor", fsmName="Cursor Movement", variableName="Box Offset X", setValue=var "Box Offset X", everyFrame=false)
    SetFsmFloat(gameObject=var "Cursor", fsmName="Cursor Movement", variableName="Box Offset Y", setValue=var "Box Offset Y", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Cursor"), sendEvent="CURSOR MOVE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UPDATE TEXT", delay=0, everyFrame=false)  // disabled
    on FINISHED → Idle
  }

  state Idle {
    on UPDATE CURSOR → Update
  }

  var Box Offset X: float = 0
  var Box Offset Y: float = 0
  var Box Bounds: vector2 = (0, 0)
  var Item Pos: vector3 = (0, 0, 0)
  var Cursor: gameObject = <null>
  var Item: gameObject = <null>
}
