# Vessel Fragment UI
# 1 location(s):
#   sharedassets10.assets / Vessel Fragment UI

fsm Vessel Fragment UI {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=heartpiece_decorative_ui, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    FindChild(gameObject=Self, childName="Blanker", storeResult=var "Blanker")
    FindChild(gameObject=Self, childName="Fleur", storeResult=var "Fleur")
    FindChild(gameObject=Self, childName="Piece", storeResult=var "Piece")
    FindChild(gameObject=Self, childName="Mover", storeResult=var "Mover")
    FindChild(gameObject=Self, childName="Mover Trail", storeResult=var "Mover Trail")
    FindChild(gameObject=Self, childName="Get Effect", storeResult=var "Get Effect")
    FindChild(gameObject=Self, childName="Get Glow", storeResult=var "Get Glow")
    on FINISHED → Darken
  }

  state Darken {
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.5176471), colorVariable=var "Blanker Colour", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Blanker", color=var "Blanker Colour", everyframe=true)
    on FINISHED → Fleur
  }

  state Fleur {
    SetMeshRenderer(gameObject=var "Fleur", active=true)
    Tk2dPlayAnimation(gameObject=var "Fleur", animLibName="", clipName="Fleur Appear")
    IntSwitch(intVariable=var "Pieces", compareTo=[1, 2, 3], sendEvent=[→"ONE", →"TWO", →"THREE"], everyFrame=false)
    on ONE → Get One
    on TWO → Get Two
    on THREE → Get Three
  }

  state Get One {
    SetStringValue(stringVariable=var "Get Anim", stringValue="Get 1", everyFrame=false)
    SetStringValue(stringVariable=var "Piece Land Particles", stringValue="Piece Land 1", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Get 0")
    SetPosition(gameObject=var "Piece", vector=(unset), x=(unset), y=0.36, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    on FINISHED → Piece Ready
  }

  state Get Anim {
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName=var "Get Anim")
    Tk2dPlayAnimationWithEvents(gameObject=var "Piece", clipName=var "Get Anim", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Piece Get Particles
  }

  state Fade {
    SetColorValue(colorVariable=var "Piece Colour", color=(1, 1, 1, 1), everyFrame=false)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Piece Colour", time=0.5, speed=(unset), delay=0.5, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=var "Piece Colour", everyframe=true)
    Tk2dSpriteSetColor(gameObject=var "Fleur", color=var "Piece Colour", everyframe=true)
    EaseColor(fromValue=(1, 1, 1, 0.5176471), toValue=(1, 1, 1, 0), colorVariable=var "Blanker Colour", time=0.5, speed=(unset), delay=0.5, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Blanker", color=var "Blanker Colour", everyframe=true)
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on WAIT → Hero Anim End
  }

  state Destroy Self {
    SetGravity2dScale(gameObject=var "Hero", gravityScale=0.79)
    SetPlayerDataBool(boolName="isInvincible", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    SetPlayerDataBool(boolName="disablePause", value=false)
    DestroySelf(detachChildren=false)
  }

  state Piece Fade {
    SetMeshRenderer(gameObject=var "Piece", active=true)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Piece Colour", time=1, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=var "Piece Colour", everyframe=true)
    Wait(time=1.8, finishEvent=→"WAIT", realTime=false)
    on WAIT → Get Anim 2
  }

  state Get Two {
    SetStringValue(stringVariable=var "Get Anim", stringValue="Get 2", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Got 1")
    SetStringValue(stringVariable=var "Piece Land Particles", stringValue="Piece Land 2", everyFrame=false)
    SetPosition(gameObject=var "Piece", vector=(unset), x=(unset), y=-0.2, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    on FINISHED → Piece Fade
  }

  state Check Max {
    IntCompare(integer1=var "Pieces", integer2=3, equal=→"MAX UP", lessThan=→"FINISHED", greaterThan=→"MAX UP", everyFrame=false)
    on FINISHED → Fade
    on MAX UP → Fuse
  }

  state Fuse {
    SetPlayerDataInt(intName="vesselFragments", value=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=final_heart_piece, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Fleur", animLibName="", clipName="Fleur Disappear")
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Piece Colour", time=0.5, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=var "Piece Colour", everyframe=true)
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    PlayParticleEmitter(gameObject=var "Mover Trail", emit=0)
    on WAIT → Tween Mover
  }

  state Piece Get Particles {
    ActivateGameObject(gameObject=var "Get Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=new_heartpiece_puzzle_bit (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName=var "Piece Land Particles", storeResult=var "Piece Get Object")
    ActivateGameObject(gameObject=var "Piece Get Object", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    Wait(time=1.5, finishEvent=→"WAIT", realTime=false)
    on WAIT → Check Max
  }

  state Tween Mover {
    iTweenMoveTo(gameObject=var "Mover Trail", id="", transformPosition=var "", vectorPosition=(-13.76, 6.1, 0), time=0.75, delay=0, speed=(unset), space=1, easeType=easeInBack, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=true, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleTo(gameObject=var "Mover Trail", id="", transformScale=var "", vectorScale=(0.2, 0.2, 0.2), time=0.75, delay=0, speed=(unset), easeType=easeInBack, loopType=none, startEvent=(none), finishEvent=→"WAIT", realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on WAIT → Get
  }

  state Get {
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Mover Trail")
    PlayParticleEmitter(gameObject=var "Get Effect", emit=0)
    DestroyObject(gameObject=var "Piece", delay=0, detachChildren=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=spell_pickup_final_boom, pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → PD Max Up
  }

  state Max Up {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NEW SOUL ORB", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=StoryRecord_maxSoulUp())
    on FINISHED → Fade
  }

  state Hero Anim End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Heart Piece End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy Self
  }

  state Piece Ready {
    SetMeshRenderer(gameObject=var "Piece", active=true)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=(1, 1, 1, 1), everyframe=true)
    Wait(time=0.8, finishEvent=→"WAIT", realTime=false)
    on WAIT → Get Anim
  }

  state Get Anim 2 {
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName=var "Get Anim")
    Tk2dPlayAnimationWithEvents(gameObject=var "Piece", clipName=var "Get Anim", animationTriggerEvent=→"FINISHED", animationCompleteEvent=→"FINISHED")
    on FINISHED → Piece Get Particles
  }

  state Get Three {
    SetStringValue(stringVariable=var "Get Anim", stringValue="Get 3", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Got 2")
    SetStringValue(stringVariable=var "Piece Land Particles", stringValue="Piece Land 3", everyFrame=false)
    SetPosition(gameObject=var "Piece", vector=(unset), x=(unset), y=-0.2, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    on FINISHED → Piece Fade 2
  }

  state Piece Fade 2 {
    SetMeshRenderer(gameObject=var "Piece", active=true)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Piece Colour", time=1, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=var "Piece Colour", everyframe=true)
    Wait(time=1.8, finishEvent=→"WAIT", realTime=false)
    on WAIT → Get Anim 2
  }

  state PD Max Up {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddToMaxMPReserve(33))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Max Up
  }

  var HP: int = 0
  var Max HP: int = 0
  var Pieces: int = 1
  var Get Anim: string = ""
  var Piece Land Particles: string = ""
  var Blanker Colour: color = (1, 1, 1, 0)
  var Mover Colour: color = (1, 1, 1, 0)
  var Piece Colour: color = (1, 1, 1, 0)
  var Blanker: gameObject = <null>
  var Camera: gameObject = <null>
  var Fleur: gameObject = <null>
  var Get Effect: gameObject = <null>
  var Get Glow: gameObject = <null>
  var Hero: gameObject = <null>
  var Mover: gameObject = <null>
  var Mover Trail: gameObject = <null>
  var Piece: gameObject = <null>
  var Piece Get Object: gameObject = <null>
  var Self: gameObject = <null>
}
