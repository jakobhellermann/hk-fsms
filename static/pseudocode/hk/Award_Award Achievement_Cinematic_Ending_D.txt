  start Pause

  state Pause {

  }

  state Award {

    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="QueueAchievement", parameters=["ENDINGD"], storeResult=(unset var))
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SaveGame())
  }
}
