# Rotate
# 1 location(s):
#   Town / _Scenery/mine_entrance/mine_entrance_0016_17

fsm Rotate {
  start Idle

  state Idle {
    on MOVING UP → Moving Up
    on MOVING DOWN → Moving Down
  }

  state Moving Up {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=500, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on MOVING DOWN → Moving Down
    on STOP MOVING → Idle
  }

  state Moving Down {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-500, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on MOVING UP → Moving Up
    on STOP MOVING → Idle
  }
}
