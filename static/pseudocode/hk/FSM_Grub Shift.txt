  start Init

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SetPosition(gameObject=var "Bottle", vector=(unset), x=23.18028, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on ENTER → Away
  }

  state Away {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SetPosition(gameObject=var "Bottle", vector=(unset), x=999, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on EXIT → Idle
  }

  state Init {
    FindChild(gameObject=Self, childName="Grub Bottle", storeResult=var "Bottle")
    on FINISHED → Idle
  }

  var Bottle: gameObject = <null>
}
