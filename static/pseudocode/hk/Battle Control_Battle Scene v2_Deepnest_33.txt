# Battle Control
# 1 location(s):
#   Deepnest_33 / Battle Scene v2

fsm Battle Control {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Wave 1
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="CameraLockArea B", withTag="Untagged", store=var "Camera Lock")
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Wave 2", storeResult=var "Wave 2")
    FindChild(gameObject=Self, childName="Wave 3", storeResult=var "Wave 3")
    FindChild(gameObject=Self, childName="Wave 4", storeResult=var "Wave 4")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activate
  }

  state Wave 1 {
    TransitionToAudioSnapshot(snapshot=Action (resources.assets), transitionTime=1)
    SetIntValue(intVariable=var "Battle Enemies", intValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Wave 1"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=2, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    Wait(time=15, finishEvent=→"NEXT", realTime=false)
    on NEXT → Wave 2
  }

  state End Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=5)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    DestroyObject(gameObject=var "Wave 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 2", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 3", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 4", delay=0, detachChildren=false)
  }

  state Wave 2 {
    SendEventByName(eventTarget=GameObject(var "Wave 2"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Battle Enemies", add=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=2, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    Wait(time=15, finishEvent=→"NEXT", realTime=false)
    on NEXT → Wave 3
  }

  state Wave 3 {
    SendEventByName(eventTarget=GameObject(var "Wave 3"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Battle Enemies", add=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=2, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    Wait(time=15, finishEvent=→"NEXT", realTime=false)
    on NEXT → Wave 4
  }

  state Wave 4 {
    SendEventByName(eventTarget=GameObject(var "Wave 4"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Battle Enemies", add=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    on NEXT → End Wait
  }

  var Battle Enemies: int = 0
  var Activated: bool = false
  var Camera Lock: gameObject = <null>
  var Moss Knight: gameObject = <null>
  var Self: gameObject = <null>
  var Wave 1: gameObject = <null>
  var Wave 2: gameObject = <null>
  var Wave 3: gameObject = <null>
  var Wave 4: gameObject = <null>
}
