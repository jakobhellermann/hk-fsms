# Spike Hit Effect
# 3 location(s):
#   Ruins1_28 / ruind_bridge_roof_04_spikes
#   Deepnest_30 / _Scenery/Deep Spikes (11)
#   Ruins1_17 / ruind_bridge_roof_01 (1)/ruind_bridge_roof_04_spikes (1)

fsm Spike Hit Effect {
  start Initiate
  on SPIKE RECOIL → Invincible Pause  // from any state

  state Detecting {
    SetScale(gameObject=var "Corpse Instance", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Corpse Instance", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ReceivedDamage(collideTag="Nail Attack", sendEvent=→"TAKE DAMAGE", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on TAKE DAMAGE → Get Damager Parameters
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    GetScale(gameObject=var "Self", vector=(unset), xScale=var "X Scale", yScale=var "Y Scale", zScale=(unset), space=0, everyFrame=false)
    FindGameObject(objectName=var "Send KILLED to", withTag="Untagged", store=var "KILLED GameObject")
    on FINISHED → Detecting
  }

  state Get Damager Parameters {
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="magnitudeMult", storeValue=var "Attack Magnitude", everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Attack Strength", everyFrame=false)
    GetFsmBool(gameObject=var "Damager", fsmName="damages_enemy", variableName="circleDirection", storeValue=var "Circle Direction", everyFrame=false)
    on FINISHED → Damage Zero?
  }

  state Blocked Hit {
    RandomFloat(min=0.85, max=1.15, storeResult=var "Chooser")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))  // disabled
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetAudioPitch(gameObject=var "Self", pitch=var "Chooser", everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[sword_hit_reject (resources.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SendEvent(eventTarget=GameObject(var "Self"), sendEvent=→"BLOCKED HIT", delay=0, everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on RIGHT → Blocked Right
    on LEFT → Blocked Left
    on UP → Blocked Up
    on DOWN → Blocked Down
  }

  state Blocked Right {
    GetPosition(gameObject=var "Damager", vector=(unset), x=(unset), y=var "Damager Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Damager X", y=(unset), z=(unset), space=0, everyFrame=false)
    BoxColliderOffset(gameObject1=var "Self", offsetVector2=(unset), offsetX=var "Collider Origin X", offsetY=(unset), everyFrame=false)
    FloatAdd(floatVariable=var "Damager X", add=var "Collider Origin X", everyFrame=false, perSecond=false)
    BoundsBoxCollider(gameObject1=var "Self", scaleVector2=(unset), scaleX=var "Collider Width", scaleY=(unset), everyFrame=false)
    FloatDivide(floatVariable=var "Collider Width", divideBy=2, everyFrame=false)
    FloatSubtract(floatVariable=var "Damager X", subtract=var "Collider Width", everyFrame=false, perSecond=false)
    SetVector3XYZ(vector3Variable=var "Block Spawn Pos", vector3Value=(unset), x=var "Damager X", y=var "Damager Y", z=0, everyFrame=false)
    CreateObject(gameObject=Block Hit (resources.assets), spawnPoint=<null>, position=var "Block Spawn Pos", rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Broadcast
  }

  state Invincible Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detecting
  }

  state Blocked Left {
    GetPosition(gameObject=var "Damager", vector=(unset), x=(unset), y=var "Damager Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Damager X", y=(unset), z=(unset), space=0, everyFrame=false)
    BoxColliderOffset(gameObject1=var "Self", offsetVector2=(unset), offsetX=var "Collider Origin X", offsetY=(unset), everyFrame=false)
    FloatAdd(floatVariable=var "Damager X", add=var "Collider Origin X", everyFrame=false, perSecond=false)
    BoundsBoxCollider(gameObject1=var "Self", scaleVector2=(unset), scaleX=var "Collider Width", scaleY=(unset), everyFrame=false)
    FloatDivide(floatVariable=var "Collider Width", divideBy=2, everyFrame=false)
    FloatAdd(floatVariable=var "Damager X", add=var "Collider Width", everyFrame=false, perSecond=false)
    SetVector3XYZ(vector3Variable=var "Block Spawn Pos", vector3Value=(unset), x=var "Damager X", y=var "Damager Y", z=0, everyFrame=false)
    CreateObject(gameObject=Block Hit (resources.assets), spawnPoint=<null>, position=var "Block Spawn Pos", rotation=(0, 0, 180), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Broadcast
  }

  state Blocked Up {
    GetPosition(gameObject=var "Damager", vector=(unset), x=var "Damager X", y=var "Attack Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Self", vector=(unset), x=(unset), y=var "Damager Y", z=(unset), space=0, everyFrame=false)
    BoxColliderOffset(gameObject1=var "Self", offsetVector2=(unset), offsetX=(unset), offsetY=var "Collider Origin Y", everyFrame=false)
    FloatAdd(floatVariable=var "Damager Y", add=var "Collider Origin Y", everyFrame=false, perSecond=false)
    BoundsBoxCollider(gameObject1=var "Self", scaleVector2=(unset), scaleX=(unset), scaleY=var "Collider Height", everyFrame=false)
    FloatDivide(floatVariable=var "Collider Height", divideBy=2, everyFrame=false)
    FloatSubtract(floatVariable=var "Damager Y", subtract=var "Collider Height", everyFrame=false, perSecond=false)
    SetFloatToHighest(floatVariable=var "Damager Y", value1=var "Damager Y", value2=var "Attack Y", everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Block Spawn Pos", vector3Value=(unset), x=var "Damager X", y=var "Damager Y", z=0, everyFrame=false)
    CreateObject(gameObject=Block Hit (resources.assets), spawnPoint=<null>, position=var "Block Spawn Pos", rotation=(0, 0, 90), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Broadcast
  }

  state Blocked Down {
    GetPosition(gameObject=var "Damager", vector=(unset), x=var "Damager X", y=var "Attack Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Self", vector=(unset), x=(unset), y=var "Damager Y", z=(unset), space=0, everyFrame=false)
    BoxColliderOffset(gameObject1=var "Self", offsetVector2=(unset), offsetX=(unset), offsetY=var "Collider Origin Y", everyFrame=false)
    FloatAdd(floatVariable=var "Damager Y", add=var "Collider Origin Y", everyFrame=false, perSecond=false)
    BoundsBoxCollider(gameObject1=var "Self", scaleVector2=(unset), scaleX=(unset), scaleY=var "Collider Height", everyFrame=false)
    FloatDivide(floatVariable=var "Collider Height", divideBy=2, everyFrame=false)
    FloatAdd(floatVariable=var "Damager Y", add=var "Collider Height", everyFrame=false, perSecond=false)
    SetFloatToSmallest(floatVariable=var "Damager Y", value1=var "Damager Y", value2=var "Attack Y", everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Block Spawn Pos", vector3Value=(unset), x=var "Damager X", y=var "Damager Y", z=0, everyFrame=false)
    CreateObject(gameObject=Block Hit (resources.assets), spawnPoint=<null>, position=var "Block Spawn Pos", rotation=(0, 0, 270), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Broadcast
  }

  state Damage Zero? {
    IntCompare(integer1=var "Attack Type", integer2=0, equal=(none), lessThan=→"FALSE", greaterThan=→"FALSE", everyFrame=false)
    IntCompare(integer1=var "Attack Strength", integer2=0, equal=→"TRUE", lessThan=→"TRUE", greaterThan=→"FALSE", everyFrame=false)
    on FALSE → Blocked Hit
    on TRUE → Pause Frame
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detecting
  }

  state Broadcast {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SPIKE RECOIL", delay=0, everyFrame=false)
  }

  var Corpse Fling Speed: float = 10
  var Invulnerable Time: float = 0
  var Enemy Type: int = 0
  var Geo Large: int = 0
  var Geo Medium: int = 0
  var Geo Small: int = 0
  var HP: int = 0
  var Corpse Faces Right: bool = false
  var Invincible: bool = false
  var Invincible No Effect: bool = false
  var Low Corpse Arc: bool = false
  var Play Animation: bool = false
  var Rotate Corpse: bool = false
  var Special Death: bool = false
  var Animation: string = ""
  var Death Broadcast Event: string = ""
  var Send KILLED to: string = ""
  var Corpse Spawn Point: vector3 = (0, 0, 0)
  var Effect Origin: vector3 = (0, 0, 0)
  var Battle Scene: gameObject = <null>
  var Corpse: gameObject = <null>
  var Send HIT to: gameObject = <null>

  var Angle: float = 0
  var Attack Direction: float = 0
  var Attack Magnitude: float = 0
  var Attack Y: float = 0
  var Chooser: float = 0
  var Collider Height: float = 0
  var Collider Origin X: float = 0
  var Collider Origin Y: float = 0
  var Collider Width: float = 0
  var Corpse Arc Left: float = 120
  var Corpse Arc Right: float = 60
  var Damager X: float = 0
  var Damager Y: float = 0
  var Face Left Scale: float = 0
  var Face Right Scale: float = 0
  var Geo Spawn Angle Max: float = 100
  var Geo Spawn Angle Min: float = 80
  var Geo Spawn Speed Max: float = 30
  var Geo Spawn Speed Min: float = 15
  var Rotate Max: float = 0
  var Rotate Min: float = 0
  var Rotation: float = 0
  var Scale Max: float = 0
  var Scale Min: float = 0
  var X Scale: float = 0
  var Y Scale: float = 0
  var Attack Strength: int = 0
  var Attack Type: int = 0
  var Battle Enemies: int = 0
  var Current Spawned: int = 0
  var Looper: int = 0
  var Circle Direction: bool = false
  var Spawned: bool = false
  var bool test: bool = false
  var spellBurn: bool = false
  var Spawner FSM: string = ""
  var Block Spawn Pos: vector3 = (0, 0, 0)
  var Midpoint: vector3 = (0, 0, 0)
  var Audio Player: gameObject = <null>
  var Camera: gameObject = <null>
  var Corpse Instance: gameObject = <null>
  var Damager: gameObject = <null>
  var Geo Pool: gameObject = <null>
  var Hero: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var KILLED GameObject: gameObject = <null>
  var Self: gameObject = <null>
  var Spawner: gameObject = <null>
}
