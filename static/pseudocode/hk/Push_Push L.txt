fsm Push {
  start Init

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    SetParticleEmission(gameObject=var "Pt Bounce", emission=false)
    on COLLIDE → Shadow Dashing?
  }

  state Push {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=shadow_gate_repel (sharedassets208.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Black Wave Default R (resources.assets), spawnPoint=var "", position=(unset), rotation=(unset), storeObject=var "Wave")
    SetScale(gameObject=var "Wave", vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Wave", vector=(unset), x=var "Parent X", y=var "Hero Y", z=-0.1, space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Pt Bounce", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilLeftLong())
    SetParticleEmission(gameObject=var "Pt Bounce", emission=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetPosition(gameObject=Self, vector=(unset), x=var "Parent X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatSubtract(floatVariable=var "Parent X", subtract=0.56, everyFrame=false, perSecond=false)
    FindChild(gameObject=Self, childName="Pt Bounce", storeResult=var "Pt Bounce")
    on FINISHED → Idle
  }

  state Shadow Dashing? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["shadowDashing"], storeResult=var "Shadow Dashing")
    BoolTest(boolVariable=var "Shadow Dashing", isTrue=→"CANCEL", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Push
    on CANCEL → Idle
  }

  var Hero Y: float = 0
  var Parent X: float = 0
  var Shadow Dashing: bool = false
  var Parent: gameObject = <null>
  var Pt Bounce: gameObject = <null>
  var Wave: gameObject = <null>
}
