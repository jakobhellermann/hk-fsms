# Control Interpolation
# 2 location(s):
#   Knight_Pickup / Knight
#   resources.assets / Knight:1

fsm Control Interpolation {
  start Idle

  state Idle {
    on LEVEL LOADED → Check Pause
  }

  state Check Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Interpolate
  }

  state Check {
    GetTagCount(tag="Set Extrapolate", storeResult=var "Tag Count")  // disabled
    IntCompare(integer1=var "Tag Count", integer2=0, equal=→"INTERPOLATE", lessThan=→"INTERPOLATE", greaterThan=→"EXTRAPOLATE", everyFrame=false)  // disabled
    on INTERPOLATE → Set Interpolate
    on EXTRAPOLATE → Set Extrapolate
  }

  state Set Interpolate {
    SetInterpolate(gameObject=Self)
    on FINISHED → Idle
  }

  state Set Extrapolate {
    SetExtrapolate(gameObject=Self)
    on FINISHED → Idle
  }

  var Tag Count: int = 0
}
