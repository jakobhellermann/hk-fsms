# Ball Control
# 1 location(s):
#   sharedassets355.assets / Dung Ball Small

fsm Ball Control {
  start Init
  on DUNGBALL END → Break  // from any state
  on ROAR WAVE START → Break  // from any state
  on ORBIT SHIELD → Break  // from any state

  state Detect {
    Collision2dEvent(collision=0, collideTag="Untagged", sendEvent=→"COLLISION ENTER 2D", storeCollider=var "Collider", storeForce=(unset))
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"SPELL", storeCollider=var "")
    on COLLISION ENTER 2D → Break
    on SPELL → Break
  }

  state Check {
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=(unset), contactNormal=var "Contact Normal", shapeCount=(unset), physics2dMaterialName="")
    GetVector3XYZ(vector3Variable=var "Contact Normal", storeX=(unset), storeY=var "Normal Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=1, tolerance=0.1, equal=(none), lessThan=→"CANCEL", greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → Detect
    on FINISHED → Break
  }

  state Break {
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[dung_ball_death_1, dung_ball_death_2, dung_ball_death_3, dung_ball_death_4], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=true)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ball Burst", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopParticleEmitter(gameObject=var "Smoke")
    StopParticleEmitter(gameObject=var "Chunks")
    PlayParticleEmitter(gameObject=var "Break Chunks", emit=0)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Pause
  }

  state Recycle {
    RecycleSelf()
  }

  state Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ball Idle")
    ActivateGameObject(gameObject=var "Nail Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    PlayParticleEmitter(gameObject=var "Smoke", emit=0)
    PlayParticleEmitter(gameObject=var "Chunks", emit=0)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Detect
  }

  state Pause {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Nail Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  var Normal Y: float = 0
  var Contact Normal: vector3 = (0, 0, 0)
  var Break Chunks: gameObject = Dung Ball Small/Break Chunks
  var Chunks: gameObject = Dung Ball Small/Chunks
  var Collider: gameObject = <null>
  var Nail Hit: gameObject = Dung Ball Small/Nail Hit
  var Self: gameObject = <null>
  var Smoke: gameObject = Dung Ball Small/Smoke
}
