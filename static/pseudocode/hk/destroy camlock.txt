fsm destroy camlock {
  start Pause

  state State 1 {
    FindGameObject(objectName="CameraLockArea 1", withTag="Untagged", store=var "Locky")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    BoolTest(boolVariable=var "Activated", isTrue=→"HIT", isFalse=(none), everyFrame=false)
    on HIT → Destroy
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State 1
  }

  state Destroy {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DestroyObject(gameObject=var "Locky", delay=0, detachChildren=false)
  }

  var Activated: bool = false
  var Locky: gameObject = <null>
}
