# Battle Control
# 2 location(s):
#   Ruins2_01 / Battle Scene
#   Ruins2_01_b / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activate
    on KILLED → Open Pause
  }

  state Activate {
    DestroyObject(gameObject=var "Shield Zombie", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
  }

  state Open Pause {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  var Activated: bool = false
  var Shield Zombie: gameObject = <null>
}
