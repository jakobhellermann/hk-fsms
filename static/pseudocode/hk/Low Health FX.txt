# Low Health FX
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health

fsm Low Health FX {
  start Pre Init
  on HUD OUT → Idle  // from any state
  on HUD IN → Overcharmed?  // from any state

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_27", isTrue=→"FINISHED", isFalse=(none))
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    SetParticleEmissionRate(gameObject=var "Low Health Leak", emissionRate=18, everyFrame=false)
    IntCompare(integer1=var "HP", integer2=1, equal=→"LOW", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on LOW → Low Health On Entry
  }

  state Idle {
    SetParticleEmission(gameObject=var "Low Health Particles", emission=false)
    SetParticleEmission(gameObject=var "Low Health Leak", emission=false)
    SetParticleEmissionRate(gameObject=var "Low Health Particles", emissionRate=40, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Light"), sendEvent="DOWN", delay=0, everyFrame=false)
    SetPosition(gameObject=var "Health 1", vector=var "H1 Initial Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Joni Health", vector=var "H1 Initial Pos", x=(unset), y=(unset), z=-3, space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="DOWN", delay=0, everyFrame=false)
    on HERO DAMAGED → Low Health Pause
  }

  state Low Health? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue HP")
    IntCompare(integer1=var "Blue HP", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=1, equal=→"LOW", lessThan=(none), greaterThan=(none), everyFrame=false)
    on LOW → Wait
    on FINISHED → Idle
  }

  state Low Health {
    SetFloatValue(floatVariable=var "Wait Time", floatValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="HERO LOW HEALTH", delay=0, everyFrame=false)
    SetParticleEmissionRate(gameObject=var "Low Health Leak", emissionRate=18, everyFrame=false)
    SetParticleEmission(gameObject=var "Low Health Particles", emission=true)
    SetParticleEmission(gameObject=var "Low Health Leak", emission=true)
    SendEventByName(eventTarget=GameObject(var "Low Health Light"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="UP", delay=0, everyFrame=false)
    iTweenShakePosition(gameObject=var "Health 1", id="", vector=(0.075, 0.075, 0), time=1, delay=0, loopType=loop, space=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenShakePosition(gameObject=var "Joni Health", id="", vector=(0.075, 0.075, 0), time=1, delay=0, loopType=loop, space=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on HERO HEALED → Idle
    on HERO HEALED FULL → Idle
    on HERO LEAVE → Idle
    on LOW HEALTH END → Idle
    on ADD BLUE HEALTH → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Init Wait
  }

  state Low Health On Entry {
    SetParticleEmission(gameObject=var "Low Health Particles", emission=true)
    SetParticleEmissionRate(gameObject=var "Low Health Particles", emissionRate=40, everyFrame=false)
    SetParticleEmissionRate(gameObject=var "Low Health Leak", emissionRate=18, everyFrame=false)
    SetParticleEmission(gameObject=var "Low Health Leak", emission=true)
    SendEventByName(eventTarget=GameObject(var "Low Health Light"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="UP", delay=0, everyFrame=false)  // disabled
    iTweenShakePosition(gameObject=var "Health 1", id="", vector=(0.075, 0.075, 0), time=1, delay=0, loopType=loop, space=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenShakePosition(gameObject=var "Joni Health", id="", vector=(0.075, 0.075, 0), time=1, delay=0, loopType=loop, space=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on HERO HEALED → Idle
    on HERO HEALED FULL → Idle
    on HERO LEAVE → Idle
    on LOW HEALTH END → Idle
    on ADD BLUE HEALTH → Idle
  }

  state HUD In HP Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue HP")
    IntCompare(integer1=var "Blue HP", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=1, equal=→"LOW", lessThan=(none), greaterThan=(none), everyFrame=false)
    on LOW → Low Health On Entry
    on FINISHED → Idle
  }

  state Overcharmed Init {
    FindChild(gameObject=Self, childName="OC Backboard", storeResult=var "OC Backboard")
    ActivateGameObject(gameObject=var "OC Backboard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="overcharmed", isTrue=(none), isFalse=→"FINISHED")
    ActivateGameObject(gameObject=var "OC Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Init
  }

  state Overcharmed? {
    FindChild(gameObject=Self, childName="OC Backboard", storeResult=var "OC Backboard")
    ActivateGameObject(gameObject=var "OC Backboard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Was Overcharmed", boolValue=var "Is Overcharmed", everyFrame=false)
    GetPlayerDataBool(boolName="overcharmed", storeValue=var "Is Overcharmed")
    BoolTest(boolVariable=var "Is Overcharmed", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    ActivateGameObject(gameObject=var "OC Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetAnimatorBool(gameObject=var "OC Backboard", parameter="Quick Appear", Value=var "Was Overcharmed", everyFrame=false, everyFrameOption=0)
    on FINISHED → HUD In HP Check
  }

  state Init Wait {
    Wait(time=2.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Overcharmed Init
  }

  state Pre Init {
    FindChild(gameObject=Self, childName="Low Health Vignette", storeResult=var "Low Health Vignette")
    FindChild(gameObject=Self, childName="Health 1", storeResult=var "Health 1")
    FindChild(gameObject=Self, childName="Joni Health", storeResult=var "Joni Health")
    FindChild(gameObject=Self, childName="Low Health Particles", storeResult=var "Low Health Particles")
    FindChild(gameObject=Self, childName="Low Health Light", storeResult=var "Low Health Light")
    FindChild(gameObject=var "Hero", childName="Low Health Leak", storeResult=var "Low Health Leak")
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetNamedParent(gameObject=Self, parentName="HudCamera", withTag="", storeResult=var "My HUD Camera")
    SetParent(gameObject=var "Low Health Vignette", parent=var "My HUD Camera", resetLocalPosition=false, resetLocalRotation=false)
    GetPosition(gameObject=var "Health 1", vector=var "H1 Initial Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    on FINISHED → Pause
  }

  state Low Health Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Low Health?
  }

  state Wait {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Low Health
  }

  var Wait Time: float = 0
  var Blue HP: int = 0
  var HP: int = 0
  var Is Overcharmed: bool = false
  var Was Overcharmed: bool = false
  var H1 Initial Pos: vector3 = (0, 0, 0)
  var Health 1: gameObject = <null>
  var Joni Health: gameObject = <null>
  var Low Health Leak: gameObject = <null>
  var Low Health Light: gameObject = <null>
  var Low Health Particles: gameObject = <null>
  var Low Health Vignette: gameObject = <null>
  var My HUD Camera: gameObject = <null>
  var OC Backboard: gameObject = <null>
  var Parent: gameObject = <null>
}
