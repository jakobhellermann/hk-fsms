# Rumbler
# 1 location(s):
#   Mines_31 / Super Dash Get/Far Rumbler

fsm Rumbler {
  start Init
  on AWAY → Away  // from any state
  on CLOSE → Wait  // from any state

  state Away {
    on CLOSE → Wait
  }

  state Wait {
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rumble
  }

  state Rumble {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="SmallShake", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Glow"), sendEvent="PULSE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Beam Parent"), sendEvent="PULSE", delay=0, everyFrame=false)
    SetMaterialColor(gameObject=var "Glow", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)  // disabled
    on FINISHED → Wait
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Glow", storeResult=var "Glow")
    FindChild(gameObject=var "Parent", childName="Beam Parent", storeResult=var "Beam Parent")
    on FINISHED → Away
  }

  state Glow Down {
    SendEventByName(eventTarget=GameObject(var "Glow"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Wait
  }

  var Beam Parent: gameObject = <null>
  var Glow: gameObject = <null>
  var Parent: gameObject = <null>
}
