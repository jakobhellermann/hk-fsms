# Dream React
# 1 location(s):
#   GG_Blue_Room / gg_blue_core

fsm Dream React {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Impact
  }

  state Play Anim {
    ActivateGameObject(gameObject=var "Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="blueRoomActivated", value=true)
    on FINISHED → Take Control
  }

  state Wait {
    Wait(time=3.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Regain Control
  }

  state Disable {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=gg_blue_room_orb_final_boom (sharedassets474.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Impact {
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Impact Pt", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=gg_blue_room_orb_slashed (sharedassets474.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Play Anim
  }

  state Regain Control {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    on FINISHED → Disable
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Convo", setValue=true, everyFrame=false)
    on FINISHED → Wait
  }

  var No Charms: bool = false
  var Impact: gameObject = blue_core_impact sequence

  var Impact Pt: gameObject = gg_blue_core/Impact Pt
  var Self: gameObject = <null>
}
