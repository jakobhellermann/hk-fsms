# Control
# 1 location(s):
#   resources.assets / Grimmball

fsm Control {
  start Init

  state Init {
    SetCircleCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Impact", active=false)
    PlayParticleEmitter(gameObject=var "Pt Flame", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Smoke", emit=0)
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"IMPACT", storeCollider=var "", storeForce=(unset))
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    on WAIT → Shrink
    on IMPACT → Impact
  }

  state Shrink {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    StopParticleEmitter(gameObject=var "Pt Flame")
    StopParticleEmitter(gameObject=var "Pt Smoke")
    SetCircleCollider(gameObject=Self, active=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(unset), time=0.5, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RecycleSelf()
  }

  state Impact {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Impact", active=true)
    StopParticleEmitter(gameObject=var "Pt Flame")
    StopParticleEmitter(gameObject=var "Pt Smoke")
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Impact", clipName="Flameball Impact", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayVibration(lowFidelityVibration=None, highFidelityVibration=grimball_impact, motors=All, loopTime=(unset), isLooping=false, tag="", gamepadVibration=SmallImpact)
    on FINISHED → Impact Pause
  }

  state Impact Pause {
    SetMeshRenderer(gameObject=var "Impact", active=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  var Enemy Damager: gameObject = Grimmball/Enemy Damager
  var Impact: gameObject = Grimmball/Impact
  var Pt Flame: gameObject = Grimmball/lava_particles_03
  var Pt Smoke: gameObject = Grimmball/Flame_smoke
  var Self: gameObject = Grimmball
}
