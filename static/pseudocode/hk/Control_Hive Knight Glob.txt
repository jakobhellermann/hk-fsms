# Control
# 2 location(s):
#   Hive_05 / Battle Scene/Globs/Hive Knight Glob
#   GG_Hive_Knight / Battle Scene/Globs/Hive Knight Glob

// uses template: hive_knight_glob
fsm Control {
  start Init

  state Set Inert {
    SetCircleCollider(gameObject=Self, active=false)
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Idle
  }

  state Idle {
    on FIRE → Tween Up
  }

  state Init {
    FindChild(gameObject=Self, childName="Splat", storeResult=var "Splat")
    FindChild(gameObject=Self, childName="Stingers", storeResult=var "Stingers")
    FindChild(gameObject=Self, childName="Pt Burst", storeResult=var "Pt Burst")
    FindChild(gameObject=Self, childName="Pt Drip", storeResult=var "Pt Drip")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Set Inert
  }

  state Tween Up {
    RandomFloat(min=-3, max=3, storeResult=var "Translate")
    Translate(gameObject=Self, vector=(unset), x=var "Translate", y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Glob Up")
    iTweenMoveBy(gameObject=Self, id="", vector=var "Tween Vector", time=(unset), delay=0, speed=20, easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    PlayParticleEmitterInState(gameObject=var "Pt Drip")
    on FINISHED → Form 1
  }

  state Form 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Glob Form 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[yellow_ball_appear_1, yellow_ball_appear_2, yellow_ball_appear_3], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Form 2
  }

  state Form 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Glob Form 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCircleCollider(gameObject=Self, active=true)
    on FINISHED → Antic
    on TINK DOWN → Tink D
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink U
  }

  state Antic {
    Wait(time=var "Antic Time", finishEvent=→"FINISHED", realTime=false)
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false)
    on FINISHED → Burst
    on TINK DOWN → Tink D
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink U
  }

  state Burst {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Splat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Stingers", activate=true)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Burst", emit=0)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[yellow_ball_shoot_1, yellow_ball_shoot_2, yellow_ball_shoot_3], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  state End {
    ActivateAllChildren(gameObject=var "Stingers", activate=false)
    on FINISHED → Set Inert
  }

  state Tink D {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -3, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Burst
  }

  state State 1 {
  }

  state Tink U {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 3, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Burst
  }

  state Tink R {
    iTweenMoveBy(gameObject=Self, id="", vector=(3, 0, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Burst
  }

  state Tink L {
    iTweenMoveBy(gameObject=Self, id="", vector=(-3, 0, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Burst
  }

  var Antic Time: float = 0.5
  var Tween Vector: vector3 = (0, 6, 0)
  var Parent: gameObject = Battle Scene/Globs

  var Translate: float = 0
  var Pt Burst: gameObject = <null>
  var Pt Drip: gameObject = <null>
  var Self: gameObject = <null>
  var Splat: gameObject = <null>
  var Stingers: gameObject = <null>
}
