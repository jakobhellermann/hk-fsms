# Door Target
# 1 location(s):
#   Room_Tram / door1

fsm Door Target {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get Station
  }

  state Get Station {
    GetPlayerDataInt(gameObject=var "GameManager", intName="tramLowerPosition", storeValue=var "Tram Position")
    IntSwitch(intVariable=var "Tram Position", compareTo=[0, 1, 2], sendEvent=[→"LEFT", →"CENTRE", →"RIGHT"], everyFrame=false)
    on LEFT → Left
    on CENTRE → Centre
    on RIGHT → Right
  }

  state Left {
    SetFsmString(gameObject=Self, fsmName="Door Control", variableName="New Scene", setValue="Abyss_03_b", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on TRAM ARRIVE → Get Station
  }

  state Centre {
    SetFsmString(gameObject=Self, fsmName="Door Control", variableName="New Scene", setValue="Abyss_03", everyFrame=false)
    on FINISHED → Idle
  }

  state Right {
    SetFsmString(gameObject=Self, fsmName="Door Control", variableName="New Scene", setValue="Abyss_03_c", everyFrame=false)
    on FINISHED → Idle
  }

  var Tram Position: int = 0
}
