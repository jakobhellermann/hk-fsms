# Knight Damage
# 1 location(s):
#   resources.assets / Knight:1/Effects/Damage Effect

fsm Knight Damage {
  start Init

  state Gen {
    SendEventByName(eventTarget=GameObject(var "Vignette"), sendEvent="DAMAGE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HeroLight"), sendEvent="DAMAGE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="damagedBlue", isTrue=→"BLUE", isFalse=(none))
    ActivateGameObject(gameObject=var "Hit Crack", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Hit Pt 1", emit=0)
    PlayParticleEmitter(gameObject=var "Hit Pt 2", emit=0)
    SetFloatValue(floatVariable=var "Scale Min", floatValue=-1, everyFrame=false)
    SetFloatValue(floatVariable=var "Scale Max", floatValue=-1.75, everyFrame=false)
    SetFloatValue(floatVariable=var "Rotate Min", floatValue=-30, everyFrame=false)
    SetFloatValue(floatVariable=var "Rotate Max", floatValue=30, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Slash Effect GhostDark1, spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=3, speedMin=20, speedMax=35, angleMin=140, angleMax=220, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Slash Effect GhostDark2, spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=3, speedMin=20, speedMax=35, angleMin=140, angleMax=220, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    SetFloatValue(floatVariable=var "Scale Min", floatValue=1, everyFrame=false)
    SetFloatValue(floatVariable=var "Scale Max", floatValue=1.75, everyFrame=false)
    SetFloatValue(floatVariable=var "Rotate Min", floatValue=-30, everyFrame=false)
    SetFloatValue(floatVariable=var "Rotate Max", floatValue=30, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Slash Effect GhostDark1, spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=3, speedMin=20, speedMax=35, angleMin=-40, angleMax=40, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Slash Effect GhostDark2, spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=3, speedMin=20, speedMax=35, angleMin=-40, angleMax=40, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    BoolTest(boolVariable=var "Hero Leak", isTrue=→"LEAK", isFalse=(none), everyFrame=false)
    on LEAK → Check Focus Prompt
    on BLUE → Blue Effects
    on FINISHED → Check Focus Prompt
  }

  state Leak {
    BoolTest(boolVariable=var "Hazard", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    StopParticleEmitter(gameObject=var "Leak")
    PlayParticleEmitter(gameObject=var "Leak", emit=0)
    on FINISHED → Idle
  }

  state Check Focus Prompt {
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="promptFocus", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="seenFocusTablet", isTrue=(none), isFalse=→"FINISHED")
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxHealth", storeValue=var "Max HP")
    IntCompare(integer1=var "HP", integer2=var "Max HP", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPCharge", storeValue=var "MP")
    IntCompare(integer1=var "MP", integer2=33, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetPlayerDataBool(boolName="promptFocus", value=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REMINDER FOCUS", delay=0, everyFrame=false)
    on FINISHED → Last HP?
  }

  state Blue Effects {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=flashHealBlue())
    FlingObjectsFromGlobalPool(gameObject=Spatter Blue, spawnPoint=var "Self", position=(unset), spawnMin=40, spawnMax=40, speedMin=10, speedMax=25, angleMin=30, angleMax=150, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")  // disabled
    SpawnBlood(spawnPoint=var "Self", position=(unset), spawnMin=40, spawnMax=40, speedMin=10, speedMax=25, angleMin=30, angleMax=150, colorOverride=(0.1454909, 0.6560647, 0.8602941, 1))
    SpawnObjectFromGlobalPool(gameObject=Splat Explode Blue, spawnPoint=var "Self", position=(0, 0, 0.02), rotation=(unset), storeObject=var "Effect")
    SetScale(gameObject=var "Effect", vector=(unset), x=1.6, y=1.6, z=(unset), everyFrame=false, lateUpdate=false)
    SpawnObjectFromGlobalPool(gameObject=Blue Flash, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Effect")
    SetScale(gameObject=var "Effect", vector=(unset), x=2.5, y=2.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Muffle? {
    BoolTest(boolVariable=var "Muffle", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DAMAGE LOWPASS", delay=0, everyFrame=false)
    on FINISHED → Gen
  }

  state Last HP? {
    IntCompare(integer1=var "HP", integer2=1, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    ActivateGameObject(gameObject=var "Low Health Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Leak
  }

  state Idle {
    SetBoolValue(boolVariable=var "Hazard", boolValue=false, everyFrame=false)
    on DAMAGE → Muffle?
    on HAZARD DAMAGE → Set Hazard
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Hit Pt 1", storeResult=var "Hit Pt 1")
    FindChild(gameObject=Self, childName="Hit Pt 2", storeResult=var "Hit Pt 2")
    FindChild(gameObject=Self, childName="Hit Crack", storeResult=var "Hit Crack")
    on FINISHED → Idle
  }

  state Set Hazard {
    SetBoolValue(boolVariable=var "Hazard", boolValue=true, everyFrame=false)
    on FINISHED → Gen
  }

  var HeroLight: gameObject = Knight:1/HeroLight
  var Leak: gameObject = Knight:1/Effects/Damage Effect/Leak
  var Vignette: gameObject = Knight:1/Vignette

  var Rotate Max: float = 0
  var Rotate Min: float = 0
  var Scale Max: float = 0
  var Scale Min: float = 0
  var HP: int = 0
  var MP: int = 0
  var Max HP: int = 0
  var Hazard: bool = false
  var Hero Leak: bool = true
  var Muffle: bool = true
  var Effect: gameObject = <null>
  var Hit Crack: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Hit Pt 1: gameObject = <null>
  var Hit Pt 2: gameObject = <null>
  var Low Health Effect: gameObject = Knight:1/Effects/Damage Effect/low health hit effect
  var Self: gameObject = <null>
}
