// uses template: rubble_push
fsm rubble_push {
  start Init
  on END RUBBLE → End Rubble  // from any state

  state Wait {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Push
  }

  state Push {
    RandomFloat(min=-100, max=100, storeResult=var "Chooser X")
    RandomFloat(min=-50, max=50, storeResult=var "Chooser Torque")
    RandomFloat(min=0, max=40, storeResult=var "Chooser Y")
    AddForce2d(gameObject=var "Self", forceMode=0, atPosition=(unset), vector=(unset), x=var "Chooser X", y=var "Chooser Y", vector3=(unset), everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Chooser Torque", everyFrame=false)
    on FINISHED → Wait
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Wait
  }

  state End Rubble {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)  // disabled
    SetCollider(gameObject=Self, active=false)  // disabled
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 1
  }

  state State 1 {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false)
  }

  var Chooser Torque: float = 0
  var Chooser X: float = 0
  var Chooser Y: float = 0
  var Self: gameObject = <null>
}
