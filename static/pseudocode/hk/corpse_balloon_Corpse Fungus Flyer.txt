#   sharedassets167.assets / Corpse Fungus Flyer

// uses template: corpse_balloon
fsm corpse_balloon {
  start Init

  state Blow {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=fungus_balloon_death_final_pop (sharedassets140.assets), pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    RandomFloat(min=0, max=360, storeResult=var "Direction")
    SetFloatValue(floatVariable=var "Speed", floatValue=100, everyFrame=false)
    AddForce2dAsAngle(gameObject=Self, atPosition=(unset), angle=var "Direction", speed=var "Speed", maxSpeed=0, maxSpeedX=0, maxSpeedY=0, everyFrame=true)
    FloatAdd(floatVariable=var "Direction", add=var "Direction Change", everyFrame=true, perSecond=true)
    FloatAdd(floatVariable=var "Speed", add=700, everyFrame=true, perSecond=true)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.25, 0.25, 0.25), time=2, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Destroy Object
  }

  state Destroy Object {
    StopParticleEmitter(gameObject=var "Steam")
    PlayParticleEmitter(gameObject=var "Spurt", emit=0)
    DestroySelf(detachChildren=true)
  }

  state Direction? {
    SendRandomEvent(events=[→"CW", →"CCW"], weights=[1, 1], delay=0)
    on CW → CW
    on CCW → CCW
  }

  state CW {
    SetFloatValue(floatVariable=var "Direction Change", floatValue=360, everyFrame=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Spin", everyFrame=false)
    on FINISHED → Blow
  }

  state CCW {
    SetFloatValue(floatVariable=var "Direction Change", floatValue=-360, everyFrame=false)
    FloatMultiply(floatVariable=var "Spin", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Spin", everyFrame=false)
    on FINISHED → Blow
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Steam", storeResult=var "Steam")
    FindChild(gameObject=var "Self", childName="Spurt", storeResult=var "Spurt")
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[fungus_balloon_death_1, fungus_balloon_death_2, fungus_balloon_death_3 (sharedassets140.assets)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Direction?
  }

  var Spin: float = 0

  var Direction: float = 0
  var Direction Change: float = 0
  var Speed: float = 0
  var Self: gameObject = <null>
  var Spurt: gameObject = <null>
  var Steam: gameObject = <null>
}
