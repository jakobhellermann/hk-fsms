# Grate Control
# 2 location(s):
#   Crossroads_47 / Stag_Tunnel/Stag_tunnel_Grate
#   Room_Town_Stag_Station / Stag_Tunnel/Stag_tunnel_Grate

fsm Grate Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PlayerData bool name", isTrue=→"DESTROY", isFalse=(none))
    FindChild(gameObject=Self, childName="Dust Rumble", storeResult=var "Dust Rumble")
    on DESTROY → Destroy
    on FINISHED → Idle
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Idle {
    on OPEN → Open
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=stag_gate_open (sharedassets9.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Dust Rumble", emit=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Grate_disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Opened
  }

  state Opened {
    StopParticleEmitter(gameObject=var "Dust Rumble")
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var PlayerData bool name: string = "openedCrossroads"

  var Dust Rumble: gameObject = <null>
  var Self: gameObject = <null>
}
