# Prompt Control
# 1 location(s):
#   resources.assets / Arrow Prompt

fsm Prompt Control {
  start Pause
  on CHANGE LABEL → Change Label  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Shadow", storeResult=var "Shadow")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Blank")
    on UP → Go Up
  }

  state Go Up {
    FadeGroupUp(target=var "Labels Folder")
    SetSpriteRenderer(gameObject=var "Shadow", active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Up", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Up
    on DOWN → Go Down
    on ARROW DOWN → Go Down
  }

  state Up {
    on DOWN → Go Down
    on ARROW DOWN → Go Down
  }

  state Go Down {
    FadeGroupDown(target=var "Labels Folder", fast=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Down", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on UP → Go Up
    on FINISHED → Down
  }

  state Select Text {
    FindChild(gameObject=Self, childName="Labels", storeResult=var "Labels Folder")
    FindChild(gameObject=var "Labels Folder", childName=var "Prompt Name", storeResult=var "Text")
    ActivateAllChildren(gameObject=var "Labels Folder", activate=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    on FINISHED → Init
  }

  state Change Label {
    FindChild(gameObject=var "Labels Folder", childName=var "Prompt Name", storeResult=var "Text")
    ActivateAllChildren(gameObject=var "Labels Folder", activate=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    GotoPreviousState()
  }

  state Down {
    on UP → Go Up
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetSpriteRenderer(gameObject=var "Shadow", active=false)
    on FINISHED → Select Text
  }

  var Prompt Name: string = ""

  var New Colour: color = (1, 1, 1, 0)
  var Prev Colour: color = (1, 1, 1, 0)
  var Shad Colour: color = (0, 0, 0, 0)
  var Shad Colour Prev: color = (0, 0, 0, 0)
  var Labels Folder: gameObject = <null>
  var Self: gameObject = <null>
  var Shadow: gameObject = <null>
  var Text: gameObject = <null>
}
