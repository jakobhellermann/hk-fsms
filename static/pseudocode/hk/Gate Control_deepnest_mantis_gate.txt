# Gate Control
# 1 location(s):
#   Fungus2_15 / deepnest_mantis_gate

fsm Gate Control {
  start Pause
  on MLORD CHALLENGE → Challenge  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedMantisLords", isTrue=→"ACTIVATE", isFalse=(none))
    FindChild(gameObject=var "Parent", childName="Mantis Lord Throne 1", storeResult=var "Mantis Lord")  // disabled
    FindGameObject(objectName="Mantis Lord Throne 1", withTag="Untagged", store=var "Mantis Lord")
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Gesture 1
  }

  state Close {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Gate", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Dust Close", emit=0)
    Tk2dPlayAnimation(gameObject=var "Gate", animLibName="", clipName="Gate Close")
    SetMeshRenderer(gameObject=var "Gate", active=true)
    Tk2dWatchAnimationEvents(gameObject=var "Gate", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=var "Mantis Lord", animLibName="", clipName="Throne Gesture 2")
    on FINISHED → Closed
  }

  state Closed {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Gate", audioClip=jiji_door_open_pt_2 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on MLORD GATE OPEN → Open
  }

  state Gesture 1 {
    Tk2dPlayAnimationWithEvents(gameObject=var "Mantis Lord", clipName="Throne Gesture 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=Mantis_Lords_Door_close (sharedassets179.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Close
  }

  state Open {
    PlayParticleEmitter(gameObject=var "Dust Close", emit=0)
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Gate", clipName="Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Opened
  }

  state Opened {
    SetMeshRenderer(gameObject=var "Gate", active=false)
  }

  state Pause {
    WaitForBossLoad(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Gate", active=false)
  }

  state Challenge {
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on MLORD GATE OPEN → Open
  }

  var Collider: gameObject = deepnest_mantis_gate/Collider
  var Dust Close: gameObject = deepnest_mantis_gate/Dust Close
  var Gate: gameObject = deepnest_mantis_gate/Gate
  var Mantis Lord: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
