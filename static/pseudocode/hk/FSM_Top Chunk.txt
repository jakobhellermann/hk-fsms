// uses template: spin_self
fsm FSM {
  start Spin At Start?

  state Wait {
    Wait(time=0.001, finishEvent=→"FINISHED", realTime=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Randomise z
  }

  state Spin {
    GetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "Spin Factor", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Speed", everyFrame=false)
    on RESPIN → Wait
  }

  state Spin At Start? {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Random Start Rotation", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetRandomRotation(gameObject=var "Self", x=false, y=false, z=true)
    on FINISHED → Wait
  }

  state Randomise z {
    RandomFloat(min=-0.000999, max=0.000999, storeResult=var "z Translate")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "z Translate", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Spin
  }

  var Spin Factor: float = 2
  var Random Start Rotation: bool = false

  var Speed: float = 0
  var z Translate: float = 0
  var Self: gameObject = <null>
}
