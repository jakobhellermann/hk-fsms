# Blanker Control
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Blanker
#   resources.assets / _GameCameras/HudCamera/Blanker

fsm Blanker Control {
  start Init
  on LEVEL LOADED → Scene Reset  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetSpriteRenderer(gameObject=var "Self", active=false)
    SetGameObject(variable=var "HUD Blanker", gameObject=var "Self", everyFrame=false)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    on FINISHED → Idle
  }

  state Fade In {
    SetSpriteRenderer(gameObject=Self, active=true)
    SetColorValue(colorVariable=var "Colour", color=(0, 0, 0, 0), everyFrame=false)
    EaseColor(fromValue=(0, 0, 0, 0), toValue=(0, 0, 0, 1), colorVariable=var "Colour", time=var "Fade Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    on FADE OUT → Fade Out
  }

  state Fade Out {
    SetColorValue(colorVariable=var "Colour", color=(0, 0, 0, 1), everyFrame=false)
    EaseColor(fromValue=(0, 0, 0, 1), toValue=(0, 0, 0, 0), colorVariable=var "Colour", time=var "Fade Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FADE IN → Fade In
    on FINISHED → Faded Out
  }

  state Scene Reset {
    SetGameObject(variable=var "HUD Blanker", gameObject=var "Self", everyFrame=false)
    BoolTest(boolVariable=var "Cancel Reset", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SetSpriteRenderer(gameObject=var "Self", active=false)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    on FINISHED → Idle
    on CANCEL → Cancel
  }

  state Idle {
    on FADE IN → Fade In
  }

  state Cancel {
    SetBoolValue(boolVariable=var "Cancel Reset", boolValue=false, everyFrame=false)
    on FADE OUT → Fade Out
  }

  state Faded Out {
    SetSpriteRenderer(gameObject=Self, active=false)
    on FADE IN → Fade In
  }

  var Fade Time: float = 1
  var Cancel Reset: bool = false
  var Colour: color = (0, 0, 0, 1)
  var Self: gameObject = <null>
}
