fsm Control {
  start Rotate

  state Wait {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=true)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Rotate {
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    RandomFloatEither(value1=1, value2=1.25, storeResult=var "Scale")
    RandomFloatEither(value1=1, value2=-1, storeResult=var "Scale Multiplier")
    FloatMultiply(floatVariable=var "Scale", multiplyBy=var "Scale Multiplier", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  var Scale: float = 0
  var Scale Multiplier: float = 0
}
