# Control
# 2 location(s):
#   GG_Uumuu / Battle Scene
#   GG_Uumuu_V / Battle Scene

fsm Control {
  start Pause

  state Start {
    SetPlayerDataBool(boolName="encounteredMegaJelly", value=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    on BOSS DEATH → End
  }

  state End {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    EndGGBossScene()
  }

  state Pause {
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Pause 2
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pause 3
  }

  state Pause 3 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Start
  }

  var Self: gameObject = <null>
}
