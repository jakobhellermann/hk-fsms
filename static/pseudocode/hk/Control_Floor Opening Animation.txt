# Control
# 1 location(s):
#   Abyss_06_Core / Floor Opening Animation

fsm Control {
  start Pause

  state Detect {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    AudioPlayInState(gameObject=Self, volume=1)
    on ENTER → Activate
    on ACTIVATE → Activated
  }

  state Activate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BLACK FLOOR OPEN", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetPlayerDataBool(boolName="openedBlackEggPath", value=true)
    SetCollider(gameObject=Self, active=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Activated {
  }

  var Activated: bool = false
  var Self: gameObject = <null>
}
