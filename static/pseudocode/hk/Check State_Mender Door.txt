# Check State
# 1 location(s):
#   Crossroads_04 / _Transition Gates/Mender Door

fsm Check State {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Door Closed", storeResult=var "Door Closed")
    FindChild(gameObject=Self, childName="door_Mender_House", storeResult=var "Door Object")
    FindChild(gameObject=Self, childName="Door Opened", storeResult=var "Door Opened")
    on FINISHED → Opened?
  }

  state Opened? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="menderState", storeValue=var "Mender State")
    IntCompare(integer1=var "Mender State", integer2=2, equal=→"OPENED", lessThan=(none), greaterThan=→"OPENED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="menderDoorOpened", isTrue=→"OPENED", isFalse=(none))  // disabled
    on OPENED → Opened
    on FINISHED → Closed
  }

  state Opened {
    ActivateGameObject(gameObject=var "Door Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Door Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Have Key? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasMenderKey", isTrue=→"KEY", isFalse=→"NO KEY")
    on KEY → Key
    on NO KEY → No Key
  }

  state Key {
    SetCollider(gameObject=var "Door Object", active=false)
  }

  state No Key {
    ActivateGameObject(gameObject=var "Door Object", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Closed {
    ActivateGameObject(gameObject=var "Door Object", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Mender State: int = 0
  var Door Closed: gameObject = _Transition Gates/Mender Door/Door Closed
  var Door Object: gameObject = <null>
  var Door Opened: gameObject = _Transition Gates/Mender Door/Door Opened
  var Inspect: gameObject = _Transition Gates/Mender Door/Inspect
}
