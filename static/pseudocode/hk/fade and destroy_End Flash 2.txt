fsm fade and destroy {
  start Idle

  state Idle {
    EaseColor(fromValue=(1, 1, 1, 0.603), toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
  }

  var Colour: color = (1, 1, 1, 0.5176471)
}
