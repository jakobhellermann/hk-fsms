# Control
# 1 location(s):
#   resources.assets / Grubberfly BeamL R

// uses template: grubberfly_beam
fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Impact", active=true)
    ActivateGameObject(gameObject=var "Impact", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetVelocity2d(gameObject=Self, vector=var "Launch Velocity", x=(unset), y=(unset), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="beamDamage", storeValue=var "Beam Damage")
    Translate(gameObject=Self, vector=var "Start Shift", x=(unset), y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FindChild(gameObject=Self, childName="Terrain Detector", storeResult=var "Terrain Detector")
    on FINISHED → Fury Multiplier
  }

  state Active {
    Wait(time=0.18, finishEvent=→"DISSIPATE", realTime=false)
    SetPolygonCollider(gameObject=Self, active=true)
    on DISSIPATE → Dissipate
    on TERRAIN HIT → Terrain Hit
    on DEALT DAMAGE → Terrain Hit
  }

  state Dissipate {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle Pause
  }

  state Recycle {
    RecycleSelf()
  }

  state Terrain Hit {
    SetPolygonCollider(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Impact", clipName="Impact", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle Pause
  }

  state Inactive {
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=var "Beam Damage", everyFrame=false)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Active
  }

  state Fury Multiplier {
    BoolTest(boolVariable=var "Fury", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    ConvertIntToFloat(intVariable=var "Beam Damage", floatVariable=var "Dmg Float", everyFrame=false)
    FloatMultiply(floatVariable=var "Dmg Float", multiplyBy=1.5, everyFrame=false)
    ConvertFloatToInt(floatVariable=var "Dmg Float", intVariable=var "Beam Damage", rounding=Nearest, everyFrame=false)
    on FINISHED → Inactive
  }

  state Recycle Pause {
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Impact", active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  var Fury: bool = true
  var Launch Velocity: vector2 = (-30, 0)
  var Start Shift: vector3 = (0, 0, 0)
  var Impact: gameObject = Grubberfly BeamL R/Impact

  var Dmg Float: float = 0
  var Beam Damage: int = 0
  var Terrain Detector: gameObject = <null>
}
