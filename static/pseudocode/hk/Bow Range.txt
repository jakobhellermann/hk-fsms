fsm Bow Range {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedMantisLords", isTrue=→"BOW", isFalse=(none))
    on BOW → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD BOW", delay=0, everyFrame=false)
  }
}
