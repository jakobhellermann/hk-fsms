fsm Up n Down {
  start State 1

  state State 1 {
    FindChild(gameObject=Self, childName="Shadow", storeResult=var "Shad1")
    FindChild(gameObject=Self, childName="Shadow (1)", storeResult=var "Shad2")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → State 2
  }

  state State 2 {
    SetSpriteRenderer(gameObject=var "Shad1", active=true)
    SetSpriteRenderer(gameObject=var "Shad2", active=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)  // disabled
    on STAG MENU DOWN → State 3
  }

  state State 3 {
    SetSpriteRenderer(gameObject=var "Shad1", active=false)
    SetSpriteRenderer(gameObject=var "Shad2", active=false)
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  var Shad1: gameObject = <null>
  var Shad2: gameObject = <null>
}
