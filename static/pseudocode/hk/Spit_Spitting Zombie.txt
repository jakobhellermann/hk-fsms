# Spit
# 1 location(s):
#   sharedassets33.assets / Spitting Zombie

// component disabled: this FSM does not run until something enables it
fsm Spit {
  start Init

  state Idle {
    SendTrigger2DEventByName(eventTarget=GameObject(Self), trigger=OnTriggerStay2D, collideTag="", collideLayer=(unset), sendEvent="MOVESTART", storeCollider=var "")  // disabled
    on HERO IN RANGE → Attack Wait
  }

  state Attack Antic {
    AudioPlayRandom(gameObject=var "Self", audioClips=[spitting_zombie_prepare_1, spitting_zombie_prepare_2], weights=[1, 1], pitchMin=1, pitchMax=1)
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="Infected Zombie 02 Anim", clipName="Attack")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"TRIGGER", animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on TRIGGER → Shoot
  }

  state Spawn Bullet R {
    SetWalkerFacing(walkRight=true, randomStartDir=false, target=Self, everyFrame=false)
    FlingObjectsFromGlobalPoolVel(gameObject=Shot Mawlek (sharedassets32.assets), spawnPoint=var "SpawnPoint", position=(unset), spawnMin=2, spawnMax=3, speedMinX=1.5, speedMaxX=8, speedMinY=15, speedMaxY=20, originVariationX=0, originVariationY=0)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"Amin End")
    on Amin End → Cooldown
  }

  state Cooldown {
    Wait(time=0.5, finishEvent=→"WAIT", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on WAIT → MoveStart
  }

  state MoveStart {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on FINISHED → Idle
  }

  state Spawn Bullet L {
    SetWalkerFacing(walkRight=false, randomStartDir=false, target=Self, everyFrame=false)
    FlingObjectsFromGlobalPoolVel(gameObject=Shot Mawlek (sharedassets32.assets), spawnPoint=var "SpawnPoint", position=(unset), spawnMin=2, spawnMax=3, speedMinX=-1.5, speedMaxX=-8, speedMinY=15, speedMaxY=20, originVariationX=0, originVariationY=0)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"Amin End")
    on Amin End → Cooldown
  }

  state Shoot {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=spitting_zombie_spit)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="PLAY", delay=0, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"SHOOTLEFT", negativeEvent=→"SHOOTRIGHT", space=0)
    on SHOOTLEFT → Spawn Bullet L
    on SHOOTRIGHT → Spawn Bullet R
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Attack Wait {
    WaitRandom(timeMin=0, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack Antic
  }

  var SpawnPoint: gameObject = Spitting Zombie/Spawn Point

  var Self: gameObject = Spitting Zombie
}
