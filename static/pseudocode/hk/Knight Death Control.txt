# Knight Death Control
# 1 location(s):
#   resources.assets / Knight Spike Death

fsm Knight Death Control {
  start Stab
  on HAZARD RELOAD → Recycle  // from any state

  state Stab {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="KNIGHT SPIKE HIT", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Pierces", activate=true)
    GetOwner(storeGameObject=var "Self")
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="SaveLevelState", parameters=[], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClips=[hero_damage], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="HAZARD DAMAGE", delay=0, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    FloatAdd(floatVariable=var "Spike Direction", add=90, everyFrame=false, perSecond=false)  // disabled
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetRotation(gameObject=var "Pierces", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spike Direction", space=0, everyFrame=false, lateUpdate=false)  // disabled
    SetFloatValue(floatVariable=var "Angle Min", floatValue=var "Spike Direction", everyFrame=false)  // disabled
    FloatSubtract(floatVariable=var "Angle Min", subtract=20, everyFrame=false, perSecond=false)  // disabled
    SetFloatValue(floatVariable=var "Angle Max", floatValue=var "Spike Direction", everyFrame=false)  // disabled
    FloatSubtract(floatVariable=var "Angle Max", subtract=-20, everyFrame=false, perSecond=false)  // disabled
    SetParent(gameObject=var "Pierces", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Hero Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spike Death Antic")
    on FINISHED → Fling
  }

  state Fling {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spike Death")
    SetGravity2dScale(gameObject=Self, gravityScale=0.7)
    RandomFloat(min=80, max=100, storeResult=var "Hero Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Hero Angle", speed=20, everyFrame=false)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
  }

  state Recycle {
    RecycleSelf()
  }

  var Spike Direction: float = 0

  var Angle Max: float = 0
  var Angle Min: float = 0
  var Hero Angle: float = 0
  var Hero Scale: float = 0
  var Pierces: gameObject = Knight Spike Death/Pierces
  var Self: gameObject = <null>
}
