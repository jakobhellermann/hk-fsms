fsm Control {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    on START → Pour
  }

  state Pour {
    SpawnRandomObjectsOverTime(gameObject=Particle Rock Tiny Transient Plain (resources.assets), spawnPoint=var "Self", position=(unset), frequency=0.2, spawnMin=1, spawnMax=1, speedMin=0, speedMax=0, angleMin=0, angleMax=0, originVariation=0.2)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Done
  }

  state Done {
  }

  var Self: gameObject = <null>
}
