# Conversation Control
# 1 location(s):
#   Ruins1_02 / RestBench/Quirrel Bench

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → P1
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → NPC Anim
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="QUIRREL", everyFrame=false)
    on FINISHED → Choice
  }

  state Convo 1 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="quirrelRuins")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_RUINS_1", "Quirrel"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bench Talk End")
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state NPC Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bench Talk")
    on FINISHED → Box Up
  }

  state Choice {
    GetPlayerDataInt(gameObject=var "GameManager", intName="quirrelRuins", storeValue=var "State")
    IntSwitch(intVariable=var "State", compareTo=[0, 1, 2], sendEvent=[→"CONVO 1", →"CONVO 2", →"CONVO 3"], everyFrame=false)
    on CONVO 1 → Convo 1
    on CONVO 2 → Convo 2
    on CONVO 3 → Convo 3
    on FINISHED → Convo 4
  }

  state Convo 2 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="quirrelRuins")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_RUINS_2", "Quirrel"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Convo 3 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="quirrelRuins")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_RUINS_3", "Quirrel"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Convo 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_RUINS_4", "Quirrel"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_05_hmmm (sharedassets19.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state P1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → P2
  }

  state P2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Shift Bench
  }

  state Shift Bench {
    GetParent(gameObject=Self, storeResult=var "Bench")
    FindChild(gameObject=var "Bench", childName="Lit", storeResult=var "Lit")
    FindChild(gameObject=var "Bench", childName="rest_bench", storeResult=var "rest_bench")
    SetPosition(gameObject=var "Lit", vector=(unset), x=0.419, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "rest_bench", vector=(unset), x=0.419, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  var State: int = 0
  var Area Title: gameObject = <null>
  var Bench: gameObject = <null>
  var Lit: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
  var rest_bench: gameObject = <null>
}
