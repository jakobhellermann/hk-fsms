# Intro
# 1 location(s):
#   GG_Lurker / Lurker Control/Lurker Intro

fsm Intro {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Jump", storeResult=var "Pt Jump")
    on ENTER → Look Back
  }

  state Look Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookBack")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_small_jump (sharedassets15.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Lurker_startle_07 (sharedassets451.assets), pitchMin=1.2, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Roar
  }

  state Roar {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar")
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Lurker_startle_12 (sharedassets451.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=misc_rumble_impact (sharedassets13.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Roar End
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    on FINISHED → Hop Antic
  }

  state Hop Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hop Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Hop
  }

  state Hop {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hop")
    SetVelocity2d(gameObject=Self, vector=(30, 30), x=(unset), y=(unset), everyFrame=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_jump (sharedassets34.assets), pitchMin=1, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Lurker_startle_10 (sharedassets451.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitterInState(gameObject=var "Pt Jump")  // disabled
    on WALL → Cling
  }

  state Cling {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wall Cling", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_wall_land (sharedassets15.assets), pitchMin=1, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Wallhop Antic
  }

  state Wallhop Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wallhop Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hop 2
  }

  state Hop 2 {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hop")
    SetVelocity2d(gameObject=Self, vector=(-10, 60), x=(unset), y=(unset), everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_jump (sharedassets34.assets), pitchMin=1, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitterInState(gameObject=var "Pt Jump")
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Pt Jump: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Self: gameObject = <null>
}
