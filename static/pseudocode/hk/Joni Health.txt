fsm Joni Health {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Health 1", storeResult=var "Health 1")
    FindChild(gameObject=Self, childName="Joni Health", storeResult=var "Joni Health")
    FindChild(gameObject=var "Health 1", childName="Idle", storeResult=var "Health 1 Idle")
    FindChild(gameObject=var "Health 1", childName="Idle Hive", storeResult=var "Health 1 Idle Hive")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_27", isTrue=→"JONI", isFalse=→"NORMAL")
    on JONI → Joni
    on NORMAL → Normal
  }

  state Joni {
    SetMeshRenderer(gameObject=var "Joni Health", active=true)
    SendEventByName(eventTarget=GameObject(var "Joni Health"), sendEvent="ON", delay=0, everyFrame=false)
    SetPosition(gameObject=var "Health 1 Idle", vector=(unset), x=(unset), y=50, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Health 1 Idle Hive", vector=(unset), x=(unset), y=50, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Idle {
    on CHARM INDICATOR CHECK → Check
    on HUD IN → Check
  }

  state Normal {
    SendEventByName(eventTarget=GameObject(var "Joni Health"), sendEvent="OFF", delay=0, everyFrame=false)
    SetPosition(gameObject=var "Health 1 Idle", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Health 1 Idle Hive", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Health 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetMeshRenderer(gameObject=var "Health 1 Idle", active=true)  // disabled
    on FINISHED → Idle
  }

  var Health 1: gameObject = <null>
  var Health 1 Idle: gameObject = <null>
  var Health 1 Idle Hive: gameObject = <null>
  var Joni Health: gameObject = <null>
}
