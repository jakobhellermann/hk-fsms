# Send Hit
# 1 location(s):
#   sharedassets32.assets / Corpse Zote Boss/Hit Detect

fsm Send Hit {
  start Init

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Send
  }

  state Send {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="NAIL HIT", delay=0, everyFrame=false)
    on FINISHED → Detect
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Detect
  }

  var Parent: gameObject = <null>
}
