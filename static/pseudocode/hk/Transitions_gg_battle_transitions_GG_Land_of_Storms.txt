# Transitions
# 1 location(s):
#   GG_Land_of_Storms / gg_battle_transitions

fsm Transitions {
  start Disable
  on GG TRANSITION IN → In  // from any state
  on GG TRANSITION OUT → Out  // from any state
  on GG TRANSITION FINAL → Final  // from any state
  on GG TRANSITION OUT INSTANT → Out Instant  // from any state
  on GG TRANSITION OUT STATUE → Out Statue  // from any state
  on GG TRANSITION IN STATUE → In Workshop  // from any state

  state Disable {
    ActivateGameObject(gameObject=var "TransitionIn", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "TransitionOut", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "TransitionEnd", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state In {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendEventToRegister(eventName="INVENTORY CANCEL")
    ActivateGameObject(gameObject=var "TransitionIn", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "TransitionOut", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Warp Allowed", setValue=false, everyFrame=false)
    Wait(time=var "Transition In Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Event
  }

  state Out {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendEventToRegister(eventName="INVENTORY CANCEL")
    ActivateGameObject(gameObject=var "TransitionOut", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=gg_room_transition (sharedassets425.assets), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (resources.assets), spawnPoint=Self, spawnPosition=(0, 0, 0))
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Warp Allowed", setValue=false, everyFrame=false)
    SendEventToRegister(eventName="LOOK UP")
    Wait(time=var "Transition Out Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Blanker
  }

  state Send Event {
    SetPlayerDataBool(boolName="disablePause", value=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Warp Allowed", setValue=true, everyFrame=false)
    SendEventToRegister(eventName="GG TRANSITION END")
  }

  state Final {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendEventToRegister(eventName="INVENTORY CANCEL")
    ActivateGameObject(gameObject=var "TransitionEnd", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=gg_room_transition_2_bit_shorter (sharedassets425.assets), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (resources.assets), spawnPoint=Self, spawnPosition=(0, 0, 0))
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Warp Allowed", setValue=false, everyFrame=false)
    SendEventToRegister(eventName="LOOK UP")
    Wait(time=var "TransitionEndTime", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Blanker
  }

  state Out Instant {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendEventToRegister(eventName="INVENTORY CANCEL")
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Warp Allowed", setValue=false, everyFrame=false)
    ActivateGameObject(gameObject=var "TransitionOut", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=var "TransitionOut", stateName="GG_battle_transition_out", layer=(unset), normalizedTime=1, everyFrame=false)
  }

  state Set Blanker {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    on FINISHED → Send Event
  }

  state Out Statue {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendEventToRegister(eventName="INVENTORY CANCEL")
    ActivateGameObject(gameObject=var "TransitionOut", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=gg_room_transition (sharedassets425.assets), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (resources.assets), spawnPoint=Self, spawnPosition=(0, 0, 0))
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Warp Allowed", setValue=false, everyFrame=false)
    SendEventToRegister(eventName="LOOK UP")
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=var "Transition Out Time")
    Wait(time=var "Transition Out Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Blanker
  }

  state In Workshop {
    ActivateGameObject(gameObject=var "TransitionIn", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "TransitionOut", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=var "Transition In Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Event Basic
  }

  state Send Event Basic {
    SendEventToRegister(eventName="GG TRANSITION END")
  }

  var Transition In Time: float = 1
  var Transition Out Time: float = 1.5

  var TransitionEndTime: float = 3.2
  var TransitionEnd: gameObject = gg_battle_transitions/battle_final
  var TransitionIn: gameObject = gg_battle_transitions/battle_enter
  var TransitionOut: gameObject = gg_battle_transitions/battle_end
}
