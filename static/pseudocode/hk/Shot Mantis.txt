# Shot Mantis
# 1 location(s):
#   sharedassets34.assets / Shot Mantis

fsm Shot Mantis {
  start Init

  state Init {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shot")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Target X", y=var "Target Y", z=(unset), space=0, everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Force", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Force", multiplyBy=-1, everyFrame=false)
    on FINISHED → To Target
  }

  state To Target {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatTestToBool(float1=var "Self X", float2=var "Target X", tolerance=2, equalBool=var "Near X", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=true)
    FloatTestToBool(float1=var "Self Y", float2=var "Target Y", tolerance=2, equalBool=var "Near Y", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=true)
    BoolAllTrue(boolVariables=[var "Near X", var "Near Y"], sendEvent=→"TARGET", storeResult=(unset), everyFrame=true)
    on TARGET → Boomerang
    on END → End
    on ORBIT SHIELD → Stop velocity
  }

  state Boomerang {
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "X Force", y=20, vector3=(unset), everyFrame=true)
    Wait(time=4, finishEvent=→"END", realTime=false)
    on END → End
    on ORBIT SHIELD → Stop velocity
  }

  state End {
    AudioStop(gameObject=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shot End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Stop velocity {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → End
  }

  var Self X: float = 0
  var Self Y: float = 0
  var Target X: float = 0
  var Target Y: float = 0
  var X Force: float = 0
  var X Speed: float = 0
  var Near X: bool = false
  var Near Y: bool = false
  var Top Detect: bool = false
}
