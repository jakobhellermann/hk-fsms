fsm Change Sprite {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State 1
  }

  state State 1 {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotKingFragment", isTrue=→"CHANGE", isFalse=(none))
    on CHANGE → Change
  }

  state Change {
    SetSpriteRendererSprite(gameObject=Self, sprite=dusk_knight_armour (sharedassets333.assets))
  }
}
