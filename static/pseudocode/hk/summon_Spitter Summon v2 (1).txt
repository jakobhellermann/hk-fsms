// uses template: summon_spitter
fsm summon {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Spitter", storeResult=var "Buzzer")
    on SUMMON → Random Pause
  }

  state Enter {
    SetMeshRenderer(gameObject=Self, active=true)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    RandomFloat(min=0.75, max=1.2, storeResult=var "Tween Time")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=8, z=16.996, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    ColorInterpolate(colors=[(0, 0, 0, 1), (1, 1, 1, 1)], time=1, storeColor=var "Colour", finishEvent=(none), realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -8, -16.99), time=var "Tween Time", delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Summon
  }

  state Summon {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Buzzer", vector=(unset), x=var "Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=Self, vector=var "Position", x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Buzzer", vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetParent(gameObject=var "Buzzer", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(var "Buzzer"), sendEvent="ALERT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Random Pause {
    WaitRandom(timeMin=0.25, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter
  }

  var Chooser: float = 0
  var Pos X: float = 0
  var Pos Y: float = 0
  var Scale: float = 0
  var Tween Time: float = 0
  var Position: vector3 = (0, 0, 0)
  var Colour: color = (0, 0, 0, 1)
  var Battle Scene: gameObject = <null>
  var Buzzer: gameObject = <null>
  var Self: gameObject = <null>
}
