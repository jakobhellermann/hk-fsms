fsm Warp messenger {
  start Wait

  state Wait {
    WaitRandom(timeMin=2, timeMax=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WARP", delay=0, everyFrame=false)
    on FINISHED → Wait
  }
}
