# Control
# 1 location(s):
#   sharedassets447.assets / Shot HK Shadow

fsm Control {
  start Start

  state Start {
    ActivateGameObject(gameObject=var "Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Beam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    Tk2dPlayFrame(gameObject=Self, frame=0)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fire
  }

  state Fire {
    FaceAngle(gameObject=Self, angleOffset=-90, everyFrame=false)  // disabled
    FaceAngle(gameObject=Self, angleOffset=-90, everyFrame=true)  // disabled
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=(unset), everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Beam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Angle: float = 0
  var Beam: gameObject = Shot HK Shadow/Beam
  var Glow: gameObject = Shot HK Shadow/Glow
}
