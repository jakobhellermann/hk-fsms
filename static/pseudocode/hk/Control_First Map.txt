fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on FIRST MAP UP → Up
  }

  state Up {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    on FIRST MAP DOWN → Down
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 1
  }

  state State 1 {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Self: gameObject = <null>
}
