fsm Set Bretta Leave {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="greyPrinceDefeats", storeValue=var "Grey Prince Defeats")
    IntCompare(integer1=var "Grey Prince Defeats", integer2=3, equal=(none), lessThan=(none), greaterThan=→"SET", everyFrame=false)
    on SET → Set
  }

  state Set {
    SetPlayerDataBool(boolName="brettaLeftTown", value=true)
  }

  var Grey Prince Defeats: int = 0
}
