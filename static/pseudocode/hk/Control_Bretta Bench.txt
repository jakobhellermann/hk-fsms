# Control
# 1 location(s):
#   Town / Bretta Bench

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pause 2
  }

  state Init {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=130, upperValue=134, boolVariable=(unset), trueEvent=→"DESTROY", falseEvent=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brettaRescued", isTrue=(none), isFalse=→"DESTROY")
    GetPlayerDataInt(gameObject=var "GameManager", intName="brettaPosition", storeValue=var "Bretta Position")
    IntCompare(integer1=var "Bretta Position", integer2=0, equal=(none), lessThan=→"DESTROY", greaterThan=→"DESTROY", everyFrame=false)
    FindGameObject(objectName="RestBench", withTag="Untagged", store=var "RestBench")
    FindChild(gameObject=var "RestBench", childName="Lit", storeResult=var "Lit")
    SetPosition(gameObject=var "RestBench", vector=(unset), x=132.05, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetSpriteRenderer(gameObject=var "RestBench", active=false)
    SetSpriteRenderer(gameObject=var "Lit", active=false)
    on DESTROY → Destroy
    on BENCHREST → Sit
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Sit {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body Sit")
    SendEventByName(eventTarget=GameObject(var "Blush"), sendEvent="UP", delay=0, everyFrame=false)
    on BENCHREST END → Off
  }

  state Off {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body")
    SendEventByName(eventTarget=GameObject(var "Blush"), sendEvent="DOWN", delay=0, everyFrame=false)
    on BENCHREST → Sit
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Hero X: float = 0
  var Bretta Position: int = 0
  var Blush: gameObject = Bretta Bench/Blush
  var Lit: gameObject = <null>
  var RestBench: gameObject = <null>
}
