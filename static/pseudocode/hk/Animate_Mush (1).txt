# Animate
# 2 location(s):
#   Fungus2_34 / Giraffe NPC/Mush (1)
#   Fungus2_34 / Giraffe NPC/Mush

fsm Animate {
  start Idle

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mush Idle")
    on EAT START → Eat
  }

  state Eat {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mush Eat")
    on EAT END → Idle
  }
}
