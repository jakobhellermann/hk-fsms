# Control
# 3 location(s):
#   Ruins1_29 / Stag Map
#   RestingGrounds_09 / Stag Map
#   resources.assets / Stag Map

fsm Control {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=true)
    FindChild(gameObject=Self, childName="UI List Stag", storeResult=var "UI List")
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Up
  }

  state Up {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(1.264597, 1.264597, 1.264597), time=0.5, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    on UI SELECTION MADE → Close
  }

  state Pause {
    NextFrameEvent(sendEvent=→"CONT")
    on CONT → Init
  }

  state Close {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(1, 1, 1), time=0.3, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="STAG MENU DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Stag Map/UI List Stag), sendEvent="UI INACTIVE", delay=0, everyFrame=false)
    on FINISHED → Send
  }

  state Send {
    GetFsmString(gameObject=var "UI List", fsmName="ui_list", variableName="Current Selection", storeValue=var "Selection Result", everyFrame=false)
    SetFsmString(gameObject=var "Requester", fsmName="Stag Control", variableName="Selection Result", setValue=var "Selection Result", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Requester"), sendEvent="CONTINUE", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Selection Result: string = ""
  var Requester: gameObject = <null>
  var Self: gameObject = <null>
  var UI List: gameObject = <null>
}
