# Big Bee
# 1 location(s):
#   Hive_03 / Big Bee (2)

fsm Big Bee {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Fly", doNotResetCurrentClip=false)  // disabled
    BoolTestMulti(boolVariables=[var "Friendly", var "Hero in range"], boolStates=[false, true], trueEvent=→"ALERT", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTest(boolVariable=var "Hero in range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)  // disabled
    IdleBuzz(gameObject=Self, waitMin=0.5, waitMax=1.5, speedMax=4, accelerationMax=15, roamingRange=4)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    on ALERT → Charge Pos
    on TOOK DAMAGE → Unfriendly
  }

  state Charge Antic {
    SetIntValue(intVariable=var "Bounces", intValue=3, everyFrame=false)
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=big_bee_prepare (sharedassets386.assets))
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=1.2, storeAngle=var "Angle", everyFrame=false)
    SetFloatValue(floatVariable=var "Antic Angle", floatValue=var "Angle", everyFrame=false)
    FloatAdd(floatVariable=var "Antic Angle", add=180, everyFrame=false, perSecond=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Antic Angle", speed=5, everyFrame=false)
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Attack Start", doNotResetCurrentClip=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on FINISHED → Charge Start
  }

  state Charge R {
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=1, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Charging Start
  }

  state Attack Bounce {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bounce")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetFloatValue(floatVariable=var "Antic Angle", floatValue=var "Angle", everyFrame=false)
    FloatAdd(floatVariable=var "Antic Angle", add=180, everyFrame=false, perSecond=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Antic Angle", speed=5, everyFrame=true)
    ActivateGameObject(gameObject=var "Smasher", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Return to idle
  }

  state Recover {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Chase
  }

  state Return to idle {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    AudioStop(gameObject=Self)
    SetAudioClip(gameObject=Self, audioClip=big_bee_fly_loop (sharedassets306.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Charge Start {
    SetAudioClip(gameObject=Self, audioClip=big_bee_charge_loop (sharedassets386.assets))
    SetDamageHeroAmount(target=Self, damageDealt=2)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Attack Loop", doNotResetCurrentClip=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=30, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatInRange(floatVariable=var "Angle", lowerValue=90, upperValue=270, boolVariable=(unset), trueEvent=→"LEFT", falseEvent=(none), everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RIGHT", delay=0, everyFrame=false)
    on RIGHT → Charge R
    on LEFT → Charge L
  }

  state Charge L {
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=-1, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Charging Start
  }

  state Charging {
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=25, everyFrame=true)
    on COLLISION ENTER 2D → Check Dir
    on COLLISON STAY 2D → Check Dir
  }

  state Charging Start {
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=30, everyFrame=true)
    ActivateGameObject(gameObject=var "Smasher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Charging
  }

  state Start Chase {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=5, acceleration=0.06, targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Chase
  }

  state Chase {
    BoolTest(boolVariable=var "Hero in range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=5, acceleration=0.06, targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0)
    WaitRandom(timeMin=8, timeMax=10, finishEvent=→"UNALERT", realTime=false)
    on ALERT → Charge Pos
    on UNALERT → Idle
  }

  state Charge Pos {
    DistanceFly(gameObject=Self, target=var "Hero", distance=7, speedMax=5, acceleration=0.08, targetsHeight=true, height=7)
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    on FINISHED → Can See?
  }

  state Check Dir {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=zombie_guard_club (sharedassets34.assets))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=(unset), contactNormal=var "Collision Normal", shapeCount=(unset), physics2dMaterialName="")
    GetVector3XYZ(vector3Variable=var "Collision Normal", storeX=var "Normal X", storeY=var "Normal Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal X", float2=-1, tolerance=0.1, equal=→"RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Normal X", float2=1, tolerance=0.1, equal=→"LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=-1, tolerance=0.1, equal=→"UP", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=1, tolerance=0.1, equal=→"DOWN", lessThan=(none), greaterThan=(none), everyFrame=false)
    on UP → Hit Up
    on DOWN → Hit Down
    on LEFT → Hit Left
    on RIGHT → Hit Right
    on END → Attack Bounce
    on FINISHED → Attack Bounce
  }

  state Hit Vertical {
    ReflectAngle(angle=var "Angle", reflectHorizontally=false, reflectVertically=true, storeResult=var "Angle")
    IntOperator(integer1=var "Bounces", integer2=1, operation=Subtract, storeResult=var "Bounces", everyFrame=false)
    IntCompare(integer1=var "Bounces", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    on FINISHED → Charge Start
    on END → Attack Bounce
  }

  state Hit Horizontal {
    ReflectAngle(angle=var "Angle", reflectHorizontally=true, reflectVertically=false, storeResult=var "Angle")
    IntOperator(integer1=var "Bounces", integer2=1, operation=Subtract, storeResult=var "Bounces", everyFrame=false)
    IntCompare(integer1=var "Bounces", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    on FINISHED → Charge Start
    on END → Attack Bounce
  }

  state Can See? {
    BoolTest(boolVariable=var "Hero in range", isTrue=→"FINISHED", isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Charge Antic
    on CANCEL → Charge Pos
  }

  state Hit Up {
    SpawnObjectFromGlobalPool(gameObject=Slam Effect R (resources.assets), spawnPoint=var "Self", position=(0.46, 0.02, 0), rotation=(0, 0, 180), storeObject=var "")
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(0.46, 0.02, 0), spawnMin=15, spawnMax=20, speedMin=15, speedMax=25, angleMin=230, angleMax=310, originVariationX=2, originVariationY=0, FSM="", FSMEvent="")
    on FINISHED → Hit Vertical
  }

  state Hit Down {
    SpawnObjectFromGlobalPool(gameObject=Slam Effect R (resources.assets), spawnPoint=var "Self", position=(-0.02, -0.76, 0), rotation=(0, 0, 0), storeObject=var "")
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(-0.02, -0.76, 0), spawnMin=15, spawnMax=20, speedMin=15, speedMax=25, angleMin=50, angleMax=130, originVariationX=2, originVariationY=0, FSM="", FSMEvent="")
    on FINISHED → Hit Vertical
  }

  state Hit Left {
    SpawnObjectFromGlobalPool(gameObject=Slam Effect R (resources.assets), spawnPoint=var "Self", position=(-0.36, -0.08, 0), rotation=(0, 0, 270), storeObject=var "")
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(-0.36, -0.08, 0), spawnMin=15, spawnMax=20, speedMin=15, speedMax=25, angleMin=-40, angleMax=40, originVariationX=0, originVariationY=2, FSM="", FSMEvent="")
    on FINISHED → Hit Horizontal
  }

  state Hit Right {
    SpawnObjectFromGlobalPool(gameObject=Slam Effect R (resources.assets), spawnPoint=var "Self", position=(0.36, -0.68, 0), rotation=(0, 0, 90), storeObject=var "")
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(0.36, -0.68, 0), spawnMin=15, spawnMax=20, speedMin=15, speedMax=25, angleMin=140, angleMax=220, originVariationX=0, originVariationY=2, FSM="", FSMEvent="")
    on FINISHED → Hit Horizontal
  }

  state Pause {
    on FINISHED → Friendly?
  }

  state Friendly? {
    GetPlayerDataBool(boolName="equippedCharm_29", storeValue=var "Friendly")
    BoolTest(boolVariable=var "Friendly", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on FINISHED → Init
  }

  state Unfriendly {
    SetBoolValue(boolVariable=var "Friendly", boolValue=false, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=2)
    on FINISHED → Charge Pos
  }

  var Angle: float = 0
  var Antic Angle: float = 0
  var Normal X: float = 0
  var Normal Y: float = 0
  var X Scale: float = 0
  var Bounces: int = 0
  var Friendly: bool = false
  var Hero in range: bool = false
  var Collision Normal: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
  var Smasher: gameObject = Big Bee (2)/Smasher
}
