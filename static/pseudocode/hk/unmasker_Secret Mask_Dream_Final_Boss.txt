# unmasker
# 1 location(s):
#   Dream_Final_Boss / Secret Mask

// uses template: remasker
fsm unmasker {
  start Pause
  on HERO LEAVE → Hero Leave  // from any state

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="Player", collideLayer="", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Fade Out
  }

  state Fade Out {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Player", collideLayer="", sendEvent=→"COVER", storeCollider=var "")
    on COVER → Fade In
  }

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Fade In {
    iTweenFadeTo(gameObject=Self, id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="Player", collideLayer="", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Fade Out
  }

  state Hero Leave {
  }

  var Fade Time: float = 8

  var Activated: bool = false
}
