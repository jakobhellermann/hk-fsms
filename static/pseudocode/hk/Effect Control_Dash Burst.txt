# Effect Control
# 2 location(s):
#   Knight_Pickup / Knight/Effects/Dash Burst
#   resources.assets / Knight:1/Effects/Dash Burst

fsm Effect Control {
  start Idle
  on CANCEL → Idle  // from any state

  state Idle {
    SetMeshRenderer(gameObject=Self, active=false)
    on PLAY → Play
  }

  state Play {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Effect", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }
}
