# Hornet Encounter
# 1 location(s):
#   Fungus1_02 / Hornet Encounter GP1

fsm Hornet Encounter {
  start Init
  on HERO CAST SPELL → Evade Antic  // from any state

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="hornetGreenpath", storeValue=var "PlayerData")
    IntCompare(integer1=var "PlayerData", integer2=0, equal=(none), lessThan=(none), greaterThan=→"DESTROY", everyFrame=false)
    on FINISHED → Idle
    on DESTROY → Remove Cam
  }

  state Idle {
    BoolTest(boolVariable=var "Turn Range", isTrue=→"TURN", isFalse=(none), everyFrame=true)
    on TURN → Turn
  }

  state Turn {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPlayerDataInt(intName="hornetGreenpath", value=1)
    on FINISHED → Turned
  }

  state Turned {
    BoolTest(boolVariable=var "Evade Range", isTrue=→"EVADE", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=5, finishEvent=→"LEAVE", realTime=false)
    on EVADE → Evade Antic
    on LEAVE → TurnToLeave
  }

  state Evade Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Evade Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Evade
  }

  state Evade {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Evade")
    SetVelocity2d(gameObject=Self, vector=(unset), x=-25, y=(unset), everyFrame=false)
    Wait(time=3, finishEvent=→"DESTROY", realTime=false)
    PlayParticleEmitter(gameObject=var "Grass", emit=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=Hornet_Fight_Yell_03 (sharedassets105.assets))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Remove Cam {
    FindGameObject(objectName="CameraLockArea H", withTag="Untagged", store=var "CamLock")
    DestroyObject(gameObject=var "CamLock", delay=0, detachChildren=false)
    on FINISHED → Destroy
  }

  state TurnToLeave {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Leave
  }

  state Leave {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetVelocity2d(gameObject=Self, vector=(unset), x=-9, y=(unset), everyFrame=false)
    Wait(time=5, finishEvent=→"DESTROY", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on DESTROY → Destroy
  }

  var PlayerData: int = 0
  var Evade Range: bool = false
  var Turn Range: bool = false
  var CamLock: gameObject = <null>
  var Grass: gameObject = Hornet Encounter GP1/Grass
}
