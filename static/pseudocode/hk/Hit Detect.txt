# Hit Detect
# 1 location(s):
#   Ruins1_04 / Nailsmith Cliff NPC/Hit Detect

fsm Hit Detect {
  start Detect

  state Detect {
    GetParent(gameObject=Self, storeResult=var "Parent")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"SPELL", storeCollider=var "")
    on NAIL → Nail
    on SPELL → Spell
  }

  state Nail {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="NAIL KILL", delay=0, everyFrame=false)
  }

  state Spell {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="SPELL KILL", delay=0, everyFrame=false)
  }

  var Parent: gameObject = <null>
}
