fsm Worm Control {
  start Initialise

  state Up {
    SetParticleEmission(gameObject=var "Dust", emission=false)
    SetParticleEmission(gameObject=var "Churn Rocks", emission=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on WAIT → Retract
  }

  state Initialise {
    GetOwner(storeGameObject=var "Self")
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Churn Rocks", storeResult=var "Churn Rocks")
    FindChild(gameObject=Self, childName="Dust Worm", storeResult=var "Dust")
    FindChild(gameObject=Self, childName="Burst Rocks", storeResult=var "Burst Rocks")
    BoolTest(boolVariable=var "Start Down", isTrue=→"DOWN", isFalse=(none), everyFrame=false)
    on FINISHED → Up
    on DOWN → Down
  }

  state Retract {
    SetParticleEmission(gameObject=var "Dust", emission=true)
    SetParticleEmission(gameObject=var "Churn Rocks", emission=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Retract")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Down
  }

  state Down {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=worm (sharedassets51.assets), finishedEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Down")
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    SetCollider(gameObject=Self, active=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on WAIT → Burst Rocks?
  }

  state Burst {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Burst")
    SetParticleEmission(gameObject=var "Dust", emission=false)
    SetParticleEmission(gameObject=var "Churn Rocks", emission=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    SetCollider(gameObject=Self, active=true)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on WAIT → Up
  }

  state Burst Rocks? {
    BoolTest(boolVariable=var "Play Burst Rocks", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayParticleEmitter(gameObject=var "Burst Rocks", emit=0)
    on FINISHED → Burst
  }

  var Play Burst Rocks: bool = false
  var Start Down: bool = true

  var Angle: float = 0
  var Burst Rocks: gameObject = <null>
  var Churn Rocks: gameObject = <null>
  var Dust: gameObject = <null>
  var Hero Hurter: gameObject = Mage Orb/Hero Hurter (sharedassets34.assets)
  var Self: gameObject = <null>
}
