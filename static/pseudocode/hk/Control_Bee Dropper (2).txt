# Control
# 2 location(s):
#   Hive_05 / Battle Scene/Droppers/Bee Dropper (2)
#   GG_Hive_Knight / Battle Scene/Droppers/Bee Dropper (2)

// uses template: bee_dropper
fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Pt Burst", storeResult=var "Pt Burst")
    FindChild(gameObject=Self, childName="Splat", storeResult=var "Splat")
    on FINISHED → Idle
  }

  state Idle {
    on SWARM → Pause
  }

  state Swarm Start {
    RandomFloat(min=var "X Left", max=var "X Right", storeResult=var "Random")
    SetPosition(gameObject=Self, vector=(unset), x=var "Random", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    RandomFloat(min=var "X Left", max=var "X Right", storeResult=var "Target X")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    RandomFloat(min=-13, max=-13, storeResult=var "Random")
    on FINISHED → Swarm
  }

  state Pause {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    Wait(time=var "Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Swarm Start
    on BOSS DEFEATED → Reset
  }

  state Swarm {
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=9999, acceleration=0.185, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Random", everyFrame=true)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", space=1, everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=22, tolerance=0, equal=(none), lessThan=→"END", greaterThan=(none), everyFrame=true)
    FaceAngleV2(gameObject=Self, angleOffset=90, worldSpace=true, everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"SPELL", storeCollider=var "")
    on END → Reset
    on SPELL → Spell Death
    on BOSS DEFEATED → Spell Death
  }

  state Reset {
    SetPosition(gameObject=Self, vector=(unset), x=var "Random", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Spell Death {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Splat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Burst", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset
  }

  var Pause: float = 1.7

  var Random: float = 0
  var Start Y: float = 42
  var Target X: float = 0
  var X Left: float = 58.4
  var X Right: float = 80
  var Y Pos: float = 0
  var Pt Burst: gameObject = Bee Dropper/Pt Burst
  var Splat: gameObject = Bee Dropper/Splat
}
