# Grub Mimic
# 4 location(s):
#   Mines_16 / Grub Mimic Top/Grub Mimic
#   Deepnest_36 / Grub Mimic Top (2)/Grub Mimic 3
#   Deepnest_36 / Grub Mimic Top/Grub Mimic 1
#   Deepnest_36 / Grub Mimic Top (1)/Grub Mimic 2

fsm Grub Mimic {
  start Init
  on FALL → Fall  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Burst Point", storeResult=var "Burst Point")
    FindChild(gameObject=Self, childName="Slob Point", storeResult=var "Slob Point")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Burst
  }

  state Burst {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=grub_mimic_transform_start (sharedassets33.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Burst 2
  }

  state Scream {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetActive(false))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=grub_mimic_initial_roar (sharedassets33.assets))
    on FINISHED → Start Run
    on TOOK DAMAGE → Start Run
  }

  state Start Run {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="IdleToRun", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetScale(gameObject=var "Parent", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Chase
  }

  state Chase {
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=15, acceleration=0.5, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToRun", resetFrame=true, everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"FALL", rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"FALL", rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on WALL → Hero Check
  }

  state Hop {
    RandomFloat(min=10, max=10, storeResult=var "Hop Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=var "Hop Speed", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="WallRun")
    NextFrameEvent(sendEvent=→"FINISHED")
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"FALL", rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"FALL", rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on FINISHED → Cooldown
  }

  state Burst 2 {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=grub_mimic_transform_end (sharedassets33.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Burst Point", position=(unset), spawnMin=20, spawnMax=20, speedMin=15, speedMax=25, angleMin=40, angleMax=140, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    on FINISHED → Scream
  }

  state Cooldown {
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=15, acceleration=0.5, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"FALL", rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"FALL", rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Chase
    on WALL → Hero Check
  }

  state Hero Check {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    FloatSubtract(floatVariable=var "Hero Y", subtract=var "Self Y", everyFrame=false, perSecond=false)
    FloatCompare(float1=var "Hero Y", float2=-8, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on FINISHED → Hop
    on CANCEL → Fall
  }

  state Fall {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=15, acceleration=0.5, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToRun", resetFrame=true, everyFrame=true)
    on FINISHED → Chase
  }

  var Hero Y: float = 0
  var Hop Speed: float = 0
  var Self Y: float = 0
  var Burst Point: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Slob Point: gameObject = <null>
}
