# FSM
# 3 location(s):
#   Dream_Guardian_Hegemol / Play Music Strings and Choir
#   Dream_Guardian_Lurien / Play Music Strings and Choir
#   Dream_Guardian_Monomon / Play Music Strings and Choir

fsm FSM {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Pause
  }

  state Play {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    TransitionToAudioSnapshot(snapshot=Sub Area (resources.assets), transitionTime=3)
  }

  state Pause {
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Sub Area (resources.assets), transitionTime=5)
    on FINISHED → Play
  }
}
