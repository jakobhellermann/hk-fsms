    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOW MAP PINS", delay=0, everyFrame=false)
    SetPlayerDataInt(intName="mapKeyPref", value=0)
    SendEventByName(eventTarget=GameObject(var "Backboard Key"), sendEvent="UP", delay=0, everyFrame=false)
    FadeGroupUp(target=Self)
    SetTextMeshProText(gameObject=var "Action Text", textString=var "Str Hide Key")
    ListenForDreamNail(eventTarget=Self, wasPressed=→"ACTION", wasReleased=(none), isPressed=(none), isNotPressed=(none), activeBool=(unset))
    SetCameraCullingMask(gameObject=var "HUD Camera", cullingMask=[UI(5), Map Pin(30)], invertMask=false, everyFrame=false)
    on MAP KEY DOWN → Down
    on ACTION → Pin
  }

  state Down {
    SendEventByName(eventTarget=GameObject(var "Backboard Key"), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Action", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

    on FINISHED → Inactive

  state Check State {
    ActivateGameObject(gameObject=var "Action", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="mapKeyPref", storeValue=var "Map Key Pref")
    IntSwitch(intVariable=var "Map Key Pref", compareTo=[0, 1, 2], sendEvent=[→"KEY AND PIN", →"PIN", →"NONE"], everyFrame=false)
    on KEY AND PIN → Key And Pin
    on PIN → Pin
    on NONE → None
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetLanguageString(sheetName="UI", convName="CTRL_HIDE_KEY", storeValue=var "Str Hide Key")
    GetLanguageString(sheetName="UI", convName="CTRL_HIDE_PINS", storeValue=var "Str Hide Pins")
    GetLanguageString(sheetName="UI", convName="CTRL_SHOW_PINS_KEY", storeValue=var "Str Show Both")
    on FINISHED → Draw Pins
  }

  state Pin {

    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOW MAP PINS", delay=0, everyFrame=false)
    SetTextMeshProText(gameObject=var "Action Text", textString=var "Str Hide Pins")
