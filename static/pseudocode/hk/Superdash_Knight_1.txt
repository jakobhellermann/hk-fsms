# Superdash

fsm Superdash {
  start Init
  on HERO DAMAGED → Cancel  // from any state
  on LEAVING SCENE → Cancel  // from any state
  on FSM CANCEL → Cancel  // from any state
  on ENTER SUPER DASH → Enter Super Dash  // from any state

  state Inactive {
    ListenForSuperdash(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ActivateGameObject(gameObject=var "SuperDash Damage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BUTTON DOWN → Can Superdash?
  }

  state Regain Control {
    ActivateGameObject(gameObject=var "SuperDash Damage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["superDashOnWall", false], storeResult=(unset var))
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["superDashing", false], storeResult=(unset var))
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=RegainControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=StartAnimationControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    on FINISHED → Inactive
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetFloatValue(floatVariable=var "Superdash Speed neg", floatValue=var "Superdash Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Superdash Speed neg", multiplyBy=-1, everyFrame=false)
    FindGameObject(objectName="", withTag="CameraTarget", store=var "Camera Target")
    FindChild(gameObject=Self, childName="Effects", storeResult=var "FX Folder")
    FindChild(gameObject=Self, childName="SuperDash Damage", storeResult=var "SuperDash Damage")
    FindChild(gameObject=var "FX Folder", childName="SD Bling", storeResult=var "Bling Effect")
    FindChild(gameObject=var "FX Folder", childName="SD Trail", storeResult=var "Trail Effect")
    FindChild(gameObject=var "FX Folder", childName="SD Charge", storeResult=var "Charge Effect")
    FindChild(gameObject=var "FX Folder", childName="SD Charge Wall", storeResult=var "Charge Effect Wall")
    FindChild(gameObject=var "FX Folder", childName="SD Break", storeResult=var "SD Break")
    FindChild(gameObject=var "FX Folder", childName="SD Sharp Flash", storeResult=var "Sharp Flash")
    FindChild(gameObject=var "FX Folder", childName="Wall Hit Effect", storeResult=var "Wall Hit Effect")
    FindChild(gameObject=var "FX Folder", childName="Dust Wall Impact", storeResult=var "Dust Wall Impact")
    FindChild(gameObject=var "FX Folder", childName="Wall Rocks", storeResult=var "Wall Rocks")
    FindChild(gameObject=var "FX Folder", childName="SD Energy", storeResult=var "SD Energy")
    FindChild(gameObject=var "FX Folder", childName="SD Burst", storeResult=var "SD Burst")
    FindChild(gameObject=var "FX Folder", childName="SD Charge Glow", storeResult=var "Charge Glow")
    FindChild(gameObject=var "FX Folder", childName="SD Burst Glow", storeResult=var "SD Burst Glow")
    FindChild(gameObject=var "FX Folder", childName="SD Pulse Pt", storeResult=var "Pulse Pt")
    FindChild(gameObject=var "FX Folder", childName="SD Crystal Burst GL", storeResult=var "Crystal Burst GL")
    FindChild(gameObject=var "FX Folder", childName="SD Crystal Burst GR", storeResult=var "Crystal Burst GR")
    FindChild(gameObject=var "FX Folder", childName="SD Crystal Burst W", storeResult=var "Crystal Burst W")
    FindChild(gameObject=Self, childName="SD Check Bot", storeResult=var "SD Check Bot Obj")
    FindChild(gameObject=Self, childName="SD Check Top", storeResult=var "SD Check Top Obj")
    FindChild(gameObject=Self, childName="SD Ground 1", storeResult=var "SD Grd Obj 1")
    FindChild(gameObject=Self, childName="SD Ground 2", storeResult=var "SD Grd Obj 2")
    on FINISHED → Inactive
  }

  state Can Superdash? {
    CallMethodProper(gameObject=var "Self", behaviour="HeroController", methodName="CanSuperDash", parameters=[], storeResult=var "Return Bool")
    BoolTest(boolVariable=var "Return Bool", isTrue=→"FINISHED", isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Inactive
    on FINISHED → Relinquish Control
  }

  state Cancel {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingFocus", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingFocus2", setValue=false, everyFrame=false)
    SendMessage(gameObject=var "Camera Target", delivery=SendMessage, options=1, functionCall=SetSuperDash(false))
    ActivateGameObject(gameObject=var "SuperDash Damage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["superDashOnWall", false], storeResult=(unset var))
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["superDashing", false], storeResult=(unset var))
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["freezeCharge", false], storeResult=(unset var))
    ActivateGameObject(gameObject=var "SD Energy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Charge Effect", active=false)
    SetMeshRenderer(gameObject=var "Charge Effect Wall", active=false)

    SetMeshRenderer(gameObject=var "Trail Effect", active=false)
    VibrationPlayerStop(target=var "Charge Audio")
    SendEventToRegister(eventName="SUPERDASH CANCEL")
    SendEventByName(eventTarget=GameObject(var "Charge Glow"), sendEvent="DOWN", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pulse Pt")
    ActivateGameObject(gameObject=var "Trail Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    VibrationPlayerStop(target=var "Dashing Audio")
    AudioStop(gameObject=var "Dashing Audio")
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingFocus", setValue=false, everyFrame=false)
    on FINISHED → Inactive
  }

  state Relinquish Control {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    on FINISHED → On Ground?
  }

  state On Ground? {
    CallMethodProper(gameObject=var "Self", behaviour="HeroController", methodName="GetState", parameters=["onGround"], storeResult=var "Return Bool")
    BoolTest(boolVariable=var "Return Bool", isTrue=→"GROUND", isFalse=→"OFF GROUND", everyFrame=false)
    on GROUND → Ground Charge
    on OFF GROUND → Wall Charge
  }

  }

  state Charge Cancel Ground {
    VibrationPlayerStop(target=var "Charge Audio")
    AudioStop(gameObject=var "Charge Audio")
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["freezeCharge", false], storeResult=(unset var))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="SD Charge Ground End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SUPERDASH CANCEL", delay=0, everyFrame=false)

    SendEventByName(eventTarget=GameObject(var "Charge Effect"), sendEvent="DOWN", delay=0, everyFrame=false)
