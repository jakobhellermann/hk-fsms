# Msg Control
# 1 location(s):
#   sharedassets137.assets / UI Msg Get WhiteCharm

fsm Msg Control {
  start Init

  state Detect {
    GetButtonDown(buttonName="Jump", sendEvent=→"PRESS", storeResult=(unset))  // disabled
    GetButtonDown(buttonName="Attack", sendEvent=→"PRESS", storeResult=(unset))  // disabled
    GetButtonDown(buttonName="Pause", sendEvent=→"PRESS", storeResult=(unset))  // disabled
    ListenForCast(eventTarget=Self, wasPressed=→"PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), activeBool=(unset), stateEntryOnly=false)
    ListenForJump(eventTarget=Self, wasPressed=→"PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForAttack(eventTarget=Self, wasPressed=→"PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on PRESS → Down
  }

  state Done {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GET ITEM MSG END", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)
  }

  state Stop Up {
    ActivateGameObject(gameObject=var "Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Done
  }

  state Init {
    SetPlayerDataBool(boolName="hasCharm", value=true)
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Charm State")
    IntCompare(integer1=var "Charm State", integer2=4, equal=→"BLACK", lessThan=(none), greaterThan=(none), everyFrame=false)
    BoolTest(boolVariable=var "Queen", isTrue=→"QUEEN", isFalse=→"KING", everyFrame=false)
    on QUEEN → Queen
    on KING → King
    on BLACK → Set Black
  }

  state Queen {
    IntSwitch(intVariable=var "Charm State", compareTo=[0, 2], sendEvent=[→"SINGLE", →"FINAL"], everyFrame=false)
    on SINGLE → Queen S
    on FINAL → Set Full
  }

  state Queen S {
    SetSpriteRendererSprite(gameObject=var "Single Frag", sprite=whtie_charm_large_pieces_0001_left)
    GetLanguageString(sheetName="UI", convName="CHARM_NAME_36_A", storeValue=var "Game Text")  // disabled
    GetLanguageStringProcessed(sheetName="UI", convName="CHARM_NAME_36_A", storeValue=var "Game Text", fontSource=Trajan)
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_ITEM_INTRO1", storeValue=var "Game Text")  // disabled
    GetLanguageStringProcessed(sheetName="Prompts", convName="GET_ITEM_INTRO1", storeValue=var "Game Text", fontSource=Perpetua)
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Set PlayerData QS
  }

  state Single 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=spell_information_screen (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    on FINISHED → S particle
  }

  state S particle {
    PlayParticleEmitter(gameObject=var "Pt Single", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Single Icon
  }

  state Single Icon {
    ActivateGameObject(gameObject=var "Single Frag", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Text
  }

  state Text {
    SendEventByName(eventTarget=Self, sendEvent="", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name Prefix"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    StopParticleEmitter(gameObject=var "Pt Single")
    on FINISHED → Stop Up
  }

  state Set PlayerData QS {
    SetPlayerDataInt(intName="royalCharmState", value=1)
    on FINISHED → Single 1
  }

  state King {
    IntSwitch(intVariable=var "Charm State", compareTo=[0, 1], sendEvent=[→"SINGLE", →"FINAL"], everyFrame=false)
    on SINGLE → King S
    on FINAL → Set Full
  }

  state King S {
    SetSpriteRendererSprite(gameObject=var "Single Frag", sprite=whtie_charm_large_pieces_0000_right)
    GetLanguageString(sheetName="UI", convName="CHARM_NAME_36_A", storeValue=var "Game Text")  // disabled
    GetLanguageStringProcessed(sheetName="UI", convName="CHARM_NAME_36_A", storeValue=var "Game Text", fontSource=Trajan)
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_ITEM_INTRO1", storeValue=var "Game Text")  // disabled
    GetLanguageStringProcessed(sheetName="Prompts", convName="GET_ITEM_INTRO1", storeValue=var "Game Text", fontSource=Perpetua)
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Set PlayerData KS
  }

  state Set Full {
    SetPlayerDataInt(intName="royalCharmState", value=3)
    SetPlayerDataInt(intName="charmCost_36", value=5)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="charmsOwned")
    GetLanguageString(sheetName="UI", convName="CHARM_NAME_36_B", storeValue=var "Game Text")  // disabled
    GetLanguageStringProcessed(sheetName="UI", convName="CHARM_NAME_36_B", storeValue=var "Game Text", fontSource=Trajan)
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_ITEM_INTRO5", storeValue=var "Game Text")  // disabled
    GetLanguageStringProcessed(sheetName="Prompts", convName="GET_ITEM_INTRO5", storeValue=var "Game Text", fontSource=Perpetua)
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Form 1
  }

  state Set PlayerData KS {
    SetPlayerDataInt(intName="royalCharmState", value=2)
    on FINISHED → Single 1
  }

  state Form 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Anim
  }

  state Anim {
    ActivateGameObject(gameObject=var "White Form", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "White Form", clipName="White Form", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=ghost_absorb_final_impact (resources.assets), pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Pt Form
  }

  state Pt Form {
    PlayParticleEmitter(gameObject=var "Pt Form", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=spell_information_screen (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=new_heartpiece_puzzle_bit (resources.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Text
  }

  state Set Black {
    GetLanguageString(sheetName="UI", convName="CHARM_NAME_36_C", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_ITEM_INTRO5", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    SetPlayerDataBool(boolName="gotShadeCharm", value=true)
    on FINISHED → Split 1
  }

  state Split 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Anim 2
  }

  state Anim 2 {
    ActivateGameObject(gameObject=var "White Form", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "White Form", vector=(unset), x=0.19, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "White Form", clipName="White Split", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=new_heartpiece_puzzle_bit (resources.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=dark_spell_get (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=1.5, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=ghost_absorb_final_impact (resources.assets), pitchMin=0.9, pitchMax=0.9, volume=1, delay=1.5, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Hero", audioClip=spell_information_screen (resources.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=1.5, storePlayer=<null>)
    on FINISHED → Achievement
  }

  state Achievement {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["VOID"], storeResult=(unset var))
    on FINISHED → Text
  }

  var Charm State: int = 0
  var Queen: bool = false
  var Game Text: string = ""
  var Item Name: gameObject = UI Msg Get WhiteCharm/Item Name
  var Item Name Prefix: gameObject = UI Msg Get WhiteCharm/Item Name Prefix
  var Pt Form: gameObject = UI Msg Get WhiteCharm/Pt Form
  var Pt Single: gameObject = UI Msg Get WhiteCharm/Pt Single
  var Single Frag: gameObject = UI Msg Get WhiteCharm/Single Frag
  var Stop: gameObject = UI Msg Get WhiteCharm/Stop
  var White Flash: gameObject = UI Msg Get WhiteCharm/White Flash
  var White Form: gameObject = UI Msg Get WhiteCharm/White Form
}
