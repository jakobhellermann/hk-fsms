# Control
# 1 location(s):
#   Hive_05 / Battle Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="CamLock B", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Escape Blocker", storeResult=var "Escape Blocker")
    FindChild(gameObject=Self, childName="Hive Knight", storeResult=var "Hive Knight")
    FindChild(gameObject=Self, childName="Droppers", storeResult=var "Droppers")
    FindChild(gameObject=Self, childName="Swarm Audio", storeResult=var "Swarm Audio")
    FindChild(gameObject=Self, childName="Shake Drip", storeResult=var "Shake Drip")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedHiveKnight", isTrue=→"INERT", isFalse=(none))
    on FINISHED → Idle
    on INERT → Inert
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"WAKE", storeCollider=var "")
    on WAKE → Start Pause
  }

  state Start Pause {
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=4)
    Wait(time=2.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Droppers
  }

  state Droppers {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Swarm Audio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Droppers"), sendEvent="SWARM", delay=1, everyFrame=false)
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Shake Drip", emit=0)
    on FINISHED → Hive Knight
  }

  state Hive Knight {
    SendEventByName(eventTarget=GameObject(var "Hive Knight"), sendEvent="WAKE", delay=0, everyFrame=false)
    on BATTLE END → End
  }

  state End {
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Escape Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BG OPEN", delay=0, everyFrame=false)
    ApplyMusicCue(musicCue=Hive (sharedassets384.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=4)
  }

  state Inert {
    ActivateGameObject(gameObject=var "Escape Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var CamLock: gameObject = <null>
  var Droppers: gameObject = <null>
  var Escape Blocker: gameObject = <null>
  var Hive Knight: gameObject = <null>
  var Shake Drip: gameObject = <null>
  var Swarm Audio: gameObject = <null>
}
