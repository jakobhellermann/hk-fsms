# Control
# 4 location(s):
#   Room_Colosseum_Bronze / Colosseum Manager/Crowd Audio
#   Room_Colosseum_Silver / Colosseum Manager/Crowd Audio
#   Room_Colosseum_Gold / Colosseum Manager/Crowd Audio
#   Room_Colosseum_Spectate / Crowd Audio

fsm Control {
  start Start
  on CROWD STILL → Go Silent  // from any state
  on CROWD IDLE → Idle  // from any state
  on CROWD CHEER → Cheer  // from any state
  on CROWD GASP → Gasp  // from any state
  on HERO LEAVE → Hero Death  // from any state
  on CROWD LAUGH → Laugh  // from any state
  on CROWD MURMUR → Murmur Pause  // from any state
  on CROWD SHORT CHEER → Cheer  // from any state

  state Idle {
    SetAudioClip(gameObject=Self, audioClip=crowd_idle_loop (sharedassets32.assets))
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → 
    on HERO DAMAGED → Damage Pause
  }

  state Cheer {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=crowd_cheer (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Go Silent {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.45)
    on FINISHED → 
  }

  state Gasp {
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=crowd_gasp (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Start {
  }

  state Damage Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gasp
  }

  state Hero Death {
    FinishFSM()
  }

  state Laugh {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=crowd_laugh_3_long (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Murmur Pause {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.5)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Murmur
  }

  state Murmur {
    SetAudioClip(gameObject=Self, audioClip=crowd_murmur_loop (sharedassets32.assets))
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }
}
