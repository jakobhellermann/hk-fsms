# Hit Crystals
# 1 location(s):
#   Mines_33 / Crystal Crawler (2)

// component disabled: this FSM does not run until something enables it
fsm Hit Crystals {
  start Idle

  state Idle {
    SetParticleEmission(gameObject=var "Hit Crystals", emission=false)
    on BLOCKED HIT → Particle effect
  }

  state Particle effect {
    PlayParticleEmitter(gameObject=var "Hit Crystals", emit=0)  // disabled
    SetParticleEmissionRate(gameObject=Self, emissionRate=0, everyFrame=false)  // disabled
    SetParticleEmission(gameObject=var "Hit Crystals", emission=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mines_crawler_hit_top (sharedassets249.assets))
    on FINISHED → Idle
  }

  var Hit Crystals: gameObject = Crystal Crawler (2)/Hit Crystals Effect
}
