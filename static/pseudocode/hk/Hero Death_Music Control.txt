# Hero Death
# 3 location(s):
#   Room_Colosseum_Bronze / Colosseum Manager/Music Control
#   Room_Colosseum_Silver / Colosseum Manager/Music Control
#   Room_Colosseum_Gold / Colosseum Manager/Music Control

fsm Hero Death {
  start Idle

  state Idle {
    on HERO LEAVE → Pause
  }

  state Pause {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.25)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Final Sting
  }

  state Final Sting {
    AudioStop(gameObject=Self)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=S57 COLOSSEUM STING (sharedassets32.assets))
  }
}
