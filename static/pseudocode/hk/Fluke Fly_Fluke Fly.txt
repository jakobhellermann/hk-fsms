fsm Fluke Fly {
  start Init

  state Startle {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=fluke_fairy_call (sharedassets27.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Chase
  }

  state Chase {
    SetAudioClip(gameObject=Self, audioClip=fluke_fairy_charge_loop (sharedassets350.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    FaceAngle(gameObject=Self, angleOffset=180, everyFrame=true)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=15, accelerationForce=35, offsetX=0, offsetY=0)
    WaitRandom(timeMin=5, timeMax=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Fluke Mother
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetAudioClip(gameObject=Self, audioClip=fluke_fairy_swim_loop (sharedassets350.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    SetRotation(gameObject=var "Self", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=true, lateUpdate=false)
    IdleBuzzV2(gameObject=Self, waitMin=0, waitMax=0, speedMax=20, accelerationMax=50, roamingRangeX=4, roamingRangeY=4, manualStartPos=(unset))
    BoolTest(boolVariable=var "In Range", isTrue=→"HERO IN RANGE", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "FLUKE MOTHER", isTrue=→"HERO IN RANGE", isFalse=(none), everyFrame=true)
    on HERO IN RANGE → Startle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    on FINISHED → Idle
  }

  state Decelerate {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=true)
    FloatCompare(float1=var "Speed", float2=0, tolerance=1, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=true)
    on FINISHED → To Idle
  }

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Idle
  }

  state To Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="To Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetAngularVelocity2d(gameObject=var "Self", angularVelocity=0, everyFrame=false)
    SetRotation(gameObject=var "Self", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetAudioClip(gameObject=Self, audioClip=fluke_fairy_swim_loop (sharedassets350.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Wait
  }

  state Check Fluke Mother {
    BoolTest(boolVariable=var "In Range", isTrue=→"HERO IN RANGE", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Decelerate
    on HERO IN RANGE → Chase
  }

  var Speed: float = 0
  var FLUKE MOTHER: bool = false
  var In Range: bool = false
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
