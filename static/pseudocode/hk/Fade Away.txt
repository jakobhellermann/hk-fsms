# Fade Away
# 1 location(s):
#   sharedassets6.assets / Screen Flash

fsm Fade Away {
  start Fade In

  state Fade In {
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=30, space=1, everyFrame=true, lateUpdate=false)
    SetParent(gameObject=Self, parent=var "MainCamera", resetLocalPosition=false, resetLocalRotation=false)
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))
    SetColorRGBA(colorVariable=var "Prev Colour", red=(unset), green=(unset), blue=(unset), alpha=var "Opacity", everyFrame=false)
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Target Colour", fail=(none))
    SetColorRGBA(colorVariable=var "Target Colour", red=(unset), green=(unset), blue=(unset), alpha=0, everyFrame=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Target Colour", everyFrame=false)
    SetColorValue(colorVariable=var "New Colour", color=var "Target Colour", everyFrame=false)
    EaseColor(fromValue=var "Target Colour", toValue=var "Prev Colour", colorVariable=var "New Colour", time=var "In Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    on FINISHED → Fade Out
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Fade Out {
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=30, space=1, everyFrame=true, lateUpdate=false)
    SetColorValue(colorVariable=var "New Colour", color=var "Prev Colour", everyFrame=false)
    SetParent(gameObject=Self, parent=var "MainCamera", resetLocalPosition=false, resetLocalRotation=false)
    EaseColor(fromValue=var "Prev Colour", toValue=var "Target Colour", colorVariable=var "New Colour", time=var "Out Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    on FINISHED → Destroy
  }

  var In Time: float = 0.25
  var Opacity: float = 1
  var Out Time: float = 0.5
  var New Colour: color = (0, 0, 0, 1)
  var Prev Colour: color = (0, 0, 0, 1)
  var Target Colour: color = (0, 0, 0, 1)
}
