  on ZERO HP → Death Start  // from any state

  state Init {

    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="GetSceneHeight", parameters=[], storeResult=var "Scene Height")

    FindChild(gameObject=Self, childName="Hero Detector", storeResult=var "Hero Detector")
    FindChild(gameObject=Self, childName="Red Flash 1", storeResult=var "Red Flash 1")
    FindChild(gameObject=Self, childName="Red Flash 2", storeResult=var "Red Flash 2")
    FindChild(gameObject=Self, childName="Pt Orbs", storeResult=var "Pt Orbs")
    FindChild(gameObject=Self, childName="Pt Smoke", storeResult=var "Pt Smoke")
    FindChild(gameObject=Self, childName="Pt Dash", storeResult=var "Pt Dash")
    FindChild(gameObject=Self, childName="Pt Start", storeResult=var "Pt Start")
    FindChild(gameObject=Self, childName="Flame Spawn Point", storeResult=var "Flame Spawn Point")
    FindChild(gameObject=Self, childName="Follow Target", storeResult=var "Follow Target")
    FindChild(gameObject=Self, childName="Pt Dying", storeResult=var "Pt Dying")
    FindChild(gameObject=Self, childName="Explode Effects", storeResult=var "Explode Effects")

    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Spiral Fire", storeResult=var "Pt Spiral Fire")
    FindChild(gameObject=Self, childName="Pt Spiral Charge", storeResult=var "Pt Spiral Charge")
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 0), everyframe=false)
    on START → Set Level
  }

  state Set Level {
    IntSwitch(intVariable=var "Grimmchild Level", compareTo=[1, 2, 3], sendEvent=[→"LEVEL 1", →"LEVEL 2", →"LEVEL 3"], everyFrame=false)
    on LEVEL 1 → Level 1
    on LEVEL 2 → Level 2
    on LEVEL 3 → Level 3
    on FINISHED → Level 1
  }

  state Alert Pause {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Music
  }

  state Follow {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SendEventByName(eventTarget=GameObject(var "Laugh Loop"), sendEvent="IN", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Pt Smoke", emission=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)

    FloatOperator(float1=var "Hero Y", float2=var "Y Distance", operation=Add, storeResult=var "Y Target", everyFrame=true)

  }

    SetParticleEmission(gameObject=var "Pt Smoke", emission=true)  // disabled
    SetPosition(gameObject=Self, vector=(unset), x=var "X Target", y=var "Y Target", z=(unset), space=0, everyFrame=false, lateUpdate=false)  // disabled

    FloatOperator(float1=var "Hero Y", float2=var "Y Distance", operation=Add, storeResult=var "Y Target", everyFrame=true)

    ActivateGameObject(gameObject=var "Red Flash 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Follow
  }

  state Set Angle {
    SetFloatValue(floatVariable=var "X Distance", floatValue=12, everyFrame=false)
    RandomFloat(min=0, max=6, storeResult=var "Y Distance")
    FloatSubtract(floatVariable=var "X Distance", subtract=var "Y Distance", everyFrame=false, perSecond=false)
    RandomlyFlipFloat(storeResult=var "X Distance")
    RandomlyFlipFloat(storeResult=var "Y Distance")  // disabled
    BoolTest(boolVariable=var "Force Left", isTrue=→"L", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Force Right", isTrue=→"R", isFalse=(none), everyFrame=false)
    on FINISHED → Tele Pos
    on L → Stay L
    on R → Stay R
  }
