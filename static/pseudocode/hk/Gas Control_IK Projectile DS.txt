# Gas Control
# 1 location(s):
#   sharedassets343.assets / IK Projectile DS

fsm Gas Control {
  start Reset

  state Reset {
    SetPolygonCollider(gameObject=Self, active=true)
    SetScale(gameObject=Self, vector=(unset), x=2, y=2, z=2, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Particle", emit=0)
    SetFloatValue(floatVariable=var "Scaler", floatValue=1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-12, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=2, stretchMinX=1, stretchMaxY=2, scaleModifier=2, everyFrame=true)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=(unset), y=22, vector3=(unset), everyFrame=true)
    Wait(time=8, finishEvent=→"END", realTime=false)
    on END → Stop
    on ORBIT SHIELD → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Stop {
    StopParticleEmitter(gameObject=var "Particle")
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    SetPolygonCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  var Scaler: float = 0
  var Wait Time: float = 0
  var Particle: gameObject = IK Projectile DS/Gas Attack
}
