fsm Heal Soul {
  start Idle

  state Idle {
    SetFsmBool(gameObject=Self, fsmName="Sound", variableName="Soul", setValue=false, everyFrame=false)
    on START HEAL → Start Heal
  }

  state Heal Pause {
    Wait(time=0.05, finishEvent=→"WAIT", realTime=false)
    on STOP HEAL → Idle
    on WAIT → Heal
  }

  state Heal {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="TryAddMPChargeSpa", parameters=[2], storeResult=var "Did Heal")
    on STOP HEAL → Idle
    on FINISHED → Stop Heal?
  }

  state Start Heal {
    SetFsmBool(gameObject=Self, fsmName="Sound", variableName="Soul", setValue=true, everyFrame=false)
    on FINISHED → Heal
  }

  state Stop Heal? {
    BoolTest(boolVariable=var "Did Heal", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Sound", variableName="Soul", setValue=false, everyFrame=false)
    on FINISHED → Heal Pause
  }

  var Did Heal: bool = false
}
