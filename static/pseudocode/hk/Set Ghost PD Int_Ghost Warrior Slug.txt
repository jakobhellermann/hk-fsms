fsm Set Ghost PD Int {
  start Idle

  state Idle {
    on ZERO HP → Set
  }

  state Set {
    SetPlayerDataInt(intName=var "PD Int", value=1)
  }

  var PD Int: string = "aladarSlugDefeated"
}
