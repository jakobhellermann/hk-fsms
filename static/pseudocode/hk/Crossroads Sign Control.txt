# Crossroads Sign Control
# 1 location(s):
#   Crossroads_01 / _Scenery/sign_post_03/Crossroads Sign Post

fsm Crossroads Sign Control {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="Active", storeResult=var "Active")
    FindChild(gameObject=var "Active", childName="Crossroads Sign Post Broken", storeResult=var "Broken Sprite")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="menderSignBroken", isTrue=→"IS BROKEN", isFalse=(none))
    on FINISHED → Broken Previously?
    on IS BROKEN → Mender Alive?
  }

  state Idle {
    on BREAK → Break
  }

  state Mended {
    SetPlayerDataBool(boolName="menderSignBroken", value=false)  // disabled
    SetSpriteRenderer(gameObject=Self, active=false)
    SetSpriteRenderer(gameObject=var "Broken Sprite", active=true)
    on FINISHED → Idle
  }

  state Break {
    SetPlayerDataBool(boolName="menderSignBroken", value=true)
    GetPlayerDataInt(gameObject=var "GameManager", intName="menderState", storeValue=var "Mender State")
    IntSwitch(intVariable=var "Mender State", compareTo=[0], sendEvent=[→"FIRST TIME"], everyFrame=false)
    on FIRST TIME → First Time
  }

  state First Time {
    SetPlayerDataInt(intName="menderState", value=1)
  }

  state Mender Alive? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="menderState", storeValue=var "Mender State")
    IntCompare(integer1=var "Mender State", integer2=2, equal=→"PERMANENT", lessThan=→"IS BROKEN", greaterThan=→"PERMANENT", everyFrame=false)
    on IS BROKEN → Mended
    on PERMANENT → Permanent
  }

  state Permanent {
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Broken Previously? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="menderState", storeValue=var "Mender State")
    IntCompare(integer1=var "Mender State", integer2=1, equal=→"IS BROKEN", lessThan=(none), greaterThan=→"IS BROKEN", everyFrame=false)
    on IS BROKEN → Mended
    on FINISHED → Idle
  }

  var Mender State: int = 0
  var Active: gameObject = <null>
  var Broken Sprite: gameObject = <null>
}
