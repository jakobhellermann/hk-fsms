fsm Anim Control {
  start Pause

  state Detect {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO R", leftEvent=→"HERO L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on HERO R → Hero R
    on HERO L → Hero L
  }

  state Hero R {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"HERO L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Outside ToR")
    on HERO L → Hero L
  }

  state Hero L {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO R", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Outside ToL")
    on HERO R → Hero R
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }
}
