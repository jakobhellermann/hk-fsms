# Switch Active
# 2 location(s):
#   Room_Tram / Tram Console L
#   Room_Tram_RG / Tram Console L

fsm Switch Active {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on ON → On
    on OFF → Off
  }

  state On {
    ActivateGameObject(gameObject=var "Lit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Inspect Region", active=true)
    SendEventByName(eventTarget=GameObject(var "Lit"), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Off {
    ActivateGameObject(gameObject=var "Lit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Inspect Region", active=false)
    SendEventByName(eventTarget=GameObject(var "Lit"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Flash: gameObject = Tram Console L/Flash
  var Inspect Region: gameObject = Tram Console L/Inspect Region
  var Lit: gameObject = Tram Console L/Lit
}
