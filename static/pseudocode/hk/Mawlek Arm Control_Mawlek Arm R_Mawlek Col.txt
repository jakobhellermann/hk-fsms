# Mawlek Arm Control
# 1 location(s):
#   sharedassets34.assets / Mawlek Col/Mawlek Arm R

fsm Mawlek Arm Control {
  start Pause
  on SLEEP → Dormant  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetProperty(targetProperty=PolygonCollider2D.enabled on Mawlek Col/Mawlek Arm R@PolygonCollider2D, everyFrame=false)
    GetChild(gameObject=Self, childName="Attack Range", withTag="Untagged", storeResult=var "Alerter")
    on FINISHED → Dormant
  }

  state Idle {
    SetBoolValue(boolVariable=var "Active", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Arm Idle")
    GetFsmBool(gameObject=var "Alerter", fsmName="alert_range", variableName="In Attack Range", storeValue=var "In Range", everyFrame=true)  // disabled
    CheckAlertRangeByName(storeResult=var "In Range", childName="Attack Range", everyFrame=true)
    BoolTest(boolVariable=var "In Range", isTrue=→"TRUE", isFalse=(none), everyFrame=true)
    on TRUE → Swipe Antic
  }

  state Swipe Antic {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mawlek_call, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Arm Swipe Antic")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Active", boolValue=true, everyFrame=false)
    on FINISHED → Swipe
  }

  state Swipe {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mawlek_whip, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Arm Swipe")
    SetProperty(targetProperty=PolygonCollider2D.enabled on Mawlek Col/Mawlek Arm R@PolygonCollider2D, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Swipe Cooldown
  }

  state Swipe Cooldown {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Arm Swipe Cooldown")
    SetProperty(targetProperty=PolygonCollider2D.enabled on Mawlek Col/Mawlek Arm R@PolygonCollider2D, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Re attack Pause
  }

  state Dormant {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dummy Blank")
    SetProperty(targetProperty=PolygonCollider2D.enabled on Mawlek Col/Mawlek Arm R@PolygonCollider2D, everyFrame=false)
    SetBoolValue(boolVariable=var "Active", boolValue=false, everyFrame=false)
    on WAKE → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Re attack Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Active", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Active: bool = false
  var In Range: bool = false
  var Alerter: gameObject = <null>
  var Self: gameObject = <null>
}
