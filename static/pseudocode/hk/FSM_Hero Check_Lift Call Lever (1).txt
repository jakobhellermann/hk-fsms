fsm FSM {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=true, everyFrame=false)
    on FINISHED → Detect
  }

  state Exit {
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=false, everyFrame=false)
    on FINISHED → Detect
  }

  var Bool Name: string = "Hero Check"
  var FSM Name: string = "Control Lever"

  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
