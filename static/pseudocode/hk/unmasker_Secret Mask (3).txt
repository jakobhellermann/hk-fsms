// uses template: remasker_inverse
fsm unmasker {
  start Pause
  on HERO LEAVE → Hero Leave  // from any state

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="Player", collideLayer="", sendEvent=→"UNCOVER", storeCollider=var "")
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.01, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenFadeTo(gameObject=var "Inverse Mask", id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=0.01, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on UNCOVER → Fade Out
  }

  state Fade Out {
    iTweenFadeTo(gameObject=Self, id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenFadeTo(gameObject=var "Inverse Mask", id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Player", collideLayer="", sendEvent=→"COVER", storeCollider=var "")
    on COVER → Fade In
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    FindChild(gameObject=Self, childName="Inverse Mask", storeResult=var "Inverse Mask")
    iTweenFadeTo(gameObject=var "Inverse Mask", id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=false)
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Fade In {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenFadeTo(gameObject=var "Inverse Mask", id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="Player", collideLayer="", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Fade Out
  }

  state Hero Leave {
  }

  var Fade Time: float = 0.5

  var Activated: bool = false
  var Inverse Mask: gameObject = <null>
}
