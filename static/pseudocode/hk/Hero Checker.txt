# Hero Checker
# 1 location(s):
#   Mines_31 / Super Dash Get/Far Rumbler

fsm Hero Checker {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Close
  }

  state Close {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CLOSE", delay=0, everyFrame=false)
    on EXIT → Away
  }

  state Away {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="AWAY", delay=0, everyFrame=false)
    on ENTER → Close
  }

  var Glow: gameObject = <null>
  var Parent: gameObject = <null>
}
