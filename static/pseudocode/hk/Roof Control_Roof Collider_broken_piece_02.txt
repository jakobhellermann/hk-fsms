# Roof Control
# 4 location(s):
#   Ruins2_10b / dung_defender_wall_broken/broken_piece_02/Roof Collider
#   Deepnest_01 / break_roof/Roof Collider
#   Abyss_01 / dung_defender_wall_broken/broken_piece_02/Roof Collider
#   Crossroads_10 / Fk Break Wall/Broken/Roof Collider

fsm Roof Control {
  start Idle

  state Idle {
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    on COLLIDE → Collide
  }

  state Collide {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=CancelSuperDash())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=CancelHeroJump())
    on FINISHED → Idle
  }
}
