# Floor Control
# 1 location(s):
#   Dream_01_False_Knight / FK Floor

fsm Floor Control {
  start Pause

  state Idle {
    FindChild(gameObject=Self, childName="Break Floor", storeResult=var "Break Floor")
    FindChild(gameObject=Self, childName="Broken", storeResult=var "Broken")
    FindChild(gameObject=Self, childName="Destroy Dust", storeResult=var "Break Dust")
    FindChild(gameObject=Self, childName="Cracked 1", storeResult=var "Cracked 1")
    FindChild(gameObject=Self, childName="Cracked 2", storeResult=var "Cracked 2")
    FindChild(gameObject=Self, childName="Normal 1", storeResult=var "Normal 1")
    FindChild(gameObject=Self, childName="Normal 2", storeResult=var "Normal 2")
    FindChild(gameObject=Self, childName="Floor Bit 1", storeResult=var "Floor Bit 1")
    FindChild(gameObject=Self, childName="Floor Bit 2", storeResult=var "Floor Bit 2")
    FindChild(gameObject=Self, childName="Floor Bit 3", storeResult=var "Floor Bit 3")
    GetOwner(storeGameObject=var "Self")
    on CRACK → Crack
    on DESTROY → Break
    on ACTIVATE → Activate
  }

  state Crack {
    ActivateGameObject(gameObject=var "Normal 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Normal 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cracked 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cracked 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on CRACK → Break
    on DESTROY → Break
  }

  state Break {
    SpawnRandomObjectsV2(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=200, spawnMax=200, speedMin=10, speedMax=32, angleMin=80, angleMax=100, originVariationX=7, originVariationY=3)  // disabled
    SpawnRandomObjectsV2(gameObject=Particle Rock Large (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=50, spawnMax=50, speedMin=10, speedMax=32, angleMin=80, angleMax=100, originVariationX=7, originVariationY=3)  // disabled
    FlingObjects(containerObject=var "Rubble", adjustPosition=(7, 3, 0), randomisePosition=true, speedMin=10, speedMax=32, angleMin=80, angleMax=100)
    PlayParticleEmitter(gameObject=var "Break Dust", emit=0)
    ActivateGameObject(gameObject=var "Cracked 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cracked 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Floor", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Normal 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Normal 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=var "Floor Bit 1", vector=(unset), x=-3, y=8, everyFrame=false)
    SetVelocity2d(gameObject=var "Floor Bit 2", vector=(unset), x=2, y=6, everyFrame=false)
    SetVelocity2d(gameObject=var "Floor Bit 3", vector=(unset), x=-2, y=9, everyFrame=false)
    SetSpriteRenderer(gameObject=var "Floor Bit 1", active=true)
    SetSpriteRenderer(gameObject=var "Floor Bit 2", active=true)
    SetSpriteRenderer(gameObject=var "Floor Bit 3", active=true)
  }

  state Activate {
    ActivateGameObject(gameObject=var "Cracked 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cracked 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Floor", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Normal 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Normal 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Bit 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Bit 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Bit 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Floor Bit 1", vector=(unset), x=(unset), y=-22, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Floor Bit 2", vector=(unset), x=(unset), y=-22, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Floor Bit 3", vector=(unset), x=(unset), y=-22, z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  state Spawn Rubble {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Rubble", storeResult=var "Rubble")
    SpawnObjectFromGlobalPool(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "Particle")
    SetParent(gameObject=var "Particle", parent=var "Rubble", resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Loop Check
  }

  state Loop Check {
    IntAdd(intVariable=var "Counter", add=1, everyFrame=false)
    IntCompare(integer1=var "Counter", integer2=100, equal=→"END", lessThan=(none), greaterThan=→"END", everyFrame=false)
    on FINISHED → Spawn Rubble
    on END → Reset
  }

  state Reset {
    SetIntValue(intVariable=var "Counter", intValue=0, everyFrame=false)
    on FINISHED → Spawn Rubble L
  }

  state Spawn Rubble L {
    GetOwner(storeGameObject=var "Self")
    CreateObject(gameObject=Particle Rock Large (resources.assets), spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "Particle", networkInstantiate=false, networkGroup=0)
    SetParent(gameObject=var "Particle", parent=var "Rubble", resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Loop Check 2
  }

  state Loop Check 2 {
    IntAdd(intVariable=var "Counter", add=1, everyFrame=false)
    IntCompare(integer1=var "Counter", integer2=50, equal=→"END", lessThan=(none), greaterThan=→"END", everyFrame=false)
    on FINISHED → Spawn Rubble L
    on END → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Check Broken {
    BoolTest(boolVariable=var "Dream", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="falseKnightDefeated", isTrue=→"ACTIVATE", isFalse=(none))
    on ACTIVATE → Activate
    on FINISHED → Spawn Rubble
  }

  state Init {
    FindChild(gameObject=Self, childName="Break Floor", storeResult=var "Break Floor")
    FindChild(gameObject=Self, childName="Broken", storeResult=var "Broken")
    FindChild(gameObject=Self, childName="Destroy Dust", storeResult=var "Break Dust")
    FindChild(gameObject=Self, childName="Cracked 1", storeResult=var "Cracked 1")
    FindChild(gameObject=Self, childName="Cracked 2", storeResult=var "Cracked 2")
    FindChild(gameObject=Self, childName="Normal 1", storeResult=var "Normal 1")
    FindChild(gameObject=Self, childName="Normal 2", storeResult=var "Normal 2")
    FindChild(gameObject=Self, childName="Floor Bit 1", storeResult=var "Floor Bit 1")
    FindChild(gameObject=Self, childName="Floor Bit 2", storeResult=var "Floor Bit 2")
    FindChild(gameObject=Self, childName="Floor Bit 3", storeResult=var "Floor Bit 3")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check Broken
  }

  var Dream: bool = true

  var Counter: int = 0
  var Break Dust: gameObject = <null>
  var Break Floor: gameObject = <null>
  var Broken: gameObject = <null>
  var Cracked 1: gameObject = <null>
  var Cracked 2: gameObject = <null>
  var Floor Bit 1: gameObject = <null>
  var Floor Bit 2: gameObject = <null>
  var Floor Bit 3: gameObject = <null>
  var Normal 1: gameObject = <null>
  var Normal 2: gameObject = <null>
  var Particle: gameObject = <null>
  var Rubble: gameObject = <null>
  var Self: gameObject = <null>
}
