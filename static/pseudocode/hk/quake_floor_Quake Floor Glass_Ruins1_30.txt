// uses template: quake_floor
fsm quake_floor {
  start Pause

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="Active", storeResult=var "Active")
    FindChild(gameObject=Self, childName="Pt Break Rocks", storeResult=var "Pt Break Rocks")
    FindChild(gameObject=Self, childName="Debris", storeResult=var "Debris")
    FindChild(gameObject=Self, childName="Inactive", storeResult=var "Inactive")
    FindChild(gameObject=Self, childName="Effects", storeResult=var "Effects")
    on FINISHED → Solid
    on ACTIVATE → Activate
  }

  state Solid {
    SetBoxColliderTrigger(gameObject=Self, trigger=false)
    on QUAKE FALL START → Transient
  }

  state Transient {
    SetBoxColliderTrigger(gameObject=Self, trigger=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"DESTROY", storeCollider=var "")
    on QUAKE FALL END → Solid
    on DESTROY → PD Bool?
  }

  state Destroy {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var ""), sendEvent="BigShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Active", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inactive", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Debris", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Debris", activate=true)
    SetCollider(gameObject=Self, active=false)
    FlingObjects(containerObject=var "Debris", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=5, speedMax=20, angleMin=0, angleMax=360)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    DestroySelf(detachChildren=false)
  }

  state PD Bool? {
    StringCompare(stringVariable=var "Playerdata Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "Playerdata Bool", value=true)
    on FINISHED → Audio
  }

  state Audio {
    BoolTest(boolVariable=var "Glass", isTrue=→"GLASS", isFalse=→"NORMAL", everyFrame=false)
    on NORMAL → Normal
    on GLASS → Glass
  }

  state Normal {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Active", audioClips=[breakable_wall_death (sharedassets6.assets), breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Pt Break Rocks", emit=0)
    on FINISHED → Destroy
  }

  state Glass {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Active", audioClips=[crystal_break_1 (resources.assets), crystal_break_2 (resources.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Destroy
  }

  var Glass: bool = true
  var Playerdata Bool: string = ""

  var Activated: bool = false
  var Active: gameObject = Quake Floor/Active
  var Debris: gameObject = Quake Floor/Debris
  var Effects: gameObject = Quake Floor/Effects
  var Inactive: gameObject = Quake Floor/Inactive
  var Pt Break Rocks: gameObject = <null>
}
