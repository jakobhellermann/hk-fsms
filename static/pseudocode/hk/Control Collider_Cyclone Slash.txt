# Control Collider
# 2 location(s):
#   Knight_Pickup / Knight/Attacks/Cyclone Slash
#   resources.assets / Knight:1/Attacks/Cyclone Slash

fsm Control Collider {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Hits", storeResult=var "Hits")
    FindChild(gameObject=var "Hits", childName="Hit L", storeResult=var "Hit L")
    FindChild(gameObject=var "Hits", childName="Hit R", storeResult=var "Hit R")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SetFsmFloat(gameObject=var "Hit L", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Hit R", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1, everyFrame=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero X scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Neg Scale", vector3Value=var "Init Scale", everyFrame=false)
    GetVector3XYZ(vector3Variable=var "Neg Scale", storeX=var "X Scale", storeY=(unset), storeZ=(unset), everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Neg Scale", vector3Value=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false)
    on FINISHED → Check Scale
  }

  state Activate {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Cyclone Effect", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on CYCLONE END → Deactivate Hit
  }

  state Deactivate Hit {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Cyclone Effect End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Deactivate Self
  }

  state Check Scale {
    FloatCompare(float1=var "Hero X scale", float2=0, tolerance=0, equal=(none), lessThan=→"NEG", greaterThan=→"POS", everyFrame=false)
    on NEG → Neg
    on POS → Pos
  }

  state Neg {
    SetScale(gameObject=var "Hits", vector=var "Neg Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Activate
  }

  state Pos {
    SetScale(gameObject=var "Hits", vector=var "Init Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Activate
  }

  state Deactivate Self {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Hero X scale: float = 0
  var X Scale: float = 0
  var Init Scale: vector3 = (1, 1, 1)
  var Neg Scale: vector3 = (0, 0, 0)
  var Hit L: gameObject = <null>
  var Hit R: gameObject = <null>
  var Hits: gameObject = <null>
  var Self: gameObject = <null>
}
