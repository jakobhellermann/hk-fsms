fsm Detect Hero {
  start trigger

  state trigger {
    GetParent(gameObject=Self, storeResult=var "Parent")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Hit
  }

  state Hit {
    NextFrameEvent(sendEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="DETECT", delay=0, everyFrame=false)
    on FINISHED → trigger
  }

  var Parent: gameObject = <null>
}
