fsm Follow Hero {
  start State 1

  state State 1 {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 2
  }

  state State 2 {
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    Vector3AddXYZ(vector3Variable=var "Hero Pos", addX=0, addY=0, addZ=-0.001, everyFrame=true, perSecond=false)
    SetPosition(gameObject=Self, vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
  }

  var Hero Pos: vector3 = (0, 0, 0)
}
