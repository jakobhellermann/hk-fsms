fsm Bell Control {
  start Init

  state Detect {
    ReceivedDamage(collideTag="", sendEvent=→"NAIL HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on NAIL HIT → Get Parameters
  }

  state Ring R {
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="STAG BELL RUNG", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bell_ring R")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Detect
  }

  state Get Parameters {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    PlayVibrationV2(vibrationDataAsset=bell_hit (sharedassets9.assets), motors=All, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bell_hit (sharedassets9.assets), pitchMin=0.85, pitchMax=1.15, volume=0.5, delay=0, storePlayer=<null>)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    FloatCompare(float1=var "Attack Direction", float2=0, tolerance=0, equal=→"RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Attack Direction", float2=180, tolerance=0, equal=→"LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendRandomEvent(events=[→"LEFT", →"RIGHT"], weights=[1, 1], delay=0)
    on RIGHT → Ring R
    on LEFT → Ring L
    on CANCEL → Detect
  }

  state Ring L {
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="STAG BELL RUNG", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bell_ring R")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  var Attack Direction: float = 0
  var Damage: int = 0
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
