# Control
# 1 location(s):
#   sharedassets161.assets / Shot Slug Spear

fsm Control {
  start Pause

  state Fire {
    SetVelocityAsAngle(gameObject=Self, angle=var "Shot Angle", speed=25, everyFrame=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dissipate
  }

  state Recycle {
    RecycleSelf()
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Idle Pt", emit=0)
    on FINISHED → Poke Out
  }

  state Poke Out {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Shot Angle", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Shot Angle", add=90, everyFrame=false, perSecond=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Shot Angle", speed=30, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.88)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fire
  }

  state Dissipate {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Idle Pt")
    on FINISHED → Recycle
  }

  var Shot Angle: float = 0
  var Idle Pt: gameObject = Shot Slug Spear/Idle Pt
}
