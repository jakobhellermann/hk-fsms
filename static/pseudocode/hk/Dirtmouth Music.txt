# Dirtmouth Music
# 2 location(s):
#   Dream_Nailcollection / Music Fadeout
#   Crossroads_10 / Music Fadeout

fsm Dirtmouth Music {
  start Pause Frame

  state Init {
    FindChild(gameObject=var "GameManager", childName="Music", storeResult=var "Music Folder")
    FindChild(gameObject=var "Music Folder", childName="Track 1", storeResult=var "Track 1")
    on FINISHED → Idle
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Fade Out
  }

  state Fade Out {
    SetFsmFloat(gameObject=var "Track 1", fsmName="Music Player", variableName="Fade Vol End", setValue=0, everyFrame=false)
    SetFsmInt(gameObject=var "Track 1", fsmName="Music Selection", variableName="Current Track", setValue=0, everyFrame=false)
    SetFsmFloat(gameObject=var "Track 1", fsmName="Music Player", variableName="Fade Time", setValue=var "Fadeout Time", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Track 1"), sendEvent="FADE", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → 
  }

  var Fadeout Time: float = 5

  var Music Folder: gameObject = <null>
  var Track 1: gameObject = <null>
}
