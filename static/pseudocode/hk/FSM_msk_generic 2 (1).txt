# FSM
# 2 location(s):
#   Crossroads_18 / Breakable Wall Waterways/Masks/msk_generic 2 (1)
#   Crossroads_18 / Breakable Wall Waterways/Masks/msk_generic 2

// uses template: unmasker
fsm FSM {
  start Pause

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on UNCOVER → Fade
    on STAY → Idle Stay
    on FINISHED → Idle Stay
    on ACTIVATE → Activate
  }

  state Fade {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.5, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Play Sound", isTrue=→"SOUND", isFalse=(none), everyFrame=false)
    on SOUND → Sound
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Idle
  }

  state Idle Stay {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="Player", collideLayer="", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Fade
  }

  state Sound {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=secret_discovered_temp)  // disabled
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (resources.assets), spawnPoint=var "Hero", audioClip=secret_discovered_temp, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  state Activate {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.1, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
  }

  var Play Sound: bool = false

  var Activated: bool = false
}
