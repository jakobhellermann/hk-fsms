# FSM
# 1 location(s):
#   Cliffs_05 / Ghost Activator

fsm FSM {
  start Pause

  state Pause {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    PreBuildTK2DSprites(gameObject=var "Self", useChildren=true)
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activate
    on SHINY PICKED UP → Ready
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  state Ready {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Do
  }

  state Do {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  var Activated: bool = false
  var Self: gameObject = <null>
}
