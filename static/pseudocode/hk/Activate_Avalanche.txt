# Activate
# 1 location(s):
#   Room_Wyrm / Avalanche

fsm Activate {
  start Pause

  state Idle {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasKingsBrand", isTrue=→"SHINY PICKED UP", isFalse=(none))
    on SHINY PICKED UP → Activate
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHINY PICKED UP", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
}
