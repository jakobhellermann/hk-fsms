# FSM
# 1 location(s):
#   GG_Atrium / gg_atrium_hidden_path

fsm FSM {
  start Idle

  state Idle {
    on OPEN → Roar
    on ACTIVATE → Activated
  }

  state Activated {
    ActivateGameObject(gameObject=var "Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Smash {
    ActivateGameObject(gameObject=gg_atrium_hidden_path/gg_roof open_effect, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=gg_atrium_hidden_path/CameraLockArea, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=6.1333, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  state Activate {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar End
  }

  state Return {
    ActivateGameObject(gameObject=gg_atrium_hidden_path/CameraLockArea, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    on FINISHED → Return
  }

  state Roar {
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    on FINISHED → Smash
  }

  var Closed: gameObject = gg_atrium_hidden_path/top_leaves_closed
  var Open: gameObject = gg_atrium_hidden_path/top_leaves_path_glow
  var Self: gameObject = <null>
}
