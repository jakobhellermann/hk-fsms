# Unlock Anim
# 1 location(s):
#   Room_Colosseum_01 / Gold Trial Board

fsm Unlock Anim {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite")
    FindChild(gameObject=Self, childName="Burst", storeResult=var "Burst")
    on FINISHED → Idle
  }

  state Idle {
    on UNLOCK GOLD → Burst
  }

  state Burst {
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetSpriteRendererSprite(gameObject=var "Sprite", sprite=colosseum_sign_0002_3 (sharedassets30.assets))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Sprite", audioClip=chain_cut (sharedassets30.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Sprite", audioClip=grass_cut_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  var Burst: gameObject = <null>
  var Sprite: gameObject = <null>
}
