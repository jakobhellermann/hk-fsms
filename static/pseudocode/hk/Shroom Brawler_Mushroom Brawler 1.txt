# Shroom Brawler
# 1 location(s):
#   Fungus2_05 / Mushroom Brawler 1

fsm Shroom Brawler {
  start Init

  state Init {
    SetFloatValue(floatVariable=var "Roam Distance", floatValue=2, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Min X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Max X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatSubtract(floatVariable=var "Min X", subtract=var "Roam Distance", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Max X", add=var "Roam Distance", everyFrame=false, perSecond=false)
    on FINISHED → Facing Right?
  }

  state Sleep {
    ActivateGameObject(gameObject=var "Cap Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on WAKE → Wake
  }

  state Wake {
    SetCollider(gameObject=Self, active=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_brawler_awaken (sharedassets169.assets))
    ActivateGameObject(gameObject=var "Cap Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=4)
    on FINISHED → Change Movement
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Move Speed", y=(unset), everyFrame=false)
    Wait(time=var "Move Time", finishEvent=→"CHANGE DIR", realTime=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Max X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"LEFT", everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Min X", tolerance=0, equal=(none), lessThan=→"RIGHT", greaterThan=(none), everyFrame=true)
    FloatTestToBool(float1=var "Hero X", float2=var "Self X", tolerance=0, equalBool=(unset), lessThanBool=var "Hero is Left", greaterThanBool=(unset), everyFrame=true)
    BoolAllTrue(boolVariables=[var "Facing Right", var "Hero is Left"], sendEvent=→"TURN L", storeResult=(unset), everyFrame=true)
    BoolNoneTrue(boolVariables=[var "Facing Right", var "Hero is Left"], sendEvent=→"TURN R", storeResult=(unset), everyFrame=true)
    FloatAdd(floatVariable=var "Wait Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Wait Timer", float2=var "Wait Time", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SPIT", everyFrame=true)
    on CHANGE DIR → Change Movement
    on LEFT → Return Left
    on RIGHT → Return Right
    on TURN L → Turn L
    on TURN R → Turn R
    on SPIT → Spit Antic
    on BLOCKED HIT → Bounce
  }

  state Change Movement {
    RandomFloat(min=1, max=2, storeResult=var "Move Time")
    RandomFloat(min=1, max=3, storeResult=var "Move Speed")
    SendRandomEvent(events=[→"LEFT", →"RIGHT"], weights=[1, 1], delay=0)
    FloatMultiply(floatVariable=var "Move Speed", multiplyBy=-1, everyFrame=false)
    on RIGHT → Idle
    on FINISHED → Set Wait Time
  }

  state Facing Right? {
    SetFloatValue(floatVariable=var "Smash Speed", floatValue=-0.85, everyFrame=false)
    SetFloatValue(floatVariable=var "SpitSpeedMax", floatValue=-2.5, everyFrame=false)
    SetFloatValue(floatVariable=var "SpitSpeedMin", floatValue=-13, everyFrame=false)
    BoolTest(boolVariable=var "Facing Right", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "SpitSpeedMin", floatValue=2.5, everyFrame=false)
    SetFloatValue(floatVariable=var "SpitSpeedMax", floatValue=13, everyFrame=false)
    SetFloatValue(floatVariable=var "Smash Speed", floatValue=0.85, everyFrame=false)
    on FINISHED → Sleep
  }

  state Return Left {
    RandomFloat(min=1, max=2, storeResult=var "Move Time")
    RandomFloat(min=-1, max=-3, storeResult=var "Move Speed")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Return Right {
    RandomFloat(min=1, max=2, storeResult=var "Move Time")
    RandomFloat(min=1, max=3, storeResult=var "Move Speed")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Turn L {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    FloatAdd(floatVariable=var "Wait Timer", add=1, everyFrame=true, perSecond=true)
    SetFloatValue(floatVariable=var "SpitSpeedMax", floatValue=-2.5, everyFrame=false)
    SetFloatValue(floatVariable=var "SpitSpeedMin", floatValue=-13, everyFrame=false)
    SetFloatValue(floatVariable=var "Smash Speed", floatValue=-0.85, everyFrame=false)
    on FINISHED → Idle
    on BLOCKED HIT → Bounce
  }

  state Turn R {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    FloatAdd(floatVariable=var "Wait Timer", add=1, everyFrame=true, perSecond=true)
    SetFloatValue(floatVariable=var "SpitSpeedMin", floatValue=2.5, everyFrame=false)
    SetFloatValue(floatVariable=var "SpitSpeedMax", floatValue=13, everyFrame=false)
    SetFloatValue(floatVariable=var "Smash Speed", floatValue=0.85, everyFrame=false)
    on FINISHED → Idle
    on BLOCKED HIT → Bounce
  }

  state Set Wait Time {
    RandomFloat(min=1.5, max=1.75, storeResult=var "Wait Time")
    SetFloatValue(floatVariable=var "Wait Timer", floatValue=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Spit Antic {
    RandomFloat(min=1.25, max=1.5, storeResult=var "Wait Time")
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=25, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    BoolTest(boolVariable=var "Smash Range", isTrue=→"SMASH", isFalse=(none), everyFrame=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=(unset))
    SetCollider(gameObject=var "Cap Hit", active=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetFloatValue(floatVariable=var "Wait Timer", floatValue=0, everyFrame=false)
    on FINISHED → Spit
    on SMASH → Smash Antic
    on CANCEL → Set Wait Time
  }

  state Spit {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_brawler_spit (sharedassets169.assets))
    SendEventByName(eventTarget=GameObject(var "Spit Effect"), sendEvent="PLAY", delay=0, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=(unset))
    SetCollider(gameObject=var "Cap Hit", active=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Hero X", float2=var "Self X", operation=Subtract, storeResult=var "Target X", everyFrame=false)
    FloatMultiply(floatVariable=var "Target X", multiplyBy=0.8, everyFrame=false)
    FloatClamp(floatVariable=var "Target X", minValue=var "SpitSpeedMin", maxValue=var "SpitSpeedMax", everyFrame=false)
    SetFloatValue(floatVariable=var "SpitRangeMin", floatValue=var "Target X", everyFrame=false)
    SetFloatValue(floatVariable=var "SpitRangeMax", floatValue=var "Target X", everyFrame=false)
    FloatAdd(floatVariable=var "SpitRangeMin", add=-1.5, everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "SpitRangeMax", add=1.5, everyFrame=false, perSecond=false)
    FlingObjectsFromGlobalPoolVel(gameObject=Vomit Glob NoiseFix (sharedassets34.assets), spawnPoint=var "Spit Point", position=(0, 0, -0.004), spawnMin=2, spawnMax=4, speedMinX=var "SpitRangeMin", speedMaxX=var "SpitRangeMax", speedMinY=23, speedMaxY=26, originVariationX=0, originVariationY=0)
    on FINISHED → Spit CD
    on BLOCKED HIT → Bounce
  }

  state Spit CD {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on BLOCKED HIT → Bounce
  }

  state Bounce {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Smash Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_brawler_awaken (sharedassets169.assets))
    on FINISHED → Smashing
  }

  state Smashing {
    ActivateGameObject(gameObject=var "Smash Hits", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Smash Hits", animLibName="", clipName="Smash Loop")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Smash Speed", y=(unset), everyFrame=true)
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=2.25, finishEvent=→"FINISHED", realTime=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=(unset))
    SetCollider(gameObject=var "Cap Hit", active=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatTestToBool(float1=var "Hero X", float2=var "Self X", tolerance=0, equalBool=(unset), lessThanBool=var "Hero is Left", greaterThanBool=(unset), everyFrame=true)
    BoolAllTrue(boolVariables=[var "Facing Right", var "Hero is Left"], sendEvent=→"FINISHED", storeResult=(unset), everyFrame=true)
    BoolNoneTrue(boolVariables=[var "Facing Right", var "Hero is Left"], sendEvent=→"FINISHED", storeResult=(unset), everyFrame=true)
    on FINISHED → Smash Recover
  }

  state Smash Recover {
    ActivateGameObject(gameObject=var "Smash Hits", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=(unset))
    SetCollider(gameObject=var "Cap Hit", active=true)
    Tk2dPlayAnimation(gameObject=var "Smash Hits", animLibName="", clipName="Run")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Smash to Run", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Facing Right: bool = false

  var Distance: float = 0
  var Hero X: float = 0
  var Max X: float = 0
  var Min X: float = 0
  var Move Speed: float = 0
  var Move Time: float = 0
  var Roam Distance: float = 0
  var Self X: float = 0
  var Smash Speed: float = 0
  var SpitRangeMax: float = 0
  var SpitRangeMin: float = 0
  var SpitSpeedMax: float = 0
  var SpitSpeedMin: float = 0
  var Target X: float = 0
  var Wait Time: float = 0
  var Wait Timer: float = 0
  var Hero is Left: bool = false
  var Smash Range: bool = false
  var Cap Hit: gameObject = Mushroom Brawler 1/Cap Hit
  var Smash Hits: gameObject = Mushroom Brawler 1/Smash Hits
  var Spit Effect: gameObject = Mushroom Brawler 1/Spit Effect
  var Spit Point: gameObject = Mushroom Brawler 1/Spit Point
}
