fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Floor Antic", storeResult=var "Floor Antic")
    on FINISHED → Downed
  }

  state Downed {
    SetPolygonCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Floor Antic", active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on UP → Floor Antic
  }

  state Floor Antic {
    SetMeshRenderer(gameObject=var "Floor Antic", active=true)
    Tk2dPlayAnimation(gameObject=var "Floor Antic", animLibName="", clipName="Spike Floor Antic")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spike Up
    on SPIKES DOWN → Down
    on DOWN → Down
  }

  state Spike Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spike Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=var "Floor Antic", animLibName="", clipName="Spike Floor Idle")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Up
    on SPIKES DOWN → Down
    on DOWN → Down
  }

  state Up {
    SetPolygonCollider(gameObject=Self, active=true)
    on DOWN → Down
    on SPIKES DOWN → Down
  }

  state Down {
    SetPolygonCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Floor Antic", clipName="Spike Floor End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spike Down")
    on FINISHED → Downed
  }

  var Floor Antic: gameObject = <null>
}
