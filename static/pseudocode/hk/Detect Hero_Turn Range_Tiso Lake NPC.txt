# Detect Hero
# 1 location(s):
#   Crossroads_50 / Tiso Lake NPC/Turn Range

fsm Detect Hero {
  start Init

  state Out {
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="LookLeft")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="LookRight")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetParent(gameObject=var "Parent", storeResult=var "Elderbug")
    FindChild(gameObject=var "Elderbug", childName="Lit", storeResult=var "Lit")
    FindChild(gameObject=var "Elderbug", childName="Light", storeResult=var "Light")
    FindChild(gameObject=var "Elderbug", childName="Particle B", storeResult=var "Particle B")
    FindChild(gameObject=var "Elderbug", childName="Particle F", storeResult=var "Particle F")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
    on EXIT → Out
  }

  var Elderbug: gameObject = <null>
  var Light: gameObject = <null>
  var Lit: gameObject = <null>
  var Parent: gameObject = <null>
  var Particle B: gameObject = <null>
  var Particle F: gameObject = <null>
}
