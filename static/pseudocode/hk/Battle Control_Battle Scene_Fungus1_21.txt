# Battle Control
# 1 location(s):
#   Fungus1_21 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Moss Knight", storeResult=var "Moss Knight")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"COMPLETE", lessThan=→"COMPLETE", greaterThan=(none), everyFrame=true)
    SetBattleScene(target=var "Moss Knight", battleScene=var "Self")
    on ACTIVATE → Activate
    on COMPLETE → Complete
  }

  state Activate {
    ActivateGameObject(gameObject=var "Moss Knight", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
  }

  state Complete {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=1.5, everyFrame=false)
  }

  var Battle Enemies: int = 1
  var Activated: bool = false
  var Moss Knight: gameObject = <null>
  var Self: gameObject = <null>
}
