# UI Journal
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Journal
#   resources.assets / _GameCameras/HudCamera/Inventory/Journal

fsm UI Journal {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Enemy List", storeResult=var "Enemy List")
    FindChild(gameObject=Self, childName="Text Completion", storeResult=var "Completion Text")
    FindChild(gameObject=var "Completion Text", childName="Amount", storeResult=var "Amount Obj")
    FindChild(gameObject=var "Completion Text", childName="Text Encountered", storeResult=var "Encountered Text")
    FindChild(gameObject=var "Completion Text", childName="Total", storeResult=var "Total Obj")
    FindChild(gameObject=var "Encountered Text", childName="Total", storeResult=var "Enc Total")
    FindChild(gameObject=var "Encountered Text", childName="Amount", storeResult=var "Encountered Obj")
    on FINISHED → Inactive
  }

  state Inactive {
    SendEventByName(eventTarget=GameObject(var "Enemy List"), sendEvent="UI INACTIVE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemy List"), sendEvent="RESET", delay=0, everyFrame=false)
    on ACTIVATE → Completion?
  }

  state Active {
    SendEventByName(eventTarget=GameObject(var "Enemy List"), sendEvent="UI ACTIVE", delay=0, everyFrame=false)
    on PANE RESET → Inactive
  }

  state Completion? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="unlockedCompletionRate", isTrue=(none), isFalse=→"FINISHED")
    ActivateGameObject(gameObject=var "Completion Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="journalNotesCompleted", storeValue=var "Amount Int")
    ConvertIntToString(intVariable=var "Amount Int", stringVariable=var "Amount Str", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Amount Obj", textString=var "Amount Str")
    GetPlayerDataInt(gameObject=var "GameManager", intName="journalEntriesCompleted", storeValue=var "Encountered Int")
    ConvertIntToString(intVariable=var "Encountered Int", stringVariable=var "Encountered Str", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Encountered Obj", textString=var "Encountered Str")
    GetPlayerDataInt(gameObject=var "GameManager", intName="journalEntriesTotal", storeValue=var "Total Int")
    ConvertIntToString(intVariable=var "Total Int", stringVariable=var "Total Str", format="", everyFrame=false)
    BuildString(stringParts=["/ ", var "Total Str"], separator="", addToEnd=true, storeResult=var "Total Str", everyFrame=false)
    SetTextMeshProText(gameObject=var "Total Obj", textString=var "Total Str")
    SetTextMeshProText(gameObject=var "Enc Total", textString=var "Total Str")
    SendEventByName(eventTarget=GameObject(var "Completion Text"), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Active
  }

  var Amount Int: int = 0
  var Encountered Int: int = 0
  var Total Int: int = 0
  var Amount Str: string = ""
  var Encountered Str: string = ""
  var Total Str: string = ""
  var Amount Obj: gameObject = <null>
  var Completion Text: gameObject = <null>
  var Enc Total: gameObject = <null>
  var Encountered Obj: gameObject = <null>
  var Encountered Text: gameObject = <null>
  var Enemy List: gameObject = <null>
  var Total Obj: gameObject = <null>
}
