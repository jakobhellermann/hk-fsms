fsm Break Pole {
  start Initiate

  state Idle {
    ReceivedDamage(collideTag="", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on HIT → Check If Nail
  }

  state Check If Nail {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0], sendEvent=[→"TRUE"], everyFrame=false)
    on TRUE → Check Direction
    on FINISHED → Idle
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "z", space=0, everyFrame=false)
    FloatCompare(float1=var "z", float2=0, tolerance=1, equal=(none), lessThan=→"INERT", greaterThan=→"INERT", everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
    on INERT → Inert
  }

  state Check Direction {
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on RIGHT → Hit Right
    on UP → Pause Frame 2
    on LEFT → Hit Left
    on DOWN → Pause Frame 2
  }

  state Hit Right {
    CreateObject(gameObject=Slash Impact (sharedassets39.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=0.6, y=0.6, z=(unset), everyFrame=false, lateUpdate=false)
    CreateObject(gameObject=Breakable Pole Top (sharedassets41.assets), spawnPoint=var "Self", position=(0, 0.8, 0), rotation=(unset), storeObject=var "Top", networkInstantiate=false, networkGroup=0)
    SetPosition(gameObject=var "Top", vector=(unset), x=(unset), y=(unset), z=var "z", space=1, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Top", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=50, max=70, storeResult=var "Chooser")
    SetVelocityAsAngle(gameObject=var "Top", angle=var "Chooser", speed=25, everyFrame=false)
    on FINISHED → Destroy
  }

  state Hit Left {
    CreateObject(gameObject=Slash Impact (sharedassets39.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=-0.6, y=0.6, z=(unset), everyFrame=false, lateUpdate=false)
    CreateObject(gameObject=Breakable Pole Top (sharedassets41.assets), spawnPoint=var "Self", position=(0, 0.8, 0), rotation=(unset), storeObject=var "Top", networkInstantiate=false, networkGroup=0)
    SetPosition(gameObject=var "Top", vector=(unset), x=(unset), y=(unset), z=var "z", space=1, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Top", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=110, max=130, storeResult=var "Chooser")
    SetVelocityAsAngle(gameObject=var "Top", angle=var "Chooser", speed=25, everyFrame=false)
    on FINISHED → Destroy
  }

  state Inert {
  }

  state Pause Frame 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Destroy {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    CreateObject(gameObject=Breakable Pole Base (sharedassets41.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Base", networkInstantiate=false, networkGroup=0)
    SetPosition(gameObject=var "Base", vector=(unset), x=(unset), y=(unset), z=var "z", space=1, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Base", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    DestroySelf(detachChildren=false)
  }

  var Final Payout: int = 5
  var Geo Per Hit: int = 2
  var Hits: int = 5

  var Attack Direction: float = 0
  var Chooser: float = 0
  var Recoil Speed: float = 1
  var Recoil Speed Neg: float = 0
  var Rotation: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Waiter: float = 0
  var z: float = 0
  var Attack Type: int = 0
  var Base: gameObject = <null>
  var Camera: gameObject = <null>
  var Damager: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Self: gameObject = <null>
  var Top: gameObject = <null>
}
