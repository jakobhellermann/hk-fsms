# Splash Check
# 1 location(s):
#   Crossroads_30 / Spa Bugs NPC

fsm Splash Check {
  start Idle
  on SPLASH WATER → Splash Check  // from any state

  state Idle {
  }

  state Splash Check {
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero Pos", float2=20.93, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    FloatCompare(float1=var "Hero Scale", float2=1, tolerance=0, equal=→"CANCEL", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Splash
    on CANCEL → Idle
  }

  state Splash {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPLASH", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Hero Pos: float = 0
  var Hero Scale: float = 0
}
