# Spawn Orbit Shield
# 1 location(s):
#   resources.assets / Knight:1/Charm Effects

fsm Spawn Orbit Shield {
  start Init
  on LEVEL LOADED → Init  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Charms Allowed?
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_38", isTrue=→"SPAWN", isFalse=→"FINISHED")
    on SPAWN → Spawn
    on FINISHED → Idle
  }

  state Idle {
    on CHARM EQUIP CHECK → Check
    on SLASH → Send Slash Event
  }

  state Spawn {
    FindGameObject(objectName="", withTag="Orbit Shield", store=var "Shield")
    GameObjectIsNull(gameObject=var "Shield", isNull=(none), isNotNull=→"CANCEL", storeResult=(unset), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Orbit Shield, spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Shield")
    on CANCEL → Idle
    on FINISHED → Idle
  }

  state Send Slash Event {
    SendEventByName(eventTarget=GameObject(var "Shield"), sendEvent="SLASH", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Charms Allowed? {
    GetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", storeValue=var "No Charms", everyFrame=false)
    BoolTest(boolVariable=var "No Charms", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="IsGameplayScene", parameters=[], storeResult=var "Is Gameplay Scene")
    BoolTest(boolVariable=var "Is Gameplay Scene", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Check
    on CANCEL → Idle
  }

  var Is Gameplay Scene: bool = false
  var No Charms: bool = false
  var Shield: gameObject = <null>
}
