# Conversation Control
# 1 location(s):
#   Waterways_05 / Dung Defender NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on BOX DOWN → Box Down  // from any state
  on TALK FINISH → Hero Look Back  // from any state
  on TALK FINISH NO HERO ANIM → Check Direction 2  // from any state

  state Idle {
    on CONVO START → Check Direction
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Active
  }

  state End {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → Hero Look Back
  }

  state Check Direction {
    AudioStop(gameObject=Self)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Talk R
    on LEFT → Talk L
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NPC TalkRight")
    on FINISHED → Hero Look
  }

  state Talk L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NPC TalkLeft")
    on FINISHED → Hero Look
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")  // disabled
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Title
  }

  state Check Direction 2 {
    AudioStop(gameObject=var "Voice Player")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Face Right
    on LEFT → Face Left
  }

  state Face Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NPC LookRight")
    on FINISHED → End
  }

  state Face Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NPC LookLeft")
    on FINISHED → End
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DUNG_DEFENDER_REPEAT", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Dung_Def_impressed_04 (sharedassets354.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down Event
  }

  state Hero Look Back {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")
    on FINISHED → Check Direction 2
  }

  state Box Down Event {
    SendEventByName(eventTarget=Self, sendEvent="BOX DOWN", delay=0, everyFrame=false)
  }

  state Check Active {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="dungDefenderEncounterReady", isTrue=(none), isFalse=→"DESTROY")
    on FINISHED → Idle
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Title {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="DUNG_DEFENDER", everyFrame=false)
    on FINISHED → Choice
  }

  state Choice {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="dungDefenderConvo1", isTrue=(none), isFalse=→"CONVO 1")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasAcidArmour", falseBool="dungDefenderIsmaConvo", isTrue=→"ISMA", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_10", falseBool="dungDefenderCharmConvo", isTrue=→"DUNG CHARM", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="dungDefenderConvo2", isTrue=(none), isFalse=→"CONVO 2")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="dungDefenderConvo3", isTrue=(none), isFalse=→"CONVO 3")
    on FINISHED → Repeat
    on CONVO 1 → Convo 1
    on CONVO 2 → Convo 2
    on CONVO 3 → Convo 3
    on DUNG CHARM → Dung Charm
    on ISMA → Isma
  }

  state Convo 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DUNG_DEFENDER_1", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="dungDefenderConvo1", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Dung_Def_impressed_01 (sharedassets354.assets), Dung_Def_impressed_02 (sharedassets354.assets), Dung_Def_impressed_03 (sharedassets354.assets), Dung_Def_impressed_04 (sharedassets354.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down Event
  }

  state Convo 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DUNG_DEFENDER_2", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="dungDefenderConvo2", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Dung_Def_impressed_01 (sharedassets354.assets), Dung_Def_impressed_02 (sharedassets354.assets), Dung_Def_impressed_03 (sharedassets354.assets), Dung_Def_impressed_04 (sharedassets354.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down Event
  }

  state Convo 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DUNG_DEFENDER_3", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="dungDefenderConvo3", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Dung_Def_impressed_01 (sharedassets354.assets), Dung_Def_impressed_02 (sharedassets354.assets), Dung_Def_impressed_03 (sharedassets354.assets), Dung_Def_impressed_04 (sharedassets354.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down Event
  }

  state Dung Charm {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DUNG_DEFENDER_DUNG_CHARM", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="dungDefenderCharmConvo", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Dung_Def_impressed_01 (sharedassets354.assets), Dung_Def_impressed_02 (sharedassets354.assets), Dung_Def_impressed_03 (sharedassets354.assets), Dung_Def_impressed_04 (sharedassets354.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down Event
  }

  state Isma {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DUNG_DEFENDER_ISMA", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="dungDefenderIsmaConvo", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Dung_Def_impressed_01 (sharedassets354.assets), Dung_Def_impressed_02 (sharedassets354.assets), Dung_Def_impressed_03 (sharedassets354.assets), Dung_Def_impressed_04 (sharedassets354.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down Event
  }

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var History: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var mapCrossroads: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
