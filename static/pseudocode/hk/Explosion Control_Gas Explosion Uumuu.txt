# Explosion Control
# 1 location(s):
#   sharedassets156.assets / Gas Explosion Uumuu

fsm Explosion Control {
  start Start

  state Explode {
    RandomFloat(min=0.85, max=1.1, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Death Wave Infected R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=3, y=3, z=(unset), everyFrame=false, lateUpdate=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Hero Damage", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Collider End
  }

  state Collider End {
    SetCircleCollider(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Hero Damage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Start {
    SetCircleCollider(gameObject=Self, active=true)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    FindChild(gameObject=Self, childName="Hero Damage", storeResult=var "Hero Damage")
    on FINISHED → Explode
  }

  state Recycle {
    RecycleSelf()
  }

  var Pitch: float = 0
  var Hero Damage: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Self: gameObject = Gas Explosion Uumuu
}
