# Battle Control

fsm Battle Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="CamLock Battle", storeResult=var "CamLock Battle")
    FindChild(gameObject=Self, childName="CamLock Intro", storeResult=var "CamLock Intro")
    FindChild(gameObject=Self, childName="CamLock Roar", storeResult=var "CamLock Roar")
    FindChild(gameObject=Self, childName="Entry Plat", storeResult=var "Entry Plat")
    FindChild(gameObject=Self, childName="Hornet Nosk", storeResult=var "Hornet Nosk")
    FindChild(gameObject=Self, childName="Nosk Transform", storeResult=var "Nosk Transform")
    FindChild(gameObject=Self, childName="Rumble Loop", storeResult=var "Rumble Loop")
    FindChild(gameObject=Self, childName="Pt Jump", storeResult=var "Pt Jump")
    FindChild(gameObject=Self, childName="Slam Effect", storeResult=var "Slam Effect")
    on FINISHED → Idle
  }

  state Idle {
    on ENTER → Music Type
  }

  state Transform 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Nosk Transform", audioClip=mimic_spider_scream (sharedassets109.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlaySimple(gameObject=var "Nosk Transform", volume=1, oneShotClip=mimic_spider_transormation_stage_1 (sharedassets290.assets))
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Nosk Transform", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Nosk Transform", position=(-0.2, -1.2, 0), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=var "Rumble Loop", volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=var "Nosk Transform", animLibName="", clipName="Transform 1")
    on FINISHED → Transform 2
  }

  state Music Type {
    GGCheckIfBossSequence(trueEvent=→"PANTHEON", falseEvent=→"STATUE")
    on PANTHEON → GG Music
    on STATUE → Orig Music
    on FINISHED → GG Music
  }

  state GG Music {
    ApplyMusicCue(musicCue=GG Sad (sharedassets425.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=HK Decline 5 (resources.assets), transitionTime=0.5)
    on FINISHED → Arena
  }

  state Orig Music {
    ApplyMusicCue(musicCue=MimicSpider (sharedassets290.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0.5)
    on FINISHED → Arena
  }

  state Transform 2 {
    Tk2dPlayAnimation(gameObject=var "Nosk Transform", animLibName="", clipName="Transform 2")
    AudioPlaySimple(gameObject=var "Nosk Transform", volume=1, oneShotClip=mimic_spider_transormation_stage_2 (sharedassets81.assets))
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Transform 3
  }

  state Arena {
    ActivateGameObject(gameObject=var "CamLock Intro", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Entry Plat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Roar", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Transform 1
  }

  state Transform 3 {
    Tk2dPlayAnimation(gameObject=var "Nosk Transform", animLibName="", clipName="Transform 3")
    AudioPlaySimple(gameObject=var "Nosk Transform", volume=1, oneShotClip=mimic_spider_transormation_stage_4 (sharedassets290.assets))
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Transform 4
  }

  state Transform 4 {
    Tk2dPlayAnimationWithEvents(gameObject=var "Nosk Transform", clipName="Transform 4", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=var "Nosk Transform", volume=1, oneShotClip=mimic_spider_transormation_stage_5 (sharedassets290.assets))  // disabled
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Nosk Transform", audioClip=mimic_spider_scream (sharedassets109.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Nosk Transform", audioClip=mimic_spider_transormation_stage_5 (sharedassets290.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=var "Rumble Loop")
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Battle Start
  }

  state Battle Start {
    ActivateGameObject(gameObject=var "CamLock Roar", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Nosk Transform", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Battle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hornet Nosk"), sendEvent="START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MIMIC_SPIDER", everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Jump", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Nosk Transform", audioClip=false_knight_jump (sharedassets48.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  var Area Title: gameObject = <null>
  var CamLock Battle: gameObject = <null>
  var CamLock Intro: gameObject = <null>
  var CamLock Roar: gameObject = <null>
  var Entry Plat: gameObject = <null>
  var Hornet Nosk: gameObject = <null>
  var Nosk Transform: gameObject = <null>
  var Pt Jump: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Rumble Loop: gameObject = Mimic Spider/Rumble Loop (sharedassets486.assets)
  var Slam Effect: gameObject = <null>
}
