# Shake Projectiles U
# 1 location(s):
#   Dream_03_Infected_Knight / Infected Knight Dream

fsm Shake Projectiles U {
  start Idle
  on SHAKE END → Idle  // from any state

  state Idle {
    GetOwner(storeGameObject=var "Self")
    on SHAKE START → Spawn
  }

  state Spawn {
    SpawnObjectFromGlobalPool(gameObject=IK Projectile SH (sharedassets343.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Projectile")
    RandomFloat(min=6, max=25, storeResult=var "X Velocity")  // disabled
    SetFloatValue(floatVariable=var "X Velocity", floatValue=1, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=var "Multiplier", everyFrame=false)
    SetVelocity2d(gameObject=var "Projectile", vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    on FINISHED → Pause
    on REDO → 
  }

  state Pause {
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    FloatMultiply(floatVariable=var "Multiplier", multiplyBy=-1, everyFrame=false)
    on FINISHED → Spawn
  }

  var Multiplier: float = 1
  var Prev Speed: float = 0
  var X Velocity: float = 0
  var Adjacent 1: bool = false
  var Adjacent 2: bool = false
  var Prev Adjacent: bool = false
  var Was Adjacent: bool = false
  var Projectile: gameObject = <null>
  var Self: gameObject = <null>
}
