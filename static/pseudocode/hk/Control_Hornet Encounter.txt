# Control
# 1 location(s):
#   Fungus2_21 / Hornet Encounter

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hornetCityBridge_ready", falseBool="hornetCityBridge_completed", isTrue=→"PLAY", isFalse=(none))
    on PLAY → Idle
  }

  state Idle {
    SetPlayerDataBool(boolName="hornetCityBridge_completed", value=true)
    FindChild(gameObject=Self, childName="Hornet Idle", storeResult=var "Hornet Idle")
    FindChild(gameObject=Self, childName="Hornet Exit Anim", storeResult=var "Hornet Exit Anim")
    ActivateGameObject(gameObject=var "Hornet Idle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Exit
  }

  state Exit {
    ActivateGameObject(gameObject=var "Hornet Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hornet Exit Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Hornet Exit Anim: gameObject = <null>
  var Hornet Idle: gameObject = <null>
}
