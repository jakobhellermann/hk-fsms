# Javelin Control
# 1 location(s):
#   sharedassets92.assets / Sentry Javelin

fsm Javelin Control {
  start Init
  on HIT TERRAIN → Stick  // from any state
  on ORBIT SHIELD → Break  // from any state

  state Init {
    SetScale(gameObject=Self, vector=(unset), x=0.8, y=0.8, z=1, everyFrame=false, lateUpdate=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Javelin")
    ActivateGameObject(gameObject=var "Snapper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"SNAP", storeCollider=var "Collider")
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    on HIT TERRAIN → Stick
    on SNAP → Pause
    on ORBIT SHIELD → Break
  }

  state Stick {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_flying_javelin_sentry_projectile_impact)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Javelin Stick")
    ActivateGameObject(gameObject=var "Thunk", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=15, finishEvent=→"WAIT", realTime=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Snapper", active=false)
    on WAIT → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    ActivateGameObject(gameObject=var "Snapper", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Snapper", active=true)
    Tk2dPlayFrame(gameObject=var "Snapper", frame=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    Tk2dWatchAnimationEvents(gameObject=var "Snapper", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Break Audio {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_flying_javelin_sentry_projectile_impact)
    on FINISHED → Break
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Break Audio
  }

  var Collider: gameObject = <null>
  var Snapper: gameObject = Sentry Javelin/Snapper
  var Thunk: gameObject = Sentry Javelin/Thunk Effect
}
