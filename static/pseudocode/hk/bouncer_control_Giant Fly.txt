# bouncer_control
# 3 location(s):
#   Crossroads_04 / _Enemies/Giant Fly
#   GG_Gruz_Mother / _Enemies/Giant Fly
#   GG_Gruz_Mother_V / _Enemies/Giant Fly

// uses template: bouncer_control
fsm bouncer_control {
  start Initialise
  on GO DOWN → Go Down  // from any state
  on GO LEFT → Go Left  // from any state
  on GO RIGHT → Go Right  // from any state
  on STOP → Stopped  // from any state
  on GO UP → Go Up  // from any state

  state Initialise {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="MainCamera", store=var "Camera")
    BoolTest(boolVariable=var "Starts Inactive", isTrue=→"STOP", isFalse=(none), everyFrame=false)
    GetDistance(gameObject=var "Self", target=var "Camera", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=44, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Aim
  }

  state Left or Right? {
    FloatSwitch(floatVariable=var "Angle", lessThan=[90, 270, 360], sendEvent=[→"RIGHT", →"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Face Left
    on RIGHT → Face Right
  }

  state Face Left {
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    on FINISHED → Fly 2
  }

  state Face Right {
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    on FINISHED → Fly 2
  }

  state Hit Up {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Up Right
    on LEFT → Up Left
  }

  state Up Right {
    RandomFloat(min=320, max=350, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Up Left {
    RandomFloat(min=190, max=220, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Down {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Down Right
    on LEFT → Up Left 2
  }

  state Down Right {
    RandomFloat(min=10, max=40, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Up Left 2 {
    RandomFloat(min=140, max=170, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Right {
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Down Right 2
    on DOWN → Right Down
  }

  state Right Down {
    RandomFloat(min=190, max=220, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Down Right 2 {
    RandomFloat(min=140, max=170, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Left {
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Left Up
    on DOWN → Left Down
  }

  state Left Down {
    RandomFloat(min=320, max=350, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Left Up {
    RandomFloat(min=10, max=40, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Fly 2 {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    SetVelocityAsAngle(gameObject=var "Self", angle=var "Angle", speed=var "Speed", everyFrame=true)
    Collision2dEvent(collision=1, collideTag="", sendEvent=→"COLLISION STAY 2D", storeCollider=var "", storeForce=(unset))  // disabled
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=25, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=25, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    on BONK DOWN → Hit Down
    on BONK LEFT → Hit Left
    on BONK RIGHT → Hit Right
    on BONK UP → Hit Up
  }

  state Stopped {
    SetBoolValue(boolVariable=var "Starts Inactive", boolValue=false, everyFrame=false)
    on WAKE → Aim
  }

  state Aim {
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    BoolTest(boolVariable=var "Start Up", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle", floatValue=90, everyFrame=false)
    on FINISHED → Left or Right?
  }

  state Go Up {
    RandomFloat(min=20, max=160, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Go Right {
    RandomFloat(min=-70, max=70, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Go Left {
    RandomFloat(min=110, max=250, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Go Down {
    RandomFloat(min=200, max=340, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  var Speed: float = 5
  var Start Up: bool = false
  var Starts Inactive: bool = true
  var Contact Normal: vector3 = (0, 0, 0)

  var Angle: float = 0
  var Distance: float = 0
  var Normal X: float = 0
  var Normal Y: float = 0
  var Raycast Down: float = 0
  var Raycast Left: float = 0
  var Raycast Right: float = 0
  var Raycast X: float = 0
  var Facing Right: bool = false
  var Moving Up: bool = false
  var Bounds: vector2 = (0, 0)
  var Origin: vector2 = (0, 0)
  var Camera: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
