// uses template: color_fader
fsm color_fader {
  start Init
  on DOWN INSTANT → Down Instant  // from any state
  on UP INSTANT → Up Instant  // from any state
  on PULSE DOWN → Pulse Down  // from any state
  on STOP → Stopped  // from any state
  on FADE UP INSTANT → Fade Up Instant  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Start Colour", everyFrame=false)
    SetTextMeshProColor(gameObject=Self, color=var "Start Colour", everyFrame=false)
    BoolTest(boolVariable=var "Auto Up", isTrue=→"UP", isFalse=(none), everyFrame=false)
    on UP → Delay
    on DOWN → Down
  }

  state Up {
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))
    SetSpriteRenderer(gameObject=var "Self", active=true)
    SetColorValue(colorVariable=var "New Colour", color=var "Prev Colour", everyFrame=false)
    EaseColor(fromValue=var "Prev Colour", toValue=var "Up Colour", colorVariable=var "New Colour", time=var "Up Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    SetTextMeshProColor(gameObject=Self, color=var "New Colour", everyFrame=true)
    SetMeshRenderer(gameObject=Self, active=true)
    on DOWN → Down
    on FINISHED → Upped
  }

  state Down {
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))
    SetSpriteRenderer(gameObject=var "Self", active=true)  // disabled
    SetColorValue(colorVariable=var "New Colour", color=var "Prev Colour", everyFrame=false)
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", fail=(none))
    EaseColor(fromValue=var "Prev Colour", toValue=var "Down Colour", colorVariable=var "New Colour", time=var "Down Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    SetTextMeshProColor(gameObject=Self, color=var "New Colour", everyFrame=true)
    on UP → Delay
    on FINISHED → Sprite Off?
  }

  state Delay {
    Wait(time=var "Up Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
    on DOWN → Down
  }

  state Sprite Off? {
    GetColorRGBA(color=var "Down Colour", storeRed=(unset), storeGreen=(unset), storeBlue=(unset), storeAlpha=var "Aplha", everyFrame=false)
    FloatCompare(float1=var "Aplha", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetSpriteRenderer(gameObject=var "Self", active=false)
    on UP → Delay
    on FINISHED → Downed
  }

  state Upped {
    on DOWN → Down
  }

  state Up Instant {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Up Colour", everyFrame=false)
    SetTextMeshProColor(gameObject=Self, color=var "Up Colour", everyFrame=false)
    SetSpriteRenderer(gameObject=var "Self", active=true)
    on FINISHED → Upped
  }

  state Pulse Down {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Up Colour", everyFrame=false)
    SetSpriteRenderer(gameObject=var "Self", active=true)
    on FINISHED → Down
  }

  state Down Instant {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Down Colour", everyFrame=false)
    SetTextMeshProColor(gameObject=Self, color=var "Down Colour", everyFrame=false)
    on FINISHED → Sprite Off?
  }

  state Stopped {
    on UP → Delay
    on DOWN → Down
  }

  state Fade Up Instant {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Down Colour", everyFrame=false)
    SetTextMeshProColor(gameObject=Self, color=var "Down Colour", everyFrame=false)
    on FINISHED → Up
  }

  state Downed {
    on UP → Delay
  }

  var Down Time: float = 1
  var Up Delay: float = 0.5
  var Up Time: float = 1
  var Auto Up: bool = false
  var Down Colour: color = (1, 1, 1, 0)
  var Start Colour: color = (1, 1, 1, 0)
  var Up Colour: color = (1, 1, 1, 1)

  var Aplha: float = 0
  var New Colour: color = (0, 0, 0, 1)
  var Prev Colour: color = (1, 1, 1, 1)
  var Self: gameObject = <null>
}
