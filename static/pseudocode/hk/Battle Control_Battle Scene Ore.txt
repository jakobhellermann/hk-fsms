# Battle Control
# 1 location(s):
#   Abyss_17 / Battle Scene Ore

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="Lesser Mawlek 1", storeResult=var "Enemy 1")
    FindChild(gameObject=Self, childName="Music Region", storeResult=var "Music Region")
    FindChild(gameObject=Self, childName="Lesser Mawlek 2", storeResult=var "Enemy 2")
    SetFsmGameObject(gameObject=var "Enemy 1", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Self", everyFrame=false)
    SetFsmGameObject(gameObject=var "Enemy 2", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Self", everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"COMPLETE", lessThan=→"COMPLETE", greaterThan=(none), everyFrame=true)
    on ACTIVATE → Activated
    on COMPLETE → Completed
  }

  state Activated {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  state Completed {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BG OPEN", delay=1.5, everyFrame=false)
    ActivateGameObject(gameObject=var "Music Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=3)
  }

  var Battle Enemies: int = 2
  var Activated: bool = false
  var Enemy 1: gameObject = <null>
  var Enemy 2: gameObject = <null>
  var Music Region: gameObject = <null>
  var Self: gameObject = <null>
}
