# Mantis
# 4 location(s):
#   Fungus2_12 / Mantis (1)
#   Fungus2_13 / Mantis (1)
#   Fungus2_14 / Mantis (1)
#   Fungus2_15 / Mantis (1)

// component disabled: this FSM does not run until something enables it
fsm Mantis {
  start Pause
  on MANTIS DEFEAT → Pause  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Raycast X", floatValue=-2, everyFrame=false)
    SetFloatValue(floatVariable=var "Atk Speed Crt", floatValue=var "Attack Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Atk Speed Crt", multiplyBy=-1, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Scale", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatMultiply(floatVariable=var "Raycast X", multiplyBy=-1, everyFrame=false)
    FloatMultiply(floatVariable=var "Atk Speed Crt", multiplyBy=-1, everyFrame=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    on FINISHED → Lords Defeated?
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    BoolTest(boolVariable=var "Back Range", isTrue=→"BACK", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Front Range", isTrue=→"FRONT", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Above Range", isTrue=→"ABOVE", isFalse=(none), everyFrame=true)
    Wait(time=6, finishEvent=→"CHECK", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on BACK → Turn
    on FRONT → Feint?
    on ABOVE → Up Antic
    on CHECK → Return?
    on TOOK DAMAGE → Turn?
  }

  state Attack Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Moving", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Head Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Head Box F", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Set Speed
  }

  state Attack 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mantis_sword (sharedassets176.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Head Box F", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVector2XY(vector2Variable=var "Raycast Vector", vector2Value=(unset), x=var "Raycast X", y=-1, everyFrame=false)
    RayCast2d(fromGameObject=Self, fromPosition=var "Raycast Vector", direction=(0, -1), space=1, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "No Terrain", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    BoolFlipEveryFrame(boolVariable=var "No Terrain", everyFrame=true)
    BoolAllTrue(boolVariables=[var "No Terrain", var "Moving"], sendEvent=→"EDGE", storeResult=(unset), everyFrame=false)
    on FINISHED → Attack Swipe
    on EDGE → Stop At Edge
  }

  state Attack Swipe {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash Forward", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RayCast2d(fromGameObject=Self, fromPosition=var "Raycast Vector", direction=(0, -1), space=1, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "No Terrain", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    BoolFlipEveryFrame(boolVariable=var "No Terrain", everyFrame=true)
    BoolAllTrue(boolVariables=[var "No Terrain", var "Moving"], sendEvent=→"EDGE", storeResult=(unset), everyFrame=false)
    on FINISHED → Attack Recover
    on EDGE → Stop At Edge
  }

  state Attack Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash Forward", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=true)
    on FINISHED → Cooldown
  }

  state Cooldown {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateGameObject(gameObject=var "Head Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Turn {
    FloatMultiply(floatVariable=var "Atk Speed Crt", multiplyBy=-1, everyFrame=false)
    FloatMultiply(floatVariable=var "Raycast X", multiplyBy=-1, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Feint?
  }

  state Stop At Edge {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Moving", boolValue=false, everyFrame=false)
    GotoPreviousState()
  }

  state Set Speed {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Atk Speed Crt", y=(unset), everyFrame=false)
    on FINISHED → Attack 1
  }

  state Up Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Up Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Up Slash
  }

  state Up Slash {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mantis_sword (sharedassets176.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Up Attack Slash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash Up", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Up Recover
  }

  state Up Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Up Attack Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash Up", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Cooldown
  }

  state Lords Defeated? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedMantisLords", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → Friendly Idle
    on FALSE → Idle
  }

  state Friendly Idle {
    SetDamageHeroAmount(target=Self, damageDealt=0)
    BoolTest(boolVariable=var "Bow Range", isTrue=→"BOW", isFalse=(none), everyFrame=true)  // disabled
    ActivateGameObject(gameObject=var "Head Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDreamNailConvo(target=Self, title="MANTIS_PASSIVE")
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=10, tolerance=0, equal=(none), lessThan=→"BOW", greaterThan=(none), everyFrame=true)
    on BOW → Bow
    on TOOK DAMAGE → Reset
  }

  state Bow {
    BoolTest(boolVariable=var "Bow Range", isTrue=(none), isFalse=→"END", everyFrame=true)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bow")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Turn To Bow", resetFrame=true, everyFrame=true)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=12, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"END", everyFrame=true)
    on END → Anim
    on TOOK DAMAGE → Reset
  }

  state Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bow End")
    on FINISHED → Friendly Idle
  }

  state Reset {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetFloatValue(floatVariable=var "Raycast X", floatValue=-2, everyFrame=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    SetFloatValue(floatVariable=var "Atk Speed Crt", floatValue=var "Attack Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Atk Speed Crt", multiplyBy=-1, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Scale", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatMultiply(floatVariable=var "Raycast X", multiplyBy=-1, everyFrame=false)
    FloatMultiply(floatVariable=var "Atk Speed Crt", multiplyBy=-1, everyFrame=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  state Return? {
    GetPosition(gameObject=Self, vector=(unset), x=var "Current X", y=var "Current Y", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Current Y", float2=var "Start Y", tolerance=0.25, equal=(none), lessThan=→"CANCEL", greaterThan=→"CANCEL", everyFrame=false)
    FloatCompare(float1=var "Current X", float2=var "Start X", tolerance=1, equal=→"CANCEL", lessThan=(none), greaterThan=(none), everyFrame=false)
    on CANCEL → Idle
    on L → Return L
    on R → Return R
  }

  state Return L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-6, y=(unset), everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Current X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Current X", float2=var "Start X", tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    on END → Reset
    on FRONT → Feint?
    on BACK → Turn
    on ABOVE → Up Antic
  }

  state Return R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=6, y=(unset), everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Current X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Current X", float2=var "Start X", tolerance=0, equal=→"END", lessThan=(none), greaterThan=→"END", everyFrame=true)
    on END → Reset
    on FRONT → Feint?
    on BACK → Turn
    on ABOVE → Up Antic
  }

  state Feint? {
    SendRandomEvent(events=[→"FINISHED", →"FEINT"], weights=[0.65, 0.35], delay=0)
    on FINISHED → Attack Antic
    on FEINT → Feint
  }

  state Feint {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mantis_steps_back (sharedassets176.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFloatValue(floatVariable=var "Feint Speed", floatValue=var "Atk Speed Crt", everyFrame=false)
    SetFloatValue(floatVariable=var "Raycast Feint X", floatValue=var "Raycast X", everyFrame=false)
    FloatMultiply(floatVariable=var "Feint Speed", multiplyBy=-0.3, everyFrame=false)
    FloatMultiply(floatVariable=var "Raycast Feint X", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Feint Speed", y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk B")
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetVector2XY(vector2Variable=var "Raycast Vector Feint", vector2Value=(unset), x=var "Raycast Feint X", y=-1, everyFrame=false)
    RayCast2d(fromGameObject=Self, fromPosition=var "Raycast Vector Feint", direction=(0, -1), space=1, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "No Terrain", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    BoolFlipEveryFrame(boolVariable=var "No Terrain", everyFrame=true)
    BoolTest(boolVariable=var "No Terrain", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    on FINISHED → Attack Antic
    on TOOK DAMAGE → Attack Antic
  }

  state Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  state Turn? {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Facing Right", everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Hero is Right", leftBool=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hero is Right", var "Facing Right"], boolStates=[true, false], trueEvent=→"BACK", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hero is Right", var "Facing Right"], boolStates=[false, true], trueEvent=→"BACK", falseEvent=→"FINISHED", storeResult=(unset), everyFrame=false)
    on BACK → Turn
    on FINISHED → Attack Antic
  }

  var Attack Speed: float = 30
  var No Terrain: bool = false

  var Atk Speed Crt: float = 0
  var Current X: float = 0
  var Current Y: float = 0
  var Distance: float = 0
  var Feint Speed: float = 0
  var Raycast Feint X: float = 0
  var Raycast X: float = 0
  var Scale: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Above Range: bool = false
  var Back Range: bool = false
  var Bow Range: bool = false
  var Facing Right: bool = false
  var Front Range: bool = false
  var Hero is Right: bool = false
  var Moving: bool = false
  var Raycast Vector: vector2 = (0, 0)
  var Raycast Vector Feint: vector2 = (0, 0)
  var Head Box: gameObject = Mantis (1)/Head Box
  var Head Box F: gameObject = Mantis (1)/Head Box F
  var Self: gameObject = <null>
  var Slash Forward: gameObject = Mantis (1)/Slash Forward
  var Slash Up: gameObject = Mantis (1)/Slash Up
}
