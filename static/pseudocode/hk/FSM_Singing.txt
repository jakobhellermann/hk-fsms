# FSM
# 1 location(s):
#   Menu_Title / Menu_Styles/Brilliant_Style/Singing

fsm FSM {
  start Pause

  state Pause {
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → frame
  }

  state frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Transition
  }

  state Transition {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    on FINISHED → Play
  }

  state Play {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }
}
