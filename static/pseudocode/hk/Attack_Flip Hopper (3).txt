# 12 location(s):
#   Waterways_01 / _Enemies/Flip Hopper (3)
#   Waterways_01 / _Enemies/Flip Hopper (1)
#   Waterways_01 / _Enemies/Flip Hopper
#   Waterways_01 / _Enemies/Flip Hopper (2)
#   Waterways_02 / Flip Hopper
#   Waterways_02 / Flip Hopper (1)
#   Waterways_02 / Flip Hopper (2)
#   Waterways_06 / Flip Hopper
#   Waterways_06 / Flip Hopper (1)
#   Waterways_06 / Flip Hopper (2)
#   Waterways_07 / Flip Hopper (1)
#   Waterways_07 / Flip Hopper

fsm Attack {
  start Init
  on TOOK DAMAGE → Flip  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetBoolValue(boolVariable=var "Side A", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "isFlipping", boolValue=false, everyFrame=false)
    SetStringValue(stringVariable=var "FlipAnim", stringValue="Flip", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    SetBoolValue(boolVariable=var "isFlipping", boolValue=false, everyFrame=false)
    BoolTest(boolVariable=var "Side A", isTrue=→"Side A", isFalse=→"Side B", everyFrame=true)
    on Side A → Side A
    on Side B → Check Collison 2
  }

  state Side A {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Facing Right", greaterThanBool=(unset), everyFrame=false)
    SetWalkerFacing(walkRight=var "Facing Right", randomStartDir=false, target=Self, everyFrame=false)
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    SetStringValue(stringVariable=var "FlipAnim", stringValue="Flip", everyFrame=false)
    SetBoxCollider2DSize(gameObject1=Self, width=2.137, height=1.511, offsetX=-0.2667122, offsetY=-0.533)
  }

  state Side B {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Side B Hop")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=flip_hopper_jump (sharedassets349.assets))
    GetScale(gameObject=Self, vector=(unset), xScale=var "Random Sign", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    RandomFloat(min=5, max=12, storeResult=var "X Vel")
    RandomFloat(min=20, max=25, storeResult=var "Jump Height")
    FloatMultiply(floatVariable=var "X Vel", multiplyBy=var "Random Sign", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Vel", y=var "Jump Height", everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Collison
  }

  state Flip {
    BoolTest(boolVariable=var "isFlipping", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=flip_hopper_flip (sharedassets349.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    BoolFlip(boolVariable=var "Side A")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "FlipAnim")
    SetBoolValue(boolVariable=var "isFlipping", boolValue=true, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopWalker(target=Self, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Hop Start {
    SendEventByNameV2(eventTarget=GameObject(Self), sendEvent="MOVESTOP", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=flip_hopper_jump (sharedassets349.assets))  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Side B Turn", resetFrame=false, everyFrame=false)  // disabled
    SetStringValue(stringVariable=var "FlipAnim", stringValue="Flip B", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hop Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetBoxCollider2DSize(gameObject1=Self, width=2.137, height=1.512, offsetX=0.137, offsetY=-0.533)
    on FINISHED → Side B
  }

  state Hop Recover {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hop Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hero Check
  }

  state Check Collison {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"HitGround", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on HitGround → Hop Recover
  }

  state Turn? {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Facing Right", everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Hero Right", leftBool=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Right"], boolStates=[true, true], trueEvent=→"CANCEL", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Right"], boolStates=[false, false], trueEvent=→"CANCEL", falseEvent=(none), storeResult=(unset), everyFrame=false)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Side B Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hop Start
    on CANCEL → Hop Start
  }

  state Flip Scale {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Hop Start
  }

  state Hero Check {
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"FLIP", everyFrame=false)
    on FINISHED → Idle
    on FLIP → Flip
  }

  state Check Collison 2 {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"HitGround", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"HitGround", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on HitGround → Turn?
  }

  var Jump Height: float = 0
  var Random Sign: float = 0
  var X Scale: float = 0
  var X Vel: float = 0
  var Facing Right: bool = false
  var Hero Range: bool = false
  var Hero Right: bool = false
  var Side A: bool = false
  var isFlipping: bool = false
  var FlipAnim: string = ""
  var Hopper Hit: gameObject = <null>
  var Self: gameObject = <null>
  var Walker Hit: gameObject = <null>
}
