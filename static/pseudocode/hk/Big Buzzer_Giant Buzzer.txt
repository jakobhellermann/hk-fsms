fsm Big Buzzer {

    on FINISHED → Intro Roar
  }

  state Intro Roar {
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)

    RandomFloat(min=0.15, max=0.22, storeResult=var "Accel")
    WaitRandom(timeMin=3, timeMax=4, finishEvent=→"ATTACK", realTime=false)
    on OUT OF RANGE → Lurk
    on ATTACK → Choose Attack
  }

  state Lurk {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    ChaseObject(gameObject=Self, target=var "Marker Lurk", speedMax=6, acceleration=var "Accel", targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0)

    FloatCompare(float1=var "Chooser", float2=75, tolerance=0, equal=→"SUMMON", lessThan=→"SWOOP", greaterThan=→"SUMMON", everyFrame=false)
    on SWOOP → Swoop Antic
    on SUMMON → Summon Antic
  }

  }

  }
