# Challenge Start
# 1 location(s):
#   Fungus2_15_boss / Mantis Battle/Challenge Prompt

fsm Challenge Start {
  start Init 2
  on CONVO CANCEL → Idle 2  // from any state

  state Init 2 {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=(unset), x=var "Challenge X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Can Talk", boolValue=true, everyFrame=false)
    FindChild(gameObject=var "Self", childName="Prompt Marker", storeResult=var "Prompt Marker")
    FindChild(gameObject=var "Self", childName="Ranges", storeResult=var "Ranges Folder")
    SetParent(gameObject=var "Prompt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetScale(gameObject=var "Prompt", vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    FindGameObject(objectName="", withTag="MainCamera", store=var "Main Camera")
    CreateObject(gameObject=Arrow Prompt (resources.assets), spawnPoint=var "Prompt Marker", position=(unset), rotation=(unset), storeObject=var "Prompt", networkInstantiate=false, networkGroup=0)
    SetFsmString(gameObject=var "Prompt", fsmName="Prompt Control", variableName="Prompt Name", setValue=var "Prompt Name", everyFrame=false)
    DestroyObject(gameObject=var "Prompt Marker", delay=0, detachChildren=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=var "Sprite Faces Right", everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle 2
  }

  state Idle 2 {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"IN RANGE", storeCollider=var "")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD CHALLENGE EXIT", delay=0, everyFrame=false)
    on IN RANGE → Can Talk Bool?
  }

  state In Range {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD CHALLENGE ENTER", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="UP", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"OUT OF RANGE", storeCollider=var "")
    GetAxis(axisName="Vertical", multiplier=1, store=var "Axis Vert", everyFrame=true)  // disabled
    FloatCompare(float1=var "Axis Vert", float2=0.6, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UP PRESSED", everyFrame=true)  // disabled
    ListenForUp(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForDown(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    on OUT OF RANGE → Idle 2
    on UP PRESSED → Can Talk?
  }

  state Can Talk? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="CanInput", parameters=[], storeResult=var "Can Input")
    BoolTest(boolVariable=var "Can Input", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["onGround"], storeResult=var "On Ground")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["attacking"], storeResult=var "Attacking")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["upAttacking"], storeResult=var "Attacking Up")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["downAttacking"], storeResult=var "Attacking Down")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["dashing"], storeResult=var "Dashing")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["backDashing"], storeResult=var "Backdashing")
    BoolTest(boolVariable=var "On Ground", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    BoolAnyTrue(boolVariables=[var "Attacking", var "Attacking Down", var "Attacking Up", var "Backdashing", var "Dashing"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Take Control
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Range
  }

  state Take Control {
    SetBoolValue(boolVariable=var "Talking", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Ranges Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var))
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Set Facing
  }

  state In Range Turns {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="UP", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"OUT OF RANGE", storeCollider=var "")
    GetAxis(axisName="Vertical", multiplier=1, store=var "Axis Vert", everyFrame=true)  // disabled
    FloatCompare(float1=var "Axis Vert", float2=0.6, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UP PRESSED", everyFrame=true)  // disabled
    ListenForUp(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=false, stateEntryOnly=false)
    ListenForDown(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=false, stateEntryOnly=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip=var "Turn Anim Name", resetFrame=true, everyFrame=true)
    on OUT OF RANGE → Idle 2
    on UP PRESSED → Can Talk?
  }

  state Turns? {
    BoolTest(boolVariable=var "Always Faces Hero", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → In Range Turns
    on FALSE → In Range
  }

  state Can Talk Bool? {
    BoolTest(boolVariable=var "Can Talk", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Turns?
    on FALSE → Idle 2
  }

  state Send Challenge {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD CHALLENGE", delay=0.7, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → 
  }

  state Set Facing {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=30.24, tolerance=0, equal=→"ON LEFT", lessThan=→"ON LEFT", greaterThan=→"ON RIGHT", everyFrame=false)
    on ON LEFT → On Left
    on ON RIGHT → On Right
  }

  state On Left {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Challenge
  }

  state On Right {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    on FINISHED → Challenge
  }

  state Challenge {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Challenge Start")
    Tk2dWatchAnimationEvents(gameObject=var "Hero", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="CamLocks", storeResult=var "CamLock")
    ActivateAllChildren(gameObject=var "CamLock", activate=true)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Challenge Audio
  }

  state Challenge Audio {
    AudioPlay(gameObject=var "Hero", volume=1, oneShotClip=hero_unsheath (sharedassets180.assets), finishedEvent=(none))
    Tk2dWatchAnimationEvents(gameObject=var "Hero", animationTriggerEvent=(none), animationCompleteEvent=→"TRUE")
    on TRUE → Send Challenge
  }

  var Always Faces Hero: bool = false
  var Sprite Faces Right: bool = false
  var Prompt Name: string = "Challenge"
  var Turn Anim Name: string = ""

  var Axis Vert: float = 0
  var Challenge X: float = 0
  var Hero X: float = 0
  var Attacking: bool = false
  var Attacking Down: bool = false
  var Attacking Up: bool = false
  var Backdashing: bool = false
  var Can Input: bool = false
  var Can Talk: bool = true
  var Dashing: bool = false
  var Facing Right: bool = false
  var On Ground: bool = false
  var Talking: bool = false
  var CamLock: gameObject = <null>
  var Hero: gameObject = <null>
  var Main Camera: gameObject = <null>
  var Parent: gameObject = <null>
  var Prompt: gameObject = <null>
  var Prompt Marker: gameObject = <null>
  var Ranges Folder: gameObject = <null>
  var Self: gameObject = <null>
}
