# CREntryPauseFSM
# 1 location(s):
#   Crossroads_01 / Crossroad Entry Pauser

fsm CREntryPauseFSM {
  start Init

  state Pause {
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero Y", float2=27, tolerance=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=EnterWithoutInput(true))
    Wait(time=0.9, finishEvent=→"UNPAUSE", realTime=false)
    on CANCEL → Cancel
    on UNPAUSE → Force Hard Landing
  }

  state Cancel {
  }

  state Unpause {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AcceptInput", parameters=[], storeResult=(unset var))  // disabled
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=AcceptInput())
  }

  state Init {
    NextFrameEvent(sendEvent=→"INIT")  // disabled
    WaitForHeroInPosition(sendEvent=→"INIT", skipIfAlreadyPositioned=false)
    on INIT → Pause
  }

  state Force Hard Landing {
    Wait(time=1.05, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=ForceHardLanding())
    on FINISHED → Unpause
  }

  var Hero Y: float = 0
  var Hero: gameObject = <null>
}
