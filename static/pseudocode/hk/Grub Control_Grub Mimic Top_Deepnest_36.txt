# Grub Control
# 1 location(s):
#   Deepnest_36 / Grub Mimic Top

fsm Grub Control {
  start Pause
  on FREE → Free  // from any state
  on IN BOTTLE → Idle  // from any state

  state Idle {
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    WaitRandom(timeMin=3, timeMax=6, finishEvent=→"CRY", realTime=false)
    on ENTER → Hero Close
    on FREE → Free
    on CRY → Cry
  }

  state Hero Close {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[grub_alert (sharedassets33.assets), grub_idle_or_look_1 (sharedassets33.assets), grub_idle_or_look_2 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Cry")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Cry Turn", resetFrame=true, everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Sad Wait
    on FREE → Free
  }

  state Init {
    GetHero(storeResult=var "Hero")
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Dig Dust", storeResult=var "Dig Dust")
    GetPersistentBool(target=var "Mimic", storeValue=var "Is Dead")  // disabled
    GetIsDead(target=var "Mimic", storeValue=var "Is Dead")
    BoolTest(boolVariable=var "Is Dead", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Mimic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Mimic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Idle No Bottle
    on ACTIVATE → Activate
  }

  state Free {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Instant Trans?
  }

  state Sad Sound {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[grub_sad_1 (sharedassets33.assets), grub_sad_2 (sharedassets33.assets), grub_sad_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Idle Wait
  }

  state Sad Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sad Sound
  }

  state Idle Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Cry {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grub_sad_idle_01 (sharedassets33.assets), Grub_sad_idle_02 (sharedassets33.assets), Grub_sad_idle_03 (sharedassets33.assets), Grub_sad_idle_04 (sharedassets33.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Idle
  }

  state Instant Trans? {
    SendRandomEvent(events=[→"CRY", →"TRANSFORM"], weights=[1, 1], delay=0)
    on TRANSFORM → Transform
    on CRY → Fake Cry
  }

  state Transform {
    ActivateGameObject(gameObject=var "Mimic", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    GameObjectIsNull(gameObject=var "Mimic", isNull=→"ACTIVATE", isNotNull=(none), storeResult=(unset), everyFrame=true)
    on ACTIVATE → Transformed
  }

  state Fake Cry {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Freed Fake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[grub_free_2 (sharedassets39.assets), grub_free_1 (sharedassets39.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Transform
  }

  state Idle No Bottle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on ENTER → Free
  }

  state Transformed {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)  // disabled
    on KILLED → Set Active
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetParent(gameObject=var "Dream Dialogue", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Pause 2
  }

  state Activate {
    DestroySelf(detachChildren=false)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    Wait(time=1, finishEvent=(none), realTime=false)  // disabled
    on FINISHED → Init
  }

  state Set Active {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  var Activated: bool = false
  var Is Dead: bool = false
  var Dig Dust: gameObject = <null>
  var Dream Dialogue: gameObject = Grub Mimic Top/Dream Dialogue
  var Hero: gameObject = <null>
  var Mimic: gameObject = Grub Mimic Top/Grub Mimic 1
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
