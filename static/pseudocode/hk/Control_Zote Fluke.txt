# Control
# 1 location(s):
#   GG_Mighty_Zote / Battle Control/Zote Fluke

fsm Control {
  start Init
  on ZERO HP → Death  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Pt Antic", storeResult=var "Pt Antic")
    GetOwner(storeGameObject=var "Self")
    GetHP(target=Self, storeValue=var "Max HP")
    GetParent(gameObject=Self, storeResult=var "Parent")  // disabled
    on FINISHED → Dormant
  }

  state Dormant {
    on GO → Pos
  }

  state Pos {
    RandomFloat(min=90.45, max=115.64, storeResult=var "X Pos")
    FloatOperator(float1=var "X Pos", float2=1, operation=Subtract, storeResult=var "X Min", everyFrame=false)
    FloatOperator(float1=var "X Pos", float2=1, operation=Add, storeResult=var "X Max", everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=7.12, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetHP(target=Self, hp=var "Max HP")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Antic
  }

  state Antic {
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fluke Antic")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=worm (sharedassets51.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Emerge 1
  }

  state Emerge 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fluke Emerge 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Emerge 2
  }

  state Emerge 2 {
    StopParticleEmitter(gameObject=var "Pt Antic")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fluke Emerge 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Climb
  }

  state Climb {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fluke Idle")
    GhostMovement(gameObject=Self, xPosMin=var "X Min", xPosMax=var "X Max", accel_x=0.3, speedMax_x=3, yPosMin=999, yPosMax=999, accel_y=0.1, speedMax_y=8, direction_x=0, direction_y=0)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=26, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"END", everyFrame=true)
    SpawnObjectFromGlobalPoolOverTimeV2(gameObject=Vomit Glob Nosk (sharedassets290.assets), spawnPoint=var "Self", position=(0, 0, 0.001), rotation=(unset), frequency=0.85, scaleMin=0.85, scaleMax=1.2)
    on END → Sleep
  }

  state Sleep {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Sleeping
  }

  state Sleeping {
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="RETURN", delay=0, everyFrame=false)
    on GO → Pos
  }

  state Death {
    SpawnObjectFromGlobalPool(gameObject=Zote Crew Death (sharedassets481.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventToRegister(eventName="ZOTE WARRIOR KILLED")
    SetHP(target=Self, hp=var "Max HP")
    SetIsDead(target=Self, setValue=false)
    on FINISHED → Sleeping
  }

  var Parent: gameObject = Battle Control/Extra Zotes

  var X Max: float = 0
  var X Min: float = 0
  var X Pos: float = 0
  var Y Pos: float = 0
  var Max HP: int = 0
  var Pt Antic: gameObject = <null>
  var Self: gameObject = <null>
}
