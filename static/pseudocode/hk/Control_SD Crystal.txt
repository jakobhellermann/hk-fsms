# Control
# 1 location(s):
#   resources.assets / SD Crystal

fsm Control {
  start Pause
  on SUPERDASH BLAST → Blast  // from any state
  on SUPERDASH CANCEL → Cancel  // from any state

  state Init {
    RandomFloat(min=1, max=1.2, storeResult=var "Scale X")
    RandomFloat(min=-0.2, max=0.2, storeResult=var "Scale Y Add")
    SetFloatValue(floatVariable=var "Scale X Add", floatValue=var "Scale Y Add", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=(unset), yScale=var "Scale Y", zScale=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Scale Y", add=var "Scale Y Add", everyFrame=false, perSecond=false)
    RandomlyFlipFloat(storeResult=var "Scale X")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale X", y=var "Scale Y", z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=-12, max=12, storeResult=var "Rotation")
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Grow
  }

  state Grow {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="SD Crys Grow")
    on SUPERDASH READY → Glow
  }

  state Glow {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="SD Crys Flash")
  }

  state Blast {
    RecycleSelf()
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="SD Crys Grow")
    RandomFloat(min=0.005, max=0.005999, storeResult=var "z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "z Pos", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Init
  }

  state Cancel {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="SD Crys Shrink", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Current Scale: float = 0
  var Rotation: float = 0
  var Scale X: float = 0
  var Scale X Add: float = 0
  var Scale Y: float = 0
  var Scale Y Add: float = 0
  var Tween Time: float = 0
  var z Pos: float = 0
  var Scale: vector3 = (0, 0, 0)
}
