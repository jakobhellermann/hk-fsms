#   Room_Colosseum_Bronze / Colosseum Manager/Crowd Audio
#   Room_Colosseum_Silver / Colosseum Manager/Crowd Audio

#   Room_Colosseum_Spectate / Crowd Audio

fsm Hero Death {
  start Idle

  state Idle {
    on HERO LEAVE → Pause
  }

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Laugh
  }

  state Laugh {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=crowd_laugh_3_long (sharedassets32.assets))
  }
}
