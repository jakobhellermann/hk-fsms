# Control
# 1 location(s):
#   sharedassets6.assets / Grimm_flare_pillar

fsm Control {
  start Pillar

  state Pillar {
    Wait(time=0.45, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=nightmare_lantern_powering_up)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.2, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Shake
  }

  state AfterBurn {
    PlayParticleEmitter(gameObject=var "Pt Afterburn", emit=0)
    SetCircleCollider(gameObject=var "Pt Afterburn", active=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burn End
  }

  state Burn End {
    SetCircleCollider(gameObject=var "Pt Afterburn", active=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Shake {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Pt Afterburn", audioClip=grimm_explode_into_bats, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → AfterBurn
  }

  var Pt Afterburn: gameObject = Grimm_flare_pillar/Pt Afterburn
}
