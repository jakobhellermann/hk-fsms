# Control
# 1 location(s):
#   Deepnest_41 / Happy Spider NPC

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="NPC", storeResult=var "NPC")
    FindChild(gameObject=Self, childName="Damager", storeResult=var "Damager")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    SetMeshRenderer(gameObject=Self, active=false)
    BoolTest(boolVariable=var "Range", isTrue=→"ENTER APPEAR", isFalse=(none), everyFrame=true)
    on ENTER APPEAR → Enter
  }

  state Enter {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=platform_dirt_fall_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.15, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=bone_pile_movement_1 (sharedassets299.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetCollider(gameObject=var "NPC", active=true)
    on CONVO END → Attack Pause
    on BLOCKED HIT → Block Disable
  }

  state Attack Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    SetCollider(gameObject=var "NPC", active=false)
    SetCollider(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → IdleToAntic
    on BLOCKED HIT → Blocked Hit
  }

  state IdleToAntic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="IdleToAntic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=spider_soldier_open_3 (sharedassets292.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Attack Idle
    on BLOCKED HIT → Blocked Hit
  }

  state Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Attack
    on BLOCKED HIT → Blocked Hit
  }

  state Attack {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=var "Damager", active=true)
    SetCollider(gameObject=var "Damager", active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=brkn_wand_jump (sharedassets34.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClips=[Midwife_big_chomp_05 (sharedassets299.assets), Midwife_big_chomp_02 (sharedassets299.assets), Midwife_big_chomp_01 (sharedassets299.assets)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Attack Recover
    on BLOCKED HIT → Blocked Hit
  }

  state Attack Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=var "Damager", active=false)
    SetCollider(gameObject=var "Damager", active=false)
    on FINISHED → Attack Idle
    on BLOCKED HIT → Blocked Hit
  }

  state Attack Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    BoolTest(boolVariable=var "Leave Range", isTrue=(none), isFalse=→"LEAVE", everyFrame=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Range Check
    on BLOCKED HIT → Blocked Hit
    on LEAVE → Leave
  }

  state Range Check {
    BoolTest(boolVariable=var "Range", isTrue=→"ATTACK", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Attack Idle
    on ATTACK → Antic
    on BLOCKED HIT → Blocked Hit
  }

  state Blocked Hit {
    SetPolygonCollider(gameObject=var "Damager", active=false)
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Retreat", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=var "Damager", active=false)
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=platform_dirt_fall_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.45, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=bone_pile_movement_2 (sharedassets299.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.45, storePlayer=<null>)
    on FINISHED → Retreated
  }

  state Retreated {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Leave {
    SetPolygonCollider(gameObject=var "Damager", active=false)
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Quick Retreat", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=var "Damager", active=false)
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=platform_dirt_fall_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.15, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "NPC", audioClip=bone_pile_movement_2 (sharedassets299.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Retreated
  }

  state Block Disable {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    SetCollider(gameObject=var "NPC", active=false)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Blocked Hit
  }

  var Leave Range: bool = false
  var Range: bool = false
  var Damager: gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var NPC: gameObject = <null>
}
