# Hopper
# 1 location(s):
#   Deepnest_East_06 / Hopper Spawn/Giant Hopper

fsm Hopper {
  start Init
  on GO LEFT → Hit Wall R  // from any state
  on GO RIGHT → Hit Wall L  // from any state
  on TO AIR → In Air  // from any state

  state In Air {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetFloatValue(floatVariable=var "Old Velocity X", floatValue=var "Velocity X", everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=true)
    SetFloatValue(floatVariable=var "Mirror X", floatValue=var "Old Velocity X", everyFrame=true)
    FloatMultiply(floatVariable=var "Mirror X", multiplyBy=-1, everyFrame=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"RIGHT", bottomHitEvent=(none), leftHitEvent=→"LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"RIGHT", bottomHitEvent=(none), leftHitEvent=→"LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land Anim
    on LEFT → Hit Wall L
    on RIGHT → Hit Wall R
  }

  state Jump Anticipate {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    on FINISHED → Left Or Right?
  }

  state Left Or Right? {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=giant_hopper_jump (sharedassets33.assets), finishedEvent=(none))
    BoolTest(boolVariable=var "Moving Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Jump Right
    on LEFT → Jump Left
  }

  state Jump Right {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="JUMPED", delay=0, everyFrame=false)
    RandomFloat(min=8, max=10, storeResult=var "Jump X")
    RandomFloat(min=45, max=50, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    on FINISHED → In Air
  }

  state Jump Left {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="JUMPED", delay=0, everyFrame=false)
    RandomFloat(min=-8, max=-10, storeResult=var "Jump X")
    RandomFloat(min=45, max=50, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    on FINISHED → In Air
  }

  state Hit Wall L {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Moving Right", boolValue=true, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Mirror X", y=(unset), everyFrame=false)
    NextFrameEvent(sendEvent=→"WAIT")
    on WAIT → In Air
  }

  state Hit Wall R {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Moving Right", boolValue=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Mirror X", y=(unset), everyFrame=false)
    NextFrameEvent(sendEvent=→"WAIT")
    on WAIT → In Air
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Air", isTrue=→"TO AIR", isFalse=(none), everyFrame=false)
    GetDistance(gameObject=var "Self", target=var "MainCamera", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=44, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Left Or Right?
    on TO AIR → In Air
  }

  state Check Alert {
    BoolTestMulti(boolVariables=[var "Alert Range", var "Alerted"], boolStates=[true, false], trueEvent=→"ALERT", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Alerted", var "Unalert Range"], boolStates=[true, false], trueEvent=→"UNALERT", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on ALERT → Set Alert
    on FINISHED → Alerted?
    on UNALERT → Set Unalert
  }

  state Land Anim {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=giant_hopper_land (sharedassets33.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    GetDistance(gameObject=var "Self", target=var "MainCamera", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=44, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Check Alert
  }

  state Alerted? {
    BoolTest(boolVariable=var "Alerted", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Jump Anticipate Alert
    on FALSE → Jump Anticipate
  }

  state Set Alert {
    SetBoolValue(boolVariable=var "Alerted", boolValue=true, everyFrame=false)
    on FINISHED → Alerted?
  }

  state Set Unalert {
    SetBoolValue(boolVariable=var "Alerted", boolValue=false, everyFrame=false)
    on FINISHED → Alerted?
  }

  state Jump Anticipate Alert {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    RandomFloat(min=1.335, max=1.65, storeResult=var "Multiplier")
    FloatOperator(float1=var "Hero X", float2=var "Self X", operation=Subtract, storeResult=var "Jump X", everyFrame=false)
    FloatMultiply(floatVariable=var "Jump X", multiplyBy=var "Multiplier", everyFrame=false)
    FloatClamp(floatVariable=var "Jump X", minValue=-14, maxValue=14, everyFrame=false)
    RandomFloat(min=45, max=55, storeResult=var "Jump Y")
    on FINISHED → Jump
  }

  state Jump {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="JUMPED", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=giant_hopper_jump (sharedassets33.assets))
    on FINISHED → In Air
  }

  var Mirror X: float = 0
  var Velocity X: float = 0
  var Moving Right: bool = false
  var Start Air: bool = false

  var Distance: float = 0
  var Hero X: float = 16.06
  var Jump X: float = 0
  var Jump Y: float = 0
  var Multiplier: float = 0
  var Old Velocity X: float = 0
  var Self X: float = 0
  var Alert Range: bool = false
  var Alerted: bool = false
  var Unalert Range: bool = true
  var Dust: gameObject = Hopper Spawn/Giant Hopper/Dust Land Small
  var Self: gameObject = <null>
}
