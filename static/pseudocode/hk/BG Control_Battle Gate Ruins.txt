# BG Control
# 2 location(s):
#   Ruins2_09 / Battle Gate Ruins
#   Ruins1_05 / Battle Gate Ruins

fsm BG Control {
  start Init
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gate Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=gate_slam (sharedassets27.assets))
    on BG OPEN → Open
  }

  state Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Open")  // disabled
    PlayParticleEmitter(gameObject=var "Raise Dust", emit=0)  // disabled
    SetCollider(gameObject=Self, active=false)  // disabled
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=switch_gate_gate (sharedassets39.assets))
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Opened
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Gate Closed")
    SetMeshRenderer(gameObject=Self, active=true)
    on BG OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Opened")  // disabled
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Opened {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  var Start Closed: bool = false

  var Raise Dust: gameObject = Battle Gate Ruins/Raise Dust
  var Self: gameObject = <null>
}
