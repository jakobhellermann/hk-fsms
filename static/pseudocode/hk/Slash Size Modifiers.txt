# Slash Size Modifiers
# 2 location(s):
#   Knight_Pickup / Knight/Charm Effects
#   resources.assets / Knight:1/Charm Effects

fsm Slash Size Modifiers {
  start Init
  on CHARM EQUIP CHECK → Check Longnail  // from any state

  state Check Mantis {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_13", isTrue=→"EQUIPPED", isFalse=→"UNEQUIPPED")
    on EQUIPPED → Equipped
    on UNEQUIPPED → Unequipped
  }

  state Equipped {
    SendMessage(gameObject=var "Alt Slash", delivery=SendMessage, options=1, functionCall=SetMantis(true))
    SendMessage(gameObject=var "Down Slash", delivery=SendMessage, options=1, functionCall=SetMantis(true))
    SendMessage(gameObject=var "Slash", delivery=SendMessage, options=1, functionCall=SetMantis(true))
    SendMessage(gameObject=var "Up Slash", delivery=SendMessage, options=1, functionCall=SetMantis(true))
    SendMessage(gameObject=var "WallSlash", delivery=SendMessage, options=1, functionCall=SetMantis(true))
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Knight")
    FindChild(gameObject=var "Knight", childName="Attacks", storeResult=var "Attacks")
    FindChild(gameObject=var "Attacks", childName="AltSlash", storeResult=var "Alt Slash")
    FindChild(gameObject=var "Attacks", childName="DownSlash", storeResult=var "Down Slash")
    FindChild(gameObject=var "Attacks", childName="UpSlash", storeResult=var "Up Slash")
    FindChild(gameObject=var "Attacks", childName="Slash", storeResult=var "Slash")
    FindChild(gameObject=var "Attacks", childName="WallSlash", storeResult=var "WallSlash")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Longnail
  }

  state Unequipped {
    SendMessage(gameObject=var "Alt Slash", delivery=SendMessage, options=1, functionCall=SetMantis(false))
    SendMessage(gameObject=var "Down Slash", delivery=SendMessage, options=1, functionCall=SetMantis(false))
    SendMessage(gameObject=var "Slash", delivery=SendMessage, options=1, functionCall=SetMantis(false))
    SendMessage(gameObject=var "Up Slash", delivery=SendMessage, options=1, functionCall=SetMantis(false))
    SendMessage(gameObject=var "WallSlash", delivery=SendMessage, options=1, functionCall=SetMantis(false))
  }

  state Check Longnail {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_18", isTrue=→"EQUIPPED", isFalse=→"UNEQUIPPED")
    on EQUIPPED → Equipped 2
    on UNEQUIPPED → Unequipped 2
  }

  state Equipped 2 {
    SendMessage(gameObject=var "Alt Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(true))
    SendMessage(gameObject=var "Down Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(true))
    SendMessage(gameObject=var "Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(true))
    SendMessage(gameObject=var "Up Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(true))
    SendMessage(gameObject=var "WallSlash", delivery=SendMessage, options=1, functionCall=SetLongnail(true))
    on FINISHED → Check Mantis
  }

  state Unequipped 2 {
    SendMessage(gameObject=var "Alt Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(false))
    SendMessage(gameObject=var "Down Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(false))
    SendMessage(gameObject=var "Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(false))
    SendMessage(gameObject=var "Up Slash", delivery=SendMessage, options=1, functionCall=SetLongnail(false))
    SendMessage(gameObject=var "WallSlash", delivery=SendMessage, options=1, functionCall=SetLongnail(false))
    on FINISHED → Check Mantis
  }

  var Alt Slash: gameObject = <null>
  var Attacks: gameObject = <null>
  var Down Slash: gameObject = <null>
  var Great Slash: gameObject = <null>
  var Hit L: gameObject = <null>
  var Hit R: gameObject = <null>
  var Knight: gameObject = <null>
  var Slash: gameObject = <null>
  var Up Slash: gameObject = <null>
  var WallSlash: gameObject = <null>
}
