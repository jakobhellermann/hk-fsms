# Ash Control
# 3 location(s):
#   resources.assets / Ash 2
#   resources.assets / Ash 1
#   resources.assets / Ash 3

fsm Ash Control {
  start Start

  state Start {
    RandomFloat(min=1, max=2, storeResult=var "Scaler")
    SetScale(gameObject=Self, vector=(unset), x=var "Scaler", y=var "Scaler", z=var "Scaler", everyFrame=false, lateUpdate=false)
    Wait(time=0.01, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Turn
  }

  state Turn {
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=10, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Speed", everyFrame=false)
  }

  var Scaler: float = 0
  var Speed: float = 0
}
