  on GHOST DEATH → End  // from any state

    on READY → Emerge
  }

  state Emerge {

    PlayParticleEmitter(gameObject=var "Dust Emerge", emit=0)

    SendEventByName(eventTarget=GameObject(var "Rock Point"), sendEvent="START", delay=0, everyFrame=false)
    on READY → Start Spin
  }

  state Start Spin {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin Speed", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    FloatAdd(floatVariable=var "Spin Speed", add=-900, everyFrame=true, perSecond=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=true)

  }

    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-1080, space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Weapon Spin")
    ActivateGameObject(gameObject=var "Hit Spin", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on DECEL → Decel
  }

  state Decel {
    ActivateGameObject(gameObject=var "Hit Spin", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Idle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)

    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.5)
    on RECEL → Start Spin
  }

  state End {
  }

  var Pitch: float = 0
  var Spin Speed: float = 0
  var Start X: float = 0
