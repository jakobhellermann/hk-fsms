#   Grimm_Main_Tent_boss / Grimm Boss/Tele In Effects
#   Grimm_Main_Tent_boss / Grimm Boss/Tele Out Effects
#   Grimm_Nightmare / Grimm Control/Nightmare Grimm Boss/Tele Out Effects
#   Grimm_Nightmare / Grimm Control/Nightmare Grimm Boss/Tele In Effects
#   GG_Grimm / Grimm Scene/Grimm Boss/Tele Out Effects
#   GG_Grimm / Grimm Scene/Grimm Boss/Tele In Effects
#   GG_Grimm_Nightmare / Grimm Control/Nightmare Grimm Boss/Tele Out Effects
#   GG_Grimm_Nightmare / Grimm Control/Nightmare Grimm Boss/Tele In Effects

fsm Play {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Flame Pillar", storeResult=var "Flame Pillar")
    FindChild(gameObject=Self, childName="tele_smoke_back", storeResult=var "Smoke BG")
    FindChild(gameObject=Self, childName="tele_smoke_front", storeResult=var "Smoke FG")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    on PLAY → Play
  }

  state Play {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetMeshRenderer(gameObject=var "Flame Pillar", active=true)
    PlayParticleEmitter(gameObject=var "Smoke BG", emit=0)
    PlayParticleEmitter(gameObject=var "Smoke FG", emit=0)
    Tk2dPlayAnimationWithEvents(gameObject=var "Flame Pillar", clipName="Pillar", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End Pause
  }

  state End Pause {
    SetMeshRenderer(gameObject=var "Flame Pillar", active=false)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset
  }

  state Reset {
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  var Flame Pillar: gameObject = <null>
  var Parent: gameObject = <null>
  var Smoke BG: gameObject = <null>
  var Smoke FG: gameObject = <null>
}
