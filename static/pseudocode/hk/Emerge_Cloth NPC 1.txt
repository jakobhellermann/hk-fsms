# Emerge
# 1 location(s):
#   Fungus2_09 / Cloth NPC 1

fsm Emerge {
  start Idle

  state Idle {
    FindChild(gameObject=Self, childName="Burst Dust", storeResult=var "Burst Dust")
    FindChild(gameObject=Self, childName="Burst Rocks", storeResult=var "Burst Rocks")
    FindChild(gameObject=Self, childName="Churn Rocks", storeResult=var "Churn Rocks")
    FindChild(gameObject=Self, childName="Emerge Dust", storeResult=var "Emerge Dust")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    on EMERGE → Pause
  }

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emerge Start
  }

  state Emerge Start {
    PlayParticleEmitter(gameObject=var "Churn Rocks", emit=0)
    PlayParticleEmitter(gameObject=var "Emerge Dust", emit=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Emerge Dust", audioClip=loose_floor_temp (sharedassets82.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    on FINISHED → Emerge
  }

  state Emerge {
    StopParticleEmitter(gameObject=var "Churn Rocks")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Emerge Dust", audioClip=HK_Cloth_Burst02 (sharedassets173.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    StopParticleEmitter(gameObject=var "Emerge Dust")
    PlayParticleEmitter(gameObject=var "Burst Rocks", emit=0)
    PlayParticleEmitter(gameObject=var "Burst Dust", emit=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Emerge Dust", audioClip=cave_in_rock_hit_2 (sharedassets173.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Emerged Pause
  }

  state Emerged Pause {
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Ready
  }

  state Ready {
    SetCollider(gameObject=Self, active=true)
    SetCollider(gameObject=var "Dream Dialogue", active=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="READY", delay=0, everyFrame=false)
  }

  var Burst Dust: gameObject = <null>
  var Burst Rocks: gameObject = <null>
  var Churn Rocks: gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var Emerge Dust: gameObject = <null>
}
