# Lesser Mawlek
# 1 location(s):
#   Abyss_17 / Lesser Mawlek (2)

fsm Lesser Mawlek {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Sleep
  }

  state Sleep {
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest")
    AudioStop(gameObject=Self)
    on ALERT → Wake
    on TOOK DAMAGE → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest To Run", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mawlek_call (sharedassets34.assets), pitchMin=1.15, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Start Run
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run Forward")
    DistanceWalk(gameObject=Self, target=var "Hero", distance=10, speed=5, range=2, changeAnimation=false, spriteFacesRight=true, forwardAnimation="", backAnimation="")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToRun", resetFrame=true, everyFrame=true)
    BoolTest(boolVariable=var "Slash Range", isTrue=→"SLASH", isFalse=(none), everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=1.5, finishEvent=→"SPIT", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=var "Sleep Distance", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SLEEP", everyFrame=true)
    on SLASH → Slash Pause
    on SPIT → Spit Antic
    on SLEEP → Sleep
  }

  state Slash Pause {
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slash Antic
  }

  state Slash Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStop(gameObject=Self)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mawlek_call (sharedassets34.assets), pitchMin=1.35, pitchMax=1.35, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Slash
  }

  state Slash {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandom(gameObject=var "Self", audioClips=[ruin_sentry_sword_1 (sharedassets33.assets), ruin_sentry_sword_2 (sharedassets33.assets), ruin_sentry_sword_3 (sharedassets33.assets)], weights=[1, 1, 0], pitchMin=1, pitchMax=1)
    ActivateGameObject(gameObject=var "Slash Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Slash Speed Crt", floatValue=var "Slash Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Slash Speed Crt", multiplyBy=var "X Scale", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Slash Speed Crt", y=(unset), everyFrame=false)
    on FINISHED → Slash Recover
  }

  state Slash Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Recover 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=0)
    on FINISHED → Slash recover 2
  }

  state Slash recover 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Recover 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=0)
    on FINISHED → Recovery
  }

  state Recovery {
    DistanceWalk(gameObject=Self, target=var "Hero", distance=10, speed=5, range=2, changeAnimation=false, spriteFacesRight=true, forwardAnimation="", backAnimation="")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToRun", resetFrame=true, everyFrame=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run Forward")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Idle
  }

  state Spit Antic {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=20, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    AudioStop(gameObject=Self)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mawlek_big_spit (sharedassets34.assets))
    on FINISHED → Spit
    on CANCEL → Idle
  }

  state Spit {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mini_mawlek_spit (sharedassets34.assets))
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Hero X", float2=var "Self X", operation=Subtract, storeResult=var "Target X", everyFrame=false)
    FloatMultiply(floatVariable=var "Target X", multiplyBy=1.2, everyFrame=false)
    FloatClamp(floatVariable=var "Target X", minValue=var "SpitSpeedMin", maxValue=var "SpitSpeedMax", everyFrame=false)
    SetFloatValue(floatVariable=var "SpitRangeMin", floatValue=var "Target X", everyFrame=false)
    SetFloatValue(floatVariable=var "SpitRangeMax", floatValue=var "Target X", everyFrame=false)
    FloatAdd(floatVariable=var "SpitRangeMin", add=-1.5, everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "SpitRangeMax", add=1.5, everyFrame=false, perSecond=false)
    FlingObjectsFromGlobalPoolVel(gameObject=Vomit Glob NoiseFix (sharedassets34.assets), spawnPoint=var "Shot Origin", position=(0, 0, -0.004), spawnMin=2, spawnMax=4, speedMinX=var "SpitRangeMin", speedMaxX=var "SpitRangeMax", speedMinY=15, speedMaxY=18, originVariationX=0, originVariationY=0)
    on FINISHED → Recovery
  }

  state Start Run {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run Forward")
    on FINISHED → Idle
  }

  var Sleep Distance: float = 30

  var Distance: float = 0
  var Hero X: float = 0
  var Self X: float = 0
  var Slash Speed: float = 40
  var Slash Speed Crt: float = 0
  var SpitRangeMax: float = 0
  var SpitRangeMin: float = 0
  var SpitSpeedMax: float = 10
  var SpitSpeedMin: float = -10
  var Target X: float = 0
  var X Scale: float = 0
  var Alert Range: bool = false
  var Slash Range: bool = false
  var Self: gameObject = <null>
  var Shot Origin: gameObject = Lesser Mawlek (2)/Shot Origin
  var Slash Hit: gameObject = Lesser Mawlek (2)/Slash Hit
  var Spit Effect: gameObject = Lesser Mawlek (2)/Spit Effect
}
