# BG Control
# 1 location(s):
#   Deepnest_East_Hornet / Battle Gate Outskirts (1)

fsm BG Control {
  start Init
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hero_unsheath (sharedassets180.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on BG OPEN → Open
  }

  state Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Open")  // disabled
    PlayParticleEmitter(gameObject=var "Raise Dust", emit=0)  // disabled
    SetCollider(gameObject=Self, active=false)  // disabled
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hero_unsheath (sharedassets180.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Opened
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Closed")
    SetMeshRenderer(gameObject=Self, active=true)
    on BG OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Opened")
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Opened {
    SetCollider(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  var Start Closed: bool = false

  var Camera: gameObject = <null>
  var Raise Dust: gameObject = Battle Gate Outskirts (1)/Raise Dust
  var Self: gameObject = <null>
}
