# Detect Hit
# 2 location(s):
#   Fungus1_32 / Moss Knight B/Wake Box
#   Fungus1_32 / Moss Knight C/Wake Box:0

fsm Detect Hit {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(collideTag="Nail Attack", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)  // disabled
    on TAKE DAMAGE → Distance Check
  }

  state Battle? {
    BoolTest(boolVariable=var "Start Battle", isTrue=→"BATTLE", isFalse=→"WAKE", everyFrame=false)
    on BATTLE → Send Event
    on WAKE → Wake
  }

  state Send Event {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE EARLY START", delay=0, everyFrame=false)
  }

  state Wake {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="WAKE", delay=0, everyFrame=false)
  }

  state Distance Check {
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=8, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Battle?
    on CANCEL → Idle
  }

  var Start Battle: bool = true

  var Distance: float = 0
  var Damage Dealt: int = 0
  var Damager: gameObject = <null>
  var Parent: gameObject = <null>
}
