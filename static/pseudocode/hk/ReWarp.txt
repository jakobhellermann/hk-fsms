fsm ReWarp {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="Baby Centipede", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "Collider")
    on COLLIDE → Send Msg
  }

  state Send Msg {
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="REWARP", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Collider", fsmName="Centipede", variableName="Rewarp", setValue=true, everyFrame=false)
    on FINISHED → Detect
  }

  var Collider: gameObject = <null>
}
