# Battle Control
# 1 location(s):
#   Crossroads_04 / Battle Scene

fsm Battle Control {
  start Pause

  state Detect {
    on START → Music
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activate
  }

  state Start {
    SetIntValue(intVariable=var "Battle Enemies", intValue=7, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    on END → End Wait
  }

  state End Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG DESTROY", delay=0, everyFrame=false)
    FindGameObject(objectName="Giant Fly", withTag="Untagged", store=var "Giant Fly")
    DestroyObject(gameObject=var "Giant Fly", delay=0, detachChildren=false)
  }

  state Music {
    on FINISHED → Start
  }

  var Battle Enemies: int = 7
  var Activated: bool = false
  var Giant Fly: gameObject = <null>
}
