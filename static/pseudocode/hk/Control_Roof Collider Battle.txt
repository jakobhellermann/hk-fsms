# Control
# 1 location(s):
#   Ruins1_24 / Roof Collider Battle

fsm Control {
  start Pause

  state Idle {
    on MAGE WINDOW BREAK → Check Hero Y
    on ACTIVATE BATTLE ROOF → Check Hero Y
  }

  state On {
    SetPolygonCollider(gameObject=Self, active=true)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Activate Self?
  }

  state Activate Self? {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero Y", float2=29, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="quakeLevel", storeValue=var "Quake Level")
    IntCompare(integer1=var "Quake Level", integer2=0, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="completedQuakeArea", isTrue=→"FINISHED", isFalse=→"ACTIVATE")
    on FINISHED → Idle
    on ACTIVATE → On
  }

  state Check Hero Y {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=24, tolerance=0, equal=(none), lessThan=→"ACTIVATE", greaterThan=(none), everyFrame=true)
    on ACTIVATE → On
  }

  var Hero Y: float = 0
  var Quake Level: int = 0
}
