# Break
# 1 location(s):
#   Ruins1_24_boss / mage_window

fsm Break {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="Rain", storeResult=var "Rain")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="mageLordDefeated", isTrue=→"ACTIVATE", isFalse=(none))
    FindChild(gameObject=Self, childName="Pt Glass S", storeResult=var "Pt Glass S")
    FindChild(gameObject=Self, childName="Pt Glass L", storeResult=var "Pt Glass L")
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Idle {
    on MAGE WINDOW BREAK → Break
  }

  state Break {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Rain", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Ground", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Debris", activate=true)
    FlingObjects(containerObject=var "Debris", adjustPosition=(7, 1.5, 0), randomisePosition=true, speedMin=0, speedMax=20, angleMin=10, angleMax=170)
    PlayParticleEmitter(gameObject=var "Pt Glass L", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Glass S", emit=0)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    ActivateGameObject(gameObject=var "Break Ground", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rain", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Mask", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Break Ground: gameObject = mage_window/break_ground
  var Debris: gameObject = mage_window/Debris
  var Effects: gameObject = mage_window/Effects
  var Mask: gameObject = mage_window/Mask
  var Pt Glass L: gameObject = <null>
  var Pt Glass S: gameObject = <null>
  var Rain: gameObject = <null>
}
