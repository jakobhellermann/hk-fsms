fsm Prompt Control {
  start Init

  state Init {
    on LORE PROMPT UP → Up
  }

  state Up {
    SetSpriteRenderer(gameObject=Self, active=true)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.141), colorVariable=var "Colour", time=0.5, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetScale(gameObject=Self, vector=(unset), x=2.75, y=2.75, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(3.15, 3.15, 3.15), time=0.5, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on LORE PROMPT DOWN → Down
    on FINISHED → Upped
  }

  state Down {
    SetSpriteRenderer(gameObject=Self, active=true)
    EaseColor(fromValue=(1, 1, 1, 0.141), toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=0.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetScale(gameObject=Self, vector=(unset), x=3.15, y=3.15, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(2.75, 2.75, 2.75), time=0.5, delay=0, speed=(unset), easeType=easeInSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on LORE PROMPT UP → Up
    on FINISHED → Downed
  }

  state Upped {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0.141), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=3.15, y=3.15, z=(unset), everyFrame=false, lateUpdate=false)
    on LORE PROMPT DOWN → Down
  }

  state Downed {
    SetSpriteRenderer(gameObject=Self, active=false)
    on LORE PROMPT UP → Up
  }

  var Colour: color = (0, 0, 0, 1)
}
