# recoil
# 1 location(s):
#   Dream_03_Infected_Knight / Infected Knight Dream

// uses template: recoil
fsm recoil {
  start Initiate
  on CANCEL RECOIL → Idle  // from any state

  state Idle {
    SetBoolValue(boolVariable=var "Recoiling", boolValue=false, everyFrame=false)
    on BEGIN RECOIL → Receive Damager Parameters
    on RECOIL L → Recoil Left
    on RECOIL R → Recoil Right
    on RECOIL U → No Recoil Up?
    on RECOIL D → Recoil Down
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Receive Damager Parameters {
    GetFsmFloat(gameObject=var "Self", fsmName="health_manager_enemy", variableName="Attack Direction", storeValue=var "Attack Direction", everyFrame=false)
    SetBoolValue(boolVariable=var "Recoiling", boolValue=true, everyFrame=false)
    GetFsmInt(gameObject=var "Self", fsmName="health_manager_enemy", variableName="Attack Type", storeValue=var "Attack Type", everyFrame=false)
    GetFsmFloat(gameObject=var "Self", fsmName="health_manager_enemy", variableName="Attack Magnitude", storeValue=var "Attack Magnitude", everyFrame=false)
    BoolTest(boolVariable=var "Freeze In Place", isTrue=→"TRUE", isFalse=(none), everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on DOWN → Recoil Down
    on LEFT → Recoil Hero Right
    on RIGHT → Recoil Hero Left
    on UP → No Recoil Up?
    on TRUE → Freeze In Place
  }

  state Recoil Right {
    SetFloatValue(floatVariable=var "Modified Recoil per second", floatValue=var "Recoil per second", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatDivide(floatVariable=var "Modified Recoil per second", divideBy=50, everyFrame=false)
    TranslateContinuous(gameObject=Self, x=var "Modified Recoil per second", y=(unset), layerMask=[])
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SendEvent(eventTarget=GameObject(var "Self"), sendEvent=→"RECOIL HORIZONTAL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="HIT RIGHT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Recoil Left {
    SetFloatValue(floatVariable=var "Modified Recoil per second", floatValue=var "Recoil per second", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatDivide(floatVariable=var "Modified Recoil per second", divideBy=50, everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=-1, everyFrame=false)
    TranslateContinuous(gameObject=Self, x=var "Modified Recoil per second", y=(unset), layerMask=[])
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SendEvent(eventTarget=GameObject(var "Self"), sendEvent=→"RECOIL HORIZONTAL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="HIT LEFT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Recoil Up Normal {
    SetFloatValue(floatVariable=var "Modified Recoil per second", floatValue=var "Recoil per second", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatDivide(floatVariable=var "Modified Recoil per second", divideBy=50, everyFrame=false)
    TranslateContinuous(gameObject=Self, x=(unset), y=var "Modified Recoil per second", layerMask=[])
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Recoil Down {
    SetFloatValue(floatVariable=var "Modified Recoil per second", floatValue=var "Recoil per second", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatDivide(floatVariable=var "Modified Recoil per second", divideBy=50, everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=-1, everyFrame=false)
    TranslateContinuous(gameObject=Self, x=(unset), y=var "Modified Recoil per second", layerMask=[])
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="HIT DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Freezes? {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="HIT UP", delay=0, everyFrame=false)
    BoolTest(boolVariable=(unset), isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Recoil Up Freeze X
    on FALSE → Recoil Up Normal
  }

  state Recoil Up Freeze X {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SetFloatValue(floatVariable=var "Modified Recoil per second", floatValue=var "Recoil per second", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatDivide(floatVariable=var "Modified Recoil per second", divideBy=50, everyFrame=false)
    TranslateContinuous(gameObject=Self, x=(unset), y=var "Modified Recoil per second", layerMask=[])
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Freeze In Place {
    BoolTest(boolVariable=var "Stop Freezing", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=0, y=0, everyFrame=true)
    SendEventByName(eventTarget=GameObjectFSM(var "Self", fsm="Climber Control"), sendEvent="FREEZE IN PLACE", delay=0, everyFrame=false)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Recoil Hero Left {
    IntCompare(integer1=var "Attack Type", integer2=0, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    SendMessage(gameObject=var "", delivery=SendMessage, options=1, functionCall=RecoilLeft())
    on FINISHED → Recoil Right
  }

  state Recoil Hero Right {
    IntCompare(integer1=var "Attack Type", integer2=0, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    SendMessage(gameObject=var "", delivery=SendMessage, options=1, functionCall=RecoilRight())
    on FINISHED → Recoil Left
  }

  state Recoil Right N {
    SetFloatValue(floatVariable=var "Modified Recoil per second", floatValue=var "Recoil per second", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=var "Attack Magnitude", everyFrame=false)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "Modified Recoil per second", y=(unset), vector3=(unset), everyFrame=true)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SendEvent(eventTarget=GameObject(var "Self"), sendEvent=→"RECOIL HORIZONTAL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="HIT RIGHT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Recoil Left N {
    SetFloatValue(floatVariable=var "Modified Recoil per second", floatValue=var "Recoil per second", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatMultiply(floatVariable=var "Modified Recoil per second", multiplyBy=-1, everyFrame=false)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "Modified Recoil per second", y=(unset), vector3=(unset), everyFrame=true)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SendEvent(eventTarget=GameObject(var "Self"), sendEvent=→"RECOIL HORIZONTAL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="HIT RIGHT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state No Recoil Up? {
    BoolTest(boolVariable=var "No Recoil Up", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Idle
    on FALSE → Freezes?
  }

  var Recoil Time: float = 0.15
  var Recoil per second: float = 15
  var Freeze In Place: bool = false
  var Freeze Recoil Up: bool = true
  var No Recoil Up: bool = false

  var Attack Direction: float = 0
  var Attack Magnitude: float = 0
  var Modified Recoil per second: float = 0
  var Attack Type: int = 0
  var Recoiling: bool = false
  var Stop Freezing: bool = false
  var Self: gameObject = <null>
}
