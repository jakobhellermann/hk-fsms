// uses template: minebridge_plank_fall
fsm FSM {
  start Idle

  state Idle {
    on BREAK 1 → Fall
  }

  state Fall {
    RandomFloat(min=-80, max=80, storeResult=var "Torque")
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Torque", everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    Wait(time=10, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Torque: float = 0
}
