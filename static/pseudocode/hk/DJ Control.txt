# DJ Control
# 1 location(s):
#   Abyss_21 / Shiny Item DJ

fsm DJ Control {
  start Init

  state Init {
    FindGameObject(objectName="Post_Double_Jump", withTag="Untagged", store=var "Post_Double_Jump")
    FindGameObject(objectName="Pre_Double_Jump", withTag="Untagged", store=var "Pre_Double_Jump")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Check Superdashing
  }

  state Take Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SetPlayerDataBool(boolName="hasDoubleJump", value=true)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    SetPlayerDataBool(boolName="disablePause", value=true)
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=shiny_item_pickup (resources.assets))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Getting
  }

  state Getting {
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Cutscene", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Vector3Add(vector3Variable=var "Hero Pos", addVector=(0, -0.5, 0), everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Bugs Get", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Magical 1")
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Wave Get", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="GET", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Bugs Burst", emit=0)
    PlayParticleEmitter(gameObject=var "Bugs Get", emit=0)
    Wait(time=1.45, finishEvent=→"FINISHED", realTime=false)
    StopParticleEmitter(gameObject=var "Bugs Idle")
    SetSpriteRenderer(gameObject=Self, active=false)
    DestroyObject(gameObject=var "Bugs Idle", delay=0, detachChildren=false)
    on FINISHED → Get1
  }

  state Get1 {
    SetParent(gameObject=var "Cutscene", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Cutscene", vector=(unset), x=var "Hero X Scale", y=1, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Cutscene", clipName="DJ Get", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SetMeshRenderer(gameObject=var "Cutscene", active=true)
    on FINISHED → Get2
  }

  state Fall {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=var "Hero", frame=9)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=var "Hero Land Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SetMeshRenderer(gameObject=var "Cutscene", active=false)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="DJ Get Land")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Infected Crossroad
  }

  state Msg {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="brettaState")
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=_0009_emperors_wings (sharedassets345.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Double Jump", everyFrame=false)
    on GET ITEM MSG END → Rise
  }

  state Return Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SetPlayerDataBool(boolName="disablePause", value=false)
  }

  state Get2 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WIND END", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Post_Double_Jump", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pre_Double_Jump", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hero_fury_charm_burst (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=var "White Wave Get2", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "White Flash2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Wave Get2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessageByTag(tag="WindyGrass", message="SetNotWindy")
    on FINISHED → Get End
  }

  state Get End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Cutscene", clipName="DJ Get End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fall
  }

  state Set Infected Crossroad {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="crossroadsInfected", isTrue=→"FINISHED", isFalse=(none))
    SetPlayerDataBool(boolName="crossroadsInfected", value=true)
    SetPlayerDataBool(boolName="visitedCrossroads", value=false)
    on FINISHED → Msg
  }

  state Rise {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Return Control
  }

  state Check Superdashing {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetCState", parameters=["superDashing"], storeResult=var "Is Superdashing")
    BoolTest(boolVariable=var "Is Superdashing", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Idle
    on FALSE → State 1
  }

  state State 1 {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Take Control
  }

  var Hero Land Y: float = 230.6

  var Hero X Scale: float = 0
  var Hero Y: float = 0
  var Guardians Defeated: int = 0
  var Is Superdashing: bool = false
  var Hero Pos: vector3 = (0, 0, 0)
  var Bugs Burst: gameObject = Shiny Item DJ/Bugs Burst
  var Bugs Get: gameObject = Shiny Item DJ/Bugs Get
  var Bugs Idle: gameObject = Shiny Item DJ/Bugs Idle
  var Cutscene: gameObject = Shiny Item DJ/Knight Cutscene
  var Post_Double_Jump: gameObject = <null>
  var Pre_Double_Jump: gameObject = <null>
  var UI Msg: gameObject = <null>
  var White Flash: gameObject = Shiny Item DJ/White Flash
  var White Flash2: gameObject = <null>
  var White Wave Get: gameObject = Shiny Item DJ/White Wave Get
  var White Wave Get2: gameObject = Shiny Item DJ/White Wave Get
}
