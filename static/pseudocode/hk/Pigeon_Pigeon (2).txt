fsm Pigeon {
  start Set Size

  state Set Anim {
    SendRandomEvent(events=[(none), (none), (none)], weights=[1, 1, 1], delay=0)
    on 1 → 1
    on 2 → 2
    on 3 → 3
  }

  state 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 01")
    on FINISHED → Set Frame
  }

  state 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 02")
    on FINISHED → Set Frame
  }

  state 3 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 03")
    on FINISHED → Set Frame
  }

  state Idle {
    BoolTest(boolVariable=var "Hero Range", isTrue=→"PIGEON FLY", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Enemy Range", isTrue=→"PIGEON FLY", isFalse=(none), everyFrame=true)
    on PIGEON FLY → Fly
    on HERO CAST SPELL → Check
  }

  state Fly {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[pigeon_fly_away (sharedassets128.assets), pigeon_fly_away_2 (sharedassets6.assets), pigeon_fly_away_3 (sharedassets6.assets)], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1)
    SendEventByName(eventTarget=GameObject(var "Waker"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.5, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    RandomFloat(min=10, max=35, storeResult=var "Rise Force")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on 1 → Right
    on 2 → Left
  }

  state Right {
    RandomFloat(min=35, max=75, storeResult=var "Side Force")
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "Side Force", y=var "Rise Force", vector3=(unset), everyFrame=true)
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Destroy
  }

  state Set Size {
    GetOwner(storeGameObject=var "Self")
    RandomFloat(min=0.8, max=1, storeResult=var "Size")
    SetScale(gameObject=Self, vector=(unset), x=var "Size", y=var "Size", z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Set Anim
  }

  state Set Frame {
    RandomInt(min=0, max=41, storeResult=var "Start Frame", inclusiveMax=true)
    Tk2dPlayFrame(gameObject=Self, frame=var "Start Frame")
    SendRandomEvent(events=[→"FINISHED", →"NEG SCALE"], weights=[1, 1], delay=0)
    on FINISHED → Idle
    on NEG SCALE → Invert Scale
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Invert Scale {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Left {
    RandomFloat(min=-75, max=-35, storeResult=var "Side Force")
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "Side Force", y=var "Rise Force", vector3=(unset), everyFrame=true)
    FaceAngle(gameObject=Self, angleOffset=180, everyFrame=true)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetScale(gameObject=Self, scale=(-1, 1, 1), everyframe=false)
    on FINISHED → Destroy
  }

  state Check {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=60, tolerance=0, equal=→"PIGEON FLY", lessThan=→"PIGEON FLY", greaterThan=(none), everyFrame=false)
    on PIGEON FLY → Fly
    on CANCEL → Idle
  }

  var Distance: float = 0
  var Rise Force: float = 0
  var Side Force: float = 0
  var Size: float = 0
  var Start Frame: int = 0
  var Enemy Range: bool = false
  var Hero Range: bool = false
  var Self: gameObject = <null>
  var Waker: gameObject = Pigeon (2)/Waker
}
