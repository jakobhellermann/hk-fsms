# Control
# 1 location(s):
#   Deepnest_East_07 / Col Drop Corpse 02/Colosseum Drop Corpse

fsm Control {
  start Init

  state Init {
    SelectRandomString(strings=[" 1", " 2", " 3"], weights=[1, 1, 1], storeString=var "Number")  // disabled
    BuildString(stringParts=["Corpse Fall", var "Number"], separator="", addToEnd=true, storeResult=var "Fall Anim", everyFrame=false)  // disabled
    BuildString(stringParts=["Corpse Land", var "Number"], separator="", addToEnd=true, storeResult=var "Land Anim", everyFrame=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Fall Anim")  // disabled
    GetOwner(storeGameObject=var "Self")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-40, everyFrame=false)
    on FINISHED → Falling
  }

  state Falling {
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"LAND", storeCollider=var "", storeForce=(unset))
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Land Anim")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 0], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Bone", emit=0)
  }

  var Land Anim: string = "Corpse Land 3"

  var Fall Anim: string = ""
  var Number: string = ""
  var Bone: gameObject = Col Drop Corpse 01/Colosseum Drop Corpse/Bone (sharedassets311.assets)
  var Self: gameObject = <null>
}
