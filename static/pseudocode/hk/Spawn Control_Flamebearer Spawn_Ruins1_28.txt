# Spawn Control
# 3 location(s):
#   Ruins1_28 / Flamebearer Spawn
#   Fungus1_10 / Flamebearer Spawn
#   Mines_10 / Flamebearer Spawn

fsm Spawn Control {
  start Pause

  state Pause {
    GetOwner(storeGameObject=var "Self")
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Extra Wait
  }

  state State {
    BoolTest(boolVariable=var "Activated", isTrue=→"KILLED", isFalse=(none), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="flamesCollected", storeValue=var "Flames Collected")
    GetPlayerDataInt(gameObject=var "GameManager", intName="flamesRequired", storeValue=var "Flames Required")
    IntCompare(integer1=var "Flames Collected", integer2=var "Flames Required", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_40", isTrue=(none), isFalse=→"NOT EQUIPPED")
    GetPlayerDataInt(gameObject=var "GameManager", intName="grimmChildLevel", storeValue=var "Charm Level")
    IntCompare(integer1=var "Charm Level", integer2=var "Level", equal=(none), lessThan=→"KILLED", greaterThan=→"KILLED", everyFrame=false)
    IntSwitch(intVariable=var "Charm Level", compareTo=[1, 2, 3, 4], sendEvent=[→"LEVEL 1", →"LEVEL 2", →"LEVEL 3", →"LEVEL 4"], everyFrame=false)
    on NOT EQUIPPED → Do Nothing
    on LEVEL 1 → Level 1
    on LEVEL 2 → Level 2
    on LEVEL 3 → Level 3
    on LEVEL 4 → Do Nothing
    on FINISHED → Do Nothing
    on KILLED → Do Nothing
  }

  state Do Nothing {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Level 1 {
    CreateObject(gameObject=var "Grimmkin Obj", spawnPoint=var "Spawn Point Small", position=(0, 1, 0), rotation=(unset), storeObject=var "Flamebearer", networkInstantiate=false, networkGroup=0)
    on FINISHED → Ready
  }

  state Ready {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GRIMMKIN SPAWN", delay=0, everyFrame=false)
    on ALERT → Start Fight
  }

  state Start Fight {
    SetSpriteRenderer(gameObject=Self, active=false)
    FindGameObject(objectName="", withTag="Grimmchild", store=var "Grimmchild")
    ActivateGameObject(gameObject=var "Active Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Flamebearer"), sendEvent="START", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    on FLAMEBEARER KILLED → Get Pause
  }

  state Get Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flame Ring
  }

  state Flame Ring {
    SetParent(gameObject=var "Get Flame", parent=var "Grimmchild", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetFsmGameObject(gameObject=var "Get Flame", fsmName="Follow Grimmchild", variableName="Grimmchild", setValue=var "Grimmchild", everyFrame=false)
    ActivateGameObject(gameObject=var "Get Flame", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=var "Get Flame", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    PlayParticleEmitter(gameObject=var "Get Flame", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Grimmchild", audioClip=nightmare_lantern_powering_up (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Get
  }

  state Get {
    SendMessage(gameObject=var "Grimmchild", delivery=SendMessage, options=1, functionCall=FlashGrimmflame())
    SetParent(gameObject=var "Get Flash", parent=var "Grimmchild", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    ActivateGameObject(gameObject=var "Get Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPositionToObject(gameObject=var "Get Flash", targetObject=var "Grimmchild", xOffset=0, yOffset=0, zOffset=0)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="flamesCollected")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=AddToFlameList())
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Relic Get Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg")
    FindChild(gameObject=var "Msg", childName="Text", storeResult=var "Msg Text")
    FindChild(gameObject=var "Msg", childName="Icon", storeResult=var "Msg Icon")
    SetSpriteRendererSprite(gameObject=var "Msg Icon", sprite=charm_grimmkin_01_upgrade (resources.assets))
    GetLanguageString(sheetName="UI", convName="FLAME_COLLECTED", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Grimmchild", audioClip=nightmare_lantern_flame_burst (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=0.75, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="SaveLevelState", parameters=[], storeResult=(unset var))
  }

  state Level 2 {
    CreateObject(gameObject=var "Grimmkin Obj", spawnPoint=var "Spawn Point Med", position=(0, 1, 0), rotation=(unset), storeObject=var "Flamebearer", networkInstantiate=false, networkGroup=0)
    on FINISHED → Ready
  }

  state Level 3 {
    CreateObject(gameObject=var "Grimmkin Obj", spawnPoint=var "Spawn Point Med", position=(0, 2, 0), rotation=(unset), storeObject=var "Flamebearer", networkInstantiate=false, networkGroup=0)
    on FINISHED → Ready
  }

  state Extra Wait {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State
  }

  var Level: int = 1
  var Grimmkin Obj: gameObject = Flamebearer Small (sharedassets6.assets)

  var Charm Level: int = 0
  var Flames Collected: int = 0
  var Flames Required: int = 0
  var Activated: bool = false
  var Item Name: string = ""
  var Active Effects: gameObject = Flamebearer Spawn/Active Effects
  var Flamebearer: gameObject = <null>
  var Get Flame: gameObject = Flamebearer Spawn/Get Flame
  var Get Flash: gameObject = Flamebearer Spawn/Get Flash
  var Grimmchild: gameObject = <null>
  var Msg: gameObject = <null>
  var Msg Icon: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Self: gameObject = <null>
  var Spawn Point Med: gameObject = Flamebearer Spawn/Spawn Point Med
  var Spawn Point Small: gameObject = Flamebearer Spawn/Spawn Point Small
}
