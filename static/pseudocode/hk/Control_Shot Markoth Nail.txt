# Control
# 1 location(s):
#   sharedassets314.assets / Shot Markoth Nail

fsm Control {
  start Init
  on GHOST DEATH → Dissipate  // from any state

  state Init {
    GameObjectIsNull(gameObject=var "Hero", isNull=→"CANCEL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Set Pos
    on CANCEL → Dissipate
  }

  state Set Pos {
    RandomFloat(min=1, max=var "X Max", storeResult=var "Tele X")
    RandomFloat(min=1, max=var "Y Max", storeResult=var "Tele Y")
    SetVector3XYZ(vector3Variable=var "Tele Pos", vector3Value=(unset), x=var "Tele X", y=var "Tele Y", z=0, everyFrame=false)
    on FINISHED → Check Distance
  }

  state Check Distance {
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Hero Pos", point2=var "Tele Pos", ignoreZ=true, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=6, tolerance=0, equal=(none), lessThan=→"RESET", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Distance", float2=10, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"RESET", everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Tele Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on RESET → Set Pos
    on FINISHED → Antic Point
  }

  state Antic Point {
    GameObjectIsNull(gameObject=var "Hero", isNull=→"CANCEL", isNotNull=(none), storeResult=(unset), everyFrame=true)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=-0.5, storeAngle=var "Angle", everyFrame=true)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=true, lateUpdate=false)
    FloatAdd(floatVariable=var "Angle", add=180, everyFrame=true, perSecond=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=5, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[mage_knight_teleport (sharedassets6.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Antic
    on CANCEL → Dissipate
  }

  state Antic {
    Decelerate(gameObject=Self, deceleration=0.9)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    on FINISHED → Fire
  }

  state Fire {
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=30, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[mage_knight_sword (sharedassets34.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[mage_appear (sharedassets34.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Dissipate {
    SetSpriteRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Idle Pt")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
  }

  var Angle: float = 0
  var Distance: float = 0
  var Hero X: float = 0
  var Hero Y: float = 0
  var Ran Float: float = 0
  var Tele X: float = 0
  var Tele Y: float = 0
  var X Max: float = 43
  var Y Max: float = 24
  var Hero Pos: vector3 = (0, 0, 0)
  var Tele Pos: vector3 = (0, 0, 0)
  var Appear Pt: gameObject = <null>
  var Idle Pt: gameObject = Shot Markoth Nail/Idle Pt
  var Self: gameObject = <null>
}
