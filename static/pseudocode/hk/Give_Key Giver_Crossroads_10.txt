# Give
# 1 location(s):
#   Crossroads_10 / Key Giver

fsm Give {
  start Pause

  state Pause {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="falseKnightDefeated", isTrue=→"FK DEATH", isFalse=(none))
    on FK DEATH → Activate
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  var Self: gameObject = <null>
}
