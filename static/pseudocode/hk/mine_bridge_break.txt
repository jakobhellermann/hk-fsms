// uses template: mine_bridge_break
fsm mine_bridge_break {
  start Idle

  state Idle {
    on SUPERDASH BLAST → Fall
  }

  state Fall {
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    RandomFloat(min=-2, max=2, storeResult=var "X Speed")
    RandomFloat(min=-1, max=5, storeResult=var "Y Speed")
    FloatOperator(float1=var "X Speed", float2=20, operation=Multiply, storeResult=var "Rotation", everyFrame=false)
    FloatMultiply(floatVariable=var "Rotation", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Rotation", everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Rotation: float = 0
  var X Speed: float = 0
  var Y Speed: float = 0
}
