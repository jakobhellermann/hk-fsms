# Encounter
# 1 location(s):
#   Deepnest_East_Hornet_boss / Hornet Outskirts Battle Encounter

fsm Encounter {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="encounteredHornet", isTrue=→"DESTROY", isFalse=(none))
    on FINISHED → Idle
    on DESTROY → Destroy
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Point
  }

  state Point {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Point")
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Idle")
    SetPlayerDataInt(intName="hornetGreenpath", value=4)
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WAKE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dialogue
  }

  state Dialogue {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=Hornet_Dialogue_Generic_02 (sharedassets105.assets))
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["HORNET_OUTSKIRTS_1", "Hornet"], storeResult=(unset var))
    on CONVO_FINISH → Box Down 2
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Hero Active
  }

  state Set Hero Active {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    on FINISHED → Start Fight
  }

  state Destroy {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="START WINDY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BLIZZARD START", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)
    SendMessageByTag(tag="WindyGrass", message="SetWindy")
  }

  state Leap Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Evade Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Leap Back
  }

  state Leap Back {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Evade")
    iTweenMoveBy(gameObject=Self, id="", vector=(-7, 0, 0), time=0.25, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetPlayerDataInt(intName="hornetGreenpath", value=4)
    on FINISHED → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Soft Land")
    on FINISHED → Start Fight
  }

  state Start Fight {
    SetPlayerDataBool(boolName="encounteredHornet", value=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="BOSS HORNET", delay=0, everyFrame=false)
  }

  state Attacked {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on FINISHED → Start Fight
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Turn
  }

  state Turn {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on FINISHED → Idle Pause
  }

  state Idle Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Blizzard Start
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dialogue 2
  }

  state Dialogue 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["HORNET_OUTSKIRTS_2", "Hornet"], storeResult=(unset var))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=Hornet_Dialogue_Generic_04 (sharedassets325.assets))
    on CONVO_FINISH → Box Down
  }

  state Blizzard Start {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DRIFT END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="START WINDY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BLIZZARD START", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BlizzardShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=blizzard_start (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessageByTag(tag="WindyGrass", message="SetWindy")
    on FINISHED → Box Up 2
  }

  var Hornet Encounter: int = 0
  var Area Title: gameObject = <null>
  var Boss: gameObject = <null>
  var Grass: gameObject = Hornet Outskirts Battle Encounter/Grass
  var Grass Escape: gameObject = Hornet Outskirts Battle Encounter/Grass Escape
  var Hornet Saver: gameObject = <null>
  var Self: gameObject = <null>
  var Thread: gameObject = Hornet Outskirts Battle Encounter/Thread
}
