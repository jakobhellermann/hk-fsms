# Control
# 3 location(s):
#   resources.assets / Spell Fluke Dung/Knight Dung Cloud
#   resources.assets / Spell Fluke Dung Lv1/Knight Dung Cloud
#   resources.assets / Spell Fluke Dung Lv2/Knight Dung Cloud

fsm Control {
  start Pause

  state Collider On {
    SetCircleCollider(gameObject=Self, active=true)
    Wait(time=2.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Collider Off
  }

  state Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Collider On
  }

  state Collider Off {
    SetCircleCollider(gameObject=Self, active=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=EmptyDamageList())
  }
}
