# Loose Floor
# 2 location(s):
#   Mines_01 / _Scenery/mine_1_quake_floor/Loose Floor 2
#   Mines_35 / _Scenery/mine_1_quake_floor/Loose Floor 2

fsm Loose Floor {
  start Init
  on BREAK 1 → Break  // from any state

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetFsmBool(gameObject=var "Parent", fsmName="Play Sound", variableName="Floor 2", setValue=false, everyFrame=false)
    Collision2dEvent(collision=1, collideTag="Player", sendEvent=→"ENTER", storeCollider=var "Hero", storeForce=(unset))
    SetParticleEmission(gameObject=var "Dust B", emission=false)
    SetParticleEmission(gameObject=var "Dust F", emission=false)
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"ENTER", storeCollider=var "Hero", storeForce=(unset))
    SetParticleEmission(gameObject=var "Churn Rocks", emission=false)
    on ENTER → Still
  }

  state Still {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetFsmBool(gameObject=var "Parent", fsmName="Play Sound", variableName="Floor 2", setValue=false, everyFrame=false)
    Collision2dEvent(collision=2, collideTag="Player", sendEvent=→"EXIT", storeCollider=var "", storeForce=(unset))
    GetSpeed2d(gameObject=var "Hero", storeResult=var "Hero Speed", everyFrame=true)
    FloatCompare(float1=var "Hero Speed", float2=0, tolerance=0.1, equal=(none), lessThan=→"MOVING", greaterThan=→"MOVING", everyFrame=true)
    SetParticleEmission(gameObject=var "Dust B", emission=false)
    SetParticleEmission(gameObject=var "Dust F", emission=false)
    SetParticleEmission(gameObject=var "Churn Rocks", emission=false)
    on EXIT → Idle
    on MOVING → Moving
  }

  state Moving {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shake")
    SetFsmBool(gameObject=var "Parent", fsmName="Play Sound", variableName="Floor 2", setValue=true, everyFrame=false)
    Collision2dEvent(collision=2, collideTag="Player", sendEvent=→"EXIT", storeCollider=var "", storeForce=(unset))
    GetSpeed2d(gameObject=var "Hero", storeResult=var "Hero Speed", everyFrame=true)
    FloatCompare(float1=var "Hero Speed", float2=0, tolerance=0.1, equal=→"STILL", lessThan=(none), greaterThan=(none), everyFrame=true)
    ObjectJitter(gameObject=var "Bot 3", x=0.03, y=0.03, z=(unset), allowMovement=false)
    ObjectJitter(gameObject=var "Bot 2", x=0.03, y=0.03, z=(unset), allowMovement=false)
    ObjectJitter(gameObject=var "Bot 1", x=0.03, y=0.03, z=(unset), allowMovement=false)
    SetParticleEmission(gameObject=var "Dust B", emission=true)
    SetParticleEmission(gameObject=var "Dust F", emission=true)
    SetParticleEmission(gameObject=var "Churn Rocks", emission=true)
    on EXIT → Moving Cooldown
    on STILL → Moving Cooldown
  }

  state Init {
    FindChild(gameObject=Self, childName="Dust B", storeResult=var "Dust B")
    FindChild(gameObject=Self, childName="Churn Rocks", storeResult=var "Churn Rocks")
    FindChild(gameObject=Self, childName="Dust F", storeResult=var "Dust F")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Effect Centre", storeResult=var "Effect Centre")
    FindChild(gameObject=var "Parent", childName="Bot 1", storeResult=var "Bot 1")
    FindChild(gameObject=var "Parent", childName="Bot 2", storeResult=var "Bot 2")
    FindChild(gameObject=var "Parent", childName="Bot 3", storeResult=var "Bot 3")
    on FINISHED → Idle
  }

  state Break {
    SetFsmBool(gameObject=var "Parent", fsmName="Play Sound", variableName="Floor 2", setValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetParticleEmission(gameObject=var "Dust B", emission=false)
    SetParticleEmission(gameObject=var "Dust F", emission=false)
  }

  state Moving Cooldown {
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    ObjectJitter(gameObject=var "Bot 3", x=0.03, y=0.03, z=(unset), allowMovement=false)
    ObjectJitter(gameObject=var "Bot 2", x=0.03, y=0.03, z=(unset), allowMovement=false)
    ObjectJitter(gameObject=var "Bot 1", x=0.03, y=0.03, z=(unset), allowMovement=false)
    on FINISHED → Idle
  }

  var Hero Speed: float = 0
  var Bot 1: gameObject = <null>
  var Bot 2: gameObject = <null>
  var Bot 3: gameObject = <null>
  var Churn Rocks: gameObject = <null>
  var Dust B: gameObject = <null>
  var Dust F: gameObject = <null>
  var Effect Centre: gameObject = <null>
  var Hero: gameObject = <null>
  var Parent: gameObject = <null>
}
