# Dust
# 1 location(s):
#   Crossroads_10 / Fk Break Wall/Broken/Slope

fsm Dust {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Dust 1", storeResult=var "Dust 1")
    FindChild(gameObject=Self, childName="Dust 2", storeResult=var "Dust 2")
    FindChild(gameObject=Self, childName="Rock Point", storeResult=var "Rock Point")
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"HIT", storeCollider=var "", storeForce=(unset))
    on HIT → Effects
  }

  state Effects {
    PlayParticleEmitter(gameObject=var "Dust 1", emit=0)
    PlayParticleEmitter(gameObject=var "Dust 2", emit=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rocks
  }

  state Rocks {
    SpawnRandomObjectsV2(gameObject=Particle Rock Tiny Transient (resources.assets), spawnPoint=var "Rock Point", position=(unset), spawnMin=7, spawnMax=7, speedMin=0, speedMax=0, angleMin=0, angleMax=0, originVariationX=(unset), originVariationY=(unset))
  }

  var Dust 1: gameObject = <null>
  var Dust 2: gameObject = <null>
  var Rock Point: gameObject = <null>
}
