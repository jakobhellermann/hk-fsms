// uses template: constrain_x
fsm constrain_x {
  start Idle
  on END CONSTRAIN R → Constrain R Only  // from any state
  on END CONSTRAIN → Inert  // from any state
  on END CONSTRAIN L → Constrain L Only  // from any state
  on START CONSTRAIN → Idle  // from any state

  state Idle {
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=var "Edge L", tolerance=0, equal=(none), lessThan=→"CONSTRAIN L", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=var "Edge R", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CONSTRAIN R", everyFrame=true)
    on CONSTRAIN L → Constrain L
    on CONSTRAIN R → Constrain R
  }

  state Constrain L {
    SetPosition(gameObject=Self, vector=(unset), x=var "Edge L", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    GotoPreviousState()
  }

  state Constrain R {
    SetPosition(gameObject=Self, vector=(unset), x=var "Edge R", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    GotoPreviousState()
  }

  state Inert {
  }

  state Constrain L Only {
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=var "Edge R", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CONSTRAIN R", everyFrame=true)
    on CONSTRAIN R → Constrain R
  }

  state Constrain R Only {
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=var "Edge R", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CONSTRAIN R", everyFrame=true)
    on CONSTRAIN L → Constrain L
  }

  var Edge L: float = 73.35
  var Edge R: float = 115.703

  var X Pos: float = 0
}
