#   Abyss_19 / Parasite Balloon (3)

fsm Control {
  start Initiate

  state Chase - In Sight {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Chase")
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=5, accelerationForce=10, offsetX=0, offsetY=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToChase", resetFrame=false, everyFrame=true)
    on WAIT → 
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    FindChild(gameObject=Self, childName="Hero Saver", storeResult=var "Hero Saver Obj")
    FindChild(gameObject=Self, childName="Spawn Range", storeResult=var "Spawn Range Obj")
    RandomFloat(min=1, max=1.25, storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Spawn Wait
  }

  state Spawn Wait {
    SetCollider(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on SPAWN → Spawn Pause
  }

  state Spawn Pause {
    WaitRandom(timeMin=0, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Saver
  }

  state Check Saver {
    BoolTest(boolVariable=var "Hero Saver", isTrue=(none), isFalse=→"FINISHED", everyFrame=true)
    on FINISHED → Spawn
  }

  state Spawn {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_balloon_appear (sharedassets34.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Chase - In Sight
  }

  var Spawns: bool = true

  var Attention Span: float = 6
  var Scale: float = 0
  var Can See Hero: bool = false
  var Hero Saver: bool = false
  var In Alert Range: bool = false
  var Start Alert: bool = false
  var Startles: bool = false
  var Alerter: gameObject = <null>
  var Hero: gameObject = <null>
  var Hero Saver Obj: gameObject = <null>
  var Self: gameObject = <null>
  var Spawn Range Obj: gameObject = <null>
}
