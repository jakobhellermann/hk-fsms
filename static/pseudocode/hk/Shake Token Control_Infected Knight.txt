# Shake Token Control
# 2 location(s):
#   Abyss_19 / Infected Knight
#   GG_Broken_Vessel / Infected Knight

fsm Shake Token Control {
  start Idle

  state Idle {
    on TOOK DAMAGE → Pause
  }

  state Check {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Token 1", equal=→"GET", lessThan=→"GET", greaterThan=(none), everyFrame=false)
    on GET → Real Set 1
    on FINISHED → Idle
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Set 1 {
    on TOOK DAMAGE → Pause 2
  }

  state Check 2 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Token 2", equal=→"GET", lessThan=→"GET", greaterThan=(none), everyFrame=false)
    on GET → Real Set 2
    on FINISHED → Set 1
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check 2
  }

  state Set 2 {
    on TOOK DAMAGE → Pause 3
  }

  state Check 3 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Token 3", equal=→"GET", lessThan=→"GET", greaterThan=(none), everyFrame=false)
    on GET → Set 3
    on FINISHED → Set 2
  }

  state Pause 3 {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check 3
  }

  state Set 3 {
    SetFsmBool(gameObject=Self, fsmName="IK Control", variableName="Do Shake", setValue=true, everyFrame=false)
    on TOOK DAMAGE → 
  }

  state Real Set 1 {
    SetFsmBool(gameObject=Self, fsmName="IK Control", variableName="Do Shake", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="IK Control", variableName="Angry", setValue=true, everyFrame=false)
    on FINISHED → Set 1
  }

  state Real Set 2 {
    SetFsmBool(gameObject=Self, fsmName="IK Control", variableName="Do Shake", setValue=true, everyFrame=false)
    on FINISHED → Set 2
  }

  var Token 1: int = 370
  var Token 2: int = 220
  var Token 3: int = 110

  var HP: int = 0
}
