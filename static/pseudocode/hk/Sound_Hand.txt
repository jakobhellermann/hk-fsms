# Sound
# 1 location(s):
#   Room_GG_Shortcut / Fluke Hermit/Hand

fsm Sound {
  start Init

  state Idle {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame ID", everyframe=true)
    IntCompare(integer1=var "Frame ID", integer2=11, equal=→"PLAY", lessThan=(none), greaterThan=(none), everyFrame=true)
    on PLAY → Play
  }

  state Play {
    RandomFloat(min=1.1, max=1.4, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=enemy_damage_over_time (resources.assets))
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Init {
    on FINISHED → Idle
  }

  var Chooser: float = 0
  var Pitch: float = 0
  var Frame ID: int = 0
  var Dust: gameObject = <null>
  var Pt Rocks: gameObject = Miner/Pt Rocks (sharedassets471.assets)
  var Self: gameObject = <null>
}
