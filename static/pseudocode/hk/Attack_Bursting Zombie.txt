# Attack
# 1 location(s):

// component disabled: this FSM does not run until something enables it
fsm Attack {
  start Init
  on TOOK DAMAGE → Explode  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Charge Audio", storeResult=var "Chareg Audio")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ATTACK", isFalse=(none), everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"EXPLODE", storeCollider=var "")
    on ATTACK → Antic
    on EXPLODE → Explode
  }

  state Antic {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"EXPLODE", storeCollider=var "")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grave_Zombie_attack_01, Grave_Zombie_attack_02, Grave_Zombie_attack_03], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Start Run
    on EXPLODE → Explode
  }

  state Start Run {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on L → L
    on R → R
  }

  state L {
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=false)
    on FINISHED → Running
  }

  state R {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=false)
    on FINISHED → Running
  }

  state Running {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"EXPLODE", bottomHitEvent=(none), leftHitEvent=→"EXPLODE", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Chareg Audio", volume=1, oneShotClip=exploding_zombie_charge)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"EXPLODE", bottomHitEvent=(none), leftHitEvent=→"EXPLODE", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"EXPLODE", storeCollider=var "")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Speed", float2=0, tolerance=0.1, equal=→"EXPLODE", lessThan=(none), greaterThan=(none), everyFrame=true)
    on EXPLODE → Explode
  }

  state Explode {
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=20, spawnMax=20, speedMin=15, speedMax=20, angleMin=10, angleMax=170, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")  // disabled
    CreateObject(gameObject=Gas Explosion L, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    InstaDeath(target=Self, direction=var "Speed")
  }

  var Speed: float = 15
  var X Speed: float = 0
  var Alert Range: bool = false
  var Chareg Audio: gameObject = <null>
  var Self: gameObject = <null>
}
