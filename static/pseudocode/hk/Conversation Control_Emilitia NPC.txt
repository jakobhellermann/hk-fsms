# Conversation Control
# 1 location(s):
#   Ruins_House_03 / Emilitia NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=var "Voice Player", startVolume=1, endVolume=0, time=0.3)
    on FINISHED → End
  }

  state Box Up {
    SetFsmBool(gameObject=Self, fsmName="Anim", variableName="Talking", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    SetFsmBool(gameObject=Self, fsmName="Anim", variableName="Talking", setValue=false, everyFrame=false)
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Meet {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Amelitia_Generic_01 (sharedassets85.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataBool(boolName="metEmilitia", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["EMILITIA_MEET", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Title
  }

  state Convo Choice {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metEmilitia", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasKingsBrand", falseBool="emilitiaKingsBrandConvo", isTrue=→"KINGS BRAND", isFalse=(none))
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"GREET", everyFrame=false)
    on MEET → Meet
    on GREET → Greet
    on REPEAT → Repeat
    on KINGS BRAND → Kings Brand
  }

  state Title {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="EMILITIA_C", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="EMILITIA_NC", everyFrame=false)
    on FINISHED → Box Up
  }

  state Greet {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["EMILITIA_GREET", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Amelitia_Generic_01 (sharedassets85.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Repeat {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["EMILITIA_REPEAT", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Amelitia_Generic_04 (sharedassets85.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Flower?
  }

  state Kings Brand {
    SetPlayerDataBool(boolName="emilitiaKingsBrandConvo", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["EMILITIA_KING_BRAND", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Amelitia_Generic_03 (sharedassets85.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Flower? {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasXunFlower", falseBool="givenEmilitiaFlower", isTrue=→"FLOWER", isFalse=→"FINISHED")  // disabled
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasXunFlower", isTrue=(none), isFalse=→"CANCEL")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="givenEmilitiaFlower", isTrue=→"CANCEL", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFlowerBroken", isTrue=→"CANCEL", isFalse=(none))
    on CANCEL → Talk Finish
    on FINISHED → Box Down 2
  }

  state Flower {
    SetPlayerDataBool(boolName="givenEmilitiaFlower", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["AMELITIA_FLOWER", "CP3"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Amelitia_Generic_04 (sharedassets85.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Decline Pause 2 {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Talk Finish
  }

  state Box Up YN 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text 2
  }

  state Send Text 2 {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["ELDERBUG_FLOWER", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause 2
    on YES → Yes 2
  }

  state Yes 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 2
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=var "Voice Player", startVolume=1, endVolume=0, time=0.3)
    on FINISHED → Box Up YN 2
  }

  state Box Up 2 {
    SetFsmBool(gameObject=Self, fsmName="Anim", variableName="Talking", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flower
  }

  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
