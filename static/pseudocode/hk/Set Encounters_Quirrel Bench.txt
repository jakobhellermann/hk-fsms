# Set Encounters
# 1 location(s):
#   Ruins1_02 / RestBench/Quirrel Bench

fsm Set Encounters {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set
  }

  state Set {
    SetPlayerDataBool(boolName="quirrelCityEncountered", value=true)
    on FINISHED → Leave Mines
  }

  state Leave Mines {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="quirrelMinesEncountered", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="quirrelMinesLeft", value=true)
    on FINISHED → Finished
  }

  state Finished {
  }
}
