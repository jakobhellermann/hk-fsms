// uses template: no_climb_msg
fsm No Climb Msg {
  start Init

  state Idle {
    on NO CLIMB ENTER → Enter
    on NO CLIMB EXIT → Exit
  }

  state Enter {
    SetFsmBool(gameObject=var "Self", fsmName="Chase", variableName="No Climb", setValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  state Exit {
    SetFsmBool(gameObject=var "Self", fsmName="Chase", variableName="No Climb", setValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
}
