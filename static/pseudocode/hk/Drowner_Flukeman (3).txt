# Drowner
# 2 location(s):
#   Waterways_02 / Flukeman (3)
#   Waterways_08 / Flukeman (3)

fsm Drowner {
  start Detect

  state Detect {
    GetOwner(storeGameObject=var "Self")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer="", sendEvent=→"DROWN", storeCollider=var "")
    on DROWN → Drown Start
  }

  state Drown Start {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DROWNING", delay=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetCollider(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=-8, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Drown")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Splash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Self", position=(0, -1, 0), spawnMin=15, spawnMax=15, speedMin=15, speedMax=25, angleMin=80, angleMax=100, originVariationX=0.5, originVariationY=0.5, FSM="", FSMEvent="")
    on FINISHED → Bob Down
  }

  state Bob Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=-1, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Self: gameObject = <null>
  var Splash: gameObject = Flukeman (3)/Splash Out Black
}
