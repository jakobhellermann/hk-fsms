# Mega Jellyfish
# 2 location(s):
#   Fungus3_archive_02_boss / Mega Jellyfish
#   sharedassets466.assets / Mega Jellyfish

fsm Mega Jellyfish {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindGameObject(objectName="", withTag="Boss Attack", store=var "Multizaps")
    FindChild(gameObject=var "Multizaps", childName="Pattern 1", storeResult=var "Pattern 1")
    FindChild(gameObject=var "Multizaps", childName="Pattern 2", storeResult=var "Pattern 2")
    FindChild(gameObject=Self, childName="Quirrel Slash", storeResult=var "Quirrel Slash")
    FindChild(gameObject=Self, childName="Quirrel Land", storeResult=var "Quirrel Land")
    FindChild(gameObject=Self, childName="Pt Wound", storeResult=var "Pt Wound")
    FindChild(gameObject=Self, childName="Strike Nail", storeResult=var "Strike Nail")
    FindChild(gameObject=Self, childName="White Flash", storeResult=var "White Flash")
    FindChild(gameObject=Self, childName="Pt Spore", storeResult=var "Pt Spore")
    FindChild(gameObject=Self, childName="Pt Bub", storeResult=var "Pt Bub")
    FindChild(gameObject=Self, childName="Pt Steam", storeResult=var "Pt Steam")
    FindChild(gameObject=Self, childName="Emerge Spore", storeResult=var "Emerge Spore")
    FindChild(gameObject=Self, childName="Entry Sprite", storeResult=var "Entry Sprite")
    SetParent(gameObject=var "Quirrel Land", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Sleep
  }

  state Idle {
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=6, accelerationForce=6, offsetX=0, offsetY=0)
    WaitRandom(timeMin=1.5, timeMax=2, finishEvent=→"ATTACK", realTime=false)
    FloatAdd(floatVariable=var "Quirrel Timer", add=1, everyFrame=true, perSecond=true)
    on ATTACK → Attack Antic
    on QUIRREL → Set Timer
  }

  state Attack Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    SetAudioClip(gameObject=Self, audioClip=uumuu_is_zapping (sharedassets34.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=3, accelerationForce=3, offsetX=0, offsetY=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Attack", emit=0)
    on FINISHED → Choice
  }

  state Zapping {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.85)
    on FINISHED → Gen
  }

  state Gen {
    IntAdd(intVariable=var "Zaps", add=1, everyFrame=false)
    IntCompare(integer1=var "Zaps", integer2=8, equal=(none), lessThan=(none), greaterThan=→"END", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Mega Jelly Zap (sharedassets229.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Zap")
    on FINISHED → First?
    on END → Attack Recover
  }

  state Choice {
    SetIntValue(intVariable=var "Zaps", intValue=0, everyFrame=false)
    SetBoolValue(boolVariable=var "First Chaser", boolValue=false, everyFrame=false)
    SendRandomEventV2(events=[→"CHASE", →"MULTIZAP"], weights=[1, 1], trackingInts=[var "Ct Chase", var "Ct Multizap"], eventMax=[2, 2])
    on CHASE → Zapping
    on MULTIZAP → Pattern Choice
  }

  state Attack Recover {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Recover")
    SetAudioClip(gameObject=Self, audioClip=uumuu_loop (sharedassets229.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    StopParticleEmitter(gameObject=var "Pt Attack")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Quirrel?
  }

  state First? {
    SetFsmFloat(gameObject=var "Zap", fsmName="Control", variableName="Antic Time", setValue=0.65, everyFrame=false)
    BoolTest(boolVariable=var "First Chaser", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "First Chaser", boolValue=true, everyFrame=false)
    SetFsmFloat(gameObject=var "Zap", fsmName="Control", variableName="Antic Time", setValue=1, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Zapping
  }

  state Multizap {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=zap_bug (sharedassets229.assets), pitchMin=1, pitchMax=1, volume=1.5, delay=0, storePlayer=<null>)
    ActivateAllChildren(gameObject=var "Current Pattern", activate=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.85)
    on FINISHED → Attack Recover
  }

  state Pattern Choice {
    GetRandomChild(gameObject=var "Multizaps", storeResult=var "Current Pattern")
    on FINISHED → Multizap
  }

  state Arrive Pause {
    SetFsmBool(gameObject=var "Quirrel Land", fsmName="Watch", variableName="Roam", setValue=false, everyFrame=false)
    BoolTest(boolVariable=var "Quirrel Arrived", isTrue=→"ATTACK", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Quirrel Arrived", boolValue=true, everyFrame=false)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=3, accelerationForce=3, offsetX=0, offsetY=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Pos
    on ATTACK → Attack Pause
  }

  state Check Pos {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Quirrel Land", audioClip=Quirrel_battle_intro (sharedassets229.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FloatTestToBool(float1=var "Hero X", float2=var "Hero Mid X", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Above Mid X", everyFrame=false)
    FloatTestToBool(float1=var "Hero Y", float2=var "Hero Mid Y", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Above Mid Y", everyFrame=false)
    BoolTestMulti(boolVariables=[var "Above Mid X", var "Above Mid Y"], boolStates=[true, true], trueEvent=→"TR", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Above Mid X", var "Above Mid Y"], boolStates=[true, false], trueEvent=→"BR", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Above Mid X", var "Above Mid Y"], boolStates=[false, true], trueEvent=→"TL", falseEvent=(none), storeResult=(unset), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"BL", delay=0, everyFrame=false)
    on BL → BL
    on BR → BR
    on TL → TL
    on TR → TR
  }

  state BL {
    SetPosition(gameObject=var "Quirrel Land", vector=(unset), x=42.8, y=117.82, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Quirrel Land", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Q Arrive
  }

  state BR {
    SetPosition(gameObject=var "Quirrel Land", vector=(unset), x=62.5, y=117.82, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Quirrel Land", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Q Arrive
  }

  state TL {
    SetPosition(gameObject=var "Quirrel Land", vector=(unset), x=38.63, y=122.92, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Quirrel Land", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Q Arrive
  }

  state TR {
    SetPosition(gameObject=var "Quirrel Land", vector=(unset), x=66.89, y=122.92, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Quirrel Land", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Q Arrive
  }

  state Q Arrive {
    SetMeshRenderer(gameObject=var "Quirrel Land", active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Quirrel Slash", audioClip=uumuu_helper_enter (sharedassets229.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Quirrel Land", animLibName="", clipName="Quirrel Arrive")
    Wait(time=1.5, finishEvent=→"ATTACK", realTime=false)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=3, accelerationForce=3, offsetX=0, offsetY=0)
    on ATTACK → Q Leave
  }

  state Q Leave {
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=3, accelerationForce=3, offsetX=0, offsetY=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Quirrel Slash", audioClip=uumuu_helper_exit (sharedassets229.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Quirrel Land", animLibName="", clipName="Quirrel Leave")
    Tk2dPlayAnimationWithEvents(gameObject=var "Quirrel Land", clipName="Quirrel Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Slash Pause
  }

  state Slash Antic {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Quirrel Slash", audioClips=[Quirrel_battle_shout_01 (sharedassets229.assets), Quirrel_battle_shout_02 (sharedassets229.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Quirrel Slash", audioClip=uumuu_helper_enter (sharedassets229.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    RandomFloat(min=-25, max=25, storeResult=var "Slash Rotation")
    RandomFloatEither(value1=-1, value2=1, storeResult=var "Slash Scale")
    SetRotation(gameObject=var "Quirrel Slash", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Slash Rotation", space=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Quirrel Slash", vector=(unset), x=var "Slash Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=var "Quirrel Slash", active=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Quirrel Slash", clipName="Slash Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Slash
  }

  state Slash {
    Tk2dPlayAnimationWithEvents(gameObject=var "Quirrel Slash", clipName="Slash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Quirrel Slash", audioClip=uumuu_helper_slash (sharedassets229.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=uumuu_wounded (sharedassets229.assets))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wound")
    DecelerateV2(gameObject=Self, deceleration=0.85)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    DestroyComponent(gameObject=Self, component="BounceShroom")
    SetCircleCollider(gameObject=Self, active=false)
    PlayParticleEmitter(gameObject=var "Pt Wound", emit=0)
    ActivateGameObject(gameObject=var "Strike Nail", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wounded
  }

  state Slash Pause {
    Wait(time=0.27, finishEvent=→"FINISHED", realTime=false)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=3, accelerationForce=3, offsetX=0, offsetY=0)
    SetMeshRenderer(gameObject=var "Quirrel Land", active=false)
    on FINISHED → Slash Antic
  }

  state Wounded {
    SetAudioPitch(gameObject=Self, pitch=1.5, everyFrame=false)
    SetMeshRenderer(gameObject=var "Quirrel Slash", active=false)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=2, accelerationForce=-1, offsetX=0, offsetY=0)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SetRecoilSpeed(target=Self, newRecoilSpeed=15)
    on FINISHED → Recover Antic
  }

  state Recover Antic {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Recover Antic")
    PlayParticleEmitter(gameObject=var "Pt Attack", emit=0)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SetAudioPitch(gameObject=Self, pitch=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=uumuu_back_to_normal (sharedassets229.assets))
    on FINISHED → Recover
  }

  state Recover {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Recover")
    SetRecoilSpeed(target=Self, newRecoilSpeed=0)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0)
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Pt Wound", emit=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    StopParticleEmitter(gameObject=var "Pt Attack")
    AddComponent(gameObject=Self, component="BounceShroom", storeComponent=var "", removeOnExit=false)
    on FINISHED → Quirrel Roam
  }

  state Set Timer {
    SetFloatValue(floatVariable=var "Quirrel Timer", floatValue=0, everyFrame=false)
    RandomFloat(min=2, max=4, storeResult=var "Quirrel Time")
    on FINISHED → Arrive Pause
  }

  state Quirrel? {
    FloatCompare(float1=var "Quirrel Timer", float2=var "Quirrel Time", tolerance=0, equal=→"QUIRREL", lessThan=(none), greaterThan=→"QUIRREL", everyFrame=false)
    BoolTest(boolVariable=var "Quirrel Arrived", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Quirrel Land"), sendEvent="WATCH", delay=0, everyFrame=false)
    on QUIRREL → Set Timer
    on FINISHED → Idle
  }

  state Sleep {
    on BATTLE START → Wake Pause
  }

  state Wake Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake Rumble
  }

  state Wake Rumble {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Bub", emit=0)  // disabled
    PlayParticleEmitter(gameObject=var "Pt Spore", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Steam", emit=0)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst
  }

  state Burst {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=uumuu_appear (sharedassets164.assets))
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 23, 0), time=1, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetCollider(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Steam")
    StopParticleEmitter(gameObject=var "Pt Spore")
    PlayParticleEmitter(gameObject=var "Pt Bub", emit=0)
    PlayParticleEmitter(gameObject=var "Emerge Spore", emit=0)
    on FINISHED → Music
  }

  state Start {
    SetAudioClip(gameObject=Self, audioClip=uumuu_loop (sharedassets229.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetMeshRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Entry Sprite", active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    StopParticleEmitter(gameObject=var "Emerge Spore")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MEGA_JELLY", everyFrame=false)
    on FINISHED → Idle
  }

  state Quirrel Roam {
    SetFsmBool(gameObject=var "Quirrel Land", fsmName="Watch", variableName="Roam", setValue=true, everyFrame=false)
    on FINISHED → Quirrel?
  }

  state Attack Pause {
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=3, accelerationForce=3, offsetX=0, offsetY=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slash Antic
  }

  state Music {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    ApplyMusicCue(musicCue=EnemyBattle (sharedassets40.assets), delayTime=0, transitionTime=0)
    on FINISHED → Start
  }

  var Hero Mid X: float = 53.3
  var Hero Mid Y: float = 121.23
  var Hero X: float = 0
  var Hero Y: float = 0
  var Quirrel Time: float = 5
  var Quirrel Timer: float = 0
  var Slash Rotation: float = 0
  var Slash Scale: float = 0
  var Ct Chase: int = 0
  var Ct Multizap: int = 0
  var HP: int = 0
  var Zaps: int = 0
  var Above Mid X: bool = false
  var Above Mid Y: bool = false
  var First Chaser: bool = false
  var Quirrel Arrived: bool = false
  var Hero Pos: vector3 = (0, 0, 0)
  var Area Title: gameObject = <null>
  var Current Pattern: gameObject = <null>
  var Emerge Spore: gameObject = <null>
  var Entry Sprite: gameObject = <null>
  var Multizaps: gameObject = <null>
  var Pattern 1: gameObject = <null>
  var Pattern 2: gameObject = <null>
  var Pt Attack: gameObject = Mega Jellyfish/Pt Attack
  var Pt Bub: gameObject = <null>
  var Pt Spore: gameObject = <null>
  var Pt Steam: gameObject = <null>
  var Pt Wound: gameObject = <null>
  var Quirrel Land: gameObject = <null>
  var Quirrel Slash: gameObject = <null>
  var Self: gameObject = <null>
  var Strike Nail: gameObject = <null>
  var White Flash: gameObject = <null>
  var Zap: gameObject = <null>
}
