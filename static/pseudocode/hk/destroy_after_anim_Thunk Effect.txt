# 1 location(s):
#   Fungus1_02 / Thunk Effect

// component disabled: this FSM does not run until something enables it
// uses template: destroy_after_anim
fsm destroy_after_anim {
  start Pause

  state Wait {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=true)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait
  }
}
