# Destroy Self
# 2 location(s):
#   Fungus3_23_boss / Battle Scene/Cloth Entry/Cloth Fighter/Slam Effect
#   sharedassets32.assets / Slam Effect

fsm Destroy Self {
  start Wait

  state Wait {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
