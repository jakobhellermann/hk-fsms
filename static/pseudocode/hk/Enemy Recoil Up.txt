# Enemy Recoil Up
# 2 location(s):
#   Knight_Pickup / Knight/Charm Effects
#   resources.assets / Knight:1/Charm Effects

fsm Enemy Recoil Up {
  start Init
  on CHARM EQUIP CHECK → Check  // from any state

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_15", isTrue=→"EQUIPPED", isFalse=→"UNEQUIPPED")
    on EQUIPPED → Equipped
    on UNEQUIPPED → Unequipped
  }

  state Equipped {
    SetFsmFloat(gameObject=var "Alt Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Down Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Up Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Great Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=2, everyFrame=false)
    SetFsmFloat(gameObject=var "Hit L", fsmName="damages_enemy", variableName="magnitudeMult", setValue=2.5, everyFrame=false)
    SetFsmFloat(gameObject=var "Hit R", fsmName="damages_enemy", variableName="magnitudeMult", setValue=2.5, everyFrame=false)
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Knight")
    FindChild(gameObject=var "Knight", childName="Attacks", storeResult=var "Attacks")
    FindChild(gameObject=var "Attacks", childName="AltSlash", storeResult=var "Alt Slash")
    FindChild(gameObject=var "Attacks", childName="DownSlash", storeResult=var "Down Slash")
    FindChild(gameObject=var "Attacks", childName="UpSlash", storeResult=var "Up Slash")
    FindChild(gameObject=var "Attacks", childName="Slash", storeResult=var "Slash")
    FindChild(gameObject=var "Attacks", childName="Great Slash", storeResult=var "Great Slash")
    FindChild(gameObject=var "Attacks", childName="Cyclone Slash", storeResult=var "Cyclone Slash")
    FindChild(gameObject=var "Cyclone Slash", childName="Hits", storeResult=var "Hits")
    FindChild(gameObject=var "Hits", childName="Hit L", storeResult=var "Hit L")
    FindChild(gameObject=var "Hits", childName="Hit R", storeResult=var "Hit R")
    on FINISHED → Check
  }

  state Unequipped {
    SetFsmFloat(gameObject=var "Alt Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Down Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Up Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Great Slash", fsmName="damages_enemy", variableName="magnitudeMult", setValue=1.5, everyFrame=false)
    SetFsmFloat(gameObject=var "Hit L", fsmName="damages_enemy", variableName="magnitudeMult", setValue=2, everyFrame=false)
    SetFsmFloat(gameObject=var "Hit R", fsmName="damages_enemy", variableName="magnitudeMult", setValue=2, everyFrame=false)
  }

  var Alt Slash: gameObject = <null>
  var Attacks: gameObject = <null>
  var Cyclone Slash: gameObject = <null>
  var Down Slash: gameObject = <null>
  var Great Slash: gameObject = <null>
  var Hit L: gameObject = <null>
  var Hit R: gameObject = <null>
  var Hits: gameObject = <null>
  var Knight: gameObject = <null>
  var Slash: gameObject = <null>
  var Up Slash: gameObject = <null>
}
