# Box Open
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/DialogueManager
#   resources.assets / _GameCameras/HudCamera/DialogueManager

fsm Box Open {
  start Init

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Orbit Shield"), sendEvent="ORBIT SHIELD UP", delay=0, everyFrame=false)
    FadeGroupDown(target=var "Box", fast=false)
    iTweenScaleTo(gameObject=var "Box", id="", transformScale=var "", vectorScale=(0.75, 0.75, 0.75), time=0.3, delay=0, speed=(unset), easeType=easeInSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on BOX UP → Box Up
  }

  state Init {
    FindChild(gameObject=Self, childName="DialogueBox", storeResult=var "Box")
    on BOX UP → Box Up
  }

  state Box Up {
    SetPosition(gameObject=var "Box", vector=(unset), x=(unset), y=4.51, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemy Dream Msg"), sendEvent="CANCEL ENEMY DREAM", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Orbit Shield"), sendEvent="ORBIT SHIELD DOWN", delay=0, everyFrame=false)
    FadeGroupUp(target=var "Box")
    iTweenScaleTo(gameObject=var "Box", id="", transformScale=var "", vectorScale=(1, 1, 1), time=0.5, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on BOX DOWN → Box Down
    on BOX DOWN NO HUD → Box Down No HUD
  }

  state Box Down No HUD {
    FadeGroupDown(target=var "Box", fast=false)
    iTweenScaleTo(gameObject=var "Box", id="", transformScale=var "", vectorScale=(0.75, 0.75, 0.75), time=0.3, delay=0, speed=(unset), easeType=easeInSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on BOX UP → Box Up
  }

  var Box: gameObject = <null>
}
