# Phase Control
# 1 location(s):
#   Dream_Final_Boss / Boss Control/Radiance

fsm Phase Control {
  start Init

  state Init {
    on FINISHED → Idle 1
  }

  state Idle 1 {
    on TOOK DAMAGE → Pause 1
  }

  state Check 1 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "P2 Spike Waves", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Set Phase 2
    on FINISHED → Idle 1
  }

  state Set Phase 2 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SPIKE WAVES", delay=0, everyFrame=false)
    on FINISHED → Idle 2
  }

  state Pause 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check 1
  }

  state Idle 2 {
    on TOOK DAMAGE → Pause 2
  }

  state Check 2 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "P3 A1 Rage", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Set Phase 3
    on FINISHED → Idle 2
  }

  state Set Phase 3 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SPIKES OUTER", delay=0, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Rage Ready", setValue=true, everyFrame=false)
    on RAGE START → Idle 3
  }

  state Pause 2 {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check 2
  }

  state Idle 3 {
    on TOOK DAMAGE → Pause 3
  }

  state Check 3 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "P4 Stun1", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Stun 1
    on FINISHED → Idle 3
  }

  state Pause 3 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check 3
  }

  state Stun 1 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SPIKES NONE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN 1", delay=0, everyFrame=false)
    on ARENA 2 START → Idle 4
  }

  state Idle 4 {
    on TOOK DAMAGE → Pause 4
  }

  state Check 4 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "P5 Acend", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Set Ascend
    on FINISHED → Idle 4
  }

  state Pause 4 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check 4
  }

  state Set Ascend {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Ascend Ready", setValue=true, everyFrame=false)
  }

  var P2 Spike Waves: int = 2650
  var P3 A1 Rage: int = 2250
  var P4 Stun1: int = 2000
  var P5 Acend: int = 1300

  var HP: int = 0
  var Battle Corpse: gameObject = <null>
}
