# Laser Bug Mega
# 2 location(s):
#   Mines_32 / Laser Turret Mega (1)
#   GG_Crystal_Guardian_2 / Laser Turret Mega (1)

fsm Laser Bug Mega {
  start Init

  state Init {
    FindChild(gameObject=var "Beam Ball", childName="End Particle", storeResult=var "End Particle")
    FindChild(gameObject=var "Beam", childName="Beam Glow", storeResult=var "Beam Glow")
    FindChild(gameObject=var "Beam Impact", childName="Init Particle", storeResult=var "Init Particle")
    FindChild(gameObject=var "Beam Impact", childName="Impact Particle", storeResult=var "Impact Particle")
    FindChild(gameObject=var "Beam Impact", childName="Burn Particle", storeResult=var "Burn Particle")
    on FINISHED → Idle
  }

  state Idle {
    on LASER SHOOT → Aim
  }

  state Beam Antic {
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mines_pink_laser_prepare (sharedassets252.assets))
    PlayParticleEmitter(gameObject=var "Init Particle", emit=0)
    SetMeshRenderer(gameObject=var "Beam Ball", active=true)
    SetMeshRenderer(gameObject=var "Beam", active=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Beam Ball", clipName="Ball Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=var "Beam", animLibName="", clipName="Beam Antic")
    Wait(time=var "Antic Time", finishEvent=→"FINISHED", realTime=false)
    GetPosition(gameObject=var "Beam", vector=(unset), x=var "Ray Origin X", y=var "Ray Origin Y", z=(unset), space=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "Ray Origin", vector2Value=(unset), x=var "Ray Origin X", y=var "Ray Origin Y", everyFrame=false)
    RayCast2d(fromGameObject=Self, fromPosition=var "Ray Origin", direction=(unset), space=0, distance=100, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Beam
  }

  state Beam {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Wait(time=var "Beam Time", finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Beam Ball", animLibName="", clipName="Ball Shoot")
    Tk2dPlayAnimation(gameObject=var "Beam", animLibName="", clipName="Beam Shoot")
    SendEventByName(eventTarget=GameObject(var "Beam Glow"), sendEvent="UP", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Impact Particle", emit=0)
    PlayParticleEmitter(gameObject=var "Burn Particle", emit=0)  // disabled
    StopParticleEmitter(gameObject=var "Init Particle")
    SetMeshRenderer(gameObject=var "Beam Impact", active=true)
    on FINISHED → Beam End
  }

  state Beam End {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.2)
    Tk2dPlayAnimationWithEvents(gameObject=var "Beam Ball", clipName="Ball End", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=var "Beam", animLibName="", clipName="Beam End")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "End Particle", emit=0)
    SendEventByName(eventTarget=GameObject(var "Beam Glow"), sendEvent="DOWN", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Impact Particle")
    StopParticleEmitter(gameObject=var "Burn Particle")  // disabled
    SetMeshRenderer(gameObject=var "Beam Impact", active=false)
    on FINISHED → Beam Off
  }

  state Beam Off {
    AudioStop(gameObject=Self)
    SetMeshRenderer(gameObject=var "Beam Ball", active=false)
    SetMeshRenderer(gameObject=var "Beam", active=false)
    on FINISHED → Idle
  }

  state Aim {
    RandomFloat(min=var "Rot Min", max=var "Rot Max", storeResult=var "Rotation")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    WaitRandom(timeMin=0.5, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Beam Antic
  }

  var Antic Time: float = 0.6
  var Beam Time: float = 0.5
  var Rot Max: float = 190
  var Rot Min: float = 170
  var Start Pause: float = 0

  var Distance: float = 0
  var Ray Origin X: float = 0
  var Ray Origin Y: float = 0
  var Rotation: float = 0
  var Ray Origin: vector2 = (0, 0)
  var Beam: gameObject = Laser Turret Mega (1)/Beam
  var Beam Ball: gameObject = Laser Turret Mega (1)/Beam Ball
  var Beam Glow: gameObject = <null>
  var Beam Impact: gameObject = Laser Turret Mega (1)/Beam Impact
  var Burn Particle: gameObject = <null>
  var End Particle: gameObject = <null>
  var Impact Particle: gameObject = <null>
  var Init Particle: gameObject = <null>
}
