# Control
# 6 location(s):
#   GG_Waterways / Godseeker Waterways/Godseeker Fall/Dream Enter
#   GG_Waterways / Godseeker Waterways/Godseeker Dazed/Dream Enter
#   GG_Waterways / Godseeker Waterways/Godseeker Awake/Dream Enter
#   sharedassets225.assets / Godseeker Waterways/Godseeker Fall/Dream Enter
#   sharedassets225.assets / Godseeker Waterways/Godseeker Dazed/Dream Enter
#   sharedassets225.assets / Godseeker Waterways/Godseeker Awake/Dream Enter

// uses template: gg_dream_enter
fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"DREAM HIT", storeCollider=var "")
    on DREAM HIT → Take Control
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Convo", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", setValue=var "No Charms", everyFrame=false)
    on FINISHED → Impact
  }

  state Impact {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DREAM ENTER", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", value=var "Return Scene")
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    CreateObject(gameObject=dream_area_effect (sharedassets6.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=1.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Dream Box Down
  }

  state Change Scene {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=TimePasses())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=ResetSemiPersistentItems())
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=EnterWithoutInput(true))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AcceptInput())
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="enteredGGAtrium", isTrue=→"FINISHED", isFalse=→"CUTSCENE")
    on FINISHED → Wastes Cutscene?
    on CUTSCENE → Cutscene
  }

  state Dream Box Down {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BOX DOWN DREAM", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CONVO CANCEL", delay=0, everyFrame=false)
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Change Scene
  }

  state Atrium {
    BeginSceneTransition(sceneName=var "To Scene", entryGateName="door_dreamEnter", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  state Cutscene {
    GetConstantsValue(variableName="GG_ENTRANCE_SCENE", storeValue=var "To Scene")
    BeginSceneTransition(sceneName=var "To Scene", entryGateName="door_dreamEnter", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  state Wastes Cutscene? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="bossDoorCageUnlocked", isTrue=(none), isFalse=→"FINISHED")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="seenGGWastes", isTrue=→"FINISHED", isFalse=(none))
    SendRandomEvent(events=[→"CUTSCENE", →"FINISHED"], weights=[0.02, 0.98], delay=0)
    SendEvent(eventTarget=Self, sendEvent=→"CUTSCENE", delay=0, everyFrame=false)  // disabled
    on FINISHED → Atrium
    on CUTSCENE → Wastes Cutscene
  }

  state Wastes Cutscene {
    BeginSceneTransition(sceneName="GG_Unlock_Wastes", entryGateName="door_dreamEnter", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  var No Charms: bool = false
  var Return Scene: string = "GG_Waterways"
  var To Scene: string = "GG_Atrium"

  var Self: gameObject = <null>
}
