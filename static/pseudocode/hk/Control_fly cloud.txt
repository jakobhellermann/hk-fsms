# Control
# 1 location(s):
#   Ruins_Elevator / fly cloud

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_10", isTrue=→"READY", isFalse=→"INACTIVE")
    on READY → Ready
    on INACTIVE → Inactive
  }

  state Inactive {
  }

  state Ready {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"TOUCH", storeCollider=var "Player Obj")
    on TOUCH → Follow
  }

  state Follow {
    GetPosition(gameObject=var "Player Obj", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    PlayParticleEmitter(gameObject=Self, emit=0)
    SetPosition(gameObject=var "Self", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }

  var Hero Pos: vector3 = (0, 0, 0)
  var Player Obj: gameObject = <null>
  var Self: gameObject = <null>
}
