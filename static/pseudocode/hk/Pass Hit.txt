# Pass Hit
# 1 location(s):
#   Mines_29 / Mines Platform/Spikes Top

fsm Pass Hit {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="NAIL", delay=0, everyFrame=false)
    on FINISHED → Detect
  }

  var Parent: gameObject = <null>
}
