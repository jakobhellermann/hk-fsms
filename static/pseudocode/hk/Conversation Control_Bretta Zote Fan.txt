# Conversation Control
# 1 location(s):
#   Town / _NPCs/Zote Final Scene/Bretta Zote Fan

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Box Up
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    on FINISHED → Confuse Check
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    AudioStop(gameObject=var "Voice Player")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Greet
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Greet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_ZOTE", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Bretta_Zote_stare (sharedassets7.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="BRETTA", everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Confuse Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="greyPrinceDefeated", isTrue=→"CONFUSED", isFalse=(none))
    on FINISHED → Idle
    on CONFUSED → Confused
  }

  state Confused {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zote Confused")
    SetCollider(gameObject=Self, active=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="greyPrinceDefeats", storeValue=var "Zote Defeats")
    IntCompare(integer1=var "Zote Defeats", integer2=3, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on 1 → Mzote 1
    on 2 → Mzote 2
  }

  state Mzote 1 {
    SetFsmString(gameObject=var "Dream Dialogue", fsmName="npc_dream_dialogue", variableName="Convo Name", setValue="BRETTA_MZOTE_01", everyFrame=false)
    SetFsmString(gameObject=var "Dream Dialogue", fsmName="npc_dream_dialogue", variableName="Sheet Name", setValue="CP2", everyFrame=false)
  }

  state Mzote 2 {
    SetFsmString(gameObject=var "Dream Dialogue", fsmName="npc_dream_dialogue", variableName="Convo Name", setValue="BRETTA_MZOTE_02", everyFrame=false)
    SetFsmString(gameObject=var "Dream Dialogue", fsmName="npc_dream_dialogue", variableName="Sheet Name", setValue="CP2", everyFrame=false)
  }

  var Zote Defeats: int = 0
  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
