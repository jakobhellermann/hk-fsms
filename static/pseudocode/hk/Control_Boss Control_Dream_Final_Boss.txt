# Control
# 1 location(s):
#   Dream_Final_Boss / Boss Control

fsm Control {
  start Wait for Hero Pos

  state Init {
    FindChild(gameObject=Self, childName="Challenge Prompt Radiant", storeResult=var "Challenge Prompt Radiant")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Setup
  }

  state Setup {
    ActivateAllChildren(gameObject=var "Plat Sets", activate=true)
    GetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="Faced Radiance", storeValue=var "Faced Radiance", everyFrame=false)
    BoolTest(boolVariable=var "Faced Radiance", isTrue=→"REFIGHT", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Plats Init"), sendEvent="APPEAR", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=true)
    on CHALLENGE → Challenge Pause
    on REFIGHT → Refight Pause
  }

  state Challenge Pause {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=radiance_challenge (sharedassets407.assets))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="Faced Radiance", setValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=2)
    on FINISHED → Sun Antic
  }

  state Sun Antic {
    Tk2dPlayAnimationWithEvents(gameObject=var "Sun", clipName="Sun Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetSpriteRenderer(gameObject=var "Light Solid", active=true)
    SetMaterialColor(gameObject=var "Light Solid", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    iTweenFadeTo(gameObject=var "Light Solid", id="", alpha=0.25, includeChildren=true, namedValueColor="_Color", time=2, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic Rumble
  }

  state Antic Rumble {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=misc_rumble_loop (sharedassets101.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Appear Boom
  }

  state Appear Boom {
    ActivateGameObject(gameObject=var "Sun", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=radiance_scream_long (sharedassets407.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=mage_lord_onscreen_appear (sharedassets102.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "White Solid", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMaterialColor(gameObject=var "Light Solid", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    iTweenFadeTo(gameObject=var "Light Solid", id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=0.001, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    iTweenFadeTo(gameObject=var "Light Solid", id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=2, delay=0.05, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    ActivateGameObject(gameObject=var "Feather Particles", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Radiance Roar", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Radiance Roar", delivery=SendMessage, options=1, functionCall=flashDreamImpact())
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=true, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Eye Flash
  }

  state Eye Flash {
    SetSpriteRenderer(gameObject=var "White Fader", active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=radiance_laser_prepare (sharedassets407.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetMaterialColor(gameObject=var "White Fader", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    iTweenFadeTo(gameObject=var "White Fader", id="", alpha=0.5, includeChildren=true, namedValueColor="_Color", time=0.25, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "White Fader", id="", vector=(1.5, 1.5, 1), time=1, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Flash Up
  }

  state Flash Up {
    iTweenFadeTo(gameObject=var "White Fader", id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=0.25, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleTo(gameObject=var "White Fader", id="", transformScale=var "", vectorScale=(65, 43, 43), time=0.5, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=radiance_laser_burst (sharedassets261.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Title Up
  }

  state Title Up {
    AudioStop(gameObject=Self)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    ApplyMusicCue(musicCue=Radiance (sharedassets407.assets), delayTime=0, transitionTime=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    FadeGroupUp(target=var "Boss Title")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Radiance Roar", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Challenge", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Flash Down
  }

  state Flash Down {
    ActivateGameObject(gameObject=var "Radiance", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeGroupDown(target=var "Boss Title", fast=false)
    iTweenFadeTo(gameObject=var "White Fader", id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.25, delay=0.5, easeType=linear, loopType=none, startEvent=(none), finishEvent=→"NEXT", realTime=false, stopOnExit=true, loopDontFinish=true)
    on NEXT → Knight Ready
  }

  state Knight Ready {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Challenge End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Battle Start
  }

  state Battle Start {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SetPlayerDataBool(boolName="disablePause", value=false)
  }

  state Refight Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Sun", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Feather Particles", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Challenge Prompt Radiant", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Refight
  }

  state Refight {
    ActivateGameObject(gameObject=var "Radiance", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    ApplyMusicCue(musicCue=Radiance (sharedassets407.assets), delayTime=0, transitionTime=0)
  }

  state Wait for Hero Pos {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Init
  }

  var Boss Title: gameObject = Boss Control/Boss Title
  var CamLock Challenge: gameObject = Boss Control/CamLocks/CamLock Challenge
  var Feather Particles: gameObject = Boss Control/feather_particles
  var Light Solid: gameObject = Boss Control/Light Solid
  var Plat Sets: gameObject = Boss Control/Plat Sets
  var Plats Init: gameObject = Boss Control/Plat Sets/Init
  var Radiance: gameObject = Boss Control/Radiance
  var Radiance Roar: gameObject = Boss Control/Radiance Roar
  var Roar Legs: gameObject = Boss Control/Radiance Roar/Legs
  var Sun: gameObject = Boss Control/Sun
  var White Fader: gameObject = Boss Control/White Fader
  var White Solid: gameObject = Boss Control/white_solid

  var Faced Radiance: bool = false
  var Challenge Prompt Radiant: gameObject = <null>
}
