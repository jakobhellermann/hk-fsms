# Update Vessels
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels

fsm Update Vessels {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Vessel 1", storeResult=var "Vessel 1")
    FindChild(gameObject=Self, childName="Vessel 2", storeResult=var "Vessel 2")
    FindChild(gameObject=Self, childName="Vessel 3", storeResult=var "Vessel 3")
    FindChild(gameObject=Self, childName="Vessel 4", storeResult=var "Vessel 4")
    on FINISHED → Idle
  }

  state Idle {
    on MP RESERVE UP → Send Up Msg
    on MP RESERVE DOWN → Send Down Msg
  }

  state Send Up Msg {
    SendEventToGameObjectOptimized(target=var "Vessel 1", sendEvent="MP RESERVE UP", everyFrame=false)
    SendEventToGameObjectOptimized(target=var "Vessel 2", sendEvent="MP RESERVE UP", everyFrame=false)
    SendEventToGameObjectOptimized(target=var "Vessel 3", sendEvent="MP RESERVE UP", everyFrame=false)
    SendEventToGameObjectOptimized(target=var "Vessel 4", sendEvent="MP RESERVE UP", everyFrame=false)
    on FINISHED → Idle
  }

  state Send Down Msg {
    SendEventToGameObjectOptimized(target=var "Vessel 1", sendEvent="MP RESERVE DOWN", everyFrame=false)
    SendEventToGameObjectOptimized(target=var "Vessel 2", sendEvent="MP RESERVE DOWN", everyFrame=false)
    SendEventToGameObjectOptimized(target=var "Vessel 3", sendEvent="MP RESERVE DOWN", everyFrame=false)
    SendEventToGameObjectOptimized(target=var "Vessel 4", sendEvent="MP RESERVE DOWN", everyFrame=false)
    on FINISHED → Idle
  }

  var Vessel 1: gameObject = <null>
  var Vessel 2: gameObject = <null>
  var Vessel 3: gameObject = <null>
  var Vessel 4: gameObject = <null>
}
