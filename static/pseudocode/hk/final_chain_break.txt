fsm final_chain_break {
  start Idle

  state Idle {
    FindChild(gameObject=Self, childName="Pt Chain", storeResult=var "Pt Chain")
    GetOwner(storeGameObject=var "Self")
    on FINAL CHAIN BREAK → Pause
  }

  state Pause {
    WaitRandom(timeMin=0, timeMax=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Break {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[chain_cut (sharedassets30.assets), rope_cut (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    ActivateGameObject(gameObject=var "Pt Chain", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Chain", emit=0)
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  var Pt Chain: gameObject = <null>
  var Self: gameObject = <null>
}
