# FalseyControl
# 2 location(s):

fsm FalseyControl {
  start State 4
  on STUN → Check Facing  // from any state
  on STUN → Check Direction  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    SetGameObject(variable=var "Camera", gameObject=var "CameraParent", everyFrame=false)
    FindChild(gameObject=var "Battle Scene", childName="Ceiling Dust", storeResult=var "Ceiling Dust")
    FindChild(gameObject=Self, childName="Pool", storeResult=var "Pool")
    FindChild(gameObject=Self, childName="Armour Smash", storeResult=var "Armour Smash")
    FindChild(gameObject=Self, childName="Slam Rocks", storeResult=var "Slam Rocks")
    FindChild(gameObject=Self, childName="Tinger", storeResult=var "Tinger")
    FindChild(gameObject=Self, childName="Geo Pool", storeResult=var "Geo Pool")
    FindChild(gameObject=Self, childName="Rage Rocks", storeResult=var "Rage Rocks")
    FindChild(gameObject=var "Battle Scene", childName="FK Floor", storeResult=var "Floor")
    FindChild(gameObject=var "Battle Scene", childName="Crack Dust", storeResult=var "Crack Dust")
    CreateObject(gameObject=Pool (sharedassets48.assets), spawnPoint=var "Crack Dust", position=(0, 0, 0), rotation=(unset), storeObject=var "Crack Pool", networkInstantiate=false, networkGroup=0)  // disabled
    FindChild(gameObject=var "Battle Scene", childName="FK Barrel Summon", storeResult=var "Barrel Summoner")
    FindChild(gameObject=var "Battle Scene", childName="Rubble Fall", storeResult=var "Rubble Fall")
    FindChild(gameObject=var "Battle Scene", childName="Rage Dust", storeResult=var "Rage Dust")
    GetHero(storeResult=var "Hero")
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Roll Dust", storeResult=var "Roll Dust")
    FindChild(gameObject=Self, childName="Slam Effect", storeResult=var "Slam Effect")
    FindChild(gameObject=Self, childName="Death Head", storeResult=var "Death Head")
    FindChild(gameObject=Self, childName="Corpse Steam", storeResult=var "Corpse Steam")
    FindChild(gameObject=Self, childName="Staff", storeResult=var "Staff")
    FindChild(gameObject=Self, childName="Dust Land Large", storeResult=var "Dust Land Large")
    FindChild(gameObject=Self, childName="Dust Hit Plume", storeResult=var "Dust Hit Plume")
    on FINISHED → Dormant
  }

  state Idle {
    BoolTest(boolVariable=var "Hero Escaped", isTrue=→"ESCAPED", isFalse=(none), everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.39)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetCollider(gameObject=var "Hitter", active=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Hero Is Right", leftBool=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Is Right"], boolStates=[true, false], trueEvent=→"TURN L", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Is Right"], boolStates=[false, true], trueEvent=→"TURN R", falseEvent=(none), storeResult=(unset), everyFrame=true)
    WaitRandom(timeMin=var "Idle Min", timeMax=var "Idle Max", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move Choice
    on TURN R → Turn R
    on TURN L → Turn L
    on ESCAPED → Esc Jump Antic
  }

  state Turn R {
    IntCompare(integer1=var "Turns", integer2=3, equal=→"SMASH", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntAdd(intVariable=var "Turns", add=1, everyFrame=false)
    SetFloatValue(floatVariable=var "Stun X Speed", floatValue=-10, everyFrame=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Turn")
    SetScale(gameObject=Self, vector=(unset), x=1.3, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Run Speed", floatValue=14, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
    on SMASH → Move Choice
  }

  state Turn L {
    IntCompare(integer1=var "Turns", integer2=3, equal=→"SMASH", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntAdd(intVariable=var "Turns", add=1, everyFrame=false)
    SetFloatValue(floatVariable=var "Stun X Speed", floatValue=10, everyFrame=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Turn")
    SetScale(gameObject=Self, vector=(unset), x=-1.3, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Run Speed", floatValue=-14, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
    on SMASH → Move Choice
  }

  state Move Choice {
    SetIntValue(intVariable=var "Turns", intValue=0, everyFrame=false)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=21, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"RUN", everyFrame=true)
    SendEvent(eventTarget=Self, sendEvent=→"TEST", delay=0, everyFrame=false)  // disabled
    SendRandomEvent(events=[→"SMASH", →"JUMP ATTACK", →"JUMP"], weights=[1, 1, 1], delay=0)
    on JUMP → Determine Jump
    on JUMP ATTACK → Row Check
    on SMASH → S Check Hero Pos
    on RUN → Run Antic
    on TEST → S Antic
  }

  state Jump Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump Antic")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Jump
  }

  state Jump {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=false_knight_jump (sharedassets48.assets))
    PlayParticleEmitter(gameObject=var "Jump Dust", emit=0)
    PlayParticleEmitter(gameObject=var "Dust Land Large", emit=0)  // disabled
    SetGravity2dScale(gameObject=Self, gravityScale=0.125)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=90, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Land Large R (sharedassets48.assets), spawnPoint=var "Self", position=(0, -5.4, 0.5), rotation=(-90, 180, 180), storeObject=var "")
    on FINISHED → Rise
  }

    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Alter", everyFrame=true)

    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"FALL", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on FALL → Land Noise
  }

  state Land {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Land")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Determine Jump {
    IntCompare(integer1=var "Jump Count", integer2=0, equal=(none), lessThan=(none), greaterThan=→"RETURN", everyFrame=false)
    IntCompare(integer1=var "Stunned Amount", integer2=2, equal=→"RETURN", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntAdd(intVariable=var "Jump Count", add=1, everyFrame=false)
    SendRandomEvent(events=[→"RANDOM", →"TOWARDS"], weights=[0.5, 0.5], delay=0)
    on RANDOM → Walls Check
    on TOWARDS → Towards
    on RETURN → Move Choice
  }

  state Random {
    RandomFloat(min=-10, max=10, storeResult=var "Jump X")
    FloatCompare(float1=var "Jump X", float2=0, tolerance=0, equal=→"LEFT", lessThan=→"LEFT", greaterThan=→"RIGHT", everyFrame=false)
    on LEFT → Random Left
    on RIGHT → Random Right
  }

  state Towards {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Hero X", minValue=var "Range Min", maxValue=var "Range Max", everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Jump X", floatValue=var "Hero X", everyFrame=false)
    FloatSubtract(floatVariable=var "Jump X", subtract=var "Self X", everyFrame=false, perSecond=false)
    FloatMultiply(floatVariable=var "Jump X", multiplyBy=0.9, everyFrame=false)
    FloatClamp(floatVariable=var "Jump X", minValue=-12, maxValue=12, everyFrame=false)
    on FINISHED → Jump Antic
  }

  state Random Left {
    FloatClamp(floatVariable=var "Jump X", minValue=-999, maxValue=-5, everyFrame=false)
    on FINISHED → Jump Antic
  }

  state Random Right {

    SetGravity2dScale(gameObject=Self, gravityScale=0.12)

    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=90, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Land Large R (sharedassets48.assets), spawnPoint=var "Self", position=(0, -5.4, 0.5), rotation=(-90, 180, 180), storeObject=var "")

    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_swing (sharedassets48.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
