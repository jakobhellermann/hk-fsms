# Setup and Follow
# 2 location(s):
#   Deepnest_East_Hornet_boss / Hornet Boss 2/Needle/Needle Tink
#   GG_Hornet_2 / Boss Holder/Hornet Boss 2/Needle/Needle Tink

fsm Setup and Follow {
  start Setup

  state Setup {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Needle")
    GetParent(gameObject=var "Needle", storeResult=var "Hornet")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetFsmGameObject(gameObject=var "Hornet", fsmName="Control", variableName="Needle Tink", setValue=var "Self", everyFrame=false)
    GetFsmGameObject(gameObject=var "Needle", fsmName="Control", variableName="Parent", storeValue=var "Hornet", everyFrame=false)
    on FINISHED → Deparent
  }

  state Deparent {
    ActivateGameObject(gameObject=var "Needle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Needle", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Follow
  }

  state Follow {
    GetPosition(gameObject=var "Needle", vector=var "Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    GetRotation(gameObject=var "Needle", quaternion=(unset), vector=var "Rotation", xAngle=(unset), yAngle=(unset), zAngle=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=var "Rotation", xAngle=(unset), yAngle=(unset), zAngle=(unset), space=0, everyFrame=true, lateUpdate=false)
  }

  var Pos: vector3 = (0, 0, 0)
  var Rotation: vector3 = (0, 0, 0)
  var Hornet: gameObject = <null>
  var Needle: gameObject = <null>
  var Self: gameObject = <null>
}
