# Control
# 1 location(s):
#   Fungus1_04_boss / Dream Scene Activate

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="dreamerScene1", isTrue=→"INERT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDash", isTrue=→"INERT", isFalse=(none))
    on INERT → Inert
    on FINISHED → Init
  }

  state Inert {
  }

  state Init {
    FindGameObject(objectName="Dreamer Scene 1", withTag="Untagged", store=var "Dreamer Scene 1")
    FindGameObject(objectName="Hornet Saver", withTag="Untagged", store=var "Hornet Saver")
    FindGameObject(objectName="Battle Gate A", withTag="Untagged", store=var "Battle Gate A")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Activate
  }

  state Activate {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDash", isTrue=(none), isFalse=→"CANCEL")
    ActivateGameObject(gameObject=var "Hornet Saver", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Dreamer Scene 1", active=true)
    SendEventByName(eventTarget=GameObject(var "Battle Gate A"), sendEvent="BG OPEN", delay=0, everyFrame=false)
    on CANCEL → Idle
  }

  var Battle Gate A: gameObject = <null>
  var Dreamer Scene 1: gameObject = <null>
  var Hornet Saver: gameObject = <null>
}
