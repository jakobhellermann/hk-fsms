# Conversation Control
# 1 location(s):
#   Room_ruinhouse / Sly Dazed

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Convo Choice {
    GetPlayerDataBool(boolName="slyRescued", storeValue=var "Met")
    BoolTest(boolVariable=var "Met", isTrue=→"MEET 2", isFalse=→"MEET", everyFrame=false)
    on MEET → Meet
    on MEET 2 → Repeat
  }

  state Init {
    FindChild(gameObject=Self, childName="Steam", storeResult=var "Steam")
    FindChild(gameObject=Self, childName="Haze", storeResult=var "Haze")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Active?
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookDown")  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")  // disabled
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=var "Self", startVolume=1, endVolume=0, time=0.3)
    on FINISHED → Convo Choice
  }

  state Meet {
    SetPlayerDataBool(boolName="slyRescued", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_wake_01 (sharedassets20.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_RESCUE_1", "Sly"], storeResult=(unset var))
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=StoryRecord_visited("dazed Sly, and saved him"))
    on CONVO_FINISH → Meet 2
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Active? {
    GetPlayerDataBool(boolName="slyRescued", storeValue=var "Met")
    BoolTest(boolVariable=var "Met", isTrue=→"DESTROY", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Audio
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Meet 2 {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    PlayParticleEmitter(gameObject=var "Steam", emit=0)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_wake_talk (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    StopParticleEmitter(gameObject=var "Haze")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wake")
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_RESCUE_2", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Meet 3
  }

  state Repeat {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Sly_talk_03 (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="SLY", everyFrame=false)
    SetPlayerDataBool(boolName="slyRescued", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_RESCUE_REPEAT", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Meet 3 {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="SLY", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SLY_RESCUE_3", "Sly"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Audio {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Idle
  }

  var Met: bool = false
  var Area Title: gameObject = <null>
  var Haze: gameObject = <null>
  var Self: gameObject = <null>
  var Steam: gameObject = <null>
  var Voice Player: gameObject = <null>
}
