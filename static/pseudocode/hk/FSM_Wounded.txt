// uses template: color_fader_2dtk
fsm FSM {
  start Init
  on UP INSTANT → Up Instant  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetSpriteRenderer(gameObject=var "Self", active=true)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Start Colour", everyFrame=false)  // disabled
    SetTextMeshProColor(gameObject=Self, color=var "Start Colour", everyFrame=false)  // disabled
    BoolTest(boolVariable=var "Auto Up", isTrue=→"UP", isFalse=(none), everyFrame=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Start Colour", everyframe=false)
    on UP → Delay
    on DOWN → Down
  }

  state Up {
    SetColorValue(colorVariable=var "New Colour", color=var "Prev Colour", everyFrame=false)
    EaseColor(fromValue=var "Prev Colour", toValue=var "Up Colour", colorVariable=var "New Colour", time=var "Up Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)  // disabled
    SetTextMeshProColor(gameObject=Self, color=var "New Colour", everyFrame=true)  // disabled
    Tk2dSpriteSetColor(gameObject=Self, color=var "New Colour", everyframe=true)
    on DOWN → Down
    on FINISHED → Upped
  }

  state Down {
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))  // disabled
    Tk2dSpriteGetColor(gameObject=Self, color=var "Prev Colour", everyframe=false)
    SetColorValue(colorVariable=var "New Colour", color=var "Prev Colour", everyFrame=false)
    Tk2dSpriteGetColor(gameObject=Self, color=var "New Colour", everyframe=false)
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", fail=(none))  // disabled
    EaseColor(fromValue=var "Prev Colour", toValue=var "Down Colour", colorVariable=var "New Colour", time=var "Down Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)  // disabled
    Tk2dSpriteSetColor(gameObject=Self, color=var "New Colour", everyframe=true)
    on UP → Delay
    on FINISHED → Downed
  }

  state Delay {
    Wait(time=var "Up Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get Prev Colour
    on DOWN → Down
  }

  state Downed {
    on UP → Delay
  }

  state Upped {
    on DOWN → Down
  }

  state Up Instant {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Up Colour", everyFrame=false)  // disabled
    SetTextMeshProColor(gameObject=Self, color=var "Up Colour", everyFrame=false)  // disabled
    Tk2dSpriteSetColor(gameObject=Self, color=var "Up Colour", everyframe=false)
    on FINISHED → Upped
  }

  state Get Prev Colour {
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))  // disabled
    Tk2dSpriteGetColor(gameObject=Self, color=var "Prev Colour", everyframe=false)
    on FINISHED → Up
  }

  var Down Time: float = 0.5
  var Up Delay: float = 0
  var Up Time: float = 0.5
  var Auto Up: bool = true
  var Down Colour: color = (1, 1, 1, 0)
  var Start Colour: color = (1, 1, 1, 0)
  var Up Colour: color = (1, 1, 1, 0.5921569)

  var New Colour: color = (0, 0, 0, 1)
  var Prev Colour: color = (1, 1, 1, 1)
  var Self: gameObject = <null>
}
