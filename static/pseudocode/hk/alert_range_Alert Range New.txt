// uses template: hero_detect_lineofsight
fsm alert_range {
  start Initialise

  state Initialise {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    on FINISHED → Inactive
  }

  state Raycast {
    GetPosition(gameObject=var "Hero", vector=var "Hero Position", x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)
    SetFloatValue(floatVariable=var "Raycast X", floatValue=var "Hero X", everyFrame=true)
    SetFloatValue(floatVariable=var "Raycast Y", floatValue=var "Hero Y", everyFrame=true)
    FloatSubtract(floatVariable=var "Raycast X", subtract=var "Self X", everyFrame=true, perSecond=false)
    FloatSubtract(floatVariable=var "Raycast Y", subtract=var "Self Y", everyFrame=true, perSecond=false)
    SetVector2XY(vector2Variable=var "Raycast Vector", vector2Value=(unset), x=var "Raycast X", y=var "Raycast Y", everyFrame=true)
    SetVector2XY(vector2Variable=var "Hero Pos 2D", vector2Value=(unset), x=var "Hero X", y=var "Hero Y", everyFrame=true)
    SetVector2XY(vector2Variable=var "Self Pos 2D", vector2Value=(unset), x=var "Self X", y=var "Self Y", everyFrame=true)
    DistanceBetweenPoints2D(distanceResult=var "Distance", point1=var "Hero Pos 2D", point2=var "Self Pos 2D", everyFrame=true)
    RayCast2d(fromGameObject=var "Self", fromPosition=(0, 0), direction=var "Raycast Vector", space=0, distance=var "Distance", minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "View Obscured", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    SetBoolValue(boolVariable=var "Can See Hero", boolValue=var "View Obscured", everyFrame=true)
    BoolFlipEveryFrame(boolVariable=var "Can See Hero", everyFrame=true)
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=var "Can See Hero", everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"RAYCAST STOP", storeCollider=var "Collider")
    Wait(time=1, finishEvent=→"RAYCAST STOP", realTime=false)  // disabled
    on RAYCAST STOP → Inactive
  }

  state Inactive {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"RAYCAST START", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"RAYCAST START", storeCollider=var "")
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=false, everyFrame=false)
    on RAYCAST START → Raycast
  }

  var Bool Name: string = "Alert Range"
  var FSM Name: string = "Hatcher"

  var Distance: float = 0
  var Hero X: float = 0
  var Hero Y: float = 0
  var Raycast X: float = 0
  var Raycast Y: float = 0
  var Self X: float = 0
  var Self Y: float = 0
  var Can See Hero: bool = false
  var In Alert Range: bool = false
  var View Obscured: bool = false
  var Hero Pos 2D: vector2 = (0, 0)
  var Raycast Vector: vector2 = (0, 0)
  var Self Pos 2D: vector2 = (0, 0)
  var Hero Position: vector3 = (0, 0, 0)
  var Alert Range: gameObject = <null>
  var Collider: gameObject = <null>
  var Hero: gameObject = <null>
  var Hit Obj: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
