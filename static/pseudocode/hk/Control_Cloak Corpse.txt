# Control
# 1 location(s):
#   Fungus1_04 / Cloak Corpse

fsm Control {
  start Pause

  state Pause {
    FindChild(gameObject=Self, childName="Fall", storeResult=var "Fall")
    FindChild(gameObject=Self, childName="Fallen", storeResult=var "Fallen")
    FindChild(gameObject=Self, childName="Shiny Item", storeResult=var "Shiny Item")
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Check PD {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hornet1Defeated", isTrue=→"ACTIVATE", isFalse=(none))
    on ACTIVATE → Activate
    on FINISHED → Idle
  }

  state Init {
    ActivateGameObject(gameObject=var "Shiny Item", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Check PD
  }

  state Activate {
    ActivateGameObject(gameObject=var "Fallen", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state Idle {
    on HORNET LEAVE → Fall Pause
  }

  state Fall Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fall
  }

  state Fall {
    ActivateGameObject(gameObject=var "Fall", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.006, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Activate Item
  }

  state Activate Item {
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Fall: gameObject = <null>
  var Fallen: gameObject = <null>
  var Shiny Item: gameObject = <null>
}
