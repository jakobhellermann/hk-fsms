// uses template: menu_pointer
fsm menu_pointer {
  start Init
  on CHOSEN → Chosen  // from any state
  on CHOSEN UNSELECTABLE → Chosen Unselectable  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetMeshRenderer(gameObject=Self, active=false)
    BoolTest(boolVariable=var "Stag UI", isTrue=→"STAG UI CHECK", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Chosen", boolValue=false, everyFrame=false)
    on SELECTED → Selected
    on STAG UI CHECK → Stag Pointer Check
  }

  state Selected {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pointer Up")
    on UNSELECTED → Unselected
    on FINISHED → Selected Idle
  }

  state Unselected {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pointer Down")
    on SELECTED → Selected
  }

  state Chosen {
    BoolTest(boolVariable=var "Chosen", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Chosen", boolValue=true, everyFrame=false)
    GetPosition(gameObject=var "Self", vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    BoolTest(boolVariable=var "Left", isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    on LEFT → Left Anim
    on RIGHT → Right Anim
    on FINISHED → Return Anim
  }

  state Left Anim {
    iTweenMoveBy(gameObject=var "Self", id="", vector=(-0.45, 0, 0), time=0.05, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Return Anim
  }

  state Right Anim {
    iTweenMoveBy(gameObject=var "Self", id="", vector=(0.45, 0, 0), time=0.05, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Return Anim
  }

  state Return Anim {
    iTweenMoveTo(gameObject=var "Self", id="", transformPosition=var "", vectorPosition=var "Start Pos", time=0.1, delay=0, speed=(unset), space=0, easeType=easeOutSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on UNSELECTED → Unselected
  }

  state Chosen Unselectable {
    iTweenShakePosition(gameObject=var "Self", id="", vector=(0.1, 0.1, 0), time=0.35, delay=0, loopType=none, space=1, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Selected Idle
  }

  state Selected Idle {
    on UNSELECTED → Unselected
  }

  state Stag Pointer Check {
    SetBoolValue(boolVariable=var "Stag UI", boolValue=false, everyFrame=false)
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    GetParent(gameObject=var "Parent", storeResult=var "UI List Stag")
    GetFsmInt(gameObject=var "Parent", fsmName="ui_list_item", variableName="Item Number", storeValue=var "Item Number", everyFrame=false)
    GetFsmInt(gameObject=var "UI List Stag", fsmName="ui_list", variableName="Initial Item", storeValue=var "Initial Item", everyFrame=false)
    IntCompare(integer1=var "Item Number", integer2=var "Initial Item", equal=→"LEFT", lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Init
    on LEFT → Shift Left
  }

  state Shift Left {
    SetPosition(gameObject=var "Self", vector=(unset), x=-3.8, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Init
  }

  var Left: bool = true
  var Stag UI: bool = true

  var Initial Item: int = 0
  var Item Number: int = 0
  var Chosen: bool = false
  var Start Pos: vector3 = (0, 0, 0)
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var UI List Stag: gameObject = <null>
}
