fsm Up n Down {
  start State 1

  state State 1 {
    Wait(time=0.5, finishEvent=(none), realTime=false)  // disabled
    on MARKER UP → State 2
  }

  state State 2 {
    FindChild(gameObject=Self, childName="Knight Icon", storeResult=var "Knight Icon")
    ActivateGameObject(gameObject=var "Knight Icon", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetMeshRenderer(gameObject=Self, active=true)
    SetParent(gameObject=var "Knight Icon", parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    on STAG MENU DOWN → State 3
  }

  state State 3 {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var Knight Icon: gameObject = <null>
  var Parent: gameObject = <null>
}
