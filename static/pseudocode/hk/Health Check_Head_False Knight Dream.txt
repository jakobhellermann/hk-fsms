# Health Check
# 2 location(s):
#   Dream_01_False_Knight / False Knight Dream/Head
#   GG_Failed_Champion / False Knight Dream/Head

fsm Health Check {
  start PAUSE

  state Check 1 {
    GetOwner(storeGameObject=var "Self")
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=ColorReturnNeutral())
    GetHPEveryFrame(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=0, equal=→"STUN", lessThan=→"STUN", greaterThan=(none), everyFrame=true)
    on STUN → State 1
  }

  state State 1 {
    SpawnBlood(spawnPoint=var "Self", position=(0, 0, 0), spawnMin=50, spawnMax=50, speedMin=5, speedMax=35, angleMin=20, angleMax=160, colorOverride=(unset))
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=ColorReturnNeutral())
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Idle")
    SetIntValue(intVariable=var "HP", intValue=40, everyFrame=false)
    SetHP(target=Self, hp=40)
    SendEventByName(eventTarget=GameObject(var "Body"), sendEvent="STUN END", delay=0, everyFrame=false)
    on FINISHED → Check 1
  }

  state PAUSE {
    GetHP(target=Self, storeValue=var "HP")
    on FINISHED → Check 1
  }

  var HP: int = 0
  var Body: gameObject = False Knight Dream
  var Self: gameObject = <null>
}
