# Over Control
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Charms/Equipped Charms/Notches/Over Indicator
#   resources.assets / _GameCameras/HudCamera/Inventory/Charms/Equipped Charms/Notches/Over Indicator

fsm Over Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Over 1", storeResult=var "Over 1")
    FindChild(gameObject=Self, childName="Over 2", storeResult=var "Over 2")
    FindChild(gameObject=Self, childName="Over 3", storeResult=var "Over 3")
    FindChild(gameObject=Self, childName="Over 4", storeResult=var "Over 4")
    FindChild(gameObject=Self, childName="Over 5", storeResult=var "Over 5")
    FindChild(gameObject=Self, childName="Over 6", storeResult=var "Over 6")
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Display Overcharm Start", isTrue=→"DISPLAY OVERCHARM START", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Overcharmed", boolValue=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on CHARM INDICATOR CHECK → Update Pause
    on DISPLAY OVER → Display Pause
    on DISPLAY OVERCHARM START → Pause
  }

  state Update Pos {
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlotsFilled", storeValue=var "Slots Filled")
    ConvertIntToFloat(intVariable=var "Slots Filled", floatVariable=var "Slots Filled Float", everyFrame=false)
    SetFloatValue(floatVariable=var "X Shift", floatValue=var "Slots Filled Float", everyFrame=false)
    FloatMultiply(floatVariable=var "X Shift", multiplyBy=var "Notch Space", everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Shift", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Display {
    BoolTest(boolVariable=var "Overcharmed", isTrue=→"OVERCHARM", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Idle
    on OVERCHARM → Update Pos 3
  }

  state 1 {
    ActivateAllChildren(gameObject=var "Self", activate=true)  // disabled
    ActivateGameObject(gameObject=var "Over 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → 2
  }

  state 2 {
    IntCompare(integer1=var "Cost", integer2=2, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Over 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → 3
  }

  state 3 {
    IntCompare(integer1=var "Cost", integer2=3, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Over 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → 4
  }

  state 4 {
    IntCompare(integer1=var "Cost", integer2=4, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Over 4", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → 5
  }

  state 5 {
    IntCompare(integer1=var "Cost", integer2=5, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Over 5", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → 6
  }

  state 6 {
    IntCompare(integer1=var "Cost", integer2=6, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Over 6", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Display
  }

  state Overcharmed {
    Translate(gameObject=Self, vector=(unset), x=var "Notch Space", y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)  // disabled
    on OVERCHARM END → End
    on DISPLAY OVER → Redisplay
  }

  state Display Overcharm {
    FindChild(gameObject=Self, childName="Over 1", storeResult=var "Over 1")
    FindChild(gameObject=Self, childName="Over 2", storeResult=var "Over 2")
    FindChild(gameObject=Self, childName="Over 3", storeResult=var "Over 3")
    FindChild(gameObject=Self, childName="Over 4", storeResult=var "Over 4")
    FindChild(gameObject=Self, childName="Over 5", storeResult=var "Over 5")
    FindChild(gameObject=Self, childName="Over 6", storeResult=var "Over 6")
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on FINISHED → Update Pos 2
  }

  state Update Pos 2 {
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlots", storeValue=var "Slots Max")
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlots", storeValue=var "Slots Max")  // disabled
    IntOperator(integer1=var "Slots Filled", integer2=var "Slots Max", operation=Subtract, storeResult=var "Slots Filled", everyFrame=false)  // disabled
    ConvertIntToFloat(intVariable=var "Slots Max", floatVariable=var "Slots Filled Float", everyFrame=false)
    SetFloatValue(floatVariable=var "X Shift", floatValue=var "Slots Filled Float", everyFrame=false)
    FloatAdd(floatVariable=var "X Shift", add=1, everyFrame=false, perSecond=false)  // disabled
    FloatMultiply(floatVariable=var "X Shift", multiplyBy=var "Notch Space", everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Shift", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Overcharmed", boolValue=true, everyFrame=false)
    on FINISHED → 1
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Display Overcharm
  }

  state Fade In Notches? {
    BoolTest(boolVariable=var "Display Overcharm Start", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Display Overcharm Start", boolValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FADE UP INSTANT", delay=0, everyFrame=false)
    on FINISHED → Overcharmed
  }

  state Redisplay {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on FINISHED → 1
  }

  state Update Pos 3 {
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlots", storeValue=var "Slots Max")
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlots", storeValue=var "Slots Max")  // disabled
    IntOperator(integer1=var "Slots Filled", integer2=var "Slots Max", operation=Subtract, storeResult=var "Slots Filled", everyFrame=false)  // disabled
    ConvertIntToFloat(intVariable=var "Slots Max", floatVariable=var "Slots Filled Float", everyFrame=false)
    SetFloatValue(floatVariable=var "X Shift", floatValue=var "Slots Filled Float", everyFrame=false)
    FloatAdd(floatVariable=var "X Shift", add=1, everyFrame=false, perSecond=false)  // disabled
    FloatMultiply(floatVariable=var "X Shift", multiplyBy=var "Notch Space", everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Shift", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Overcharmed", boolValue=true, everyFrame=false)
    on FINISHED → Fade In Notches?
  }

  state Update Pause {
    on FINISHED → Update Pos
  }

  state Display Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → 1
  }

  state End {
    SetBoolValue(boolVariable=var "Display Overcharm Start", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Notch Space: float = 0.81
  var Slots Filled Float: float = 0
  var X Shift: float = 0
  var Cost: int = 0
  var Slots Filled: int = 0
  var Slots Max: int = 0
  var Display Overcharm Start: bool = false
  var Overcharmed: bool = false
  var Over 1: gameObject = <null>
  var Over 2: gameObject = <null>
  var Over 3: gameObject = <null>
  var Over 4: gameObject = <null>
  var Over 5: gameObject = <null>
  var Over 6: gameObject = <null>
  var Self: gameObject = <null>
}
