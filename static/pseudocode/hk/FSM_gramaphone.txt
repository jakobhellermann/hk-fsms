// uses template: breakable_object_v2
fsm FSM {
  start Pause

  state Idle {
    SetParent(gameObject=var "Pool", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    on TAKE DAMAGE → Get Parameters
    on BREAK → Self Break
    on BREAKABLE DEACTIVE → Deactive
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Active", storeResult=var "Active")
    FindChild(gameObject=var "Self", childName="Inactive", storeResult=var "Inactive")
    FindChild(gameObject=var "Self", childName="Effects", storeResult=var "Effects")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z", space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Z", lowerValue=var "Z Min", upperValue=var "Z Max", boolVariable=var "On Hero's Z Plane", trueEvent=(none), falseEvent=(none), everyFrame=false)
    BoolTest(boolVariable=var "On Hero's Z Plane", isTrue=(none), isFalse=→"INERT", everyFrame=false)
    FindChild(gameObject=var "Self", childName="Debris", storeResult=var "Debris Holder")
    IntSwitch(intVariable=var "Effect Type", compareTo=[0, 1, 2, 3, 4, 5], sendEvent=[→"DUST MED", →"DUST LARGE", →"NO EFFECT", →"SHROOM", →"NO EFFECT", →"NO EFFECT"], everyFrame=false)
    on INERT → Inert
    on DUST MED → Dust Medium
    on DUST LARGE → Dust Large
    on ACTIVATE → Activated
    on NO EFFECT → Idle
    on SHROOM → Shroom
  }

  state Inert {
    SetCollider(gameObject=var "Self", active=false)
  }

  state Get Parameters {
    GetEventSender(sentByGameObject=var "Damager")
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="magnitudeMult", storeValue=var "Attack Magnitude", everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage Dealt", everyFrame=false)  // disabled
    on FINISHED → Damage Check
  }

  state Attack Type {
    IntSwitch(intVariable=var "Attack Type", compareTo=[0, 1, 2], sendEvent=[→"NAIL", →"NAIL", →"SPELL"], everyFrame=false)
    on NAIL → Strike?
    on SPELL → Spell
    on FINISHED → Other
  }

  state Nail Effect {
    FloatMultiply(floatVariable=var "Fling Speed Min", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatMultiply(floatVariable=var "Fling Speed Max", multiplyBy=var "Attack Magnitude", everyFrame=false)
    GetMidPoint(gameObject=var "Self", target=var "Damager", midPoint=var "Midpoint", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=<null>, position=var "Midpoint", rotation=(unset), storeObject=var "Hit Effect")
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on DOWN → Nail Down
    on UP → Nail Up
    on LEFT → Nail Left
    on RIGHT → Nail Right
  }

  state Check Direction {
    BoolTest(boolVariable=var "Dont Rotate Particle", isTrue=→"NO ROTATE", isFalse=(none), everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Left
    on UP → Up
    on DOWN → Down
    on RIGHT → Right
    on NO ROTATE → 
  }

  state Left {
    SetFloatValue(floatVariable=var "Angle Min", floatValue=110, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=150, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=var "Dust Hit", spawnPoint=var "Self", position=var "Effect Origin", rotation=(180, 90, 270), storeObject=var "")
    on FINISHED → Spider Egg?
  }

  state Right {
    SpawnObjectFromGlobalPool(gameObject=var "Dust Hit", spawnPoint=var "Self", position=var "Effect Origin", rotation=(0, 90, 270), storeObject=var "")
    SetFloatValue(floatVariable=var "Angle Min", floatValue=30, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=70, everyFrame=false)
    on FINISHED → Spider Egg?
  }

  state Up {
    SetFloatValue(floatVariable=var "Angle Min", floatValue=70, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=110, everyFrame=false)
    FloatMultiply(floatVariable=var "Fling Speed Min", multiplyBy=1.5, everyFrame=false)
    FloatMultiply(floatVariable=var "Fling Speed Max", multiplyBy=1.5, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=var "Dust Hit", spawnPoint=var "Self", position=var "Effect Origin", rotation=(270, 90, 270), storeObject=var "")
    on FINISHED → Spider Egg?
  }

  state Down {
    SetFloatValue(floatVariable=var "Angle Min", floatValue=160, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=380, everyFrame=false)
    SetBoolValue(boolVariable=var "Hit Down", boolValue=true, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=var "Dust Hit Down", spawnPoint=var "Self", position=var "Effect Origin", rotation=(-72.5, -180, -180), storeObject=var "")
    on FINISHED → Spider Egg?
  }

  state Spell {
    FloatMultiply(floatVariable=var "Fling Speed Min", multiplyBy=var "Attack Magnitude", everyFrame=false)
    FloatMultiply(floatVariable=var "Fling Speed Max", multiplyBy=var "Attack Magnitude", everyFrame=false)
    BoolTest(boolVariable=var "spellInvulnerable", isTrue=→"TRUE", isFalse=(none), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Fireball Hit (resources.assets), spawnPoint=var "Self", position=var "Effect Origin", rotation=(unset), storeObject=var "Hit Effect")
    SetPosition(gameObject=var "Hit Effect", vector=(unset), x=(unset), y=(unset), z=0.0031, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Check Direction
    on TRUE → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state Nail Down {
    RandomFloat(min=250, max=290, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Direction
  }

  state Nail Up {
    RandomFloat(min=70, max=110, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Direction
  }

  state Nail Right {
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Direction
  }

  state Nail Left {
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=-1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Direction
  }

  state Dust Medium {
    SetGameObject(variable=var "Dust Hit", gameObject=Dust Hit Med R (resources.assets), everyFrame=false)
    SetGameObject(variable=var "Dust Hit Down", gameObject=Dust Hit Med Down R (resources.assets), everyFrame=false)
    on FINISHED → Idle
  }

  state Dust Large {
    SetGameObject(variable=var "Dust Hit", gameObject=Dust Hit Large R (resources.assets), everyFrame=false)
    SetGameObject(variable=var "Dust Hit Down", gameObject=Dust Hit Large Down R (resources.assets), everyFrame=false)
    on FINISHED → Idle
  }

  state Break {
    ActivateGameObject(gameObject=var "Inactive", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=var "Sound", pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    DestroyObject(gameObject=var "Active", delay=0, detachChildren=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "Debris Holder", activate=true)
    ActivateAllChildren(gameObject=var "Effects", activate=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    FindGameObject(objectName=var "Send HIT to", withTag="Untagged", store=var "Hit Target")
    SendEventByName(eventTarget=GameObject(var "Hit Target"), sendEvent="HIT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="BREAK", delay=0, everyFrame=false)
    FlingObjects(containerObject=var "Debris Holder", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=var "Fling Speed Min", speedMax=var "Fling Speed Max", angleMin=var "Angle Min", angleMax=var "Angle Max")
    IntCompare(integer1=var "Effect Type", integer2=5, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Pause Frame
  }

  state Activated {
    SetSpriteRenderer(gameObject=Self, active=false)
    FindGameObject(objectName=var "Send HIT to", withTag="Untagged", store=var "Hit Target")
    SendEventByName(eventTarget=GameObject(var "Hit Target"), sendEvent="HIT", delay=0, everyFrame=false)
    DestroyObject(gameObject=var "Active", delay=0, detachChildren=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="BROKEN", delay=0, everyFrame=false)
  }

  state Damage Check {
    IntCompare(integer1=var "Damage Dealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)  // disabled
    on CANCEL → Cancel Pause
    on FINISHED → Attack Type
  }

  state Cancel Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Shroom {
    SetGameObject(variable=var "Dust Hit", gameObject=Shroom Spore Med, everyFrame=false)
    SetGameObject(variable=var "Dust Hit Down", gameObject=Shroom Spore Med, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Effect Rotation", vector3Value=(unset), x=270, y=270, z=90, everyFrame=false)
    on FINISHED → Idle
  }

  state No Rotate Check {
    BoolTest(boolVariable=var "Dont Rotate Particle", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetRotation(gameObject=var "Effect Object", quaternion=(unset), vector=var "Effect Rotation", xAngle=(unset), yAngle=(unset), zAngle=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Break
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Disable
  }

  state Disable {
    SetCollider(gameObject=Self, active=false)
  }

  state Spider Egg? {
    IntCompare(integer1=var "Effect Type", integer2=4, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=12, spawnMax=16, speedMin=10, speedMax=20, angleMin=0, angleMax=360, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    on FINISHED → No Rotate Check
  }

  state Self Break {
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    SetFloatValue(floatVariable=var "Angle Min", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=360, everyFrame=false)
    on FINISHED → Spider Egg?
  }

  state Deactive {
    ActivateGameObject(gameObject=var "Active", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Other {
    SetFloatValue(floatVariable=var "Attack Direction", floatValue=90, everyFrame=false)
    SetFloatValue(floatVariable=var "Attack Magnitude", floatValue=1, everyFrame=false)
    on FINISHED → Strike?
  }

  state Strike? {
    IntCompare(integer1=var "Effect Type", integer2=5, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Nail Effect
  }

  var Fling Speed Max: float = 17
  var Fling Speed Min: float = 10
  var Effect Type: int = 2
  var Dont Rotate Particle: bool = false
  var Send HIT to: string = ""
  var Effect Origin: vector3 = (0, 0, 0)
  var Sound: object = breakable_wall_hit_1 (sharedassets6.assets)

  var Angle Max: float = 0
  var Angle Min: float = 0
  var Attack Direction: float = 0
  var Attack Magnitude: float = 0
  var Chooser: float = 0
  var Z: float = 0
  var Z Max: float = 1
  var Z Min: float = -1
  var Attack Type: int = 0
  var Damage Dealt: int = 0
  var Activated: bool = false
  var Hit Down: bool = false
  var On Hero's Z Plane: bool = false
  var spellInvulnerable: bool = false
  var Effect Rotation: vector3 = (0, 0, 0)
  var Midpoint: vector3 = (0, 0, 0)
  var Active: gameObject = <null>
  var Camera: gameObject = <null>
  var Damager: gameObject = <null>
  var Debris Holder: gameObject = <null>
  var Dust Hit: gameObject = <null>
  var Dust Hit Down: gameObject = <null>
  var Effect Object: gameObject = <null>
  var Effects: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Hit Target: gameObject = <null>
  var Inactive: gameObject = <null>
  var Pool: gameObject = <null>
  var Self: gameObject = <null>
}
