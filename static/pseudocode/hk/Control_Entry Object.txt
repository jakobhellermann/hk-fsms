# Control
# 1 location(s):
#   GG_God_Tamer / Entry Object

fsm Control {
  start Init

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    FindChild(gameObject=Self, childName="Lancer", storeResult=var "Lancer")
    FindChild(gameObject=Self, childName="Lobster", storeResult=var "Lobster")
    FindChild(gameObject=Self, childName="Roar Point", storeResult=var "Roar Point")
    GetOwner(storeGameObject=var "Self")
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gate Open
    on GG BOSS → Enable Mesh Renderer
  }

  state Enter {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Enter")
    SetMeshRenderer(gameObject=Self, active=true)
    iTweenMoveBy(gameObject=Self, id="", vector=(-14, 0, 0), time=(unset), delay=0, speed=6, easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on FINISHED → Roar Pause
  }

  state Roar Pause {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="R GATE CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ENTRY END", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Enter Idle")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar Antic
  }

  state Gate Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="R GATE OPEN", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter
  }

  state Roar Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Title
  }

  state Roar {
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Roar Point", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD CHEER", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Lancer", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Lancer", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=big_buzzer_scream (sharedassets32.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=big_buzzer_scream (sharedassets32.assets), pitchMin=0.85, pitchMax=0.85, volume=0.75, delay=0, storePlayer=<null>)
    on FINISHED → Roar End
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Lobster Roar End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fight Start
  }

  state Fight Start {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC 6", delay=0, everyFrame=false)
    SetPositionToObject(gameObject=var "Lobster", targetObject=var "Self", xOffset=0, yOffset=0, zOffset=0)
    ActivateGameObject(gameObject=var "Lobster", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetParent(gameObject=var "Lobster", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(var "Lobster"), sendEvent="WAKE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Title {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="LOBSTER_LANCER_C", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="LOBSTER_LANCER_NC", everyFrame=false)
    on FINISHED → Roar
  }

  state Enable Mesh Renderer {
    SetMeshRenderer(gameObject=Self, active=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar Pause
  }

  var Area Title: gameObject = <null>
  var Lancer: gameObject = <null>
  var Lobster: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Roar Point: gameObject = <null>
  var Self: gameObject = <null>
}
