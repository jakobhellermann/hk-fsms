# Mender Bug Ctrl
# 1 location(s):
#   Crossroads_01 / _Scenery/Mender Bug

fsm Mender Bug Ctrl {
  start Init
  on ZERO HP → Killed  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Dead?
  }

  state Sign Broken? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="menderSignBroken", isTrue=(none), isFalse=→"DESTROY")
    on DESTROY → Destroy
    on FINISHED → Chance
  }

  state Destroy {
    DestroySelf(detachChildren=false)  // disabled
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Dead? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="menderState", storeValue=var "State")
    IntCompare(integer1=var "State", integer2=2, equal=→"DESTROY", lessThan=→"FINISHED", greaterThan=→"DESTROY", everyFrame=false)
    on DESTROY → Destroy
    on FINISHED → Sign Broken?
  }

  state Chance {
    SetPlayerDataBool(boolName="menderSignBroken", value=false)
    RandomInt(min=1, max=50, storeResult=var "Chooser Int", inclusiveMax=true)
    IntCompare(integer1=var "Chooser Int", integer2=50, equal=→"FINISHED", lessThan=→"DESTROY", greaterThan=→"FINISHED", everyFrame=false)
    on DESTROY → Destroy
    on FINISHED → Idle
  }

  state Idle {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    AudioPlay(gameObject=var "Hammer", volume=0.15, oneShotClip=<null>, finishedEvent=(none))
    on HERO ENTER → Direction
  }

  state Startle {
    AudioStop(gameObject=Self)
    DestroyObject(gameObject=var "Hammer", delay=0, detachChildren=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Mender_startle (sharedassets37.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fly
  }

  state Fly {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetAudioClip(gameObject=Self, audioClip=flap_fast (sharedassets37.assets))
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetCollider(gameObject=Self, active=false)
    iTweenMoveBy(gameObject=Self, id="", vector=var "Flight Vector", time=1, delay=0, speed=(unset), easeType=easeInSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"DESTROY", realTime=false, stopOnExit=true, loopDontFinish=true)
    on DESTROY → Destroy
  }

  state Direction {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Fly Left
    on LEFT → Fly Right
  }

  state Fly Left {
    SetVector3Value(vector3Variable=var "Flight Vector", vector3Value=(-15, 20, -30), everyFrame=false)
    on FINISHED → Startle
  }

  state Fly Right {
    SetVector3Value(vector3Variable=var "Flight Vector", vector3Value=(15, 20, -30), everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Startle
  }

  state Killed {
    SetPlayerDataInt(intName="menderState", value=2)
  }

  var X Scale: float = 0
  var Chooser Int: int = 0
  var State: int = 0
  var Flight Vector: vector3 = (0, 0, 0)
  var Hammer: gameObject = _Scenery/Mender Bug/Hammer
  var Self: gameObject = <null>
  var Shiny Item: gameObject = <null>
}
