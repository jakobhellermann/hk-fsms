// component disabled: this FSM does not run until something enables it
fsm Fall Check {
  start Grounded

  state Grounded {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=var "Grounded", leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolTest(boolVariable=var "Grounded", isTrue=(none), isFalse=→"FALL", everyFrame=true)
    on FALL → Fall
  }

  state Fall {
    StopWalker(target=Self, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK STOP", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolTest(boolVariable=var "Grounded", isTrue=→"LAND", isFalse=(none), everyFrame=true)  // disabled
    on LAND → Land
  }

  state Land {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK START", delay=0, everyFrame=false)
    on FINISHED → Grounded
  }

  var Grounded: bool = false
}
