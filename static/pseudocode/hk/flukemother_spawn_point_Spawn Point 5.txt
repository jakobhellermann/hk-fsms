# flukemother_spawn_point
# 2 location(s):
#   Waterways_12_boss / Fluke Mother/Spawn Points/Spawn Point 5
#   GG_Flukemarm / Fluke Mother/Spawn Points/Spawn Point 5

// uses template: flukemother_spawn_point
fsm flukemother_spawn_point {
  start Init

  state Idle {
    on ANTIC → Antic
  }

  state Antic {
    AudioPlaySimple(gameObject=var "Gush Audio", volume=1, oneShotClip=<null>)
    PlayParticleEmitter(gameObject=var "Pt Blood", emit=0)
    on END → Spit
    on CANCEL → End
  }

  state Init {
    SetFloatValue(floatVariable=var "Spit Min", floatValue=var "Angle", everyFrame=false)
    SetFloatValue(floatVariable=var "Spit Max", floatValue=var "Angle", everyFrame=false)
    FloatAdd(floatVariable=var "Spit Min", add=-30, everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Spit Max", add=30, everyFrame=false, perSecond=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Spit Effect", storeResult=var "Spit Effect")
    FindChild(gameObject=Self, childName="Pt Blood", storeResult=var "Pt Blood")
    on FINISHED → Idle
  }

  state End {
    AudioStop(gameObject=var "Gush Audio")
    StopParticleEmitter(gameObject=var "Pt Blood")
    on FINISHED → Idle
  }

  state Spit {
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End
  }

  var Angle: float = -24.4198
  var Gush Audio: gameObject = Fluke Mother/Gush Audio

  var Spit Max: float = 0
  var Spit Min: float = 0
  var Pt Blood: gameObject = <null>
  var Self: gameObject = <null>
  var Spit Effect: gameObject = <null>
}
