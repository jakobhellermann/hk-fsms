# Control
# 2 location(s):
#   Dream_Mighty_Zote / Audience/CamLock Roar
#   GG_Grey_Prince_Zote / CamLock Roar

fsm Control {
  start Inactive

  state Inactive {
    SetCollider(gameObject=Self, active=false)
    on ZOTE APPEAR → Active
  }

  state Active {
    SetCollider(gameObject=Self, active=true)
    on ZOTE ROAR END → Inactive
  }
}
