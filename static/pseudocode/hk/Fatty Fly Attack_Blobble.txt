# Fatty Fly Attack
# 1 location(s):
#   sharedassets33.assets / Blobble

fsm Fatty Fly Attack {
  start Sleep
  on TAKE DAMAGE → Attack Antic  // from any state

  state Wait {
    GetOwner(storeGameObject=var "Self")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WAKE", delay=0, everyFrame=false)
    WaitRandom(timeMin=1.25, timeMax=2.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack Antic
  }

  state Attack Antic {
    Decelerate(gameObject=Self, deceleration=0.1)
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack Antic 2
  }

  state Attack {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=spitting_zombie_spit, pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlingObjectsFromGlobalPool(gameObject=Spitter Shot R (sharedassets32.assets), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=45, angleMax=45, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spitter Shot R (sharedassets32.assets), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=135, angleMax=135, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spitter Shot R (sharedassets32.assets), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=225, angleMax=225, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spitter Shot R (sharedassets32.assets), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=315, angleMax=315, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → CD
  }

  state CD {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait
  }

  state Attack Antic 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=→"FINISHED", animationCompleteEvent=→"FINISHED")
    Decelerate(gameObject=Self, deceleration=0.1)
    Wait(time=0.42, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack
  }

  state Sleep {
    on START → Wait
  }

  var Shot Speed: float = 12
  var Self: gameObject = <null>
}
