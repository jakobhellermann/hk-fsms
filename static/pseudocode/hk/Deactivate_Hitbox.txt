# Deactivate
# 2 location(s):
#   Knight_Pickup / Knight/Dream Effects/Hitbox
#   resources.assets / Knight:1/Dream Effects/Hitbox

fsm Deactivate {
  start Wait

  state Wait {
    Wait(time=0.06, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pause Frame
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pause Frame 2
  }

  state Pause Frame 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Deactivate
  }
}
