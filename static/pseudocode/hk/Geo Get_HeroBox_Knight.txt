# Geo Get
# 1 location(s):
#   Knight_Pickup / Knight/HeroBox

fsm Geo Get {
  start Init

  state Inert {
    SetMeshRenderer(gameObject=var "Geo Get", active=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Geo", collideLayer="Particle", sendEvent=→"GEO GET", storeCollider=var "")
    on GEO GET → Get 1
  }

  state Get 1 {
    SetRandomRotation(gameObject=var "Geo Get", x=false, y=false, z=true)
    AudioStop(gameObject=Self)
    AudioPlayRandom(gameObject=var "Geo Get", audioClips=[geo_small_collect_1 (resources.assets), geo_small_collect_2 (resources.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15)
    SetMeshRenderer(gameObject=var "Geo Get", active=true)
    Tk2dPlayFrame(gameObject=var "Geo Get", frame=0)
    Tk2dPlayAnimationWithEvents(gameObject=var "Geo Get", clipName="Get", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get 2
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Geo Get", storeResult=var "Geo Get")
    on FINISHED → Inert
  }

  state Get 2 {
    Tk2dWatchAnimationEvents(gameObject=var "Geo Get", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetPosition(gameObject=Self, vector=var "World Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)  // disabled
    on FINISHED → Inert
    on GEO GET → Get 1
  }

  var World Pos: vector3 = (0, 0, 0)
  var Geo Get: gameObject = <null>
  var Self: gameObject = <null>
}
