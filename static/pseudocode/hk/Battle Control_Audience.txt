# Battle Control
# 1 location(s):
#   Dream_Mighty_Zote / Audience

fsm Battle Control {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="Darkness", storeResult=var "Darkness")
    FindChild(gameObject=Self, childName="Audio", storeResult=var "Audio")
    GetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="Faced Zote", storeValue=var "Faced Zote", everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="Faced Zote", setValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Faced Zote", isTrue=→"SHORT", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
    on SHORT → Short Init
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Entry Pause
  }

  state Entry Pause {
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "Faced Zote", isTrue=→"SHORT", isFalse=(none), everyFrame=false)
    on FINISHED → Darkness Up
    on SHORT → Short Pause
  }

  state Darkness Up {
    SendEventByName(eventTarget=GameObject(var "Darkness"), sendEvent="UP", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Audio", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=secret_discovered_temp (sharedassets6.assets), pitchMin=0.75, pitchMax=0.75, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=hero_fury_charm_burst (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=hero_fury_charm_burst (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=queen_bretta_summon_02 (sharedassets399.assets), pitchMin=1, pitchMax=1, volume=1, delay=1, storePlayer=<null>)
    on FINISHED → Zote Start
  }

  state Zote Start {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ZOTE APPEAR", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Short Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Audio", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Zote Start
  }

  state Short Init {
    ActivateGameObject(gameObject=var "Darkness", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Faced Zote: bool = false
  var Audio: gameObject = <null>
  var Darkness: gameObject = <null>
}
