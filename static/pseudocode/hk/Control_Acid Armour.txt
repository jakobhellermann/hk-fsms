# Control
# 2 location(s):
#   Knight_Pickup / Knight/Effects/Acid Armour
#   resources.assets / Knight:1/Effects/Acid Armour

fsm Control {
  start Init
  on LEAVING SCENE → Out of Acid  // from any state
  on FSM CANCEL → Out of Acid  // from any state
  on LEVEL LOADED → Out of Acid  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Bub Cloud", storeResult=var "Bub Cloud")
    FindChild(gameObject=Self, childName="Acid Steam", storeResult=var "Acid Steam")
    SetGameObjectSelf(variable=var "Acid Armour", gameObject=Self, everyFrame=false)
    on FINISHED → Out of Acid
  }

  state Out of Acid {
    SetMeshRenderer(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Bub Cloud")  // disabled
    StopParticleEmitter(gameObject=var "Acid Steam")  // disabled
    SetParticleEmission(gameObject=var "Acid Steam", emission=false)
    SetParticleEmission(gameObject=var "Bub Cloud", emission=false)
    VibrationPlayerStop(target=Self)
    on ACID ARMOUR START → Enter Acid
  }

  state Enter Acid {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Acid Armour")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetParticleEmission(gameObject=var "Acid Steam", emission=true)  // disabled
    SetParticleEmission(gameObject=var "Bub Cloud", emission=true)
    VibrationPlayerPlay(target=Self)
    on ACID ARMOUR END → Out of Acid
  }

  var Acid Steam: gameObject = <null>
  var Bub Cloud: gameObject = <null>
}
