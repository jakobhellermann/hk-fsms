// uses template: radiant_beam
fsm Control {
  start Init

  state Init {
    on FINISHED → Inert
  }

  state Inert {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on ANTIC → Antic
  }

  state Antic {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Beam Antic")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FIRE → Fire
  }

  state Fire {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Beam")
    SetCollider(gameObject=Self, active=true)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on END → End
  }

  state End {
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Beam End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Inert
    on ANTIC → Antic
  }

  state Recycle {
    RecycleSelf()
  }
}
