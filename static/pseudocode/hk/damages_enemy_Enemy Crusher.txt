// uses template: damages_enemy
fsm damages_enemy {
  start Idle

  state Idle {
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"HIT", storeCollider=var "Collider", storeForce=(unset))
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "Collider")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="Enemies", sendEvent=→"HIT", storeCollider=var "Collider")
    on HIT → Send Event
  }

  state Send Event {
    IntCompare(integer1=var "damageDealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    GetLayer(gameObject=var "Collider", storeResult=var "Layer", everyFrame=false)
    IntCompare(integer1=var "Layer", integer2=20, equal=→"CANCEL", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Layer", integer2=9, equal=→"CANCEL", lessThan=(none), greaterThan=(none), everyFrame=false)
    CheckSendEventLimit(gameObject=var "Collider", target=Self, trueEvent=(none), falseEvent=→"FINISHED")
    GetName(gameObject=var "Collider", storeName=var "Name", everyFrame=false)
    CompareNames(name=var "Name", strings=var "Ignore Names", target=Self, trueEvent=→"CANCEL", falseEvent=(none))
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="TAKE DAMAGE", delay=0, everyFrame=false)
    TakeDamage(Target=var "Collider", AttackType=var "attackType", CircleDirection=var "circleDirection", DamageDealt=var "damageDealt", Direction=var "direction", IgnoreInvulnerable=var "Ignore Invuln", MagnitudeMultiplier=var "magnitudeMult", MoveAngle=var "Move Angle", MoveDirection=var "moveDirection", Multiplier=var "Multiplier", SpecialType=var "Special Type")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Parent
    on CANCEL → Idle
  }

  state Parent {
    GetParent(gameObject=var "Collider", storeResult=var "Parent")
    GameObjectIsNull(gameObject=var "Parent", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckSendEventLimit(gameObject=var "Parent", target=Self, trueEvent=(none), falseEvent=→"FINISHED")
    GetName(gameObject=var "Parent", storeName=var "Name", everyFrame=false)
    CompareNames(name=var "Name", strings=var "Ignore Names", target=Self, trueEvent=→"FINISHED", falseEvent=(none))
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TAKE DAMAGE", delay=0, everyFrame=false)
    TakeDamage(Target=var "Parent", AttackType=var "attackType", CircleDirection=var "circleDirection", DamageDealt=var "damageDealt", Direction=var "direction", IgnoreInvulnerable=var "Ignore Invuln", MagnitudeMultiplier=var "magnitudeMult", MoveAngle=var "Move Angle", MoveDirection=var "moveDirection", Multiplier=var "Multiplier", SpecialType=var "Special Type")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Grandparent
  }

  state Grandparent {
    GetParent(gameObject=var "Parent", storeResult=var "G Parent")
    GameObjectIsNull(gameObject=var "G Parent", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckSendEventLimit(gameObject=var "G Parent", target=Self, trueEvent=(none), falseEvent=→"FINISHED")
    GetName(gameObject=var "G Parent", storeName=var "Name", everyFrame=false)
    CompareNames(name=var "Name", strings=var "Ignore Names", target=Self, trueEvent=→"FINISHED", falseEvent=(none))
    SendEventByName(eventTarget=GameObject(var "G Parent"), sendEvent="TAKE DAMAGE", delay=0, everyFrame=false)
    TakeDamage(Target=var "G Parent", AttackType=var "attackType", CircleDirection=var "circleDirection", DamageDealt=var "damageDealt", Direction=var "direction", IgnoreInvulnerable=var "Ignore Invuln", MagnitudeMultiplier=var "magnitudeMult", MoveAngle=var "Move Angle", MoveDirection=var "moveDirection", Multiplier=var "Multiplier", SpecialType=var "Special Type")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var direction: float = 0
  var magnitudeMult: float = 0
  var Special Type: int = 0
  var attackType: int = 4
  var damageDealt: int = 9999
  var Ignore Invuln: bool = false
  var circleDirection: bool = false
  var moveDirection: bool = false

  var Move Angle: float = 0
  var Multiplier: float = 1
  var Layer: int = 0
  var Name: string = ""
  var Collider: gameObject = <null>
  var G Parent: gameObject = <null>
  var Parent: gameObject = <null>
  var Ignore Names: array = var "Ignore Names"
}
