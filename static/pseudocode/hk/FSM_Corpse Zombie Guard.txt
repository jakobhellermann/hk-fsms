fsm FSM {
  start Initiate

  state Initiate {
    SetProperty(targetProperty=ParticleSystem on <null>, everyFrame=false)
    Wait(time=0.05, finishEvent=(none), realTime=false)
    GetOwner(storeGameObject=var "Self")
    SetFsmBool(gameObject=Self, fsmName="", variableName="", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    on FINISHED → Sound
  }

  state In Air {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LANDED", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)

    on LANDED → Chinese?
  }

  state Landed {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Emit
  }

  state Check Scale {
    GetScale(gameObject=Self, vector=(unset), xScale=var "scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "scale", float2=0, tolerance=0, equal=→"RIGHT", lessThan=→"LEFT", greaterThan=→"RIGHT", everyFrame=false)
    on LEFT → Club L
    on RIGHT → Club R
  }

  state Club L {
    CreateObject(gameObject=Zombie Guard Club, spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "Club", networkInstantiate=false, networkGroup=0)

    AddTorque2d(gameObject=var "Club", forceMode=0, torque=-400, everyFrame=false)
    on FINISHED → Death Stun
  }

  state Club R {
    CreateObject(gameObject=Zombie Guard Club, spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "Club", networkInstantiate=false, networkGroup=0)
    SetVelocityAsAngle(gameObject=var "Club", angle=105, speed=20, everyFrame=false)
    AddTorque2d(gameObject=var "Club", forceMode=0, torque=400, everyFrame=false)
    on FINISHED → Death Stun
  }

  state Stop Emit {
    SetProperty(targetProperty=ParticleSystem.enableEmission on Corpse Zombie Guard/Corpse Steam@ParticleSystem, everyFrame=false)
  }

  state Flame Check {
    BoolTest(boolVariable=var "spellBurn", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Start Flame
    on FALSE → Check Scale
  }

  state Start Flame {
    SetProperty(targetProperty=ParticleSystem on <null>, everyFrame=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.19607843, 0.19607843, 0.19607843, 1), everyframe=false)
    on FINISHED → Death Stun
  }

  state Sound {
    GetOwner(storeGameObject=var "Self")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=zombie_guard_death, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Check Scale
  }

  state Death Stun {
    GetVelocity2d(gameObject=var "Self", vector=var "Velocity", x=(unset), y=(unset), space=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=true)

    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stun End
  }

  state Stun End {

    StopParticleEmitter(gameObject=var "Stun Steam")
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=2, y=2, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → In Air
  }

  state Chinese? {
    CheckIsChineseBuild(trueEvent=(none), falseEvent=→"FINISHED")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Landed
  }

  var Chooser: float = 0
  var scale: float = 0
  var spellBurn: bool = false
  var Velocity: vector2 = (0, 0)
  var Audio Player: gameObject = <null>

  var Stun Steam: gameObject = <null>
}
