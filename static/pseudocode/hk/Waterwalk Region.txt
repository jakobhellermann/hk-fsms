# Waterwalk Region
# 1 location(s):
#   RestingGrounds_08 / WaterWalk Region

fsm Waterwalk Region {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"ENTER", storeCollider=var "Hero")
    on ENTER → Enter
  }

  state Enter {
    SetPlayerDataInt(intName="environmentType", value=3)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(0.2, -0.7, 0), spawnMin=6, spawnMax=8, speedMin=5, speedMax=15, angleMin=45, angleMax=70, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Hero", position=(-0.2, -0.7, 0), spawnMin=6, spawnMax=8, speedMin=5, speedMax=15, angleMin=115, angleMax=135, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=0.8, pitchMax=0.95, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Spa Ripple", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Splash In Black (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Splash")
    SetScale(gameObject=var "Splash", vector=(unset), x=1.75, y=1.75, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=var "Splash", color=(1, 1, 1, 1), everyframe=false)
    on FINISHED → Stay
  }

  state Stay {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Y", add=0.31, everyFrame=true, perSecond=false)
    SetPosition(gameObject=var "Spa Ripple", vector=(unset), x=var "Hero X", y=var "Self Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Player", collideLayer="", sendEvent=→"LEAVE", storeCollider=var "Hero")
    on LEAVE → Leave
  }

  state Leave {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=1, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataInt(intName="environmentType", value=var "Scene Enviro Type")
    ActivateGameObject(gameObject=var "Spa Ripple", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Splash Out White (resources.assets), spawnPoint=var "Hero", position=(0, -1, 0), rotation=(unset), storeObject=var "Splash")
    on FINISHED → Detect
  }

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="environmentType", storeValue=var "Scene Enviro Type")
    FindChild(gameObject=Self, childName="Ripple", storeResult=var "Spa Ripple")
    on FINISHED → Detect
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Hero X: float = 0
  var Self Y: float = 0
  var Scene Enviro Type: int = 0
  var Hero: gameObject = <null>
  var Spa Ripple: gameObject = <null>
  var Splash: gameObject = <null>
  var Splash Out Obj: gameObject = <null>
}
