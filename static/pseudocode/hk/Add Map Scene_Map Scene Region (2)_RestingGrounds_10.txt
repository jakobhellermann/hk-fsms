# Add Map Scene
# 1 location(s):
#   RestingGrounds_10 / Map Scene Region (2)

fsm Add Map Scene {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Add
  }

  state Add {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AddToScenesVisited", parameters=[var "Scene Name"], storeResult=(unset var))
  }

  var Scene Name: string = "RestingGrounds_10_d"
}
