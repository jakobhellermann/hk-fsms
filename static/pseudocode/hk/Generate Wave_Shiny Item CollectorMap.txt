# Generate Wave
# 1 location(s):
#   Ruins2_11 / Shiny Item CollectorMap

fsm Generate Wave {
  start Init
  on GET → Get  // from any state
  on STOP → Stopped  // from any state

  state Gen {
    CreateObject(gameObject=White Wave Default (sharedassets22.assets), spawnPoint=var "Self", position=(0, 0, 0.5), rotation=(unset), storeObject=var "Wave", networkInstantiate=false, networkGroup=0)  // disabled
    SetScale(gameObject=var "Wave", vector=(unset), x=0.5, y=0.5, z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    Tk2dSpriteSetColor(gameObject=var "Wave", color=(1, 1, 1, 0.74509805), everyframe=false)  // disabled
    ActivateGameObject(gameObject=var "Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetScale(gameObject=var "Wave", vector=(unset), x=0.35, y=0.35, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=var "Wave", color=(1, 1, 1, 0.74509805), everyframe=false)
    on FINISHED → Wait
  }

  state Wait {
    Wait(time=2.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gen
  }

  state Get {
  }

  state Stopped {
    on START → Gen
  }

  var Self: gameObject = <null>
  var Wave: gameObject = Shiny Item CollectorMap/White Wave
}
