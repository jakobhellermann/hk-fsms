# Conversation Control
# 1 location(s):
#   Crossroads_47 / _NPCs/Tiso Bench NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="RestBench", withTag="RespawnPoint", store=var "Bench")  // disabled
    SetFsmVector3(gameObject=var "Bench", fsmName="Bench Control", variableName="Adjust Vector", setValue=(-0.45, 0, 0), everyFrame=false)  // disabled
    on FINISHED → Wait
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sit Talk End")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sit Talk")
    on FINISHED → Title
  }

  state Greet {
    SetPlayerDataBool(boolName="tisoEncounteredBench", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["TISO_BENCH_GREET", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Tiso_talk_02 (sharedassets77.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk Finish
  }

  state Convo Choice {
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"GREET", everyFrame=false)
    on GREET → Greet
    on REPEAT → Repeat
  }

  state Repeat {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["TISO_BENCH_REPEAT", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Tiso_talk_03 (sharedassets77.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk Finish
  }

  state Title {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="TISO_C", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="TISO_NC", everyFrame=false)
    on FINISHED → Box Up
  }

  state Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Alter Bench
  }

  state Alter Bench {
    on FINISHED → Idle
  }

  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Bench: gameObject = <null>
  var Self: gameObject = <null>
}
