fsm break_floor {
  start Pause

  state Idle {
    ReceivedDamage(collideTag="", sendEvent=→"NAIL HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on NAIL HIT → Check If Nail
  }

  state Check If Nail {
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0], sendEvent=[→"TRUE"], everyFrame=false)
    on TRUE → Hit
    on FINISHED → Idle
  }

  state Hit {
    IntSwitch(intVariable=var "Hits", compareTo=[0, 1, 2], sendEvent=[→"HIT 1", →"HIT 2", →"HIT 3"], everyFrame=false)
    on HIT 1 → Hit 1
    on HIT 2 → Hit 2
    on HIT 3 → Break
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="msk_generic", storeResult=var "mask")
    FindChild(gameObject=Self, childName="wood small", storeResult=var "Wood 1")
    FindChild(gameObject=Self, childName="wood large", storeResult=var "Wood 2")
    FindChild(gameObject=Self, childName="floor 1", storeResult=var "Floor 1")
    FindChild(gameObject=Self, childName="floor 2", storeResult=var "Floor 2")
    FindChild(gameObject=Self, childName="Dust Hit 1", storeResult=var "Dust Hit 1")
    FindChild(gameObject=Self, childName="Dust Hit 2", storeResult=var "Dust Hit 2")
    FindChild(gameObject=Self, childName="Dust Hit 3", storeResult=var "Dust Hit 3")
    FindChild(gameObject=Self, childName="Pt Bits 1", storeResult=var "Pt Bits 1")
    FindChild(gameObject=Self, childName="Pt Bits 2", storeResult=var "Pt Bits 2")
    FindChild(gameObject=Self, childName="Pt Bits 3", storeResult=var "Pt Bits 3")
    FindChild(gameObject=Self, childName="Pt Wood 1", storeResult=var "Pt Wood 1")
    FindChild(gameObject=Self, childName="Pt Wood 2", storeResult=var "Pt Wood 2")
    FindChild(gameObject=Self, childName="Pt Wood 3", storeResult=var "Pt Wood 3")
    FindChild(gameObject=Self, childName="Solid", storeResult=var "Solid")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activated
  }

  state Break {
    SendEventByName(eventTarget=GameObject(var "mask"), sendEvent="HIT", delay=0, everyFrame=false)
    SetParent(gameObject=var "mask", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust Hit 3", emit=0)
    SetParent(gameObject=var "Dust Hit 3", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=barrel_death_1 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=breakable_wall_death (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Bits 3", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Wood 3", emit=0)
    ActivateGameObject(gameObject=var "Floor 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wood 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wood 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Solid", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false)
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Large (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=3, speedMin=0, speedMax=0, angleMin=0, angleMax=0, originVariationX=2, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=8, spawnMax=8, speedMin=0, speedMax=0, angleMin=0, angleMax=0, originVariationX=2, originVariationY=1, FSM="", FSMEvent="")
  }

  state Hit 1 {
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(0, -2, 0), rotation=(unset), storeObject=var "Strike Effect")
    Translate(gameObject=var "Floor 1", vector=(unset), x=(unset), y=-0.05, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Floor 2", vector=(unset), x=(unset), y=-0.1, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Rotate(gameObject=var "Floor 1", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-2.5, space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Rotate(gameObject=var "Floor 2", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=3.5, space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=barrel_death_1 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust Hit 1", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Bits 1", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Wood 1", emit=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit 2 {
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(0, -2, 0), rotation=(unset), storeObject=var "Strike Effect")
    Translate(gameObject=var "Floor 1", vector=(unset), x=(unset), y=-0.1, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Floor 2", vector=(unset), x=(unset), y=-0.2, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Rotate(gameObject=var "Floor 1", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-2.5, space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Rotate(gameObject=var "Floor 2", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=2.5, space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=barrel_death_1 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Translate(gameObject=var "Floor 2", vector=(unset), x=(unset), y=-0.15, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust Hit 2", emit=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Bits 2", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Wood 2", emit=0)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state Activated {
    DestroyAllChildren(gameObject=var "Self", disable=false)
    SetPolygonCollider(gameObject=Self, active=false)
  }

  var Attack Type: int = 0
  var Hits: int = 0
  var Activated: bool = false
  var Hero Range: bool = false
  var Damager: gameObject = <null>
  var Dust Hit 1: gameObject = <null>
  var Dust Hit 2: gameObject = <null>
  var Dust Hit 3: gameObject = <null>
  var Floor 1: gameObject = <null>
  var Floor 2: gameObject = <null>
  var Pt Bits 1: gameObject = <null>
  var Pt Bits 2: gameObject = <null>
  var Pt Bits 3: gameObject = <null>
  var Pt Wood 1: gameObject = <null>
  var Pt Wood 2: gameObject = <null>
  var Pt Wood 3: gameObject = <null>
  var Self: gameObject = <null>
  var Solid: gameObject = <null>
  var Strike Effect: gameObject = <null>
  var Wood 1: gameObject = <null>
  var Wood 2: gameObject = <null>
  var mask: gameObject = <null>
}
