# Darkness Region
# 1 location(s):
#   GG_Atrium_Roof / GG_Arena_Prefab/Darkness Region

fsm Darkness Region {
  start Pause

  state Init {
    FindGameObject(objectName="_SceneManager", withTag="SceneManager", store=var "Scene Manager")
    FindGameObject(objectName="Vignette", withTag="Vignette", store=var "Vignette")
    CallMethodProper(gameObject=var "Scene Manager", behaviour="SceneManager", methodName="GetDarknessLevel", parameters=[], storeResult=var "Scene Darkness")
    on FINISHED → Out
  }

  state Out {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter
  }

  state Enter {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SetDarkness(var "Darkness"))
    SetFsmInt(gameObject=var "Vignette", fsmName="Darkness Control", variableName="Darkness Level", setValue=var "Darkness", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Vignette"), sendEvent="SCENE RESET", delay=0, everyFrame=false)
    on EXIT → Exit
  }

  state Exit {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SetDarkness(var "Scene Darkness"))
    SetFsmInt(gameObject=var "Vignette", fsmName="Darkness Control", variableName="Darkness Level", setValue=var "Scene Darkness", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Vignette"), sendEvent="SCENE RESET", delay=0, everyFrame=false)
    on ENTER → Enter
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Darkness: int = 0

  var Scene Darkness: int = 0
  var Scene Manager: gameObject = <null>
  var Vignette: gameObject = <null>
}
