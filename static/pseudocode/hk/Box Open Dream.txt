# Box Open Dream
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/DialogueManager
#   resources.assets / _GameCameras/HudCamera/DialogueManager

fsm Box Open Dream {
  start Init

  state Box Down {
    Tk2dPlayAnimation(gameObject=var "Box", animLibName="", clipName="Dream Down")
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    FadeAudio(gameObject=var "Dialogue Audio", startVolume=1, endVolume=0, time=0.5)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Orbit Shield"), sendEvent="ORBIT SHIELD UP", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Box", active=true)
    on BOX UP DREAM → Box Up
    on FINISHED → Stop Audio
  }

  state Init {
    FindChild(gameObject=Self, childName="Box Dream", storeResult=var "Box")
    FindChild(gameObject=var "MainCamera", childName="Audio", storeResult=var "Audio")
    FindChild(gameObject=var "Audio", childName="Dream Dialogue", storeResult=var "Dialogue Audio")
    on BOX UP DREAM → Box Up
  }

  state Box Up {
    Tk2dPlayAnimation(gameObject=var "Box", animLibName="", clipName="Dream Up")
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=var "Dialogue Audio", volume=1, oneShotClip=<null>)
    FadeAudio(gameObject=var "Dialogue Audio", startVolume=0, endVolume=1, time=1)
    SendEventByName(eventTarget=GameObject(var "Orbit Shield"), sendEvent="ORBIT SHIELD DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemy Dream Msg"), sendEvent="CANCEL ENEMY DREAM", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Box", active=true)
    on BOX DOWN DREAM → Box Down
    on HERO DAMAGED → Box Down
  }

  state Stop Audio {
    AudioStop(gameObject=var "Dialogue Audio")
    SetMeshRenderer(gameObject=var "Box", active=false)
    on BOX UP DREAM → Box Up
  }

  var Audio: gameObject = <null>
  var Box: gameObject = <null>
  var Dialogue Audio: gameObject = <null>
}
