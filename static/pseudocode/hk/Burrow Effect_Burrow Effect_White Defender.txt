# Burrow Effect
# 3 location(s):
#   Dream_04_White_Defender / White Defender/Burrow Effect
#   GG_White_Defender / White Defender/Burrow Effect
#   sharedassets398.assets / White Defender:0/Burrow Effect

fsm Burrow Effect {
  start Init
  on DUNG DEFENDER DEFEAT → Defeat  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", z=(unset), space=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Particles", storeResult=var "Particles")
    on FINISHED → Inactive
  }

  state Inactive {
    SetMeshRenderer(gameObject=Self, active=false)
    on BURROW START → Burrowing
  }

  state Burrowing {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Burrow Effect")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)  // disabled
    PlayParticleEmitter(gameObject=var "Particles", emit=0)
    on BURROW END → End
  }

  state End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burrow End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)  // disabled
    StopParticleEmitter(gameObject=var "Particles")
    on FINISHED → Inactive
  }

  state Defeat {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Ground Y: float = 0
  var Particles: gameObject = <null>
}
