  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Battle Sub", storeResult=var "Battle Sub")
    FindChild(gameObject=Self, childName="Mantis Lord Throne 1", storeResult=var "Throne 1")
    FindChild(gameObject=Self, childName="Mantis Lord Throne 2", storeResult=var "Throne 2")
    FindChild(gameObject=Self, childName="Mantis Lord Throne 3", storeResult=var "Throne 3")
    FindChild(gameObject=var "Battle Sub", childName="Mantis Lord S1", storeResult=var "Lord S1")
    FindChild(gameObject=var "Battle Sub", childName="Mantis Lord S2", storeResult=var "Lord S2")
    SetFsmGameObject(gameObject=var "Lord S1", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Self", everyFrame=false)  // disabled
    SetFsmGameObject(gameObject=var "Lord S2", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Self", everyFrame=false)  // disabled
    on FINISHED → Idle
  }

  state Idle {
    IntCompare(integer1=var "Battle Enemies", integer2=1, equal=→"KILL1", lessThan=→"KILL1", greaterThan=(none), everyFrame=true)
    on KILL1 → Return 1
  }

  state Return 1 {
    SendEventByName(eventTarget=GameObject(var "Throne 3"), sendEvent="RETURN", delay=0, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"KILL2", lessThan=→"KILL2", greaterThan=(none), everyFrame=true)
    on KILL2 → Journal
  }

  state Return 2 {
    SendEventByName(eventTarget=GameObject(var "Throne 1"), sendEvent="RETURN", delay=0, everyFrame=false)
    Wait(time=3.75, finishEvent=→"WAIT", realTime=false)
    SetPlayerDataBool(boolName="defeatedMantisLords", value=true)  // disabled
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    on FINISHED → Bow
  }

  state Bow {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=StoryRecord_defeated("Mantis Lords"))
    Tk2dPlayAnimation(gameObject=var "Throne 1", animLibName="", clipName="Throne Bow")
    Tk2dPlayAnimation(gameObject=var "Throne 2", animLibName="", clipName="Throne Bow")
    Tk2dPlayAnimation(gameObject=var "Throne 3", animLibName="", clipName="Throne Bow")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MANTIS DEFEAT", delay=0, everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Self", audioClip=Boss Defeat (sharedassets40.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=3, finishEvent=→"WAIT", realTime=false)
    on WAIT → Floor + Cage
    on GG BOSS → GG
  }

  state Floor + Cage {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["MANTIS_LORDS"], storeResult=(unset var))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD CAGE UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD FLOOR EXTEND", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MLORD GATE OPEN", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_gate (sharedassets30.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=misc_rumble_impact (sharedassets13.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  state Journal {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    SetPlayerDataBool(boolName="killedMantisLord", value=true)
    SetPlayerDataBool(boolName="newDataMantisLord", value=true)
    SetPlayerDataInt(intName="killsMantisLord", value=0)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasJournal", isTrue=(none), isFalse=→"FINISHED")
    CreateObject(gameObject=Journal Update Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Return 2
  }

  state GG {
    EndGGBossScene()
  }

  var Battle Enemies: int = 2
  var Battle Sub: gameObject = <null>
  var Lord S1: gameObject = <null>

  var Throne 2: gameObject = <null>
  var Throne 3: gameObject = <null>
}
