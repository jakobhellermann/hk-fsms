fsm Fungus Zombie Attack {
  start Initialise
  on ZERO HP → Death  // from any state

  state Initialise {
    GetHero(storeResult=var "")
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Gas Hit Box", storeResult=var "Gas Hit")
    FindChild(gameObject=Self, childName="Lines", storeResult=var "Lines")
    FindChild(gameObject=Self, childName="Gas Attack", storeResult=var "Gas Particles")
    FindChild(gameObject=Self, childName="Attack Range", storeResult=var "Attack Region")
    FindChild(gameObject=Self, childName="Gas Antic", storeResult=var "Gas Antic")
    FindChild(gameObject=Self, childName="Gas Idle", storeResult=var "Gas Idle")
    on FINISHED → Ready
  }

  state Ready {
    CheckCanSeeHero(storeResult=var "Can See Hero", everyFrame=true)
    CheckAlertRangeByName(storeResult=var "In Attack Range", childName="Attack Range", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Attack Range"], sendEvent=→"ATTACK ALERT", storeResult=(unset), everyFrame=true)
    on ATTACK ALERT → Attack Delay
  }

  state Attack Antic {
    SetParticleEmission(gameObject=var "Gas Idle", emission=false)
    SetParticleEmission(gameObject=var "Gas Antic", emission=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mossman_shaker_explode_prepare (sharedassets128.assets))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mosskin_attack_01 (sharedassets128.assets), Mosskin_attack_02 (sharedassets128.assets), Mosskin_attack_03 (sharedassets128.assets)], weights=[1, 1, 1], pitchMin=1.1, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Attack
  }

  state Attack {
    ActivateAllChildren(gameObject=var "Lines", activate=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=zombie_fungus_a_explode (sharedassets128.assets))
    PlayParticleEmitter(gameObject=var "Gas Particles", emit=0)
    SetParticleEmission(gameObject=var "Gas Antic", emission=false)
    ActivateGameObject(gameObject=var "Gas Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetScale(gameObject=var "Gas Hit", vector=(unset), x=0.2, y=0.2, z=0.2, everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=var "Gas Hit", id="", transformScale=var "", vectorScale=(1, 1, 1), time=0.4, delay=0.005, speed=(unset), easeType=easeOutCirc, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → CD
  }

  state CD {
    ActivateGameObject(gameObject=var "Gas Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle Pause
  }

  state Idle Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetParticleEmission(gameObject=var "Gas Idle", emission=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset
  }

  state Reset {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on FINISHED → Ready
  }

  state Death {
    StopParticleEmitter(gameObject=var "Gas Antic")
    StopParticleEmitter(gameObject=var "Gas Idle")
    DestroyObject(gameObject=var "Gas Hit", delay=0, detachChildren=false)
    StopParticleEmitter(gameObject=var "Gas Particles")
    DetachChildren(gameObject=Self)
  }

  state Attack Delay {
    WaitRandom(timeMin=0, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack Antic
  }

  var Can See Hero: bool = false
  var In Attack Range: bool = false
  var Attack Region: gameObject = <null>
  var Gas Antic: gameObject = <null>
  var Gas Hit: gameObject = <null>
  var Gas Idle: gameObject = <null>
  var Gas Particles: gameObject = <null>
  var Lines: gameObject = <null>
  var Self: gameObject = <null>
}
