# Check Journal Placement
# 1 location(s):
#   Fungus1_08 / Hunter Eyes

fsm Check Journal Placement {
  start Init

  state Init {
    on FINISHED → Check Journal
  }

  state Check Journal {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="metHunter", falseBool="hasJournal", isTrue=→"PLACE", isFalse=(none))
    on PLACE → Place
  }

  state Place {
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Shiny Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Shiny Item: gameObject = Hunter Eyes/Shiny Item
}
