# Conversation Control
# 1 location(s):
#   Crossroads_46b / Tram Call Box

fsm Conversation Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Check Tram {
    GetPlayerDataInt(gameObject=var "GameManager", intName=var "PD Int", storeValue=var "Tram Position")
    IntCompare(integer1=var "Tram Position", integer2=var "Station Position", equal=→"HERE", lessThan=→"AWAY", greaterThan=→"AWAY", everyFrame=false)
    on HERE → Tram Here
    on AWAY → Tram Away
  }

  state Tram Here {
    SetCollider(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Lit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Tram Away {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    on CONVO START → Got Pass?
  }

  state Got Pass? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasTramPass", isTrue=→"YES", isFalse=→"NO")
    on YES → Box Up YN
    on NO → Box Up
  }

  state No Pass {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["TRAM_CALL_NOPASS", "Prompts"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → No Pass
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    on FINISHED → Tram Away
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["TRAM_CALL", "Prompts"], storeResult=(unset var))
    on NO → Box Down YN
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetPlayerDataInt(intName=var "PD Int", value=var "Station Position")
    SetPlayerDataBool(boolName=var "Activated Bool", value=true)
    on FINISHED → Flash
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check Tram
  }

  state Box Down YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Flash {
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=white_seal_1 (sharedassets19.assets))
    on FINISHED → Start Shake
  }

  state Start Shake {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=true, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=jiji_door_open_pt_2 (sharedassets7.assets))
    on FINISHED → Call Tram
  }

  state Call Tram {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CALL TRAM", delay=0, everyFrame=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Shake
  }

  state Tram Called {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
  }

  state Stop Shake {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    on TRAM ARRIVE → Tram Called
  }

  var Station Position: int = 1
  var Activated Bool: string = "openedTramRestingGrounds"
  var PD Int: string = "tramRestingGroundsPosition"

  var Something Cost: int = 0
  var Tram Position: int = 0
  var Flash: gameObject = Tram Call Box/Flash
  var Lit: gameObject = Tram Call Box/Lit
  var Self: gameObject = <null>
}
