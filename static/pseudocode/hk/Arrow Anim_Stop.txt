# Arrow Anim
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/DialogueManager/Text/Stop
#   resources.assets / _GameCameras/HudCamera/DialogueManager/Text/Stop

fsm Arrow Anim {
  start Idle

  state Idle {
    SetMeshRenderer(gameObject=Self, active=false)
    on UP → Up
  }

  state Up {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stop Up")
    on DOWN → Down
  }

  state Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stop Down")
    on UP → Up
  }
}
