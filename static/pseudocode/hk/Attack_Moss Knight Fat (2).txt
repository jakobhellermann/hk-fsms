# Attack
# 3 location(s):
#   Fungus3_39 / Moss Knight Fat (2)
#   Fungus3_39 / Moss Knight Fat
#   Fungus3_39 / Moss Knight Fat (1)

// component disabled: this FSM does not run until something enables it
fsm Attack {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Particle", storeResult=var "Particle")
    BoolTest(boolVariable=var "Praying", isTrue=→"LOOK AT", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Bounce On Spot
    on LOOK AT → Pray
  }

  state Idle {
    SetFloatValue(floatVariable=var "Speed", floatValue=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetIntValue(intVariable=var "Bounces", intValue=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    BoolTest(boolVariable=var "HERO IN RANGE", isTrue=→"HERO IN RANGE", isFalse=(none), everyFrame=true)
    on HERO IN RANGE → Wait
  }

  state Wait {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Speed
  }

  state Run {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=var "Self", vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    GetDistance(gameObject=var "Hero", target=var "Self", storeResult=var "Distance", everyFrame=true)
    SetFloatValue(floatVariable=var "Speed", floatValue=var "Run Speed", everyFrame=false)
    FloatMultiplyV2(floatVariable=var "Speed", multiplyBy=var "Scale", everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=true)
    FloatCompare(float1=var "Distance", float2=10, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Wait to jump
  }

  state Jump Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump Start")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    RandomInt(min=3, max=4, storeResult=var "Max Bounces", inclusiveMax=true)
    FloatMultiplyV2(floatVariable=var "Speed", multiplyBy=3, everyFrame=false, fixedUpdate=false)  // disabled
    FloatMultiplyV2(floatVariable=var "Speed", multiplyBy=var "Scale", everyFrame=false, fixedUpdate=false)  // disabled
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStop(gameObject=Self)
    on FINISHED → Wait Land
  }

  state Wait to jump {
    WaitRandom(timeMin=0.1, timeMax=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Jump Start
  }

  state Wait Land {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_fat_land (sharedassets219.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Jump Land", doNotResetCurrentClip=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Launch
  }

  state Jump Air {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"SIDE BOUNCE", bottomHitEvent=→"BOUNCE", leftHitEvent=→"SIDE BOUNCE", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on BOUNCE → Bounce
    on SIDE BOUNCE → Reflect
  }

  state Bounce {
    IntAddV2(intVariable=var "Bounces", add=1, everyFrame=false)
    IntCompare(integer1=var "Bounces", integer2=var "Max Bounces", equal=→"LAND", lessThan=→"BOUNCE", greaterThan=→"LAND", everyFrame=false)
    on BOUNCE → Wait Land
    on LAND → Landing
  }

  state Reflect {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Vel", y=(unset), space=0, everyFrame=false)
    FloatMultiplyV2(floatVariable=var "X Vel", multiplyBy=-1, everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Vel", y=(unset), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Jump Air
  }

  state Wait 2 {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Idle", doNotResetCurrentClip=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Landing {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_fat_land (sharedassets219.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Jump Land", doNotResetCurrentClip=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"LAND")
    on LAND → Wait 2
  }

  state Pray {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest")
    BoolTest(boolVariable=var "Alert Range", isTrue=→"LOOK AT", isFalse=(none), everyFrame=true)
    on LOOK AT → Look At
    on TOOK DAMAGE → Look At
    on MOSS FAT WAKE → Look At
  }

  state Look At {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look At")  // disabled
    PlayParticleEmitter(gameObject=var "Particle", emit=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MOSS FAT WAKE", delay=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    BoolTest(boolVariable=var "Alert Range", isTrue=(none), isFalse=→"FINISHED", everyFrame=true)  // disabled
    BoolTest(boolVariable=var "Bounce Range", isTrue=→"BOUNCE", isFalse=(none), everyFrame=true)  // disabled
    on FINISHED → Bounce Begin
  }

  state Bounce On Spot {
    IntAdd(intVariable=var "Start Bounces", add=1, everyFrame=false)
    IntCompare(integer1=var "Start Bounces", integer2=var "Max Start Bounces", equal=→"ATTACK", lessThan=(none), greaterThan=→"ATTACK", everyFrame=false)
    on ATTACK → Idle
    on FINISHED → Bounce Start
  }

  state Bounce Begin {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look Jump Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Bounce Start
  }

  state Bounce Start {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Jump Start", doNotResetCurrentClip=false)
    RandomFloat(min=14, max=20, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Jump Y", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bounce Air
  }

  state Bounce Air {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_fat_jump (sharedassets219.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump Air")
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"FINISHED", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on FINISHED → Bounce Land
  }

  state Bounce Land {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_fat_land (sharedassets219.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Jump Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Bounce On Spot
  }

  state Look Back {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look Back", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Pray
  }

  state Set Speed {
    RandomFloat(min=-4, max=-6, storeResult=var "Run Speed")
    on FINISHED → Run
  }

  state Launch {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_fat_jump (sharedassets219.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Jump Air", doNotResetCurrentClip=false)
    RandomFloat(min=10, max=16, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=var "Jump Y", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Jump Air
  }

  var Praying: bool = true

  var Distance: float = 0
  var Jump Y: float = 0
  var Run Speed: float = -5
  var Scale: float = 0
  var Speed: float = 0
  var X Vel: float = 0
  var Bounces: int = 0
  var Max Bounces: int = 0
  var Max Start Bounces: int = 3
  var Start Bounces: int = 0
  var Alert Range: bool = false
  var Bounce Range: bool = false
  var HERO IN RANGE: bool = false
  var Particle: gameObject = <null>
  var Self: gameObject = <null>
}
