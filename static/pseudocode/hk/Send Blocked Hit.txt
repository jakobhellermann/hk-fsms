# Send Blocked Hit
# 1 location(s):
#   Deepnest_41 / Happy Spider NPC/Damager

fsm Send Blocked Hit {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    on BLOCKED HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="BLOCKED HIT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Parent: gameObject = <null>
}
