# Remove
# 1 location(s):
#   Abyss_06_Core / Inspect Region

fsm Remove {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue Health")
    IntCompare(integer1=var "Blue Health", integer2=0, equal=(none), lessThan=(none), greaterThan=→"OFF", everyFrame=false)
    on OFF → Off
  }

  state Off {
    SetCollider(gameObject=Self, active=false)
  }

  var Blue Health: int = 0
}
