# Effect Control
# 5 location(s):
#   Dream_01_False_Knight / False Knight Dream/Armour Hit
#   GG_Failed_Champion / False Knight Dream/Armour Hit
#   GG_False_Knight / Battle Scene/False Knight New/Armour Hit
#   Crossroads_10_boss / Battle Scene/False Knight New/Armour Hit
#   GG_Mighty_Zote / Battle Control/Zote Thwomp/Armour Hit

fsm Effect Control {
  start Idle

  state Idle {
    SetParticleEmission(gameObject=Self, emission=false)
    on ARMOUR HIT U → Armour Hit U
    on ARMOUR HIT D → Armour Hit D
    on ARMOUR HIT L → Armour Hit R
    on ARMOUR HIT R → Armour Hit L
  }

  state Emitting {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmission(gameObject=Self, emission=true)
    on FINISHED → Idle
    on ARMOUR HIT U → Armour Hit U
    on ARMOUR HIT L → Armour Hit R
    on ARMOUR HIT D → Armour Hit D
    on ARMOUR HIT R → Armour Hit L
  }

  state Armour Hit U {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=45, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Emitting
  }

  state Armour Hit D {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=225, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Emitting
  }

  state Armour Hit L {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=315, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Emitting
  }

  state Armour Hit R {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=135, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Emitting
  }
}
