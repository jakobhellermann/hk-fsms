# zap control
# 1 location(s):
#   Fungus3_archive_02 / Zap Cloud (6)

fsm zap control {
  start Pause

  state Init {
    GetHero(storeResult=var "")
    GetOwner(storeGameObject=var "Self")
    SetFloatValue(floatVariable=var "Waiter", floatValue=var "Waiter Start", everyFrame=false)
    on FINISHED → Ready
  }

  state Ready {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetFsmFloat(gameObject=Self, fsmName="zap drift", variableName="Speed Max", setValue=0.8, everyFrame=false)
    FloatAdd(floatVariable=var "Waiter", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Waiter", float2=var "Wait Time", tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Close Check
  }

  state Attack Antic {
    SetFsmFloat(gameObject=Self, fsmName="zap drift", variableName="Speed Max", setValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Waiter", floatValue=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=zap_bug_pt1 (sharedassets34.assets))
    on FINISHED → Attack
  }

  state Attack {
    PlayParticleEmitter(gameObject=var "Zaps", emit=0)
    SetCircleCollider(gameObject=Self, active=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=zap_bug_pt2 (sharedassets34.assets))
    on FINISHED → Attack CD
  }

  state Attack CD {
    StopParticleEmitter(gameObject=var "Zaps")
    SetCircleCollider(gameObject=Self, active=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Ready
  }

  state Close Check {
    SetFloatValue(floatVariable=var "Waiter", floatValue=0, everyFrame=false)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=9.5, tolerance=0, equal=→"READY", lessThan=→"READY", greaterThan=(none), everyFrame=true)
    on READY → Attack Antic
    on FINISHED → Ready
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Wait Time: float = 0.8
  var Waiter Start: float = 0

  var Distance: float = 0
  var Waiter: float = 0
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Zaps: gameObject = Zap Cloud (6)/Zaps
}
