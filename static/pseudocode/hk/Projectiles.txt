# Projectiles
# 1 location(s):
#   Room_Colosseum_Spectate / Spectate Effects

fsm Projectiles {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Pause
  }

  state Pause {
    WaitRandom(timeMin=1, timeMax=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Spawn
  }

  state Type {
    SendRandomEvent(events=[→"SICKLE", →"SLASH", →"TINK"], weights=[1, 1, 1], delay=0)
    on SICKLE → Sickle
    on SLASH → Slash
    on TINK → Tink
  }

  state Set Spawn {
    RandomFloat(min=var "Spawn X Min", max=var "Spawn X Max", storeResult=var "Spawn X")
    RandomFloat(min=var "Spawn Y Min", max=var "Spawn Y Max", storeResult=var "Spawn Y")
    SetVector3XYZ(vector3Variable=var "Spawn Vector", vector3Value=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), everyFrame=false)
    on FINISHED → Type
  }

  state Sickle {
    FlingObjectsFromGlobalPool(gameObject=Shot GladiatorSickle (sharedassets32.assets), spawnPoint=<null>, position=var "Spawn Vector", spawnMin=1, spawnMax=1, speedMin=15, speedMax=20, angleMin=20, angleMax=70, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[zombie_miner_throw (sharedassets32.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=0.75, delay=0, storePlayer=<null>)
    on FINISHED → Pause
  }

  state Slash {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[ruin_sentry_sword_1 (sharedassets33.assets), ruin_sentry_sword_2 (sharedassets33.assets), ruin_sentry_sword_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[ruin_sentry_sword_1 (sharedassets33.assets), ruin_sentry_sword_2 (sharedassets33.assets), ruin_sentry_sword_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0.5, storePlayer=<null>)
    on FINISHED → Pause
  }

  state Tink {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[ruin_sentry_sword_1 (sharedassets33.assets), ruin_sentry_sword_2 (sharedassets33.assets), ruin_sentry_sword_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[hero_parry (resources.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0.25, storePlayer=<null>)
    on FINISHED → Pause
  }

  var Spawn X: float = 0
  var Spawn X Max: float = 112.5
  var Spawn X Min: float = 108
  var Spawn Y: float = 0
  var Spawn Y Max: float = 11.99
  var Spawn Y Min: float = 8.43
  var Wounds: int = 0
  var Spawn Vector: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
  var Shot: gameObject = <null>
}
