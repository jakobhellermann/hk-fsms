# Conversation Control
# 1 location(s):
#   Fungus3_49 / gg_white_flower/gg_white_flower_collect

fsm Conversation Control {
  start Pause
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasXunFlower", isTrue=→"INACTIVE", isFalse=→"FINISHED")
    on FINISHED → Idle
    on INACTIVE → Inactive
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Idle")
    on FINISHED → Box Down
  }

  state Hero Anim {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hero Anim 2
  }

  state Activate {
    SetCollider(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    on FINISHED → 
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inactive {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFlowerBroken", isTrue=→"NO", isFalse=(none))
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on NO → Idle
  }

  state Hero Anim 2 {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Get Item
  }

  state Get Item {
    PlayParticleEmitter(gameObject=gg_white_flower/gg_white_flower_collect/Spa Glows, emit=0)
    SetPlayerDataBool(boolName="hasXunFlower", value=true)
    SetPlayerDataBool(boolName="xunFlowerBroken", value=false)
    SpawnObjectFromGlobalPool(gameObject=Relic Get Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg")
    FindChild(gameObject=var "Msg", childName="Text", storeResult=var "Msg Text")
    FindChild(gameObject=var "Msg", childName="Icon", storeResult=var "Msg Item")
    SetSpriteRendererSprite(gameObject=var "Msg Item", sprite=White_Flower_Full (resources.assets))
    GetLanguageString(sheetName="UI", convName="INV_NAME_FLOWER", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero Anim 3
  }

  state Hero Anim 3 {
    SetSpriteRenderer(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Activate
  }

  var Decrement: int = -1
  var Simple Keys: int = 0
  var Item Name: string = ""
  var Godseeker: gameObject = Godseeker Waterways/Godseeker Awake (sharedassets225.assets)
  var Godseeker Fall: gameObject = Godseeker Waterways/Godseeker Fall (sharedassets225.assets)
  var Key: gameObject = Godseeker Waterways/Coffin/GG_Waterways_Coffin_0003_1/Key (sharedassets225.assets)
  var Msg: gameObject = <null>
  var Msg Item: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Pt Shake: gameObject = <null>
  var Self: gameObject = <null>
  var Unlock Glow: gameObject = <null>
}
