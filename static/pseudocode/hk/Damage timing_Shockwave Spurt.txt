# Damage timing
# 1 location(s):
#   sharedassets48.assets / Shockwave Spurt

fsm Damage timing {
  start Wait

  state Wait {
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  state Activate {
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Deactivate
  }

  state Deactivate {
    SetDamageHeroAmount(target=Self, damageDealt=0)
  }
}
