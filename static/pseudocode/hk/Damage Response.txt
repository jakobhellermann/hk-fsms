# Damage Response
# 3 location(s):
#   Fungus1_35 / Warrior/Ghost Warrior No Eyes
#   GG_Ghost_No_Eyes / Warrior/Ghost Warrior No Eyes
#   GG_Ghost_No_Eyes_V / Warrior/Ghost Warrior No Eyes

fsm Damage Response {
  start Idle

  state Idle {
    on TOOK DAMAGE → Decide
  }

  state Decide {
    SendRandomEvent(events=[→"FINISHED", →"SEND"], weights=[0.65, 0.35], delay=0)
    on FINISHED → Idle
    on SEND → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WARP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }
}
