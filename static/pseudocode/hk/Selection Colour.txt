// uses template: map_select_colour
fsm Selection Colour {
  start Init

  state Unselected {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP", delay=0, everyFrame=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(0.3602941, 0.3602941, 0.3602941, 1), everyFrame=false)
    SetTextMeshProColor(gameObject=var "Area Name", color=(0.35686275, 0.35686275, 0.35686275, 1), everyFrame=false)
    on SELECT → Selected
    on UNSELECT → Unselected
  }

  state Selected {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP", delay=0, everyFrame=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    SetTextMeshProColor(gameObject=var "Area Name", color=(1, 1, 1, 1), everyFrame=false)
    on UNSELECT → Unselected
    on SELECT → Selected
  }

  state Init {
    FindChild(gameObject=Self, childName="Area Name", storeResult=var "Area Name")
    on SELECT → Selected
    on UNSELECT → Unselected
  }

  var Area Name: gameObject = <null>
}
