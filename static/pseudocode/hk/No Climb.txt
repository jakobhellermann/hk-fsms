fsm No Climb {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "Enemy")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "Enemy")
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SendEventByName(eventTarget=GameObject(var "Enemy"), sendEvent="NO CLIMB ENTER", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Exit {
    SendEventByName(eventTarget=GameObject(var "Enemy"), sendEvent="NO CLIMB EXIT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Enemy: gameObject = <null>
}
