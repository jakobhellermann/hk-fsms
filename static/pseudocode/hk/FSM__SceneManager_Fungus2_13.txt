# FSM
# 1 location(s):
#   Fungus2_13 / _SceneManager

// uses template: map_manual_tilemap
fsm FSM {
  start Pause
  on REFRESH → Pause  // from any state

  state Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Set
  }

  state Set {
    CallMethodProper(gameObject=var "Game Map", behaviour="GameMap", methodName="SetManualTilemap", parameters=[var "Offset X", var "Offset Y", var "Width", var "Height"], storeResult=(unset var))
    SetPlayerDataFloat(gameObject=var "GameManager", floatName="gMap_doorOriginOffsetX", value=var "Offset X")
    SetPlayerDataFloat(gameObject=var "GameManager", floatName="gMap_doorOriginOffsetY", value=var "Offset Y")
    SetPlayerDataFloat(gameObject=var "GameManager", floatName="gMap_doorSceneWidth", value=var "Width")
    SetPlayerDataFloat(gameObject=var "GameManager", floatName="gMap_doorSceneHeight", value=var "Height")
  }

  var Height: float = 116
  var Offset X: float = 0
  var Offset Y: float = 0
  var Width: float = 41
}
