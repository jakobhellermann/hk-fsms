# Control
# 1 location(s):
#   Fungus3_archive_02 / Dreamer Monomon/Particle Control

fsm Control {
  start Init

  state Init {
    Wait(time=1, finishEvent=→"INACTIVE", realTime=false)
    on INACTIVE → Inactive
  }

  state Inactive {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ACTIVE", storeCollider=var "")
    on ACTIVE → Active
  }

  state Active {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"INACTIVE", storeCollider=var "")
    on INACTIVE → Inactive
  }

  var Self: gameObject = <null>
}
