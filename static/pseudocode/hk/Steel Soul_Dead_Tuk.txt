# Steel Soul
# 1 location(s):
#   Waterways_03 / Dead_Tuk

fsm Steel Soul {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="permadeathMode", storeValue=var "Permadeath Mode")
    IntCompare(integer1=var "Permadeath Mode", integer2=1, equal=(none), lessThan=→"INACTIVE", greaterThan=(none), everyFrame=false)
    on INACTIVE → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Permadeath Mode: int = 0
}
