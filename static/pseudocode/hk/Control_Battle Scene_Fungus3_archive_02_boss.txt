# Control
# 1 location(s):
#   Fungus3_archive_02_boss / Battle Scene

fsm Control {
  start Pause

  state Idle {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedMegaJelly", isTrue=→"ACTIVATE", isFalse=(none))
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Start
    on ACTIVATE → Activate
  }

  state Start {
    SetPlayerDataBool(boolName="encounteredMegaJelly", value=true)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    on BOSS DEATH → End Pause
  }

  state End {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Activate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  state End Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  var Self: gameObject = <null>
}
