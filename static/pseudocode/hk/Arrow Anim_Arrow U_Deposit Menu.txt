fsm Arrow Anim {
  start Init

  state Init {
    SetVector3Value(vector3Variable=var "Move Pos", vector3Value=var "Init Pos", everyFrame=false)
    SetVector3Value(vector3Variable=var "Idle Pos", vector3Value=var "Init Pos", everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Move Pos", addX=0, addY=0.15, addZ=0, everyFrame=false, perSecond=false)
    Vector3AddXYZ(vector3Variable=var "Idle Pos", addX=0, addY=0.7, addZ=0, everyFrame=false, perSecond=false)
    on FINISHED → Idle
  }

  state Idle {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Init Pos", time=0.1, delay=0, speed=(unset), space=1, easeType=easeOutSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=false)
    Wait(time=1, finishEvent=→"IDLE", realTime=false)
    on MOVE → Up
    on IDLE → 
  }

  state Up {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Move Pos", time=0.05, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Idle
    on MOVE → Repeat
  }

  state Repeat {
    on FINISHED → Up
  }

  var Idle Pos: vector3 = (0, 0, 0)
  var Init Pos: vector3 = (-3.41, -1.72, -1)
  var Move Pos: vector3 = (-6.83, -0.92, -1)
}
