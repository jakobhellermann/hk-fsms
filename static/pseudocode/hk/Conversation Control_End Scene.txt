# Conversation Control
# 1 location(s):
#   White_Palace_20 / End Scene

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="CamLock", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="initial_loop", storeResult=var "initial_loop")
    FindChild(gameObject=Self, childName="watch_scene", storeResult=var "watch_scene")
    on FINISHED → Idle
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Look At King")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="FaceLeft", parameters=[], storeResult=(unset var))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Disappear Scne
  }

  state Journal {
    SetPlayerDataBool(boolName="killedBindingSeal", value=true)
    SetPlayerDataBool(boolName="newDataBindingSeal", value=true)
    SetPlayerDataInt(intName="killsBindingSeal", value=0)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasJournal", isTrue=(none), isFalse=→"FINISHED")
    CreateObject(gameObject=Journal Update Msg (resources.assets), spawnPoint=<null>, position=(0, 0, -4), rotation=(unset), storeObject=var "Msg", networkInstantiate=false, networkGroup=0)
    SetFsmVector3(gameObject=var "Msg", fsmName="Journal Msg", variableName="Position", setValue=(-11.91, -6.22, -4), everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → New Scene
  }

  state Orb Away {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="FRIENDS OUT", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=dream_plant_emit_orbs (sharedassets228.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Out
  }

  state Fade Out {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WHITE PALACE END", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_enter (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=5.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Journal
  }

  state New Scene {
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)  // disabled
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    SetTextMeshProColor(gameObject=var "DialogueText", color=(1, 1, 1, 1), everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=["White_Palace_06", "door_dreamReturn", 0], storeResult=(unset var))  // disabled
    BeginSceneTransition(sceneName="White_Palace_06", entryGateName="door_dreamReturn", entryDelay=0, visualization=Dream, preventCameraFadeOut=true)
  }

  state Disappear Scne {
    ActivateGameObject(gameObject=var "initial_loop", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "watch_scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=10, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Out
  }

  var CamLock: gameObject = <null>
  var Msg: gameObject = <null>
  var Self: gameObject = <null>
  var initial_loop: gameObject = <null>
  var watch_scene: gameObject = <null>
}
