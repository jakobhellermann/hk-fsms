# Send Hit Msg
# 1 location(s):
#   Room_Jinn / Jinn NPC/Tinker

fsm Send Hit Msg {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    on BLOCKED HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="HIT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Parent: gameObject = <null>
}
