  }

    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPAWN START", delay=0, everyFrame=false)
    Wait(time=0.6, finishEvent=→"HIT", realTime=false)
    on HIT → End Pause
    on WALL → End Pause
  }

  state End Particle {
    StopParticleEmitter(gameObject=var "Duster")
    StopParticleEmitter(gameObject=var "Burst Rocks Stomp")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPAWN END", delay=0, everyFrame=false)

  }

    SetGameObject(variable=var "Shockwave Object", gameObject=Shockwave Spurt L Z (sharedassets399.assets), everyFrame=false)

  }

    SetFsmGameObject(gameObject=Self, fsmName="Spawn", variableName="Shockwave Object", setValue=var "Shockwave Object", everyFrame=false)

  }

  }
