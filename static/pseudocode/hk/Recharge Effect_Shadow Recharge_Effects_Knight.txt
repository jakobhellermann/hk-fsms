# Recharge Effect
# 1 location(s):
#   Knight_Pickup / Knight/Effects/Shadow Recharge

fsm Recharge Effect {
  start Init
  on RESET → Init  // from any state

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    SetFloatValue(floatVariable=var "Wait time", floatValue=var "Shadow Recharge Time", everyFrame=false)
    FloatSubtract(floatVariable=var "Wait time", subtract=0.85, everyFrame=false, perSecond=false)
    on FINISHED → Wait
  }

  state Play anim {
    AudioPlay(gameObject=var "Knight", volume=1, oneShotClip=hero_shade_dash_charge_pt_1 (resources.assets), finishedEvent=(none))
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shadow Recharge", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Burst
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait {
    Wait(time=var "Wait time", finishEvent=→"WAIT", realTime=false)
    on WAIT → Play anim
  }

  state Burst {
    AudioPlay(gameObject=var "Knight", volume=1, oneShotClip=hero_shade_dash_charge_pt_2 (resources.assets), finishedEvent=(none))
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Deactivate
  }

  var Shadow Recharge Time: float = 1.5
  var Knight: gameObject = Knight

  var Wait time: float = 0
}
