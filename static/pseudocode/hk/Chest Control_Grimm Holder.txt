# Chest Control
# 1 location(s):
#   Grimm_Main_Tent / Grimm Holder

fsm Chest Control {
  start Init

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    FindChild(gameObject=Self, childName="Grimm Chest", storeResult=var "Grimm Chest")
    FindChild(gameObject=Self, childName="Grimm Chest Effect", storeResult=var "Grimm Chest Effect")
    FindChild(gameObject=Self, childName="Chest Tweener", storeResult=var "Chest Tweener")
    ActivateGameObject(gameObject=var "Grimm Chest", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PreBuildTK2DSprites(gameObject=var "Self", useChildren=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Grimm Killed?
  }

  state Grimm Killed? {
    ActivateGameObject(gameObject=var "Grimm Chest", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedGrimm", isTrue=→"KILLED", isFalse=(none))
    on KILLED → Killed
    on CHEST APPEAR → Appear
  }

  state Killed {
    ActivateGameObject(gameObject=var "Grimm Chest Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Grimm Chest", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Appear {
    ActivateGameObject(gameObject=var "Grimm Chest Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=grimm_teleport_in (sharedassets391.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    iTweenMoveBy(gameObject=var "Chest Tweener", id="", vector=(0, 4, 0), time=0.25, delay=0, speed=(unset), easeType=easeOutElastic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Chest Active
  }

  state Chest Active {
    ActivateGameObject(gameObject=var "Grimm Chest", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Chest Tweener", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Chest Tweener: gameObject = <null>
  var Grimm Chest: gameObject = <null>
  var Grimm Chest Effect: gameObject = <null>
  var Self: gameObject = <null>
}
