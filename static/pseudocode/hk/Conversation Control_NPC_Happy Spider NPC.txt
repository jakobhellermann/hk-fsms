# Conversation Control
# 1 location(s):
#   Deepnest_41 / Happy Spider NPC/NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MIDWIFE", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Meet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SPIDER_MEET", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="midwifeMet", value=true)
    on CONVO_FINISH → Talk Finish
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Convo Choice {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Midwife_talk_01 (sharedassets299.assets), Midwife_talk_02 (sharedassets299.assets), Midwife_talk_03 (sharedassets299.assets), Midwife_talk_04 (sharedassets299.assets), Midwife_talk_05 (sharedassets299.assets), Midwife_talk_06 (sharedassets299.assets)], weights=[1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="midwifeMet", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_39", falseBool="midwifeWeaverlingConvo", isTrue=→"WEAVERSONG", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasKingsBrand", falseBool="midwifeConvo2", isTrue=→"CONVO2", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="midwifeConvo1", isTrue=(none), isFalse=→"CONVO1")
    on MEET → Meet
    on CONVO1 → Convo1
    on CONVO2 → Convo2
    on FINISHED → Repeat
    on WEAVERSONG → Weaversong
  }

  state Convo1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SPIDER_GREET", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="midwifeConvo1", value=true)
    on CONVO_FINISH → Talk Finish
  }

  state Convo2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SPIDER_GREET2", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="midwifeConvo2", value=true)
    on CONVO_FINISH → Talk Finish
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SPIDER_REPEAT", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Weaversong {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MIDWIFE_WEAVERSONG", "CP2"], storeResult=(unset var))
    SetPlayerDataBool(boolName="midwifeWeaverlingConvo", value=true)
    on CONVO_FINISH → Talk Finish
  }

  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
