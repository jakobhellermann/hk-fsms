# Control
# 1 location(s):
#   Deepnest_39 / Spider Flyer (8)

fsm Control {
  start Initiate

  state Idle {
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName="Idle")
    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=1, speedMax=2, accelerationMin=5, accelerationMax=15, roamingRangeX=2, roamingRangeY=2, manualStartPos=(unset))
    FaceDirection(gameObject=var "Self", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToIdle", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=12, tolerance=0, equal=(none), lessThan=→"ALERT", greaterThan=(none), everyFrame=true)
    on ALERT → Chase Start
  }

  state Chase - In Sight {
    FaceDirection(gameObject=var "Self", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToChase", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=7, accelerationForce=15, offsetX=0, offsetY=0)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=12, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"WAIT", everyFrame=true)
    on WAIT → Chase - Out of Sight
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    BoolTest(boolVariable=var "Start Alert", isTrue=→"ALERT", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Spawns", isTrue=→"SPAWNS", isFalse=(none), everyFrame=false)
    on FINISHED → Deactivate ranges
    on ALERT → Deactivate ranges
    on SPAWNS → Spawn Wait
  }

  state Chase - Out of Sight {
    FaceDirection(gameObject=var "Self", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToChase", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=7, accelerationForce=15, offsetX=0, offsetY=0)
    Wait(time=var "Attention Span", finishEvent=→"WAIT", realTime=false)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=12, tolerance=0, equal=(none), lessThan=→"ALERT", greaterThan=(none), everyFrame=true)
    on ALERT → Chase - In Sight
    on WAIT → Stop
  }

  state Stop {
    SetAudioPitch(gameObject=var "Self", pitch=1, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName="Idle")
    Decelerate(gameObject=var "Self", deceleration=0.12)
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on WAIT → Idle
  }

  state Chase Start {
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName="Chase")
    on FINISHED → Chase - In Sight
  }

  state Spawn Wait {
    SetCollider(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on SPAWN → Spawn Pause
  }

  state Spawn Pause {
    WaitRandom(timeMin=0, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Saver
  }

  state Check Saver {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=3.5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SPAWN", everyFrame=true)
    on SPAWN → Spawn
  }

  state Spawn {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=weaver_spider_scream (sharedassets292.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    on FINISHED → Deactivate ranges
  }

  state Deactivate ranges {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Spawn Range Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Spawns: bool = true

  var Attention Span: float = 6
  var Distance: float = 0
  var Start Alert: bool = false
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Spawn Range Obj: gameObject = Spider Flyer (8)/Spawn Range
}
