# Detect Hero
# 2 location(s):
#   GG_Engine / Godseeker EngineRoom NPC/Turn Range
#   GG_Engine_Prime / Godseeker EngineRoom NPC/Turn Range

fsm Detect Hero {
  start Init

  state Out {
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="Turn Left")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="Turn Right")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
  }

  state Init {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
    on EXIT → Out
  }

  var Parent: gameObject = Godseeker EngineRoom NPC
}
