# Set
# 1 location(s):
#   Waterways_15 / Dung Defender_Awake/Convo Setter

fsm Set {
  start Pause

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Set
  }

  state Set {
    SetPlayerDataBool(boolName="dungDefenderAwakeConvo", value=true)
  }
}
