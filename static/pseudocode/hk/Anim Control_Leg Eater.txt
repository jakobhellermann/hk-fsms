# Anim Control
# 1 location(s):
#   Fungus2_26 / Leg Eater

fsm Anim Control {
  start Idle
  on REST → Rest  // from any state
  on SHOP TALK START → Talk Left  // from any state

  state Idle {
    GetOwner(storeGameObject=var "Self")
    AudioPlayInState(gameObject=Self, volume=1)
    on HERO LEFT → See Hero
  }

  state See Hero {
    AudioStop(gameObject=Self)
    AudioPlayRandom(gameObject=var "Self", audioClips=[Leg_Eater_alert_01 (sharedassets189.assets), Leg_Eater_alert_02 (sharedassets189.assets), Leg_Eater_alert_03 (sharedassets189.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="IdleToLookLeft")
    on HERO RIGHT → Left to Right
    on TALK START → Talk Left
  }

  state Left to Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LeftToRight")
    on HERO LEFT → RightToLeft
    on TALK START → Talk Right
  }

  state RightToLeft {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RightToLeft")
    on HERO RIGHT → Left to Right
    on TALK START → Talk Left
  }

  state Talk Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk Right")
    on TALK END → Right
  }

  state Talk Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk Left")
    on TALK END → Left
    on SHOP TALK END → Left
  }

  state Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Left")
    on HERO RIGHT → Left to Right
    on TALK START → Talk Left
  }

  state Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Right")
    on HERO LEFT → RightToLeft
    on TALK START → Talk Right
  }

  state Rest {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ToRest")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)  // disabled
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
}
