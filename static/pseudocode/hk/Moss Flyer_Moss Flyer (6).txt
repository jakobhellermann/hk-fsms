# Moss Flyer
# 1 location(s):
#   Fungus3_22 / Moss Flyer (6)

fsm Moss Flyer {
  start Init

  state Init {
    BoolTest(boolVariable=var "Start Flying", isTrue=→"WAKE", isFalse=(none), everyFrame=false)
    on FINISHED → Rest
    on WAKE → Start Flying
  }

  state Rest {
    BoolTest(boolVariable=var "Wake Range", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    on WAKE → Wake Pause
  }

  state Wake Pause {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  state Shake {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hide Grass", audioClip=moss_flyer_walker_emerge (sharedassets128.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shake")
    SetParticleEmissionRate(gameObject=var "Shake Grass", emissionRate=35, everyFrame=false)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetParticleEmissionRate(gameObject=var "Shake Grass", emissionRate=0, everyFrame=false)
    RandomFloat(min=3, max=5, storeResult=var "Hide Timer")
    on FINISHED → Activate
  }

  state Activate {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Move Up
  }

  state Move Up {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    FloatAddV2(floatVariable=var "Angle", add=90, everyFrame=false, perSecond=false, fixedUpdate=false, activeBool=(unset))
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=1, everyFrame=true)
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fly
  }

  state Fly {
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=2, acceleration=0.05, targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0)
    BoolTest(boolVariable=var "Wake Range", isTrue=(none), isFalse=→"IDLE", everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    on IDLE → Idle
  }

  state Idle {
    IdleBuzz(gameObject=Self, waitMin=0.75, waitMax=1, speedMax=1.75, accelerationMax=15, roamingRange=1)
    BoolTest(boolVariable=var "Wake Range", isTrue=→"FLY", isFalse=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    on FLY → Fly
  }

  state Start Flying {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Move Up
  }

  var Start Flying: bool = true

  var Angle: float = 0
  var Hide Timer: float = 0
  var X Scale: float = 0
  var Y Scale: float = 0
  var Wake Range: bool = false
  var LD Velocity: vector2 = (0, 0)
  var UR Velocity: vector2 = (0, 0)
  var Hide Grass: gameObject = Moss Flyer (6)/Hide Grass
  var Shake Grass: gameObject = Moss Flyer (6)/Shake Grass
}
