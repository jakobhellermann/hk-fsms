# Trail Control
# 1 location(s):
#   Knight_Pickup / Knight/Effects/Fall Trail

fsm Trail Control {
  start Idle

  state Playing {
    PlayParticleEmitter(gameObject=Self, emit=0)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.2, 0.2, 0.2), time=0.2, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FALLING", realTime=false, stopOnExit=true, loopDontFinish=true)
    EaseFloat(fromValue=400, toValue=50, floatVariable=var "Emission Rate", time=0.2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y Velocity", space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y Velocity", float2=0, tolerance=0, equal=→"FALLING", lessThan=(none), greaterThan=→"FALLING", everyFrame=true)
    on FALLING → End
  }

  state End {
    StopParticleEmitter(gameObject=Self)
    on FINISHED → Idle
  }

  state Idle {
    on PLAY → Playing
  }

  var Hero: gameObject = Knight

  var Emission Rate: float = 0
  var Hero Y Velocity: float = 0
  var Environment Type: int = 0
}
