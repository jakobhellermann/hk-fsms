# Music Control
# 4 location(s):
#   Room_Tram / gramaphone
#   Room_Tram / gramaphone (1)
#   Room_Tram_RG / gramaphone
#   Room_Tram_RG / gramaphone (1)

fsm Music Control {
  start Init
  on BROKEN → Stop  // from any state

  state Init {
    on FINISHED → Start Pause
  }

  state Start Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Play
  }

  state Play {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on BREAK → Stop
  }

  state Stop {
    AudioStop(gameObject=Self)
  }
}
