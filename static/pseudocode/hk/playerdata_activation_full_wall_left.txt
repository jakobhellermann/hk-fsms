// uses template: playerdata_activation
fsm playerdata_activation {
  start Check

  state Check {
    GetPlayerDataBool(boolName=var "Player Data name", storeValue=var "Player Data Bool")
    BoolTest(boolVariable=var "Check if Player Data is", isTrue=→"CHECK TRUE", isFalse=→"CHECK FALSE", everyFrame=false)
    on CHECK TRUE → Check True
    on CHECK FALSE → Check False
  }

  state Check True {
    BoolTest(boolVariable=var "Player Data Bool", isTrue=→"CHANGE", isFalse=(none), everyFrame=false)
    on CHANGE → Check Activation
  }

  state Check False {
    BoolTest(boolVariable=var "Player Data Bool", isTrue=(none), isFalse=→"CHANGE", everyFrame=false)
    on CHANGE → Check Activation
  }

  state Check Activation {
    ActivateGameObject(gameObject=Self, activate=var "Activate", recursive=true, resetOnExit=false, everyFrame=false)
  }

  var Activate: bool = false
  var Check if Player Data is: bool = true
  var Player Data name: string = "crossroadsMawlekWall"

  var Player Data Bool: bool = false
}
