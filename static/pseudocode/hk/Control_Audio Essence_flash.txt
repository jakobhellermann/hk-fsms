# Control
# 1 location(s):
#   resources.assets / dream_essence_corpse_get/Audio Essence_flash

fsm Control {
  start Wait
  on A RECYCLE → Reset  // from any state

  state Wait {
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Recycle
    on STOP → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Reset {
    AudioStop(gameObject=Self)
    on FINISHED → Dormant
  }

  state Dormant {
    AudioStop(gameObject=Self)
    on A SPAWN → Wait
  }
}
