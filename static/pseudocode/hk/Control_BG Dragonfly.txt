# Control
# 1 location(s):
#   Room_nailmaster_02 / BG Dragonfly

fsm Control {
  start Init

  state Init {
    RandomFloat(min=0.6, max=0.8, storeResult=var "Scale")
    RandomFloat(min=-0.001, max=0.001, storeResult=var "Z Shift")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Shift", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Hovering
  }

  state Hovering {
  }

  var Scale: float = 0
  var Z Shift: float = 0
}
