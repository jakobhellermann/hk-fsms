# Control
# 9 location(s):
#   Dream_Mighty_Zote / Zote Balloon (3)
#   Dream_Mighty_Zote / Zote Balloon (2)
#   Dream_Mighty_Zote / Zote Balloon (1)
#   Dream_Mighty_Zote / Zote Balloon
#   GG_Grey_Prince_Zote / Zote Balloon (3)
#   GG_Grey_Prince_Zote / Zote Balloon (2)
#   GG_Grey_Prince_Zote / Zote Balloon (1)
#   GG_Grey_Prince_Zote / Zote Balloon
#   GG_Mighty_Zote / Zote Balloon (1)

fsm Control {
  start Init
  on ZOTE KILLED → Reset  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Burst", storeResult=var "Burst")
    GetHP(target=Self, storeValue=var "Max HP")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.00555, space=0, everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Init Pos X", y=var "Init Pos Y", z=(unset), space=0, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="greyPrinceDefeats", storeValue=var "Level")
    IntAdd(intVariable=var "Level", add=1, everyFrame=false)
    IntOperator(integer1=var "Level", integer2=2, operation=Subtract, storeResult=var "Damage", everyFrame=false)
    IntClamp(intVariable=var "Damage", minValue=1, maxValue=99, everyFrame=false)
    on FINISHED → GG Damage
  }

  state Reset {
    SetHP(target=Self, hp=var "Max HP")
    SetMeshRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    SetIsDead(target=Self, setValue=false)
    on FINISHED → Dormant
  }

  state Dormant {
    on BALLOON SPAWN → Spawn Pos
  }

  state Die {
    FlingObjectsFromGlobalPool(gameObject=Spatter Black (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=15, speedMin=15, speedMax=20, angleMin=0, angleMax=360, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    SetCircleCollider(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_death_sword (resources.assets), pitchMin=0.75, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_damage (resources.assets), pitchMin=0.75, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Balloon Burst", animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Reset
  }

  state Set Pos {
    RandomFloat(min=1, max=1.3, storeResult=var "Random")
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=var "Random", z=(unset), everyFrame=false, lateUpdate=false)
    RandomlyFlipFloat(storeResult=var "Random")
    SetScale(gameObject=Self, vector=(unset), x=var "Random", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=-4, max=4, storeResult=var "Random")
    FloatOperator(float1=var "Init Pos X", float2=var "Random", operation=Add, storeResult=var "Pos X", everyFrame=false)
    RandomFloat(min=-2.25, max=2.25, storeResult=var "Random")
    FloatOperator(float1=var "Init Pos Y", float2=var "Random", operation=Add, storeResult=var "Pos Y", everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=2.5, tolerance=0, equal=(none), lessThan=→"ZERO HP", greaterThan=(none), everyFrame=false)
    on FINISHED → Appear
    on ZERO HP → Reset
  }

  state Appear {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Balloon Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=mage_balloon_appear (sharedassets34.assets), pitchMin=0.7, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Balloon Idle")
    SetCircleCollider(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    IdleBuzzV3(gameObject=Self, waitMin=0.5, waitMax=0.6, speedMax=1.75, accelerationMin=6, accelerationMax=10, roamingRangeX=1, roamingRangeY=1, manualStartPos=(unset))
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zote_Balloon_idle_02 (sharedassets399.assets), Zote_Balloon_idle_03 (sharedassets399.assets)], weights=[1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=var "Voice Player")
    on ZERO HP → Explode
    on FINISHED → Antic
  }

  state Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Balloon Explode Antic")
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false)
    Wait(time=0.6, finishEvent=→"WAIT", realTime=false)
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zote_Balloon_pre_burst_01 (sharedassets399.assets), Zote_Balloon_pre_burst_02 (sharedassets399.assets), Zote_Balloon_pre_burst_03 (sharedassets399.assets)], weights=[1, 1, 1], pitchMin=1.2, pitchMax=1.3, volume=1, delay=0, storePlayer=var "Voice Player")
    on ZERO HP → Explode
    on WAIT → Explode
  }

  state Explode {
    AudioStop(gameObject=var "Voice Player")
    CreateObject(gameObject=Gas Explosion M (sharedassets33.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Explosion", networkInstantiate=false, networkGroup=0)
    EnableFSM(gameObject=var "Explosion", fsmName="damages_enemy", enable=false, resetOnExit=false)
    SetDamageHeroAmount(target=var "Explosion", damageDealt=var "Damage")
    on FINISHED → Die
  }

  state Spawn Pos {
    WaitRandom(timeMin=0, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Pos
  }

  state GG Damage {
    GGCheckIfBossScene(bossSceneEvent=(none), regularSceneEvent=→"FINISHED")
    SetIntValue(intVariable=var "Damage", intValue=2, everyFrame=false)
    on FINISHED → Set Damage
  }

  state Set Damage {
    SetDamageHeroAmount(target=Self, damageDealt=var "Damage")
    on FINISHED → Dormant
  }

  var Distance: float = 0
  var Init Pos X: float = 0
  var Init Pos Y: float = 0
  var Pitch: float = 0
  var Pos X: float = 0
  var Pos Y: float = 0
  var Random: float = 0
  var Damage: int = 0
  var Level: int = 0
  var Max HP: int = 0
  var Burst: gameObject = <null>
  var Explosion: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
