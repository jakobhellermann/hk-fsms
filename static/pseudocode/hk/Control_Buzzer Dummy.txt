# Control
# 2 location(s):
#   Ruins2_11 / Break Jar (3)/Buzzer Dummy
#   Ruins2_11 / Break Jar (2)/Buzzer Dummy

fsm Control {
  start Pause F

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    on BREAK → Pause
  }

  state Pause {
    Wait(time=0.16, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Spawn {
    FindChild(gameObject=Self, childName="Buzzer", storeResult=var "Child")
    ActivateGameObject(gameObject=var "Child", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DestroySelf(detachChildren=true)
  }

  state Pause F {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Child: gameObject = <null>
  var Self: gameObject = <null>
}
