# Shiny Control
# 1 location(s):
#   Fungus1_08 / Hunter Eyes/Shiny Item

fsm Shiny Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="Inspect Region", storeResult=var "Inspect Region")
    FindChild(gameObject=Self, childName="Appear Trail", storeResult=var "Trail")
    on FINISHED → PD Bool?
    on ACTIVATE → Destroy
  }

  state PD Bool? {
    StringCompare(stringVariable=var "PD Bool Name", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool Name", isTrue=→"COLLECTED", isFalse=(none))
    on FINISHED → Fling?
    on COLLECTED → Destroy
  }

  state Idle {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopBounce())
    SetFsmBool(gameObject=var "Inspect Region", fsmName="inspect", variableName="Inspectable", setValue=true, everyFrame=false)
    on START INSPECT → Hero Down
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Charm? {
    BoolTest(boolVariable=var "Charm", isTrue=→"CHARM", isFalse=(none), everyFrame=false)
    on CHARM → Get Charm
    on FINISHED → Big Item?
  }

  state Hero Down {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 1")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Charm?
    on HERO DAMAGED → Idle
  }

  state Flash {
    CreateObject(gameObject=Item Get Effect (sharedassets7.assets), spawnPoint=var "Hero", position=(0, -0.76, -1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetSpriteRenderer(gameObject=var "Self", active=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 2")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero Up
    on HERO DAMAGED → Finish
  }

  state Hero Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Tute Msg"), sendEvent="CLOSE", delay=0, everyFrame=false)
    on FINISHED → Finish
    on HERO DAMAGED → Finish
  }

  state Finish {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Inspect Region", fsmName="inspect", variableName="Inspectable", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Tute Msg"), sendEvent="CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Inspect Region"), sendEvent="END INSPECT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHINY ITEM GET", delay=0, everyFrame=false)
  }

  state Get Charm {
    SetPlayerDataBool(boolName=var "PD Bool Name", value=true)
    on FINISHED → Show Tute?
  }

  state Show Tute? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasCharm", isTrue=→"FINISHED", isFalse=(none))
    SetPlayerDataBool(boolName="hasCharm", value=true)
    BoolTest(boolVariable=var "Show Charm Tute", isTrue=→"TUTE", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Normal Msg
    on TUTE → Tute
  }

  state Normal Msg {
    CreateObject(gameObject=Charm Get Msg (sharedassets6.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg", networkInstantiate=false, networkGroup=0)
    SetFsmInt(gameObject=var "Msg", fsmName="Charm Msg", variableName="ID", setValue=var "Charm ID", everyFrame=false)
    on FINISHED → Flash
  }

  state Tute {
    CreateObject(gameObject=Charm Tutorial Msg (sharedassets6.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Tute Msg", networkInstantiate=false, networkGroup=0)
    Wait(time=4.5, finishEvent=→"FINISHED", realTime=false)
    CreateObject(gameObject=Item Get Effect (sharedassets7.assets), spawnPoint=var "Hero", position=(0, -0.76, -1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetSpriteRenderer(gameObject=var "Self", active=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 2")
    on FINISHED → Wait for Input
    on HERO DAMAGED → Finish
  }

  state Wait for Input {
    ListenForInventory(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForCast(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on HERO DAMAGED → Finish
    on BUTTON DOWN → Hero Up
  }

  state Close Tute {
  }

  state Fling? {
    BoolTest(boolVariable=var "Fling On Start", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayParticleEmitter(gameObject=var "Trail", emit=0)
    SetFsmBool(gameObject=var "Inspect Region", fsmName="inspect", variableName="Inspectable", setValue=false, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.85)
    SendRandomEvent(events=[→"FLING L", →"FLING R"], weights=[1, 1], delay=0)
    on FINISHED → Idle
    on FLING L → Fling R
    on FLING R → Fling R
  }

  state Fling L {
    FlingObject(flungObject=Self, speedMin=25, speedMax=25, angleMin=80, angleMax=85)
    on FINISHED → In Air
  }

  state Fling R {
    FlingObject(flungObject=Self, speedMin=20, speedMax=20, angleMin=95, angleMax=100)
    on FINISHED → In Air
  }

  state In Air {
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=true)
    FloatCompare(float1=var "Speed", float2=1, tolerance=0, equal=(none), lessThan=→"STOPPED", greaterThan=(none), everyFrame=true)
    on STOPPED → Land
  }

  state Land {
    SetGravity2dScale(gameObject=Self, gravityScale=0)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Trail")
    on FINISHED → Idle
  }

  state Big Item? {
    BoolTest(boolVariable=var "Dash Cloak", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Mantis Claw", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Journal", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    on BIG → Big Get Flash
    on FINISHED → Trinket?
  }

  state Big Get Flash {
    CreateObject(gameObject=Item Get Effect (sharedassets7.assets), spawnPoint=var "Hero", position=(0, -0.76, -1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetSpriteRenderer(gameObject=var "Self", active=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 2")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=shiny_item_pickup (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    on FINISHED → Item Choice
  }

  state Item Choice {
    BoolTest(boolVariable=var "Dash Cloak", isTrue=→"DASH", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Mantis Claw", isTrue=→"WALLJUMP", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Journal", isTrue=→"JOURNAL", isFalse=(none), everyFrame=false)
    on DASH → 
    on WALLJUMP → 
    on JOURNAL → Journal
  }

  state Trinket? {
    IntCompare(integer1=var "Trinket Num", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"TRINKET", everyFrame=false)
    IntSwitch(intVariable=var "Trinket Num", compareTo=[1, 2, 3, 4], sendEvent=[→"TRINK 1", →"TRINK 2", →"TRINK 3", →"TRINK 4"], everyFrame=false)
    on TRINKET → Trink Flash
    on FINISHED → 
  }

  state Trinket Type {
    IntSwitch(intVariable=var "Trinket Num", compareTo=[1, 2, 3, 4], sendEvent=[→"TRINK 1", →"TRINK 2", →"TRINK 3", →"TRINK 4"], everyFrame=false)
    on TRINK 1 → Trink 1
    on TRINK 2 → Trink 2
    on TRINK 3 → Trink 3
    on TRINK 4 → Trink 4
  }

  state Trink Flash {
    CreateObject(gameObject=Item Get Effect (sharedassets7.assets), spawnPoint=var "Hero", position=(0, -0.76, -1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetSpriteRenderer(gameObject=var "Self", active=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 2")
    CreateObject(gameObject=Relic Get Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg", networkInstantiate=false, networkGroup=0)
    FindChild(gameObject=var "Msg", childName="Text", storeResult=var "Msg Text")
    FindChild(gameObject=var "Msg", childName="Icon", storeResult=var "Msg Icon")
    on FINISHED → Trinket Type
  }

  state Trink 1 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="trinket1")
    SetPlayerDataBool(boolName="foundTrinket1", value=true)
    SetSpriteRendererSprite(gameObject=var "Msg Icon", sprite=inv_item__0013_wanderers-journal (resources.assets))
    GetLanguageString(sheetName="UI", convName="INV_NAME_TRINKET1", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    on FINISHED → Trink Pause
  }

  state Trink 2 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="trinket2")
    SetPlayerDataBool(boolName="foundTrinket2", value=true)
    SetSpriteRendererSprite(gameObject=var "Msg Icon", sprite=inv_item__0012_hallownest-seal (resources.assets))
    GetLanguageString(sheetName="UI", convName="INV_NAME_TRINKET2", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    on FINISHED → Trink Pause
  }

  state Trink 3 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="trinket3")
    SetPlayerDataBool(boolName="foundTrinket3", value=true)
    SetSpriteRendererSprite(gameObject=var "Msg Icon", sprite=inv_item__0010_Kings_idol (resources.assets))
    GetLanguageString(sheetName="UI", convName="INV_NAME_TRINKET3", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    on FINISHED → Trink Pause
  }

  state Trink 4 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="trinket4")
    SetPlayerDataBool(boolName="foundTrinket4", value=true)
    SetSpriteRendererSprite(gameObject=var "Msg Icon", sprite=inv_item__0011_arcane-egg (resources.assets))
    GetLanguageString(sheetName="UI", convName="INV_NAME_TRINKET4", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    on FINISHED → Trink Pause
  }

  state Trink Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero Up
    on HERO DAMAGED → Finish
  }

  state Journal {
    SetPlayerDataBool(boolName="hasJournal", value=true)
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=Journal_Prompt (sharedassets137.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Journal", everyFrame=false)
    on GET ITEM MSG END → Hero Up
  }

  var Charm ID: int = 0
  var Trinket Num: int = 0
  var Charm: bool = false
  var Dash Cloak: bool = false
  var Fling On Start: bool = true
  var Journal: bool = true
  var Mantis Claw: bool = false
  var Show Charm Tute: bool = false
  var PD Bool Name: string = "hasJournal"

  var Speed: float = 0
  var Type: int = 0
  var Activated: bool = false
  var Item Name: string = ""
  var Inspect Region: gameObject = <null>
  var Msg: gameObject = <null>
  var Msg Icon: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Self: gameObject = <null>
  var Trail: gameObject = <null>
  var Tute Msg: gameObject = <null>
  var UI Msg: gameObject = <null>
}
