# Hornet Encounter
# 1 location(s):
#   Fungus1_30 / Hornet Encounter GP3

fsm Hornet Encounter {
  start Init

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="hornetGreenpath", storeValue=var "PlayerData")
    IntCompare(integer1=var "PlayerData", integer2=2, equal=(none), lessThan=(none), greaterThan=→"DESTROY", everyFrame=false)
    on FINISHED → Idle
    on DESTROY → Destroy
  }

  state Idle {
    BoolTest(boolVariable=var "Evade Range", isTrue=→"EVADE", isFalse=(none), everyFrame=true)
    on EVADE → Leave
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Leave {
    SetPlayerDataInt(intName="hornetGreenpath", value=3)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump Full")
    SetVelocity2d(gameObject=Self, vector=(unset), x=7, y=28, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hornet_jump (resources.assets))
    on FINISHED → In Air
  }

  state In Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LANDED", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LANDED", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LANDED → Land
  }

  state Land {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hornet_ground_land (sharedassets28.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Soft Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Jump Out
  }

  state Jump Out {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=28, everyFrame=true)
    Wait(time=2, finishEvent=→"DESTROY", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hornet_jump (resources.assets))
    on DESTROY → Destroy
  }

  var PlayerData: int = 0
  var Evade Range: bool = false
}
