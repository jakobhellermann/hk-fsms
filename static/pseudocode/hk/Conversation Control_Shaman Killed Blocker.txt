# Conversation Control
# 1 location(s):
#   Crossroads_ShamanTemple / _Props/Shaman Killed Blocker

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on UPDATE SHAMAN → Check Active  // from any state

  state Idle {
    on CONVO START → Convo Choice
  }

  state Box Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → Open gate
  }

  state Convo Choice {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="SHAMAN", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntSwitch(intVariable=var "Shaman State", compareTo=[4, 5, 6], sendEvent=[→"MEET", →"SUMMONED 1", →"SUMMONED 2"], everyFrame=false)
    on MEET → Turn Wait
    on SUMMONED 1 → Turn Wait 2
    on SUMMONED 2 → Turn Wait 3
  }

  state Init {
    GetHero(storeResult=var "Hero")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check Active
  }

  state Killed Blocker 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Shaman_01 (sharedassets76.assets), Shaman_02 (sharedassets76.assets), Shaman_03 (sharedassets76.assets), Shaman_04 (sharedassets76.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_KILLED_BLOCKER_1", "Shaman"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Turn Wait {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)
    IntCompare(integer1=var "Frame", integer2=9, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Killed Blocker 1
  }

  state Turn Wait 2 {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)
    IntCompare(integer1=var "Frame", integer2=9, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Killed Blocker 2
  }

  state Killed Blocker 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Shaman_01 (sharedassets76.assets), Shaman_02 (sharedassets76.assets), Shaman_03 (sharedassets76.assets), Shaman_04 (sharedassets76.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_KILLED_BLOCKER_2", "Shaman"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state Turn Wait 3 {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame", everyframe=true)
    IntCompare(integer1=var "Frame", integer2=9, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Opened
  }

  state Opened {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Shaman_04 (sharedassets76.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_OPENED_GATE", "Shaman"], storeResult=(unset var))
    on CONVO_FINISH → Box Down 3
  }

  state Box Down 3 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ACTIVATE RESPAWN TRIGGER", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Check Active {
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=4, equal=(none), lessThan=→"DEACTIVATE", greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Shaman State", integer2=6, equal=→"DEACTIVATE", lessThan=(none), greaterThan=→"DEACTIVATE", everyFrame=false)
    on FINISHED → Enable
    on DEACTIVATE → Disable
  }

  state Disable {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
  }

  state Enable {
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Idle
  }

  state Open gate {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    FindGameObject(objectName="Bone Gate", withTag="Untagged", store=var "Bone Gate")
    SendEventByName(eventTarget=GameObject(var "Bone Gate"), sendEvent="OPEN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open Gate Dial
  }

  state Open Gate Dial {
    SetPlayerDataInt(intName="shaman", value=6)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    SetBoxCollider2DSize(gameObject1=Self, width=6, height=(unset), offsetX=(unset), offsetY=(unset))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHAMAN_OPEN_GATE", "Shaman"], storeResult=(unset var))
    on CONVO_FINISH → Box Down 3
  }

  var Emission Rate: float = 0
  var Hero X Scale: float = 0
  var Self X: float = 0
  var Self X Scale: float = 0
  var Fireball Level: int = 0
  var Frame: int = 0
  var Shaman State: int = 0
  var Area Title: gameObject = <null>
  var Bone Gate: gameObject = <null>
  var CameraParent: gameObject = <null>
  var Hero: gameObject = <null>
  var MainCamera: gameObject = <null>
  var Self: gameObject = <null>
  var Summon FX: gameObject = <null>
  var Summon Pt 1: gameObject = <null>
  var Vengeful Spirit: gameObject = <null>
}
