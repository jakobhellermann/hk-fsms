# Control
# 1 location(s):
#   Ruins_Bathhouse / Superdash Splasher

fsm Control {
  start Init
  on SUPERDASH CANCEL → Dormant  // from any state
  on SUPERDASH END → Dormant  // from any state
  on LEFT RANGE → Dormant  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Splash Pt", storeResult=var "Splash Pt")
    on FINISHED → Dormant
  }

  state Dormant {
    StopParticleEmitter(gameObject=var "Splash Pt")
    on SUPERDASH BLAST → Out of Range
  }

  state Out of Range {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Splash X Min", upperValue=var "Splash X Max", boolVariable=var "In X Range", trueEvent=(none), falseEvent=(none), everyFrame=true)
    FloatInRange(floatVariable=var "Hero Y", lowerValue=var "Splash Y Min", upperValue=var "Splash Y Max", boolVariable=var "In Y Range", trueEvent=(none), falseEvent=(none), everyFrame=true)
    BoolAllTrue(boolVariables=[var "In X Range", var "In Y Range"], sendEvent=→"IN RANGE", storeResult=(unset), everyFrame=true)
    on IN RANGE → Dir Check
  }

  state Dir Check {
    PlayParticleEmitter(gameObject=var "Splash Pt", emit=0)
    SendEventByScale(gameObject=var "Hero", eventTarget=Self, xScale=true, positiveEvent=→"LEFT", negativeEvent=→"RIGHT", space=0)
    on LEFT → Effect Left
    on RIGHT → Effect Right
  }

  state Effect Left {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Splash X Min", upperValue=var "Splash X Max", boolVariable=(unset), trueEvent=(none), falseEvent=→"LEFT RANGE", everyFrame=true)
    FloatInRange(floatVariable=var "Hero Y", lowerValue=var "Splash Y Min", upperValue=var "Splash Y Max", boolVariable=(unset), trueEvent=(none), falseEvent=→"LEFT RANGE", everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    SetRotation(gameObject=var "Splash Pt", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-6.575, space=0, everyFrame=false, lateUpdate=false)
    AudioPlayInState(gameObject=Self, volume=1)
  }

  state Effect Right {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Splash X Min", upperValue=var "Splash X Max", boolVariable=var "In X Range", trueEvent=(none), falseEvent=→"LEFT RANGE", everyFrame=true)
    FloatInRange(floatVariable=var "Hero Y", lowerValue=var "Splash Y Min", upperValue=var "Splash Y Max", boolVariable=(unset), trueEvent=(none), falseEvent=→"LEFT RANGE", everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    SetRotation(gameObject=var "Splash Pt", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=124.267, space=0, everyFrame=false, lateUpdate=false)
    AudioPlayInState(gameObject=Self, volume=1)
  }

  var Splash X Max: float = 61.28
  var Splash X Min: float = 47.53
  var Splash Y Max: float = 32.95105
  var Splash Y Min: float = 30

  var Hero X: float = 0
  var Hero Y: float = 0
  var In X Range: bool = false
  var In Y Range: bool = false
  var Self: gameObject = <null>
  var Splash Pt: gameObject = <null>
}
