# Charm Control
# 1 location(s):
#   Waterways_05 / Charm Container

fsm Charm Control {
  start Pause

  state Idle {
    ActivateGameObject(gameObject=var "Charm", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Charm R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedDungDefender", isTrue=→"ACTIVATE", isFalse=(none))
    on CHARM DROP → Drop
    on ACTIVATE → R Spawn
  }

  state Drop {
    GetOwner(storeGameObject=var "Self")  // disabled
    ActivateAllChildren(gameObject=var "Self", activate=true)  // disabled
    ActivateGameObject(gameObject=var "Charm", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    FindChild(gameObject=Self, childName="Shiny Item", storeResult=var "Charm")
    ActivateGameObject(gameObject=var "Charm", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Shiny Item R", storeResult=var "Charm R")
    ActivateGameObject(gameObject=var "Charm R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state R Spawn {
    ActivateGameObject(gameObject=var "Charm R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Charm: gameObject = <null>
  var Charm R: gameObject = <null>
  var Self: gameObject = <null>
}
