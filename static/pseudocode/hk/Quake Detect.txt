# Quake Detect
# 1 location(s):
#   Room_Mask_Maker / Maskmaker NPC

fsm Quake Detect {
  start Init

  state Idle {
    on QUAKE FALL END → Range Check
  }

  state Range Check {
    BoolTest(boolVariable=var "Quake Range", isTrue=→"FALL", isFalse=→"CANCEL", everyFrame=false)
    on FALL → Fall
    on CANCEL → Idle
  }

  state Fall {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FALL", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    SetFsmBool(gameObject=Self, fsmName="Conversation Control", variableName="Unmasked", setValue=true, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[Mask_Maker_unmask_startle (sharedassets21.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=var "Voice Loop")
    on FINISHED → Recover
  }

  state Recover {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetAudioClip(gameObject=var "Voice Loop", audioClip=Mask_Maker_unmasked_loop (sharedassets21.assets))
    AudioPlaySimple(gameObject=var "Voice Loop", volume=1, oneShotClip=<null>)
  }

  state Init {
    FindChild(gameObject=Self, childName="Voice Loop", storeResult=var "Voice Loop")
    on FINISHED → Idle
  }

  var Quake Range: bool = false
  var Voice Loop: gameObject = <null>
}
