# Conversation Control
# 2 location(s):
#   GG_Engine / Godseeker EngineRoom NPC
#   GG_Engine_Prime / Godseeker EngineRoom NPC

fsm Conversation Control {
  start Pause Frame
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state
  on LOOK UP → Look Up  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"REVISIT", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    ActivateGameObject(gameObject=var "AreaTitle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "AreaTitle", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "AreaTitle", fsmName="", variableName="Area Event", setValue="GODSEEKER", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Revisit?
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
    on FINISHED → 
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Look Up Return
  }

  state Greet 2 {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Repeat Cell", var "GameText Sheet"], storeResult=(unset var))
    AudioPlayRandom(gameObject=var "Self", audioClips=[GS_standard_02 (sharedassets475.assets)], weights=[1], pitchMin=1, pitchMax=1)
    on CONVO_FINISH → Talk Finish
  }

  state Greet 1 {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "First Cell", var "GameText Sheet"], storeResult=(unset var))
    AudioPlayRandom(gameObject=var "Self", audioClips=[GS_standard_05 (sharedassets475.assets), GS_engine_room_03 (sharedassets475.assets), GS_standard_07 (sharedassets475.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on CONVO_FINISH → Talk Finish
  }

  state Spoken? {
    BoolTest(boolVariable=var "Spoken", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Greet 1
    on TRUE → Greet 2
    on REVISIT → Revisit
  }

  state Revisit {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Revisit Cell", var "GameText Sheet"], storeResult=(unset var))
    AudioPlayRandom(gameObject=var "Self", audioClips=[GS_engine_room_04 (sharedassets475.assets)], weights=[1], pitchMin=1, pitchMax=1)
    on CONVO_FINISH → Talk Finish
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Tier
  }

  state Look Up Return {
    BoolTest(boolVariable=var "Do Look Up", isTrue=(none), isFalse=→"NEXT", everyFrame=false)
    SetBoolValue(boolVariable=var "Do Look Up", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Woken", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Turn Range", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Looking Up Return", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.1)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=GS_engine_room (sharedassets475.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"NEXT", realTime=false)
    on NEXT → Box Up
  }

  state Look Up {
    BoolTest(boolVariable=var "Woken", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Up")
    on FINISHED → Cancel
  }

  state Check Tier {
    GGCheckIsBossRushMode(trueEvent=→"GG MODE", falseEvent=(none))
    GGCheckBossSequenceList(tierList=Boss Sequence Tier 1 (resources.assets), trueEvent=→"TIER 1", falseEvent=(none))
    GGCheckBossSequenceList(tierList=Boss Sequence Tier 2 (resources.assets), trueEvent=→"TIER 2", falseEvent=(none))
    GGCheckBossSequenceList(tierList=Boss Sequence Tier 3 (resources.assets), trueEvent=→"TIER 3", falseEvent=(none))
    GGCheckBossSequenceList(tierList=Boss Sequence Tier 4 (resources.assets), trueEvent=→"TIER 4", falseEvent=(none))
    GGCheckBossSequenceList(tierList=Boss Sequence Tier 5 (resources.assets), trueEvent=→"TIER 5", falseEvent=(none))
    on TIER 1 → Set Tier 1
    on TIER 2 → Set Tier 2
    on TIER 3 → Set Tier 3
    on TIER 4 → Set Tier 4
    on FINISHED → Set Tier 1
    on TIER 5 → Set Tier 5
    on GG MODE → Set GG Mode
  }

  state Set Tier 1 {
    SetStringValue(stringVariable=var "First Cell", stringValue="GODSEEKER_ENGINE", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Cell", stringValue="GODSEEKER_ENGINE_REPEAT", everyFrame=false)
    SetStringValue(stringVariable=var "Revisit Cell", stringValue="GODSEEKER_ENGINE_REVISIT", everyFrame=false)
    on FINISHED → Init
  }

  state Set Tier 2 {
    SetStringValue(stringVariable=var "First Cell", stringValue="GODSEEKER_ENGINE_2", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Cell", stringValue="GODSEEKER_ENGINE_REPEAT_2", everyFrame=false)
    SetStringValue(stringVariable=var "Revisit Cell", stringValue="GODSEEKER_ENGINE_REVISIT", everyFrame=false)
    on FINISHED → Init
  }

  state Set Tier 3 {
    SetStringValue(stringVariable=var "First Cell", stringValue="GODSEEKER_ENGINE_3", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Cell", stringValue="GODSEEKER_ENGINE_REPEAT_3", everyFrame=false)
    SetStringValue(stringVariable=var "Revisit Cell", stringValue="GODSEEKER_ENGINE_REVISIT", everyFrame=false)
    on FINISHED → Init
  }

  state Set Tier 4 {
    SetStringValue(stringVariable=var "First Cell", stringValue="GODSEEKER_ENGINE_PRIME", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Cell", stringValue="GODSEEKER_ENGINE_PRIME_REPEAT", everyFrame=false)
    SetStringValue(stringVariable=var "Revisit Cell", stringValue="GODSEEKER_ENGINE_PRIME_REVISIT", everyFrame=false)
    on FINISHED → Init
  }

  state Set Tier 5 {
    SetStringValue(stringVariable=var "First Cell", stringValue="GODSEEKER_ENGINE_T5", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Cell", stringValue="GODSEEKER_ENGINE_REPEAT_T5", everyFrame=false)
    SetBoolValue(boolVariable=var "No Revisit", boolValue=true, everyFrame=false)
    on FINISHED → Init
  }

  state Revisit? {
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    AudioStop(gameObject=Self)
    BoolTest(boolVariable=var "No Revisit", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    GetStaticVariable(variableName="currentBossDoorPD", storeValue=var "Boss Door PD")
    GGGetBossDoorCompletion(playerDataVariable=var "Boss Door PD", unlocked=(unset), completed=var "Door Completed", allBindings=(unset), noHits=(unset), boundNail=(unset), boundShell=(unset), boundCharms=(unset), boundSoul=(unset))
    BoolTest(boolVariable=var "Door Completed", isTrue=→"REVISIT", isFalse=(none), everyFrame=false)
    on FINISHED → Spoken?
    on REVISIT → Revisit
  }

  state Cancel {
  }

  state Set GG Mode {
    SetStringValue(stringVariable=var "First Cell", stringValue="GODSEEKER_ENGINE_GGMODE", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Cell", stringValue="GODSEEKER_ENGINE_REPEAT_GGMODE", everyFrame=false)
    SetStringValue(stringVariable=var "Revisit Cell", stringValue="GODSEEKER_ENGINE_REVISIT_GGMODE", everyFrame=false)
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    SetFsmString(gameObject=var "Dream Dialogue", fsmName="npc_dream_dialogue", variableName="Convo Name", setValue="GODSEEKER_ENGINE_DREAM_GGMODE", everyFrame=false)
    on FINISHED → Init
  }

  var Do Look Up: bool = true

  var Activated: bool = false
  var Door Completed: bool = false
  var No Revisit: bool = false
  var Revisiting: bool = false
  var Spoken: bool = false
  var Woken: bool = false
  var Boss Door PD: string = ""
  var First Cell: string = "GODSEEKER_ENGINE"
  var GameText Sheet: string = "CP3"
  var Repeat Cell: string = "GODSEEKER_ENGINE_REPEAT"
  var Revisit Cell: string = "GODSEEKER_ENGINE_REVISIT"
  var Area Title: gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var Self: gameObject = <null>
  var Turn Range: gameObject = Godseeker EngineRoom NPC/Turn Range
}
