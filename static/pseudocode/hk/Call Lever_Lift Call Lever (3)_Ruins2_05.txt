# Call Lever
# 1 location(s):
#   Ruins2_05 / Lift Call Lever (3)

fsm Call Lever {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "Collider")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Call Switch Idle")
    on NAIL HIT → Get Direction
  }

  state Get Direction {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_switch (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    GetFsmFloat(gameObject=var "Collider", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", (none), →"LEFT", (none)], everyFrame=false)
    SendRandomEvent(events=[→"LEFT", →"RIGHT"], weights=[1, 1], delay=0)
    on RIGHT → Right
    on LEFT → Left
  }

  state Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Call Switch Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Check Already Called
  }

  state Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Call Switch Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Check Already Called
  }

  state Send Msg {
    BoolTest(boolVariable=var "Was In Range", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)  // disabled
    SetFsmInt(gameObject=var "Lift", fsmName="Lift Control", variableName="Call Stop", setValue=var "Stop Number", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lift"), sendEvent="CALL", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Hero Check {
    SetBoolValue(boolVariable=var "Was In Range", boolValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Hero Check", isTrue=→"IN RANGE", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Was In Range", boolValue=false, everyFrame=false)
    on FINISHED → Get Direction
  }

  state Check Already Called {
    GetFsmBool(gameObject=var "Lift", fsmName="Lift Control", variableName="Called", storeValue=var "Called", everyFrame=false)
    BoolTest(boolVariable=var "Called", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Idle
    on FALSE → Send Msg
  }

  var Stop Number: int = 3
  var Lift: gameObject = Ruins Lift (1)

  var Attack Direction: float = 0
  var Called: bool = false
  var Hero Check: bool = false
  var Was In Range: bool = false
  var Collider: gameObject = <null>
  var Self: gameObject = <null>
}
