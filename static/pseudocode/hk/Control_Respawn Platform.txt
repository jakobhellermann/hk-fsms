# Control
# 2 location(s):
#   Room_Colosseum_Silver / Colosseum Manager/Waves/Respawn Plat/Respawn Platform
#   Room_Colosseum_Gold / Colosseum Manager/Waves/Respawn Plat/Respawn Platform

fsm Control {
  start Init
  on HERO RESPAWNED → Sit for Respawn  // from any state
  on KNIGHT SPIKE HIT → Expand  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Platform", storeResult=var "Platform")
    SetCollider(gameObject=Self, active=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    on FINISHED → Retracted
  }

  state Retracted {
    on KNIGHT SPIKE HIT → Expand
  }

  state Fold {
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Platform", clipName="Retract", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPosition(gameObject=var "Platform", vector=(unset), x=(unset), y=(unset), z=0.007, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Retract
  }

  state Retract {
    iTweenMoveTo(gameObject=var "Platform", id="", transformPosition=var "", vectorPosition=(0, -18, 0.007), time=0.6, delay=0, speed=(unset), space=1, easeType=easeInSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Retracted
  }

  state Retract Antic {
    ObjectJitter(gameObject=var "Platform", x=0.1, y=0.1, z=(unset), allowMovement=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fold
  }

  state Sit for Respawn {
    SetPosition(gameObject=var "Platform", vector=(unset), x=0, y=0, z=0.007, space=1, everyFrame=false, lateUpdate=false)
    SetCollider(gameObject=Self, active=true)
    SetPosition(gameObject=var "Platform", vector=(unset), x=(unset), y=(unset), z=0.001, space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Platform", animLibName="", clipName="Expand")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Retract Antic
  }

  state Expand {
    Tk2dPlayAnimation(gameObject=var "Platform", animLibName="", clipName="Retracted")
    SetCollider(gameObject=Self, active=true)
    iTweenMoveTo(gameObject=var "Platform", id="", transformPosition=var "", vectorPosition=(0, 0, 0.007), time=0.6, delay=0, speed=(unset), space=1, easeType=easeOutSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Sit for Respawn
  }

  var Platform: gameObject = <null>
}
