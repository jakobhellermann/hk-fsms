# fat fly bounce
# 2 location(s):
#   Room_Colosseum_Silver / Colosseum Manager/Waves/Wave 30 Obble/Mega Fat Bee (1)
#   GG_Oblobbles / Mega Fat Bee (1)

fsm fat fly bounce {
  start Initialise
  on STOP → Stopped  // from any state
  on GO UP → Go Up  // from any state
  on RAGING → Stopped  // from any state

  state Initialise {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    FindGameObject(objectName="Colosseum Manager", withTag="Untagged", store=var "Col Manager")
    on FINISHED → Swoop In
  }

  state Left or Right? {
    FloatSwitch(floatVariable=var "Angle", lessThan=[90, 270, 360], sendEvent=[→"RIGHT", →"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Face Left
    on RIGHT → Face Right
  }

  state Face Left {
    SetScale(gameObject=var "Self", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    BoolTest(boolVariable=var "Rage", isTrue=→"RAGE", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Fly 2
    on RAGE → Fly Rage
  }

  state Face Right {
    SetScale(gameObject=var "Self", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Rage", isTrue=→"RAGE", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Fly 2
    on RAGE → Fly Rage
  }

  state Hit Up {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Up Right
    on LEFT → Up Left
  }

  state Up Right {
    RandomFloat(min=300, max=301, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Up Left {
    RandomFloat(min=230, max=240, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Hit Down {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Down Right
    on LEFT → Down Left
  }

  state Down Right {
    RandomFloat(min=60, max=70, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Down Left {
    RandomFloat(min=120, max=130, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Hit Right {
    Translate(gameObject=Self, vector=(unset), x=-0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Down Right 2
    on DOWN → Right Down
  }

  state Right Down {
    RandomFloat(min=210, max=220, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Down Right 2 {
    RandomFloat(min=140, max=150, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Hit Left {
    Translate(gameObject=Self, vector=(unset), x=0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Left Up
    on DOWN → Left Down
  }

  state Left Down {
    RandomFloat(min=320, max=330, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Left Up {
    RandomFloat(min=30, max=40, storeResult=var "Angle")
    on FINISHED → From Bounce
  }

  state Stopped {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    SetBoolValue(boolVariable=var "Starts Inactive", boolValue=false, everyFrame=false)
    on WAKE → Left or Right?
    on RAGE END → Left or Right?
  }

  state Aim {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="START", delay=0, everyFrame=false)
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Go Up {
    RandomFloat(min=20, max=160, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Fly 2 {
    SetVelocityAsAngle(gameObject=var "Self", angle=var "Angle", speed=var "Speed", everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn Fly", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=var "X Min", tolerance=0, equal=(none), lessThan=→"HIT LEFT", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "X Pos", float2=var "X Max", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"HIT RIGHT", everyFrame=true)
    on HIT DOWN → Hit Down
    on HIT LEFT → Hit Left
    on HIT RIGHT → Hit Right
    on HIT UP → Hit Up
  }

  state From Bounce {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BOUNCE", delay=0, everyFrame=false)
    on FINISHED → Left or Right?
  }

  state Swoop In {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetCollider(gameObject=Self, active=false)
    RandomFloat(min=25.006, max=25.0069, storeResult=var "Z Random")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Random", space=1, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=15, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -15, -25), time=2, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"SUMMON", realTime=false, stopOnExit=true, loopDontFinish=true)
    on SUMMON → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Aim
  }

  state Fly Rage {
    SetVelocityAsAngle(gameObject=var "Self", angle=var "Angle", speed=var "Speed", everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn Fly", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT UP", rightHitEvent=→"HIT RIGHT", bottomHitEvent=→"HIT DOWN", leftHitEvent=→"HIT LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    on HIT DOWN → Hit Down
    on HIT LEFT → Hit Left
    on HIT RIGHT → Hit Right
    on HIT UP → Hit Up
  }

  var Speed: float = 6
  var X Max: float = 118.5412
  var X Min: float = 86.45936
  var Starts Inactive: bool = false

  var Angle: float = 0
  var X Pos: float = 0
  var Z Random: float = 102.85
  var Facing Right: bool = false
  var Rage: bool = false
  var Col Manager: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
