# Generate Wave
# 1 location(s):
#   Waterways_05 / Charm Container/Shiny Item R

fsm Generate Wave {
  start Init
  on GET → Get  // from any state
  on STOP → Stopped  // from any state

  state Gen {
    ActivateGameObject(gameObject=var "Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=var "Wave", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetScale(gameObject=var "Wave", vector=(unset), x=0.35, y=0.35, z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    on FINISHED → Can Gen
  }

  state Wait {
    Wait(time=2.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Can Gen
  }

  state Get {
  }

  state Stopped {
    on START → Gen
  }

  state Can Gen {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "World Z Pos", space=0, everyFrame=false)
    FloatCompare(float1=var "World Z Pos", float2=0, tolerance=1, equal=→"TRUE", lessThan=→"FALSE", greaterThan=→"FALSE", everyFrame=false)
    on TRUE → Gen
    on FALSE → Wait
  }

  var World Z Pos: float = 0
  var Self: gameObject = <null>
  var Wave: gameObject = Charm Container/Shiny Item R/White Wave
}
