# superdash_crystal_gen
# 2 location(s):
#   Knight_Pickup / Knight/Effects/SD Crystal Gen G1
#   resources.assets / Knight:1/Effects/SD Crystal Gen G1

// uses template: superdash_crystal_gen
fsm superdash_crystal_gen {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Floor Check", storeResult=var "Floor Check Obj")
    on FINISHED → Idle
  }

  state Idle {
    ActivateGameObject(gameObject=var "Floor Check Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on SUPERDASH CHARGING G → Reset Pos
  }

  state Charging {
    GetParent(gameObject=var "Hero", storeResult=var "Hero Parent")
    GameObjectIsNull(gameObject=var "Hero Parent", isNull=(none), isNotNull=→"SUPERDASH CANCEL", storeResult=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Floor Check", boolValue=true, everyFrame=false)
    iTweenMoveBy(gameObject=Self, id="", vector=var "Move Vector", time=0.8, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    BoolTest(boolVariable=var "Floor Check", isTrue=(none), isFalse=→"NO FLOOR", everyFrame=true)
    on FINISHED → Idle
    on SUPERDASH READY → Idle
    on SUPERDASH CANCEL → Send Cancel
    on NO FLOOR → No Floor
  }

  state No Floor {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="NO FLOOR", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Reset Pos {
    SetPosition(gameObject=Self, vector=var "Init Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Floor Check Obj", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Charging
  }

  state Send Cancel {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SUPERDASH CANCEL", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Positive Direction: bool = false
  var Move Vector: vector3 = (3, 0, 0)

  var Speed: float = 4
  var Speed Crt: float = 0
  var Floor Check: bool = false
  var Init Pos: vector3 = (0, -1.42, -0.001)
  var Floor Check Obj: gameObject = <null>
  var Hero Parent: gameObject = <null>
  var Self: gameObject = <null>
}
