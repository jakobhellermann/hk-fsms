# Control
# 1 location(s):
#   Room_Mask_Maker / Maskmaker NPC/Mask

fsm Control {
  start Init

  state Init {
    SendRandomEvent(events=[→"ONE", →"TWO", →"THREE"], weights=[1, 1, 1], delay=0)
    on ONE → One
    on TWO → Two
    on THREE → Three
  }

  state One {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mask Idle 1")
    on FALL → Fall 1
  }

  state Two {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mask Idle 2")
    on FALL → Fall 2
  }

  state Three {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mask Idle 3")
    on FALL → Fall 3
  }

  state Fall 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mask Fall 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Disable
  }

  state Fall 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mask Fall 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Disable
  }

  state Fall 3 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mask Fall 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Disable
  }

  state Disable {
    SetMeshRenderer(gameObject=Self, active=false)
  }
}
