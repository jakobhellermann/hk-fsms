# Control
# 1 location(s):
#   GG_Pipeway / Fat Fluke

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Crawling
  }

  state Crawling {
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Behind Range", isTrue=→"BEHIND", isFalse=(none), everyFrame=true)
    AudioPlayInState(gameObject=var "Crawl Loop", volume=1)
    on ALERT → Roar
    on BEHIND → Turn Anim
    on TOOK DAMAGE → Turn?
  }

  state Roar {
    StopWalker(target=Self, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mawlek_call (sharedassets34.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)  // disabled
    AudioPlaySimple(gameObject=var "Angry Loop", volume=1, oneShotClip=<null>)  // disabled
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=ff_startle_short_02 (sharedassets458.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Charge
  }

  state Charge {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Charge")
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Charge Speed", float2=var "X Scale", operation=Multiply, storeResult=var "Crt Speed", everyFrame=false)
    AudioPlayInState(gameObject=var "Charge Loop", volume=1)
    AudioPlayInState(gameObject=var "Charge Loop 2", volume=1)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Crt Speed", y=(unset), everyFrame=false)
    SetParticleEmission(gameObject=var "Pt Spatter", emission=true)
    AudioPlaySimple(gameObject=var "Angry Loop", volume=1, oneShotClip=<null>)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on WALL → Charge End
  }

  state Turn Anim {
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Flip
  }

  state Flip {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Roar
  }

  state Charge End {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStop(gameObject=var "Angry Loop")
    SetParticleEmission(gameObject=var "Pt Spatter", emission=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_strike_ground (sharedassets48.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Resume Walker
  }

  state Resume Walker {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on FINISHED → Crawling
  }

  state Turn? {
    BoolTest(boolVariable=var "Long Behind Range", isTrue=→"TURN", isFalse=(none), everyFrame=false)
    on FINISHED → Roar
    on TURN → Turn Anim
  }

  var Charge Speed: float = -14
  var Crt Speed: float = 0
  var X Scale: float = 0
  var Alert Range: bool = false
  var Behind Range: bool = false
  var Long Behind Range: bool = false
  var Angry Loop: gameObject = Fat Fluke/Angry Loop
  var Charge Loop: gameObject = Fat Fluke/Charge Loop
  var Charge Loop 2: gameObject = Fat Fluke/Charge Loop 2
  var Crawl Loop: gameObject = Fat Fluke/Crawl Loop
  var Pt Spatter: gameObject = Fat Fluke/Pt Spatter
  var Self: gameObject = <null>
}
