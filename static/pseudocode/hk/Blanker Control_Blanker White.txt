# Blanker Control
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Blanker White
#   resources.assets / _GameCameras/HudCamera/Blanker White

fsm Blanker Control {
  start Init
  on RESET → Reset  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetSpriteRenderer(gameObject=var "Self", active=false)
    SetGameObject(variable=var "HUD Blanker White", gameObject=var "Self", everyFrame=false)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    on FINISHED → Idle
  }

  state Fade In {
    SetColorValue(colorVariable=var "Colour", color=(1, 1, 1, 0), everyFrame=false)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Colour", time=var "Fade Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FADE OUT → Fade Out
  }

  state Fade Out {
    SetColorValue(colorVariable=var "Colour", color=(1, 1, 1, 1), everyFrame=false)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=var "Fade Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FADE IN → Fade In
    on FINISHED → Faded Out
  }

  state Reset {
    SetSpriteRenderer(gameObject=var "Self", active=true)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on FADE IN → Fade In
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Faded Out {
    SetSpriteRenderer(gameObject=Self, active=true)
    on FADE IN → Fade In
  }

  var Fade Time: float = 1
  var Colour: color = (0, 0, 0, 1)
  var Self: gameObject = <null>
}
