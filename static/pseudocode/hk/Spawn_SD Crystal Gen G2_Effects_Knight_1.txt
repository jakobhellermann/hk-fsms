# Spawn
# 1 location(s):
#   resources.assets / Knight:1/Effects/SD Crystal Gen G2

fsm Spawn {
  start Init
  on SUPERDASH BLAST → Idle  // from any state
  on SUPERDASH CANCEL → Idle  // from any state
  on SUPERDASH READY → Idle  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on SUPERDASH CHARGING G → Gen Init
  }

  state Gen Pause {
    Wait(time=0.09, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gen
  }

  state Gen Init {
    SetFloatValue(floatVariable=var "Scale", floatValue=0.5, everyFrame=false)
    on FINISHED → Gen Pause
  }

  state Gen {
    SpawnObjectFromGlobalPool(gameObject=SD Crystal, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Crystal")
    SetRotation(gameObject=var "Crystal", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tilt", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Crystal", vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    FloatAdd(floatVariable=var "Scale", add=0.12, everyFrame=false, perSecond=false)
    on FINISHED → Gen Pause
  }

  var Tilt: float = 10

  var Scale: float = 0
  var Init Pos: vector3 = (0, -1.42, -0.001)
  var Crystal: gameObject = <null>
  var Self: gameObject = <null>
}
