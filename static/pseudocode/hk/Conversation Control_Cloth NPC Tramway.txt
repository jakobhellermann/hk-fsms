# Conversation Control
# 1 location(s):
#   Abyss_17 / Cloth NPC Tramway

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Club", storeResult=var "Club")
    FindChild(gameObject=Self, childName="Dream Dialogue Happy", storeResult=var "Dream Dialogue Happy")
    FindChild(gameObject=Self, childName="Dream Dialogue Sad", storeResult=var "Dream Dialogue Sad")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Meet {
    SetPlayerDataBool(boolName="metCloth", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CLOTH_TRAMWAY_MEET", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[HK_Cloth_Sad01 (sharedassets283.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down 2
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Title
  }

  state Convo Choice {
    SetCollider(gameObject=var "Dream Dialogue Happy", active=true)
    SetCollider(gameObject=var "Dream Dialogue Sad", active=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metCloth", isTrue=(none), isFalse=→"MEET")
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"GREET", everyFrame=false)
    on MEET → Meet
    on GREET → Greet
    on REPEAT → Repeat
  }

  state Title {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="CLOTH_C", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="CLOTH_NC", everyFrame=false)
    on FINISHED → Box Up
  }

  state Greet {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CLOTH_TRAMWAY_GREET", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[HK_Cloth_Sad01 (sharedassets283.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down 2
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CLOTH_TRAMWAY_REPEAT", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[HK_Cloth_Happy02 (sharedassets7.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Triumph Antic {
    ActivateGameObject(gameObject=var "Club", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Triumph Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=StoryRecord_visited("Cloth in the Ancient Basin, and cheered him up"))
    on FINISHED → Triumph
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Triumph Antic
  }

  state Triumph {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Triumph")
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar", networkInstantiate=false, networkGroup=0)
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[HK_Cloth_Battle_Cry (sharedassets209.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[new_heartpiece_puzzle_bit (resources.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Triumph End
  }

  state Triumph End {
    SendEventByName(eventTarget=GameObject(var "Roar"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT SCENE", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up 2
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Meet 2
  }

  state Meet 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="READY", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CLOTH_TRAMWAY_SAVED", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[HK_Cloth_Combat01 (sharedassets341.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk Finish
  }

  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Club: gameObject = <null>
  var Dream Dialogue Happy: gameObject = <null>
  var Dream Dialogue Sad: gameObject = <null>
  var Roar: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
