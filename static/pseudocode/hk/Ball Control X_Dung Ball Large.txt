# Ball Control X
# 1 location(s):
#   sharedassets355.assets / Dung Ball Large

fsm Ball Control X {
  start Init
  on DUNGBALL END → Burst  // from any state

  state Detect {
    Collision2dEvent(collision=0, collideTag="Untagged", sendEvent=→"COLLISION ENTER 2D", storeCollider=var "Collider", storeForce=(unset))
    Collision2dEvent(collision=1, collideTag="Untagged", sendEvent=→"COLLISION STAY 2D", storeCollider=var "Collider", storeForce=(unset))
    on COLLISION ENTER 2D → Check
    on COLLISION STAY 2D → Check
    on TINK RIGHT → Get Speed
    on TINK LEFT → Get Speed
  }

  state Check {
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=(unset), contactNormal=var "Contact Normal", shapeCount=(unset), physics2dMaterialName="")
    GetVector3XYZ(vector3Variable=var "Contact Normal", storeX=var "Normal X", storeY=var "Normal Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=1, tolerance=0.3, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatTestToBool(float1=var "X Speed", float2=0, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Moving Right", everyFrame=false)
    FloatTestToBool(float1=var "Normal X", float2=-1, tolerance=0.3, equalBool=var "Hit Right", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    FloatTestToBool(float1=var "Normal X", float2=1, tolerance=0.1, equalBool=var "Hit Left", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hit Left", var "Moving Right"], boolStates=[true, false], trueEvent=→"FLIP", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Hit Right", var "Moving Right"], boolStates=[true, true], trueEvent=→"FLIP", falseEvent=(none), storeResult=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=1, tolerance=0.1, equal=(none), lessThan=→"CANCEL", greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → Detect
    on FINISHED → Detect
    on FLIP → Flip
  }

  state Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ball Idle")
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    PlayParticleEmitter(gameObject=var "Smoke", emit=0)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    SetIntValue(intVariable=var "Bounces", intValue=7, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get Speed
  }

  state Flip {
    FloatMultiply(floatVariable=var "X Speed", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), everyFrame=false)
    on FINISHED → Pause Frame
  }

  state Pause Frame {
    FloatCompare(float1=var "Prev X Speed", float2=0, tolerance=0, equal=→"BURST", lessThan=(none), greaterThan=(none), everyFrame=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
    on BURST → Burst
  }

  state Burst {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="JUST BURST", delay=0, everyFrame=false)
  }

  state Get Speed {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=false)
    on FINISHED → Detect
  }

  var Normal X: float = 0
  var Normal Y: float = 0
  var Prev X Speed: float = 0
  var X Speed: float = 0
  var Bounces: int = 0
  var Hit Left: bool = false
  var Hit Right: bool = false
  var Moving Right: bool = false
  var Contact Normal: vector3 = (0, 0, 0)
  var Collider: gameObject = <null>
  var Smoke: gameObject = Dung Ball Large/Smoke
}
