fsm Grub Control {
  start Init
  on FREE → Free  // from any state

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    WaitRandom(timeMin=3, timeMax=6, finishEvent=→"CRY", realTime=false)
    on ENTER → Hero Close
    on FREE → Free
    on CRY → Cry
  }

  state Hero Close {
    AudioStop(gameObject=Self)
    AudioPlayRandom(gameObject=var "Self", audioClips=[grub_alert (sharedassets33.assets), grub_idle_or_look_1 (sharedassets33.assets), grub_idle_or_look_2 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Cry")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Cry Turn", resetFrame=true, everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Sad Wait
    on FREE → Free
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    FindChild(gameObject=var "Self", childName="Dig Dust", storeResult=var "Dig Dust")
    on FINISHED → Idle
  }

  state Free {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Leave
  }

  state Leave {
    AudioPlayRandom(gameObject=var "Self", audioClips=[grub_free_1 (sharedassets39.assets), grub_free_2 (sharedassets39.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Freed")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dig
  }

  state Dig {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=grub_burrow (sharedassets39.assets), finishedEvent=(none))
    PlayParticleEmitter(gameObject=var "Dig Dust", emit=0)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy
  }

  state Destroy {
    StopParticleEmitter(gameObject=var "Dig Dust")
    DestroySelf(detachChildren=true)
  }

  state Sad Sound {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    AudioStop(gameObject=Self)
    AudioPlayRandom(gameObject=var "Self", audioClips=[grub_sad_1 (sharedassets33.assets), grub_sad_2 (sharedassets33.assets), grub_sad_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on FINISHED → Idle Wait
  }

  state Sad Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sad Sound
  }

  state Idle Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Cry {
    AudioStop(gameObject=Self)
    AudioPlayRandom(gameObject=var "Self", audioClips=[Grub_sad_idle_01 (sharedassets33.assets), Grub_sad_idle_02 (sharedassets33.assets), Grub_sad_idle_03 (sharedassets33.assets), Grub_sad_idle_04 (sharedassets33.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1)
    on FINISHED → Idle
  }

  var Dig Dust: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
