# Flying Sentry Nail
# 1 location(s):
#   Ruins1_05 / Battle Scene v2/Wave 2/Ruins Flying SentryB

fsm Flying Sentry Nail {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    on FINISHED → Wall
  }

  state Wall {
    BoolTest(boolVariable=var "Start Idle", isTrue=→"IDLE", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Wake Region", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    on WAKE → Wake
    on IDLE → Idle Buzz
    on TOOK DAMAGE → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wall Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Scale", multiplyBy=-1, everyFrame=false)
    FloatMultiply(floatVariable=var "Kick Off Force", multiplyBy=var "Scale", everyFrame=false)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "Kick Off Force", y=(unset), vector3=(unset), everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Ruins_Flying_Sentry_startle_01 (sharedassets90.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Idle {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateGameObject(gameObject=var "Charge Javelin Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DistanceFly(gameObject=var "Self", target=var "Hero", distance=8, speedMax=6, acceleration=0.3, targetsHeight=true, height=4.25)
    WaitRandom(timeMin=1.5, timeMax=2, finishEvent=→"ATTACK", realTime=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn To Idle", resetFrame=true, everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on ATTACK → Lost Check
  }

  state Idle Buzz {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    IdleBuzz(gameObject=Self, waitMin=0.75, waitMax=1, speedMax=4, accelerationMax=20, roamingRange=2)
    BoolTest(boolVariable=var "Wake Region", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn To Idle", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    on WAKE → Idle
    on TOOK DAMAGE → Idle
  }

  state Position {
    DistanceFly(gameObject=Self, target=var "Hero", distance=5, speedMax=6, acceleration=0.3, targetsHeight=true, height=-0.5)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn To Idle", resetFrame=true, everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatTestToBool(float1=var "Hero Y", float2=var "Self Y", tolerance=0.75, equalBool=var "Same Y", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=true)
    FloatTestToBool(float1=var "Distance", float2=5.25, tolerance=0, equalBool=(unset), lessThanBool=var "In Slash Range", greaterThanBool=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "In Slash Range", var "Same Y"], boolStates=[true, true], trueEvent=→"ATTACK", falseEvent=(none), storeResult=(unset), everyFrame=true)
    Wait(time=8, finishEvent=→"CANCEL", realTime=false)
    on ATTACK → Slash Antic
    on CANCEL → Can See?
  }

  state Slash Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DistanceFly(gameObject=Self, target=var "Hero", distance=4, speedMax=6, acceleration=0.3, targetsHeight=true, height=-0.5)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Ruins_Flying_Sentry_attack_01 (sharedassets90.assets), Ruins_Flying_Sentry_attack_02 (sharedassets90.assets), Ruins_Flying_Sentry_attack_03 (sharedassets90.assets), Ruins_Flying_Sentry_attack_04 (sharedassets90.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Check Dir
  }

  state Slash {
    AudioPlayRandom(gameObject=var "Self", audioClips=[ruin_sentry_sword_1 (sharedassets33.assets), ruin_sentry_sword_2 (sharedassets33.assets), ruin_sentry_sword_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slash Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Slash Recover
  }

  state Slash Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.95)
    ActivateGameObject(gameObject=var "Slash Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Check Dir {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Scale", float2=0, tolerance=0, equal=(none), lessThan=→"RIGHT", greaterThan=→"LEFT", everyFrame=false)
    on RIGHT → Right
    on LEFT → Left
  }

  state Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=18, y=(unset), everyFrame=false)
    on FINISHED → Slash Forward
  }

  state Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-18, y=(unset), everyFrame=false)
    on FINISHED → Slash Forward
  }

  state Slash Forward {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Forward", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Slash
  }

  state Can See? {
    BoolTest(boolVariable=var "Wake Region", isTrue=→"FINISHED", isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Idle
    on CANCEL → Idle Buzz
  }

  state Lost Check {
    GetFsmFloat(gameObject=Self, fsmName="Lost Hero Check", variableName="Lost Counter", storeValue=var "Lost Counter", everyFrame=false)
    FloatCompare(float1=var "Lost Counter", float2=6, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"LOST HERO", everyFrame=false)
    on LOST HERO → Idle Buzz
    on FINISHED → Position
  }

  state Go Up {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Start Idle: bool = false

  var Distance: float = 0
  var Hero Y: float = 0
  var Kick Off Force: float = 200
  var Lost Counter: float = 0
  var Scale: float = 0
  var Self Y: float = 0
  var In Slash Range: bool = false
  var Same Y: bool = false
  var Wake Region: bool = false
  var Charge Javelin Hit: gameObject = <null>
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Slash Hit: gameObject = Battle Scene v2/Wave 2/Ruins Flying SentryB/Slash Hit
}
