# Knight Height Check
# 1 location(s):
#   RestingGrounds_10 / Breakable Wall (2)

fsm Knight Height Check {
  start Pause
  on UNCOVER → Uncover  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Away
  }

  state Away {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    SetCollider(gameObject=Self, active=false)
    FloatCompare(float1=var "Hero Y", float2=13.5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CHANGE", everyFrame=true)
    on CHANGE → Near
  }

  state Near {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    SetCollider(gameObject=Self, active=true)
    FloatCompare(float1=var "Hero Y", float2=13.5, tolerance=0, equal=(none), lessThan=→"CHANGE", greaterThan=(none), everyFrame=true)
    on CHANGE → Away
  }

  state Uncover {
    SetCollider(gameObject=Self, active=false)
  }

  var Hero Y: float = 0
}
