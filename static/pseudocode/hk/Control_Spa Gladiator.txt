# Control
# 1 location(s):
#   Room_Colosseum_02 / Spa Gladiator

fsm Control {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Look
  }

  state Look {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look L")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Back
    on SPLASH WATER → Splash
  }

  state Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look R")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Look
  }

  state Splashed {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Back
    on SPLASH WATER → Splash
  }

  state Splash {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[Col_miner_attack_01 (sharedassets31.assets), Col_miner_attack_02 (sharedassets31.assets), Col_miner_attack_03 (sharedassets31.assets), Col_miner_attack_04 (sharedassets31.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Splash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Splashed
  }
}
