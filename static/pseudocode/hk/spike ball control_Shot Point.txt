# spike ball control
# 6 location(s):
#   Fungus1_21 / Moss Knight/Shot Point
#   Fungus1_21 / Battle Scene/Moss Knight/Shot Point
#   Fungus1_21 / Moss Knight (1)/Shot Point
#   Fungus1_26 / Moss Knight/Shot Point
#   Fungus1_32 / Moss Knight B/Shot Point
#   Fungus1_32 / Moss Knight C/Shot Point

fsm spike ball control {
  start Init

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"TERRAIN", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer=Attack(17), sendEvent=→"TERRAIN", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=Attack(17), sendEvent=→"NAIL", storeCollider=var "Nail Object")
    on TERRAIN → Break
    on NAIL → Nail Hit
  }

  state Break {
    StopParticleEmitter(gameObject=var "Grass")
    PlayParticleEmitter(gameObject=var "Death Spore", emit=0)
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    FloatMultiply(floatVariable=var "Angle", multiplyBy=-1, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=var "Angle", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=var "Angle", everyFrame=false)
    FloatOperator(float1=var "Angle Min", float2=45, operation=Subtract, storeResult=var "Angle Min", everyFrame=false)
    FloatOperator(float1=var "Angle Max", float2=45, operation=Add, storeResult=var "Angle Max", everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=var "Bit1", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bit2", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bit3", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    RandomFloat(min=10, max=15, storeResult=var "Speed")
    RandomFloat(min=var "Angle Min", max=var "Angle Max", storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=var "Bit1", angle=var "Angle", speed=var "Speed", everyFrame=false)
    RandomFloat(min=10, max=15, storeResult=var "Speed")
    RandomFloat(min=var "Angle Min", max=var "Angle Max", storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=var "Bit2", angle=var "Angle", speed=var "Speed", everyFrame=false)
    RandomFloat(min=10, max=15, storeResult=var "Speed")
    RandomFloat(min=var "Angle Min", max=var "Angle Max", storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=var "Bit3", angle=var "Angle", speed=var "Speed", everyFrame=false)
    DestroySelf(detachChildren=true)
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetScale(gameObject=Self, vector=(unset), x=0.5, y=0.5, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(1, 1, 1), time=0.1, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    AddTorque2d(gameObject=Self, forceMode=0, torque=0, everyFrame=false)
    on FINISHED → Spin Choose
  }

  state Activate {
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=5, everyFrame=false)
    on FINISHED → Idle
  }

  state Spin Choose {
    SendRandomEvent(events=[→"LEFT", →"RIGHT"], weights=[1, 1], delay=0)
    on LEFT → Left
    on RIGHT → Right
  }

  state Left {
    AddTorque2d(gameObject=Self, forceMode=0, torque=-400, everyFrame=false)
    on FINISHED → Activate
  }

  state Right {
    AddTorque2d(gameObject=Self, forceMode=0, torque=400, everyFrame=false)
    on FINISHED → Activate
  }

  state Nail Hit {
    GetFsmFloat(gameObject=var "Nail Object", fsmName="damages_enemy", variableName="direction", storeValue=var "Direction", everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetVelocityAsAngle(gameObject=Self, angle=var "Direction", speed=22, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Angle: float = 0
  var Angle Max: float = 0
  var Angle Min: float = 0
  var Direction: float = 0
  var Speed: float = 0
  var Bit1: gameObject = <null>
  var Bit2: gameObject = <null>
  var Bit3: gameObject = <null>
  var Death Spore: gameObject = <null>
  var Grass: gameObject = <null>
  var Nail Object: gameObject = <null>
  var Self: gameObject = <null>
}
