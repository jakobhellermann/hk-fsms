#   Fungus2_05 / Mushroom Brawler 1/Cap Hit
#   Fungus2_29 / Mushroom Brawler/Cap Hit
#   Fungus2_29 / Mushroom Brawler (1)/Cap Hit
#   Fungus2_30 / Mushroom Brawler/Cap Hit
#   Fungus2_30 / Mushroom Brawler (1)/Cap Hit

fsm Unparent, follow and notify {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Follow
  }

  state Follow {
    GameObjectIsNull(gameObject=var "Parent", isNull=→"NULL", isNotNull=(none), storeResult=(unset), everyFrame=true)
    GetPosition(gameObject=var "Parent", vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=var "Parent Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    GetScale(gameObject=var "Parent", vector=(unset), xScale=var "Parent Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Parent Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on BLOCKED HIT → Blocked Hit
    on NULL → Deactivate
  }

  state Blocked Hit {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="BLOCKED HIT", delay=0, everyFrame=false)
    on FINISHED → Follow
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Parent Scale: float = 0
  var Parent Pos: vector3 = (0, 0, 0)
  var Parent: gameObject = <null>
}
