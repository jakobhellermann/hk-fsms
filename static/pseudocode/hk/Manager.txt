# Manager
# 3 location(s):
#   Room_Colosseum_Bronze / Colosseum Manager
#   Room_Colosseum_Silver / Colosseum Manager
#   Room_Colosseum_Gold / Colosseum Manager

// uses template: colosseum_manager
fsm Manager {
  start Pause

  state Pause {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    PreSpawnCorpsesInChildren(target=Self)
    PreBuildTK2DSprites(gameObject=var "Self", useChildren=true)
    EnsurePersonalPoolCreatedInChildren(Target=Self)
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Area Title Controller", storeResult=var "Title")
    FindChild(gameObject=Self, childName="CameraLockArea", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Walls", storeResult=var "Walls")
    FindChild(gameObject=Self, childName="Remasker", storeResult=var "Remasker")
    ActivateGameObject(gameObject=var "Remasker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Tiso Event
  }

  state Idle {
    on HERO ENTER → Entry
  }

  state Entry {
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GATE CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD STILL", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=gate_slam (sharedassets30.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    on HERO CENTRE → Init Cheer
  }

  state Init Cheer {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD CHEER", delay=0.5, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC STING", delay=1, everyFrame=false)
    ActivateAllChildren(gameObject=var "Walls", activate=true)
    ActivateGameObject(gameObject=var "Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="seenColosseumTitle", isTrue=(none), isFalse=→"FIRST TITLE")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Pause
    on FIRST TITLE → Extra Title Pause
  }

  state Start Pause {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD IDLE", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Waves Start
  }

  state Waves Start {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WAVES START", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Remasker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on WAVES COMPLETED → Reward Antic
  }

  state Extra Title Pause {
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD CHEER", delay=0.5, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC STING", delay=1, everyFrame=false)
    on FINISHED → Start Pause
  }

  state Reward Antic {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD STILL", delay=0, everyFrame=false)  // disabled
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reward Cheer
  }

  state Reward Cheer {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD CHEER", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0.25, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Give Geo
  }

  state Give Geo {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="GIVE GEO", delay=0, everyFrame=false)
  }

  state Set Respawn {
    SetPlayerDataString(gameObject=var "GameManager", stringName="respawnMarkerName", value="RestBench")
    SetPlayerDataString(gameObject=var "GameManager", stringName="respawnScene", value="Room_Colosseum_02")
    SetPlayerDataInt(intName="respawnType", value=1)
    SetPlayerDataBool(boolName="respawnFacingRight", value=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SetCurrentMapZoneAsRespawn())
    on FINISHED → Idle
  }

  state Tiso Event {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="tisoEncounteredColosseum", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="tisoDead", value=true)
    on FINISHED → Set Respawn
  }

  var CamLock: gameObject = <null>
  var Remasker: gameObject = <null>
  var Self: gameObject = <null>
  var Title: gameObject = <null>
  var Walls: gameObject = <null>
}
