# Control
# 1 location(s):
#   resources.assets / Knight:1/Charm Effects/Dung

fsm Control {
  start Init
  on CHARM EQUIP CHECK → Check  // from any state
  on ALL CHARMS END → Unequipped  // from any state
  on LEVEL LOADED → Wait for Hero Pos  // from any state

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_10", isTrue=→"EQUIPPED", isFalse=→"UNEQUIPPED")
    on EQUIPPED → Emit Pause
    on UNEQUIPPED → Unequipped
  }

  state Equipped {
    SpawnObjectFromGlobalPoolOverTime(gameObject=Knight Dung Trail, spawnPoint=var "Particle 1", position=(unset), rotation=(unset), frequency=0.75)
  }

  state Init {
    FindChild(gameObject=Self, childName="Particle 1", storeResult=var "Particle 1")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Charms Allowed?
  }

  state Unequipped {
    SetParticleEmission(gameObject=var "Particle 1", emission=false)
    AudioStop(gameObject=Self)
  }

  state Charms Allowed? {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="IsGameplayScene", parameters=[], storeResult=var "Is Gameplay Scene")
    GetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", storeValue=var "No Charms", everyFrame=false)
    BoolTest(boolVariable=var "No Charms", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Is Gameplay Scene", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Check
    on CANCEL → Unequipped
  }

  state Wait for Hero Pos {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Charms Allowed?
  }

  state Emit Pause {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetParticleEmission(gameObject=var "Particle 1", emission=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Equipped
  }

  var Is Gameplay Scene: bool = false
  var No Charms: bool = false
  var Particle 1: gameObject = <null>
}
