fsm Spin Control {
  start Init

  state Spin {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=240, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GameObjectIsNull(gameObject=var "Parent", isNull=→"PARENT NULL", isNotNull=(none), storeResult=(unset), everyFrame=true)
    on STOP → Idle
    on PARENT NULL → Mage Killed
  }

  state Idle {
    GameObjectIsNull(gameObject=var "Parent", isNull=→"PARENT NULL", isNotNull=(none), storeResult=(unset), everyFrame=true)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on SPIN → Spin
    on PARENT NULL → Mage Killed
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Mage Killed {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DISSIPATE", delay=0, everyFrame=false)
  }

  var Spin Speed: float = 240
  var Parent: gameObject = <null>
}
