# shop_control
# 2 location(s):
#   Fungus3_35 / Bank Menu
#   sharedassets218.assets / Bank Menu

// uses template: bank_control
fsm shop_control {
  start Init
  on RESET SHOP WINDOW → Item List Up  // from any state
  on CLOSE SHOP WINDOW → Special Close  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Window", storeResult=var "Window")
    FindChild(gameObject=Self, childName="Item List", storeResult=var "Item List")
    FindChild(gameObject=Self, childName="Arrow D", storeResult=var "Arrow D")
    FindChild(gameObject=Self, childName="Arrow U", storeResult=var "Arrow U")
    FindChild(gameObject=Self, childName="Confirm", storeResult=var "Confirm")
    FindChild(gameObject=var "Confirm", childName="UI List", storeResult=var "Confirm UI List")
    FindChild(gameObject=Self, childName="Menu 1", storeResult=var "Menu 1")
    on FINISHED → Set Figurehead
  }

  state Idle {
    SetIntValue(intVariable=var "Current Item", intValue=0, everyFrame=false)
    on SHOP UP → Open Window
  }

  state Open Window {
    SendEventByName(eventTarget=GameObject(var "Window"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerBalance", storeValue=var "Balance")
    ConvertIntToString(intVariable=var "Balance", stringVariable=var "Balance String", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Txt Balance Amount", textString=var "Balance String")
    on FINISHED → Item List Up
  }

  state Item List Up {
    SendEventByName(eventTarget=GameObject(var "Item List"), sendEvent="UP", delay=0, everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "In Second Menu", boolValue=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Confirm UI List", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Confirm", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Arrow D", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Arrow U", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Opened {
    ListenForMenuActions(eventTarget=Self, submitPressed=(none), cancelPressed=→"CLOSE", ignoreAttack=false)
    on CLOSE → In Second Menu?
  }

  state Down {
    DestroyAllChildren(gameObject=var "Item List", disable=false)
    SendEventByName(eventTarget=GameObject(var "Item List"), sendEvent="UI INACTIVE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Arrow D", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Arrow U", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Menu 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Event
  }

  state Send Event {
    ActivateGameObject(gameObject=var "Item List", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item List"), sendEvent="RESET", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOP CLOSED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Set Figurehead {
    FindChild(gameObject=var "Window", childName="Figurehead", storeResult=var "Current Figurehead")
    on FINISHED → Idle
  }

  state In Second Menu? {
    BoolTest(boolVariable=var "In Second Menu", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    on FINISHED → Down
    on CANCEL → Opened
  }

  state Special Close {
    DestroyAllChildren(gameObject=var "Item List", disable=false)
    SendEventByName(eventTarget=GameObject(var "Item List"), sendEvent="UI INACTIVE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Arrow D", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Arrow U", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Confirm UI List", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Confirm", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Item List", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item List"), sendEvent="RESET", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOP CLOSED QUICK", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Open {
    ActivateGameObject(gameObject=var "Menu 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Opened
  }

  var Balance: int = 0
  var Current Item: int = 0
  var In Second Menu: bool = false
  var Balance String: string = ""
  var Arrow D: gameObject = <null>
  var Arrow U: gameObject = <null>
  var Confirm: gameObject = <null>
  var Confirm UI List: gameObject = <null>
  var Current Figurehead: gameObject = <null>
  var Item List: gameObject = <null>
  var Menu 1: gameObject = Bank Menu/Menu 1
  var Self: gameObject = <null>
  var Txt Balance Amount: gameObject = Bank Menu/Txt Balance Amount
  var Window: gameObject = <null>
}
