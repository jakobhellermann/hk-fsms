# Avoid Knight
# 2 location(s):
#   White_Palace_03_hub / Progress Gate (1)
#   White_Palace_03_hub / Progress Gate (2)

fsm Avoid Knight {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check
  }

  state Check {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=1, tolerance=0, equal=(none), lessThan=→"DISAPPEAR", greaterThan=(none), everyFrame=false)
    on DISAPPEAR → Disappear
  }

  state Disappear {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Hero X: float = 0
}
