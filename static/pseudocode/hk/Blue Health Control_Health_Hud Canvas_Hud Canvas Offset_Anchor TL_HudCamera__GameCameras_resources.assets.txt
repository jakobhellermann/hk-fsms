# Blue Health Control
# 1 location(s):
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health

fsm Blue Health Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Idle {
    on ADD BLUE HEALTH → Add Blue Health
    on UPDATE BLUE HEALTH → Hive Check
  }

  state Add Blue Health {
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue HP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxHealth", storeValue=var "Max HP")
    IntAdd(intVariable=var "Max HP", add=var "Blue HP", everyFrame=false)
    ConvertIntToFloat(intVariable=var "Max HP", floatVariable=var "Spawn Offset Multiplier", everyFrame=false)
    IntAdd(intVariable=var "Blue HP", add=1, everyFrame=false)
    SetPlayerDataInt(intName="healthBlue", value=var "Blue HP")
    SetFloatValue(floatVariable=var "Blue Spawn X", floatValue=-10.32, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn X Offset", floatValue=0.94, everyFrame=false)
    FloatMultiply(floatVariable=var "Spawn X Offset", multiplyBy=var "Spawn Offset Multiplier", everyFrame=false)
    FloatAdd(floatVariable=var "Blue Spawn X", add=var "Spawn X Offset", everyFrame=false, perSecond=false)
    CreateObject(gameObject=Blue Health, spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Blue HP Object", networkInstantiate=false, networkGroup=0)
    SetParent(gameObject=var "Blue HP Object", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Blue HP Object", vector=(unset), x=var "Blue Spawn X", y=7.68, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetFsmBool(gameObject=var "Blue HP Object", fsmName="blue_health_display", variableName="Start Idle", setValue=false, everyFrame=false)
    SetFsmInt(gameObject=var "Blue HP Object", fsmName="blue_health_display", variableName="Health Number", setValue=var "Blue HP", everyFrame=false)
    on FINISHED → Idle
  }

  state Add Existing? {
    IntCompare(integer1=var "Joni HP Counter", integer2=0, equal=(none), lessThan=(none), greaterThan=→"ADD JONI HEALTH", everyFrame=false)
    IntCompare(integer1=var "Blue HP counter", integer2=0, equal=(none), lessThan=(none), greaterThan=→"ADD BLUE HEALTH", everyFrame=false)
    on ADD BLUE HEALTH → Add Blue Health 2
    on FINISHED → Idle
    on ADD JONI HEALTH → Add Joni Health
  }

  state Add Blue Health 2 {
    IntOperator(integer1=var "Blue HP counter", integer2=1, operation=Subtract, storeResult=var "Blue HP counter", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxHealth", storeValue=var "Max HP")
    IntAdd(intVariable=var "Max HP", add=var "Blues Added", everyFrame=false)
    ConvertIntToFloat(intVariable=var "Max HP", floatVariable=var "Spawn Offset Multiplier", everyFrame=false)
    SetFloatValue(floatVariable=var "Blue Spawn X", floatValue=-10.32, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn X Offset", floatValue=0.94, everyFrame=false)
    FloatMultiply(floatVariable=var "Spawn X Offset", multiplyBy=var "Spawn Offset Multiplier", everyFrame=false)
    FloatAdd(floatVariable=var "Blue Spawn X", add=var "Spawn X Offset", everyFrame=false, perSecond=false)
    CreateObject(gameObject=Blue Health, spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Blue HP Object", networkInstantiate=false, networkGroup=0)
    SetParent(gameObject=var "Blue HP Object", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Blue HP Object", vector=(unset), x=var "Blue Spawn X", y=7.68, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    IntAdd(intVariable=var "Blues Added", add=1, everyFrame=false)
    SetFsmBool(gameObject=var "Blue HP Object", fsmName="blue_health_display", variableName="Start Idle", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Blue HP Object", fsmName="blue_health_display", variableName="Init", setValue=true, everyFrame=false)
    SetFsmInt(gameObject=var "Blue HP Object", fsmName="blue_health_display", variableName="Health Number", setValue=var "Blues Added", everyFrame=false)
    on FINISHED → Add Existing?
  }

  state Set Blue {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REMOVE BLUE HEALTH", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Blues Added", add=0, everyFrame=false)  // disabled
    SetIntValue(intVariable=var "Blues Added", intValue=0, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=UpdateBlueHealth())
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue HP counter")
    GetPlayerDataInt(gameObject=var "GameManager", intName="joniHealthBlue", storeValue=var "Joni HP Counter")
    on FINISHED → Add Existing?
  }

  state Add Joni Health {
    IntOperator(integer1=var "Joni HP Counter", integer2=1, operation=Subtract, storeResult=var "Joni HP Counter", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxHealth", storeValue=var "Max HP")
    IntAdd(intVariable=var "Max HP", add=var "Blues Added", everyFrame=false)
    ConvertIntToFloat(intVariable=var "Max HP", floatVariable=var "Spawn Offset Multiplier", everyFrame=false)
    SetFloatValue(floatVariable=var "Blue Spawn X", floatValue=-10.32, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn X Offset", floatValue=0.94, everyFrame=false)
    FloatMultiply(floatVariable=var "Spawn X Offset", multiplyBy=var "Spawn Offset Multiplier", everyFrame=false)
    FloatAdd(floatVariable=var "Blue Spawn X", add=var "Spawn X Offset", everyFrame=false, perSecond=false)
    GGCheckBoundHeart(healthNumber=var "Blues Added", checkSource=Joni, trueEvent=→"FINISHED", falseEvent=(none))
    IntAdd(intVariable=var "Blues Added", add=1, everyFrame=false)
    CreateObject(gameObject=var "Joni Health Object", spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Blue HP Object", networkInstantiate=false, networkGroup=0)
    SetParent(gameObject=var "Blue HP Object", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Blue HP Object", vector=(unset), x=var "Blue Spawn X", y=7.68, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetFsmBool(gameObject=var "Blue HP Object", fsmName="blue_health_display", variableName="Start Idle", setValue=false, everyFrame=false)
    SetFsmInt(gameObject=var "Blue HP Object", fsmName="blue_health_display", variableName="Health Number", setValue=var "Blues Added", everyFrame=false)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="healthBlue")
    on FINISHED → Add Existing?
  }

  state Wait {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)  // disabled
    BoolTest(boolVariable=var "Init", isTrue=(none), isFalse=→"LAST HP ADDED", everyFrame=false)  // disabled
    on LAST HP ADDED → Hive Check
    on INVENTORY OPENED → Hive Check
  }

  state Hive Check {
    SetGameObject(variable=var "Joni Health Object", gameObject=Blue Health, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_29", isTrue=(none), isFalse=→"FINISHED")
    SetGameObject(variable=var "Joni Health Object", gameObject=Blue Health Hive, everyFrame=false)
    on FINISHED → Set Blue
  }

  var Blue Spawn X: float = 0
  var Blue Spawn Y: float = 0
  var Spawn Offset Multiplier: float = 0
  var Spawn X Offset: float = 0
  var Blue HP: int = 0
  var Blue HP counter: int = 0
  var Blues Added: int = 0
  var Joni HP Counter: int = 0
  var Max HP: int = 0
  var Init: bool = false
  var Blue HP Object: gameObject = <null>
  var Joni Health Object: gameObject = <null>
  var Self: gameObject = <null>
}
