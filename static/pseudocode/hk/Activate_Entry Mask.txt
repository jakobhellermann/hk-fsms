# Activate
# 1 location(s):
#   Grimm_Main_Tent / Entry Mask

fsm Activate {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    on LIGHTS UP → Activate
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetCollider(gameObject=Self, active=true)
    on BATTLE END → Deactivate
  }

  state Deactivate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SetCollider(gameObject=Self, active=false)
  }

  var Self: gameObject = <null>
}
