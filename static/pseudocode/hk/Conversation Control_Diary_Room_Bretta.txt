# Conversation Control
# 1 location(s):
#   Room_Bretta / Diary

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    CreateObject(gameObject=Area Title (sharedassets9.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Area Title", networkInstantiate=false, networkGroup=0)  // disabled
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)  // disabled
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="RELICDEALER", everyFrame=false)  // disabled
    on FINISHED → Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Diary 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_DIARY_1", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Choice {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brettaLeftTown", isTrue=→"LEFT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="zoteDefeated", isTrue=→"DIARY 4", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brettaSeenBedDiary", isTrue=→"DIARY 3", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brettaSeenBenchDiary", isTrue=→"DIARY 2", isFalse=(none))
    on FINISHED → Diary 1
    on DIARY 2 → Diary 2
    on DIARY 3 → Diary 3
    on DIARY 4 → Diary 4
    on LEFT → Diary Left
  }

  state Diary 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_DIARY_2", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Diary 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_DIARY_3", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Diary 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_DIARY_4", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Diary Left {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRETTA_DIARY_LEAVE", "CP2"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
}
