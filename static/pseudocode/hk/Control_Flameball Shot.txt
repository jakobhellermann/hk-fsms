# Control
# 1 location(s):
#   sharedassets6.assets / Flameball Shot

fsm Control {
  start Init

  state Init {
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Pt Flame", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Smoke", emit=0)
    Wait(time=5, finishEvent=→"WAIT", realTime=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Accel", floatValue=2, everyFrame=false)
    SetFloatValue(floatVariable=var "Speed", floatValue=12, everyFrame=false)
    on WAIT → Shrink
    on ORBIT SHIELD → Shield
    on SPIRAL → Spiral
  }

  state Shrink {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    StopParticleEmitter(gameObject=var "Pt Flame")
    StopParticleEmitter(gameObject=var "Pt Smoke")
    SetCircleCollider(gameObject=Self, active=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(unset), time=0.5, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Shield {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Shrink
  }

  state Spiral {
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=true)
    FloatAdd(floatVariable=var "Angle", add=100, everyFrame=true, perSecond=true)
    FloatAdd(floatVariable=var "Speed", add=var "Accel", everyFrame=true, perSecond=true)
    FloatAdd(floatVariable=var "Accel", add=2, everyFrame=true, perSecond=true)
    Wait(time=3, finishEvent=→"WAIT", realTime=false)
    on ORBIT SHIELD → Shield
    on WAIT → Shrink
  }

  var Angle: float = 0

  var Accel: float = 1
  var Speed: float = 12
  var Pt Flame: gameObject = Flameball Shot/lava_particles_03
  var Pt Smoke: gameObject = Flameball Shot/Flame_smoke
}
