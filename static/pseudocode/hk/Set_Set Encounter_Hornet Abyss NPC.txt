# Set
# 1 location(s):
#   Abyss_06_Core / Hornet Abyss NPC/Set Encounter

fsm Set {
  start Wait For Hero

  state State 1 {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → State 2
  }

  state State 2 {
    SetPlayerDataBool(boolName="hornetAbyssEncounter", value=true)
  }

  state Wait For Hero {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → State 1
  }
}
