# Control
# 1 location(s):
#   GG_Lurker / Lurker Control/Pale Lurker/Slash Ball

fsm Control {
  start In

  state In {
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 0), everyframe=false)
    SetPolygonCollider(gameObject=Self, active=true)
    EaseFloat(fromValue=0.75, toValue=1.4, floatVariable=var "Scale", time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    SetFloatValue(floatVariable=var "Scale", floatValue=0.75, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=true, lateUpdate=false)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Colour", time=0.15, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Slash Ball")
    on OUT → Out
  }

  state Out {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 0), duration=0.15)
    SetPolygonCollider(gameObject=Self, active=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Scale: float = 0
  var Colour: color = (0, 0, 0, 1)
}
