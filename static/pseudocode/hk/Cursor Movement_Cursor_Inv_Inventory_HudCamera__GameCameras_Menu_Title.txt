# Cursor Movement
# 1 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Inv/Cursor

fsm Cursor Movement {
  start Init
  on DOWN → Deactivate  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=Self, childName="BL", storeResult=var "BL")
    FindChild(gameObject=Self, childName="BR", storeResult=var "BR")
    FindChild(gameObject=Self, childName="TL", storeResult=var "TL")
    FindChild(gameObject=Self, childName="TR", storeResult=var "TR")
    FindChild(gameObject=Self, childName="Back", storeResult=var "Back")
    on FINISHED → Inactive
  }

  state Idle {
    on CURSOR MOVE → Move
  }

  state Move {
    SetVector3XYZ(vector3Variable=var "Back To Pos", vector3Value=(unset), x=var "Box Offset X", y=var "Box Offset Y", z=2, everyFrame=false)
    iTweenMoveTo(gameObject=var "Back", id="", transformPosition=var "", vectorPosition=var "Back To Pos", time=0.15, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=true)
    GetVector2XY(vector2Variable=var "ColliderBounds", storeX=var "Left X", storeY=var "Bot Y", everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Back Scale", vector3Value=(unset), x=var "Left X", y=var "Bot Y", z=1, everyFrame=false)
    iTweenScaleTo(gameObject=var "Back", id="", transformScale=var "", vectorScale=var "Back Scale", time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=true)
    FloatMultiply(floatVariable=var "Left X", multiplyBy=-0.5, everyFrame=false)
    FloatMultiply(floatVariable=var "Bot Y", multiplyBy=-0.5, everyFrame=false)
    GetVector2XY(vector2Variable=var "ColliderBounds", storeX=var "Right X", storeY=var "Top Y", everyFrame=false)
    FloatMultiply(floatVariable=var "Right X", multiplyBy=0.5, everyFrame=false)
    FloatMultiply(floatVariable=var "Top Y", multiplyBy=0.5, everyFrame=false)
    FloatAdd(floatVariable=var "Bot Y", add=var "Box Offset Y", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Top Y", add=var "Box Offset Y", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Right X", add=var "Box Offset X", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Left X", add=var "Box Offset X", everyFrame=false, perSecond=false)
    SetVector3XYZ(vector3Variable=var "BL Pos", vector3Value=(unset), x=var "Left X", y=var "Bot Y", z=-4.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "BR Pos", vector3Value=(unset), x=var "Right X", y=var "Bot Y", z=-4.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "TL Pos", vector3Value=(unset), x=var "Left X", y=var "Top Y", z=-4.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "TR Pos", vector3Value=(unset), x=var "Right X", y=var "Top Y", z=-4.5, everyFrame=false)
    iTweenMoveTo(gameObject=var "BL", id="", transformPosition=var "", vectorPosition=var "BL Pos", time=0.15, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=true)
    iTweenMoveTo(gameObject=var "BR", id="", transformPosition=var "", vectorPosition=var "BR Pos", time=0.15, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=true)
    iTweenMoveTo(gameObject=var "TL", id="", transformPosition=var "", vectorPosition=var "TL Pos", time=0.15, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=true)
    iTweenMoveTo(gameObject=var "TR", id="", transformPosition=var "", vectorPosition=var "TR Pos", time=0.15, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=true)
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "MoveToPos", time=0.15, delay=0, speed=(unset), space=1, easeType=linear, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=true)
    GetPosition(gameObject=Self, vector=var "Current Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    Vector3Compare(vector3Variable1=var "Current Pos", vector3Variable2=var "MoveToPos", tolerance=0.01, equal=→"FINISHED", notEqual=(none), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (resources.assets), spawnPoint=var "Self", audioClip=ui_change_selection (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=0.75, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="UPDATE TEXT", delay=0, everyFrame=false)
    on FINISHED → Idle
    on CURSOR MOVE → Quick Move
  }

  state Quick Move {
    on FINISHED → Move
  }

  state Inactive {
    on CURSOR ACTIVATE → Cursor Activate
  }

  state Cursor Activate {
    SetPosition(gameObject=var "Back", vector=(unset), x=0, y=0, z=2, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "BL", vector=(unset), x=-0.5, y=-0.5, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "BR", vector=(unset), x=0.5, y=-0.5, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "TL", vector=(unset), x=-0.5, y=0.5, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "TR", vector=(unset), x=0.5, y=0.5, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UP", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=<null>, spawnPoint=var "Self", audioClip=ui_change_selection (sharedassets0.assets), pitchMin=1, pitchMax=1, volume=0.75, delay=0, storePlayer=<null>)  // disabled
    on FINISHED → Idle
  }

  state Deactivate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on FINISHED → Inactive
  }

  var Bot Y: float = 0
  var Box Offset X: float = 0
  var Box Offset Y: float = 0
  var Left X: float = 0
  var Right X: float = 0
  var Top Y: float = 0
  var ColliderBounds: vector2 = (0, 0)
  var BL Pos: vector3 = (0, 0, 0)
  var BR Pos: vector3 = (0, 0, 0)
  var Back Scale: vector3 = (0, 0, 0)
  var Back To Pos: vector3 = (0, 0, 0)
  var Current Pos: vector3 = (0, 0, 0)
  var MoveToPos: vector3 = (0, 0, 0)
  var TL Pos: vector3 = (0, 0, 0)
  var TR Pos: vector3 = (0, 0, 0)
  var BL: gameObject = <null>
  var BR: gameObject = <null>
  var Back: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var TL: gameObject = <null>
  var TR: gameObject = <null>
}
