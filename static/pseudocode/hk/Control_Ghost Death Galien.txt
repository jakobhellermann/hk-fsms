# Control
# 1 location(s):
#   sharedassets298.assets / Ghost Death Galien

fsm Control {
  start Music

  state Pause {
    GetOwner(storeGameObject=var "Self")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=boss_final_hit (sharedassets32.assets))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Blow
  }

  state Blow {
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Galien_death, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=dream_ghost_appear (resources.assets))
    PlayParticleEmitter(gameObject=var "Burst Pt", emit=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    StopParticleEmitter(gameObject=var "Burst Pt")
    SetMeshRenderer(gameObject=Self, active=false)
    CreateObject(gameObject=Ghost Death (sharedassets161.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=boss_explode_clean (resources.assets))
  }

  state Music {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="NONE", delay=0, everyFrame=false)
    on FINISHED → Pause
  }

  var Burst Pt: gameObject = Ghost Death Galien/Vanish Burst Pt
  var Self: gameObject = <null>
}
