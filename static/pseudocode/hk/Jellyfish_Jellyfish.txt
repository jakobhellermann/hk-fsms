fsm Jellyfish {
  start Init
  on ZERO HP → Detach  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "X Min", floatValue=var "Start X", everyFrame=false)
    SetFloatValue(floatVariable=var "X Max", floatValue=var "Start X", everyFrame=false)
    FloatOperator(float1=var "X Min", float2=var "X Roam", operation=Subtract, storeResult=var "X Min", everyFrame=false)
    FloatOperator(float1=var "X Max", float2=var "X Roam", operation=Add, storeResult=var "X Min", everyFrame=false)
    RandomFloat(min=0, max=3, storeResult=var "Chooser")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=var "Chooser", z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Sleep
  }

  state Idle {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Current Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Current Y", float2=var "Start Y", tolerance=0, equal=(none), lessThan=→"JUMP", greaterThan=(none), everyFrame=true)
    on JUMP → Sleep?
  }

  state Check Dir {
    GetPosition(gameObject=Self, vector=(unset), x=var "Current X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Current X", float2=var "X Min", tolerance=0, equal=(none), lessThan=→"RIGHT", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Current X", float2=var "X Max", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"LEFT", everyFrame=false)
    on RIGHT → Jump R
    on LEFT → Jump L
    on FINISHED → Jump Either
  }

  state Jump {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Jump", animationTriggerEvent=→"JUMP", animationCompleteEvent=(none))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on JUMP → Shoot
  }

  state Shoot {
    RandomFloat(min=7.7, max=8.3, storeResult=var "Y Veloc")
    PlayParticleEmitter(gameObject=var "Ring F", emit=0)
    PlayParticleEmitter(gameObject=var "Ring B", emit=0)
    PlayParticleEmitter(gameObject=var "Trail", emit=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Veloc", y=var "Y Veloc", everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Jump R {
    RandomFloat(min=0, max=1, storeResult=var "X Veloc")
    on FINISHED → Jump
  }

  state Jump L {
    RandomFloat(min=-1, max=0, storeResult=var "X Veloc")
    on FINISHED → Jump
  }

  state Jump Either {
    RandomFloat(min=-1, max=1, storeResult=var "X Veloc")
    on FINISHED → Jump
  }

  state Detach {
    DestroyObject(gameObject=var "Tent Box", delay=0, detachChildren=false)
    DetachChildren(gameObject=Self)
  }

  state Sleep {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=31, tolerance=0, equal=(none), lessThan=→"WAKE", greaterThan=(none), everyFrame=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on WAKE → Idle
  }

  state Sleep? {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=32, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SLEEP", everyFrame=false)
    on SLEEP → Sleep
    on FINISHED → Check Dir
  }

  var Chooser: float = 0
  var Current X: float = 0
  var Current Y: float = 0
  var Distance: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var X Roam: float = 0.5
  var X Veloc: float = 0
  var Y Veloc: float = 0
  var Ring B: gameObject = Jellyfish/Ring B
  var Ring F: gameObject = Jellyfish/Ring F
  var Tent Box: gameObject = Jellyfish/Tentacle Box
  var Trail: gameObject = Jellyfish/Trail
}
