# Display
# 1 location(s):
#   resources.assets / _GameCameras/HudCamera/DialogueManager/Dream Msg

fsm Display {
  start Init
  on DISPLAY ENEMY DREAM → Check Convo  // from any state
  on DISPLAY DREAM MSG → Cancel  // from any state
  on DISPLAY DREAM MSG ALT → Set Sprite  // from any state
  on CANCEL ENEMY DREAM → Full Cancel  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Text", storeResult=var "Text")
    FindChild(gameObject=Self, childName="Backboard", storeResult=var "Backboard")
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "Enemy Dream Msg", gameObject=var "Self", everyFrame=false)
    FindChild(gameObject=Self, childName="Dream_Nail_Dialogue_Backboard", storeResult=var "Backboard Sprite Obj")
    on FINISHED → Idle
  }

  state Idle {
    on DISPLAY ENEMY DREAM → Check Convo
  }

  state Cancel Existing {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN INSTANT", delay=0, everyFrame=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    FadeGroupDown(target=Self, fast=true)
    FadeGroupDown(target=var "Backboard", fast=true)
    on FINISHED → Set Convo
  }

  state Set Convo {
    RandomInt(min=1, max=var "Convo Amount", storeResult=var "Convo Num", inclusiveMax=true)
    ConvertIntToString(intVariable=var "Convo Num", stringVariable=var "Convo Num Str", format="", everyFrame=false)
    BuildString(stringParts=[var "Convo Title", "_", var "Convo Num Str"], separator="", addToEnd=true, storeResult=var "Convo Title", everyFrame=false)
    GetLanguageString(sheetName="Enemy Dreams", convName=var "Convo Title", storeValue=var "Convo Text")
    SetTextMeshProText(gameObject=var "Text", textString=var "Convo Text")
    on FINISHED → Display Text
  }

  state Check Convo {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN INSTANT", delay=0, everyFrame=false)  // disabled
    StringCompare(stringVariable=var "Convo Title", compareTo="", equalEvent=→"CANCEL", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Cancel Existing
    on CANCEL → Idle
  }

  state Display Text {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    FadeGroupUp(target=Self)
    FadeGroupUp(target=var "Backboard")
    on FINISHED → Text Down
    on LEVEL LOADED → Text Down
  }

  state Text Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)  // disabled
    FadeGroupDown(target=Self, fast=false)
    FadeGroupDown(target=var "Backboard", fast=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return Sprite
  }

  state Cancel {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN INSTANT", delay=0, everyFrame=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    FadeGroupDown(target=Self, fast=true)
    FadeGroupDown(target=var "Backboard", fast=true)
    on FINISHED → Msg Text
  }

  state Msg Text {
    GetLanguageString(sheetName=var "Sheet", convName=var "Convo Title", storeValue=var "Convo Text")
    SetTextMeshProText(gameObject=var "Text", textString=var "Convo Text")
    on FINISHED → Display Text
  }

  state Set Sprite {
    GetSpriteRendererSprite(gameObject=var "Backboard Sprite Obj", sprite=var "Backboard Sprite")
    SetSpriteRendererSprite(gameObject=var "Backboard Sprite Obj", sprite=Dream_Nail_Dialogue_Backboard_White)
    SetBoolValue(boolVariable=var "Changed Backboard Sprite", boolValue=true, everyFrame=false)
    on FINISHED → Cancel
  }

  state Return Sprite {
    BoolTest(boolVariable=var "Changed Backboard Sprite", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetSpriteRendererSprite(gameObject=var "Backboard Sprite Obj", sprite=var "Backboard Sprite")
    on FINISHED → Idle
  }

  state Full Cancel {
    FadeGroupDown(target=Self, fast=true)
    FadeGroupDown(target=var "Backboard", fast=true)
    on FINISHED → Idle
  }

  var Convo Amount: int = 0
  var Convo Num: int = 0
  var Changed Backboard Sprite: bool = false
  var Convo Num Str: string = ""
  var Convo Text: string = ""
  var Convo Title: string = ""
  var Sheet: string = ""
  var Backboard: gameObject = <null>
  var Backboard Sprite Obj: gameObject = <null>
  var Self: gameObject = <null>
  var Text: gameObject = <null>
  var Backboard Sprite: object = <null>
}
