# Activate Lantern
# 1 location(s):
#   Cliffs_06 / Sycophant Dream

fsm Activate Lantern {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Impact Pt", storeResult=var "Impact Pt")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="nightmareLanternAppeared", isTrue=→"ACTIVATED", isFalse=(none))
    on FINISHED → Idle
    on ACTIVATED → Activated
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Set PD
  }

  state Set PD {
    SetPlayerDataBool(boolName="nightmareLanternAppeared", value=true)
    on FINISHED → Impact
  }

  state Impact {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="TORCH APPEAR", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Impact Pt", emit=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM DIALOGUE START", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    SetCollider(gameObject=var "Inspect Region", active=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Inspect Region"), sendEvent="CONVO CANCEL", delay=0, everyFrame=false)  // disabled
  }

  state Flare {
    ActivateGameObject(gameObject=var "Burn Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "particle_flame", emit=0)
    PlayParticleEmitter(gameObject=var "Smoke", emit=0)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(0, 0, 0, 1), colorVariable=var "Colour", time=0.7, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=var "Sprite", materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    on FINISHED → Burn
  }

  state Burn {
    EaseColor(fromValue=(0, 0, 0, 1), toValue=(0, 0, 0, 0), colorVariable=var "Colour", time=0.7, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMaterialColor(gameObject=var "Sprite", materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    on FINISHED → End
  }

  state End {
    StopParticleEmitter(gameObject=var "Smoke")
    StopParticleEmitter(gameObject=var "particle_flame")
  }

  state Wait {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=77.7, tolerance=0, equal=(none), lessThan=→"BURN", greaterThan=(none), everyFrame=true)
    on BURN → Flare
  }

  state Activated {
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Hero X: float = 0
  var Colour: color = (0, 0, 0, 1)
  var Burn Flash: gameObject = Sycophant Dream/Burn Flash
  var Hit Flash: gameObject = Sycophant Dream/Hit Flash
  var Impact Pt: gameObject = Dream Dialogue/Impact Pt (sharedassets6.assets)
  var Inspect Region: gameObject = Sycophant Dream/Inspect Region
  var Self: gameObject = <null>
  var Smoke: gameObject = Sycophant Dream/Smoke
  var Sprite: gameObject = Sycophant Dream/Sprite
  var particle_flame: gameObject = Sycophant Dream/particle_flame
}
