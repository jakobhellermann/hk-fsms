# Thorn Counter
# 1 location(s):
#   Knight_Pickup / Knight/Charm Effects

fsm Thorn Counter {
  start Init
  on LEVEL LOADED → Idle  // from any state

  state Idle {
    ActivateGameObject(gameObject=var "Thorn Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on THORN COUNTER → Check Equipped
  }

  state Check Equipped {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_12", isTrue=→"EQUIPPED", isFalse=→"CANCEL")
    on CANCEL → Idle
    on EQUIPPED → Counter Start
  }

  state Counter Start {
    GetPosition(gameObject=var "Knight", vector=var "Thorn Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Knight", vector=var "Thorn Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    SetVelocity2d(gameObject=var "Knight", vector=(unset), x=0, y=0, everyFrame=true)
    SendMessage(gameObject=var "Knight", delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    SendMessage(gameObject=var "Knight", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Knight", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (resources.assets), spawnPoint=var "Knight", audioClip=hero_thorn_counter (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=var "Knight", clipName="Thorn Attack", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.06, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Thorn Scale
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Knight")
    FindChild(gameObject=Self, childName="Thorn Hit", storeResult=var "Thorn Hit")
    on FINISHED → Idle
  }

  state Counter {
    Tk2dWatchAnimationEvents(gameObject=var "Knight", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPosition(gameObject=var "Knight", vector=var "Thorn Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Knight", vector=(unset), x=0, y=0, everyFrame=true)
    ActivateGameObject(gameObject=var "Thorn Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Counter End
  }

  state Counter End {
    SendMessage(gameObject=var "Knight", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Knight", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SendMessage(gameObject=var "Knight", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    ActivateGameObject(gameObject=var "Thorn Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Set Thorn Scale {
    GetScale(gameObject=var "Knight", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Thorn Hit", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetScale(gameObject=var "Thorn Hit", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Counter
  }

  var X Scale: float = 0
  var Thorn Pos: vector3 = (0, 0, 0)
  var Knight: gameObject = <null>
  var Thorn Hit: gameObject = <null>
}
