# Focus Speedup
# 1 location(s):
#   resources.assets / Orbit Shield

fsm Focus Speedup {
  start Idle

  state Idle {
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Speed", setValue=110, everyFrame=false)
    on HERO FOCUS START → Focus
  }

  state Focus {
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Speed", setValue=300, everyFrame=false)
    AudioPlayInState(gameObject=Self, volume=0.6)
    on HERO FOCUS END → Idle
  }
}
