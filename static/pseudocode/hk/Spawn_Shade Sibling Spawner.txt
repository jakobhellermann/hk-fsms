# Spawn
# 1 location(s):
#   Abyss_06_Core / Shade Sibling Spawner

fsm Spawn {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Spawn Y", z=(unset), space=0, everyFrame=false)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_36", falseBool="openedBlackEggPath", isTrue=→"INERT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="visitedAbyssLower", isTrue=→"SHADE SPAWN READY", isFalse=(none))
    on SHADE SPAWN READY → Idle
    on BIG TITLE END → Idle
    on INERT → Inert
  }

  state Idle {
    BoolTest(boolVariable=var "Hero Range", isTrue=→"IN RANGE", isFalse=(none), everyFrame=true)
    on IN RANGE → In Range
  }

  state In Range {
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"OUT OF RANGE", everyFrame=true)
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=var "Spawn Time", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SPAWN", everyFrame=true)
    on OUT OF RANGE → Idle
    on SPAWN → Check
  }

  state Check {
    SetFloatValue(floatVariable=var "Timer", floatValue=0, everyFrame=false)
    GetTagCount(tag="Boss", storeResult=var "Siblings")
    IntCompare(integer1=var "Siblings", integer2=2, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → In Range
    on FINISHED → Spawn
  }

  state Spawn {
    RandomFloat(min=5, max=7, storeResult=var "Spawn Time")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Spawn X", y=(unset), z=(unset), space=0, everyFrame=false)
    RandomFloat(min=-10, max=10, storeResult=var "X Modifier")
    FloatAdd(floatVariable=var "Spawn X", add=var "X Modifier", everyFrame=false, perSecond=false)
    FloatClamp(floatVariable=var "Spawn X", minValue=var "Spawn X Min", maxValue=var "Spawn X Max", everyFrame=false)
    RandomFloat(min=var "Spawn X Min", max=var "Spawn X Max", storeResult=var "Spawn X")  // disabled
    SetVector3XYZ(vector3Variable=var "Spawn Pos", vector3Value=(unset), x=var "Spawn X", y=var "Spawn Y", z=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Shade Sibling (sharedassets334.assets), spawnPoint=<null>, position=var "Spawn Pos", rotation=(unset), storeObject=var "")
    on FINISHED → In Range
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inert {
  }

  var Spawn X Max: float = 88.41
  var Spawn X Min: float = 10.44041

  var Hero X: float = 0
  var Spawn Time: float = 3
  var Spawn X: float = 89.03
  var Spawn Y: float = 0
  var Timer: float = 0
  var X Modifier: float = 0
  var Siblings: int = 0
  var Hero Range: bool = false
  var Spawn Pos: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
}
