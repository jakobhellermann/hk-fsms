# charm_cost_indicator
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Charms/Equipped Charms/Notches/Charm Cost 4
#   resources.assets / _GameCameras/HudCamera/Inventory/Charms/Equipped Charms/Notches/Charm Cost 4

// uses template: charm_cost_indicator
fsm charm_cost_indicator {
  start Init
  on CHARM INDICATOR CHECK → Check  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Sprite Empty", storeResult=var "Sprite Empty")
    FindChild(gameObject=var "Self", childName="Sprite Full", storeResult=var "Sprite Full")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlots", storeValue=var "Charm Slots")
    IntCompare(integer1=var "Charm Slots", integer2=var "Indicator Number", equal=(none), lessThan=→"INACTIVE", greaterThan=(none), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlotsFilled", storeValue=var "Charm Slots Filled")
    IntCompare(integer1=var "Charm Slots Filled", integer2=var "Indicator Number", equal=→"FULL", lessThan=→"EMPTY", greaterThan=→"FULL", everyFrame=false)
    on EMPTY → Empty
    on FULL → Full
    on INACTIVE → Inactive
  }

  state Empty {
    SetPosition(gameObject=var "Sprite Empty", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Sprite Full", vector=(unset), x=(unset), y=50, z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  state Full {
    SetPosition(gameObject=var "Sprite Empty", vector=(unset), x=(unset), y=50, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Sprite Full", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  state Inactive {
    SetSpriteRenderer(gameObject=var "Sprite Empty", active=false)
    SetSpriteRenderer(gameObject=var "Sprite Full", active=false)
    SetPosition(gameObject=var "Sprite Empty", vector=(unset), x=(unset), y=50, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Sprite Full", vector=(unset), x=(unset), y=50, z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  var Indicator Number: int = 4

  var Charm Slots: int = 0
  var Charm Slots Filled: int = 0
  var Self: gameObject = <null>
  var Sprite Empty: gameObject = <null>
  var Sprite Full: gameObject = <null>
}
