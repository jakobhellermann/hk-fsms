# vessel_orb
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels/Vessel 2
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels/Vessel 2

// uses template: vessel_orb
fsm vessel_orb {
  start Setup
  on BIND VESSEL ORB → Bound  // from any state
  on UNBIND VESSEL ORB → Init  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=var "Binding Cap", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=var "Self", childName="Flash", storeResult=var "Flash")
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserveMax", storeValue=var "Reserve MP Max")
    SetIntValue(intVariable=var "Quarter Amount", intValue=var "Empty Amount", everyFrame=false)
    SetIntValue(intVariable=var "Half Amount", intValue=var "Empty Amount", everyFrame=false)
    SetIntValue(intVariable=var "3Quarter Amount", intValue=var "Empty Amount", everyFrame=false)
    IntAdd(intVariable=var "Quarter Amount", add=8, everyFrame=false)
    IntAdd(intVariable=var "Half Amount", add=16, everyFrame=false)
    IntAdd(intVariable=var "3Quarter Amount", add=24, everyFrame=false)
    SetFsmInt(gameObject=var "Self", fsmName="vessel_size", variableName="Empty Amount", setValue=var "Empty Amount", everyFrame=false)
    SetFsmInt(gameObject=var "Self", fsmName="vessel_size", variableName="Full Amount", setValue=var "Full Amount", everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Self", fsm="vessel_size"), sendEvent="START", delay=0, everyFrame=false)
    IntCompare(integer1=var "Reserve MP Max", integer2=var "Full Amount", equal=(none), lessThan=→"NOT OBTAINED", greaterThan=(none), everyFrame=false)
    on NOT OBTAINED → Not Obtained
    on FINISHED → Load Anim?
    on NEW SOUL ORB → Pause
  }

  state Not Obtained {
    SetMeshRenderer(gameObject=Self, active=false)
    on NEW SOUL ORB → Pause
    on LEVEL LOADED → Pause
  }

  state Down Check {
    SetMeshRenderer(gameObject=Self, active=true)
    SendEventByName(eventTarget=GameObjectFSM(var "Self", fsm="vessel_size"), sendEvent="SIZE CHECK", delay=0, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserve", storeValue=var "Reserve MP")
    IntCompare(integer1=var "Reserve MP", integer2=var "Full Amount", equal=→"FULL", lessThan=(none), greaterThan=→"FULL", everyFrame=false)
    IntCompare(integer1=var "Reserve MP", integer2=var "3Quarter Amount", equal=→"3QUARTER", lessThan=(none), greaterThan=→"3QUARTER", everyFrame=false)
    IntCompare(integer1=var "Reserve MP", integer2=var "Half Amount", equal=→"HALF", lessThan=(none), greaterThan=→"HALF", everyFrame=false)
    IntCompare(integer1=var "Reserve MP", integer2=var "Quarter Amount", equal=→"QUARTER", lessThan=→"EMPTY", greaterThan=→"QUARTER", everyFrame=false)
    on EMPTY → Down To Empty
    on QUARTER → Down To Quarter
    on HALF → Down To Half
    on 3QUARTER → Down To 3Quarter
    on FULL → Down To Full
  }

  state Down To Empty {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_DownToEmpty", doNotResetCurrentClip=true)
    on MP RESERVE UP → Flash?
    on MP RESERVE DOWN → Down Check
    on MP RESERVE ZERO → Init Anim
  }

  state Up Check {
    SetMeshRenderer(gameObject=Self, active=true)
    SendEventByName(eventTarget=GameObjectFSM(var "Self", fsm="vessel_size"), sendEvent="SIZE CHECK", delay=0, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserve", storeValue=var "Reserve MP")
    IntCompare(integer1=var "Reserve MP", integer2=var "Full Amount", equal=→"FULL", lessThan=(none), greaterThan=→"FULL", everyFrame=false)
    IntCompare(integer1=var "Reserve MP", integer2=var "3Quarter Amount", equal=→"3QUARTER", lessThan=(none), greaterThan=→"3QUARTER", everyFrame=false)
    IntCompare(integer1=var "Reserve MP", integer2=var "Half Amount", equal=→"HALF", lessThan=(none), greaterThan=→"HALF", everyFrame=false)
    IntCompare(integer1=var "Reserve MP", integer2=var "Quarter Amount", equal=→"QUARTER", lessThan=→"EMPTY", greaterThan=→"QUARTER", everyFrame=false)
    on EMPTY → Empty
    on QUARTER → Up To Quarter
    on HALF → Up To Half
    on 3QUARTER → Up To 3Quarter
    on FULL → Up To Full
  }

  state Up To Quarter {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_UpToQuarter", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE UP → Flash?
    on MP RESERVE ZERO → Init Anim
  }

  state Down To Quarter {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_DownToQuarter", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE UP → Flash?
    on MP RESERVE ZERO → Init Anim
  }

  state Down To Half {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_DownToHalf", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE UP → Flash?
    on MP RESERVE ZERO → Init Anim
  }

  state Up To Half {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_UpToHalf", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE UP → Flash?
    on MP RESERVE ZERO → Init Anim
  }

  state Down To 3Quarter {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_DownTo3Quarter", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE UP → Flash?
    on MP RESERVE ZERO → Init Anim
  }

  state Up To 3Quarter {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_UpTo3Quarter", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE UP → Flash?
    on MP RESERVE ZERO → Init Anim
  }

  state Down To Full {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_Full", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE ZERO → Init Anim
  }

  state Up To Full {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_UpToFull", doNotResetCurrentClip=true)
    on MP RESERVE DOWN → Down Check
    on MP RESERVE UP → Flash?
    on MP RESERVE ZERO → Init Anim
  }

  state Flash? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserve", storeValue=var "Reserve MP")
    IntCompare(integer1=var "Reserve MP", integer2=var "Empty Amount", equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Reserve MP", integer2=var "Full Amount", equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Flash"), sendEvent="FLASH", delay=0, everyFrame=false)
    on FINISHED → Up Check
  }

  state Appear? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserveMax", storeValue=var "Reserve MP Max")
    IntCompare(integer1=var "Reserve MP Max", integer2=var "Full Amount", equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="V_New", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on CANCEL → Not Obtained
    on FINISHED → Empty
  }

  state Load Anim? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="isFirstGame", isTrue=(none), isFalse=→"FULL")  // disabled
    BoolTest(boolVariable=var "Initialised", isTrue=→"FULL", isFalse=(none), everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "Initialised", boolValue=true, everyFrame=false)  // disabled
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=var "Load Anim Pause", finishEvent=→"FINISHED", realTime=false)
    on FULL → Init Anim
    on FINISHED → Load Anim
  }

  state Load Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="V_New", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Empty
  }

  state Init Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="V_Empty")
    on FINISHED → Down Check
  }

  state Empty {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="V_Empty", doNotResetCurrentClip=true)
    on MP RESERVE UP → Up Check
    on MP RESERVE DOWN → Down Check
    on MP RESERVE ZERO → Init Anim
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Appear?
  }

  state Bound {
    IntCompare(integer1=var "Reserve MP Max", integer2=var "Full Amount", equal=(none), lessThan=→"NOT OBTAINED", greaterThan=(none), everyFrame=false)
    on NOT OBTAINED → Not obtained
    on FINISHED → Obtained
  }

  state Setup {
    FindChild(gameObject=Self, childName="Binding", storeResult=var "Binding Cap")
    on FINISHED → Init
  }

  state Obtained {
    ActivateGameObject(gameObject=var "Binding Cap", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Not obtained {
  }

  var Load Anim Pause: float = 2.2
  var Empty Amount: int = 33
  var Full Amount: int = 66

  var 3Quarter Amount: int = 0
  var Half Amount: int = 0
  var Quarter Amount: int = 0
  var Reserve MP: int = 0
  var Reserve MP Max: int = 0
  var Initialised: bool = false
  var Binding Cap: gameObject = <null>
  var Flash: gameObject = <null>
  var Self: gameObject = <null>
}
