# Flukeman Bot
# 1 location(s):
#   sharedassets350.assets / Corpse Flukeman/Flukeman Bot

fsm Flukeman Bot {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Climb Sprite", storeResult=var "Climb Sprite")
    on FINISHED → Falling
  }

  state Falling {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LANDED", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LANDED", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    RayCast2d(fromGameObject=Self, fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=→"LANDED", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on LANDED → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bot Land")
    WaitRandom(timeMin=3, timeMax=4, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Revive
  }

  state Revive {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=(unset))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bot Revive", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(false))
    SetAudioVolume(gameObject=var "Climb Sprite", volume=1, everyFrame=false)
    on FINISHED → Chase
  }

  state Chase {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))  // disabled
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)  // disabled
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Anim Run", variableName="Anim", setValue=true, everyFrame=false)
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=15, acceleration=0.4, animateTurnAndRun=true, runAnimation="Bot TurnToRun", turnAnimation="Bot Turn", turnRange=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bot Run")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=true, everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on TOOK DAMAGE → Hit Slow
    on WALL → Hop
    on GO DOWN → Go Down Msg
  }

  state Hit Slow {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Chase
    on GO DOWN → Go Down Msg
  }

  state Hop {
    RandomFloat(min=5, max=10, storeResult=var "Hop Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Hop Speed", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bot Climb")
    CheckCollisionSide(topHit=(unset), rightHit=var "Wall R", bottomHit=(unset), leftHit=var "Wall L", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolNoneTrue(boolVariables=[var "Wall L", var "Wall R"], sendEvent=→"STOP", storeResult=(unset), everyFrame=true)
    on STOP → Chase
    on GO DOWN → Go Down Msg
  }

  state Go Down Msg {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bot Run")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Chase
  }

  var Bump Vel: float = 0
  var Hop Speed: float = 0
  var Wall L: bool = false
  var Wall R: bool = false
  var Climb Sprite: gameObject = <null>
  var Self: gameObject = <null>
}
