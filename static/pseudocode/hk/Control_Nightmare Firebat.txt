# Control
# 1 location(s):
#   sharedassets393.assets / Nightmare Firebat

fsm Control {
  start Init
  on DESTROY → Dissipate  // from any state

  state Init {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    PlayParticleEmitter(gameObject=var "Particle System", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Smoke", emit=0)
    SetDamageHeroAmount(target=var "Hero Hurter", damageDealt=0)
    SetMeshRenderer(gameObject=Self, active=true)
    NextFrameEvent(sendEvent=→"FIRE")
    ActivateGameObject(gameObject=var "Spawn Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flash Damage", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Self Y", add=-0.5, everyFrame=false, perSecond=false)
    FloatClamp(floatVariable=var "Self Y", minValue=7.25, maxValue=8.9, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=var "Translate", z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    RandomFloat(min=-0.3, max=0.3, storeResult=var "Translate")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    on FIRE → Fire
    on HIGH → High
    on MID → Mid
  }

  state Fire {
    SetDamageHeroAmount(target=var "Hero Hurter", damageDealt=2)
    ActivateGameObject(gameObject=var "Flash Damage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"END", storeCollider=var "", storeForce=(unset))
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=30, accelerationForce=16, offsetX=0, offsetY=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    on DISSIPATE → Dissipate
    on ORBIT SHIELD → Impact
    on END → Impact
  }

  state Recycle {
    RecycleSelf()
  }

  state Impact {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_lord_projectile_impact (resources.assets))
    SetMeshRenderer(gameObject=Self, active=false)
    SetDamageHeroAmount(target=var "Hero Hurter", damageDealt=0)
    ActivateGameObject(gameObject=var "Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Impact Particles", emit=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Particle System")
    StopParticleEmitter(gameObject=var "Pt Smoke")
    Tk2dPlayAnimationWithEvents(gameObject=var "Impact", clipName="Impact", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Impact End
  }

  state Dissipate {
    AudioStop(gameObject=Self)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.1, 0.1, 0.1), time=0.2, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetDamageHeroAmount(target=var "Hero Hurter", damageDealt=0)
    DecelerateV2(gameObject=Self, deceleration=0.9)
    on FINISHED → Stop Particles
  }

  state Stop Particles {
    StopParticleEmitter(gameObject=var "Particle System")
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Pt Smoke")
    on FINISHED → Recycle
  }

  state Impact End {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Impact", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Tween Up {
    iTweenMoveBy(gameObject=Self, id="", vector=var "Tween Vector", time=0.3, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=true)
    SetDamageHeroAmount(target=var "Hero Hurter", damageDealt=2)
    ActivateGameObject(gameObject=var "Flash Damage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"END", storeCollider=var "", storeForce=(unset))
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    on FINISHED → Fire
    on END → Impact
    on ORBIT SHIELD → Impact
  }

  state High {
    SetVector3XYZ(vector3Variable=var "Tween Vector", vector3Value=(unset), x=0, y=6, z=0, everyFrame=false)
    on FINISHED → Tween Up
  }

  state Mid {
    SetVector3XYZ(vector3Variable=var "Tween Vector", vector3Value=(unset), x=0, y=3, z=0, everyFrame=false)
    on FINISHED → Tween Up
  }

  var Self Y: float = 0
  var Translate: float = 0
  var X Velocity: float = 0
  var Tween Vector: vector3 = (0, 0, 0)
  var Flash Damage: gameObject = Nightmare Firebat/Flash Damage
  var Hero Hurter: gameObject = Nightmare Firebat/Hero Hurter
  var Impact: gameObject = Nightmare Firebat/Impact
  var Impact Particles: gameObject = <null>
  var Particle System: gameObject = Nightmare Firebat/Pt Fire
  var Pt Smoke: gameObject = Nightmare Firebat/Flame_smoke
  var Spawn Flash: gameObject = Nightmare Firebat/Spawn Flash
}
