# xero_nail
# 3 location(s):
#   RestingGrounds_02_boss / Warrior/Ghost Warrior Xero/Sword 4
#   GG_Ghost_Xero / Warrior/Ghost Warrior Xero/Sword 4
#   GG_Ghost_Xero_V / Warrior/Ghost Warrior Xero/Sword 4

// uses template: xero_nail
fsm xero_nail {
  start Init
  on GHOST DEAD → Dissipate  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetChild(gameObject=var "Parent", childName=var "Home Name", withTag="Untagged", storeResult=var "Home")
    GetChild(gameObject=Self, childName="Idle Pt", withTag="Untagged", storeResult=var "Idle Pt")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    iTweenRotateBy(gameObject=Self, id="", vector=(0, 0, 30), time=1, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Home
  }

  state Home {
    DistanceFlySmooth(gameObject=Self, target=var "Home", distance=0, speedMax=15, accelerationForce=100, targetRadius=0.5, deceleration=0.9, offset=(0, 0, 0))
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0.5, storeAngle=var "Angle To Hero", everyFrame=true)  // disabled
    RotateTo(gameObject=Self, targetAngle=var "Angle To Hero", speed=360)  // disabled
    RotateTo(gameObject=Self, targetAngle=270, speed=360)
    SetBoolValue(boolVariable=var "Attacking", boolValue=false, everyFrame=false)
    on ATTACK → Antic Spin
  }

  state Antic Spin {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_fat_sentry_sword (sharedassets33.assets))
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0.5, storeAngle=var "Angle To Hero", everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=())
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    FloatAdd(floatVariable=var "Angle To Hero", add=180, everyFrame=false, perSecond=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle To Hero", speed=20, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.85)
    iTweenRotateBy(gameObject=Self, id="", vector=(0, 0, 360), time=0.4, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Attacking", boolValue=true, everyFrame=false)
    on FINISHED → Antic Point
  }

  state Antic Point {
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0.5, storeAngle=var "Angle To Hero", everyFrame=true)
    RotateTo(gameObject=Self, targetAngle=var "Angle To Hero", speed=360)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shoot
  }

  state Shoot {
    AudioPlayRandom(gameObject=var "Self", audioClips=[ruin_sentry_sword_1 (sharedassets33.assets), ruin_sentry_sword_2 (sharedassets33.assets), ruin_sentry_sword_3 (sharedassets33.assets)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.1)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle To Hero", speed=45, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=10.25, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Decelerate
  }

  state Decelerate {
    DecelerateV2(gameObject=Self, deceleration=0.85)
    iTweenRotateBy(gameObject=Self, id="", vector=(0, 0, 360), time=0.4, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=false, loopDontFinish=false)
    on FINISHED → Return Pause
  }

  state Returning {
    DistanceFlySmooth(gameObject=Self, target=var "Home", distance=0, speedMax=20, accelerationForce=100, targetRadius=0.5, deceleration=0.9, offset=(0, 0, 0))
    RotateTo(gameObject=Self, targetAngle=270, speed=360)  // disabled
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    FaceAngle(gameObject=Self, angleOffset=180, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=14.8, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Home
  }

  state Return Pause {
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.8)
    on FINISHED → Returning
  }

  state Dissipate {
    StopParticleEmitter(gameObject=var "Idle Pt")
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    DecelerateV2(gameObject=Self, deceleration=0.8)
  }

  var Home Name: string = "S4 Home"

  var Angle To Hero: float = 0
  var Self Y: float = 0
  var Attacking: bool = false
  var Home: gameObject = <null>
  var Idle Pt: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
