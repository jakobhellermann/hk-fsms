# Control
# 1 location(s):
#   Room_Final_Boss_Core / Scene Darken Object

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="plates", storeResult=var "plates")
    on FINISHED → Idle
  }

  state Idle {
    on SUSPENSE FINAL → Turn Off
  }

  state Turn Off {
    ActivateGameObject(gameObject=var "plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindGameObject(objectName="Vignette", withTag="Vignette", store=var "Vignette")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SetDarkness(-1))
    SetFsmInt(gameObject=var "Vignette", fsmName="Darkness Control", variableName="Darkness Level", setValue=-1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Vignette"), sendEvent="SCENE RESET", delay=0, everyFrame=false)
  }

  var Vignette: gameObject = <null>
  var plates: gameObject = <null>
}
