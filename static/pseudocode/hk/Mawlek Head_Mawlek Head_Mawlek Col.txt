# Mawlek Head
# 1 location(s):
#   sharedassets34.assets / Mawlek Col/Mawlek Head

fsm Mawlek Head {
  start Init
  on SLEEP → Dormant  // from any state

  state Init {
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Spit Effect", storeResult=var "Spit Effect")
    on FINISHED → Dormant
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Idle")
    RandomFloat(min=0.3, max=0.6, storeResult=var "Chooser")
    Wait(time=var "Chooser", finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Active", boolValue=false, everyFrame=false)
    on FINISHED → Shoot Antic
  }

  state Shoot Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Spit")
    Wait(time=0.083, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Active", boolValue=true, everyFrame=false)
    on FINISHED → Detect Hero Pos
  }

  state Shoot {
    SendEventByName(eventTarget=GameObject(var "Spit Effect"), sendEvent="PLAY", delay=0, everyFrame=false)
    SpawnRandomObjects(gameObject=Shot Mawlek (sharedassets32.assets), spawnPoint=var "Self", position=(0, 0.5, 0), spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=var "Angle Min", angleMax=var "Angle Max", originVariation=0)  // disabled
    FlingObjectsFromGlobalPool(gameObject=Shot Mawlek NoDrip, spawnPoint=var "Self", position=(0, 0.5, 0), spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=var "Angle Min", angleMax=var "Angle Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(0, 0.5, 0), spawnMin=6, spawnMax=8, speedMin=10, speedMax=27, angleMin=var "Angle Min", angleMax=var "Angle Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    Tk2dWatchAnimationEvents(gameObject=var "Self", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[mawlek_spit (sharedassets33.assets), mawlek_spit_b], weights=[1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Detect Hero Pos {
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X", float2=var "Hero X", tolerance=0, equal=→"LEFT", lessThan=→"RIGHT", greaterThan=→"LEFT", everyFrame=false)
    on LEFT → L
    on RIGHT → R
  }

  state L {
    SetFloatValue(floatVariable=var "Angle Min", floatValue=95, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=105, everyFrame=false)
    on FINISHED → Shoot
  }

  state R {
    SetFloatValue(floatVariable=var "Angle Min", floatValue=75, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=85, everyFrame=false)
    on FINISHED → Shoot
  }

  state Dormant {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dummy Blank")
    SetBoolValue(boolVariable=var "Active", boolValue=false, everyFrame=false)
    on WAKE → Idle
  }

  var Angle Max: float = 0
  var Angle Min: float = 0
  var Chooser: float = 0
  var Hero X: float = 0
  var Self X: float = 0
  var Shot Speed: float = 27
  var Active: bool = false
  var Hero: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Spit Effect: gameObject = <null>
}
