# Flukeman Top
# 1 location(s):
#   sharedassets350.assets / Corpse Flukeman/Flukeman Top

fsm Flukeman Top {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Blood Point", storeResult=var "Blood Point")
    FindChild(gameObject=Self, childName="Pt Blood", storeResult=var "Pt Blood")
    on FINISHED → Fall
  }

  state Fall {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Top Fall", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.95)
    on FINISHED → Startle
  }

  state Startle {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    DecelerateV2(gameObject=Self, deceleration=0.8)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Top Cry", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(false))
    PlayParticleEmitter(gameObject=var "Pt Blood", emit=0)
    on FINISHED → Speed
  }

  state Chase {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Top Fly")
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=var "Speed", acceleration=0.35, targetSpread=2, spreadResetTimeMin=0.5, spreadResetTimeMax=1)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0.5)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Speed
  }

  state Speed {
    RandomFloat(min=4.5, max=5.5, storeResult=var "Speed")
    on FINISHED → Chase
  }

  var Speed: float = 0
  var Blood Point: gameObject = <null>
  var Pt Blood: gameObject = <null>
  var Self: gameObject = <null>
}
