# Big Buzzer
# 3 location(s):
#   GG_Vengefly / Giant Buzzer Col
#   GG_Vengefly_V / Giant Buzzer Col (1)
#   GG_Vengefly_V / Giant Buzzer Col

fsm Big Buzzer {
  start Init

  state Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    on FINISHED → Swoop In
    on GG BOSS → Wait Frame
  }

  state Intro Roar Antic {
    SetCollider(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    AudioStop(gameObject=Self)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar")
    Wait(time=0.332, finishEvent=→"FINISHED", realTime=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    on FINISHED → Title?
  }

  state Intro Roar Left {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=big_buzzer_scream (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(-1.49, -1.61, 0), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SpawnBloodTime(delay=0.1, spawnPoint=var "Self", position=(-1.49, -1.61, 0), spawnMin=2, spawnMax=3, speedMin=10, speedMax=18, angleMin=140, angleMax=250, colorOverride=(unset))
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Col Music
  }

  state Idle {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    DistanceFly(gameObject=Self, target=var "Hero", distance=13, speedMax=6, acceleration=var "Accel", targetsHeight=true, height=5)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn Fly", resetFrame=false, everyFrame=true)
    GetFsmBool(gameObject=var "Alert Range", fsmName="", variableName="In Alert Range", storeValue=var "In Range", everyFrame=true)
    BoolTest(boolVariable=var "In Range", isTrue=(none), isFalse=→"OUT OF RANGE", everyFrame=true)
    RandomFloat(min=0.15, max=0.22, storeResult=var "Accel")
    WaitRandom(timeMin=1.5, timeMax=2.5, finishEvent=→"ATTACK", realTime=false)
    on ATTACK → Choose Attack
  }

  state Lurk {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    ChaseObject(gameObject=Self, target=var "Marker Lurk", speedMax=6, acceleration=var "Accel", targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn Fly", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    GetFsmBool(gameObject=var "Alert Range", fsmName="", variableName="In Alert Range", storeValue=var "In Range", everyFrame=true)
    BoolTest(boolVariable=var "In Range", isTrue=→"IN RANGE", isFalse=(none), everyFrame=true)
    RandomFloat(min=0.15, max=0.22, storeResult=var "Accel")
    on IN RANGE → Idle
  }

  state Choose Attack {
    RandomFloat(min=0, max=100, storeResult=var "Chooser")
    FloatCompare(float1=var "Chooser", float2=75, tolerance=0, equal=→"SUMMON", lessThan=→"SWOOP", greaterThan=→"SUMMON", everyFrame=false)
    on SWOOP → Swoop Antic
    on SUMMON → Check Summon
  }

  state Swoop Antic {
    IntCompare(integer1=var "Swoops in A Row", integer2=3, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    IntAdd(intVariable=var "Swoops in A Row", add=1, everyFrame=false)
    SetIntValue(intVariable=var "Summons In A Row", intValue=0, everyFrame=false)
    AudioStop(gameObject=Self)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Face Right", leftBool=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Swoop Antic")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Swoop Direction
    on CANCEL → Check Summon
  }

  state Swoop L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Swoop")
    CreateObject(gameObject=Dash Burst (sharedassets32.assets), spawnPoint=var "Self", position=(-0.03, -0.19, -0.02), rotation=(0, 0, 47), storeObject=var "", networkInstantiate=false, networkGroup=0)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=-20, y=(unset), everyFrame=false)
    iTweenMoveBy(gameObject=Self, id="", vector=var "Swoop Vector", time=0.8, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Swoop Rise
  }

  state Swoop Direction {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=big_buzzer_swoop (sharedassets32.assets), pitchMin=1, pitchMax=0.9, volume=1.1, delay=0, storePlayer=<null>)
    SetBoolValue(boolVariable=var "Swooping", boolValue=true, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Swoop Target", floatValue=var "Swoop Height", everyFrame=false)
    FloatSubtract(floatVariable=var "Swoop Target", subtract=var "Self Y", everyFrame=false, perSecond=false)
    SetVector3XYZ(vector3Variable=var "Swoop Vector", vector3Value=(unset), x=0, y=var "Swoop Target", z=0, everyFrame=false)
    BoolTest(boolVariable=var "Face Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on LEFT → Swoop L
    on RIGHT → Swoop R
  }

  state Swoop R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Swoop")
    CreateObject(gameObject=Dash Burst (sharedassets32.assets), spawnPoint=var "Self", position=(-0.03, -0.19, -0.02), rotation=(0, 0, 128), storeObject=var "Dash Burst", networkInstantiate=false, networkGroup=0)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=20, y=(unset), everyFrame=false)
    iTweenMoveBy(gameObject=Self, id="", vector=var "Swoop Vector", time=0.8, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Swoop Rise
  }

  state Swoop Rise {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 3.5, 0), time=0.4, delay=0, speed=(unset), easeType=easeInOutQuad, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Swooping", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Summon Antic {
    IntOperator(integer1=var "Summons", integer2=1, operation=Subtract, storeResult=var "Summons", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar")
    Wait(time=0.332, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Dir
    on CANCEL → Swoop Antic
  }

  state Check Dir {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=big_buzzer_scream (sharedassets32.assets), pitchMin=1, pitchMax=0.85, volume=1.1, delay=0, storePlayer=<null>)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=→"RIGHT", lessThan=→"RIGHT", greaterThan=→"LEFT", everyFrame=false)
    on LEFT → Roar Left
    on RIGHT → Roar Right
  }

  state Roar Left {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SpawnBloodTime(delay=0.1, spawnPoint=var "Self", position=(-1.49, -1.61, 0), spawnMin=2, spawnMax=3, speedMin=10, speedMax=18, angleMin=140, angleMax=250, colorOverride=(unset))
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(-1.49, -1.61, 0), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Summon
  }

  state Roar Right {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SpawnBloodTime(delay=0.1, spawnPoint=var "Self", position=(1.49, -1.61, 0), spawnMin=2, spawnMax=3, speedMin=10, speedMax=18, angleMin=400, angleMax=290, colorOverride=(unset))
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(1.49, -1.61, 0), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Summon
  }

  state Summon {
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    CreateObject(gameObject=Buzzer Summon (sharedassets32.assets), spawnPoint=<null>, position=(94.63, 25.87, 17), rotation=(unset), storeObject=var "Buzzer Instance", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Buzzer Instance", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    CreateObject(gameObject=Buzzer Summon (sharedassets32.assets), spawnPoint=<null>, position=(109.85, 25.87, 17), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Roar End
    on GG BOSS → GG Summon
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Swoop In {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetCollider(gameObject=Self, active=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=16.006, space=1, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=15, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -15, -16), time=1, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"SUMMON", realTime=false, stopOnExit=true, loopDontFinish=true)
    on SUMMON → Intro Roar Antic
  }

  state Col Music {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MUSIC 4", delay=0, everyFrame=false)
    on FINISHED → Roar End
  }

  state Set GG {
    PreSpawnGameObjects(prefab=Buzzer Summon (sharedassets32.assets), storeArray=var "PreSpawned", spawnAmount=var "Summons", spawnAmountMultiplier=2)
    SetIntValue(intVariable=var "SpawnedIndex", intValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Swoop Height", floatValue=14, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetIntValue(intVariable=var "Spawned Alive", intValue=0, everyFrame=false)
    on FINISHED → Intro Roar Antic
  }

  state Title? {
    GGCheckIfBossScene(bossSceneEvent=(none), regularSceneEvent=→"FINISHED")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="VENGEFLY", everyFrame=false)
    on FINISHED → Check Dir 2
  }

  state GG Summon {
    GetNextPreSpawnedGameObject(storedArray=var "PreSpawned", spawnPosition=(38, 25, 17), storeObject=var "Buzzer Instance", currentIndex=var "SpawnedIndex")
    TrackSpawnedEnemiesAdd(Target=Self, SpawnedEnemy=var "Buzzer Instance", UsesEnemySpawner=true)
    SetScale(gameObject=var "Buzzer Instance", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    GetNextPreSpawnedGameObject(storedArray=var "PreSpawned", spawnPosition=(57, 25, 17), storeObject=var "Buzzer Instance", currentIndex=var "SpawnedIndex")
    TrackSpawnedEnemiesAdd(Target=Self, SpawnedEnemy=var "Buzzer Instance", UsesEnemySpawner=true)
    SetScale(gameObject=var "Buzzer Instance", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Roar End
  }

  state Check Dir 2 {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=→"RIGHT", lessThan=→"RIGHT", greaterThan=→"LEFT", everyFrame=false)
    on LEFT → Intro Roar Left
    on RIGHT → Intro Roar Right
  }

  state Intro Roar Right {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=big_buzzer_scream (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(1.49, -1.61, 0), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SpawnBloodTime(delay=0.1, spawnPoint=var "Self", position=(1.49, -1.61, 0), spawnMin=2, spawnMax=3, speedMin=10, speedMax=18, angleMin=400, angleMax=290, colorOverride=(unset))
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Col Music
  }

  state Check Summon {
    IntCompare(integer1=var "Summons In A Row", integer2=2, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    IntAdd(intVariable=var "Summons In A Row", add=1, everyFrame=false)
    SetIntValue(intVariable=var "Swoops in A Row", intValue=0, everyFrame=false)
    AudioStop(gameObject=Self)
    IntCompare(integer1=var "Summons", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on FINISHED → Check Summon GG
    on CANCEL → Swoop Antic
  }

  state Check Summon GG {
    GGCheckIfBossScene(bossSceneEvent=(none), regularSceneEvent=→"FINISHED")
    TrackSpawnedEnemiesGetInfo(Target=Self, TotalTracked=(unset), TotalAlive=var "Spawned Alive")
    IntCompare(integer1=var "Spawned Alive", integer2=4, equal=→"CANCEL", lessThan=→"FINISHED", greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Summon Antic
    on CANCEL → Swoop Antic
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set GG
  }

  var Summons: int = 15

  var Accel: float = 0
  var Chooser: float = 0
  var Self Y: float = 0
  var Swoop Height: float = 16.5
  var Swoop Target: float = 0
  var X Scale: float = 0
  var Spawned Alive: int = 30
  var SpawnedIndex: int = 20
  var Summons In A Row: int = 0
  var Swoops in A Row: int = 0
  var Face Right: bool = false
  var In Range: bool = false
  var Swooping: bool = false
  var Local Scale: vector3 = (0, 0, 0)
  var Roar Point: vector3 = (0, 0, 0)
  var Swoop Vector: vector3 = (0, 0, 0)
  var Alert Range: gameObject = <null>
  var Area Title: gameObject = <null>
  var Buzzer Instance: gameObject = <null>
  var Dash Burst: gameObject = <null>
  var Hero: gameObject = <null>
  var Marker Lurk: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Self: gameObject = <null>
  var PreSpawned: array = var "PreSpawned"
}
