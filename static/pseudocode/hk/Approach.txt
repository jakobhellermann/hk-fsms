# Approach
# 1 location(s):
#   Room_Ouiji / Jiji NPC

fsm Approach {
  start Idle

  state Idle {
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Land", storeResult=var "Pt Land")
    on HERO ENTER → Approach
  }

  state Approach {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Approach", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Jiji_retreat_01 (sharedassets28.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[hornet_jump (resources.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[hornet_ground_land (sharedassets28.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0.65, storePlayer=var "Voice Player")
    PlayParticleEmitter(gameObject=var "Pt Land", emit=0)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ANIM START", delay=0, everyFrame=false)
  }

  var Dream Dialogue: gameObject = <null>
  var Pt Land: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
