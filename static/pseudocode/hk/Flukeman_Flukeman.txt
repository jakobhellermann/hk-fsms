fsm Flukeman {
  start Init
  on DROWNING → Drowning  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    on ALERT → Startle
    on TOOK DAMAGE → Startle
  }

  state Startle {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Flukeman_startle_02 (sharedassets350.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    StopWalker(target=Self, everyFrame=false)
    SetAudioClip(gameObject=Self, audioClip=Flukeman_aggressive_loop (resources.assets))
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on FINISHED → Footsteps
  }

  state Chase {
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=var "Speed", acceleration=0.45, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=true)
    WaitRandom(timeMin=1, timeMax=5, finishEvent=→"WAIT", realTime=false)
    on TOOK DAMAGE → Hit Slow
    on WAIT → Set Speed
  }

  state Hit Slow {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Chase
  }

  state Set Speed {
    RandomFloat(min=11, max=13, storeResult=var "Speed")
    on FINISHED → Chase
  }

  state Drowning {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Flukeman_startle_01 (sharedassets350.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  state Footsteps {
    AudioPlaySimple(gameObject=var "Footsteps", volume=1, oneShotClip=<null>)
    on FINISHED → Set Speed
  }

  var Speed: float = 0
  var Alert Range: bool = false
  var Footsteps: gameObject = Flukeman/Footsteps
  var Self: gameObject = <null>
}
