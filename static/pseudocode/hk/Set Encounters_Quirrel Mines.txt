# Set Encounters
# 1 location(s):
#   Mines_13 / Quirrel Mines

fsm Set Encounters {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set
  }

  state Set {
    SetPlayerDataBool(boolName="quirrelMinesEncountered", value=true)
    on FINISHED → Leave City
  }

  state Leave City {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="quirrelCityEncountered", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="quirrelCityLeft", value=true)
    on FINISHED → Finished
  }

  state Finished {
  }
}
