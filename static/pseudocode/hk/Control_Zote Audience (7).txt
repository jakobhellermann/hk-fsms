# Control
# 8 location(s):
#   Dream_Mighty_Zote / Audience/Zote Audience (7)
#   Dream_Mighty_Zote / Audience/Zote Audience (3)
#   Dream_Mighty_Zote / Audience/Zote Audience (4)
#   Dream_Mighty_Zote / Audience/Zote Audience (6)
#   Dream_Mighty_Zote / Audience/Zote Audience (2)
#   Dream_Mighty_Zote / Audience/Zote Audience (5)
#   Dream_Mighty_Zote / Audience/Zote Audience (1)
#   Dream_Mighty_Zote / Audience/Zote Audience

fsm Control {
  start Init
  on CHANGE → Choice  // from any state
  on ZOTE STUNNED → Stun Shock  // from any state
  on ZOTE KILLED → Zote Killed  // from any state
  on ZOTE EXPLODE → Explode  // from any state

  state Init {
    FindGameObject(objectName="", withTag="Extra Tag", store=var "Grey Prince")
    RandomFloatEither(value1=-1, value2=1, storeResult=var "X Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Anim Pause
  }

  state Pre Fight {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zote Idle")
    on FIGHT START → Choice
  }

  state Anim Pause {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Look Anim
  }

  state Choice {
    SendRandomEvent(events=[→"WATCH HERO", →"WATCH ZOTE", →"IDLE"], weights=[1, 1, 1], delay=0)
    on IDLE → Idle
    on WATCH HERO → Target Hero
    on WATCH ZOTE → Target Zote
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zote Idle")
    WaitRandom(timeMin=4, timeMax=8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choice
  }

  state Target L {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Look L")
    FloatAdd(floatVariable=(unset), add=0, everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=var "Time", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CHANGE", everyFrame=true)
    on R → Pause R
  }

  state Watch Target {
    SetFloatValue(floatVariable=var "Timer", floatValue=0, everyFrame=false)
    RandomFloat(min=4, max=10, storeResult=var "Time")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on L → Target L
    on R → 
    on FINISHED → Target L
  }

  state Target R {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Look R")
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=var "Time", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CHANGE", everyFrame=true)
    on L → Pause L
  }

  state Pause R {
    WaitRandom(timeMin=0.2, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=var "Time", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CHANGE", everyFrame=true)
    on FINISHED → Target R
  }

  state Pause L {
    WaitRandom(timeMin=0.2, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=var "Time", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CHANGE", everyFrame=true)
    on FINISHED → Target L
  }

  state Target Hero {
    SetGameObject(variable=var "Target", gameObject=var "Hero", everyFrame=false)
    on FINISHED → Watch Target
  }

  state Target Zote {
    SetGameObject(variable=var "Target", gameObject=var "Hero", everyFrame=false)
    on FINISHED → Watch Target
  }

  state Stun Shock {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zote Shock")
    WaitRandom(timeMin=1.5, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    RandomFloatEither(value1=-1, value2=1, storeResult=var "X Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Choice
  }

  state Set Look Anim {
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)
    SetStringValue(stringVariable=var "Look L", stringValue="Zote R", everyFrame=false)
    SetStringValue(stringVariable=var "Look R", stringValue="Zote L", everyFrame=false)
    on FINISHED → Pre Fight
  }

  state Zote Killed {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zote Shock")
    RandomFloatEither(value1=-1, value2=1, storeResult=var "X Scale")
  }

  state Explode {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Zote Collapse")
  }

  var Time: float = 0
  var Timer: float = 0
  var X Scale: float = 0
  var Look L: string = "Zote L"
  var Look R: string = "Zote R"
  var Grey Prince: gameObject = <null>
  var Target: gameObject = <null>
}
