# Control
# 1 location(s):
#   Knight_Pickup / Knight/Charm Effects/Blocker Shield

fsm Control {
  start Pause
  on CHARM EQUIP CHECK → Init  // from any state
  on HERO REVIVED → Init  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Shell Anim", storeResult=var "Shell Anim")
    FindChild(gameObject=Self, childName="Pusher", storeResult=var "Pusher")
    FindChild(gameObject=Self, childName="Hit Crack", storeResult=var "Crack")
    GetParent(gameObject=Self, storeResult=var "Charm Effects")
    GetParent(gameObject=var "Charm Effects", storeResult=var "Knight")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check Equipped
  }

  state Unequipped {
    SetMeshRenderer(gameObject=var "Blocker HUD", active=false)
    on CHARM EQUIP CHECK → Check Equipped
  }

  state Check Equipped {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_5", isTrue=→"EQUIPPED", isFalse=→"UNEQUIPPED")
    on EQUIPPED → Icon Pause
    on UNEQUIPPED → Unequip Pause
  }

  state Equipped {
    on FOCUS START → Hits Left?
  }

  state Focusing {
    on FOCUS END → Focus End
    on BLOCKER HIT → Blocker Hit
    on FSM CANCEL → Shell Off
  }

  state Focus End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Shell Anim", clipName="Disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=blocker_open (resources.assets), pitchMin=1.15, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Shell Off
  }

  state Shell Off {
    SetMeshRenderer(gameObject=var "Shell Anim", active=false)
    on FINISHED → Equipped
  }

  state Blocker Hit {
    PlayerDataIntAdd(gameObject=var "GameManager", intName="blockerHits", amount=var "Decrement")
    GetPlayerDataInt(gameObject=var "GameManager", intName="blockerHits", storeValue=var "Blocks")
    IntCompare(integer1=var "Blocks", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Shell Anim", animLibName="", clipName="Impact")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Pusher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    IntSwitch(intVariable=var "Blocks", compareTo=[1, 2, 3], sendEvent=[(none), (none), (none)], everyFrame=false)
    on 1 → HUD 1
    on 2 → HUD 2
    on 3 → HUD 3
    on BREAK → Break
  }

  state Shell Up {
    Tk2dPlayAnimation(gameObject=var "Shell Anim", animLibName="", clipName="Appear")
    SetMeshRenderer(gameObject=var "Shell Anim", active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=blocker_close (resources.assets), pitchMin=1.15, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Focusing
    on FSM CANCEL → Shell Off
  }

  state Hits Left? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="blockerHits", storeValue=var "Blocks")
    IntCompare(integer1=var "Blocks", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Equipped
    on FINISHED → Shell Up
  }

  state Break {
    Tk2dPlayAnimationWithEvents(gameObject=var "Shell Anim", clipName="Break", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Pusher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Crack", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.83, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Blocker HUD", animLibName="", clipName="UI Break 4")
    on FINISHED → Bits
  }

  state Bits {
    Tk2dWatchAnimationEvents(gameObject=var "Shell Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlingObjectsFromGlobalPool(gameObject=Blocker Bit 1 (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=2, speedMin=15, speedMax=22, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Blocker Bit 2 (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=2, speedMin=15, speedMax=22, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Blocker Bit 3 (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=2, speedMin=20, speedMax=25, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Blocker Bit 4 (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=2, speedMin=15, speedMax=22, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Blocker Bit 5 (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=2, speedMin=20, speedMax=25, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    on FINISHED → Shell Off
  }

  state HUD Icon Up {
    SetMeshRenderer(gameObject=var "Blocker HUD", active=true)
    Tk2dPlayAnimation(gameObject=var "Blocker HUD", animLibName="", clipName="UI Appear")
    on FINISHED → Equipped
  }

  state HUD 1 {
    Tk2dPlayAnimation(gameObject=var "Blocker HUD", animLibName="", clipName="UI Break 3")
    on FINISHED → Impact End
  }

  state HUD 2 {
    Tk2dPlayAnimation(gameObject=var "Blocker HUD", animLibName="", clipName="UI Break 2")
    on FINISHED → Impact End
  }

  state HUD 3 {
    Tk2dPlayAnimation(gameObject=var "Blocker HUD", animLibName="", clipName="UI Break 1")
    on FINISHED → Impact End
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  state Unequip Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Unequipped
  }

  state Impact End {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Tk2dWatchAnimationEvents(gameObject=var "Shell Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Crack", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Shell Off
  }

  state Icon Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → HUD Icon Up
  }

  var Blocks: int = 0
  var Decrement: int = -1
  var Charm Effects: gameObject = <null>
  var Crack: gameObject = <null>
  var Knight: gameObject = <null>
  var Pusher: gameObject = <null>
  var Self: gameObject = <null>
  var Shell Anim: gameObject = <null>
}
