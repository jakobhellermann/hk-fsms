# Update Sprite
# 1 location(s):
#   resources.assets / _GameCameras/HudCamera/Inventory/Charms/Details/Detail Sprite

fsm Update Sprite {
  start Init
  on UPDATE → To Update  // from any state

  state Init {
    on FINISHED → Update
  }

  state Update {
    IntCompare(integer1=var "ID", integer2=0, equal=→"NO CHARM", lessThan=→"NO CHARM", greaterThan=(none), everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=true)
    CallMethodProper(gameObject=var "Charm Icons", behaviour="CharmIconList", methodName="GetSprite", parameters=[var "ID"], storeResult=var "Sprite")
    SetSpriteRendererSprite(gameObject=Self, sprite=var "Sprite")
    on NO CHARM → No Charm
    on FINISHED → Glass HP
  }

  state No Charm {
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state To Update {
    on FINISHED → Update
  }

  state Glass HP {
    IntCompare(integer1=var "ID", integer2=23, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brokenCharm_23", isTrue=(none), isFalse=→"FINISHED")
    SetSpriteRendererSprite(gameObject=Self, sprite=_0002_charm_glass_heal_broken)
    on FINISHED → Glass Geo
  }

  state Glass Geo {
    IntCompare(integer1=var "ID", integer2=24, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brokenCharm_24", isTrue=(none), isFalse=→"FINISHED")
    SetSpriteRendererSprite(gameObject=Self, sprite=_0003_charm_glass_geo_broken)
    on FINISHED → Glass Attack
  }

  state Glass Attack {
    IntCompare(integer1=var "ID", integer2=25, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="brokenCharm_25", isTrue=(none), isFalse=→"FINISHED")
    SetSpriteRendererSprite(gameObject=Self, sprite=_0002_charm_glass_attack_up_broken)
    on FINISHED → Royal Charm
  }

  state Updated {
  }

  state Royal Charm {
    IntCompare(integer1=var "ID", integer2=36, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    IntSwitch(intVariable=var "Royal Charm State", compareTo=[1, 2, 3, 4], sendEvent=[→"R QUEEN", →"R KING", →"R FINAL", →"R SHADE"], everyFrame=false)
    on FINISHED → Updated
    on R QUEEN → R Queen
    on R KING → R King
    on R FINAL → R Final
    on R SHADE → R Shade
  }

  state R Queen {
    SetSpriteRendererSprite(gameObject=Self, sprite=charm_white_left)
    on FINISHED → Updated
  }

  state R King {
    SetSpriteRendererSprite(gameObject=Self, sprite=charm_white_right)
    on FINISHED → Updated
  }

  state R Final {
    SetSpriteRendererSprite(gameObject=Self, sprite=charm_white_full)
    on FINISHED → Updated
  }

  state R Shade {
    SetSpriteRendererSprite(gameObject=Self, sprite=charm_black)
    on FINISHED → Updated
  }

  var ID: int = 0
  var Royal Charm State: int = 0
  var Charm Sprite: gameObject = <null>
  var Sprite: object = <null>
}
