fsm Fader {
  start Init

  state Init {
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    on SELECTED → Up
    on UNSELECTED → Down
  }

  state Up {
    SetSpriteRenderer(gameObject=Self, active=true)
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))
    EaseColor(fromValue=var "Prev Colour", toValue=(1, 1, 1, 1), colorVariable=var "New Colour", time=0.2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    on UNSELECTED → Down
  }

  state Down {
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", fail=(none))
    EaseColor(fromValue=var "Prev Colour", toValue=(1, 1, 1, 0), colorVariable=var "New Colour", time=0.2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    on SELECTED → Up
  }

  var New Colour: color = (0, 0, 0, 1)
  var Prev Colour: color = (1, 1, 1, 1)
}
