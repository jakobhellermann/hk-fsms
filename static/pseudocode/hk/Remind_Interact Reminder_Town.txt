# Remind
# 1 location(s):
#   Town / Interact Reminder

fsm Remind {
  start Pause
  on BIG TITLE START → Inert  // from any state
  on HIDE INTERACT REMINDER → Activate  // from any state

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ENTER → State 1
    on ACTIVATE → Activate
  }

  state State 1 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REMINDER INTERACT", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  state Pause {
    GetPlayerDataBool(boolName="metElderbug", storeValue=var "Visited Dirtmouth")
    BoolTest(boolVariable=var "Visited Dirtmouth", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on ACTIVATE → Set Active
  }

  state Activate {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  state Set Active {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Inert {
    on BIG TITLE END → Wait
  }

  state Wait {
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Activated: bool = false
  var Visited Dirtmouth: bool = false
}
