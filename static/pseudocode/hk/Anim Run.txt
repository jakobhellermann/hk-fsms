# Anim Run
# 1 location(s):
#   sharedassets350.assets / Corpse Flukeman/Flukeman Bot

fsm Anim Run {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Climb Sprite", storeResult=var "Climb Sprite")
    SetMeshRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Climb Sprite", active=false)
    BoolTest(boolVariable=var "Anim", isTrue=→"START", isFalse=(none), everyFrame=true)
    on START → Check
  }

  state Check {
    SetAudioClip(gameObject=Self, audioClip=flukeman_run_loop)  // disabled
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Vel", space=0, everyFrame=true)
    SetMeshRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Climb Sprite", active=false)
    FloatCompare(float1=var "Y Vel", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UP", everyFrame=true)
    on UP → Climbing
  }

  state Climbing {
    SetAudioClip(gameObject=Self, audioClip=flukeman_crawling_half_rise)  // disabled
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Climb Sprite", active=true)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Vel", space=0, everyFrame=true)
    FloatCompare(float1=var "Y Vel", float2=1, tolerance=0, equal=(none), lessThan=→"STOP", greaterThan=(none), everyFrame=true)
    on STOP → Check
  }

  var Y Vel: float = 0
  var Anim: bool = false
  var Climb Sprite: gameObject = <null>
}
