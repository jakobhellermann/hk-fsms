# Attack
# 1 location(s):
#   resources.assets / Grimmball/Enemy Damager

fsm Attack {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="Enemies", sendEvent=→"ATTACK", storeCollider=var "Enemy")
    on ATTACK → Invincible?
  }

  state Hit {
    SendMessage(gameObject=var "Enemy", delivery=SendMessage, options=1, functionCall=FlashGrimmHit())
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClip=enemy_damage_over_time, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Enemy"), sendEvent="TOOK DAMAGE", delay=0, everyFrame=false)
    GetHP(target=var "Enemy", storeValue=var "Enemy HP")
    IntOperator(integer1=var "Enemy HP", integer2=var "Damage", operation=Subtract, storeResult=var "Enemy HP", everyFrame=false)
    SetHP(target=var "Enemy", hp=var "Enemy HP")
    IntCompare(integer1=var "Enemy HP", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    InstaDeath(target=var "Enemy", direction=0)
    on FINISHED → Parent?
  }

  state Invincible? {
    SendEventByName(eventTarget=GameObject(var "Enemy"), sendEvent="GRIMMBALL", delay=0, everyFrame=false)
    GetTag(gameObject=var "Enemy", storeResult=var "Tag", everyFrame=false)
    StringCompare(stringVariable=var "Tag", compareTo="Spell Vulnerable", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    CheckInvincibility(target=var "Enemy", storeValue=var "Invincible")
    BoolTest(boolVariable=var "Invincible", isTrue=→"INVINCIBLE", isFalse=(none), everyFrame=false)
    on INVINCIBLE → Tink
    on FINISHED → Hit
  }

  state Parent? {
    GetParent(gameObject=var "Enemy", storeResult=var "Enemy Parent")
    GameObjectIsNull(gameObject=var "Enemy Parent", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemy Parent"), sendEvent="TOOK DAMAGE", delay=0, everyFrame=false)
    GetHP(target=var "Enemy Parent", storeValue=var "Enemy HP")
    IntOperator(integer1=var "Enemy HP", integer2=var "Damage", operation=Subtract, storeResult=var "Enemy HP", everyFrame=false)
    SetHP(target=var "Enemy Parent", hp=var "Enemy HP")
    SendMessage(gameObject=var "Enemy Parent", delivery=SendMessage, options=1, functionCall=FlashGrimmHit())
    IntCompare(integer1=var "Enemy HP", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    InstaDeath(target=var "Enemy Parent", direction=0)
    on FINISHED → G Parent?
  }

  state G Parent? {
    GetParent(gameObject=var "Enemy Parent", storeResult=var "Enemy Grandparent")
    GameObjectIsNull(gameObject=var "Enemy Grandparent", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemy Grandparent"), sendEvent="TOOK DAMAGE", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Enemy Grandparent", delivery=SendMessage, options=1, functionCall=FlashGrimmHit())
    GetHP(target=var "Enemy Grandparent", storeValue=var "Enemy HP")
    IntOperator(integer1=var "Enemy HP", integer2=var "Damage", operation=Subtract, storeResult=var "Enemy HP", everyFrame=false)
    SetHP(target=var "Enemy Grandparent", hp=var "Enemy HP")
    IntCompare(integer1=var "Enemy HP", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    InstaDeath(target=var "Enemy Grandparent", direction=0)
    on FINISHED → Send Impact
  }

  state Send Impact {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="IMPACT", delay=0, everyFrame=false)
  }

  state Tink {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClip=sword_hit_reject, pitchMin=1.2, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Send Impact
  }

  var Attack Timer: float = 0
  var Damage: int = 5
  var Enemy HP: int = 0
  var Invincible: bool = false
  var Tag: string = ""
  var Enemy: gameObject = <null>
  var Enemy Grandparent: gameObject = <null>
  var Enemy Parent: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
