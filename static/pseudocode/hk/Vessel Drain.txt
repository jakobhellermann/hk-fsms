# Vessel Drain

#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels

fsm Vessel Drain {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    SetParticleEmission(gameObject=var "Particle 1", emission=false)
    SetParticleEmission(gameObject=var "Particle 2", emission=false)
    SetParticleEmission(gameObject=var "Particle 3", emission=false)
    SetParticleEmission(gameObject=var "Particle 4", emission=false)
    SetParticleEmission(gameObject=var "Particle Fill", emission=false)
    on MP DRAIN → Drain Pause
    on SOUL LIMITER DOWN → Drain Pause
  }

  state Drain Pause {
    SetParticleEmission(gameObject=var "Particle 1", emission=false)
    SetParticleEmission(gameObject=var "Particle 2", emission=false)
    SetParticleEmission(gameObject=var "Particle 3", emission=false)
    SetParticleEmission(gameObject=var "Particle 4", emission=false)
    SetParticleEmission(gameObject=var "Particle Fill", emission=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on MP DRAIN → Reset Pause
    on FINISHED → Particle Check
    on SOUL LIMITER DOWN → Reset Pause
  }

  state Reset Pause {
    on FINISHED → Drain Pause
  }

  state Drain {
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserve", storeValue=var "MP Reserve")
    GetPlayerDataInt(gameObject=var "GameManager", intName="maxMP", storeValue=var "Max MP")
    IntCompare(integer1=var "MP Reserve", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPCharge", storeValue=var "MP")
    IntCompare(integer1=var "MP", integer2=var "Max MP", equal=→"CANCEL", lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddMPChargeSpa(1))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=TakeReserveMP(1))
    Wait(time=0.005, finishEvent=→"FINISHED", realTime=false)
    on MP DRAIN → Drain Pause
    on CANCEL → Idle
    on FINISHED → Drain Recheck
    on SOUL LIMITER DOWN → Drain Pause
  }

  state Drain Recheck {
    on FINISHED → Particle Check
  }

  state Particle Check {
    SetParticleEmission(gameObject=var "Particle Fill", emission=true)
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPReserve", storeValue=var "MP Reserve")
    IntCompare(integer1=var "MP Reserve", integer2=34, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "MP Reserve", integer2=67, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "MP Reserve", integer2=100, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "MP Reserve", integer2=133, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on 1 → 1
    on 2 → 2
    on 3 → 3
    on 4 → 4
  }

  state 1 {
    SetParticleEmission(gameObject=var "Particle 1", emission=true)
    SetParticleEmission(gameObject=var "Particle 2", emission=false)
    SetParticleEmission(gameObject=var "Particle 3", emission=false)
    SetParticleEmission(gameObject=var "Particle 4", emission=false)
    on FINISHED → Drain
  }

  state 2 {
    SetParticleEmission(gameObject=var "Particle 2", emission=true)
    SetParticleEmission(gameObject=var "Particle 1", emission=false)
    SetParticleEmission(gameObject=var "Particle 3", emission=false)
    SetParticleEmission(gameObject=var "Particle 4", emission=false)
    on FINISHED → Drain
  }

  state 3 {
    SetParticleEmission(gameObject=var "Particle 3", emission=true)
    SetParticleEmission(gameObject=var "Particle 2", emission=false)
    SetParticleEmission(gameObject=var "Particle 1", emission=false)
    SetParticleEmission(gameObject=var "Particle 4", emission=false)
    on FINISHED → Drain
  }

  state 4 {
    SetParticleEmission(gameObject=var "Particle 4", emission=true)
    SetParticleEmission(gameObject=var "Particle 2", emission=false)
    SetParticleEmission(gameObject=var "Particle 3", emission=false)
    SetParticleEmission(gameObject=var "Particle 1", emission=false)
    on FINISHED → Drain
  }

  var Particle 1: gameObject = _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels/Particle 1
  var Particle 2: gameObject = _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels/Particle 2
  var Particle 3: gameObject = _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels/Particle 3
  var Particle 4: gameObject = _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels/Particle 4
  var Particle Fill: gameObject = _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Soul Orb/Vessels/Particle Fill

  var MP: int = 0
  var MP Reserve: int = 0
  var Max MP: int = 0
}
