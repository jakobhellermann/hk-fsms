// uses template: npc_control
fsm npc_control {
  start Start Pause
  on BIG TITLE START → Inert  // from any state
  on NPC CONTROL OFF → Off  // from any state
  on CONVO CANCEL → Cancel Pause  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetBoolValue(boolVariable=var "Can Talk", boolValue=true, everyFrame=false)
    FindChild(gameObject=var "Self", childName="Prompt Marker", storeResult=var "Prompt Marker")
    SetSpriteRenderer(gameObject=var "Prompt Marker", active=false)
    FindChild(gameObject=var "Self", childName="Ranges", storeResult=var "Ranges Folder")
    SetGameObject(variable=var "Hero Obj", gameObject=var "Hero", everyFrame=false)
    SetBoolValue(boolVariable=var "Facing Right", boolValue=var "Sprite Faces Right", everyFrame=false)
    AddTrackTrigger(target=Self, skipIfPresent=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Idle {
    HidePromptMarker(storedObject=var "Prompt")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"IN RANGE", storeCollider=var "")
    on IN RANGE → Can Talk Bool?
  }

  state In Range {
    ShowPromptMarker(prefab=Arrow Prompt New (resources.assets), labelName=var "Prompt Name", spawnPoint=var "Prompt Marker", storeObject=var "Prompt")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"OUT OF RANGE", storeCollider=var "")
    ListenForDown(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForUp(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    CheckTrackTriggerCount(target=Self, count=0, test=LessThanOrEqual, everyFrame=true, successEvent=→"OUT OF RANGE")
    on OUT OF RANGE → Idle
    on UP PRESSED → Can Talk?
    on DREAM DIALOGUE START → Dream Dialogue
  }

  state Can Talk? {
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="CanTalk", parameters=[], storeResult=var "Talk Allowed")
    BoolTest(boolVariable=var "Talk Allowed", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Take Control
  }

  state Cancel Frame {
    HidePromptMarker(storedObject=var "Prompt")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Convo End {
    ActivateGameObject(gameObject=var "Ranges Folder", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Talking", boolValue=false, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    on FINISHED → Regain Control
  }

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Check Proximity {
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero Obj", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Left Prox", floatValue=var "Self X", everyFrame=false)
    FloatSubtract(floatVariable=var "Left Prox", subtract=var "Move To Offset", everyFrame=false, perSecond=false)
    SetFloatValue(floatVariable=var "Right Prox", floatValue=var "Self X", everyFrame=false)
    FloatAdd(floatVariable=var "Right Prox", add=var "Move To Offset", everyFrame=false, perSecond=false)
    FloatTestToBool(float1=var "Hero X", float2=var "Self X", tolerance=0, equalBool=(unset), lessThanBool=var "Hero On Left", greaterThanBool=var "Hero On Right", everyFrame=false)
    BoolAllTrue(boolVariables=[var "Hero Always Left", var "Hero On Right"], sendEvent=→"MOVE LEFT", storeResult=(unset), everyFrame=false)
    BoolAllTrue(boolVariables=[var "Hero Always Right", var "Hero On Left"], sendEvent=→"MOVE RIGHT", storeResult=(unset), everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Self X", upperValue=var "Right Prox", boolVariable=(unset), trueEvent=→"MOVE RIGHT", falseEvent=(none), everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Left Prox", upperValue=var "Self X", boolVariable=(unset), trueEvent=→"MOVE LEFT", falseEvent=(none), everyFrame=false)
    GetScale(gameObject=var "Hero Obj", vector=(unset), xScale=var "Hero X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Hero X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Hero Facing Right", greaterThanBool=var "Hero Facing Left", everyFrame=false)
    BoolAllTrue(boolVariables=[var "Hero Facing Right", var "Hero On Right"], sendEvent=→"TURN LEFT", storeResult=(unset), everyFrame=false)
    BoolAllTrue(boolVariables=[var "Hero Facing Left", var "Hero On Left"], sendEvent=→"TURN RIGHT", storeResult=(unset), everyFrame=false)
    on MOVE RIGHT → Move Hero Right
    on FINISHED → Convo Start
    on TURN LEFT → Turn Hero Left
    on MOVE LEFT → Move Hero Left
    on TURN RIGHT → Turn Hero Right
  }

  state Move Hero Right {
    ForceHeroFootstepSound(forceFootstepSound=Run, endOnStateExit=true)
    SetScale(gameObject=var "Hero Obj", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Hero Obj", animLibName="", clipName="Walk")
    SetVelocity2d(gameObject=var "Hero Obj", vector=(unset), x=6, y=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero Obj", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Right Prox", tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Turn Hero Left
  }

  state Take Control {
    SetBoolValue(boolVariable=var "Talking", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Ranges Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var))
    HidePromptMarker(storedObject=var "Prompt")
    on FINISHED → Turn Self?
  }

  state Turn Hero Left {
    Tk2dPlayAnimation(gameObject=var "Hero Obj", animLibName="", clipName="Turn")
    SetVelocity2d(gameObject=var "Hero Obj", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetScale(gameObject=var "Hero Obj", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="FaceLeft", parameters=[], storeResult=(unset var))
    Tk2dWatchAnimationEvents(gameObject=var "Hero Obj", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Convo Start
  }

  state Move Hero Left {
    ForceHeroFootstepSound(forceFootstepSound=Run, endOnStateExit=true)
    SetScale(gameObject=var "Hero Obj", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Hero Obj", animLibName="", clipName="Walk")
    SetVelocity2d(gameObject=var "Hero Obj", vector=(unset), x=-6, y=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero Obj", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=var "Left Prox", tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Turn Hero Right
  }

  state Turn Hero Right {
    Tk2dPlayAnimation(gameObject=var "Hero Obj", animLibName="", clipName="Turn")
    SetVelocity2d(gameObject=var "Hero Obj", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetScale(gameObject=var "Hero Obj", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="FaceRight", parameters=[], storeResult=(unset var))
    Tk2dWatchAnimationEvents(gameObject=var "Hero Obj", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Convo Start
  }

  state Convo Start {
    Tk2dPlayAnimation(gameObject=var "Hero Obj", animLibName="", clipName="Idle")
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="CONVO START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC CONVO START", delay=0, everyFrame=false)
    on CONVO END → Convo End
    on HERO DAMAGED → Reset Convo
  }

  state Turn Self R {
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName=var "Turn Anim Name")
    Tk2dPlayFrame(gameObject=var "Self", frame=0)
    SetScale(gameObject=var "Self", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Turned", boolValue=true, everyFrame=false)
    on FINISHED → Check Proximity
  }

  state Turn Self L {
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName=var "Turn Anim Name")
    SetScale(gameObject=var "Self", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Turned", boolValue=true, everyFrame=false)
    on FINISHED → Check Proximity
  }

  state Turn Back {
    SetBoolValue(boolVariable=var "Turned", boolValue=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName=var "Turn Anim Name")
    on FINISHED → Pause
  }

  state Turn Self? {
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero Obj", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Hero X", float2=var "Self X", tolerance=0, equalBool=(unset), lessThanBool=var "Hero On Left", greaterThanBool=var "Hero On Right", everyFrame=false)
    BoolTestMulti(boolVariables=[var "Sprite Faces Right", var "Hero On Right", var "Turns To Speak"], boolStates=[false, true, true], trueEvent=→"TURN RIGHT", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolAllTrue(boolVariables=[var "Hero On Left", var "Sprite Faces Right", var "Turns To Speak"], sendEvent=→"TURN LEFT", storeResult=(unset), everyFrame=false)
    on TURN RIGHT → Turn Self R
    on FINISHED → Check Proximity
    on TURN LEFT → Turn Self L
  }

  state In Range Turns {
    ShowPromptMarker(prefab=Arrow Prompt New (resources.assets), labelName=var "Prompt Name", spawnPoint=var "Prompt Marker", storeObject=var "Prompt")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"OUT OF RANGE", storeCollider=var "")
    ListenForDown(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForUp(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    FaceObject(objectA=var "Self", objectB=var "Hero Obj", spriteFacesRight=false, playNewAnimation=true, newAnimationClip=var "Turn Anim Name", resetFrame=true, everyFrame=true)
    on OUT OF RANGE → Idle
    on UP PRESSED → Can Talk?
    on DREAM DIALOGUE START → Dream Dialogue
  }

  state Turns? {
    BoolTest(boolVariable=var "Always Faces Hero", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → In Range Turns
    on FALSE → In Range
  }

  state Inert {
    on BIG TITLE END → Idle
  }

  state Can Talk Bool? {
    BoolTest(boolVariable=var "Can Talk", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Turns?
    on FALSE → Idle
  }

  state Dream Dialogue {
    HidePromptMarker(storedObject=var "Prompt")
    on DREAM DIALOGUE END → Idle
  }

  state Off {
    HidePromptMarker(storedObject=var "Prompt")
  }

  state Start Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Cancel Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Reset Convo {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="RESET CONVO", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → Convo End
  }

  state Regain Control {
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    BoolTest(boolVariable=var "Turned", isTrue=→"TURN BACK", isFalse=(none), everyFrame=false)
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="PreventCastByDialogueEnd", parameters=[], storeResult=(unset var))
    on FINISHED → Pause
    on TURN BACK → Turn Back
  }

  var Move To Offset: float = 2
  var Always Faces Hero: bool = false
  var Hero Always Left: bool = false
  var Hero Always Right: bool = false
  var Sprite Faces Right: bool = false
  var Turns To Speak: bool = false
  var Prompt Name: string = "Listen"
  var Turn Anim Name: string = ""

  var Hero X: float = 0
  var Hero X Scale: float = 0
  var Left Prox: float = 0
  var Right Prox: float = 0
  var Self X: float = 0
  var Can Talk: bool = true
  var Facing Right: bool = false
  var Hero Facing Left: bool = false
  var Hero Facing Right: bool = false
  var Hero On Left: bool = false
  var Hero On Right: bool = false
  var Talk Allowed: bool = false
  var Talking: bool = false
  var Turned: bool = false
  var Hero Obj: gameObject = <null>
  var Prompt: gameObject = <null>
  var Prompt Marker: gameObject = <null>
  var Ranges Folder: gameObject = <null>
  var Self: gameObject = <null>
}
