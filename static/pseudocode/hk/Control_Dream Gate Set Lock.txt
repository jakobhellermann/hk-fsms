fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Gate Allowed", setValue=false, everyFrame=false)
    on EXIT → Out
  }

  state Out {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Gate Allowed", setValue=true, everyFrame=false)
    on ENTER → In
  }
}
