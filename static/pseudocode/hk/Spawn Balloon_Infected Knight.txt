# Spawn Balloon
# 2 location(s):
#   Abyss_19 / Infected Knight
#   GG_Broken_Vessel / Infected Knight

fsm Spawn Balloon {
  start Inert

  state Inert {
    on START SPAWN → Spawn Pause
  }

  state Spawn Pause {
    WaitRandom(timeMin=var "Wait Min", timeMax=var "Wait Max", finishEvent=→"SPAWN", realTime=false)
    on SPAWN → Spawn
    on STOP SPAWN → Stop
  }

  state Spawn {
    GetTagCount(tag="Extra Tag", storeResult=var "Enemy Count")
    IntCompare(integer1=var "Enemy Count", integer2=3, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=420, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    RandomFloat(min=var "X Min", max=var "X Max", storeResult=var "X Pos")
    RandomFloat(min=var "Y Min", max=var "Y Max", storeResult=var "Y Pos")
    RandomFloat(min=0.006001, max=0.008, storeResult=var "Z Pos")
    SetVector3XYZ(vector3Variable=var "Spawn Vector", vector3Value=(unset), x=var "X Pos", y=var "Y Pos", z=var "Z Pos", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Parasite Balloon Spawner (sharedassets343.assets), spawnPoint=<null>, position=var "Spawn Vector", rotation=(unset), storeObject=var "Spawned Enemy")
    EnemyPusherIgnore(target=Self, other=var "Spawned Enemy")
    on CANCEL → Spawn Pause
    on FINISHED → Spawn Pause
    on STOP SPAWN → Stop
  }

  state Stop {
    on START SPAWN → Spawn Pause
  }

  var Wait Max: float = 5
  var Wait Min: float = 4
  var X Max: float = 36.46
  var X Min: float = 17.36
  var X Pos: float = 0
  var Y Max: float = 37.42
  var Y Min: float = 32.16
  var Y Pos: float = 0
  var Z Pos: float = 0
  var Enemy Count: int = 0
  var HP: int = 0
  var Spawn Vector: vector3 = (0, 0, 0)
  var Spawned Enemy: gameObject = <null>
}
