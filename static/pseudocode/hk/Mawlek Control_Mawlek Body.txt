fsm Mawlek Control {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "Camera", gameObject=var "CameraParent", everyFrame=false)
    GetHero(storeResult=var "Hero")
    GetChild(gameObject=Self, childName="Dummy", withTag="Untagged", storeResult=var "Dummy")
    GetChild(gameObject=Self, childName="Mawlek Arm L", withTag="Untagged", storeResult=var "Arm L")

    FindChild(gameObject=Self, childName="Spit Effect", storeResult=var "Spit Effect")

    GetChild(gameObject=Self, childName="Mawlek Head", withTag="Untagged", storeResult=var "Head")

    StopWalker(target=Self, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Dormant
  }

  state Dormant {

  }

  state Wake {

    Wait(time=0.166, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake Jump
  }

  state Wake Jump {

    on FINISHED → Super Select
  }

  state Super Select {
    SetCollider(gameObject=var "Head", active=false)
    SendRandomEvent(events=[→"TRUE", →"FALSE"], weights=[1, 1], delay=0)

    on FALSE → Super Jump

    SetMeshRenderer(gameObject=Self, active=true)
