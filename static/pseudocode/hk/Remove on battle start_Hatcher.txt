# Remove on battle start
# 1 location(s):
#   Crossroads_22 / Hatcher

fsm Remove on battle start {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    on BG CLOSE → Die
  }

  state Die {
    InstaDeath(target=Self, direction=0)  // disabled
    SetGeoDrop(target=Self, smallGeo=0, mediumGeo=0, largeGeo=0)  // disabled
    FindChild(gameObject=Self, childName="Corpse Flukeman Bot", storeResult=var "Corpse Flukeman Bot")
    ActivateGameObject(gameObject=var "Corpse Flukeman Bot", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DestroyObject(gameObject=var "Self", delay=0, detachChildren=true)
  }

  var Corpse Flukeman Bot: gameObject = <null>
  var Self: gameObject = <null>
}
