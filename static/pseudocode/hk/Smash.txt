// uses template: hive_smash_platform
fsm Smash {
  start Pause

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Particles", storeResult=var "Particles")
    ActivateGameObject(gameObject=var "Particles", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Smash Checker", storeResult=var "Smash Checker")
    FindChild(gameObject=Self, childName="Roof Collider", storeResult=var "Roof Collider")
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Idle {
    on HIT → Smash
  }

  state Smash {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=honey_pillar_final_break (sharedassets384.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=honey_pillar_hit_1 (sharedassets384.assets))
    SetPolygonCollider(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=var "Smash Checker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Roof Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(unset), spawnMin=40, spawnMax=40, speedMin=10, speedMax=20, angleMin=0, angleMax=360, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    ActivateGameObject(gameObject=var "Particles", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Hit Objects: gameObject = <null>
  var Particles: gameObject = <null>
  var Roof Collider: gameObject = <null>
  var Self: gameObject = <null>
  var Smash Checker: gameObject = <null>
}
