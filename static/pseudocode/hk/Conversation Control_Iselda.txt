# Conversation Control
# 1 location(s):
#   Room_mapper / Iselda

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → NPC Anim
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ISELDA", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Meet {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=Iselda_Shop_talk_01 (sharedassets12.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ISELDA_MEET", "Iselda"], storeResult=(unset var))
    SetPlayerDataBool(boolName="metIselda", value=true)
    on CONVO_FINISH → To Shop
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk Convo End")
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state NPC Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk Convo Start")
    on FINISHED → Box Up
  }

  state Convo Choice {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="corniferAtHome", falseBool="iseldaCorniferHomeConvo", isTrue=→"CORNIFER HOME", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metIselda", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="troupeInTown", falseBool="iseldaConvoGrimm", isTrue=→"GRIMM", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="nymmInTown", falseBool="iseldaNymmConvo", isTrue=→"NYMM", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="iseldaConvo1", isTrue=(none), isFalse=→"CONVO 1")
    on MEET → Meet
    on CONVO 1 → Convo 1
    on CORNIFER HOME → Cornifer Home
    on FINISHED → Repeat
    on GRIMM → Grimm
    on NYMM → Nymm
  }

  state Convo 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ISELDA_GREET1", "Iselda"], storeResult=(unset var))
    SetPlayerDataBool(boolName="iseldaConvo1", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Iselda_Shop_talk_02 (sharedassets10.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Cornifer Home {
    SetPlayerDataBool(boolName="iseldaCorniferHomeConvo", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ISELDA_CORNIFER_HOME", "Iselda"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Iselda_Shop_talk_01 (sharedassets12.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state To Shop {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DO SHOP OPEN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO CANCEL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ISELDA_REPEAT", "Iselda"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Iselda_Shop_talk_03 (sharedassets12.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Grimm {
    SetPlayerDataBool(boolName="iseldaConvoGrimm", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ISELDA_GRIMM", "CP2"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Iselda_Shop_talk_01 (sharedassets12.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Nymm {
    SetPlayerDataBool(boolName="iseldaNymmConvo", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ISELDA_NYMM", "CP2"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Iselda_Shop_talk_01 (sharedassets12.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var History: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var mapCrossroads: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Shop Region: gameObject = <null>
  var Voice Player: gameObject = <null>
}
