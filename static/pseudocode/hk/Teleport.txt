# Teleport
# 2 location(s):
#   Dream_Final_Boss / Boss Control/Radiance
#   GG_Radiance / Boss Control/Absolute Radiance

fsm Teleport {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Halo", storeResult=var "Halo")
    FindChild(gameObject=Self, childName="Legs", storeResult=var "Legs")
    FindChild(gameObject=Self, childName="Tele Flash", storeResult=var "Tele Flash")
    FindChild(gameObject=Self, childName="Pt Tele Out", storeResult=var "Pt Tele Out")
    FindChild(gameObject=Self, childName="Pt Feather Burst", storeResult=var "Pt Feather Burst")
    FindChild(gameObject=Self, childName="Pt AscendRise", storeResult=var "Pt AscendRise")
    on FINISHED → Idle
  }

  state Idle {
    SetBoolValue(boolVariable=var "Ascending", boolValue=false, everyFrame=false)
    on TELEPORT → Antic
    on TELEPORT ASCEND → Set Ascend
  }

  state Antic {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)  // disabled
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=radiance_teleport (sharedassets407.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Halo"), sendEvent="DOWN", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Legs", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Ascend Burst?
  }

  state Flash {
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    PlayParticleEmitter(gameObject=var "Pt Tele Out", emit=0)
    ActivateGameObject(gameObject=var "Tele Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.11, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Music?
  }

  state Arrive {
    SetPosition(gameObject=Self, vector=var "Destination", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetPolygonCollider(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele In", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Feather Burst", emit=0)
    SendEventByName(eventTarget=GameObject(<null>), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Halo"), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Reset
  }

  state Reset {
    ActivateGameObject(gameObject=var "Legs", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Notify
  }

  state Notify {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TELEPORTED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Set Ascend {
    SetBoolValue(boolVariable=var "Ascending", boolValue=true, everyFrame=false)
    on FINISHED → Antic
  }

  state Ascend Burst? {
    BoolTest(boolVariable=var "Ascending", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt AscendRise", emit=0)
    on FINISHED → Flash
  }

  state Music? {
    BoolTest(boolVariable=var "First Tele", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)  // disabled
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="RADIANCE", delay=0, everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "First Tele", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)  // disabled
    on FINISHED → Arrive
  }

  var Destination: vector3 = (0, 0, 0)

  var Ascending: bool = false
  var First Tele: bool = false
  var Halo: gameObject = <null>
  var Legs: gameObject = <null>
  var Pt AscendRise: gameObject = <null>
  var Pt Feather Burst: gameObject = <null>
  var Pt Tele Out: gameObject = <null>
  var Self: gameObject = <null>
  var Tele Flash: gameObject = <null>
}
