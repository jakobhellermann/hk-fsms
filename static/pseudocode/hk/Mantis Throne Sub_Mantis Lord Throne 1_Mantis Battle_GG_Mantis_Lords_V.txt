# Mantis Throne Sub
# 2 location(s):
#   GG_Mantis_Lords_V / Mantis Battle/Mantis Lord Throne 1
#   GG_Mantis_Lords_V / Mantis Battle/Mantis Lord Throne 3

// uses template: mlord_throne_sub_triple
fsm Mantis Throne Sub {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on MLORD CHALLENGE → First battle
  }

  state First battle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Stand")  // disabled
    on SUBS STAND → Stand
  }

  state Leave {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Throne Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_misc_jump_2 (sharedassets180.assets))
    on FINISHED → Fighting
  }

  state Fighting {
    SetMeshRenderer(gameObject=Self, active=false)
    on RETURN → Return Pause
  }

  state Return Pause {
    Wait(time=2.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return
  }

  state Return {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lord_land_on_ground (sharedassets180.assets))
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Wounded")
  }

  state Stand {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Stand")
    on SUBS LOOK → Look
  }

  state Look {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Look")
    on SUBS ROAR 1 → Roar 1
  }

  state Roar 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Triple Gesture 1")
    on SUBS ROAR 2 → Roar 2
  }

  state Roar 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Triple Gesture 2")
    on MLORD START SUB → Leave
  }
}
