# Land Detector
# 1 location(s):
#   Ruins1_28 / Chain Platform/Landing Box

fsm Land Detector {
  start State 1

  state State 1 {
    GetParent(gameObject=Self, storeResult=var "Parent")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Check Type
    on ACID → Check Type
  }

  state Hard Land {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="HARDLAND", delay=0, everyFrame=false)
    on ACID → Acid Land
  }

  state Acid Land {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="ACIDLAND", delay=0, everyFrame=false)
  }

  state Check Type {
    BoolTest(boolVariable=var "Acid", isTrue=→"ACID", isFalse=→"COLLIDE", everyFrame=false)
    on ACID → Acid Land
    on COLLIDE → Hard Land
  }

  var Acid: bool = false

  var Parent: gameObject = <null>
}
