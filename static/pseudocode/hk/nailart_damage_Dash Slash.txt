#   Knight_Pickup / Knight/Attacks/Great Slash
#   resources.assets / Knight:1/Attacks/Dash Slash
#   resources.assets / Knight:1/Attacks/Great Slash

// uses template: nailart_damage
fsm nailart_damage {
  start Init

  state Init {
    GetNailDamage(storeValue=var "nailDamage")
    ConvertIntToFloat(intVariable=var "nailDamage", floatVariable=var "Damage Float", everyFrame=false)
    FloatMultiply(floatVariable=var "Damage Float", multiplyBy=var "Multiplier", everyFrame=false)
    FormatString(format="nailDamage: {0}, multiplied: {1}", variables=[var "nailDamage", var "Damage Float"], storeResult=var "Log", everyFrame=false)
    DebugLogConsole(logLevel=0, text=var "Log", sendToUnityLog=true)
    on FINISHED → Fury?
  }

  state Fury? {
    SendEventToRegister(eventName="FURY REFRESH")
    BoolTest(boolVariable=var "Fury", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    FloatMultiply(floatVariable=var "Damage Float", multiplyBy=1.75, everyFrame=false)
    FormatString(format="Applied fury, new: {0}", variables=[var "Damage Float"], storeResult=var "Log", everyFrame=false)
    DebugLogConsole(logLevel=0, text=var "Log", sendToUnityLog=true)
    on FINISHED → Set
  }

  state Set {
    ConvertFloatToInt(floatVariable=var "Damage Float", intVariable=var "nailDamage", rounding=Nearest, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=var "nailDamage", everyFrame=false)
  }

  var Multiplier: float = 2.5

  var Damage Float: float = 0
  var nailDamage: int = 0
  var Fury: bool = false
  var Log: string = ""
}
