// uses template: equip_position
fsm equip_position {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetIntValue(intVariable=var "Item Counter", intValue=var "Item Number", everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Init X Pos", y=var "Init Y Pos", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=var "Y Pos", z=(unset), space=1, everyFrame=false)
    SetIntValue(intVariable=var "Loops", intValue=var "Item Number", everyFrame=false)
    SetIntValue(intVariable=var "Current Column", intValue=1, everyFrame=false)
    BoolTest(boolVariable=var "Break", isTrue=→"BREAK", isFalse=(none), everyFrame=false)
    on FINISHED → Loop Check
    on BREAK → Break
  }

  state Loop Check {
    IntCompare(integer1=var "Loops", integer2=1, equal=(none), lessThan=(none), greaterThan=→"LOOP", everyFrame=false)
    on LOOP → Increment X
    on FINISHED → Set Position
  }

  state Increment X {
    FloatAdd(floatVariable=var "X Pos", add=var "X Increment", everyFrame=false, perSecond=false)
    IntAddV2(intVariable=var "Loops", add=-1, everyFrame=false)
    IntAddV2(intVariable=var "Current Column", add=1, everyFrame=false)
    IntCompare(integer1=var "Current Column", integer2=var "Columns", equal=(none), lessThan=(none), greaterThan=→"SHIFT DOWN", everyFrame=false)
    on SHIFT DOWN → Shift Y Down
    on FINISHED → Loop Check
  }

  state Shift Y Down {
    SetFloatValue(floatVariable=var "X Pos", floatValue=var "Init X Pos", everyFrame=false)
    FloatAdd(floatVariable=var "Y Pos", add=var "Y Increment", everyFrame=false, perSecond=false)
    SetIntValue(intVariable=var "Current Column", intValue=1, everyFrame=false)
    on FINISHED → Loop Check
  }

  state Set Position {
    SetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=var "Y Pos", z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  state Break {
    on FINISHED → Loop Check
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Item Number: int = 0
  var Break: bool = false

  var Init X Pos: float = 1.1
  var Init Y Pos: float = -3.8
  var X Increment: float = 2.08
  var X Pos: float = 0
  var Y Increment: float = -2.18
  var Y Pos: float = 0
  var Columns: int = 4
  var Current Column: int = 1
  var Item Counter: int = 0
  var Loops: int = 0
  var Self: gameObject = <null>
}
