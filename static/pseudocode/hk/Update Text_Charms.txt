# Update Text
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Charms
#   resources.assets / _GameCameras/HudCamera/Inventory/Charms

fsm Update Text {
  start Init
  on PANE RESET → Pane Reset  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    FindChild(gameObject=Self, childName="Text Desc", storeResult=var "Text Desc")
    FindChild(gameObject=Self, childName="Text Name", storeResult=var "Text Name")
    on UPDATE TEXT → Change Text
  }

  state Fade Down {
    SendEventByName(eventTarget=GameObject(var "Text Desc"), sendEvent="CHANGE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Text Name"), sendEvent="CHANGE DOWN", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Change Text
  }

  state Fade Up {
    SendEventByName(eventTarget=GameObject(var "Text Desc"), sendEvent="CHANGE UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Text Name"), sendEvent="CHANGE UP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Pane Reset {
    SendEventByName(eventTarget=GameObject(var "Text Desc"), sendEvent="CHANGE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Text Name"), sendEvent="CHANGE DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Change Text {
    GetLanguageString(sheetName=var "Sheet", convName=var "Convo Name", storeValue=var "Name String")
    GetLanguageString(sheetName=var "Sheet", convName=var "Convo Desc", storeValue=var "Desc String")
    SetTextMeshProText(gameObject=var "Text Name", textString=var "Name String")
    SetTextMeshProText(gameObject=var "Text Desc", textString=var "Desc String")
    on FINISHED → Idle
  }

  var Convo Desc: string = ""
  var Convo Name: string = ""
  var Desc String: string = ""
  var Name String: string = ""
  var Sheet: string = "UI"
  var Text Desc: gameObject = <null>
  var Text Name: gameObject = <null>
}
