# Cancel
# 1 location(s):
#   resources.assets / dream_gate_object

fsm Cancel {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Other Pieces", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dream_gate", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Particle System", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "beam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "beam (1)", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "haze", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Disappear", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Disappear", activate=true)
    on FINISHED → Idle
  }

  state Idle {
    on DREAM GATE SET → Disappear
  }

  state Recycle {
    RecycleSelf()
  }

  state Disappear {
    ActivateGameObject(gameObject=var "Other Pieces", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dream_gate", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Particle System", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "beam", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "beam (1)", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "haze", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Disappear", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  var Disappear: gameObject = dream_gate_object/Disappear
  var Dream_gate: gameObject = dream_gate_object/Dream_Gate
  var Other Pieces: gameObject = dream_gate_object/Dream_Gate_Other_pieces_0001_2 (1)
  var Particle System: gameObject = dream_gate_object/Particle System
  var beam: gameObject = dream_gate_object/beam
  var beam (1): gameObject = dream_gate_object/beam (1)
  var haze: gameObject = dream_gate_object/haze2 (1)
}
