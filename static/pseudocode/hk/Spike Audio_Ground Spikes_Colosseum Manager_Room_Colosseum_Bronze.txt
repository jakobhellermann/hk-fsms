# Spike Audio
# 1 location(s):
#   Room_Colosseum_Bronze / Colosseum Manager/Ground Spikes

fsm Spike Audio {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on EXPAND → Antic
  }

  state Antic {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=col_spikes_appear_and_charge (sharedassets32.assets))
    on FINISHED → Expand
  }

  state Expand {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Expand")
    StopParticleEmitter(gameObject=var "Dust")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=col_spikes_snap_up (sharedassets32.assets))
    on RETRACT → Retract
  }

  state Retract {
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Retract", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=col_spikes_disappear (sharedassets32.assets))
    on FINISHED → Idle
  }

  var Dust: gameObject = <null>
}
