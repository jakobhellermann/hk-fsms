# Disable
# 1 location(s):
#   sharedassets6.assets / grimm_area_effect

fsm Disable {
  start Idle

  state Idle {
    on DREAM AREA DISABLE → Disable
  }

  state Disable {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt 1")
    StopParticleEmitter(gameObject=var "Pt 2")
    StopParticleEmitter(gameObject=var "Pt 3")
    StopParticleEmitter(gameObject=var "Pt 4")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)  // disabled
  }

  var Pt 1: gameObject = grimm_area_effect/dream_particle_03 (1)
  var Pt 2: gameObject = grimm_area_effect/dream_particle_03 (2)
  var Pt 3: gameObject = <null>
  var Pt 4: gameObject = <null>
}
