# Crystal Form
# 1 location(s):
#   sharedassets251.assets / Crystal Shot

fsm Crystal Form {
  start Idle

  state Idle {
    ActivateGameObject(gameObject=var "Growth", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=0, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL", storeCollider=var "Nail Object")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "")
    FaceAngle(gameObject=Self, angleOffset=90, everyFrame=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetCollider(gameObject=Self, active=true)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    SetParticleEmission(gameObject=var "Shatter", emission=false)
    on COLLIDE → Hit
    on NAIL → Nail Hit
  }

  state Hit {
    SetSpriteRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Growth", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Growth", audioClip=mines_crystal_expand, pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=var "Growth", animLibName="", clipName="Crystal Grow")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Wait(time=0.13, finishEvent=→"FINISHED", realTime=false)
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    RandomFloat(min=0.9, max=1.3, storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=var "Scale", everyFrame=false, lateUpdate=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    on FINISHED → Grown
  }

  state Grown {
    SetPolygonCollider(gameObject=Self, active=true)
    SendMessage(gameObject=var "Growth", delivery=SendMessage, options=1, functionCall=flashFocusHeal())
    ReceivedDamage(collideTag="", sendEvent=→"COLLIDE", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    WaitRandom(timeMin=10, timeMax=12, finishEvent=→"SMASH", realTime=false)
    on COLLIDE → Damaged?
    on SMASH → Smash
  }

  state Damaged? {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage Dealt", everyFrame=false)
    IntCompare(integer1=var "Damage Dealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"SMASH", everyFrame=false)
    on SMASH → Shake
    on CANCEL → Grown
  }

  state Smash {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Growth", audioClips=[crystal_break_1 (resources.assets), crystal_break_2 (resources.assets)], weights=[1, 1], pitchMin=0.8, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Strike Nail (sharedassets6.assets), spawnPoint=var "Growth", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetParticleEmission(gameObject=var "Shatter", emission=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Animation
  }

  state Particle {
    ActivateGameObject(gameObject=var "Growth", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmission(gameObject=var "Shatter", emission=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Shake {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Smash
  }

  state Animation {
    SetCollider(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Growth", clipName="Crystal Smash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Particle
  }

  state Nail Hit {
    GetFsmFloat(gameObject=var "Nail Object", fsmName="damages_enemy", variableName="direction", storeValue=var "Direction", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Direction", speed=30, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)  // disabled
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=10, everyFrame=false)
    on FINISHED → Idle
  }

  var Direction: float = 0
  var Scale: float = 0
  var Damage Dealt: int = 0
  var Damager: gameObject = <null>
  var Growth: gameObject = Crystal Shot/Growth
  var Nail Object: gameObject = <null>
  var Shatter: gameObject = Crystal Shot/Hit Crystals
}
