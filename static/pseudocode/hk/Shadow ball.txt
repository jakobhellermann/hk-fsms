# Shadow ball
# 1 location(s):
#   resources.assets / Shadow Ball

fsm Shadow ball {
  start Wait

  state Wait {
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on WAIT → Fade
  }

  state Fade {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=true)
    FloatMultiplyV2(floatVariable=var "Speed", multiplyBy=0.1, everyFrame=true, fixedUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fireball End", animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Destroy
  }

  state Destroy {
    SetParticleEmission(gameObject=var "Particles", emission=false)
    FindChild(gameObject=Self, childName="HeroLight", storeResult=var "Light")
    DestroyObject(gameObject=var "Light", delay=0, detachChildren=false)
    DestroySelf(detachChildren=true)
  }

  var Speed: float = 0
  var Light: gameObject = <null>
  var Particles: gameObject = Shadow Ball/Particle System
}
