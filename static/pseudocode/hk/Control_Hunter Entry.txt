  start Init

  state Init {
    FindChild(gameObject=Self, childName="Hunter", storeResult=var "Hunter")
    FindChild(gameObject=Self, childName="Blocker", storeResult=var "Blocker")
    FindChild(gameObject=Self, childName="Slam", storeResult=var "Slam")
    FindChild(gameObject=Self, childName="Dust Hit Plume", storeResult=var "Dust Plume")
    FindChild(gameObject=Self, childName="Roar Point", storeResult=var "Roar Point")

    FindGameObject(objectName="Break Floor 1 (1)", withTag="Untagged", store=var "Break Floor 1 (1)")

  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    on HUNTER ENTRY → Fall Pause
  }

  state Fall Pause {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    SetPolygonCollider(gameObject=var "Break Floor 1 (1)", active=false)
    on FINISHED → Fall
  }

  state Fall {
    Tk2dPlayAnimation(gameObject=var "Hunter", animLibName="", clipName="Fall")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hollow_knight_self_stab_prepare (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

    Translate(gameObject=var "Hunter", vector=(unset), x=(unset), y=15, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    iTweenMoveBy(gameObject=var "Hunter", id="", vector=(0, -15, 0), time=0.28, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)

  }

    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)

    ActivateGameObject(gameObject=var "Slam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust Plume", emit=0)
    ActivateGameObject(gameObject=var "Hunter Tink", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Roar
  }

  state Roar {
    ApplyMusicCue(musicCue=Boss1 (sharedassets48.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=mawlek_scream (sharedassets45.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

    on FINISHED → Roar End
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Roar"), sendEvent="END", delay=0, everyFrame=false)

    on GIVE → Give Anim
  }

  state Give Anim {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hunter", clipName="Give", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    TransitionToAudioSnapshot(snapshot=Normal Flangier (resources.assets), transitionTime=0.1)

  }

  state Hand NPC {
    SetCollider(gameObject=var "Hand NPC", active=true)
    ActivateGameObject(gameObject=var "Hand Plat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hunter", animLibName="", clipName="Give End")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=moss_knight_burst_leaves (sharedassets34.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  var Area Title: gameObject = <null>
  var Blocker: gameObject = <null>
  var Break Floor 1 (1): gameObject = <null>
