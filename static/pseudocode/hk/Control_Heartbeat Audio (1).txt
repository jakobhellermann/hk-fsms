# Control
# 1 location(s):
#   Grimm_Main_Tent / Heartbeat Audio (1)

fsm Control {
  start Init
  on HEARTBEAT STOP → Stop  // from any state
  on HEARTBEAT START → Pause 1  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Pause 1
  }

  state Pause 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Beat 1
  }

  state Beat 1 {
    AudioPlayRandom(gameObject=var "Self", audioClips=[heartbeat_B_01 (sharedassets391.assets), heartbeat_B_02 (sharedassets391.assets), heartbeat_B_03 (sharedassets391.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)  // disabled
    on FINISHED → Pause 1
  }

  state Stop {
  }

  var Self: gameObject = <null>
}
