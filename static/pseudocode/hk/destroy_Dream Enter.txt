# destroy
# 1 location(s):
#   Deepnest_Spider_Town / Dream Enter

// uses template: destroy_if_playerdatabool
fsm destroy {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var playerData bool: string = "hegemolDefeated"
}
