# Control
# 1 location(s):
#   Abyss_Lighthouse_room / Light Control

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Pt Enter", storeResult=var "Pt Enter")
    FindChild(gameObject=Self, childName="Pt Glow", storeResult=var "Pt Glow")
    on OPEN → Start Pause
    on ACTIVATE → Activate
  }

  state Start Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fly Start
  }

  state Activate {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Glow", emit=0)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
  }

  state Fly Start {
    PlayParticleEmitter(gameObject=var "Pt Enter", emit=0)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=lighthouse_lantern (sharedassets347.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Glow Start
  }

  state Glow Start {
    PlayParticleEmitter(gameObject=var "Pt Glow", emit=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Glow Up
  }

  state Glow Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Enter")
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=1)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
  }

  var Pt Enter: gameObject = <null>
  var Pt Glow: gameObject = <null>
}
