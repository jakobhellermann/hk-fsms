# White Charm
# 2 location(s):
#   Knight_Pickup / Knight/Charm Effects
#   resources.assets / Knight:1/Charm Effects

fsm White Charm {
  start Pause
  on CHARM INDICATOR CHECK → Check  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_36", isTrue=(none), isFalse=→"INACTIVE")
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    IntCompare(integer1=var "Royal Charm State", integer2=3, equal=→"ACTIVE", lessThan=→"INACTIVE", greaterThan=→"INACTIVE", everyFrame=false)
    on ACTIVE → Wait
    on INACTIVE → Inactive
  }

  state Inactive {
  }

  state Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → SOUL Check
  }

  state SOUL Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="disablePause", isTrue=→"CANCEL", isFalse=→"SOUL UP")
    on CANCEL → Wait
    on SOUL UP → Soul UP
  }

  state Soul UP {
    SendMessageV2(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddMPCharge(4), everyFrame=false)
    on FINISHED → Wait
  }

  var Royal Charm State: int = 0
}
