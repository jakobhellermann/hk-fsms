# Battle Control
# 1 location(s):
#   Dream_01_False_Knight / Battle Scene

fsm Battle Control {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Start
  }

  state Init {
    FindGameObject(objectName="Zombie Runner", withTag="Untagged", store=var "Runner")
    FindGameObject(objectName="Zombie Barger", withTag="Untagged", store=var "Barger")
    FindGameObject(objectName="Music Region B", withTag="Untagged", store=var "Tension Region")
    FindGameObject(objectName="Zombie Hornhead", withTag="Untagged", store=var "Hornhead")
    FindGameObject(objectName="False Knight New", withTag="Untagged", store=var "False Knight")
    FindGameObject(objectName="CameraLockArea B", withTag="Untagged", store=var "CameraLock 1")
    FindGameObject(objectName="CameraLockArea B2", withTag="Untagged", store=var "CameraLock 2")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "CameraLock 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindGameObject(objectName="FK Armour", withTag="Untagged", store=var "Armour")
    DestroyObject(gameObject=var "Armour", delay=0, detachChildren=false)
    FindChild(gameObject=var "GameManager", childName="Music", storeResult=var "Music Folder")
    FindChild(gameObject=var "Music Folder", childName="Track 1", storeResult=var "Track 1")
    on FINISHED → Detect
    on ACTIVATED → Activate
  }

  state Start {
    SetIntValue(intVariable=var "Battle Enemies", intValue=1, everyFrame=false)
    DestroyObject(gameObject=var "Tension Region", delay=0, detachChildren=false)
    ActivateGameObject(gameObject=var "CameraLock 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CameraLock 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Track 1"), sendEvent="BOSS 1", delay=0, everyFrame=false)  // disabled
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    on END → End Wait
  }

  state End Wait {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DestroyObject(gameObject=var "CameraLock 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "CameraLock 2", delay=0, detachChildren=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    FindGameObject(objectName="False Knight New", withTag="Untagged", store=var "False Knight")
    DestroyObject(gameObject=var "Tension Region", delay=0, detachChildren=false)
    FindGameObject(objectName="FK Floor", withTag="Untagged", store=var "Floor")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    DestroyObject(gameObject=var "CameraLock 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "CameraLock 2", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "False Knight", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Hornhead", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Barger", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Runner", delay=0, detachChildren=false)
    SendEventByName(eventTarget=GameObject(var "Floor"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
  }

  state Init 2 {
    FindChild(gameObject=var "GameManager", childName="Music", storeResult=var "Music Folder")
    FindChild(gameObject=var "Music Folder", childName="Track 1", storeResult=var "Track 1")
    on FINISHED → 
  }

  state Start 2 {
    SendEventByName(eventTarget=GameObject(var "Track 1"), sendEvent="STOP FADE", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Track 1"), sendEvent=var "Track Event", delay=0, everyFrame=false)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Track 1"), sendEvent="PLAY", delay=0, everyFrame=false)  // disabled
    NextFrameEvent(sendEvent=→"EXIT")  // disabled
    on EXIT → 
  }

  state Music {
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="BOSS 1", delay=0, everyFrame=false)
    on FINISHED → Start
  }

  var Track Event: string = ""

  var Battle Enemies: int = 0
  var Activated: bool = false
  var Armour: gameObject = <null>
  var Barger: gameObject = <null>
  var CameraLock 1: gameObject = <null>
  var CameraLock 2: gameObject = <null>
  var False Knight: gameObject = <null>
  var Floor: gameObject = <null>
  var Hornhead: gameObject = <null>
  var Music Folder: gameObject = <null>
  var Runner: gameObject = <null>
  var Tension Region: gameObject = <null>
  var Track 1: gameObject = <null>
}
