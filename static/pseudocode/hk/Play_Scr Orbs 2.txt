fsm Play {
  start Idle

  state Idle {
    SetParticleEmission(gameObject=Self, emission=false)
    on PLAY → Play
  }

  state Play {
    SetParticleEmission(gameObject=Self, emission=true)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on PLAY → Replay
  }

  state Replay {
    on FINISHED → Play
  }
}
