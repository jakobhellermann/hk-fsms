# Manhole Control
# 1 location(s):
#   Ruins1_05b / Waterways Manhole

fsm Manhole Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="openedWaterwaysManhole", isTrue=→"OPENED", isFalse=(none))
    on OPENED → Opened
    on MANHOLE OPEN → Open
  }

  state Opened {
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Open {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    ActivateGameObject(gameObject=var "Open Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  var Open Anim: gameObject = Waterways Manhole/Open Anim
  var Opened: gameObject = Waterways Manhole/Opened
}
