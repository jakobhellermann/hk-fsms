# Mantis Flyer
# 4 location(s):
#   Fungus2_12 / Mantis Flyer Child
#   Fungus2_13 / Mantis Flyer Child
#   Fungus2_14 / Mantis Flyer Child
#   Fungus2_15 / Mantis Flyer Child

fsm Mantis Flyer {
  start Init
  on MANTIS DEFEAT → Disappear  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    BoolTest(boolVariable=var "Start Idle", isTrue=→"START IDLE", isFalse=(none), everyFrame=false)
    on FINISHED → Rest
    on START IDLE → Idle Buzz
  }

  state Chase {
    DistanceFly(gameObject=var "Self", target=var "Hero", distance=7, speedMax=6, acceleration=0.2, targetsHeight=true, height=4)
    SetRecoilSpeed(target=Self, newRecoilSpeed=15)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Turn To Fly", resetFrame=true, everyFrame=true)
    BoolTest(boolVariable=var "In Range", isTrue=→"ATTACK", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    Wait(time=8, finishEvent=→"CHECK", realTime=false)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedMantisLords", isTrue=→"LORDS", isFalse=(none))
    on ATTACK → Pause
    on CHECK → Check Near
    on GO UP → Go up
  }

  state Attack Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.95)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mantis_flyer_child_awake (sharedassets176.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Attack
  }

  state Attack {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mantis_flyer_child_charge (sharedassets34.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    SetVelocityAsAngle(gameObject=Self, angle=var "Attack Angle", speed=var "Attack Speed", everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Sting Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Return
  }

  state Recover {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → CD
  }

  state Return {
    FloatAdd(floatVariable=var "Attack Angle", add=180, everyFrame=false, perSecond=false)
    ActivateGameObject(gameObject=var "Sting Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Attack Angle", speed=var "Attack Speed", everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recover
  }

  state CD {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    DistanceFly(gameObject=var "Self", target=var "Hero", distance=7, speedMax=6, acceleration=0.2, targetsHeight=true, height=4)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Turn To Fly", resetFrame=true, everyFrame=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    WaitRandom(timeMin=0.35, timeMax=0.65, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Chase
  }

  state Rest {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedMantisLords", isTrue=→"LORDS", isFalse=(none))
    BoolTest(boolVariable=var "Wake Region", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    on WAKE → Lord Def?
    on TOOK DAMAGE → Wake
    on LORDS → Lords Defeated
  }

  state Wake {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mantis_flyer_child_awake (sharedassets176.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHeroAmount(target=Self, damageDealt=1)
    iTweenRotateTo(gameObject=Self, id="", transformRotation=var "", vectorRotation=(0, 0, 0), time=0.2, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Chase
  }

  state Idle Buzz {
    BoolTest(boolVariable=var "Wake Region", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    IdleBuzz(gameObject=Self, waitMin=0.75, waitMax=1, speedMax=1.75, accelerationMax=15, roamingRange=1.5)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Turn To Fly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    on WAKE → Chase
  }

  state Get Angle {
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Attack Angle", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on R → Right
    on L → Left
  }

  state Right {
    FloatClamp(floatVariable=var "Attack Angle", minValue=270, maxValue=360, everyFrame=false)
    on FINISHED → Attack Antic
  }

  state Left {
    FloatClamp(floatVariable=var "Attack Angle", minValue=180, maxValue=270, everyFrame=false)
    on FINISHED → Attack Antic
  }

  state Pause {
    WaitRandom(timeMin=0.15, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    DistanceFly(gameObject=var "Self", target=var "Hero", distance=7, speedMax=6, acceleration=0.2, targetsHeight=true, height=4)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Turn To Fly", resetFrame=true, everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    on FINISHED → Recheck
  }

  state Recheck {
    BoolTest(boolVariable=var "In Range", isTrue=→"FINISHED", isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Get Angle
    on CANCEL → Pause
  }

  state Check Near {
    BoolTest(boolVariable=var "Wake Region", isTrue=→"FINISHED", isFalse=→"HERO LOST", everyFrame=false)
    on FINISHED → Chase
    on HERO LOST → Idle Buzz
  }

  state Lords Defeated {
    SetDamageHeroAmount(target=Self, damageDealt=0)
    SetRecoilSpeed(target=Self, newRecoilSpeed=0)
    SetDreamNailConvo(target=Self, title="MANTIS_YOUNG_PASSIVE")
    on TOOK DAMAGE → Wake
  }

  state Lord Def? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedMantisLords", isTrue=→"LORDS", isFalse=(none))
    on FINISHED → Wake
    on LORDS → Lords Defeated
  }

  state Go up {
    SetVelocity2d(gameObject=Self, vector=(0, 3), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Idle Buzz
  }

  state Disappear {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Attack Speed: float = 25
  var Start Idle: bool = false

  var Attack Angle: float = 0
  var X Scale: float = 0
  var In Range: bool = false
  var Wake Region: bool = false
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Sting Hit: gameObject = Mantis Flyer Child/Sting Hit
}
