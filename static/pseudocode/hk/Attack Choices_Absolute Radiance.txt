# Attack Choices
# 1 location(s):
#   GG_Radiance / Boss Control/Absolute Radiance

fsm Attack Choices {
  start Init
  on ARENA 1 END → A1 End  // from any state
  on ARENA 2 END → A2 End  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on ARENA 1 START → A1 Pause
  }

  state A1 Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → A1 Choice
  }

  state A1 Choice {
    SendEventByName(eventTarget=Self, sendEvent="ORBS", delay=0, everyFrame=false)  // disabled
    SendRandomEventV3(events=[→"NAIL R SWEEP", →"NAIL L SWEEP", →"NAIL TOP SWEEP", →"EYE BEAMS", →"BEAM SWEEP L", →"BEAM SWEEP R", →"NAIL FAN", →"ORBS"], weights=[0.5, 0.5, 0.75, 1, 0.75, 0.75, 1, 1], trackingInts=[var "Ct Nail R Sweep", var "Ct Nail L Sweep", var "Ct Nail Top Sweep", var "Ct Eye Beam", var "Ct Beam Sweep L", var "Ct Beam Sweep R", var "Ct Nail Fan", var "Ct Orbs"], eventMax=[1, 1, 1, 2, 1, 1, 2, 1], trackingIntsMissed=[var "Ms Nail R Sweep", var "Ms Nail L Sweep", var "Ms Nail Top Sweep", var "Ms Eye Beam", var "Ms Beam Sweep L", var "Ms Beam Sweep R", var "Ms Nail Fan", var "Ms Orbs"], missedMax=[12, 12, 12, 10, 12, 12, 10, 12])
    on NAIL TOP SWEEP → Nail Top Sweep
    on NAIL L SWEEP → Nail L Sweep
    on NAIL R SWEEP → Nail R Sweep
    on EYE BEAMS → Eye Beam Wait
    on BEAM SWEEP L → Beam Sweep L
    on BEAM SWEEP R → Beam Sweep R
    on NAIL FAN → Nail Fan Wait
    on ORBS → Orb Wait
  }

  state Nail Top Sweep {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP", delay=0.75, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP", delay=1.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP", delay=2.25, everyFrame=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Nail L Sweep {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB L", delay=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB L", delay=1.7, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB L", delay=2.9, everyFrame=false)
    Wait(time=4.4, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Nail R Sweep {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB R", delay=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB R", delay=1.7, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB R", delay=2.9, everyFrame=false)
    Wait(time=4.4, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Eye Beam Wait {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Please Cast", setValue=true, everyFrame=false)
    on CAST READY → Eye Beam
  }

  state Eye Beam {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EYE BEAMS", delay=0, everyFrame=false)
    on ATTACK END → Eye Beam Recover
  }

  state Eye Beam Recover {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CAST END", delay=0, everyFrame=false)
    on FINISHED → Return
  }

  state Beam Sweep L {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BEAM SWEEP L", delay=0, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Beam Sweep R {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BEAM SWEEP R", delay=0, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Nail Fan Wait {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Please Cast", setValue=true, everyFrame=false)
    on CAST READY → Nail Fan
  }

  state Nail Fan {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="NAIL FAN", delay=0, everyFrame=false)
    on ATTACK END → Nail Fan Recover
  }

  state Nail Fan Recover {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CAST END", delay=0, everyFrame=false)
    on FINISHED → Return
  }

  state Orb Wait {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Please Cast", setValue=true, everyFrame=false)
    on CAST READY → Orbs
  }

  state Orbs {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ORBS", delay=0, everyFrame=false)
    on ATTACK END → Orb Recover
  }

  state Orb Recover {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CAST END", delay=0, everyFrame=false)
    on FINISHED → Return
  }

  state A1 End {
    SetIntValue(intVariable=var "Arena", intValue=2, everyFrame=false)
    on ARENA 2 START → A2 Pause
  }

  state Return {
    IntCompare(integer1=var "Arena", integer2=1, equal=→"A1", lessThan=→"A1", greaterThan=→"A2", everyFrame=false)
    on A1 → A1 Choice
    on A2 → A2 Choice
  }

  state A2 Choice {
    SendEventByName(eventTarget=Self, sendEvent="ORBS", delay=0, everyFrame=false)  // disabled
    SendRandomEventV3(events=[→"NAIL LR SWEEP", →"NAIL FAN", →"ORBS", →"EYE BEAMS", →"BEAM SWEEP L", →"BEAM SWEEP R"], weights=[0.5, 1, 1, 1, 0.75, 0.75], trackingInts=[var "Ct Nail R Sweep", var "Ct Nail Fan", var "Ct Orbs", var "Ct Eye Beam", var "Ct Beam Sweep L", var "Ct Beam Sweep R"], eventMax=[1, 2, 1, 2, 1, 1], trackingIntsMissed=[var "Ms Nail R Sweep", var "Ms Nail Fan", var "Ms Orbs", var "Ms Eye Beam", var "Ms Beam Sweep L", var "Ms Beam Sweep R"], missedMax=[12, 10, 10, 10, 12, 12])
    on NAIL LR SWEEP → L or R Choice
    on EYE BEAMS → Eye Beam Wait
    on BEAM SWEEP L → Beam Sweep L 2
    on BEAM SWEEP R → Beam Sweep R 2
    on NAIL FAN → Nail Fan Wait
    on ORBS → Orb Wait
  }

  state A2 Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → A2 Choice
  }

  state Nail Top Sweep 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP2", delay=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP2", delay=1.5, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP2", delay=2.5, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB TOP2", delay=3.5, everyFrame=false)  // disabled
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Nail L Sweep 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB L2", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB L2", delay=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB L2", delay=3.5, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB L2", delay=5, everyFrame=false)  // disabled
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Nail R Sweep 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB R2", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB R2", delay=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB R2", delay=3.5, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="COMB R2", delay=5, everyFrame=false)  // disabled
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Beam Sweep L 2 {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BEAM SWEEP L2", delay=0, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state Beam Sweep R 2 {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BEAM SWEEP R2", delay=0, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Last Attack Was Cast", setValue=false, everyFrame=false)
    on FINISHED → Return
  }

  state L or R Choice {
    SendRandomEvent(events=[→"NAIL L SWEEP", →"NAIL R SWEEP"], weights=[1, 1], delay=0)
    on NAIL R SWEEP → Nail R Sweep 2
    on NAIL L SWEEP → Nail L Sweep 2
  }

  state A2 End {
  }

  var Arena: int = 1
  var Ct Beam Sweep L: int = 0
  var Ct Beam Sweep R: int = 0
  var Ct Eye Beam: int = 0
  var Ct Nail Fan: int = 0
  var Ct Nail L Sweep: int = 0
  var Ct Nail R Sweep: int = 0
  var Ct Nail Top Sweep: int = 0
  var Ct Orbs: int = 0
  var Ms Beam Sweep L: int = 0
  var Ms Beam Sweep R: int = 0
  var Ms Eye Beam: int = 0
  var Ms Nail Fan: int = 0
  var Ms Nail L Sweep: int = 0
  var Ms Nail R Sweep: int = 0
  var Ms Nail Top Sweep: int = 0
  var Ms Orbs: int = 0
}
