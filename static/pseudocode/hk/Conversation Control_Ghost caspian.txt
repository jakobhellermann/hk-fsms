# Conversation Control
# 1 location(s):
#   RestingGrounds_08 / Ghost caspian

fsm Conversation Control {
  start Pause
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state
  on REVEK DEJECTED → Dejected  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDreamNail", isTrue=(none), isFalse=→"INERT")
    FindChild(gameObject=Self, childName="Character Sprite", storeResult=var "Character Sprite")
    on FINISHED → Revek Failure?
    on INERT → Inert
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BOX DOWN DREAM", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Hero Look {
    BoolTest(boolVariable=var "Hero Looks Up", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BOX UP DREAM", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue=var "Title Convo", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Talk {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Talk Convo", "Ghosts"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Anim End {
    BoolTest(boolVariable=var "Hero Looks Up", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Convo Choice {
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"TALK", everyFrame=false)
    on TALK → Talk
    on REPEAT → Repeat
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Repeat Convo", "Ghosts"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Set Convos {
    BuildString(stringParts=[var "Ghost Name", "_TALK"], separator="", addToEnd=true, storeResult=var "Talk Convo", everyFrame=false)
    BuildString(stringParts=[var "Ghost Name", "_REPEAT"], separator="", addToEnd=true, storeResult=var "Repeat Convo", everyFrame=false)
    BuildString(stringParts=["GH_", var "Ghost Name", "_C"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    BuildString(stringParts=["GH_", var "Ghost Name", "_NC"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inert {
    SetCollider(gameObject=Self, active=false)
  }

  state Revek Failure? {
    BoolTest(boolVariable=var "z_Revek", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="gladeGhostsKilled", storeValue=var "Glade Ghosts Killed")
    IntCompare(integer1=var "Glade Ghosts Killed", integer2=17, equal=→"DEJECTED", lessThan=(none), greaterThan=→"DEJECTED", everyFrame=false)
    on FINISHED → Set Convos
    on DEJECTED → Dejected
  }

  state Dejected {
    SetStringValue(stringVariable=var "Talk Convo", stringValue="REVEK_TALK_DEFEAT", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Convo", stringValue="REVEK_TALK_DEFEAT_REPEAT", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Character Sprite", animLibName="", clipName="Revek Sad")
    SetFsmBool(gameObject=Self, fsmName="ghost_npc_death", variableName="z_Revek", setValue=false, everyFrame=false)
    BuildString(stringParts=["GH_", var "Ghost Name", "_C"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    BuildString(stringParts=["GH_", var "Ghost Name", "_NC"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    on FINISHED → Idle
  }

  var Hero Looks Up: bool = true
  var z_Revek: bool = false
  var Ghost Name: string = "CASPIAN"

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var Glade Ghosts Killed: int = 0
  var History: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Spoken: bool = false
  var Visited Crossroads: bool = false
  var mapCrossroads: bool = false
  var Repeat Convo: string = ""
  var Talk Convo: string = ""
  var Title Convo: string = ""
  var Area Title: gameObject = <null>
  var Character Sprite: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
