# Conversation Control
# 1 location(s):
#   Mines_13 / Quirrel Mines

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Stop Anim  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Convo Choice {
    GetPlayerDataInt(gameObject=var "GameManager", intName="quirrelMines", storeValue=var "Mines State")
    IntSwitch(intVariable=var "Mines State", compareTo=[0, 1, 2], sendEvent=[→"CONVO 1", →"CONVO 2", →"CONVO 3"], everyFrame=false)
    on CONVO 1 → Quirrel 1
    on CONVO 2 → Quirrel 2
    on CONVO 3 → Quirrel 3
    on FINISHED → Quirrel 4
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="LookUpEnd", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")  // disabled
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Quirrel 1 {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="quirrelMines")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    SetPlayerDataBool(boolName="metQuirrel", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_MINES_1", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Stop Anim {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Play Talk End", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Quirrel 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_MINES_2", "Quirrel"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="quirrelMines")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Quirrel 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_MINES_3", "Quirrel"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="quirrelMines")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Quirrel 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_MINES_4", "Quirrel"], storeResult=(unset var))
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="quirrelMines")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_05_hmmm (sharedassets19.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var Egg Temple: int = 0
  var Elderbug State: int = 0
  var Mines State: int = 0
  var Met: bool = false
  var Play Talk End: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
