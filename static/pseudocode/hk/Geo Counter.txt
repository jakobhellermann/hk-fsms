# Geo Counter
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Geo Counter
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Geo Counter

fsm Geo Counter {
  start Init

  state Idle {
    on TO ZERO → To Zero
    on ADD GEO → Add Geo
    on TAKE GEO → Take Geo
    on ZERO DEATH → Zero Death
    on LEVEL LOADED → Reset Counter
  }

  state To Zero {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=ToZero())
    on FINISHED → Zero Death
  }

  state Add Geo {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddGeo(var "Amount"))
    on FINISHED → Idle
  }

  state Take Geo {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=TakeGeo(var "Amount"))
    on FINISHED → Idle
  }

  state Zero Death {
    SendEventByName(eventTarget=GameObject(var "Add Text"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Subtract Text"), sendEvent="DOWN", delay=0, everyFrame=false)
    DestroyObject(gameObject=var "Add Text", delay=0, detachChildren=false)  // disabled
    DestroyObject(gameObject=var "Subtract Text", delay=0, detachChildren=false)  // disabled
    on FINISHED → Idle
  }

  state Rand Geo {
    RandomInt(min=167, max=2689, storeResult=var "Amount", inclusiveMax=false)  // disabled
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddGeo(var "Amount"))  // disabled
    on FINISHED → Idle
  }

  state Reset Counter {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=NewSceneRefresh())
    SendEventByName(eventTarget=GameObject(var "Add Text"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Subtract Text"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Geo Text"), sendEvent="UP", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Geo Sprite", active=true)
    Tk2dPlayAnimation(gameObject=var "Geo Sprite", animLibName="", clipName="Coin Idle")
    on FINISHED → Idle
  }

  state Init {
    FindChild(gameObject=Self, childName="Add Text", storeResult=var "Add Text")
    FindChild(gameObject=Self, childName="Subtract Text", storeResult=var "Subtract Text")
    FindChild(gameObject=Self, childName="Geo Text", storeResult=var "Geo Text")
    FindChild(gameObject=Self, childName="Geo Sprite", storeResult=var "Geo Sprite")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rand Geo
  }

  var Amount: int = 50

  var Add Text: gameObject = <null>
  var Geo Sprite: gameObject = <null>
  var Geo Text: gameObject = <null>
  var Subtract Text: gameObject = <null>
}
