# Fireball Control
# 1 location(s):
#   resources.assets / Fireball

fsm Fireball Control {
  start Set Damage

  state Idle {
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"WALL", storeCollider=var "Collider", storeForce=(unset))
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"WALL", storeCollider=var "Collider", storeForce=(unset))
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"WALL", storeCollider=var "Collider", storeForce=(unset))
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"WALL", storeCollider=var "Collider", storeForce=(unset))
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Crt X Velocity", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Crt X Velocity", float2=0, tolerance=0.1, equal=→"WALL", lessThan=(none), greaterThan=(none), everyFrame=true)
    Wait(time=0.45, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dissipate
    on WALL → Wall Impact
  }

  state Dissipate {
    PlayParticleEmitter(gameObject=var "Dribble", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ball End")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Terrain Checker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Dissipate End
  }

  state Diss R {
    RecycleSelf()
  }

  state Wall Impact {
    SetCollider(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Particle Emitter")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Checker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Break
    on CANCEL → Fireball Safe Return
  }

  state Break R {
    RecycleSelf()
  }

  state Init {
    ActivateGameObject(gameObject=var "White Light", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Velocity", float2=0, tolerance=0, equal=→"RIGHT", lessThan=→"LEFT", greaterThan=→"RIGHT", everyFrame=false)
    on LEFT → L
    on RIGHT → R
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Terrain Checker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Init
  }

  state L {
    FindChild(gameObject=Self, childName="Dribble L", storeResult=var "Dribble")
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state R {
    FindChild(gameObject=Self, childName="Dribble R", storeResult=var "Dribble")
    on FINISHED → Idle
  }

  state Fireball Safe Return {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Set Damage {
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ball")
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=15, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_19", isTrue=(none), isFalse=→"FINISHED")
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=20, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=var "Y Scale", zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1.3, y=1.6, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state Break {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "White Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Break R
  }

  state Dissipate End {
    StopParticleEmitter(gameObject=var "Dribble")
    StopParticleEmitter(gameObject=var "Particle Emitter")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Diss R
  }

  var Crt X Velocity: float = 0
  var Velocity: float = 0
  var X Scale: float = 0
  var Y Scale: float = 0
  var Collider: gameObject = <null>
  var Dribble: gameObject = <null>
  var Particle Emitter: gameObject = Fireball/Particle System
  var Terrain Checker: gameObject = Fireball/Terrain Checker
  var Wall Impact: gameObject = Fireball/Wall Impact
  var White Light: gameObject = Fireball/white_light
}
