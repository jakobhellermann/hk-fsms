# Control Dream Particles
# 1 location(s):
#   Waterways_15 / Dung Defender_Sleep

fsm Control Dream Particles {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindChild(gameObject=Self, childName="Base Glow", storeResult=var "Base Glow")
    FindChild(gameObject=Self, childName="Idle Pt", storeResult=var "Idle Pt")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDreamNail", isTrue=→"GLOW", isFalse=→"NO")
    on GLOW → Glow
    on NO → Off
  }

  state Off {
  }

  state Glow {
    PlayParticleEmitter(gameObject=var "Idle Pt", emit=0)
    SendEventByName(eventTarget=GameObject(var "Base Glow"), sendEvent="UP", delay=0, everyFrame=false)
  }

  var Base Glow: gameObject = <null>
  var Dream Enter: gameObject = <null>
  var Idle Pt: gameObject = <null>
}
