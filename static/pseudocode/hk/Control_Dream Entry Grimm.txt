# Control
# 1 location(s):
#   Grimm_Nightmare / Dream Entry Grimm

fsm Control {
  start Pause

  state Take Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    NextFrameEvent(sendEvent=→"FINISHED")
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=MaxHealth())
    SendEventByName(eventTarget=BroadcastAll, sendEvent="UPDATE BLUE HEALTH", delay=0, everyFrame=false)
    on FINISHED → Facing
  }

  state Start Fade {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Arrive Anim
  }

  state Return Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AcceptInput())
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SetMeshRenderer(gameObject=var "Cutscene", active=false)
  }

  state Arrive Anim {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SetWieldingLantern(false))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    FindChild(gameObject=Self, childName="Knight Dream Arrival", storeResult=var "Cutscene")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Cutscene", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Cutscene", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Cutscene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait for Anim
  }

  state Facing {
    BoolTest(boolVariable=var "Hero Faces left", isTrue=(none), isFalse=(none), everyFrame=false)
    on L → Left
    on R → Right
  }

  state Left {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Start Fade
  }

  state Right {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    on FINISHED → Start Fade
  }

  state Wait for Anim {
    Tk2dWatchAnimationEvents(gameObject=var "Cutscene", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Return Control
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    on FINISHED → Door Entry
  }

  state Door Entry {
    StringCompare(stringVariable=var "Door Entry", compareTo="", equalEvent=→"ACTIVATE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetEntryGateName", parameters=[], storeResult=var "Door Entered")
    StringCompare(stringVariable=var "Door Entry", compareTo=var "Door Entered", equalEvent=→"ACTIVATE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on ACTIVATE → Take Control
  }

  var Hero Faces left: bool = true
  var Door Entry: string = ""

  var X Scale: float = 0
  var Door Entered: string = ""
  var Hero Pos: vector3 = (0, 0, 0)
  var Cutscene: gameObject = <null>
}
