# Grub Control
# 1 location(s):
#   sharedassets33.assets / Grub Mimic Bottle Col/Grub Mimic Top

fsm Grub Control {
  start Pause
  on FREE → Free  // from any state
  on IN BOTTLE → Idle  // from any state

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")  // disabled
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    WaitRandom(timeMin=3, timeMax=6, finishEvent=→"CRY", realTime=false)
    on ENTER → Hero Close
    on FREE → Free
    on CRY → Cry
  }

  state Hero Close {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[grub_alert, grub_idle_or_look_1, grub_idle_or_look_2], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Cry")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Cry Turn", resetFrame=true, everyFrame=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Sad Wait
    on FREE → Free
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    FindChild(gameObject=var "Self", childName="Dig Dust", storeResult=var "Dig Dust")
    FindChild(gameObject=Self, childName="Grub Mimic", storeResult=var "Mimic")
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Free {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Instant Trans?
  }

  state Sad Sound {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[grub_sad_1, grub_sad_2, grub_sad_3], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Idle Wait
  }

  state Sad Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sad Sound
  }

  state Idle Wait {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Cry {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grub_sad_idle_01, Grub_sad_idle_02, Grub_sad_idle_03, Grub_sad_idle_04], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Idle
  }

  state Instant Trans? {
    SendRandomEvent(events=[→"CRY", →"TRANSFORM"], weights=[1, 1], delay=0)
    on TRANSFORM → Transform
    on CRY → Fake Cry
  }

  state Transform {
    ActivateGameObject(gameObject=var "Mimic", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DestroySelf(detachChildren=true)  // disabled
    SetMeshRenderer(gameObject=Self, active=false)
    GameObjectIsNull(gameObject=var "Mimic", isNull=→"ACTIVATE", isNotNull=(none), storeResult=(unset), everyFrame=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on ACTIVATE → Set Active
  }

  state Fake Cry {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Freed Fake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Transform
  }

  state Idle No Bottle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Free
  }

  state Set Active {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    DestroySelf(detachChildren=false)
  }

  var Activated: bool = false
  var Dig Dust: gameObject = <null>
  var Hero: gameObject = <null>
  var Mimic: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
