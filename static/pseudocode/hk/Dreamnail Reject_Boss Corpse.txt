# Dreamnail Reject
# 1 location(s):
#   Room_Final_Boss_Core / Boss Control/Hollow Knight Boss/Boss Corpse

fsm Dreamnail Reject {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Attack Pt", storeResult=var "Attack Pt")
    FindChild(gameObject=Self, childName="Dream Impact", storeResult=var "Dream Impact")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"DREAM HIT", storeCollider=var "Collider")
    on DREAM HIT → Tink
  }

  state Tink {
    GetTag(gameObject=var "Collider", storeResult=var "Tag", everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=sword_hit_reject (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashArmoured())
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP INSTANT", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)  // disabled
    SetParticleEmission(gameObject=var "Attack Pt", emission=true)  // disabled
    ActivateGameObject(gameObject=var "Dream Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventByScale(gameObject=var "Hero", eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on L → Recoil L
    on R → Recoil R
    on CANCEL → Cancel Pause
  }

  state End emission {
    SetParticleEmission(gameObject=var "Attack Pt", emission=false)  // disabled
    on FINISHED → Idle
  }

  state Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End emission
  }

  state Recoil L {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilLeft())  // disabled
    Translate(gameObject=var "Hero", vector=(unset), x=-1.5, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Pause
  }

  state Recoil R {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilRight())  // disabled
    Translate(gameObject=var "Hero", vector=(unset), x=1.5, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Pause
  }

  state Cancel Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Tag: string = ""
  var Attack Pt: gameObject = Dream Dialogue/Active Pt (sharedassets6.assets)
  var Collider: gameObject = <null>
  var Dream Impact: gameObject = <null>
  var Self: gameObject = <null>
}
