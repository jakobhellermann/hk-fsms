# Grub Mimic
# 1 location(s):
#   sharedassets33.assets / Grub Mimic Bottle Col/Grub Mimic Top/Grub Mimic

fsm Grub Mimic {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Burst Point", storeResult=var "Burst Point")
    FindChild(gameObject=Self, childName="Slob Point", storeResult=var "Slob Point")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Burst
  }

  state Burst {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=grub_mimic_transform_start)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Burst 2
  }

  state Scream {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD LAUGH", delay=0.5, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetActive(false))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=grub_mimic_initial_roar)
    on FINISHED → Start Run
    on TOOK DAMAGE → Start Run
  }

  state Start Run {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="IdleToRun", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetScale(gameObject=var "Parent", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Chase
  }

  state Chase {
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=15, acceleration=0.5, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToRun", resetFrame=true, everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on TOOK DAMAGE → Hit Slow
    on WALL → Hop
  }

  state Hit Slow {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)  // disabled
    on FINISHED → Chase
  }

  state Hop {
    RandomFloat(min=10, max=10, storeResult=var "Hop Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Hop Speed", everyFrame=false)
    on FINISHED → Chase
  }

  state Burst 2 {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=grub_mimic_transform_end)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Burst Point", position=(unset), spawnMin=20, spawnMax=20, speedMin=15, speedMax=25, angleMin=40, angleMax=140, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    on FINISHED → Scream
  }

  var Hop Speed: float = 0
  var Burst Point: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Slob Point: gameObject = <null>
}
