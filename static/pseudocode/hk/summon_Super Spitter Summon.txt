# summon
# 4 location(s):
#   Deepnest_East_04 / Spawn 1/Super Spitter Summon
#   Deepnest_East_04 / Spawn 1/Super Spitter Summon (1)
#   Deepnest_East_11 / Spawn 1/Super Spitter Summon (1)
#   Deepnest_East_11 / Spawn 1/Super Spitter Summon

fsm summon {
  start Init
  on SPAWN → Check Activated  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    CreateObject(gameObject=Super Spitter (sharedassets33.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Buzzer", networkInstantiate=false, networkGroup=0)
    SetMeshRenderer(gameObject=Self, active=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Disable
  }

  state Enter {
    SetMeshRenderer(gameObject=Self, active=true)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    RandomFloat(min=0.75, max=1.2, storeResult=var "Tween Time")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=8, z=16.996, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    ColorInterpolate(colors=[(0, 0, 0, 1), (1, 1, 1, 1)], time=1, storeColor=var "Colour", finishEvent=(none), realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -8, -16.99), time=var "Tween Time", delay=0, speed=(unset), easeType=easeOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Summon
  }

  state Summon {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Buzzer", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPositionToObject(gameObject=var "Buzzer", targetObject=var "Self", xOffset=0, yOffset=0, zOffset=0)
    SetScale(gameObject=var "Buzzer", vector=(unset), x=var "Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFsmBool(gameObject=var "Buzzer", fsmName="spitter", variableName="startAlert", setValue=true, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Disable {
    GetPersistentBool(target=var "Buzzer", storeValue=var "Activated")
    ActivateGameObject(gameObject=var "Buzzer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Check Activated {
    BoolTest(boolVariable=var "Activated", isTrue=→"CANCEL", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Enter
    on CANCEL → Inert
  }

  state Inert {
  }

  var Chooser: float = 0
  var Scale: float = 0
  var Tween Time: float = 0
  var Activated: bool = false
  var Colour: color = (0, 0, 0, 1)
  var Battle Scene: gameObject = <null>
  var Buzzer: gameObject = <null>
  var Self: gameObject = <null>
}
