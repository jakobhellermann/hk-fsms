# Slam Effects
# 2 location(s):
#   Fungus2_29 / Mushroom Brawler
#   Fungus2_30 / Mushroom Brawler

fsm Slam Effects {
  start Init

  state Init {
    on FINISHED → Check
  }

  state Check {
    Tk2dSpriteGetId(gameObject=var "Cap Hit", spriteID=var "Sprite ID", everyframe=true)
    IntCompare(integer1=var "Sprite ID", integer2=36, equal=→"SMASH", lessThan=(none), greaterThan=(none), everyFrame=true)
    on SMASH → Smash
  }

  state Smash {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_brawler_head_bang (sharedassets169.assets))
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Check
  }

  var Sprite ID: int = 0
  var Cap Hit: gameObject = Mushroom Brawler/Smash Hits
  var Slam Effect: gameObject = Mushroom Brawler/Slam Effect
}
