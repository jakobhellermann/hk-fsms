# Blink
# 1 location(s):
#   Room_Queen / Queen

fsm Blink {
  start Init

  state Init {
    on AWAKE → Pause
  }

  state Pause {
    WaitRandom(timeMin=4, timeMax=10, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Blink
  }

  state Blink {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Blink", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Pause
  }
}
