# Fade
# 3 location(s):
#   Dream_Guardian_Hegemol / Dreamer Appear orb
#   Dream_Guardian_Lurien / Dreamer Appear orb
#   Dream_Guardian_Monomon / Dreamer Appear orb

fsm Fade {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Dream Orb Get S", storeResult=var "Dream Orb Get S")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Pause
  }

  state Pause {
    Wait(time=0.85, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade
  }

  state Fade {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=1.7, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    StopParticleEmitter(gameObject=var "Dream Orb Get S")
  }

  var Dream Orb Get S: gameObject = <null>
}
