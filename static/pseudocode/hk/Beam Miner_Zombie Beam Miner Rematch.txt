    SetParent(gameObject=var "Beam Ball", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Beam", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)

    SetParent(gameObject=var "Beam Point L", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)

  }

    PlayParticleEmitter(gameObject=var "End Particle", emit=0)

    StopParticleEmitter(gameObject=var "Impact Particle")

    SendEventByName(eventTarget=GameObject(var "Beam Glow"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Check Facing {
    SetFsmBool(gameObject=Self, fsmName="walk_left_and_right", variableName="Facing Right", setValue=false, everyFrame=false)

    on FINISHED → Battle Init
  }

  state Roar Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar Start")
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Action (resources.assets), transitionTime=1)
    on FINISHED → Laser Shoot
    on GG BOSS → Laser Shoot?
  }

  state Roar {

    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=radiance_laser_burst (sharedassets261.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

  }

  state Jump Antic {

  }

  state Aim Jump {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)

  }

  state Rising {

  }

  state Falling {

    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", everyFrame=true)
