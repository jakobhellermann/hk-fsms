// component disabled: this FSM does not run until something enables it
fsm Mush Roller {
  start Init

  state Init {
    SetFloatValue(floatVariable=var "RollAnticSpeedCrt", floatValue=var "Roll Antic Speed", everyFrame=false)
    SetFloatValue(floatVariable=var "RollSpeedCrt", floatValue=var "Roll Speed", everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Sleep
  }

  state Sleep {
    BoolTest(boolVariable=var "Starts Awake", isTrue=→"START AWAKE", isFalse=(none), everyFrame=false)
    on WAKE → Emerge Flip?
    on START AWAKE → Init Idle
    on TOOK DAMAGE → Emerge Flip?
  }

  state Emerge {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)  // disabled
    AudioPlayRandom(gameObject=var "Self", audioClips=[fungus_mushroom_hit (sharedassets56.assets)], weights=[1], pitchMin=0.75, pitchMax=0.95)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mushroom_Roller_startle_01 (sharedassets171.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Emerge Cooldown
  }

  state Idle {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetBoxCollider2DSize(gameObject1=Self, width=1.94, height=2.77, offsetX=0, offsetY=-0.43)
    BoolTest(boolVariable=var "Hero Behind", isTrue=→"BEHIND", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Hero In Front", isTrue=→"FRONT", isFalse=(none), everyFrame=true)
    on BEHIND → Turn
    on FRONT → Attack Choice
  }

  state Turn {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "RollAnticSpeedCrt", floatValue=var "Roll Antic Speed", everyFrame=false)
    SetFloatValue(floatVariable=var "RollSpeedCrt", floatValue=var "Roll Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "RollAnticSpeedCrt", multiplyBy=var "X Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "RollSpeedCrt", multiplyBy=var "X Scale", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Attack Choice {
    AudioStop(gameObject=Self)
    SetBoxCollider2DSize(gameObject1=Self, width=1.94, height=1.9, offsetX=0, offsetY=-0.865)
    SendRandomEventV2(events=[→"JUMP", →"ROLL"], weights=[1, 1], trackingInts=[var "Jump Count", var "Roll Count"], eventMax=[3, 3])
    on ROLL → Roll Antic
    on JUMP → Jump Antic
  }

  state Roll Antic {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_roller_roll_up (sharedassets171.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "RollAnticSpeedCrt", y=(unset), everyFrame=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mushroom_Roller_startle_01 (sharedassets171.assets), Mushroom_Roller_startle_02 (sharedassets171.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Roll
  }

  state Roll {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_roller_charge (sharedassets171.assets))
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "RollSpeedCrt", y=(unset), everyFrame=true)
    Wait(time=var "Roll Time", finishEvent=→"WAIT", realTime=false)
    FlingObjectsFromGlobalPoolTime(gameObject=Gas Projectile Quick (sharedassets171.assets), spawnPoint=var "Self", position=(0, -0.3, 0), frequency=0.05, spawnMin=2, spawnMax=2, speedMin=0, speedMax=0, angleMin=0, angleMax=0, originVariationX=0, originVariationY=0)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on WAIT → Roll Last
    on WALL → Roll End
  }

  state Roll End {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Turn?
  }

  state Cooldown {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Emerge Cooldown {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Idle
  }

  state Roll Last {
    Wait(time=0.05, finishEvent=→"WAIT", realTime=false)
    on WAIT → Roll End
  }

  state Turn? {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetBoxCollider2DSize(gameObject1=Self, width=1.94, height=2.77, offsetX=0, offsetY=-0.43)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Turn To Idle", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "RollAnticSpeedCrt", floatValue=var "Roll Antic Speed", everyFrame=false)
    SetFloatValue(floatVariable=var "RollSpeedCrt", floatValue=var "Roll Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "RollAnticSpeedCrt", multiplyBy=var "X Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "RollSpeedCrt", multiplyBy=var "X Scale", everyFrame=false)
    on FINISHED → Cooldown
  }

  state Jump Antic {
    BoolTest(boolVariable=var "Roof Check", isTrue=→"ROLL", isFalse=(none), everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_roller_roll_up (sharedassets171.assets))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mushroom_Roller_startle_01 (sharedassets171.assets), Mushroom_Roller_startle_02 (sharedassets171.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=10, everyFrame=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Jump
    on ROLL → Roll Antic
  }

  state Jump {
    SetFloatValue(floatVariable=var "JumpSpeed", floatValue=var "RollSpeedCrt", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mushroom_roller_charge (sharedassets171.assets))
    FloatMultiply(floatVariable=var "JumpSpeed", multiplyBy=0.8, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "JumpSpeed", y=28, everyFrame=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Air
  }

  state Land {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Turn?
  }

  state In Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "JumpSpeed", y=(unset), everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    FlingObjectsFromGlobalPoolTime(gameObject=Gas Projectile Quick (sharedassets171.assets), spawnPoint=var "Self", position=(0, -0.3, 0), frequency=0.05, spawnMin=2, spawnMax=2, speedMin=0, speedMax=0, angleMin=0, angleMax=0, originVariationX=0, originVariationY=0)
    on LAND → Land
  }

  state Emerge Flip? {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Hero Is Right", leftBool=(unset), everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Facing Right", greaterThanBool=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Is Right"], boolStates=[true, true], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing Right", var "Hero Is Right"], boolStates=[false, false], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "RollAnticSpeedCrt", floatValue=var "Roll Antic Speed", everyFrame=false)
    SetFloatValue(floatVariable=var "RollSpeedCrt", floatValue=var "Roll Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "RollAnticSpeedCrt", multiplyBy=var "X Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "RollSpeedCrt", multiplyBy=var "X Scale", everyFrame=false)
    on FINISHED → Emerge
  }

  state Init Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetBoxCollider2DSize(gameObject1=Self, width=1.94, height=2.77, offsetX=0, offsetY=-0.43)
    BoolTest(boolVariable=var "Hero Behind", isTrue=→"BEHIND", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Hero In Front", isTrue=→"FRONT", isFalse=(none), everyFrame=true)
    on FRONT → Attack Choice
    on BEHIND → Turn
  }

  var Roll Antic Speed: float = 5
  var Roll Speed: float = -15
  var Roll Time: float = 0.75
  var Starts Awake: bool = false

  var JumpSpeed: float = 0
  var RollAnticSpeedCrt: float = 0
  var RollSpeedCrt: float = 0
  var X Scale: float = 0
  var Jump Count: int = 0
  var Roll Count: int = 0
  var Facing Right: bool = false
  var Hero Behind: bool = false
  var Hero In Front: bool = true
  var Hero Is Right: bool = false
  var Roof Check: bool = false
  var Self: gameObject = <null>
}
