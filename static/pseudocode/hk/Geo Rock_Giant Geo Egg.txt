# Geo Rock
# 1 location(s):
#   Deepnest_East_17 / Giant Geo Egg

// uses template: geo_rock_huge
fsm Geo Rock {
  start Pause Frame Init

  state Idle {
    RandomFloat(min=1.5, max=4, storeResult=var "Waiter")
    Wait(time=var "Waiter", finishEvent=→"WAIT", realTime=false)
    ReceivedDamage(collideTag="", sendEvent=→"NAIL HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on WAIT → Gleam
    on NAIL HIT → Check If Nail
  }

  state Gleam {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Gleam Anim")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    ReceivedDamage(collideTag="", sendEvent=→"NAIL HIT", fsmName="damages_enemy", storeGameObject=var "", ignoreAcid=false, ignoreWater=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on NAIL HIT → Check If Nail
  }

  state Check If Nail {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0], sendEvent=[→"TRUE"], everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0, 2], sendEvent=[→"TRUE", →"TRUE"], everyFrame=false)
    on TRUE → Hit
    on FINISHED → Idle
  }

  state Hit {
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    FlingObjectsFromGlobalPool(gameObject=Geo Med (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Per Hit", spawnMax=var "Geo Per Hit", speedMin=23, speedMax=30, angleMin=80, angleMax=100, originVariationX=2, originVariationY=1, FSM="", FSMEvent="")
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"NAIL HIT", lessThan=→"NAIL HIT", greaterThan=(none), everyFrame=false)
    on FINISHED → Check Direction
    on NAIL HIT → Pause Frame
    on SKIP → Check Direction
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    IntCompare(integer1=var "Hits", integer2=1, equal=(none), lessThan=→"BROKEN", greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Recoil Speed Neg", floatValue=var "Recoil Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Recoil Speed Neg", multiplyBy=-1, everyFrame=false)
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    GetRotation(gameObject=var "Self", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    on FINISHED → Idle
    on BROKEN → Broken
  }

  state Check Direction {
    AudioPlayRandom(gameObject=var "Self", audioClips=[geo_rock_hit_1 (sharedassets6.assets), geo_rock_hit_2 (sharedassets6.assets), geo_rock_hit_3 (sharedassets6.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on RIGHT → Hit Right
    on UP → Hit Down
    on LEFT → Hit Left
    on DOWN → Hit Down
  }

  state Destroy {
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Geo Med (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Final Payout", spawnMax=var "Final Payout", speedMin=23, speedMax=30, angleMin=80, angleMax=100, originVariationX=2, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=8, spawnMax=12, speedMin=20, speedMax=28, angleMin=60, angleMax=120, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Broken Anim")
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med Down R (resources.assets), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(-72.5, -180, -180), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (resources.assets), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(-72.5, -180, -180), storeObject=var "")
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Death Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=breakable_wall_death (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  state Hit Right {
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false)
    FloatCompare(float1=var "Angle", float2=270, tolerance=10, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (resources.assets), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(0, 90, 270), storeObject=var "")
    on FINISHED → Return Left
  }

  state Hit Left {
    FloatCompare(float1=var "Angle", float2=90, tolerance=10, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (resources.assets), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(180, 90, 270), storeObject=var "")
    on FINISHED → Return Left
  }

  state Return Left {
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit Down {
    FloatCompare(float1=var "Angle", float2=180, tolerance=10, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (resources.assets), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(-72.5, -180, -180), storeObject=var "")
    on FINISHED → Return Left
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Destroy
  }

  state Broken {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Broken Anim")
    SetCollider(gameObject=Self, active=false)
  }

  state Pause Frame Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Death Effects", storeResult=var "Death Effects")
    on FINISHED → Initiate
  }

  var Final Payout: int = 4
  var Geo Per Hit: int = 8
  var Hits: int = 10
  var Broken Anim: string = "Broken 1"
  var Gleam Anim: string = "Gleam 1"

  var Angle: float = 0
  var Attack Direction: float = 0
  var Recoil Speed: float = 1
  var Recoil Speed Neg: float = 0
  var Recoil Time: float = 0.1
  var Start X: float = 0
  var Start Y: float = 0
  var Waiter: float = 0
  var Attack Type: int = 0
  var Damager: gameObject = <null>
  var Death Effects: gameObject = <null>
  var Self: gameObject = <null>
}
