# Battle Control
# 1 location(s):
#   White_Palace_02 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Royal Gaurd", storeResult=var "Royal Gaurd")
    on FINISHED → Idle
  }

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="CameraLockAreaB", storeResult=var "CamLock")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ACTIVATE → Activate
    on ENTER → Start
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    ActivateGameObject(gameObject=var "Royal Gaurd", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Start {
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on ROYAL GUARD KILLED → End Pause
  }

  state End Pause {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var CamLock: gameObject = <null>
  var Royal Gaurd: gameObject = <null>
  var Self: gameObject = <null>
}
