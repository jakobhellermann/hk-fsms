# Detect Bounce
# 2 location(s):
#   sharedassets32.assets / Colosseum_Armoured_Roller/Bouncer
#   sharedassets495.assets / Colosseum_Armoured_Roller R/Bouncer

fsm Detect Bounce {
  start Init

  state Detect {
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=(unset), sendEvent=→"COLLIDE", storeCollider=var "Collider", storeForce=(unset))
    on COLLIDE → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="BOUNCE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="BOUNCE", delay=0, everyFrame=false)
    on FINISHED → Detect
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Detect
  }

  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
}
