fsm Door Control {
  start Init

  state Init {
    on DOORS OPEN → Open
  }

  state Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lift Door Open")
    on DOORS CLOSE → Close
  }

  state Close {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Lift Door Close")
    on DOORS OPEN → Open
  }
}
