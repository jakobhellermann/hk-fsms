# Control
# 1 location(s):
#   sharedassets395.assets / Dream Orb Get

fsm Control {
  start Fade Out

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM NAIL HUD", delay=0, everyFrame=false)
    on FINISHED → Increment Orbs
  }

  state Increment Orbs {
    PlayerDataIntAdd(gameObject=var "GameManager", intName="dreamOrbs", amount=var "Orb Amount")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Orb In
  }

  state Orb In {
    PlayParticleEmitter(gameObject=var "Vanish Get Pt", emit=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get
  }

  state Get {
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Attack Pt", emit=0)
    StopParticleEmitter(gameObject=var "Vanish Get Pt")
    SetPosition(gameObject=var "White Flash Get", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Attack Pt", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "White Flash Get", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM ORB COLLECT", delay=0, everyFrame=false)
    on FINISHED → Fade Out
  }

  state Fade Out {
    FindGameObject(objectName="Dream Fall Catcher", withTag="Untagged", store=var "Dream Fall Catcher")
    ActivateGameObject(gameObject=var "Dream Fall Catcher", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0.9, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → New Scene
  }

  state New Scene {
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    GetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", storeValue=var "Return Scene")
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=[var "Return Scene", "door_dreamReturn", 0], storeResult=(unset var))  // disabled
    BeginSceneTransition(sceneName=var "Return Scene", entryGateName="door_dreamReturn", entryDelay=0, visualization=Dream, preventCameraFadeOut=true)
  }

  var Orb Amount: int = 50
  var Return Scene: string = "Crossroads_10"
  var Hero Pos: vector3 = (0, 0, 0)
  var Attack Pt: gameObject = <null>
  var Dream Fall Catcher: gameObject = <null>
  var Vanish Burst Pt: gameObject = Dream Orb Get/Vanish Burst Pt
  var Vanish Get Pt: gameObject = Dream Orb Get/Vanish Get Pt
  var White Flash Get: gameObject = Dream Orb Get/White Flash Get
}
