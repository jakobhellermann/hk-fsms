# Attack Range Detect
# 1 location(s):
#   Fungus1_20_v02 / _Enemies/Buzzer Range

// uses template: alert_range
fsm Attack Range Detect {
  start Initialise

  state Initialise {
    NextFrameEvent(sendEvent=→"WAIT")
    on WAIT → Detecting
  }

  state Detecting {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"COLLIDE ENTER", storeCollider=var "Hero")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"COLLIDE ENTER", storeCollider=var "Hero")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"COLLIDE EXIT", storeCollider=var "")
    on COLLIDE ENTER → In Range
    on COLLIDE EXIT → Out of Range
  }

  state In Range {
    SetBoolValue(boolVariable=var "In Alert Range", boolValue=true, everyFrame=false)
    on FINISHED → Detecting
  }

  state Out of Range {
    SetBoolValue(boolVariable=var "In Alert Range", boolValue=false, everyFrame=false)
    on FINISHED → Detecting
  }

  var In Alert Range: bool = false
  var Hero: gameObject = <null>
}
