fsm Recoil {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "Damager")
    on NAIL HIT → Recoil
  }

  state Recoil {
    SendMessageV2(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SoulGain(), everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="magnitudeMult", storeValue=var "Attack Magnitude", everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    BeginRecoil(target=Self, attackDirection=var "Attack Direction", attackType=var "Attack Type", attackMagnitude=var "Attack Magnitude")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashInfected())
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=enemy_damage (resources.assets), pitchMin=1.15, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Attack Direction: float = 0
  var Attack Magnitude: float = 0
  var Attack Type: int = 0
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
