# Battle Control
# 1 location(s):
#   Ruins1_09 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Init
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Started
    on KILLED → Enemy Killed
  }

  state Init {
    FindGameObject(objectName="Mage", withTag="Untagged", store=var "Mage")  // disabled
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activated
  }

  state Activated {
    ActivateGameObject(gameObject=var "Mage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Started {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="WAKE", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Action Only (resources.assets), transitionTime=0)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="WATERWAYS", delay=0, everyFrame=false)
    on KILLED → Enemy Killed
  }

  state Enemy Killed {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=2)
  }

  var Mage: gameObject = Battle Scene/Mage

  var Activated: bool = false
  var Self: gameObject = <null>
}
