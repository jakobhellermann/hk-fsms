# Conversation Control
# 1 location(s):
#   Room_nailmaster / NM Mato NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Stop Anim  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Convo Choice {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metNailmasterMato", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasCyclone", isTrue=(none), isFalse=→"REOFFER")
    BoolTest(boolVariable=var "Bowing", isTrue=→"BOWING", isFalse=(none), everyFrame=false)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasUpwardSlash", falseBool="matoConvoOro", isTrue=→"ORO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasDashSlash", falseBool="matoConvoSheo", isTrue=→"SHEO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gotCharm_26", falseBool="matoConvoSly", isTrue=→"SLY", isFalse=(none))
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"GREET", everyFrame=false)
    on MEET → Meet
    on BOWING → Bowing
    on GREET → Greet
    on REPEAT → Repeat
    on ORO → Oro
    on SHEO → Sheo
    on SLY → Sly
    on REOFFER → Reoffer
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Cinematic", store=var "Cinematic Player")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="LookUpEnd", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    BoolTest(boolVariable=var "Bowing", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest Talk")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="NM_MATO", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Meet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_MEET", "Nailmasters"], storeResult=(unset var))
    SetPlayerDataBool(boolName="metNailmasterMato", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_Look_to_talk_03 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down2
  }

  state Greet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_GREET", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_talk_01 (sharedassets13.assets), Nailmaster_talk_02 (sharedassets13.assets), Nailmaster_talk_03 (sharedassets13.assets), Nailmaster_talk_06 (sharedassets13.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Activate Sit
  }

  state Stop Anim {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Get Msg {
    FindGameObject(objectName="Arrow Prompt(Clone)", withTag="Untagged", store=var "Arrow Prompt")
    Translate(gameObject=Self, vector=(unset), x=3, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Arrow Prompt", vector=(unset), x=2, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetPlayerDataBool(boolName="hasCyclone", value=true)
    SetPlayerDataBool(boolName="hasNailArt", value=true)
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=_0001_charge_cyclone (sharedassets13.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="NA Cyclone", everyFrame=false)
    on GET ITEM MSG END → Fade Back
  }

  state Box Down2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up YN
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest Look")
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["NAILMASTER_FREE", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → Roar
  }

  state Decline Pause {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Box Up 2
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Decline
  }

  state Decline {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_REFUSE", "Nailmasters"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_talk_01 (sharedassets13.assets), Nailmaster_talk_02 (sharedassets13.assets), Nailmaster_talk_03 (sharedassets13.assets), Nailmaster_talk_06 (sharedassets13.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk End
  }

  state Roar {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar")
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar", networkInstantiate=false, networkGroup=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_roar_02 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[misc_rumble_impact (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Fade
  }

  state Fade {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Scene Blanker
  }

  state Cinematic {
    SendEventByName(eventTarget=GameObject(var "Roar"), sendEvent="END", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Cinematic Player", delivery=SendMessage, options=1, functionCall=TriggerStartVideo())
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    on CINEMATIC END → Cinematic End
  }

  state Fade Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Battle Idle")
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER OFF", delay=0, everyFrame=false)
    on FINISHED → Box Up 3
  }

  state Box Up 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Taught Msg
  }

  state Taught Msg {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_TAUGHT", "Nailmasters"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_talk_05 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down 2
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bow
  }

  state Bow {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bow")
    SetBoolValue(boolVariable=var "Bowing", boolValue=true, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_Bama_02 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Talk Finish
  }

  state Bowing {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_BOW", "Nailmasters"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_Bama_02 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk Finish
  }

  state Talk End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rest Talk End")
    on FINISHED → Talk Finish
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_REPEAT", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_talk_05 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Activate Sit
  }

  state Oro {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_ORO", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="matoConvoOro", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_talk_01 (sharedassets13.assets), Nailmaster_talk_02 (sharedassets13.assets), Nailmaster_talk_03 (sharedassets13.assets), Nailmaster_talk_06 (sharedassets13.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk End
  }

  state Sheo {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_SHEO", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="matoConvoSheo", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_talk_01 (sharedassets13.assets), Nailmaster_talk_02 (sharedassets13.assets), Nailmaster_talk_03 (sharedassets13.assets), Nailmaster_talk_06 (sharedassets13.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk End
  }

  state Sly {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_SLY", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="matoConvoSly", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_talk_01 (sharedassets13.assets), Nailmaster_talk_02 (sharedassets13.assets), Nailmaster_talk_03 (sharedassets13.assets), Nailmaster_talk_06 (sharedassets13.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk End
  }

  state Reoffer {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MATO_ART", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_Look_to_talk_02 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down2
  }

  state Cinematic End {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get Msg
  }

  state Activate Sit {
    SetCollider(gameObject=Self, active=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SIT START", delay=0, everyFrame=false)
    on FINISHED → Talk End
  }

  state Scene Blanker {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER ON", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Cinematic
  }

  var Bowing: bool = false
  var Equipped Sly Charm: bool = false
  var Got Dash Slash: bool = false
  var Got Great Slash: bool = false
  var Oro Convo: bool = false
  var Sheo Convo: bool = false
  var Sly Convo: bool = false
  var Spoken: bool = false
  var Visited Crossroads: bool = false
  var Area Title: gameObject = <null>
  var Arrow Prompt: gameObject = <null>
  var Cinematic Player: gameObject = <null>
  var Roar: gameObject = <null>
  var Self: gameObject = <null>
  var UI Msg: gameObject = <null>
  var Voice Player: gameObject = <null>
}
