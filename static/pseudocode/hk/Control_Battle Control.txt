# Control
# 1 location(s):
#   GG_Mighty_Zote / Battle Control

fsm Control {
  start Init

  state Init {
    on FINISHED → Start Pause
  }

  state Start Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open Cage
  }

  state Open Cage {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "First Zote", audioClip=col_cage_open (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "First Zote", audioClip=Zote_ceiling_drop_02 (sharedassets399.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → First Zote
  }

  state First Zote {
    ActivateGameObject(gameObject=var "First Zote", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Music Init
  }

  state Start Warriors {
    SendEventByName(eventTarget=GameObject(var "Dormant Warriors"), sendEvent="START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ZOTE WARRIOR KILLED", delay=0, everyFrame=false)  // disabled
    IntCompare(integer1=var "Kills", integer2=2, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on NEXT → Add Zoteling
  }

  state Add Warrior 2 {
    SetFsmInt(gameObject=var "Dormant Warriors", fsmName="Spawn", variableName="Active Max", setValue=2, everyFrame=false)
    IntCompare(integer1=var "Kills", integer2=5, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on NEXT → Add Fatties
  }

  state Add Fatties {
    IntCompare(integer1=var "Kills", integer2=15, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    SetParent(gameObject=var "Fat 1", parent=var "Dormant Warriors", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Fat 2", parent=var "Dormant Warriors", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Fat 3", parent=var "Dormant Warriors", resetLocalPosition=false, resetLocalRotation=false)
    on NEXT → Add Turret
  }

  state Add Zoteling {
    SendEventByName(eventTarget=GameObject(var "Zoteling 1"), sendEvent="SPAWN", delay=0, everyFrame=false)
    IntCompare(integer1=var "Kills", integer2=8, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on NEXT → Add Fatties
  }

  state Add Turret {
    IntCompare(integer1=var "Kills", integer2=20, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Extra Zotes"), sendEvent="START", delay=0, everyFrame=false)
    on NEXT → Add Tall
  }

  state Add Tall {
    IntCompare(integer1=var "Kills", integer2=25, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    SetParent(gameObject=var "Tall 1", parent=var "Dormant Warriors", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Tall 2", parent=var "Dormant Warriors", resetLocalPosition=false, resetLocalRotation=false)
    on NEXT → Add Thwomp
  }

  state Add Balloon {
    IntCompare(integer1=var "Kills", integer2=35, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Balloon"), sendEvent="BALLOON SPAWN", delay=0, everyFrame=false)
    on NEXT → Add Fluke
  }

  state Add Thwomp {
    IntCompare(integer1=var "Kills", integer2=30, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    SetParent(gameObject=var "Thwomp 1", parent=var "Extra Zotes", resetLocalPosition=false, resetLocalRotation=false)
    on NEXT → Add Balloon
  }

  state Add Zoteling 2 {
    SendEventByName(eventTarget=GameObject(var "Zoteling 2"), sendEvent="SPAWN", delay=0, everyFrame=false)
    IntCompare(integer1=var "Kills", integer2=57, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on NEXT → Menu Unlock
  }

  state Add Ghost {
    IntCompare(integer1=var "Kills", integer2=52, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Zote Salubra"), sendEvent="START", delay=0, everyFrame=false)
    on NEXT → Add Zoteling 2
  }

  state Add Fluke {
    IntCompare(integer1=var "Kills", integer2=40, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    SetParent(gameObject=var "Zote Fluke", parent=var "Extra Zotes", resetLocalPosition=false, resetLocalRotation=false)
    on NEXT → Add Ghost
  }

  state Add Warrior 3 {
    SetFsmInt(gameObject=var "Dormant Warriors", fsmName="Spawn", variableName="Active Max", setValue=3, everyFrame=false)
    IntCompare(integer1=var "Kills", integer2=50, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
  }

  state Menu Unlock {
    MenuStyleUnlockAction(unlockKey="eternalOrdealMenu")
    SetPlayerDataBool(boolName="ordealAchieved", value=true)
    IntCompare(integer1=var "Kills", integer2=80, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on FINISHED → Add Warrior 3
  }

  state Music Init {
    AudioPlaySimple(gameObject=var "Music Initial", volume=1, oneShotClip=<null>)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    on FIRST ZOTE END → Title Zoom 3
  }

  state Title Zoom 1 {
    ActivateGameObject(gameObject=var "Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetScale(gameObject=var "Title", vector=(unset), x=0.57, y=0.57, z=(unset), everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=col_moving_wall_impact_into_place (sharedassets32.assets), pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Title Zoom 2
  }

  state Title Zoom 2 {
    SetScale(gameObject=var "Title", vector=(unset), x=1.41, y=1.41, z=(unset), everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=col_moving_wall_impact_into_place (sharedassets32.assets), pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Title Zoom 3
  }

  state Title Zoom 3 {
    SetScale(gameObject=var "Title", vector=(unset), x=2.38, y=2.38, z=(unset), everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=explosion_1 (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Crash Out
  }

  state Crash Out {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=col_moving_wall_impact_into_place (sharedassets32.assets), pitchMin=1.15, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ZOTE WARRIOR KILLED", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

    SendEventByName(eventTarget=BroadcastAll, sendEvent="CORPSE END", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=var "Music", volume=0.8, oneShotClip=<null>)
    Wait(time=2.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    on FINISHED → Start Warriors
  }

  state Music {
    AudioPlaySimple(gameObject=var "Music", volume=0.8, oneShotClip=<null>)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Crash Out
  }

  var Balloon: gameObject = Battle Control/Zote Balloon Ordeal
  var Dormant Warriors: gameObject = Battle Control/Dormant Warriors
  var Extra Zotes: gameObject = Battle Control/Extra Zotes
  var Fat 1: gameObject = Battle Control/Fat Zotes/Zote Crew Fat (1)
  var Fat 2: gameObject = Battle Control/Fat Zotes/Zote Crew Fat (2)
  var Fat 3: gameObject = Battle Control/Fat Zotes/Zote Crew Fat (3)
  var First Zote: gameObject = Battle Control/First Zote
  var Music: gameObject = Battle Control/Music
  var Music Initial: gameObject = Battle Control/Music Initial
  var Tall 1: gameObject = Battle Control/Tall Zotes/Zote Crew Tall
  var Tall 2: gameObject = Battle Control/Tall Zotes/Zote Crew Tall (1)
  var Thwomp 1: gameObject = Battle Control/Zote Thwomp
  var Title: gameObject = Battle Control/Title
  var Zote Fluke: gameObject = Battle Control/Zote Fluke
  var Zote Salubra: gameObject = Battle Control/Zote Salubra

  var Zoteling 2: gameObject = Battle Control/Zotelings/Ordeal Zoteling (1)

  var Kills: int = 0
}
