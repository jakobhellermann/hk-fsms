fsm Shop Anim {
  start Init

  state Init {
    on SHOP START → Start
  }

  state Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Inside TalkL")
    on SHOP STOP → Stop
  }

  state Stop {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Inside L")
    on SHOP START → Start
  }
}
