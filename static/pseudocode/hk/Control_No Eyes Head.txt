# Control
# 4 location(s):
#   Fungus1_35 / No Eyes Head
#   GG_Ghost_No_Eyes / No Eyes Head
#   GG_Ghost_No_Eyes_V / No Eyes Head
#   sharedassets161.assets / No Eyes Head

fsm Control {
  start Init
  on GHOST DEATH → Dissipate  // from any state

  state Init {
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetCollider(gameObject=Self, active=true)
    SetFloatValue(floatVariable=var "Self X Pos", floatValue=30, everyFrame=false)
    SetMeshRenderer(gameObject=var "White Light", active=true)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.1, space=0, everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Idle Pt", emit=0)
    FloatCompare(float1=var "Speed", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on L → L
    on R → R
  }

  state L {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Travel
  }

  state R {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Travel
  }

  state Travel {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X Pos", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X Pos", float2=72.16, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"END", everyFrame=true)  // disabled
    FloatCompare(float1=var "Self X Pos", float2=23.71, tolerance=0, equal=(none), lessThan=→"END", greaterThan=(none), everyFrame=true)  // disabled
    SetFloatValue(floatVariable=var "Tween Speed", floatValue=var "Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Tween Speed", multiplyBy=0.75, everyFrame=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 5, 0), time=(unset), delay=0, speed=var "Tween Speed", easeType=easeInOutSine, loopType=pingPong, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=8, finishEvent=→"END", realTime=false)
    on FINISHED → 
    on END → Dissipate
  }

  state Dissipate {
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "White Light", active=false)
    StopParticleEmitter(gameObject=var "Idle Pt")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Self X Pos: float = 0
  var Speed: float = 9
  var Tween Speed: float = 0
  var Idle Pt: gameObject = No Eyes Head/Idle Pt
  var White Light: gameObject = No Eyes Head/white_light
}
