# Control
# 1 location(s):
#   sharedassets343.assets / Parasite Balloon Spawner

fsm Control {
  start Initiate
  on ZERO HP → Death  // from any state

  state Chase - In Sight {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Chase")
    RandomFloat(min=3.5, max=6.5, storeResult=var "Speed Max")
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=var "Speed Max", accelerationForce=10, offsetX=0, offsetY=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToChase", resetFrame=false, everyFrame=true)
    on WAIT → 
    on STOP SPAWN → Leave
  }

  state Initiate {
    SetBoolValue(boolVariable=var "Hero Saver", boolValue=false, everyFrame=false)
    RandomFloat(min=1, max=1.25, storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Spawn Wait
    on STOP SPAWN → Recycle
  }

  state Spawn Wait {
    SetCollider(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Spawn Pause
    on STOP SPAWN → Recycle
  }

  state Spawn Pause {
    WaitRandom(timeMin=0.1, timeMax=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Saver
    on STOP SPAWN → Recycle
  }

  state Check Saver {
    BoolTest(boolVariable=var "Hero Saver", isTrue=(none), isFalse=→"FINISHED", everyFrame=true)
    on FINISHED → Spawn
    on CANCEL → Spawn Pause
    on STOP SPAWN → Recycle
  }

  state Spawn {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_balloon_appear (sharedassets34.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Activate
    on STOP SPAWN → Leave
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Chase - In Sight
    on STOP SPAWN → Leave
  }

  state Recycle {
    RecycleSelf()
  }

  state Leave {
    SetCollider(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  state Death {
    PlayParticleEmitter(gameObject=var "Pt Death", emit=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCircleCollider(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Death 2
  }

  state Death 2 {
    Wait(time=1.2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  var Scale: float = 0
  var Speed Max: float = 0
  var Hero Saver: bool = false
  var Pt Death: gameObject = Parasite Balloon Spawner/Pt Death
  var Self: gameObject = Parasite Balloon Spawner
}
