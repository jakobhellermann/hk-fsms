# Update Text
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Inv
#   resources.assets / _GameCameras/HudCamera/Inventory/Inv

fsm Update Text {
  start Init
  on PANE RESET → Pane Reset  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Text Name", storeResult=var "Text Name")
    FindChild(gameObject=Self, childName="Text Desc", storeResult=var "Text Desc")
    FindChild(gameObject=Self, childName="Text Desc Low", storeResult=var "Text Desc Low")
    on FINISHED → Idle
  }

  state Idle {
    on UPDATE TEXT → State 1
  }

  state Pane Reset {
    SendEventByName(eventTarget=GameObject(var "Text Desc"), sendEvent="CHANGE DOWN", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Text Name"), sendEvent="CHANGE DOWN", delay=0, everyFrame=false)  // disabled
    SetTextMeshProText(gameObject=var "Text Name", textString="")
    SetTextMeshProText(gameObject=var "Text Desc", textString="")
    on FINISHED → Idle
  }

  state Change Text {
    GetLanguageString(sheetName=var "Sheet", convName=var "Convo Name", storeValue=var "Name String")
    GetLanguageString(sheetName=var "Sheet", convName=var "Convo Desc", storeValue=var "Desc String")
    SetTextMeshProText(gameObject=var "Text Name", textString=var "Name String")
    SetTextMeshProText(gameObject=var "Text Desc", textString=var "Desc String")
    SetTextMeshProText(gameObject=var "Text Desc Low", textString=var "Desc String")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SET BUTTON CONTROL2", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state State 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Change Text
  }

  var Convo Desc: string = ""
  var Convo Name: string = ""
  var Desc String: string = ""
  var Name String: string = ""
  var Sheet: string = "UI"
  var Text Desc: gameObject = <null>
  var Text Desc Low: gameObject = <null>
  var Text Name: gameObject = <null>
}
