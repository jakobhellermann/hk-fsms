# Control
# 1 location(s):
#   RestingGrounds_08 / Ghost Battle Revek

fsm Control {
  start Init
  on REVEK DEJECTED → Dejected  // from any state
  on REVEK KILLED → Killed  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetMeshRenderer(gameObject=Self, active=false)
    FindChild(gameObject=Self, childName="Idle Pt", storeResult=var "Idle Pt")
    FindChild(gameObject=Self, childName="Slash Pt", storeResult=var "Slash Pt")
    FindChild(gameObject=Self, childName="Slash Hit", storeResult=var "Slash Hit")
    FindChild(gameObject=Self, childName="Flare In", storeResult=var "Flare In")
    FindChild(gameObject=Self, childName="A Dash Effect", storeResult=var "Dash Burst")
    FindChild(gameObject=Self, childName="Dreamnail Detect", storeResult=var "Dreamnail Detect")
    SetCollider(gameObject=var "Slash Hit", active=false)
    SetCollider(gameObject=var "Dreamnail Detect", active=false)
    on FINISHED → Ghost Check Pause
  }

  state Dormant {
    on GHOST NPC KILLED → Appear Pause
  }

  state Appear Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Angle
  }

  state Set Angle {
    SetFloatValue(floatVariable=var "X Distance", floatValue=12, everyFrame=false)
    RandomFloat(min=0, max=4, storeResult=var "Y Distance")
    FloatSubtract(floatVariable=var "X Distance", subtract=var "Y Distance", everyFrame=false, perSecond=false)
    RandomlyFlipFloat(storeResult=var "X Distance")
    RandomlyFlipFloat(storeResult=var "Y Distance")
    on FINISHED → Slash Tele In
  }

  state Slash Tele In {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_knight_teleport (sharedassets6.assets), pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    SetCollider(gameObject=var "Dreamnail Detect", active=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Target", y=var "Y Target", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    PlayParticleEmitter(gameObject=var "Idle Pt", emit=0)  // disabled
    StopParticleEmitter(gameObject=var "Slash Pt")  // disabled
    SetParticleEmission(gameObject=var "Idle Pt", emission=true)
    SetParticleEmission(gameObject=var "Slash Pt", emission=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    FloatOperator(float1=var "Hero X", float2=var "X Distance", operation=Add, storeResult=var "X Target", everyFrame=true)
    FloatOperator(float1=var "Hero Y", float2=var "Y Distance", operation=Add, storeResult=var "Y Target", everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Target", y=var "Y Target", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Tele", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=true)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Flare In", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Slash Idle
  }

  state Slash Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Slash Antic")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Hero X", float2=var "X Distance", operation=Add, storeResult=var "X Target", everyFrame=true)
    FloatOperator(float1=var "Hero Y", float2=var "Y Distance", operation=Add, storeResult=var "Y Target", everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Target", y=var "Y Target", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slash Antic
    on DREAM IMPACT → Hit
    on TAKE DAMAGE → Hit
  }

  state Slash {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Col_miner_attack_04 (sharedassets31.assets), Col_miner_attack_03 (sharedassets31.assets), Col_miner_attack_02 (sharedassets31.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_horizontal_dash_dash (sharedassets243.assets), pitchMin=1.1, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    FireAtTarget(gameObject=Self, target=var "Hero", speed=var "Speed", position=(0, -0.5, 0), spread=0, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9)
    FaceAngle(gameObject=Self, angleOffset=var "Angle Adjust", everyFrame=true)
    SetParticleEmission(gameObject=var "Idle Pt", emission=false)
    SetParticleEmission(gameObject=var "Slash Pt", emission=true)
    SetCollider(gameObject=var "Slash Hit", active=true)
    ActivateGameObject(gameObject=var "Dash Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Slash Tele Out
    on TAKE DAMAGE → Hit
    on DREAM IMPACT → Hit
  }

  state Slash Tele Out {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=var "Dreamnail Detect", active=false)
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Flare In", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_knight_teleport (sharedassets6.assets), pitchMin=1.1, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetParticleEmission(gameObject=var "Slash Pt", emission=false)
    SetCollider(gameObject=var "Slash Hit", active=false)
    on FINISHED → Attack Pause
  }

  state Attack Pause {
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Set Angle
  }

  state L or R? {
    SetFloatValue(floatVariable=var "Angle Adjust", floatValue=0, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=→"FINISHED", space=0)
    SetFloatValue(floatVariable=var "Angle Adjust", floatValue=180, everyFrame=false)
    on FINISHED → Slash
  }

  state Slash Antic {
    FireAtTarget(gameObject=Self, target=var "Hero", speed=-20, position=(0, -0.5, 0), spread=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[hornet_parry_prepare (sharedassets243.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → L or R?
    on DREAM IMPACT → Hit
    on TAKE DAMAGE → Hit
  }

  state Hit {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=var "Dreamnail Detect", active=false)
    FlingObjectsFromGlobalPool(gameObject=Slash Effect Ghost1 (resources.assets), spawnPoint=var "Self", position=var "Effect Origin", spawnMin=6, spawnMax=6, speedMin=20, speedMax=35, angleMin=0, angleMax=360, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Slash Effect Ghost2 (resources.assets), spawnPoint=var "Self", position=var "Effect Origin", spawnMin=6, spawnMax=6, speedMin=20, speedMax=35, angleMin=0, angleMax=360, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Flare In", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_knight_teleport (sharedassets6.assets), pitchMin=1.1, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_damage (resources.assets), pitchMin=1.1, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Col_miner_death_01 (sharedassets32.assets), col_miner_death_03 (sharedassets32.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetParticleEmission(gameObject=var "Slash Pt", emission=false)
    SetParticleEmission(gameObject=var "Idle Pt", emission=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Remove Slash
  }

  state Damaged Pause {
    WaitRandom(timeMin=4.5, timeMax=4.5, finishEvent=→"END", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on END → Set Angle
  }

  state Remove Slash {
    SetCollider(gameObject=var "Slash Hit", active=false)
    on FINISHED → Damaged Pause
  }

  state Dejected {
  }

  state Killed {
  }

  state Ghost Check Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → None Remaining?
  }

  state None Remaining? {
    IntCompare(integer1=var "Ghosts Alive", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REVEK DEJECTED", delay=0, everyFrame=false)
    on FINISHED → Dormant
  }

  var Speed: float = 180
  var Effect Origin: vector3 = (0, 0, 0)

  var Angle Adjust: float = 0
  var Hero X: float = 0
  var Hero Y: float = 0
  var X Distance: float = 0
  var X Target: float = 0
  var Y Distance: float = 0
  var Y Target: float = 0
  var Ghosts Alive: int = 0
  var Dash Burst: gameObject = <null>
  var Dreamnail Detect: gameObject = <null>
  var Flare In: gameObject = <null>
  var Idle Pt: gameObject = <null>
  var Self: gameObject = <null>
  var Slash Hit: gameObject = <null>
  var Slash Pt: gameObject = <null>
}
