# Music Region
# 1 location(s):
#   Ruins1_24 / Music Region (1)

// uses template: music_regionv2
fsm Music Region {
  start Pause

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Mines Delay
  }

  state Enter {
    TransitionToAudioSnapshot(snapshot=var "Enter Music Snapshot", transitionTime=var "Enter Transition Time")
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent=var "Enter Track Event", delay=0, everyFrame=false)  // disabled
    ApplyMusicCue(musicCue=var "Enter Music Cue", delayTime=0, transitionTime=0)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ENTER", delay=0, everyFrame=false)
    on EXIT → Exit
  }

  state Exit {
    TransitionToAudioSnapshot(snapshot=var "Exit Music Snapshot", transitionTime=var "Exit Transition Time")
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent=var "Exit Track Event", delay=0, everyFrame=false)  // disabled
    ApplyMusicCue(musicCue=var "Exit Music Cue", delayTime=0, transitionTime=0)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Mines Delay
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Mines Delay {
    BoolTest(boolVariable=var "Mines Delay", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Wait(time=2.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dirtmouth Check
  }

  state Dirtmouth Check {
    BoolTest(boolVariable=var "Dirtmouth", isTrue=(none), isFalse=→"NON", everyFrame=false)
    GetCurrentMusicCueName(musicCueName=var "Music Cue")
    GetFsmInt(gameObject=var "AudioManager", fsmName="Load Music Into Groups", variableName="Current Track", storeValue=var "Current Track", everyFrame=false)  // disabled
    IntCompare(integer1=var "Current Track", integer2=1, equal=→"NON", lessThan=(none), greaterThan=(none), everyFrame=false)  // disabled
    StringCompare(stringVariable=var "Music Cue", compareTo="Dirtmouth", equalEvent=→"NON", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    TransitionToAudioSnapshot(snapshot=var "Enter Music Snapshot", transitionTime=0)
    ApplyMusicCue(musicCue=var "Enter Music Cue", delayTime=var "Enter Delay", transitionTime=var "Enter Transition Time")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on NON → Enter
    on EXIT → Exit
  }

  state State 1 {
  }

  var Enter Transition Time: float = 3
  var Exit Transition Time: float = 3
  var Dirtmouth: bool = false
  var Mines Delay: bool = false
  var Enter Track Event: string = "SOUL SOCIETY"
  var Exit Track Event: string = "SOUL SOCIETY"
  var Enter Music Cue: object = <null>
  var Enter Music Snapshot: object = Normal (resources.assets)
  var Exit Music Cue: object = <null>
  var Exit Music Snapshot: object = Normal (resources.assets)

  var Enter Delay: float = 0
  var Current Track: int = 0
  var Music Cue: string = ""
}
