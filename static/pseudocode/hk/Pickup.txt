# Pickup
# 1 location(s):
#   Ruins1_24 / Quake Real Parent/Quake Pickup

fsm Pickup {
  start Idle

  state Idle {
    GetPlayerDataInt(gameObject=var "GameManager", intName="quakeLevel", storeValue=var "Quake Level")
    IntCompare(integer1=var "Quake Level", integer2=0, equal=(none), lessThan=(none), greaterThan=→"DORMANT", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="mageLordDefeated", isTrue=→"INSTANT", isFalse=(none))
    on QUAKE PICKUP START → Wait
    on DORMANT → Dormant
    on INSTANT → Instant Activate
  }

  state Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Cascade Start
  }

  state Cascade Start {
    PlayParticleEmitter(gameObject=var "Cascade", emit=0)
    Wait(time=1.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Arrival Start
  }

  state Arrival Start {
    PlayParticleEmitter(gameObject=var "Arrival", emit=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=2)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on FINISHED → Cascade End
  }

  state Cascade End {
    StopParticleEmitter(gameObject=var "Cascade")
    Wait(time=1.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Appear
  }

  state Appear {
    AudioStop(gameObject=Self)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Quake Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmitter(gameObject=var "Arrival")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=shaman_spell (sharedassets43.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=White Wave Default R (sharedassets101.assets), spawnPoint=var "Quake Item", position=(unset), rotation=(unset), storeObject=var "")
  }

  state Dormant {
  }

  state Instant Activate {
    ActivateGameObject(gameObject=var "Quake Item", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Quake Level: int = 0
  var Arrival: gameObject = Quake Real Parent/Quake Pickup/Arrival
  var Cascade: gameObject = Quake Real Parent/Quake Pickup/Cascade
  var Quake Item: gameObject = Quake Real Parent/Quake Pickup/Quake Item
  var White Flash: gameObject = Quake Real Parent/Quake Pickup/White Flash
}
