# Sit Up
# 2 location(s):
#   Fungus2_15_boss / Mantis Battle/Mantis Lord Throne 1
#   Fungus2_15_boss_defeated / Mantis Lord Throne 1

// uses template: mlord_sit_up
fsm Sit Up {
  start Idle
  on MLORD BOW → Bow  // from any state

  state Idle {
    on MLORD CHALLENGE ENTER → Enter
  }

  state Enter {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Look Up")
    on MLORD CHALLENGE EXIT → Exit Pause
  }

  state Exit {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Look Down")
    on MLORD CHALLENGE ENTER → Enter
  }

  state Exit Pause {
    WaitRandom(timeMin=0.1, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "No Pause", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    on FINISHED → Exit
    on MLORD CHALLENGE ENTER → Enter
  }

  state Bow {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Throne Bow")
  }

  var No Pause: bool = true
}
