# Crowd Control
# 1 location(s):
#   Grimm_Main_Tent_boss / Crowd

fsm Crowd Control {
  start Init
  on CROWD CLAP → Clap  // from any state
  on CROWD STILL → Still  // from any state
  on CROWD IDLE → Idle  // from any state
  on CROWD CLAP DELAY → Delay  // from any state
  on BATTLE END → Go Silent  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Clap", storeResult=var "Clap")
    FindChild(gameObject=Self, childName="Idle", storeResult=var "Idle")
    FindChild(gameObject=Self, childName="Still", storeResult=var "Still")
    FindChild(gameObject=Self, childName="Murmur", storeResult=var "Murmur")
    on FINISHED → Idle
  }

  state Idle {
    ActivateGameObject(gameObject=var "Idle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Clap", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Still", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=1)
  }

  state Clap {
    Wait(time=3.6, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Clap", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Still", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=0.5)
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    AudioStop(gameObject=var "Murmur")
    on FINISHED → Play Murmur
  }

  state Still {
    ActivateGameObject(gameObject=var "Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Clap", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Still", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStop(gameObject=var "Murmur")
  }

  state Delay {
    Wait(time=1.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Clap
  }

  state Play Murmur {
    AudioPlaySimple(gameObject=var "Murmur", volume=1, oneShotClip=<null>)
    on FINISHED → Idle
  }

  state Go Silent {
    FadeAudio(gameObject=var "Murmur", startVolume=1, endVolume=0, time=1)
  }

  var Clap: gameObject = <null>
  var Idle: gameObject = <null>
  var Murmur: gameObject = <null>
  var Still: gameObject = <null>
}
