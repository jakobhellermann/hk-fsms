# Check if midwarp or completed
# 1 location(s):
#   Abyss_05 / Dusk Knight/Dream Enter 2

fsm Check if midwarp or completed {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="whitePalaceMidWarp", isTrue=→"ACTIVE", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotKingFragment", isTrue=→"ACTIVE", isFalse=(none))
    on FINISHED → Inactive
    on ACTIVE → Active
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Active {
  }
}
