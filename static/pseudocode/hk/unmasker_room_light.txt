# unmasker
# 2 location(s):
#   Ruins1_05b / _Scenery/room_light parent/room_light
#   Ruins1_05b / Shop Mask

// uses template: mask_and_unmask_reverse
fsm unmasker {
  start Pause

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Fade Out
  }

  state Fade Out {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.5, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="Player", collideLayer="", sendEvent=→"LEAVE", storeCollider=var "")
    on LEAVE → Fade In
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Fade In {
    iTweenFadeTo(gameObject=Self, id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=0.5, delay=0, easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Fade Out
  }
}
