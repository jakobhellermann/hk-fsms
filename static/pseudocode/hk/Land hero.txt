# Land hero
# 1 location(s):
#   Ruins1_24_boss / Quake Fake Parent/Knight Get Quake Fake

fsm Land hero {
  start Idle

  state Idle {
    on LAND HERO → Fall
  }

  state Fall {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=var "Hero", frame=9)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=29.5, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Dreamer Land")
  }

  var Hero Y: float = 0
}
