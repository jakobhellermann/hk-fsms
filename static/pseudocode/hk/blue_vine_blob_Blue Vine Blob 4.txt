# blue_vine_blob
# 1 location(s):
#   Abyss_06_Core / Blue Plinth/Blue Vine Blob 4

// uses template: blue_vine_blob
fsm blue_vine_blob {
  start Init
  on LIGHT ALL → Light All  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Flash", storeResult=var "Flash")
    FindChild(gameObject=Self, childName="Lit", storeResult=var "Lit")
    FindChild(gameObject=Self, childName="Wave", storeResult=var "Wave")
    GetParent(gameObject=Self, storeResult=var "Plinth")
    on FINISHED → Idle
  }

  state Idle {
    on LIGHT → Check
  }

  state Light {
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=var "Lit", active=true)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=flashHealBlue())
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Wave", audioClips=[soul_totem_awake (sharedassets6.assets)], weights=[1], pitchMin=0.8, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
  }

  state Check {
    GetFsmInt(gameObject=var "Plinth", fsmName="Control", variableName="Current Number", storeValue=var "Current Number", everyFrame=false)
    IntCompare(integer1=var "Current Number", integer2=var "Sequence Number", equal=→"FINISHED", lessThan=→"CANCEL", greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → Idle
    on FINISHED → Light
  }

  state Light All {
    SetSpriteRenderer(gameObject=var "Lit", active=true)
  }

  var Sequence Number: int = 4

  var Current Number: int = 0
  var Flash: gameObject = <null>
  var Lit: gameObject = <null>
  var Plinth: gameObject = <null>
  var Wave: gameObject = <null>
}
