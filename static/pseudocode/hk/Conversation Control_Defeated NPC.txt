# Conversation Control
# 1 location(s):
#   Grimm_Main_Tent / Grimm Holder/Defeated NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite")
    FindChild(gameObject=Self, childName="Tele In Effects", storeResult=var "Tele In Effects")
    FindChild(gameObject=Self, childName="Spotlight", storeResult=var "Spotlight")
    FindChild(gameObject=Self, childName="Grimm Hand", storeResult=var "Grimm Hand")
    FindChild(gameObject=Self, childName="Get Flame", storeResult=var "Get Flame")
    FindChild(gameObject=Self, childName="Get Flash", storeResult=var "Get Flash")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Grimm Scene", storeResult=var "Grimm Scene")
    FindChild(gameObject=var "Grimm Scene", childName="CamLock Battle", storeResult=var "CamLock Battle")
    FindChild(gameObject=var "Grimm Scene", childName="CamLock Normal", storeResult=var "CamLock Normal")
    on FINISHED → Inactive
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="LookBack")
    on FINISHED → Greet
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Greet {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GRIMM_DEFEAT_1", "CP2"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[Grimm_talk_05 (sharedassets391.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down 2
  }

  state Inactive {
    on GRIMM DEFEATED → Appear
  }

  state Appear {
    ActivateGameObject(gameObject=var "Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=grimm_teleport_in (sharedassets391.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Tele In", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Tele In Effects"), sendEvent="PLAY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Spotlight"), sendEvent="UP", delay=0, everyFrame=false)
    FaceObject(objectA=var "Sprite", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on FINISHED → Bow Pause
  }

  state Bow Pause {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="LookBack")
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bow
  }

  state Bow {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Bow")
    Wait(time=1.25, finishEvent=→"WAIT", realTime=false)
    FaceObject(objectA=var "Sprite", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_jump (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=accordion_sting_03 (sharedassets391.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=var "Voice Player")
    on WAIT → Applause
  }

  state Applause {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD CLAP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"WAIT", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=accordion_sting_03 (sharedassets391.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)  // disabled
    on WAIT → Bow Return
  }

  state Bow Return {
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Bow Return", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → NPC Activated
  }

  state NPC Activated {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Idle")
    SetCollider(gameObject=Self, active=true)
    FindGameObject(objectName="", withTag="Grimmchild", store=var "Grimmchild")
    SetPosition(gameObject=var "Grimmchild", vector=(unset), x=80.6, y=9.8, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Sprite", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN NO HUD", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hand Out
  }

  state Hand Out {
    ActivateGameObject(gameObject=var "Grimm Hand", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Grimm Hand", animLibName="", clipName="Hand Out")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Click
  }

  state Click {
    Tk2dPlayAnimation(gameObject=var "Grimm Hand", animLibName="", clipName="Hand Click")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=Grimm_click (sharedassets391.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Flame Ring
  }

  state Flame Ring {
    SetParent(gameObject=var "Get Flame", parent=var "Grimmchild", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=var "Get Flame", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    SetFsmGameObject(gameObject=var "Get Flame", fsmName="Follow Grimmchild", variableName="Grimmchild", setValue=var "Grimmchild", everyFrame=false)
    ActivateGameObject(gameObject=var "Get Flame", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Get Flame", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=nightmare_lantern_powering_up (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Level Up To 3
  }

  state Level Up To 3 {
    SendMessage(gameObject=var "Grimmchild", delivery=SendMessage, options=1, functionCall=FlashGrimmflame())
    SetPositionToObject(gameObject=var "Get Flash", targetObject=var "Grimmchild", xOffset=0, yOffset=0, zOffset=0)
    SetParent(gameObject=var "Get Flash", parent=var "Grimmchild", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    ActivateGameObject(gameObject=var "Get Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=nightmare_lantern_flame_burst (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPosition(gameObject=var "Get Flash", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    SetPlayerDataInt(intName="grimmChildLevel", value=3)
    SetPlayerDataInt(intName="flamesCollected", value=0)
    SetPlayerDataInt(intName="flamesRequired", value=3)
    SendEventByName(eventTarget=GameObject(var "Grimmchild"), sendEvent="LEVEL UP 2", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 5
  }

  state Box Up 5 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Grimm Hand", animLibName="", clipName="Hand In")
    on FINISHED → Levelup 3
  }

  state Levelup 3 {
    ActivateGameObject(gameObject=var "Grimm Hand", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GRIMM_DEFEAT_2", "CP2"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[Grimm_talk_02 (sharedassets391.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down 3
  }

  state Box Down 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN NO HUD", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")
    on FINISHED → Tele Out Anim
  }

  state Tele Out Anim {
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Tele Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Dark_room"), sendEvent="DOWN", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=grimm_teleport_out (sharedassets391.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Tele Poof
  }

  state Tele Poof {
    ActivateGameObject(gameObject=var "Grimm_teleport_effect2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Dark_room"), sendEvent="UP", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Sprite", active=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Tele In Effects"), sendEvent="PLAY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE END", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Final Pause
  }

  state End 2 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHEST APPEAR", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SetPlayerDataBool(boolName="disablePause", value=false)
    SetPlayerDataBool(boolName="killedGrimm", value=true)
    SetPlayerDataBool(boolName="newDataGrimm", value=true)
    SetPlayerDataInt(intName="killsGrimm", value=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ACCORDION START", delay=0, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "CamLock Normal", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Battle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["GRIMM"], storeResult=(unset var))
  }

  state Final Pause {
    Wait(time=2.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 6
  }

  state Box Up 6 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Levelup 4
  }

  state Levelup 4 {
    ActivateGameObject(gameObject=var "Grimm Hand", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GRIMM_DEFEAT_3", "CP2"], storeResult=(unset var))
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClips=[Grimm_parting_voice (sharedassets391.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down 4
  }

  state Box Down 4 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")  // disabled
    on FINISHED → Gates Open
  }

  state Gates Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End 2
  }

  var Dark_room: gameObject = Grimm Holder/Grimm Scene/Dark_room

  var Spoken: bool = false
  var CamLock Battle: gameObject = <null>
  var CamLock Normal: gameObject = <null>
  var Get Flame: gameObject = Flamebearer Spawn/Get Flame (sharedassets391.assets)
  var Get Flash: gameObject = Flamebearer Spawn/Get Flash (sharedassets391.assets)
  var Grimm Hand: gameObject = <null>
  var Grimm NPC: gameObject = Grimm Scene/Grimm NPC (sharedassets391.assets)
  var Grimm Scene: gameObject = <null>
  var Grimm_teleport_effect2: gameObject = Grimm Scene/Grimm_teleport_effect2 (sharedassets391.assets)
  var Grimmchild: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Spotlight: gameObject = <null>
  var Sprite: gameObject = <null>
  var Tele In Effects: gameObject = <null>
  var Voice Player: gameObject = <null>
}
