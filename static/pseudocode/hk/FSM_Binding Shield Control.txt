# FSM
# 5 location(s):
#   Deepnest_Spider_Town / Binding Shield Control
#   Abyss_06_Core / _Props/abyss_door/Binding Shield Control:1
#   Room_Queen / Binding Shield Control
#   Grimm_Main_Tent / Binding Shield Control
#   GG_Engine_Root / Binding Shield Control

fsm FSM {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on HERO CAST SPELL → Distance
  }

  state Flash {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on FINISHED → Idle
  }

  state Distance {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=10, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Flash
    on CANCEL → Idle
  }

  var Distance: float = 0
  var Self: gameObject = <null>
}
