# Check Opened
# 1 location(s):
#   Town / _Props/Stag_station

fsm Check Opened {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="open", storeResult=var "open")
    FindChild(gameObject=Self, childName="closed", storeResult=var "closed")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="openedTownBuilding", isTrue=→"OPENED", isFalse=→"CLOSED")
    on OPENED → Opened
    on CLOSED → Closed
  }

  state Opened {
    ActivateGameObject(gameObject=var "closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Closed {
    ActivateGameObject(gameObject=var "open", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var closed: gameObject = <null>
  var open: gameObject = <null>
}
