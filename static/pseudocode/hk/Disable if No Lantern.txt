# Disable if No Lantern
# 3 location(s):
#   Mines_33 / Toll Gate Machine
#   Mines_33 / Toll Gate Machine (1)
#   Deepnest_41 / Toll Gate Machine

fsm Disable if No Lantern {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasLantern", isTrue=(none), isFalse=→"DISABLE")
    on DISABLE → Disable
  }

  state Disable {
    SetCollider(gameObject=Self, active=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.2647059, 0.2647059, 0.2647059, 1), everyframe=false)
  }
}
