# Conversation Control
# 1 location(s):
#   Ruins1_04 / Nailsmith Cliff NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state
  on NPC CONTROL OFF → Killed  // from any state

  state Idle {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="nailsmithCliff", isTrue=(none), isFalse=→"INACTIVE")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="nailsmithKilled", isTrue=→"INACTIVE", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="nailsmithSpared", isTrue=→"INACTIVE", isFalse=(none))
    on CONVO START → Hero Anim
    on INACTIVE → Inactive
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    FindChild(gameObject=Self, childName="Splash Point", storeResult=var "Splash Point")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="NAILSMITH", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Kill Ready {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Outside Idle")
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Talk {
    SetPlayerDataBool(boolName="nailsmithKillSpeech", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["NAILSMITH_KILL", "Nailsmith"], storeResult=(unset var))
    on CONVO_FINISH → Anim End 2
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Convo Choice {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Outside Talk")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Nailmsith_greet (sharedassets14.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="nailsmithKillSpeech", isTrue=→"REPEAT", isFalse=→"TALK")
    on TALK → Talk
    on REPEAT → Repeat
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["NAILSMITH_KILL_REPEAT", "Nailsmith"], storeResult=(unset var))
    on CONVO_FINISH → Anim End 2
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Kill Ready
  }

  state Anim End 2 {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down 2
  }

  state Killed {
  }

  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var Self: gameObject = <null>
  var Splash Point: gameObject = <null>
  var Voice Player: gameObject = <null>
}
