# FSM
# 1 location(s):
#   Deepnest_East_Hornet / Battle Scene

fsm FSM {
  start Pause

  state Inactive {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on WAKE → Active
  }

  state Active {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on BATTLE END → Inactive
    on HORNET LEAVE → Open
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hornetOutskirtsDefeated", isTrue=→"ACTIVATE", isFalse=(none))
    on FINISHED → Inactive
    on ACTIVATE → Activate
  }

  state Open {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    SendMessageByTag(tag="WindyGrass", message="SetWindy")
  }

  var Self: gameObject = <null>
}
