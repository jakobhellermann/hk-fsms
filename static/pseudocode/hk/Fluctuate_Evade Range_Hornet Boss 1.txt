# Fluctuate
# 2 location(s):
#   Fungus1_04_boss / Hornet Boss 1/Evade Range
#   GG_Hornet_1 / Boss Holder/Hornet Boss 1/Evade Range

fsm Fluctuate {
  start Off

  state Off {
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → On
  }

  state On {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Off
    on EVADED → Off
  }
}
