# Conversation Control
# 1 location(s):
#   Dream_Nailcollection / Witch Control/Moth NPC

fsm Conversation Control {
  start Pause
  on RESET CONVO → Idle  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Witch", storeResult=var "Witch")
    FindChild(gameObject=var "Witch", childName="Particles", storeResult=var "W Particles")
    FindChild(gameObject=var "W Particles", childName="Trail Pt", storeResult=var "W Trail")
    FindChild(gameObject=var "W Particles", childName="Idle Pt", storeResult=var "W Idle")
    FindChild(gameObject=var "Parent", childName="Dream Nail Get", storeResult=var "DN Get")
    FindChild(gameObject=var "DN Get", childName="Dream Nail", storeResult=var "Dream Nail")
    on FINISHED → Set Convos
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN DREAM", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Moth Leave
  }

  state Hero Look {
    BoolTest(boolVariable=var "Hero Looks Up", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP DREAM", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Talk
  }

  state Talk {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="dreamNailConvo", isTrue=→"REPEAT", isFalse=(none))
    SetPlayerDataBool(boolName="dreamNailConvo", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["WITCH_DREAM1", "Dream Witch"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Moth_Seer_Silhouette_Talk_option_01 (sharedassets394.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down
    on REPEAT → Repeat
  }

  state Set Convos {
    BuildString(stringParts=[var "Ghost Name", "_TALK"], separator="", addToEnd=true, storeResult=var "Talk Convo", everyFrame=false)
    BuildString(stringParts=[var "Ghost Name", "_REPEAT"], separator="", addToEnd=true, storeResult=var "Repeat Convo", everyFrame=false)
    BuildString(stringParts=["GH_", var "Ghost Name", "_C"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    BuildString(stringParts=["GH_", var "Ghost Name", "_NC"], separator="", addToEnd=true, storeResult=var "Title Convo", everyFrame=false)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Moth Leave {
    AudioStop(gameObject=Self)
    SendEventByName(eventTarget=GameObject(var "Witch"), sendEvent="DOWN", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "W Trail")
    StopParticleEmitter(gameObject=var "W Idle")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Nail Appear
  }

  state Nail Appear {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_plant_hit (sharedassets43.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Dream Nail", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "DN Get", active=true)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → End
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["WITCH_DREAM_FALL", "Dream Witch"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Moth_Seer_Silhouette_Talk_option_01 (sharedassets394.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down
  }

  var Hero Looks Up: bool = true
  var Ghost Name: string = ""

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var History: int = 0
  var Met Elderbug: bool = false
  var Reintro Played: bool = false
  var Spoken: bool = false
  var Visited Crossroads: bool = false
  var mapCrossroads: bool = false
  var Repeat Convo: string = ""
  var Talk Convo: string = ""
  var Title Convo: string = ""
  var Area Title: gameObject = <null>
  var DN Get: gameObject = <null>
  var Dream Nail: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
  var W Idle: gameObject = <null>
  var W Particles: gameObject = <null>
  var W Trail: gameObject = <null>
  var Witch: gameObject = <null>
}
