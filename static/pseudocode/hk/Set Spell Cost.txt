# Set Spell Cost
# 2 location(s):
#   Knight_Pickup / Knight/Charm Effects
#   resources.assets / Knight:1/Charm Effects

fsm Set Spell Cost {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_33", isTrue=→"MAGE", isFalse=→"NORMAL")
    on NORMAL → Normal
    on MAGE → Mage
  }

  state Normal {
    SetFsmInt(gameObject=var "Parent", fsmName="Spell Control", variableName="MP Cost", setValue=33, everyFrame=false)
    on FINISHED → Idle
  }

  state Mage {
    SetFsmInt(gameObject=var "Parent", fsmName="Spell Control", variableName="MP Cost", setValue=24, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on CHARM INDICATOR CHECK → Check
  }

  var Parent: gameObject = <null>
}
