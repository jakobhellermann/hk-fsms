fsm PassToParent {
  start Inert
  on HIT LANDED → Pass Hit Landed  // from any state

  state Inert {
  }

  state Pass Hit Landed {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="HIT LANDED", delay=0, everyFrame=false)
    on FINISHED → Inert
  }

  var Parent: gameObject = <null>
}
