# Charm Tute Msg
# 1 location(s):
#   sharedassets6.assets / Charm Tutorial Msg

fsm Charm Tute Msg {
  start Init
  on CLOSE → Down  // from any state

  state Init {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHARM MSG", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOP REGION INACTIVE", delay=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=2.77, z=-20, space=0, everyFrame=false, lateUpdate=false)
    FindChild(gameObject=Self, childName="Text", storeResult=var "Text")
    FindChild(gameObject=Self, childName="Subtitle", storeResult=var "Subtitle")
    FindChild(gameObject=Self, childName="Title", storeResult=var "Title")
    FindChild(gameObject=Self, childName="Stop", storeResult=var "Stop")
    GetLanguageString(sheetName="Prompts", convName="CHARM_REMINDER", storeValue=var "Msg String")
    GetLanguageString(sheetName="Prompts", convName="CHARM_TUTE_TITLE", storeValue=var "Title Str")
    GetLanguageString(sheetName="Prompts", convName="CHARM_TUTE_SUB", storeValue=var "Subtitle Str")
    SetTextMeshProText(gameObject=var "Text", textString=var "Msg String")
    SetTextMeshProText(gameObject=var "Subtitle", textString=var "Subtitle Str")
    SetTextMeshProText(gameObject=var "Title", textString=var "Title Str")
    on FINISHED → Up
  }

  state Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop
  }

  state Down {
    ActivateGameObject(gameObject=var "Stop", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Regain Control?
  }

  state Destroy {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOP REGION ACTIVE", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)
  }

  state Stop {
    ActivateGameObject(gameObject=var "Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Input", isTrue=→"GET INPUT", isFalse=(none), everyFrame=false)
    on CLOSE → Down
    on GET INPUT → Wait For Input
  }

  state Wait For Input {
    ListenForInventory(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForCast(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on BUTTON DOWN → Down
  }

  state Regain Control? {
    BoolTest(boolVariable=var "Input", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    on FINISHED → Destroy
  }

  var Input: bool = false
  var Msg String: string = ""
  var Subtitle Str: string = ""
  var Title Str: string = ""
  var Shop Region: gameObject = <null>
  var Stop: gameObject = <null>
  var Subtitle: gameObject = <null>
  var Text: gameObject = <null>
  var Title: gameObject = <null>
}
