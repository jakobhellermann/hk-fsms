fsm Bee {
  start Pause

  state Chase {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Bee TurnToFly", resetFrame=true, everyFrame=true)
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=5, acceleration=0.1, targetSpread=1.5, spreadResetTimeMin=1, spreadResetTimeMax=2)
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"OUT OF RANGE", everyFrame=true)
    on OUT OF RANGE → Idle
  }

  state Idle {
    BoolTestMulti(boolVariables=[var "Friendly", var "Hero Range"], boolStates=[false, true], trueEvent=→"IN RANGE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    IdleBuzzV2(gameObject=Self, waitMin=1, waitMax=2, speedMax=3, accelerationMax=15, roamingRangeX=2, roamingRangeY=2, manualStartPos=(unset))
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Bee TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.4)
    on IN RANGE → Chase
    on TOOK DAMAGE → Unfriendly
  }

  state Pause {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    on FINISHED → Friendly?
  }

  state Friendly? {
    GetPlayerDataBool(boolName="equippedCharm_29", storeValue=var "Friendly")
    BoolTest(boolVariable=var "Friendly", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on FINISHED → Idle
  }

  state Unfriendly {
    SetBoolValue(boolVariable=var "Friendly", boolValue=false, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Chase
  }

  var Angle: float = 0
  var Speed: float = 0
  var Friendly: bool = false
  var Hero Range: bool = false
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
