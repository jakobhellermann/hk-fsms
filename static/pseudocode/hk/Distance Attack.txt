# Distance Attack
# 4 location(s):
#   Cliffs_02_boss / Warrior/Ghost Warrior Slug
#   GG_Ghost_Gorb / Warrior/Ghost Warrior Slug
#   GG_Ghost_Gorb_V / Warrior/Ghost Warrior Slug
#   sharedassets194.assets / Ghost Warrior Slug

fsm Distance Attack {
  start Init

  state Close {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=35, tolerance=0, equal=(none), lessThan=→"AWAY", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=78, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"AWAY", everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=25, tolerance=0, equal=(none), lessThan=→"AWAY", greaterThan=(none), everyFrame=true)
    on AWAY → To Away
  }

  state To Away {
    SetFsmBool(gameObject=Self, fsmName="Attacking", variableName="Hero Away", setValue=true, everyFrame=false)
    on FINISHED → Away
  }

  state Away {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatTestToBool(float1=var "Hero X", float2=35, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Above X Min", everyFrame=true)
    FloatTestToBool(float1=var "Hero X", float2=78, tolerance=0, equalBool=(unset), lessThanBool=var "Below X Max", greaterThanBool=(unset), everyFrame=true)
    FloatTestToBool(float1=var "Hero Y", float2=25, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Above Y Min", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Above X Min", var "Above Y Min", var "Below X Max"], sendEvent=→"CLOSE", storeResult=(unset), everyFrame=true)
    Wait(time=1, finishEvent=→"ATTACK", realTime=false)  // disabled
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"ATTACK", realTime=false)
    on CLOSE → To Close
    on ATTACK → Attack
  }

  state To Close {
    SetFsmBool(gameObject=Self, fsmName="Attacking", variableName="Hero Away", setValue=false, everyFrame=false)
    on FINISHED → Close
  }

  state Attack {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=50, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=-0.5, storeAngle=var "Angle", everyFrame=false)
    FloatSubtract(floatVariable=var "Angle", subtract=90, everyFrame=false, perSecond=false)
    SpawnObjectFromGlobalPool(gameObject=Shot Slug Spear (sharedassets161.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Shot")
    SetRotation(gameObject=var "Shot", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Away
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Close
  }

  var Angle: float = 0
  var Distance: float = 0
  var Hero X: float = 0
  var Hero Y: float = 0
  var Above X Min: bool = false
  var Above Y Min: bool = false
  var Below X Max: bool = false
  var Self: gameObject = <null>
  var Shot: gameObject = <null>
}
