# Death Detect
# 2 location(s):
#   Room_Colosseum_Gold / Colosseum Manager/Waves/Lobster Lancer/Entry Object/Lancer
#   GG_God_Tamer / Entry Object/Lancer

fsm Death Detect {
  start Idle

  state Idle {
    on LOBSTER KILLED → Set
    on ZERO HP → Set
  }

  state Set {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Death", setValue=true, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="recoil", variableName="Recoil per second", setValue=0, everyFrame=false)
  }
}
