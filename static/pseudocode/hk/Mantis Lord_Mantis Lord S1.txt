fsm Mantis Lord {

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Dash", storeResult=var "Pt Dash")
    FindChild(gameObject=Self, childName="Pt Land", storeResult=var "Pt Land")

    GetPosition(gameObject=Self, vector=var "Away Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)

  }

  state Idle {

    on SUB → Sub Idle
  }
