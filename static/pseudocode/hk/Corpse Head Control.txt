# Corpse Head Control
# 1 location(s):
#   resources.assets / Corpse Head

fsm Corpse Head Control {
  start Detect
  on ACID → Splash  // from any state

  state Detect {
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    on COLLIDE → Chip
  }

  state Chip {
    GetOwner(storeGameObject=var "Self")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Head Cracked")
    SpawnRandomObjects(gameObject=skull_chip1, spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=10, speedMax=20, angleMin=20, angleMax=160, originVariation=0)
    SpawnRandomObjects(gameObject=skull_chip2, spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=2, speedMin=10, speedMax=20, angleMin=20, angleMax=160, originVariation=0)
  }

  state Acid {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor, spawnPoint=var "Acid Bubble", audioClip=acid_splash, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Corpse Acid Steam, spawnPoint=var "Acid Bubble", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetGravity2dScale(gameObject=Self, gravityScale=0.05)
    DecelerateV2(gameObject=var "Acid Bubble", deceleration=0.65)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Acid Spore", emit=0)
    PlayParticleEmitter(gameObject=var "Corpse Flame", emit=0)
    on FINISHED → Sizzle
  }

  state Sizzle {
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    DecelerateV2(gameObject=var "Acid Bubble", deceleration=0.65)
    ColorInterpolate(colors=[(1, 1, 1, 1), (1, 1, 1, 0)], time=1, storeColor=var "Colour", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  state Finish {
    StopParticleEmitter(gameObject=var "Acid Spore")
    StopParticleEmitter(gameObject=var "Corpse Flame")
  }

  state Splash {
    GetOwner(storeGameObject=var "Acid Bubble")
    FindChild(gameObject=Self, childName="Spore Cloud", storeResult=var "Corpse Flame")
    FindChild(gameObject=Self, childName="Bub Cloud", storeResult=var "Acid Spore")
    SetPolygonCollider(gameObject=Self, active=false)
    CreateObject(gameObject=Acid Splash, spawnPoint=var "Acid Bubble", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    on FINISHED → Acid
  }

  var Colour: color = (1, 1, 1, 1)
  var Acid Bubble: gameObject = <null>
  var Acid Spore: gameObject = <null>
  var Corpse Flame: gameObject = <null>
  var Self: gameObject = <null>
}
