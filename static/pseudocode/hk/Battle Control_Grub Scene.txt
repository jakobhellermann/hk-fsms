# Battle Control
# 2 location(s):
#   Ruins2_03 / Grub Scene
#   Ruins2_03b / Grub Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"OPEN", lessThan=→"OPEN", greaterThan=(none), everyFrame=true)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="Great Shield Zombie", storeResult=var "Shield Zombie")
    GetOwner(storeGameObject=var "Self")
    SetFsmGameObject(gameObject=var "Shield Zombie", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Self", everyFrame=false)
    on OPEN → Open Pause
    on ACTIVATE → Activate
  }

  state Open Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Activate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Battle Enemies: int = 1
  var Activated: bool = false
  var Self: gameObject = <null>
  var Shield Zombie: gameObject = <null>
}
