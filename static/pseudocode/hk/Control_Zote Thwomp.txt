# Control
# 1 location(s):
#   GG_Mighty_Zote / Battle Control/Zote Thwomp

fsm Control {
  start Init
  on ZERO HP → Break Antic  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Slam Effect", storeResult=var "Slam Effect")
    FindChild(gameObject=Self, childName="Pt Fall", storeResult=var "Pt Fall")
    FindChild(gameObject=Self, childName="Pt Break", storeResult=var "Pt Break")
    FindChild(gameObject=Self, childName="Enemy Crusher", storeResult=var "Enemy Crusher")
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")  // disabled
    on FINISHED → Dormant
  }

  state Dormant {
    on GO → Set Pos
  }

  state Set Pos {
    RandomFloat(min=92.43, max=113.85, storeResult=var "X Pos")
    SetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=23.4, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetHP(target=Self, hp=200)
    SetIsDead(target=Self, setValue=false)
    SetPolygonCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Tween Down
  }

  state Tween Down {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -3, 0), time=0.7, delay=0, speed=(unset), easeType=easeInOutSine, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Skull Idle")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_pirouette_01 (sharedassets399.assets), pitchMin=0.8, pitchMax=0.8, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Antic Shake
  }

  state Antic Shake {
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bone_pile_movement_1 (sharedassets299.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Down
  }

  state Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-50, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Skull Fall")
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    PlayParticleEmitterInState(gameObject=var "Pt Fall")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_down_stab_dash (sharedassets209.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Enemy Crusher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on LAND → Waves
  }

  state Slam {
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_strike_ground (sharedassets48.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Skull Land")
    WaitRandom(timeMin=5, timeMax=10, finishEvent=→"NEXT", realTime=false)
    ActivateGameObject(gameObject=var "Enemy Crusher", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on NEXT → Rise Antic
  }

  state Rise Antic {
    ObjectJitter(gameObject=Self, x=0.1, y=(unset), z=(unset), allowMovement=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bone_pile_movement_2 (sharedassets299.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bone_pile_movement_2 (sharedassets299.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=bone_pile_movement_4 (sharedassets406.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Skull Idle")
    on FINISHED → Rise
  }

  state Rise {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Skull Fall")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=60, everyFrame=false)
    PlayParticleEmitterInState(gameObject=var "Pt Fall")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_horizontal_dash_dash (sharedassets243.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=21.4, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Crusher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Out
  }

  state Out {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=99, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="RETURN", delay=0, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Enemy Crusher", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait
  }

  state Waves {
    SpawnObjectFromGlobalPool(gameObject=Shockwave Wave (sharedassets48.assets), spawnPoint=var "Self", position=(0, -2.3, -0.001), rotation=(unset), storeObject=var "Shockwave")
    SetScale(gameObject=var "Shockwave", vector=(unset), x=2.5, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFsmBool(gameObject=var "Shockwave", fsmName="shockwave", variableName="Facing Right", setValue=true, everyFrame=false)
    SetFsmFloat(gameObject=var "Shockwave", fsmName="shockwave", variableName="Speed", setValue=8, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Shockwave Wave (sharedassets48.assets), spawnPoint=var "Self", position=(0, -2.3, -0.001), rotation=(unset), storeObject=var "Shockwave")
    SetScale(gameObject=var "Shockwave", vector=(unset), x=2.5, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFsmBool(gameObject=var "Shockwave", fsmName="shockwave", variableName="Facing Right", setValue=false, everyFrame=false)
    SetFsmFloat(gameObject=var "Shockwave", fsmName="shockwave", variableName="Speed", setValue=12, everyFrame=false)
    on FINISHED → Slam
  }

  state Break Antic {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventToRegister(eventName="ZOTE WARRIOR KILLED")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Skull Crack")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=false_knight_damage_armour_final (sharedassets46.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    SetPolygonCollider(gameObject=Self, active=false)
    ObjectJitter(gameObject=Self, x=0.15, y=0.15, z=(unset), allowMovement=false)
    on FINISHED → Break
  }

  state Break {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ghost Burst", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Out
  }

  state Wait {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="RETURN", delay=0, everyFrame=false)
    on GO → Set Pos
  }

  var Parent: gameObject = Battle Control/Extra Zotes

  var X Pos: float = 0
  var Y Pos: float = 0
  var Enemy Crusher: gameObject = Battle Control/Extra Zotes
  var Pt Break: gameObject = <null>
  var Pt Fall: gameObject = <null>
  var Self: gameObject = <null>
  var Shockwave: gameObject = <null>
  var Slam Effect: gameObject = <null>
}
