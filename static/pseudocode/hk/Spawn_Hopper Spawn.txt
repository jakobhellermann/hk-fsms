# Spawn
# 1 location(s):
#   Deepnest_East_06 / Hopper Spawn

fsm Spawn {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Active?
  }

  state Active? {
    GetIsDead(target=var "Hopper", storeValue=var "Is Dead")
    BoolTest(boolVariable=var "Is Dead", isTrue=→"INACTIVE", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Hopper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on INACTIVE → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Hopper: gameObject = Hopper Spawn/Giant Hopper

  var Is Dead: bool = false
}
