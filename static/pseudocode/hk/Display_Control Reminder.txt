# Display
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Prompts/Control Reminder
#   resources.assets / _GameCameras/HudCamera/Prompts/Control Reminder

fsm Display {
  start Init
  on LEVEL LOADED → Idle  // from any state
  on REMINDER FOCUS → To Focus  // from any state

  state Idle {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on REMINDER FOCUS → Focus
    on REMINDER ATTACK → Attack
    on REMINDER JUMP → First Jump
    on REMINDER INTERACT → Interact
    on REMINDER LOOK DOWN → Look Down Mawlek
    on REMINDER FIREBALL → Japanese?
    on REMINDER DASH → Dash
    on REMINDER SUPERDASH → Superdash
    on REMINDER DREAM NAIL → Dreamnail
  }

  state Down {
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeGroupDown(target=Self, fast=false)
    on FINISHED → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="ActionButton Cast JA", storeResult=var "ActionButton Cast JA")
    on FINISHED → Idle
  }

  state Attack {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_ATTACK", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton Attack", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    FadeGroupUp(target=Self)
    ListenForAttack(eventTarget=Self, wasPressed=→"DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on DOWN → Down Wait
  }

  state Down Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  state Down Wait {
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Down Counter", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Down Pause
  }

  state First Jump {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_JUMP", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton Jump", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Txt Hold", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    FadeGroupUp(target=Self)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=109, tolerance=0, equal=(none), lessThan=→"DOWN", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=114, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"DOWN", everyFrame=true)
    on DOWN → Down Wait
  }

  state Interact {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_INTERACT", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton Up", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeGroupUp(target=Self)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=4, finishEvent=→"DOWN", realTime=false)
    on DOWN → Down Wait
    on LORE PROMPT UP → Down
    on HIDE INTERACT REMINDER → Down
  }

  state Focus {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_FOCUS", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton Cast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Txt Hold", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    FadeGroupUp(target=Self)
    on DOWN → 
    on PROMPT FOCUS END → Down Pause
    on DREAM FOCUS START → Down Pause
    on PROMPT FOCUS END QUICK → Down
  }

  state To Focus {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on FINISHED → Focus
  }

  state Look Down Mawlek {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_LOOK_DOWN", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton Down", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Txt Hold", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeGroupUp(target=Self)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    GetPosition(gameObject=var "MainCamera", vector=(unset), x=(unset), y=var "Camera Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Camera Y", float2=31, tolerance=0, equal=(none), lessThan=→"DOWN", greaterThan=(none), everyFrame=true)
    CheckHeroLookingDown(storeValue=(unset), everyFrame=true, notLookingDownEvent=(none), lookingDownEvent=→"DOWN")
    on DOWN → Down Wait
    on LEVEL LOADED → Down
  }

  state Fireball {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_FIREBALL", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeGroupUp(target=Self)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=4, finishEvent=→"DOWN", realTime=false)
    on DOWN → Down Wait
    on STOP HERO EXIT → Down Wait
  }

  state Dash {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_DASH", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton Dash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    FadeGroupUp(target=Self)
    ListenForDash(eventTarget=Self, wasPressed=→"DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on DOWN → Down Wait
  }

  state Superdash {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_SUPERDASH", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton SD", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Txt Hold", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    FadeGroupUp(target=Self)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=211.92, tolerance=0, equal=(none), lessThan=→"DOWN", greaterThan=(none), everyFrame=true)
    on DOWN → Down Wait
  }

  state Dreamnail {
    SetFloatValue(floatVariable=var "Down Counter", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Down Counter", add=1, everyFrame=true, perSecond=true)
    GetLanguageString(sheetName="Prompts", convName="PROMPT_DREAMNAIL", storeValue=var "Label String")
    SetTextMeshProText(gameObject=var "Text", textString=var "Label String")
    ActivateGameObject(gameObject=var "ActionButton DreamNail", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Txt Hold", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Backboard"), sendEvent="UP", delay=0, everyFrame=false)
    FadeGroupUp(target=Self)
    on DOWN → Down Wait
    on DREAM PLANT HIT → Down Pause
  }

  state Japanese? {
    GetCurrentLanguageAsString(stringVariable=var "Language")
    SendEventByName(eventTarget=Self, sendEvent=var "Language", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "ActionButton Cast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Txt Tap", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on JA → Fireball JA
    on FINISHED → Fireball
  }

  state Fireball JA {
    ActivateGameObject(gameObject=var "ActionButton Cast JA", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fireball
  }

  var ActionButton Attack: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton Attack
  var ActionButton Cast: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton Cast
  var ActionButton Dash: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton Dash
  var ActionButton Down: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton Down
  var ActionButton DreamNail: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton DreamNail
  var ActionButton Jump: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton Jump
  var ActionButton QuickMap: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton QuickMap
  var ActionButton SD: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton SD
  var ActionButton Up: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/ActionButton Up
  var Backboard: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/Backboard
  var Text: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/Text
  var Txt Hold: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/Txt Hold
  var Txt Tap: gameObject = _GameCameras/HudCamera/Prompts/Control Reminder/Txt Tap

  var Camera Y: float = 0
  var Down Counter: float = 0
  var Hero X: float = 0
  var Label String: string = ""
  var Language: string = ""
  var ActionButton Cast JA: gameObject = <null>
  var Self: gameObject = <null>
}
