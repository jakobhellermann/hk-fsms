# Pane Names
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Border
#   resources.assets / _GameCameras/HudCamera/Inventory/Border

fsm Pane Names {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Pane Name", storeResult=var "Current Pane")
    FindChild(gameObject=Self, childName="Pane Name L", storeResult=var "Pane L")
    FindChild(gameObject=Self, childName="Pane Name R", storeResult=var "Pane R")
    on FINISHED → Idle
  }

  state Idle {
    on UPDATE → Down
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CHANGE DOWN", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Update text
  }

  state Update text {
    SetTextMeshProText(gameObject=var "Current Pane", textString=var "Current Pane Str")
    SetTextMeshProText(gameObject=var "Pane L", textString=var "L Pane Str")
    SetTextMeshProText(gameObject=var "Pane R", textString=var "R Pane Str")
    on FINISHED → Up
  }

  state Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CHANGE UP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Current Pane Str: string = ""
  var L Pane Str: string = ""
  var R Pane Str: string = ""
  var Current Pane: gameObject = <null>
  var Pane L: gameObject = <null>
  var Pane R: gameObject = <null>
}
