# corpse plant trap
# 1 location(s):
#   sharedassets128.assets / Corpse Plant Trap

fsm corpse plant trap {
  start Retract

  state Retract {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Grass Puff", storeResult=var "Grass Puff")
    FindChild(gameObject=Self, childName="Orange Puff", storeResult=var "Orange Puff")
    on FINISHED → Death
  }

  state Death {
    PlayParticleEmitter(gameObject=var "Grass Puff", emit=0)
    PlayParticleEmitter(gameObject=var "Orange Puff", emit=0)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var Grass Puff: gameObject = <null>
  var Orange Puff: gameObject = <null>
}
