# Conversation Control
# 1 location(s):
#   Room_Mansion / Xun NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Wake Dust", storeResult=var "Wake Dust")
    FindChild(gameObject=Self, childName="Leave Dust", storeResult=var "Leave Dust")
    FindChild(gameObject=Self, childName="Depart Beam", storeResult=var "Depart Beam")
    FindChild(gameObject=Self, childName="White Flash", storeResult=var "White Flash")
    FindGameObject(objectName="Heart Piece Folder", withTag="Untagged", store=var "Heart Piece Folder")
    FindChild(gameObject=var "Heart Piece Folder", childName="Heart Piece", storeResult=var "Heart Piece")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    on FINISHED → Idle
  }

  state End {
    AudioStop(gameObject=var "Voice Player")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Meet {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_talk_01 (sharedassets22.assets), Grey_Mourner_talk_02 (sharedassets22.assets), Grey_Mourner_talk_03 (sharedassets22.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataBool(boolName="metXun", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_MEET", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Box Down2
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Wake
  }

  state Convo Choice {
    GetPlayerDataBool(boolName="hasXunFlower", storeValue=var "Has Flower")
    GetPlayerDataBool(boolName="xunFlowerBroken", storeValue=var "Flower Broken")
    GetPlayerDataBool(boolName="xunFailedConvo1", storeValue=var "FailedConvo1")
    GetPlayerDataBool(boolName="xunFailedConvo2", storeValue=var "FailedConvo2")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFlowerGiven", isTrue=→"SUCCESS", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metXun", isTrue=(none), isFalse=→"MEET")
    BoolTestMulti(boolVariables=[var "FailedConvo1", var "Flower Broken"], boolStates=[false, true], trueEvent=→"FAILED1", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "FailedConvo2", var "Flower Broken"], boolStates=[false, true], trueEvent=→"FAILED2", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Flower Broken", var "Has Flower"], boolStates=[false, true], trueEvent=→"HAS FLOWER", falseEvent=(none), storeResult=(unset), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"REOFFER", delay=0, everyFrame=false)
    on MEET → Meet
    on REOFFER → Reoffer
    on HAS FLOWER → Has Flower
    on FAILED1 → Failed
    on SUCCESS → Success 1
    on FAILED2 → Failed 2
  }

  state Title {
    CreateObject(gameObject=Area Title (sharedassets9.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Area Title", networkInstantiate=false, networkGroup=0)  // disabled
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="XUN_C", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="backerCredits", isTrue=→"FINISHED", isFalse=(none))
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="XUN_NC", everyFrame=false)
    on FINISHED → Box Up
  }

  state Wake {
    BoolTest(boolVariable=var "Woken", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Wake Dust", emit=0)
    SetBoolValue(boolVariable=var "Woken", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wake")
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Grey_Mourner_wake (sharedassets22.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=stalactite_break (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=platform_dirt_fall_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    on FINISHED → Title
  }

  state Box Down2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Box Up YN
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_OFFER", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes Pause
  }

  state Decline Pause {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Box Up 2
  }

  state Yes Pause {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Give Flower
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Decline
  }

  state Decline {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_REFUSE", "Minor NPC"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_talk_01 (sharedassets22.assets), Grey_Mourner_talk_02 (sharedassets22.assets), Grey_Mourner_talk_03 (sharedassets22.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Give Flower {
    SetPlayerDataBool(boolName="hasXunFlower", value=true)
    SetPlayerDataBool(boolName="xunFlowerBroken", value=false)
    CreateObject(gameObject=Item Get Effect (sharedassets7.assets), spawnPoint=var "Hero", position=(0, -0.76, -1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=shiny_item_pickup (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Relic Get Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg", networkInstantiate=false, networkGroup=0)
    FindChild(gameObject=var "Msg", childName="Text", storeResult=var "Msg Text")
    FindChild(gameObject=var "Msg", childName="Icon", storeResult=var "Msg Item")
    SetSpriteRendererSprite(gameObject=var "Msg Item", sprite=White_Flower_Full (resources.assets))
    GetLanguageString(sheetName="UI", convName="INV_NAME_FLOWER", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 3
  }

  state Box Up 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Accept
  }

  state Accept {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_talk_01 (sharedassets22.assets), Grey_Mourner_talk_02 (sharedassets22.assets), Grey_Mourner_talk_03 (sharedassets22.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFailedConvo1", isTrue=→"SECOND", isFalse=(none))
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_ACCEPT", "Minor NPC"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    on CONVO_FINISH → Talk Finish
    on SECOND → Accept 2
  }

  state Reoffer {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_talk_01 (sharedassets22.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFailedConvo1", isTrue=→"SECOND", isFalse=(none))
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_REOFFER", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Box Down2
    on SECOND → Reoffer 2
  }

  state Has Flower {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_talk_03 (sharedassets22.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFailedConvo1", isTrue=→"SECOND", isFalse=(none))
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_HAVE_FLOWER", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
    on SECOND → Has Flower 2
  }

  state Failed {
    SetPlayerDataBool(boolName="xunFailedConvo1", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_FAILED", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_Flower_Destroyed (sharedassets22.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down2
  }

  state Success 1 {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_death_version_02 (sharedassets22.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_SUCCESS1", "Minor NPC"], storeResult=(unset var))
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CONVO_FINISH → Success 2
  }

  state Success 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_SUCCESS2", "Minor NPC"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bask")
    on CONVO_FINISH → Box Down2 2
  }

  state Box Down2 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=0.9, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    on FINISHED → Crumble Antic
  }

  state Crumble {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["MOURNER"], storeResult=(unset var))
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Depart Beam"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Leave Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="xunRewardGiven", value=true)
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Heart Piece", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmGameObject(gameObject=var "Heart Piece", fsmName="Heart Container Control", variableName="Appear Source", setValue=var "Self", everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=stalactite_break (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=platform_dirt_fall_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  state Reoffer 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_REOFFER2", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Box Down2
  }

  state Has Flower 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_HAVE_FLOWER2", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Accept 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_ACCEPT2", "Minor NPC"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    SetPlayerDataBool(boolName="xunFailedConvo2", value=false)
    on CONVO_FINISH → Talk Finish
  }

  state Failed 2 {
    SetPlayerDataBool(boolName="xunFailedConvo2", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["XUN_FAILED2", "Minor NPC"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Grey_Mourner_Flower_Destroyed (sharedassets22.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Box Down2
  }

  state Crumble Antic {
    SendEventByName(eventTarget=GameObject(var "Depart Beam"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Grey_Mourner_death_version_01 (sharedassets22.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Crumble
  }

  var FailedConvo1: bool = false
  var FailedConvo2: bool = true
  var Flower Broken: bool = false
  var Has Flower: bool = false
  var Spoken: bool = false
  var Woken: bool = false
  var Item Name: string = ""
  var Area Title: gameObject = <null>
  var Depart Beam: gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var Heart Piece: gameObject = <null>
  var Heart Piece Folder: gameObject = <null>
  var Leave Dust: gameObject = <null>
  var Msg: gameObject = <null>
  var Msg Item: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
  var Wake Dust: gameObject = <null>
  var White Flash: gameObject = <null>
}
