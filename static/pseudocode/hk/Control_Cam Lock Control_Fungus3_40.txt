# Control
# 1 location(s):
#   Fungus3_40 / Cam Lock Control

fsm Control {
  start Inactive

  state Inactive {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on DREAM GATE CLOSE → Active
  }

  state Active {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on DREAM GATE OPEN → Inactive
  }

  var Self: gameObject = <null>
}
