# FSM
# 2 location(s):
#   Fungus2_25 / mantis_big_door
#   Fungus2_31 / Lords Undefeated Group

// uses template: destroy_if_playerdatabool
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var playerData bool: string = "defeatedMantisLords"
}
