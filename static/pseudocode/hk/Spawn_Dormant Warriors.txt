# Spawn
# 1 location(s):
#   GG_Mighty_Zote / Battle Control/Dormant Warriors

fsm Spawn {
  start Init

  state Init {
    on FINISHED → Dormant
  }

  state Dormant {
    on START → Spawn
  }

  state Idle {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Spawn {
    GetChildCount(gameObject=var "Active Warriors", storeResult=var "Current Active")
    IntCompare(integer1=var "Current Active", integer2=var "Active Max", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Zote To Spawn")
    GameObjectIsNull(gameObject=var "Zote To Spawn", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SetParent(gameObject=var "Zote To Spawn", parent=var "Active Warriors", resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(var "Zote To Spawn"), sendEvent="SPAWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Active Warriors: gameObject = Battle Control/Active Warriors

  var Active Max: int = 2
  var Current Active: int = 0
  var Next Spawn: gameObject = <null>
  var Zote To Spawn: gameObject = <null>
}
