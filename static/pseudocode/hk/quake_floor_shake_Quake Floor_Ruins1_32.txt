// uses template: quake_floor_shake
fsm quake_floor_shake {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Active", storeResult=var "Active")
    FindChild(gameObject=Self, childName="Pt Rocks", storeResult=var "Pt Rocks")
    FindChild(gameObject=var "Active", childName="Sprite", storeResult=var "Sprite")
    GetPosition(gameObject=var "Sprite", vector=var "Sprite Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    SetPosition(gameObject=var "Sprite", vector=var "Sprite Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Collision2dEvent(collision=0, collideTag="Player", sendEvent=→"HERO ON", storeCollider=var "", storeForce=(unset))
    AudioStop(gameObject=Self)
    SetParticleEmission(gameObject=var "Pt Rocks", emission=false)
    on HERO ON → Shaking
  }

  state Shaking {
    SetPosition(gameObject=var "Sprite", vector=var "Sprite Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Collision2dEvent(collision=2, collideTag="Player", sendEvent=→"HERO OFF", storeCollider=var "", storeForce=(unset))
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=var "Hero X Speed", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X Speed", float2=0, tolerance=0.01, equal=→"STILL", lessThan=(none), greaterThan=(none), everyFrame=true)
    ObjectJitter(gameObject=var "Sprite", x=0.025, y=0.025, z=(unset), allowMovement=false)
    SetParticleEmission(gameObject=var "Pt Rocks", emission=true)
    AudioPlaySimple(gameObject=Self, volume=0.65, oneShotClip=<null>)
    on HERO OFF → Idle
    on STILL → Shaking Cooldown
  }

  state Shaking Cooldown {
    SetPosition(gameObject=var "Sprite", vector=var "Sprite Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Collision2dEvent(collision=2, collideTag="Player", sendEvent=→"HERO OFF", storeCollider=var "", storeForce=(unset))
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=var "Hero X Speed", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X Speed", float2=0, tolerance=0.01, equal=(none), lessThan=→"MOVING", greaterThan=→"MOVING", everyFrame=true)
    Wait(time=0.35, finishEvent=→"WAIT", realTime=false)
    ObjectJitter(gameObject=var "Sprite", x=0.025, y=0.025, z=(unset), allowMovement=false)
    on HERO OFF → Idle
    on MOVING → Shaking
    on WAIT → Still
  }

  state Still {
    SetPosition(gameObject=var "Sprite", vector=var "Sprite Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Collision2dEvent(collision=2, collideTag="Player", sendEvent=→"HERO OFF", storeCollider=var "", storeForce=(unset))
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=var "Hero X Speed", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X Speed", float2=0, tolerance=0.01, equal=(none), lessThan=→"MOVING", greaterThan=→"MOVING", everyFrame=true)
    AudioStop(gameObject=Self)
    SetParticleEmission(gameObject=var "Pt Rocks", emission=false)
    on HERO OFF → Idle
    on MOVING → Shaking
  }

  var Hero X Speed: float = 0
  var Sprite Pos: vector3 = (0, 0, 0)
  var Active: gameObject = <null>
  var Pt Rocks: gameObject = <null>
  var Sprite: gameObject = <null>
}
