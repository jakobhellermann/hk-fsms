fsm Tele Out {
  start Idle

  state Idle {
    on TELE OUT → Anim
  }

  state Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out Quick", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
