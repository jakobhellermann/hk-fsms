# collide_with_wall
# 1 location(s):
#   sharedassets203.assets / Shot Spine

fsm collide_with_wall {
  start Init

  state In Air {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"COLLIDE", storeCollider=var "")
    FaceAngle(gameObject=Self, angleOffset=-90, everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Impact
    on ORBIT SHIELD → Break
  }

  state Impact {
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[garden_zombie_projectile_impact_1, garden_zombie_projectile_impact_2, garden_zombie_projectile_impact_3], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Impact", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.48235294, 0.48235294, 0.48235294, 1), everyframe=false)
    on FINISHED → Wait
  }

  state Init {
    FaceAngle(gameObject=Self, angleOffset=-90, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fire")
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    GetOwner(storeGameObject=var "Self")
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Pause
  }

  state Recycle {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RecycleSelf()
  }

  state Wait {
    WaitRandom(timeMin=8, timeMax=12, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Break
  }

  state Break {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Air
  }

  var Impact: gameObject = Shot Spine/Strike Nail
  var Self: gameObject = <null>
}
