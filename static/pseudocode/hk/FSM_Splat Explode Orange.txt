# FSM
# 1 location(s):
#   sharedassets138.assets / Splat Explode Orange

fsm FSM {
  start State 1

  state State 1 {
    SendRandomEvent(events=[→"FLIP", →"NO FLIP"], weights=[1, 1], delay=0)
    on FLIP → Fliup
    on NO FLIP → Rotate
  }

  state Fliup {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Scale", multiplyBy=-1, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Rotate
  }

  state Rotate {
    RandomFloat(min=-25, max=25, storeResult=var "Chooser")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Chooser: float = 0
  var Scale: float = 0
}
