# Detect
# 1 location(s):
#   sharedassets324.assets / Hornet Barb/Spell Detect

fsm Detect {
  start Init

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Send Event
    on ORBIT SHIELD → Send Event
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Detect
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="SPELL", delay=0, everyFrame=false)
    on FINISHED → Detect
  }

  var Parent: gameObject = <null>
}
