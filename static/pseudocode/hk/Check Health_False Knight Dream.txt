# Check Health
# 2 location(s):
#   Dream_01_False_Knight / False Knight Dream
#   GG_Failed_Champion / False Knight Dream

fsm Check Health {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get Health
  }

  state Check {
    GetHPEveryFrame(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=0, equal=→"STUN", lessThan=→"STUN", greaterThan=(none), everyFrame=true)
    on STUN → Stun
  }

  state Stun {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="FalseyControl"), sendEvent="STUN", delay=0, everyFrame=false)
    GetHP(target=Self, storeValue=var "HP")
    SetHP(target=Self, hp=var "Recover HP")
    on FINISHED → Check
  }

  state Get Health {
    GetHP(target=Self, storeValue=var "Recover HP")
    on FINISHED → Check
  }

  var HP: int = 0
  var Recover HP: int = 0
  var Stun 1: bool = false
  var Stun 2: bool = false
}
