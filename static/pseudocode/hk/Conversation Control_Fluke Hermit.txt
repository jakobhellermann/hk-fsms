# Conversation Control
# 1 location(s):
#   Room_GG_Shortcut / Fluke Hermit

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    RandomInt(min=1, max=5, storeResult=var "Convo", inclusiveMax=true)
    FindChild(gameObject=Self, childName="Hand", storeResult=var "Hand")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Title
  }

  state Convo 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["FLUKE_HERMIT_IDLE_1", "CP3"], storeResult=(unset var))
    SetIntValue(intVariable=var "Convo", intValue=2, everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Choice {
    BoolTest(boolVariable=var "Flukemarmed", isTrue=→"FLUKEMARM REPEAT", isFalse=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_11", isTrue=→"FLUKEMARM", isFalse=(none))
    IntSwitch(intVariable=var "Convo", compareTo=[1, 2, 3, 4, 5], sendEvent=[(none), (none), (none), (none), (none)], everyFrame=false)
    on FLUKEMARM → Flukemarm
    on FLUKEMARM REPEAT → Flukemarm 2
    on 1 → Convo 1
    on 2 → Convo 2
    on 3 → Convo 3
    on 4 → Convo 4
    on 5 → Convo 5
  }

  state Convo 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["FLUKE_HERMIT_IDLE_2", "CP3"], storeResult=(unset var))
    SetIntValue(intVariable=var "Convo", intValue=3, everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Convo 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["FLUKE_HERMIT_IDLE_3", "CP3"], storeResult=(unset var))
    SetIntValue(intVariable=var "Convo", intValue=4, everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Convo 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["FLUKE_HERMIT_IDLE_4", "CP3"], storeResult=(unset var))
    SetIntValue(intVariable=var "Convo", intValue=5, everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Convo 5 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["FLUKE_HERMIT_IDLE_5", "CP3"], storeResult=(unset var))
    SetIntValue(intVariable=var "Convo", intValue=1, everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Title {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="FLUKE_HERMIT", everyFrame=false)
    on FINISHED → Box Up
  }

  state Flukemarm {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=junk_fluke_pray_no_loop (sharedassets471.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioStop(gameObject=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pray")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["FLUKE_HERMIT_PRAY", "CP3"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Flukemarmed", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Hand", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on CONVO_FINISH → Pray End
  }

  state Pray End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pray End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FadeAudio(gameObject=var "Voice Player", startVolume=1, endVolume=0, time=0.15)
    on FINISHED → Reset
  }

  state Flukemarm 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=junk_fluke_pray_no_loop (sharedassets471.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioStop(gameObject=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pray")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["FLUKE_HERMIT_PRAY_REPEAT", "CP3"], storeResult=(unset var))
    ActivateGameObject(gameObject=var "Hand", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on CONVO_FINISH → Pray End
  }

  state Reset {
    ActivateGameObject(gameObject=var "Hand", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Talk Finish
  }

  var Convo: int = 0
  var Flukemarmed: bool = false
  var Area Title: gameObject = <null>
  var Hand: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
