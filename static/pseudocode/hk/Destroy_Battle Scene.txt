# Destroy
# 1 location(s):
#   Crossroads_ShamanTemple / Battle Scene

fsm Destroy {
  start Pause

  state Detect {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on KILLED → Set Activated
    on ACTIVATED → Activate
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Set Activated {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Activate
  }

  state Activate {
    ActivateGameObject(gameObject=var "Cam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cam B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Reminder Cast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Init {
    FindChild(gameObject=Self, childName="CameraLockArea 6", storeResult=var "Cam")
    FindChild(gameObject=Self, childName="CameraLockArea 6 B", storeResult=var "Cam B")
    FindChild(gameObject=Self, childName="Reminder Cast", storeResult=var "Reminder Cast")
    ActivateGameObject(gameObject=var "Cam", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Detect
  }

  var Activated: bool = false
  var Blocker: gameObject = <null>
  var Cam: gameObject = <null>
  var Cam B: gameObject = <null>
  var Reminder Cast: gameObject = <null>
  var Self: gameObject = <null>
}
