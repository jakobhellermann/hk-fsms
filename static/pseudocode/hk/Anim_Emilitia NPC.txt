# Anim
# 1 location(s):
#   Ruins_House_03 / Emilitia NPC

fsm Anim {
  start Init

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    BoolTest(boolVariable=var "Range R", isTrue=→"LOOK R", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Range L", isTrue=→"LOOK L", isFalse=(none), everyFrame=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on LOOK R → Idle To R
    on LOOK L → Idle To L
  }

  state Idle To R {
    BoolTest(boolVariable=var "Talking", isTrue=→"TALK", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="IdleToR")
    BoolTest(boolVariable=var "Range L", isTrue=→"LOOK L", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Range R", isTrue=(none), isFalse=→"IDLE", everyFrame=true)
    AudioPlayRandom(gameObject=var "Self", audioClips=[Amelitia_Startle_01 (sharedassets85.assets), Amelitia_Startle_02 (sharedassets85.assets), Amelitia_Startle_03 (sharedassets85.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on IDLE → R To Idle
    on LOOK L → R To L
    on TALK → Talk R
  }

  state R To Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RToIdle")
    BoolTest(boolVariable=var "Range R", isTrue=→"LOOK R", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Range L", isTrue=→"LOOK L", isFalse=(none), everyFrame=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on LOOK R → Idle To R
    on LOOK L → Idle To L
  }

  state Idle To L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="IdleToL")
    BoolTest(boolVariable=var "Talking", isTrue=→"TALK", isFalse=(none), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Range L", var "Range R"], boolStates=[false, true], trueEvent=→"LOOK R", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Range L", var "Range R"], boolStates=[false, false], trueEvent=→"IDLE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    AudioPlayRandom(gameObject=var "Self", audioClips=[Amelitia_Startle_01 (sharedassets85.assets), Amelitia_Startle_02 (sharedassets85.assets), Amelitia_Startle_03 (sharedassets85.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on IDLE → L To Idle
    on LOOK R → L To R
    on TALK → Talk L
  }

  state L To Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LToIdle")
    BoolTest(boolVariable=var "Range R", isTrue=→"LOOK R", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Range L", isTrue=→"LOOK L", isFalse=(none), everyFrame=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on LOOK R → Idle To R
    on LOOK L → Idle To L
  }

  state R To L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RToL")
    BoolTest(boolVariable=var "Talking", isTrue=→"TALK", isFalse=(none), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Range L", var "Range R"], boolStates=[false, true], trueEvent=→"LOOK R", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Range L", var "Range R"], boolStates=[false, false], trueEvent=→"IDLE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on IDLE → L To Idle
    on LOOK R → L To R
    on TALK → Talk L
  }

  state L To R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LToR")
    BoolTest(boolVariable=var "Talking", isTrue=→"TALK", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Range L", isTrue=→"LOOK L", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Range R", isTrue=(none), isFalse=→"IDLE", everyFrame=true)
    on IDLE → R To Idle
    on LOOK L → R To L
    on TALK → Talk R
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk R")
    BoolTest(boolVariable=var "Talking", isTrue=(none), isFalse=→"LOOK R", everyFrame=true)
    on LOOK R → Idle R
  }

  state Talk L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk L")
    BoolTest(boolVariable=var "Talking", isTrue=(none), isFalse=→"LOOK L", everyFrame=true)
    on LOOK L → Idle L
  }

  state Idle L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle L")
    BoolTest(boolVariable=var "Talking", isTrue=→"TALK", isFalse=(none), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Range L", var "Range R"], boolStates=[false, true], trueEvent=→"LOOK R", falseEvent=(none), storeResult=(unset), everyFrame=true)

    on LOOK R → L To R

  }

    BoolTest(boolVariable=var "Talking", isTrue=→"TALK", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle R")
    BoolTest(boolVariable=var "Range L", isTrue=→"LOOK L", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Range R", isTrue=(none), isFalse=→"IDLE", everyFrame=true)
    on IDLE → R To Idle
    on LOOK L → R To L
    on TALK → Talk R
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Range L: bool = false
  var Range R: bool = false
  var Talking: bool = false
  var Self: gameObject = <null>
}
