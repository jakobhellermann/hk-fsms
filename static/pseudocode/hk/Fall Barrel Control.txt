# Fall Barrel Control
# 1 location(s):
#   sharedassets48.assets / Falling Barrel

fsm Fall Barrel Control {
  start Init

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"NAIL HIT", storeCollider=var "Nail Object")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Player", collideLayer=(unset), sendEvent=→"BREAK", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"BREAK", storeCollider=var "")
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin Dir", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on BREAK → Pause
    on NAIL HIT → Check Direct
    on ORBIT SHIELD → Break
  }

  state Break {
    SetSpriteRenderer(gameObject=Self, active=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Bits", emit=0)
    PlayParticleEmitter(gameObject=var "Dust Puff", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=wagon_death (sharedassets37.assets), pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    StopParticleEmitter(gameObject=var "Duster")
    ActivateGameObject(gameObject=var "Splat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnBlood(spawnPoint=var "Self", position=(unset), spawnMin=6, spawnMax=8, speedMin=10, speedMax=20, angleMin=75, angleMax=105, colorOverride=(unset))
    SetCollider(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Break
  }

  state Check Direct {
    GetEventSender(sentByGameObject=var "Attacker")  // disabled
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=10, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    GetFsmFloat(gameObject=var "Nail Object", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Attack Direction", speed=45, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.3)
    on FINISHED → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=var "Splat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=0, everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    PlayParticleEmitter(gameObject=var "Duster", emit=0)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FINISHED → Set Spin Dir
  }

  state Recycle {
    RecycleSelf()
    on FINISHED → 
  }

  state Set Spin Dir {
    RandomlyFlipFloat(storeResult=var "Spin Dir")
    on FINISHED → Idle
  }

  var Attack Direction: float = 0
  var Spin Dir: float = 720
  var Attacker: gameObject = <null>
  var Bits: gameObject = Falling Barrel/Bits
  var Dust Puff: gameObject = Falling Barrel/Dust Puff Med
  var Duster: gameObject = Falling Barrel/Dust Trail
  var Nail Object: gameObject = <null>
  var Self: gameObject = <null>
  var Splat: gameObject = Falling Barrel/Splat
}
