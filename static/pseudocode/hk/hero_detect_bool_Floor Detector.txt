// uses template: hero_detect_bool_stay
fsm hero_detect_bool {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=var "Self", storeResult=var "Parent")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    NextFrameEvent(sendEvent=→"EXIT")  // disabled
    SetIntValue(intVariable=var "Frames", intValue=0, everyFrame=false)
    IntAddV2(intVariable=var "Frames", add=1, everyFrame=true)
    IntCompare(integer1=var "Frames", integer2=2, equal=→"EXIT", lessThan=(none), greaterThan=→"EXIT", everyFrame=true)
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Exit {
    SetFsmBool(gameObject=var "Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=false, everyFrame=false)
    on FINISHED → Detect
  }

  var Bool Name: string = "Floor Detector"
  var FSM Name: string = "Zombie Miner"

  var Frames: int = 0
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
