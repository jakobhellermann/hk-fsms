# FSM
# 1 location(s):
#   Hive_05 / Battle Scene/Activate Vespa

fsm FSM {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Vespa NPC", storeResult=var "Vespa NPC")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Activate
  }

  state Activate {
    ActivateGameObject(gameObject=var "Vespa NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Parent: gameObject = <null>
  var Vespa NPC: gameObject = <null>
}
