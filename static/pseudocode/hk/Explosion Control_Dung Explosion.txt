# Explosion Control
# 1 location(s):
#   resources.assets / Dung Explosion

fsm Explosion Control {
  start Explode

  state Explode {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClip=explosion_4_wet, pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=3, y=3, z=(unset), everyFrame=false, lateUpdate=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Hit Effect: gameObject = <null>
  var Self: gameObject = Dung Explosion
}
