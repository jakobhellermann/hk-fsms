# Stun Control
# 1 location(s):
#   Room_Final_Boss_Core / Boss Control/Hollow Knight Boss

// uses template: stun_control
fsm Stun Control {
  start Init
  on STUN CONTROL STOP → Stop  // from any state
  on STUN CONTROL RESET → Reset  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Heavy Blow
  }

  state Idle {
    on STUN DAMAGE → Max Check
  }

  state In Combo {
    IntAdd(intVariable=var "Combo Counter", add=1, everyFrame=false)
    IntAdd(intVariable=var "Hits Total", add=1, everyFrame=false)
    IntCompare(integer1=var "Combo Counter", integer2=var "Stun Combo", equal=→"STUN", lessThan=(none), greaterThan=(none), everyFrame=false)
    Wait(time=var "Combo Time", finishEvent=→"TIME OUT", realTime=false)
    on TIME OUT → Reset Counter
    on STUN DAMAGE → Continue Combo
    on STUN → Stun
  }

  state Reset Counter {
    SetIntValue(intVariable=var "Combo Counter", intValue=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Continue Combo {
    IntCompare(integer1=var "Hits Total", integer2=var "Stun Hit Max", equal=→"STUN", lessThan=(none), greaterThan=→"STUN", everyFrame=false)
    on FINISHED → In Combo
    on STUN → Stun
  }

  state Stun {
    IntCompare(integer1=var "Stuns Total", integer2=var "Stuns Max", equal=→"MAX", lessThan=(none), greaterThan=→"MAX", everyFrame=false)
    IntAdd(intVariable=var "Stuns Total", add=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Combo Counter", intValue=0, everyFrame=false)
    SetIntValue(intVariable=var "Hits Total", intValue=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Max Check {
    GetTag(gameObject=<null>, storeResult="", everyFrame=false)
    IntCompare(integer1=var "Hits Total", integer2=var "Stun Hit Max", equal=→"STUN", lessThan=→"FINISHED", greaterThan=→"STUN", everyFrame=false)
    on FINISHED → In Combo
    on STUN → Stun
  }

  state Stop {
    on STUN CONTROL START → Reset Counter
    on STUN DAMAGE → Unstun Increment
  }

  state Unstun Increment {
    IntAdd(intVariable=var "Hits Total", add=1, everyFrame=false)
    on FINISHED → Stop
  }

  state Reset {
    on STUN CONTROL START → Reset Counter
  }

  state Heavy Blow {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_15", isTrue=(none), isFalse=→"FINISHED")
    IntOperator(integer1=var "Stun Hit Max", integer2=1, operation=Subtract, storeResult=var "Stun Hit Max", everyFrame=false)
    IntOperator(integer1=var "Stun Combo", integer2=1, operation=Subtract, storeResult=var "Stun Combo", everyFrame=false)
    on FINISHED → Idle
  }

  var Combo Time: float = 1
  var Stun Combo: int = 10
  var Stun Hit Max: int = 12
  var Stuns Max: int = 5

  var Combo Counter: int = 0
  var Decrement: int = 0
  var Hits Total: int = 0
  var Stuns Total: int = 0
  var Tag: string = ""
  var Self: gameObject = <null>
}
