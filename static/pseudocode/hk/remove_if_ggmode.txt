# remove_if_ggmode
# 4 location(s):
#   GG_Engine_Prime / Godseeker EngineRoom NPC
#   GG_Engine_Root / Godseeker Young NPC
#   GG_Wyrm / Godseeker Young NPC
#   GG_Unn / Godseeker Young NPC

// uses template: remove_if_ggmode
fsm remove_if_ggmode {
  start Check

  state Check {
    GGCheckIsBossRushMode(trueEvent=→"DISABLE", falseEvent=(none))
    on DISABLE → Disable
  }

  state Disable {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
