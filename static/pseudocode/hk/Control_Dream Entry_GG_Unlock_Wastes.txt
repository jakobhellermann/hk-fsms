# Control
# 1 location(s):
#   GG_Unlock_Wastes / Dream Entry

fsm Control {
  start Pause

  state Take Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Start Kneeling
  }

  state Start Fade {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return from Kneeling
  }

  state Return Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AcceptInput())
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="ResetQuakeDamage", parameters=[], storeResult=(unset var))
  }

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Take Control
  }

  state Start Kneeling {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 3")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pause Kneeling
  }

  state Return from Kneeling {
    Tk2dPauseAnimation(gameObject=var "Hero", pause=false, everyframe=false)
    Tk2dWatchAnimationEvents(gameObject=var "Hero", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Return Control
  }

  state Pause Kneeling {
    Tk2dPauseAnimation(gameObject=var "Hero", pause=true, everyframe=false)
    on FINISHED → Start Fade
  }
}
