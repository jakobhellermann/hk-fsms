// uses template: nail_cancel_check
fsm nail_cancel_check {
  start Init

  state Idle {
    on HERO DAMAGED → Cancel
    on HERO RESPAWNED → Cancel
  }

  state Cancel {
    SendMessage(gameObject=var "Self", delivery=SendMessage, options=1, functionCall=CancelAttack())
    on FINISHED → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
}
