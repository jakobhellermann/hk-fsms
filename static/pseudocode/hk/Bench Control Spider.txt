# Bench Control Spider
# 1 location(s):
#   Deepnest_Spider_Town / RestBench Spider

fsm Bench Control Spider {
  start Pause Frame

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Lit", storeResult=var "Lit")
    FindChild(gameObject=Self, childName="Knight Anim", storeResult=var "Knight Anim")
    FindChild(gameObject=Self, childName="Splat Point L", storeResult=var "Splat Point L")
    FindChild(gameObject=Self, childName="Splat Point R", storeResult=var "Splat Point R")
    FindChild(gameObject=Self, childName="Webbed", storeResult=var "Webbed")
    FindChild(gameObject=var "Self", childName="Prompt Marker", storeResult=var "Prompt Marker")
    CreateObject(gameObject=Arrow Prompt (resources.assets), spawnPoint=var "Prompt Marker", position=(unset), rotation=(unset), storeObject=var "Prompt", networkInstantiate=false, networkGroup=0)
    SetFsmString(gameObject=var "Prompt", fsmName="Prompt Control", variableName="Prompt Name", setValue="Rest", everyFrame=false)
    ActivateGameObject(gameObject=var "Prompt Marker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    on FINISHED → Idle
  }

  state Idle {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"IN RANGE", storeCollider=var "")
    on IN RANGE → In Range
  }

  state In Range {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="UP", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"OUT OF RANGE", storeCollider=var "")
    GetAxis(axisName="Vertical", multiplier=1, store=var "Axis Vert", everyFrame=true)  // disabled
    FloatCompare(float1=var "Axis Vert", float2=0.6, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UP PRESSED", everyFrame=true)  // disabled
    ListenForUp(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=false, stateEntryOnly=false)
    ListenForDown(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=false, stateEntryOnly=false)
    on OUT OF RANGE → Idle
    on UP PRESSED → Can Rest?
  }

  state Can Rest? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="CanInput", parameters=[], storeResult=var "Can Input")
    BoolTest(boolVariable=var "Can Input", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["onGround"], storeResult=var "On Ground")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["attacking"], storeResult=var "Attacking")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["upAttacking"], storeResult=var "Attacking Up")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["downAttacking"], storeResult=var "Attacking Down")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["dashing"], storeResult=var "Dashing")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["backDashing"], storeResult=var "Backdashing")
    BoolTest(boolVariable=var "On Ground", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    BoolAnyTrue(boolVariables=[var "Attacking", var "Attacking Down", var "Attacking Up", var "Backdashing", var "Dashing"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Start Rest
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Range
  }

  state Start Rest {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var))
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Sit")
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Sit Vector", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Sit Vector", vector3Value=(unset), x=var "Self X", y=(unset), z=(unset), everyFrame=false)
    Vector3Add(vector3Variable=var "Sit Vector", addVector=var "Adjust Vector", everyFrame=false, perSecond=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    iTweenMoveTo(gameObject=var "Hero", id="", transformPosition=var "", vectorPosition=var "Sit Vector", time=0.2, delay=0, speed=(unset), space=0, easeType=easeOutSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.3, finishEvent=→"WAIT", realTime=false)
    on WAIT → Sit Start
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Sit Start {
    SetMeshRenderer(gameObject=var "Hero", active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=grass_cut_1 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetMeshRenderer(gameObject=var "Knight Anim", active=true)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Splat Point L", position=(unset), spawnMin=3, spawnMax=4, speedMin=10, speedMax=15, angleMin=100, angleMax=150, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (resources.assets), spawnPoint=var "Splat Point L", position=(unset), spawnMin=3, spawnMax=4, speedMin=10, speedMax=15, angleMin=30, angleMax=80, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    SetSpriteRenderer(gameObject=var "Webbed", active=true)
    on FINISHED → Neutral
  }

  state Neutral {
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"RIGHT", isNotPressed=(none))
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"LEFT", isNotPressed=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"STRUGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForAttack(eventTarget=Self, wasPressed=→"STRUGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForDown(eventTarget=Self, wasPressed=→"STRUGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForUp(eventTarget=Self, wasPressed=→"STRUGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    on LEFT → Start Left
    on RIGHT → Start Right
    on STRUGGLE → Struggle
  }

  state Struggle {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="FIRST STRUGGLE", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Knight Anim", clipName="Shake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Set Flange
  }

  state Left {
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"FINISHED")
    on FINISHED → Return Left
  }

  state Start Left {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="FIRST STRUGGLE", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Knight Anim", animLibName="", clipName="Struggle Left")
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Left
  }

  state Start Right {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="FIRST STRUGGLE", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Knight Anim", animLibName="", clipName="Struggle Right")
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Right
  }

  state Right {
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"FINISHED")
    on FINISHED → Return Right
  }

  state Return Left {
    Tk2dPlayAnimationWithEvents(gameObject=var "Knight Anim", clipName="Return Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Set Flange
  }

  state Return Right {
    Tk2dPlayAnimationWithEvents(gameObject=var "Knight Anim", clipName="Return Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Set Flange
  }

  state Set Flange {
    BoolTest(boolVariable=var "Flanged", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Flanged", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal Flangier (resources.assets), transitionTime=3)
    on FINISHED → Neutral
  }

  var Adjust Vector: vector3 = (0, 0, 0)

  var Axis Vert: float = 0
  var Self X: float = 0
  var Attacking: bool = false
  var Attacking Down: bool = false
  var Attacking Up: bool = false
  var Backdashing: bool = false
  var Can Input: bool = false
  var Dashing: bool = false
  var Flanged: bool = false
  var On Ground: bool = false
  var Sit Vector: vector3 = (0, 0, 0)
  var Hero: gameObject = <null>
  var Knight Anim: gameObject = <null>
  var Lit: gameObject = <null>
  var Prompt: gameObject = <null>
  var Prompt Marker: gameObject = <null>
  var Self: gameObject = <null>
  var Splat Point L: gameObject = <null>
  var Splat Point R: gameObject = <null>
  var Webbed: gameObject = <null>
}
