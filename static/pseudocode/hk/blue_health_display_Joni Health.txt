# blue_health_display
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health/Joni Health
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health/Joni Health

// uses template: blue_health_display_joni
fsm blue_health_display {
  start Init
  on OFF → Off  // from any state
  on CHARM INDICATOR CHECK → Hive Check  // from any state

  state Init {
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 0), everyframe=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Idle", storeResult=var "Idle Sprite")
    FindChild(gameObject=var "Self", childName="Max Up", storeResult=var "Max Up")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Health 1", storeResult=var "Health 1")
    Tk2dSpriteSetColor(gameObject=var "Health 1", color=(1, 1, 1, 0), everyframe=false)
    SetScale(gameObject=Self, vector=(unset), x=0.75, y=0.75, z=(unset), everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=-3, space=1, everyFrame=false, lateUpdate=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hive Check
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName=var "Anim Idle")
    RandomInt(min=0, max=26, storeResult=var "Chooser", inclusiveMax=true)  // disabled
    Tk2dPlayFrame(gameObject=var "Self", frame=var "Chooser")  // disabled
    on HERO LEAVE → Break?
    on BENCHREST → Reappear
  }

  state Off {
    SetMeshRenderer(gameObject=Self, active=false)
    Tk2dSpriteSetColor(gameObject=var "Health 1", color=(1, 1, 1, 1), everyframe=false)
    on ON → Hive Check
  }

  state Break? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Current HP")  // disabled
    IntCompare(integer1=var "Current HP", integer2=var "Health Number", equal=→"FULL", lessThan=→"EMPTY", greaterThan=→"FULL", everyFrame=false)  // disabled
    Tk2dPlayAnimationWithEvents(gameObject=var "Self", clipName=var "Anim Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Off
    on FULL → Idle
  }

  state Hive Check {
    SetStringValue(stringVariable=var "Anim Break", stringValue="Blue Break", everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=7.7, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetStringValue(stringVariable=var "Anim Idle", stringValue="Blue Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Appear", stringValue="Blue Appear", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_29", isTrue=(none), isFalse=→"FINISHED")
    SetStringValue(stringVariable=var "Anim Break", stringValue="Hive Health Break B", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Idle", stringValue="Hive Health Idle B", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Appear", stringValue="Hive Health Appear B", everyFrame=false)
    on FINISHED → Appear
  }

  state Appear {
    BoolTest(boolVariable=var "Appeared?", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Appeared?", boolValue=true, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    Tk2dSpriteSetColor(gameObject=var "Health 1", color=(1, 1, 1, 1), everyframe=false)
    on FINISHED → Idle
  }

  state Reappear {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Idle
  }

  var Health Number: int = 0

  var Chooser: int = 0
  var Current HP: int = 0
  var Appeared?: bool = false
  var Start Idle: bool = false
  var Anim Appear: string = ""
  var Anim Break: string = ""
  var Anim Idle: string = ""
  var Health 1: gameObject = <null>
  var Idle Sprite: gameObject = <null>
  var Max Up: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
