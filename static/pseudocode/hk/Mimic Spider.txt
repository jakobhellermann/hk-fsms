fsm Mimic Spider {
  start Init
  on ENCOUNTERED → Encountered  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetBoolValue(boolVariable=var "Transformed", boolValue=false, everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    on FINISHED → Hollow Idle
    on GG BOSS → GG Pause
  }

  state Idle {
    SetGravity2dScale(gameObject=Self, gravityScale=2)
    SetBoolValue(boolVariable=var "Will Spit", boolValue=false, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Self X", lowerValue=88.32, upperValue=103.78, boolVariable=(unset), trueEvent=→"NO SPIT", falseEvent=(none), everyFrame=false)
    SendRandomEventV3(events=[→"CHARGE", →"JUMP", →"SPIT"], weights=[0.25, 0.5, 0.25], trackingInts=[var "Ct Charge", var "Ct Jump", var "Ct Spit"], eventMax=[2, 3, 2], trackingIntsMissed=[var "Ms Charge", var "Ms Jump", var "Ms Spit"], missedMax=[5, 4, 3])
    SendEventByName(eventTarget=Self, sendEvent="ROOF SPIT", delay=0, everyFrame=false)
    on CHARGE → Charge Init
    on JUMP → Aim Jump
    on ROAR → Roar Init
    on SPIT → Set Spit
    on NO SPIT → Idle No Spit
  }

  state Charge Init {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Charge Antic 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Facing", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Speed", floatValue=var "Charge Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "Facing", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mimic_spider_short_scream (sharedassets290.assets))
    on FINISHED → C Antic 2
  }

  state Charge R {
    AudioPlayInState(gameObject=Self, volume=1)  // disabled
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Jump Max X", tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    FloatTestToBool(float1=var "Self X", float2=var "Spit X", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Spit X Pos", everyFrame=true)

    AudioPlayInState(gameObject=var "Loop Run", volume=1)
    SetParticleEmission(gameObject=var "Run Dust", emission=true)
    on FINISHED → Wait
    on SPIT → Spit Antic
  }

  state Wait {
    AudioStop(gameObject=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Antic")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CHARGE END", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Run Dust", emission=false)
    on FINISHED → Idle

    on FINISHED → Jump
  }

  state Jump {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=var "Jump Height", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")

  }

    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=→"FINISHED", animationCompleteEvent=→"FINISHED")
    on FINISHED → Jump Init
  }

  state Jump Wait {

    on FINISHED → Jump Land
  }

  state Roar Init {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar Init", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MIMIC_SPIDER", everyFrame=false)
    on FINISHED → Roar Loop
  }

  state Roar Loop {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar Loop")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)

    SetParent(gameObject=var "Roar Emitter", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Roar Finish
  }

  state Roar Finish {
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar Finish", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

    AudioPlaySimple(gameObject=var "Rumble Loop", volume=1, oneShotClip=<null>)

  }

  state Trans 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Transform 2")

  }

    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Transform 3")

  state Jump Antic {

    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mimic_spider_jump (sharedassets290.assets))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")

    on FINISHED → Rising
  }

  state Aim Jump {

    FloatOperator(float1=var "Target X", float2=var "Self X", operation=Subtract, storeResult=var "Jump Distance", everyFrame=false)
    FloatMultiply(floatVariable=var "Jump Distance", multiplyBy=1.65, everyFrame=false)
    FloatClamp(floatVariable=var "Jump Distance", minValue=-20, maxValue=20, everyFrame=false)
    FloatCompare(float1=var "Jump Distance", float2=0, tolerance=0, equal=→"LEFT", lessThan=→"LEFT", greaterThan=→"RIGHT", everyFrame=false)
    on LEFT → Face L
    on RIGHT → Face R
  }

  state Rising {
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", space=0, everyFrame=true)  // disabled

    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump Distance", y=var "Velocity", everyFrame=true)  // disabled
    FloatCompare(float1=var "Velocity", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Falling
  }

  state Falling {
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", space=0, everyFrame=true)  // disabled
    FloatDivideV2(floatVariable=var "Velocity", divideBy=0.7, everyFrame=true, fixedUpdate=true)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump Distance", y=var "Velocity", everyFrame=true)  // disabled
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
