# Avoid Knight
# 1 location(s):
#   White_Palace_03_hub / Progress Gate

fsm Avoid Knight {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check
  }

  state Check {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero Y", float2=5, tolerance=0, equal=(none), lessThan=→"DISAPPEAR", greaterThan=(none), everyFrame=false)
    on DISAPPEAR → Disappear
  }

  state Disappear {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Hero Y: float = 0
}
