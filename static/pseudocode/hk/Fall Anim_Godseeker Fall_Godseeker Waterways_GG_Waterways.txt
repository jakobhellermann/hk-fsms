# Fall Anim
# 1 location(s):
#   GG_Waterways / Godseeker Waterways/Godseeker Fall

fsm Fall Anim {
  start Wait Frame
  on DO FALL → Wait  // from any state

  state Wait {
    GetOwner(storeGameObject=var "Self")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.006, space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dPauseAnimation(gameObject=Self, pause=true, everyframe=false)
    Wait(time=1.85, finishEvent=→"FINISHED", realTime=false)
    FindChild(gameObject=Self, childName="Slam Effect", storeResult=var "Slam Effect")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=godseeker_groan_01 (sharedassets225.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Fall
  }

  state Shake {
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fall
  }

  state Fall {
    Tk2dPauseAnimation(gameObject=Self, pause=false, everyframe=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=coccoon_god_body_fall_out (sharedassets225.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Land Effects
  }

  state Has Dream Nail? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDreamNail", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → Enable Dream Effects
    on FALSE → Inert
  }

  state Inert {
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Has Dream Nail?
  }

  state Idle {
    ActivateGameObject(gameObject=Godseeker Waterways/Godseeker Fall/Dream Enter, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Enable Dream Effects {
    ActivateGameObject(gameObject=Godseeker Waterways/Godseeker Fall/Dream Effects, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Land Effects {
    ActivateGameObject(gameObject=Godseeker Waterways/Godseeker Fall/Slam Dust, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Godseeker Waterways/Shiny Item Godfinder, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=false_knight_land_1st_time (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=godseeker_ground_hit (sharedassets225.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Has Dream Nail?
  }

  var Self: gameObject = <null>
  var Slam Effect: gameObject = <null>
}
