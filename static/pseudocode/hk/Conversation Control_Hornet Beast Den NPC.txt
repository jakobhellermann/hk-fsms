# Conversation Control
# 1 location(s):
#   Deepnest_Spider_Town / Hornet Beast Den NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Defeated in Outskirts?
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="HORNET", everyFrame=false)
    on FINISHED → Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim End")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on L → L
    on R → R
  }

  state Greet {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Hornet_Dialogue_Generic_05 (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Audio Player")
    SetPlayerDataBool(boolName="hornetDenEncounter", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["HORNET_SPIDER_TOWN_01", "Hornet"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Choice {
    SetAudioVolume(gameObject=var "Audio Player", volume=0, everyFrame=false)
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"GREET", everyFrame=false)
    on GREET → Greet
    on REPEAT → Repeat
  }

  state Repeat {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Hornet_Dialogue_Generic_03 (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Audio Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["HORNET_SPIDER_TOWN_REPEAT", "Hornet"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Den Talk L")
    SetStringValue(stringVariable=var "Anim End", stringValue="Den End L", everyFrame=false)
    on FINISHED → Box Up
  }

  state R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Den Talk R")
    SetStringValue(stringVariable=var "Anim End", stringValue="Den End R", everyFrame=false)
    on FINISHED → Box Up
  }

  state Defeated in Outskirts? {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hegemolDefeated", falseBool="hornetOutskirtsDefeated", isTrue=→"ABSENT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hornetOutskirtsDefeated", isTrue=(none), isFalse=→"ABSENT")  // disabled
    on ABSENT → Never Meet
    on FINISHED → Idle
  }

  state Never Meet {
    SetPlayerDataBool(boolName="hornetDenEncounter", value=true)
    ActivateGameObject(gameObject=Self, activate=false, recursive=true, resetOnExit=false, everyFrame=false)
  }

  var Guardians Defeated: int = 0
  var Royal Charm State: int = 0
  var Spoken: bool = false
  var Anim End: string = ""
  var Area Title: gameObject = <null>
  var Audio Player: gameObject = <null>
  var Self: gameObject = <null>
}
