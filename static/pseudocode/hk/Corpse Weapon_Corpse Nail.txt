# Corpse Weapon
# 3 location(s):
#   sharedassets54.assets / Corpse Nail
#   sharedassets88.assets / Corpse Ruins Sentry 1/Corpse Nail
#   sharedassets90.assets / Corpse Ruins Flying Sentry/Corpse Nail

fsm Corpse Weapon {
  start Initialise

  state Detect {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=true)
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"COLLISION ENTER 2D", storeCollider=var "", storeForce=(unset))
    on COLLISION ENTER 2D → Check
  }

  state Check {
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=var "Contact Point", contactNormal=var "Normal", shapeCount=(unset), physics2dMaterialName="")
    GetVector3XYZ(vector3Variable=var "Normal", storeX=var "Contact X", storeY=var "Contact Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Contact Y", float2=1, tolerance=0, equal=→"HIT GROUND", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Detect
    on HIT GROUND → Ground Check
  }

  state Ground Check {
    SetFloatValue(floatVariable=var "Clamp Min", floatValue=80, everyFrame=false)
    SetFloatValue(floatVariable=var "Clamp Max", floatValue=120, everyFrame=false)
    FloatInRange(floatVariable=var "Angle", lowerValue=var "Clamp Min", upperValue=var "Clamp Max", boolVariable=var "In Range", trueEvent=(none), falseEvent=(none), everyFrame=false)
    BoolTest(boolVariable=var "In Range", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Detect
    on TRUE → Stick
  }

  state Initialise {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Detect
  }

  state Stick {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Thunk")
    CreateObject(gameObject=Dust Land Small (sharedassets32.assets), spawnPoint=<null>, position=var "Contact Point", rotation=(unset), storeObject=var "Dust", networkInstantiate=false, networkGroup=0)
    CreateObject(gameObject=Thunk Effect (sharedassets33.assets), spawnPoint=var "", position=var "Contact Point", rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    GetPosition(gameObject=var "Self", vector=(unset), x=(unset), y=(unset), z=var "Self Z", space=0, everyFrame=false)
    SetPosition(gameObject=var "Dust", vector=(unset), x=(unset), y=(unset), z=var "Self Z", space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=0, y=0, everyFrame=true)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=true)
    RandomFloat(min=80, max=90, storeResult=var "Clamp Min")
    RandomFloat(min=110, max=120, storeResult=var "Clamp Max")
    FloatClamp(floatVariable=var "Angle", minValue=var "Clamp Min", maxValue=var "Clamp Max", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    SetIsFixedAngle2d(gameObject=var "Self", isFixedAngle=true, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetProperty(targetProperty=ObjectBounce on <null>, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Dormant
  }

  state Dormant {
  }

  var Normal: vector3 = (0, 0, 0)

  var Angle: float = 0
  var Chooser: float = 0
  var Clamp Max: float = 0
  var Clamp Min: float = 0
  var Contact X: float = 0
  var Contact Y: float = 0
  var Self Z: float = 0
  var In Range: bool = false
  var Contact Point: vector3 = (0, 0, 0)
  var Dust: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Self: gameObject = <null>
}
