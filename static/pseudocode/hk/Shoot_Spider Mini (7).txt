# Shoot
# 2 location(s):
#   Deepnest_03 / Spider Mini (7)
#   Deepnest_16 / Spider Mini (7)

fsm Shoot {
  start Pause

  state Init {
    GetHero(storeResult=var "Hero")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Hero Range", isTrue=→"RANGE", isFalse=(none), everyFrame=true)
    on RANGE → Wait
  }

  state Wait {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recheck
  }

  state Recheck {
    GetFsmBool(gameObject=Self, fsmName="Spider", variableName="Walking", storeValue=var "Walking", everyFrame=false)
    BoolTest(boolVariable=var "Walking", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    BoolTest(boolVariable=var "Hero Range", isTrue=→"RANGE", isFalse=→"CANCEL", everyFrame=false)
    on RANGE → Check Orientation 2
    on CANCEL → Idle
  }

  state Antic {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVESTOP", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mantis_flyer_child_awake (sharedassets176.assets), pitchMin=0.8, pitchMax=0.8, volume=1, delay=0, storePlayer=<null>)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=var "Glow", active=true)
    Tk2dPlayAnimation(gameObject=var "Glow", animLibName="", clipName="Glow Antic")
    Tk2dPlayFrame(gameObject=var "Glow", frame=0)
    on FINISHED → Antic 2
  }

  state Fire {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit Fire", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Spitter Shot R (sharedassets32.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Shot")
    FireAtTarget(gameObject=var "Shot", target=var "Hero", speed=15, position=(unset), spread=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetMeshRenderer(gameObject=var "Glow", active=false)
    on FINISHED → Spit
  }

  state Movestart {
    SetScale(gameObject=Self, vector=(unset), x=var "Prev X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVESTART", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Antic 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spit Antic 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=var "Glow", animLibName="", clipName="Glow Antic 2")
    on FINISHED → Check Orientation
  }

  state Recover {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Check Orientation {
    IntSwitch(intVariable=var "Orientation", compareTo=[0, 1, 2, 3], sendEvent=[→"FLOOR", →"L WALL", →"ROOF", →"R WALL"], everyFrame=false)
    on FLOOR → Floor
    on L WALL → L Wall
    on ROOF → Roof
    on R WALL → R Wall
  }

  state Floor {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"SCALE -1", leftEvent=→"SCALE 1", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 1
    on SCALE -1 → Scale -1
  }

  state Scale 1 {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Fire
  }

  state Scale -1 {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Fire
  }

  state L Wall {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"SCALE 1", belowEvent=→"SCALE -1", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 1
    on SCALE -1 → Scale -1
  }

  state R Wall {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"SCALE -1", belowEvent=→"SCALE 1", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 1
    on SCALE -1 → Scale -1
  }

  state Roof {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"SCALE 1", leftEvent=→"SCALE -1", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 1
    on SCALE -1 → Scale -1
  }

  state Check Orientation 2 {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Prev X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    IntSwitch(intVariable=var "Orientation", compareTo=[0, 1, 2, 3], sendEvent=[→"FLOOR", →"L WALL", →"ROOF", →"R WALL"], everyFrame=false)
    on FLOOR → Floor 2
    on L WALL → L Wall 2
    on ROOF → Roof 2
    on R WALL → R Wall 2
  }

  state Floor 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"SCALE -1", leftEvent=→"SCALE 1", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 2
    on SCALE -1 → Scale -1 2
  }

  state L Wall 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"SCALE 1", belowEvent=→"SCALE -1", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 2
    on SCALE -1 → Scale -1 2
  }

  state R Wall 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"SCALE -1", belowEvent=→"SCALE 1", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 2
    on SCALE -1 → Scale -1 2
  }

  state Roof 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"SCALE 1", leftEvent=→"SCALE -1", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on SCALE 1 → Scale 2
    on SCALE -1 → Scale -1 2
  }

  state Scale 2 {
    SetScale(gameObject=Self, vector=(unset), x=0.85, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Antic
  }

  state Scale -1 2 {
    SetScale(gameObject=Self, vector=(unset), x=-0.85, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Antic
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Spit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=spitter_spit (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetRotation(gameObject=var "Shot", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "AngleToHero", space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Dribble Spread Min", floatValue=var "AngleToHero", everyFrame=false)
    SetFloatValue(floatVariable=var "Dribble Spread Max", floatValue=var "AngleToHero", everyFrame=false)
    FloatAdd(floatVariable=var "Dribble Spread Max", add=25, everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Dribble Spread Min", subtract=25, everyFrame=false, perSecond=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=5, speedMin=10, speedMax=20, angleMin=var "Dribble Spread Min", angleMax=var "Dribble Spread Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    on FINISHED → Movestart
  }

  var AngleToHero: float = 0
  var Dribble Spread Max: float = 0
  var Dribble Spread Min: float = 0
  var Prev X Scale: float = 0
  var Orientation: int = 0
  var Hero Range: bool = false
  var Walking: bool = false
  var Glow: gameObject = Spider Mini (7)/Glow
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Shot: gameObject = <null>
  var Spit Effect: gameObject = Spider Mini (7)/Spit Effect
}
