# Attack
# 1 location(s):
#   Fungus3_10 / Battle Scene/Wave 2/Garden Zombie

fsm Attack {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Spawn Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Hero in range", isTrue=→"HERO IN RANGE", isFalse=(none), everyFrame=true)
    on HERO IN RANGE → Attack Pause
  }

  state Attack Start {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=garden_zombie_prepare (sharedassets203.assets))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack Start")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Attack Fire
  }

  state Attack Fire {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fire
  }

  state Attack Recover {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack Recover")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Fire {
    FlingObjectsFromGlobalPool(gameObject=Shot Spine (sharedassets203.assets), spawnPoint=var "Self", position=var "Shot Origin", spawnMin=1, spawnMax=1, speedMin=var "Speed Min", speedMax=var "Speed Max", angleMin=165, angleMax=180, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=garden_zombie_shoot (sharedassets203.assets))
    FlingObjectsFromGlobalPool(gameObject=Shot Spine (sharedassets203.assets), spawnPoint=var "Self", position=var "Shot Origin", spawnMin=1, spawnMax=1, speedMin=var "Speed Min", speedMax=var "Speed Max", angleMin=130, angleMax=145, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Shot Spine (sharedassets203.assets), spawnPoint=var "Self", position=var "Shot Origin", spawnMin=1, spawnMax=1, speedMin=var "Speed Min", speedMax=var "Speed Max", angleMin=90, angleMax=110, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Shot Spine (sharedassets203.assets), spawnPoint=var "Self", position=var "Shot Origin", spawnMin=1, spawnMax=1, speedMin=var "Speed Min", speedMax=var "Speed Max", angleMin=90, angleMax=75, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Shot Spine (sharedassets203.assets), spawnPoint=var "Self", position=var "Shot Origin", spawnMin=1, spawnMax=1, speedMin=var "Speed Min", speedMax=var "Speed Max", angleMin=35, angleMax=55, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Shot Spine (sharedassets203.assets), spawnPoint=var "Self", position=var "Shot Origin", spawnMin=1, spawnMax=1, speedMin=var "Speed Min", speedMax=var "Speed Max", angleMin=0, angleMax=15, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    on FINISHED → Attack Recover
  }

  state Movestart {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on FINISHED → Idle
  }

  state Wait {
    Tk2dPlayAnimationV2(gameObject=Self, animLibName="", clipName="Idle", doNotResetCurrentClip=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Movestart
  }

  state Attack Pause {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack Start
  }

  state Spawn Idle {
    BoolTest(boolVariable=var "Spawner", isTrue=(none), isFalse=→"ACTIVE", everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spawn Idle")
    SetCollider(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    BoolAllTrue(boolVariables=[var "Hero in range", var "Self Spawn"], sendEvent=→"SPAWN", storeResult=(unset), everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on ACTIVE → Idle
    on SPAWN → Spawn Antic
  }

  state Spawn Antic {
    PlayParticleEmitter(gameObject=var "Shake Grass", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spawn Antic")
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_rustle (sharedassets137.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Spawn
  }

  state Spawn {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopParticleEmitter(gameObject=var "Shake Grass")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_burst_leaves (sharedassets34.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVESTART", delay=0, everyFrame=false)  // disabled
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on FINISHED → Idle
  }

  var Speed Max: float = 25
  var Speed Min: float = 20
  var Self Spawn: bool = false
  var Spawner: bool = true
  var Shot Origin: vector3 = (0, -0.5, 0)

  var Hero in range: bool = false
  var Self: gameObject = <null>
  var Shake Grass: gameObject = Battle Scene/Wave 2/Garden Zombie/Shake Grass
}
