# Control
# 1 location(s):
#   White_Palace_01 / White Palace Gate

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Closed", storeResult=var "Closed")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool", isTrue=→"WHITE GATE OPEN", isFalse=(none))
    on WHITE GATE OPEN → Bool2?
    on FINISHED → Idle
  }

  state Idle {
    on WHITE GATE OPEN → Open Pause
  }

  state Opened {
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Open Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Bool2? {
    StringCompare(stringVariable=var "PD Bool 2", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool 2", isTrue=→"FINISHED", isFalse=→"CANCEL")
    on FINISHED → Opened
    on CANCEL → Idle
  }

  var Start Open: bool = false
  var PD Bool: string = "whitePalaceOrb_1"
  var PD Bool 2: string = ""

  var Closed: gameObject = White Palace Gate/Closed
  var Open: gameObject = White Palace Gate/Open
}
