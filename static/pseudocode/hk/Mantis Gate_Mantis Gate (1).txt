# Mantis Gate
# 2 location(s):
#   Fungus2_04 / Mantis Gate (1)
#   Fungus2_15 / Mantis Gate (1)

fsm Mantis Gate {
  start Pause
  on ACTIVATE → Activate  // from any state
  on OPEN → Open Pause  // from any state

  state Idle {
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Activate {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Open {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Door Open 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopParticleEmitter(gameObject=var "Dust")
    on FINISHED → Open 2
  }

  state Open 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Door Open 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Opened
  }

  state Opened {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Open Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Dust
  }

  state Dust {
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Open
  }

  var Dust: gameObject = Mantis Gate (1)/Dust
  var Self: gameObject = <null>
}
