# Get Cut
# 1 location(s):
#   Ruins2_03 / chandelier/chandelier_top/rope_top

fsm Get Cut {
  start Idle

  state Idle {
    on CUT → Cut
  }

  state Cut {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.01, 1, 1), time=0.25, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Remove
  }

  state Remove {
    SetSpriteRenderer(gameObject=Self, active=false)
  }
}
