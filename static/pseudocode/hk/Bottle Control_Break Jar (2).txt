# Bottle Control
# 1 location(s):
#   Ruins2_11 / Break Jar (2)

fsm Bottle Control {
  start Pause

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on NAIL HIT → Shatter
    on ACTIVATE → Activate
  }

  state Shatter {
    DestroyObject(gameObject=var "Phys Box", delay=0, detachChildren=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[crystal_break_1 (resources.assets), crystal_break_2 (resources.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    PlayParticleEmitter(gameObject=var "Pt Glass L", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Glass S", emit=0)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BREAK", delay=0, everyFrame=false)
    on FINISHED → Destroy Self
  }

  state Destroy Self {
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Lid", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObject(flungObject=var "Lid", speedMin=7, speedMax=8, angleMin=85, angleMax=95)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pool
  }

  state Activate {
    SpawnRandomObjects(gameObject=Bottle Lid (sharedassets33.assets), spawnPoint=var "Self", position=(0, -0.5, 0), spawnMin=1, spawnMax=1, speedMin=0, speedMax=0, angleMin=85, angleMax=95, originVariation=0)
    DestroySelf(detachChildren=false)
  }

  state Pool {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Activated: bool = false
  var Lid: gameObject = Break Jar (2)/Bottle Lid
  var Phys Box: gameObject = Break Jar (2)/Bottle Physical
  var Pt Glass L: gameObject = Break Jar (2)/Pt Glass L
  var Pt Glass S: gameObject = Break Jar (2)/Pt Glass S
  var Self: gameObject = <null>
}
