# Check Active
# 1 location(s):
#   resources.assets / _GameCameras/HudCamera/Inventory/Inv/Inv_Items/Spell Fireball

fsm Check Active {
  start Pause Frame

  state Check {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasNailArt", isTrue=→"ART", isFalse=→"NO ART")
    on NO ART → Pos Down
    on ART → Pos Up
  }

  state Pos Down {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Down Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Check Spell?
  }

  state Pos Up {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Up Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Check Spell?
  }

  state Check Spell? {
    BoolTest(boolVariable=var "Check Spell", isTrue=→"CHECK", isFalse=(none), everyFrame=false)
    on CHECK → Appear?
  }

  state Appear? {
    GetPlayerDataInt(gameObject=var "GameManager", intName=var "Spell Name", storeValue=var "Spell Level")
    IntCompare(integer1=var "Spell Level", integer2=1, equal=→"ACTIVE", lessThan=→"INACTIVE", greaterThan=→"ACTIVE", everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UP", delay=0, everyFrame=false)
    IntCompare(integer1=var "Spell Level", integer2=1, equal=→"LEVEL 1", lessThan=(none), greaterThan=→"LEVEL 2", everyFrame=false)
    on INACTIVE → Inactive
    on LEVEL 1 → Lv 1
    on LEVEL 2 → Lv 2
  }

  state Inactive {
    SetSpriteRenderer(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SetSpriteRendererSprite(gameObject=Self, sprite=Inv_Empty)
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Lv 1 {
    SetSpriteRendererSprite(gameObject=Self, sprite=Inv_0025_spell_fireball_01)
  }

  state Lv 2 {
    SetSpriteRendererSprite(gameObject=Self, sprite=Inv_0025_spell_fireball_01_level2)
  }

  var Down Y: float = -7.46
  var Up Y: float = -6.21
  var Check Spell: bool = true
  var Spell Name: string = "fireballLevel"

  var Spell Level: int = 0
  var Self: gameObject = <null>
}
