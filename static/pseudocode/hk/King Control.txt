# King Control
# 1 location(s):
#   Crossroads_38 / Grub King

fsm King Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init Check
  }

  state Init Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="fatGrubKing", isTrue=→"FAT", isFalse=(none))
    FindChild(gameObject=Self, childName="Audio Point", storeResult=var "Audio Point")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=StoryRecord_visited("Grub King"))
    GetPlayerDataInt(gameObject=var "GameManager", intName="grubsCollected", storeValue=var "Grubs Saved")
    GetPlayerDataInt(gameObject=var "GameManager", intName="grubRewards", storeValue=var "Rewards Given")
    IntCompare(integer1=var "Grubs Saved", integer2=0, equal=→"ALONE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Grubs Saved", integer2=var "Rewards Given", equal=→"ALL GIVEN", lessThan=(none), greaterThan=(none), everyFrame=false)
    on ALONE → Alone
    on ALL GIVEN → All Given
    on FINISHED → Beckon
    on FAT → Fat
  }

  state Alone {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Cry")
  }

  state All Given {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
  }

  state Beckon {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Audio Point", audioClip=Grub_King_wave (sharedassets68.assets), pitchMin=1, pitchMax=1, volume=1, delay=1, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Beckon")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=0, tolerance=0, equal=(none), lessThan=→"GIVE", greaterThan=(none), everyFrame=true)
    FloatInRange(floatVariable=var "Hero X", lowerValue=30, upperValue=42, boolVariable=(unset), trueEvent=→"GIVE", falseEvent=(none), everyFrame=true)
    on GIVE → Cheer?
  }

  state Gift Anim {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Audio Point", audioClips=[Grub_king_cheer_03 (sharedassets68.assets), Grub_king_cheer_02 (sharedassets68.assets)], weights=[1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gift", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Activate Reward
  }

  state Activate Reward {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="grubRewards")
    GetPlayerDataInt(gameObject=var "GameManager", intName="grubRewards", storeValue=var "Rewards Given")
    ConvertIntToString(intVariable=var "Rewards Given", stringVariable=var "Reward String", format="", everyFrame=false)
    BuildString(stringParts=["Reward ", var "Reward String"], separator="", addToEnd=true, storeResult=var "Reward Name", everyFrame=false)
    DebugLog(logLevel=1, text=var "Reward Name", sendToUnityLog=false)  // disabled
    FindChild(gameObject=var "Rewards Parent", childName=var "Reward Name", storeResult=var "Reward Folder")
    SendEventByName(eventTarget=GameObject(var "Reward Folder"), sendEvent="GIVE", delay=0, everyFrame=false)
    on FINISHED → Final Reward?
  }

  state Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gift Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recover Pause
  }

  state Recover Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=1.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recheck
  }

  state Recheck {
    GetPlayerDataInt(gameObject=var "GameManager", intName="grubsCollected", storeValue=var "Grubs Saved")
    GetPlayerDataInt(gameObject=var "GameManager", intName="grubRewards", storeValue=var "Rewards Given")
    IntCompare(integer1=var "Grubs Saved", integer2=0, equal=→"ALONE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Grubs Saved", integer2=var "Rewards Given", equal=→"ALL GIVEN", lessThan=(none), greaterThan=(none), everyFrame=false)
    on ALONE → 
    on ALL GIVEN → All Given
    on FINISHED → At cap?
  }

  state Cheer? {
    BoolTest(boolVariable=var "Given", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Given", boolValue=true, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Audio Point", audioClips=[Grub_king_cheer_01 (sharedassets68.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Cheer", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Cheer Pause
    on GIVE → Gift Anim
  }

  state Cheer Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gift Anim
  }

  state Final Reward? {
    IntCompare(integer1=var "Rewards Given", integer2=46, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetPlayerDataBool(boolName="finalGrubRewardCollected", value=true)
    on FINISHED → Recover
  }

  state Fat {
    SetParent(gameObject=var "Rewards Parent", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state At cap? {
    CheckGeoCap(IsUnderCapEvent=→"FALSE", IsAtCapEvent=→"TRUE", StoreIsUnderCap=(unset), StoreIsAtCap=(unset), EveryFrame=false)
    on TRUE → Recover Pause
    on FALSE → Gift Anim
  }

  var Hero X: float = 0
  var Grubs Saved: int = 0
  var Rewards Given: int = 0
  var Given: bool = false
  var Reward Name: string = ""
  var Reward String: string = ""
  var Audio Point: gameObject = <null>
  var Reward Folder: gameObject = <null>
  var Rewards Parent: gameObject = Grub King/Rewards Parent
}
