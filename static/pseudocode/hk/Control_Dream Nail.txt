# Control
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Extras/Dream Nail
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Extras/Dream Nail

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Amount", storeResult=var "Amount")
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite")
    on FINISHED → Out
  }

  state Out {
    on DREAM ORB COLLECT → Pos Y
    on DREAM NAIL HUD → Pos Y
  }

  state Activate {
    SetMeshRenderer(gameObject=var "Sprite", active=true)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="DN In")
    Tk2dPlayFrame(gameObject=var "Sprite", frame=0)
    SendEventByName(eventTarget=GameObject(var "Amount"), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Update
  }

  state Idle {
    Wait(time=8, finishEvent=→"DOWN", realTime=false)
    on DREAM ORB COLLECT → Anim
    on LEVEL LOADED → Deactivate
    on DOWN → Deactivate
  }

  state Update {
    GetPlayerDataInt(gameObject=var "GameManager", intName="dreamOrbs", storeValue=var "Orbs Int")
    ConvertIntToString(intVariable=var "Orbs Int", stringVariable=var "Orbs String", format="", everyFrame=false)
    SetTextMeshProText(gameObject=var "Amount", textString=var "Orbs String")
    on FINISHED → Achievement?
  }

  state Anim {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="DN Get")
    Tk2dPlayFrame(gameObject=var "Sprite", frame=0)
    on FINISHED → Update
  }

  state Deactivate {
    SendEventByName(eventTarget=GameObject(var "Amount"), sendEvent="DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="DN Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on DREAM ORB COLLECT → Pos Y
    on FINISHED → Sprite Off
  }

  state Sprite Off {
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    on FINISHED → Out
  }

  state Pos Y {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=-0.5, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_5", isTrue=(none), isFalse=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=-1.42, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Activate
  }

  state Achievement? {
    IntCompare(integer1=var "Orbs Int", integer2=600, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["ATTUNEMENT"], storeResult=(unset var))
    on FINISHED → Idle
  }

  var Orbs Int: int = 0
  var Orbs String: string = ""
  var Amount: gameObject = <null>
  var Sprite: gameObject = <null>
}
