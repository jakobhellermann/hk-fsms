# Blessing Control
# 2 location(s):
#   Knight_Pickup / Knight/Blessing Ghost
#   resources.assets / Knight:1/Blessing Ghost

fsm Blessing Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Blessing_Ghost", storeResult=var "Blessing_Ghost")
    FindChild(gameObject=var "Blessing_Ghost", childName="Particle appear", storeResult=var "Particle appear")
    FindChild(gameObject=var "Blessing_Ghost", childName="Particle rain", storeResult=var "Particle rain")
    GetPosition(gameObject=Self, vector=var "Init Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on BENCHREST → Blessed?
    on STARTED BENCHREST → Blessed?
  }

  state Blessed? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="salubraBlessing", isTrue=→"BLESSED", isFalse=→"UNBLESSED")
    on BLESSED → Blessed Pause
    on UNBLESSED → Idle
  }

  state Blessed Pause {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Blessing
    on BENCHREST END → Idle
    on HERO DAMAGED → End
    on LEVEL LOADED → End
  }

  state Start Blessing {
    ActivateGameObject(gameObject=var "Blessing_Ghost", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeAudio(gameObject=var "Blessing_Ghost", startVolume=0, endVolume=1, time=0.1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetSpriteRenderer(gameObject=var "Blessing_Ghost", active=true)
    PlayParticleEmitter(gameObject=var "Particle rain", emit=0)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Fill Soul
    on BENCHREST END → End
    on HERO DAMAGED → End
    on LEVEL LOADED → End
  }

  state Fill Soul {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="TryAddMPChargeSpa", parameters=[2], storeResult=var "Did Add")
    BoolTest(boolVariable=var "Did Add", isTrue=(none), isFalse=→"FULL", everyFrame=false)
    on FINISHED → Fill Pause
    on BENCHREST END → End
    on HERO DAMAGED → End
    on LEVEL LOADED → End
    on FULL → Fill End Pause
  }

  state Fill Pause {
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fill Soul
    on BENCHREST END → End
    on HERO DAMAGED → End
    on LEVEL LOADED → End
  }

  state End {
    SetSpriteRenderer(gameObject=var "Blessing_Ghost", active=false)
    PlayParticleEmitter(gameObject=var "Particle appear", emit=0)
    StopParticleEmitter(gameObject=var "Particle rain")
    FadeAudio(gameObject=var "Blessing_Ghost", startVolume=1, endVolume=0, time=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    SetParent(gameObject=Self, parent=var "Hero", resetLocalPosition=true, resetLocalRotation=true)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=Self, vector=var "Init Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Blessing_Ghost", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Fill End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
    on BENCHREST END → End
    on HERO DAMAGED → End
    on LEVEL LOADED → End
  }

  var Did Add: bool = false
  var Init Pos: vector3 = (0, 0, 0)
  var Blessing_Ghost: gameObject = <null>
  var Particle appear: gameObject = <null>
  var Particle rain: gameObject = <null>
}
