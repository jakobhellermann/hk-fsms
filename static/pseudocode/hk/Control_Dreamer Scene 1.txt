# Control
# 2 location(s):
#   Fungus1_04_boss / Dreamer Scene 1
#   GG_Hornet_1 / Dreamer Scene 1

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="dreamerScene1", isTrue=→"ACTIVATE", isFalse=(none))
    CreateObject(gameObject=Cutscene Dreamer (sharedassets133.assets), spawnPoint=<null>, position=(42.83, -0.51, 0), rotation=(unset), storeObject=var "Hegemol", networkInstantiate=false, networkGroup=0)
    SetFsmString(gameObject=var "Hegemol", fsmName="Control", variableName="Name", setValue="Hegemol", everyFrame=false)
    CreateObject(gameObject=Cutscene Dreamer (sharedassets133.assets), spawnPoint=<null>, position=(44.29, -2.86, 0), rotation=(unset), storeObject=var "Lurien", networkInstantiate=false, networkGroup=0)
    SetFsmString(gameObject=var "Lurien", fsmName="Control", variableName="Name", setValue="Lurien", everyFrame=false)
    CreateObject(gameObject=Cutscene Dreamer (sharedassets133.assets), spawnPoint=<null>, position=(39.9, 1.02, 0), rotation=(unset), storeObject=var "Monomon", networkInstantiate=false, networkGroup=0)
    SetFsmString(gameObject=var "Monomon", fsmName="Control", variableName="Name", setValue="Monomon", everyFrame=false)
    on ACTIVATE → Inert
    on FINISHED → Idle
  }

  state Inert {
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Take Control
  }

  state Take Control {
    SetPlayerDataBool(boolName="dreamerScene1", value=true)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    on FINISHED → Blast
  }

  state Blast {
    CreateObject(gameObject=dream_area_effect (sharedassets6.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hero_fury_charm_burst (resources.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hero_quake_spell_impact (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Init Blast", vector=var "Hero Pos", x=(unset), y=(unset), z=0.1, space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Init Blast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Magical Fall")
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=-10, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    on FINISHED → Hero Fall
  }

  state Hero Fall {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Knight Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Knight Y", float2=var "Knight Scene Y", tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Dreamer Land")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up
  }

  state Lurien Appear {
    SetPosition(gameObject=var "Lurien", vector=(unset), x=23.37, y=33.05, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetScale(gameObject=var "Lurien", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Lurien"), sendEvent="FORM", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Monomon Appear
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP DREAM", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo 1
  }

  state Convo 1 {
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=false, topRight=false, topCentre=true, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DREAMERS_EC1_1", "Dreamers"], storeResult=(unset var))
    on CONVO_FINISH → Convo 2
  }

  state Box Down {
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN DREAM", delay=0, everyFrame=false)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Lurien Appear
  }

  state Monomon Appear {
    SetPosition(gameObject=var "Monomon", vector=(unset), x=30.64, y=35.15, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=1.15, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SetScale(gameObject=var "Monomon", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Monomon"), sendEvent="FORM", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hegemol Appear
  }

  state Hegemol Appear {
    SetPosition(gameObject=var "Hegemol", vector=(unset), x=19.22, y=34.92, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    SetScale(gameObject=var "Hegemol", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Hegemol"), sendEvent="FORM", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Casting
  }

  state Casting {
    SendEventByName(eventTarget=GameObject(var "Hegemol"), sendEvent="CAST", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=ghost_dialogue_death_initial_white (sharedassets46.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Lurien"), sendEvent="CAST", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Monomon"), sendEvent="CAST", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=true, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0.1, everyFrame=false)
    on FINISHED → Blast Down
  }

  state Fade Out {
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Remove Dreamers
  }

  state Fade In {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DREAM AREA DISABLE", delay=0, everyFrame=false)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=4.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Aftermath", emit=0)
    on FINISHED → Get Up
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Prostrate Rise", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End
  }

  state End {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
  }

  state Remove Dreamers {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=false, everyFrame=false)
    DestroyObject(gameObject=var "Hegemol", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Monomon", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Lurien", delay=0, detachChildren=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Prostrate")
    on FINISHED → Fade In
  }

  state Convo 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DREAMERS_EC1_2", "Dreamers"], storeResult=(unset var))
    on CONVO_FINISH → Convo 3
  }

  state Convo 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DREAMERS_EC1_3", "Dreamers"], storeResult=(unset var))
    on CONVO_FINISH → Convo 4
  }

  state Convo 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DREAMERS_EC1_4", "Dreamers"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state Blast Down {
    FindChild(gameObject=Self, childName="Pt Blast 1", storeResult=var "Pt Blast 1")
    FindChild(gameObject=Self, childName="Pt Blast 2", storeResult=var "Pt Blast 2")
    FindChild(gameObject=Self, childName="Pt Aftermath", storeResult=var "Pt Aftermath")
    PlayParticleEmitter(gameObject=var "Pt Blast 1", emit=0)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=radiance_burst_white_stuff_move_up (sharedassets133.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Blast Impact
  }

  state Blast Impact {
    SpawnObjectFromGlobalPool(gameObject=White Flash R (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    PlayParticleEmitter(gameObject=var "Pt Blast 2", emit=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=ghost_absorb_final_impact (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=boss_explode_clean (resources.assets), pitchMin=1.15, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Fade Out
  }

  var Knight Scene Y: float = 28.5
  var Knight Y: float = 0
  var Hero Pos: vector3 = (0, 0, 0)
  var Hegemol: gameObject = <null>
  var Init Blast: gameObject = Dreamer Scene 1/Init Blast
  var Lurien: gameObject = <null>
  var Monomon: gameObject = <null>
  var Pt Aftermath: gameObject = <null>
  var Pt Blast 1: gameObject = <null>
  var Pt Blast 2: gameObject = <null>
}
