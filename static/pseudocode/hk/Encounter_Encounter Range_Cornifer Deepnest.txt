# Encounter
# 2 location(s):
#   Fungus2_25 / Cornifer Deepnest/Encounter Range
#   Deepnest_01b / Cornifer Deepnest/Encounter Range

fsm Encounter {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Set
  }

  state Set {
    SetPlayerDataBool(boolName=var "Encountered Bool", value=true)
    SetPlayerDataBool(boolName=var "Deepnest Met", value=true)
  }

  var Deepnest Met: string = ""
  var Encountered Bool: string = ""
}
