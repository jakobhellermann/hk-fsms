# Fade Control
# 1 location(s):
#   resources.assets / dream_gate_object

fsm Fade Control {
  start Init
  on DREAM GATE SET → Disappear  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Away
  }

  state Away {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"CLOSE", storeCollider=var "")
    SetParticleEmissionRate(gameObject=var "Particle System", emissionRate=10, everyFrame=false)
    FadeGroupUp(target=Self)
    on CLOSE → Close
  }

  state Close {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"AWAY", storeCollider=var "")
    SetParticleEmissionRate(gameObject=var "Particle System", emissionRate=4, everyFrame=false)
    FadeGroupDown(target=Self, fast=false)
    on AWAY → Away
  }

  state Disappear {
  }

  var Disappear: gameObject = dream_gate_object/Disappear
  var Dream_gate: gameObject = dream_gate_object/Dream_Gate
  var Other Pieces: gameObject = dream_gate_object/Dream_Gate_Other_pieces_0001_2 (1)
  var Particle System: gameObject = dream_gate_object/Particle System
  var beam: gameObject = dream_gate_object/beam
  var beam (1): gameObject = dream_gate_object/beam (1)
  var haze: gameObject = dream_gate_object/haze2 (1)
}
