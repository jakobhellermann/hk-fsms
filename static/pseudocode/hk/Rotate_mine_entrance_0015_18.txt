fsm Rotate {
  start Idle

  state Idle {
    on MOVING UP → Moving Up
    on MOVING DOWN → Moving Down
  }

  state Moving Up {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=500, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=true)
    on MOVING DOWN → Moving Down
    on STOP MOVING → Idle
  }

  state Moving Down {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-500, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=true)
    on MOVING UP → Moving Up
    on STOP MOVING → Idle
  }
}
