# Control
# 1 location(s):
#   Fungus2_31 / Lords Undefeated Group

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Gate", storeResult=var "Gate")
    FindChild(gameObject=Self, childName="Hit", storeResult=var "Hit")
    FindChild(gameObject=Self, childName="Lever", storeResult=var "Lever")
    FindChild(gameObject=Self, childName="Mantis", storeResult=var "Mantis")
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="encounteredGatekeeper", isTrue=→"ENCOUNTERED", isFalse=(none))
    on FINISHED → Idle
    on ENCOUNTERED → Encountered
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Mantis
  }

  state Mantis {
    SetPlayerDataBool(boolName="encounteredGatekeeper", value=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Mantis", clipName="Quick Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hit
  }

  state Hit {
    Tk2dPlayAnimation(gameObject=var "Mantis", animLibName="", clipName="Slash Recover")
    ActivateGameObject(gameObject=var "Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Lever", animLibName="", clipName="Lever Unhit")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_switch (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Close
  }

  state Close {
    SetMeshRenderer(gameObject=var "Gate", active=true)
    Tk2dPlayAnimation(gameObject=var "Gate", animLibName="", clipName="Gate Close")
  }

  state Encountered {
    SetMeshRenderer(gameObject=var "Gate", active=true)
    Tk2dPlayAnimation(gameObject=var "Lever", animLibName="", clipName="Lever Activated")
    Tk2dPlayAnimation(gameObject=var "Gate", animLibName="", clipName="Gate Idle")
  }

  var Gate: gameObject = <null>
  var Hit: gameObject = <null>
  var Lever: gameObject = <null>
  var Mantis: gameObject = <null>
  var Self: gameObject = <null>
}
