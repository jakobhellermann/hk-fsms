# Border Display
# 3 location(s):
#   Ruins1_29 / Stag Map/Border
#   RestingGrounds_09 / Stag Map/Border
#   resources.assets / Stag Map/Border

fsm Border Display {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Stag_Border_Bottom", storeResult=var "Bot")
    FindChild(gameObject=Self, childName="Stag_Border_Top", storeResult=var "Top")
    SetMeshRenderer(gameObject=var "Top", active=false)
    SetMeshRenderer(gameObject=var "Bot", active=false)
    on UP → Up
  }

  state Up {
    SetMeshRenderer(gameObject=var "Top", active=true)
    SetMeshRenderer(gameObject=var "Bot", active=true)
    Tk2dPlayAnimation(gameObject=var "Bot", animLibName="", clipName="Stag_Border_Bottom Up")
    Tk2dPlayAnimation(gameObject=var "Top", animLibName="", clipName="Stag_Border_Top Up")
    on DOWN → Down
  }

  state Down {
    Tk2dPlayAnimation(gameObject=var "Bot", animLibName="", clipName="Stag_Border_Bottom Down")
    Tk2dPlayAnimation(gameObject=var "Top", animLibName="", clipName="Stag_Border_Top Down")
    on UP → Up
  }

  var Bot: gameObject = <null>
  var Top: gameObject = <null>
}
