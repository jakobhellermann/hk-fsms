// uses template: enemy_message
fsm enemy_message {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"TOUCH", storeCollider=var "Collider")
    Collision2dEvent(collision=0, collideTag="", sendEvent=→"TOUCH", storeCollider=var "Collider", storeForce=(unset))
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"TOUCH", storeCollider=var "Collider")
    on TOUCH → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent=var "Event", delay=0, everyFrame=false)
    SendEnemyMessage(Target=var "Collider", EventString=var "Event")
    on FINISHED → Idle
  }

  var Event: string = "GO RIGHT"

  var Collider: gameObject = <null>
  var Enemy: gameObject = <null>
}
