# Close
# 1 location(s):
#   Room_Colosseum_01 / Colosseum Hatch

fsm Close {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Close Anim", storeResult=var "Close Anim")
    on FINISHED → Idle
  }

  state Idle {
    on GATE OPEN → Close
  }

  state Close {
    ActivateGameObject(gameObject=var "Close Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=true)
  }

  var Close Anim: gameObject = <null>
}
