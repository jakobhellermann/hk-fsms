fsm Anvil Tink {
  start Init

  state Idle {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame ID", everyframe=true)
    IntCompare(integer1=var "Frame ID", integer2=25, equal=→"PLAY", lessThan=(none), greaterThan=(none), everyFrame=true)
    on PLAY → Play
  }

  state Play {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=nailsmith_anvil (sharedassets16.assets))
    on FINISHED → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  var Frame ID: int = 0
  var Self: gameObject = <null>
}
