fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=false)
    RandomFloatEither(value1=-1, value2=1, storeResult=var "X Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=-5, max=5, storeResult=var "Rotation")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    on SPIKES READY → Ready
  }

  state Ready {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spike Ready")
    on SPIKES UP → Antic
    on SPIKES DOWN → Cancel
  }

  state Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spike Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Up
    on SPIKES DOWN → Cancel
  }

  state Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spike Up")
    SetPolygonCollider(gameObject=Self, active=true)
    on SPIKES DOWN → Down
  }

  state Down {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spike Down", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false)
    on FINISHED → Dormant
  }

  state Cancel {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spike Cancel", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false)
    on FINISHED → Dormant
  }

  var Rotation: float = 0
  var X Scale: float = 1
}
