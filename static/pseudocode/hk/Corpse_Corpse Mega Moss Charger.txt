# Corpse
# 1 location(s):
#   sharedassets156.assets / Corpse Mega Moss Charger

fsm Corpse {
  start Save Death?

  state In Air {
    GetOwner(storeGameObject=var "Self")
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Steam
  }

  state Steam {
    SpawnBloodTime(delay=0.1, spawnPoint=var "Self", position=(0, 0, 0.002), spawnMin=12, spawnMax=15, speedMin=24, speedMax=30, angleMin=0, angleMax=360, colorOverride=(unset))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_gushing (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CreateObject(gameObject=Death Puff Boss (sharedassets32.assets), spawnPoint=var "Self", position=(0, -1, -6), rotation=(unset), storeObject=var "Smoker", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    FloatAdd(floatVariable=var "emissionRate", add=5, everyFrame=true, perSecond=false)
    FloatAdd(floatVariable=var "emissionSpeed", add=1, everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "emissionSpeed", minValue=0, maxValue=110, everyFrame=true)
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Ready
  }

  state Blow {
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_explode (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="StopRumble", delay=0, everyFrame=false)
    CreateObject(gameObject=Death Explode Boss (sharedassets32.assets), spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Smoker", emission=false)
    PlayParticleEmitter(gameObject=var "Grass Explode", emit=0)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Nibblers", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → 
  }

  state Ready {
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    EaseFloat(fromValue=var "emissionRate", toValue=0, floatVariable=var "emissionRate", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "emissionSpeed", toValue=0, floatVariable=var "emissionSpeed", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Blow
  }

  state Save Death? {
    CheckCurrentMapZone(mapZone="DREAM_WORLD", equalEvent=→"NO", notEqualEvent=→"YES")
    on YES → Save
    on NO → In Air
  }

  state Save {
    SetPlayerDataBool(boolName="megaMossChargerDefeated", value=true)
    on FINISHED → In Air
  }

  var emissionRate: float = 50
  var emissionSpeed: float = 5
  var Battle Scene: gameObject = <null>
  var Camera: gameObject = <null>
  var Grass Explode: gameObject = Corpse Mega Moss Charger/Grass Explode
  var Nibblers: gameObject = Corpse Mega Moss Charger/Nibblers
  var Self: gameObject = <null>
  var Smoker: gameObject = <null>
}
