# Activate Boss
# 1 location(s):
#   GG_Brooding_Mawlek_V / Battle Scene

fsm Activate Boss {
  start Init

  state Init {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FindChild(gameObject=Self, childName="Mawlek Body", storeResult=var "Mawlek")
    FindChild(gameObject=Self, childName="Tiso Boss", storeResult=var "Tiso")
    on FINISHED → In Pantheon?
  }

  state Call Mawlek {
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Mawlek", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GG MUSIC", delay=0, everyFrame=false)  // disabled
    ApplyMusicCue(musicCue=GG Sad (sharedassets425.assets), delayTime=0, transitionTime=0)  // disabled
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    on FINISHED → Music Up
  }

  state In Pantheon? {
    GGCheckIfBossSequence(trueEvent=→"PANTHEON", falseEvent=→"STATUE")
    on PANTHEON → Tiso?
    on STATUE → Call Mawlek 2
    on FINISHED → Tiso?
  }

  state Tiso {
    ActivateGameObject(gameObject=var "Tiso", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Mawlek", fsmName="Mawlek Control", variableName="Skip Title", setValue=true, everyFrame=false)
    SetFsmGameObject(gameObject=var "Godseeker", fsmName="Control", variableName="Target", setValue=var "Tiso", everyFrame=false)
    on TISO END → Mawlek Fall
  }

  state Mawlek Fall {
    ActivateGameObject(gameObject=var "Mawlek", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPosition(gameObject=var "Mawlek", vector=(unset), x=(unset), y=var "Mawlek Pos", z=(unset), space=1, everyFrame=true)
    FloatCompare(float1=var "Mawlek Pos", float2=-25.12, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Crush Tiso
  }

  state Crush Tiso {
    SendEventByName(eventTarget=GameObject(var "Tiso"), sendEvent="CRUSHED", delay=0, everyFrame=false)
    ApplyMusicCue(musicCue=None (resources.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MAWLEK", everyFrame=false)
    on FINISHED → Music
  }

  state Tiso Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tiso
  }

  state Music {
    ApplyMusicCue(musicCue=GG Sad (sharedassets425.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Action and Sub (resources.assets), transitionTime=0.1)
    SetFsmGameObject(gameObject=var "Godseeker", fsmName="Control", variableName="Target", setValue=var "Hero", everyFrame=false)
  }

  state Music Up {
    ApplyMusicCue(musicCue=GG Sad (sharedassets425.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Action and Sub (resources.assets), transitionTime=1)
  }

  state Tiso? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="tisoEncounteredColosseum", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → Tiso Pause
    on FALSE → Call Mawlek
  }

  state Call Mawlek 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Mawlek", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GG MUSIC", delay=0, everyFrame=false)  // disabled
    ApplyMusicCue(musicCue=GG Sad (sharedassets425.assets), delayTime=0, transitionTime=0)  // disabled
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    on FINISHED → Music Up 2
  }

  state Music Up 2 {
    ApplyMusicCue(musicCue=EnemyBattle (sharedassets40.assets), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=1)
  }

  var Mawlek Pos: float = 0
  var Area Title: gameObject = <null>
  var Godseeker: gameObject = <null>
  var Mawlek: gameObject = <null>
  var Tiso: gameObject = <null>
}
