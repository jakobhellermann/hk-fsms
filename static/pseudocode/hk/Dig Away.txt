fsm Dig Away {
  start Away

  state Away {
    BoolTest(boolVariable=var "In Range", isTrue=→"IN", isFalse=(none), everyFrame=true)
    WaitRandom(timeMin=8, timeMax=12, finishEvent=→"DIG", realTime=false)
    on IN → In
    on DIG → Dig
  }

  state In {
    BoolTest(boolVariable=var "In Range", isTrue=(none), isFalse=→"AWAY", everyFrame=true)
    on AWAY → Away
  }

  state Dig {
    SetFsmBool(gameObject=Self, fsmName="Centipede", variableName="Will Dig", setValue=true, everyFrame=false)
    on FINISHED → Away
  }

  var In Range: bool = false
}
