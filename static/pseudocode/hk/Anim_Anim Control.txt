# Anim
# 1 location(s):
#   RestingGrounds_07 / Dream Moth/Anim Control

fsm Anim {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Left
  }

  state Left {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="Look Left")
    on EXIT → Right
  }

  state Right {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Tk2dPlayAnimation(gameObject=var "Parent", animLibName="", clipName="Look Right")
    on ENTER → Left
  }

  var Parent: gameObject = <null>
}
