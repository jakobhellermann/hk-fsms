# Hit effect
# 1 location(s):
#   Fungus3_archive_02 / Dreamer Monomon/Shield

fsm Hit effect {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Shield Hit", storeResult=var "Shield Hit")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"BLOCKED HIT", storeCollider=var "")
    on BLOCKED HIT → Hit
  }

  state Hit {
    SendEventByName(eventTarget=GameObject(var "Shield Hit"), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  state Down {
    SendEventByName(eventTarget=GameObject(var "Shield Hit"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Shield Hit: gameObject = <null>
}
