# Control
# 1 location(s):
#   GG_Mighty_Zote / Battle Control/Extra Zotes

fsm Control {
  start Init

  state Init {
    on FINISHED → Inactive
  }

  state Inactive {
    on START → Wait
  }

  state Wait {
    RandomWait(min=1, max=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send
  }

  state Send {
    GetRandomChild(gameObject=Self, storeResult=var "Next Zote")
    GameObjectIsNull(gameObject=var "Next Zote", isNull=→"CANCEL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Next Zote"), sendEvent="GO", delay=0, everyFrame=false)
    on CANCEL → Wait
    on RETURN → Wait
  }

  var Next Zote: gameObject = <null>
}
