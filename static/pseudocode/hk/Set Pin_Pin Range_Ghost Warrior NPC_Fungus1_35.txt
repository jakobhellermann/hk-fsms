    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Set
  }

  state Set {
    SetPlayerDataBool(boolName=var "Pinned Bool", value=true)
  }

  var Pinned Bool: string = "noEyesPinned"
}
