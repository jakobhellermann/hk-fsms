# Conversation Control
# 1 location(s):
#   Room_Colosseum_02 / Zote Colosseum

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Check Direction
  }

  state Convo Choice {
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="zoteSpokenColosseum", isTrue=→"GREET", isFalse=→"MEET")
    on MEET → Meet
    on GREET → Greet
    on REPEAT → Repeat
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    AudioStop(gameObject=var "Voice Player")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    on FINISHED → Check Direction 2
  }

  state Check Direction {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Talk R
    on LEFT → Talk Back
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk L")
    on FINISHED → Box Up
  }

  state Talk Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk R")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ZOTE", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Check Direction 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Face Right
    on LEFT → Face Left
  }

  state Face Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Right")
    on FINISHED → End
  }

  state Face Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Left")
    on FINISHED → End
  }

  state Meet {
    SetPlayerDataBool(boolName="zoteSpokenColosseum", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_COLOSSEUM_MEET", "Zote"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_05 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Greet {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_COLOSSEUM_GREET", "Zote"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_05 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_COLOSSEUM_REPEAT", "Zote"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_03 (sharedassets31.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  var NPC State: int = 0
  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
