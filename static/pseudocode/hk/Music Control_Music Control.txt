# Music Control
# 1 location(s):
#   resources.assets / Hollow Shade:0/Music Control

fsm Music Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Shade
  }

  state Shade {
    TransitionToAudioSnapshot(snapshot=Close, transitionTime=var "Transition To Time")
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Return
  }

  state Return {
    TransitionToAudioSnapshot(snapshot=Away, transitionTime=var "Transition Back Time")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Shade
  }

  var Transition Back Time: float = 3.5
  var Transition To Time: float = 2.5
}
