fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="brokenMageWindow", falseBool="brokenMageWindowGlass", isTrue=→"ACTIVATE", isFalse=(none))
    on ACTIVATE → Activate
  }

  state Activate {
    GetOwner(storeGameObject=var "Self")
    SetPlayerDataBool(boolName="brokenMageWindowGlass", value=true)
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  var Self: gameObject = <null>
}
