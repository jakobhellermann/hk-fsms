# Sequence Control
# 1 location(s):
#   GG_Door_5_Finale / abyss_door_5_cutscene_sequence

fsm Sequence Control {
  start Start

  state Start {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    on FINISHED → Wait
  }

  state End {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Next Scene
  }

  state Wait {
    Wait(time=21, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade
  }

  state Fade {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Next Scene
  }

  state Next Scene {
    SetStaticVariable(variableName="skipEndWhiteFader", setValue=true)
    GetConstantsValue(variableName="BOSSRUSH_END_SCENE", storeValue=var "End Scene")
    BeginSceneTransition(sceneName=var "End Scene", entryGateName="", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=false)
  }

  var End Scene: string = ""
}
