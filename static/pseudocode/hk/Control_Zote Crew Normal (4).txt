# Control
# 1 location(s):

  on HIT RIGHT → Hit Right  // from any state
  on HIT LEFT → Hit Left  // from any state
  on HIT UP → Hit Up  // from any state
  on ZERO HP → Death  // from any state

  state Init {

    FindChild(gameObject=Self, childName="Pt Run", storeResult=var "Pt Run")
    FindChild(gameObject=Self, childName="Pt Fall", storeResult=var "Pt Fall")
    FindChild(gameObject=Self, childName="Bonker", storeResult=var "Bonker")
    FindChild(gameObject=Self, childName="Attack Voice", storeResult=var "Attack Voice")
    FindChild(gameObject=Self, childName="Spawn Lines", storeResult=var "Spawn Lines")
    ActivateGameObject(gameObject=var "Bonker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetHP(target=Self, storeValue=var "HP")
    on FINISHED → Dormant
  }

  state Tumble Out {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll")

    on FINISHED → In Air
  }

  state In Air {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Prev X Speed", y=(unset), space=0, everyFrame=false)

    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=true)
    on WALL → Reflect
    on LAND → Grav
  }

  state Reflect {

    on FINISHED → In Air
  }

  state Fall Over {

    StopParticleEmitter(gameObject=var "Pt Run")
    ActivateGameObject(gameObject=var "Bonker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Get Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    WaitRandom(timeMin=var "Idle Time Min", timeMax=var "Idle Time Max", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move Choice
  }

  state Move Choice {
    SendRandomEventV2(events=[→"ATTACK", →"JUMP"], weights=[0.35, 0.65], trackingInts=[var "Ct Attack", var "Ct Jump"], eventMax=[2, 3])
    on ATTACK → Attack Antic
    on JUMP → Jump Aim
  }

  state Jump Aim {
    RandomFloat(min=-18, max=18, storeResult=var "Jump X")

    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)

    on FINISHED → Jump Antic
  }

  state Jump Antic {

  }

  state Trip? {

    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
