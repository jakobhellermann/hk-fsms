# Battle Control
# 1 location(s):
#   Crossroads_08 / Battle Scene

fsm Battle Control {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Wave 1
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Wave 2", storeResult=var "Wave 2")
    FindGameObject(objectName="CameraLockArea B", withTag="Untagged", store=var "Camera Lock")
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activate
  }

  state Wave 1 {
    TransitionToAudioSnapshot(snapshot=Action (resources.assets), transitionTime=1)
    SetIntValue(intVariable=var "Battle Enemies", intValue=3, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=2, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    ActivateGameObject(gameObject=var "Camera Lock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on NEXT → Wave Pause
  }

  state End Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=5)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    ActivateGameObject(gameObject=var "Camera Lock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wave Pause {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave 2
  }

  state Wave 2 {
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    ActivateGameObject(gameObject=var "Wave 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Wave 2"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on NEXT → End Wait
  }

  var Battle Enemies: int = 0
  var Activated: bool = false
  var Camera Lock: gameObject = <null>
  var Moss Knight: gameObject = <null>
  var Self: gameObject = <null>
  var Wave 1: gameObject = <null>
  var Wave 2: gameObject = <null>
}
