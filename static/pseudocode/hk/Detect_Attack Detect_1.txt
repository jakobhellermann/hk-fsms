fsm Detect {
  start Detect

  state Detect {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="ATTACKED", delay=0, everyFrame=false)
  }

  var Parent: gameObject = <null>
}
