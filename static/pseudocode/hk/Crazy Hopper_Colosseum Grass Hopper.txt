# Crazy Hopper
# 1 location(s):
#   sharedassets34.assets / Colosseum Grass Hopper

fsm Crazy Hopper {
  start Init

  state Idle {
    SetParticleEmissionRate(gameObject=var "Grass Trail", emissionRate=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Land")
    WaitRandom(timeMin=0.15, timeMax=0.25, finishEvent=→"JUMP", realTime=false)
    SetRecoilSpeed(target=Self, newRecoilSpeed=0)
    on JUMP → Jump Antic
  }

  state Init {
    FindChild(gameObject=Self, childName="Grass Trail", storeResult=var "Grass Trail")
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite")
    on FINISHED → Jump
  }

  state Jump Antic {
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Jump Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Jump
  }

  state Jump {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Grass Trail", audioClip=grass_hopper_jump, pitchMin=0.85, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    SetParticleEmissionRate(gameObject=var "Grass Trail", emissionRate=50, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.5)
    RandomFloat(min=var "Angle Min", max=var "Angle Max", storeResult=var "Angle")
    RandomFloat(min=var "Speed Min", max=var "Speed Max", storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Jump")
    NextFrameEvent(sendEvent=→"FINISHED")
    SetRecoilSpeed(target=Self, newRecoilSpeed=12)
    on FINISHED → In Air
  }

  state In Air {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Sprite Rotation", everyFrame=true)
    SetRotation(gameObject=var "Sprite", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Sprite Rotation", space=0, everyFrame=true, lateUpdate=false)
    Rotate(gameObject=var "Sprite", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-90, space=1, perSecond=false, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"UP", rightHitEvent=→"RIGHT", bottomHitEvent=→"DOWN", leftHitEvent=→"LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"UP", rightHitEvent=→"RIGHT", bottomHitEvent=→"DOWN", leftHitEvent=→"LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on RIGHT → Land Right
    on LEFT → Land Left
    on UP → Land Up
    on DOWN → Land Down
  }

  state Land Right {
    SetRotation(gameObject=var "Sprite", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=90, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=135, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=225, everyFrame=false)
    on FINISHED → Idle
  }

  state Land Left {
    SetRotation(gameObject=var "Sprite", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=270, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=-45, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=45, everyFrame=false)
    on FINISHED → Idle
  }

  state Land Down {
    SetRotation(gameObject=var "Sprite", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=45, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=135, everyFrame=false)
    on FINISHED → Idle
  }

  state Land Up {
    SetRotation(gameObject=var "Sprite", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=180, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=225, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=315, everyFrame=false)
    on FINISHED → Idle
  }

  var Angle: float = 0
  var Angle Max: float = 135
  var Angle Min: float = 45
  var Speed: float = 0
  var Speed Max: float = 25
  var Speed Min: float = 20
  var Sprite Rotation: float = 0
  var Grass Trail: gameObject = <null>
  var Sprite: gameObject = <null>
}
