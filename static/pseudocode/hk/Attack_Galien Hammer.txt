# Attack
# 2 location(s):
#   Deepnest_40 / Warrior/Galien Hammer
#   GG_Ghost_Galien / Warrior/Galien Hammer

fsm Attack {
  start Idle
  on GHOST DEATH → End  // from any state

  state Idle {
    GetOwner(storeGameObject=var "Self")
    on HAMMER ATTACK → Antic
  }

  state Antic {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 3, 0), time=1, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    FindGameObject(objectName="Ghost Warrior Galien", withTag="Untagged", store=var "Ghost Warrior Galien")
    on FINISHED → Chase
  }

  state Chase {
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=10, accelerationForce=50, offsetX=0, offsetY=-0.5)  // disabled
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Floor Y", tolerance=0, equal=→"FLOOR", lessThan=→"FLOOR", greaterThan=(none), everyFrame=true)
    ChaseObjectGround(gameObject=Self, target=var "Hero", speedMax=20, acceleration=0.5, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0)
    AccelerateVelocity(gameObject=Self, xAccel=(unset), yAccel=-1, xMaxSpeed=(unset), yMaxSpeed=999)
    FloatCompare(float1=var "Self X", float2=var "Wall L X", tolerance=0, equal=→"WALL L", lessThan=→"WALL L", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Wall R X", tolerance=0, equal=→"WALL R", lessThan=(none), greaterThan=→"WALL R", everyFrame=true)
    FloatAdd(floatVariable=var "Active Timer", add=1, everyFrame=true, perSecond=true)
    FloatTestToBool(float1=var "Active Timer", float2=10, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Timer Expired", everyFrame=true)
    FloatTestToBool(float1=var "Self Y", float2=21.06, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Over Topline", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Over Topline", var "Timer Expired"], sendEvent=→"DECEL", storeResult=(unset), everyFrame=true)
    on FLOOR → Floor Bounce
    on WALL L → Wall L
    on WALL R → Wall R
    on DECEL → Decel
  }

  state Floor Bounce {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetAudioVolume(gameObject=var "Audio Player", volume=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[false_knight_land (sharedassets32.assets), false_knight_land_1st_time (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=var "Audio Player")
    NextFrameEvent(sendEvent=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Floor Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=26, max=32, storeResult=var "Ran Float")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Ran Float", everyFrame=true)
    SpawnObjectFromGlobalPool(gameObject=Slam Effect R (resources.assets), spawnPoint=var "Self", position=(0, -1.5, 0), rotation=(0, 0, 0), storeObject=var "")
    on FINISHED → Chase
  }

  state Wall L {
    SetAudioVolume(gameObject=var "Audio Player", volume=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[<null>, <null>], weights=[1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=var "Audio Player")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=var "Wall L X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    RandomFloat(min=26, max=32, storeResult=var "Ran Float")  // disabled
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=var "Y Velocity", space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=-0.75, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=true)
    FloatClamp(floatVariable=var "X Velocity", minValue=1, maxValue=99999, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Slam Effect R (resources.assets), spawnPoint=var "Self", position=(-1.5, 0, 0), rotation=(unset), storeObject=var "Slam")
    SetRotation(gameObject=var "Slam", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-90, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Wall Y
  }

  state Wall R {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetAudioVolume(gameObject=var "Audio Player", volume=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[<null>, <null>], weights=[1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=var "Audio Player")
    NextFrameEvent(sendEvent=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=var "Wall R X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=-0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    RandomFloat(min=26, max=32, storeResult=var "Ran Float")  // disabled
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=var "Y Velocity", space=0, everyFrame=false)
    FloatClamp(floatVariable=var "X Velocity", minValue=-9999, maxValue=-1, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=-0.75, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=true)
    SpawnObjectFromGlobalPool(gameObject=Slam Effect R (resources.assets), spawnPoint=var "Self", position=(1.5, 0, 0), rotation=(unset), storeObject=var "Slam")
    SetRotation(gameObject=var "Slam", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=90, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Wall Y
  }

  state Wall Y {
    FloatCompare(float1=var "Y Velocity", float2=5, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[<null>, <null>], weights=[1, 1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=5, everyFrame=false)
    on FINISHED → Chase
  }

  state Decel {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Weapon Idle")
    DecelerateV2(gameObject=Self, deceleration=0.89)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DECEL", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Spin Speed", floatValue=-800, everyFrame=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    FloatAddV2(floatVariable=var "Spin Speed", add=1600, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=(unset))
    on FINISHED → Recel
  }

  state Recel {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="RECEL", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Active Timer", floatValue=0, everyFrame=false)
    on FINISHED → Anim
  }

  state Anim {
    Tk2dPlayAnimation(gameObject=var "Ghost Warrior Galien", animLibName="", clipName="AttackFullIdle")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Ghost Warrior Galien", audioClips=[Galien_attack_01 (sharedassets298.assets), Galien_attack_02 (sharedassets298.assets), Galien_attack_03 (sharedassets298.assets), Galien_attack_04 (sharedassets298.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Chase
  }

  state End {
  }

  var Active Timer: float = 0
  var Floor Y: float = 14.88
  var Ran Float: float = 0
  var Self X: float = 0
  var Self Y: float = 0
  var Slam Y: float = 15.24
  var Spin Speed: float = 0
  var Wall L X: float = 36.08
  var Wall R X: float = 68.89
  var X Velocity: float = 0
  var Y Velocity: float = 0
  var Over Topline: bool = false
  var Timer Expired: bool = false
  var Audio Player: gameObject = <null>
  var Ghost Warrior Galien: gameObject = <null>
  var Self: gameObject = <null>
  var Slam: gameObject = <null>
}
