# Shrink N Destroy
# 1 location(s):
#   resources.assets / skull_chip1

fsm Shrink N Destroy {
  start Wait
  on HAZARD RELOAD → Destroy  // from any state

  state Wait {
    WaitRandom(timeMin=5, timeMax=5.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shrink
  }

  state Shrink {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.1, 0.1, 0.1), time=0.25, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var X Speed: float = 0
}
