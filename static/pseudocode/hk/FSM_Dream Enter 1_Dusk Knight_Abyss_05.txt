// uses template: deactivate_ifnot_playerdatabool
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=(none), isFalse=→"DEACTIVATE")
    on DEACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var playerData bool: string = "dreamNailUpgraded"
}
