# Control
# 1 location(s):
#   Dream_Abyss / abyss_door

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindChild(gameObject=Self, childName="Collider", storeResult=var "Collider")
    FindChild(gameObject=Self, childName="Gate", storeResult=var "Gate")
    FindChild(gameObject=Self, childName="HeroLight", storeResult=var "HeroLight")
    FindChild(gameObject=Self, childName="Spark 1", storeResult=var "Spark 1")
    FindChild(gameObject=Self, childName="Spark 2", storeResult=var "Spark 2")
    FindChild(gameObject=Self, childName="White Flash", storeResult=var "White Flash")
    FindChild(gameObject=Self, childName="Gate Orbs", storeResult=var "Gate Orbs")
    FindChild(gameObject=Self, childName="Shatter Orbs", storeResult=var "Shatter Orbs")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="abyssGateOpened", isTrue=→"OPENED", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasKingsBrand", isTrue=→"READY", isFalse=(none))
    on OPENED → Opened
    on READY → Ready
  }

  state Opened {
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Ready {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Open Pause
  }

  state Open Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetPlayerDataBool(boolName="abyssGateOpened", value=true)
    on FINISHED → Open Start
  }

  state Open Start {
    Tk2dPlayAnimation(gameObject=var "Gate", animLibName="", clipName="Open Antic")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Spark 1", emit=0)
    PlayParticleEmitter(gameObject=var "Spark 2", emit=0)
    SendEventByName(eventTarget=GameObject(var "HeroLight"), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Opening
  }

  state Opening {
    Tk2dPlayAnimation(gameObject=var "Gate", animLibName="", clipName="Opening")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Orbing
  }

  state Orbing {
    PlayParticleEmitter(gameObject=var "Gate Orbs", emit=0)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Gate"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Shatter
  }

  state Shatter {
    Tk2dPlayAnimationWithEvents(gameObject=var "Gate", clipName="Disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Shatter Orbs", emit=0)
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate Orbs", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HeroLight"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Gate Off
  }

  state Gate Off {
    ActivateGameObject(gameObject=var "Gate", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Collider: gameObject = <null>
  var Gate: gameObject = <null>
  var Gate Orbs: gameObject = <null>
  var HeroLight: gameObject = <null>
  var Shatter Orbs: gameObject = <null>
  var Spark 1: gameObject = <null>
  var Spark 2: gameObject = <null>
  var White Flash: gameObject = <null>
}
