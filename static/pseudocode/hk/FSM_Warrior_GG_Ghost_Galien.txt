# FSM
# 1 location(s):
#   GG_Ghost_Galien / Warrior

fsm FSM {
  start Wait

  state Wait {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hammer Rise
  }

  state Enable {
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Enable GameObject", position=(unset), rotation=(unset), storeObject=var "")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Enable GameObject", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Enable GameObject", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByNameV2(eventTarget=GameObject(var "Hammer"), sendEvent="READY", delay=0, everyFrame=false)
    SendEventByNameV2(eventTarget=GameObject(var "Hammer"), sendEvent="READY", delay=0, everyFrame=false)  // disabled
  }

  state Hammer Rise {
    SendEventByNameV2(eventTarget=GameObject(var "Hammer"), sendEvent="READY", delay=0, everyFrame=false)
    Wait(time=2.1, finishEvent=(none), realTime=false)
    on FINISHED → Enable
  }

  var Wait Time: float = 1.5
  var Enable GameObject: gameObject = Warrior/Ghost Warrior Galien
  var Hammer: gameObject = Warrior/Galien Hammer
}
