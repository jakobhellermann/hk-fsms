# Rage Check
# 4 location(s):
#   Deepnest_East_10 / Warrior/Ghost Warrior Markoth
#   Deepnest_East_16 / Warrior/Ghost Warrior Markoth
#   GG_Ghost_Markoth / Warrior/Ghost Warrior Markoth
#   GG_Ghost_Markoth_V / Warrior/Ghost Warrior Markoth

fsm Rage Check {
  start Wait

  state Idle {
    on TOOK DAMAGE → Check
  }

  state Check {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Rage HP", equal=→"RAGE", lessThan=→"RAGE", greaterThan=(none), everyFrame=false)
    on RAGE → Set Rage
    on FINISHED → Idle
  }

  state Set Rage {
    SetFsmBool(gameObject=Self, fsmName="Attacking", variableName="Rage", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Shield Attack", variableName="Rage", setValue=true, everyFrame=false)
  }

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set HP
  }

  state Set HP {
    GetHP(target=Self, storeValue=var "Rage HP")
    IntOperator(integer1=var "Rage HP", integer2=2, operation=Divide, storeResult=var "Rage HP", everyFrame=false)
    on FINISHED → Idle
  }

  var HP: int = 0
  var Rage HP: int = 0
}
