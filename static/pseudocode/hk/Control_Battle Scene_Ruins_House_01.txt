# Control
# 1 location(s):
#   Ruins_House_01 / Battle Scene

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Great Shield Zombie", storeResult=var "Zombie")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"DEAD", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Activate
    on DEAD → Already Dead
  }

  state Activate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Action and Sub (resources.assets), transitionTime=1)
    on KILLED → Killed Pause
  }

  state Already Dead {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Killed Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Open
  }

  state Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Sub Area (resources.assets), transitionTime=3)
  }

  var Activated: bool = false
  var Zombie: gameObject = <null>
}
