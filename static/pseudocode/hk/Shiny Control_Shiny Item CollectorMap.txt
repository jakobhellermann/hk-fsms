# Shiny Control
# 1 location(s):
#   Ruins2_11 / Shiny Item CollectorMap

fsm Shiny Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="Inspect Region", storeResult=var "Inspect Region")
    FindChild(gameObject=Self, childName="Appear Trail", storeResult=var "Trail")
    on FINISHED → PD Bool?
    on ACTIVATE → Destroy
  }

  state PD Bool? {
    StringCompare(stringVariable=var "PD Bool Name", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool Name", isTrue=→"COLLECTED", isFalse=(none))
    on FINISHED → Fling?
    on COLLECTED → Destroy
  }

  state Idle {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopBounce())
    SetFsmBool(gameObject=var "Inspect Region", fsmName="inspect", variableName="Inspectable", setValue=true, everyFrame=false)
    on START INSPECT → Hero Down
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Charm? {
    BoolTest(boolVariable=var "Charm", isTrue=→"CHARM", isFalse=(none), everyFrame=false)
    on CHARM → 
    on FINISHED → Big Item?
  }

  state Hero Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect SD 1")
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Big Get Flash
    on HERO DAMAGED → Idle
  }

  state Hero Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect SD 1 Back", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Tute Msg"), sendEvent="CLOSE", delay=0, everyFrame=false)
    on FINISHED → Finish
    on HERO DAMAGED → Finish
    on DREAM EXIT → 
  }

  state Finish {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Inspect Region", fsmName="inspect", variableName="Inspectable", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Tute Msg"), sendEvent="CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Inspect Region"), sendEvent="END INSPECT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHINY PICKED UP", delay=0, everyFrame=false)
  }

  state Fling? {
    BoolTest(boolVariable=var "Fling On Start", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayParticleEmitter(gameObject=var "Trail", emit=0)
    SetFsmBool(gameObject=var "Inspect Region", fsmName="inspect", variableName="Inspectable", setValue=false, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.85)
    BoolTest(boolVariable=var "Slug Fling", isTrue=→"SLUG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Fling L", isTrue=→"FLING L", isFalse=(none), everyFrame=false)
    SendRandomEvent(events=[→"FLING L", →"FLING R"], weights=[1, 1], delay=0)
    on FINISHED → Idle
    on FLING L → Fling L
    on FLING R → Fling R
    on SLUG → Slug Fling
  }

  state Fling R {
    FlingObject(flungObject=Self, speedMin=25, speedMax=25, angleMin=82, angleMax=86)
    on FINISHED → In Air
  }

  state Fling L {
    FlingObject(flungObject=Self, speedMin=20, speedMax=20, angleMin=98, angleMax=102)
    on FINISHED → In Air
  }

  state In Air {
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=true)
    FloatCompare(float1=var "Speed", float2=1, tolerance=0, equal=(none), lessThan=→"STOPPED", greaterThan=(none), everyFrame=true)
    on STOPPED → Land
  }

  state Land {
    SetGravity2dScale(gameObject=Self, gravityScale=0)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Trail")
    on FINISHED → Idle
  }

  state Big Item? {
    BoolTest(boolVariable=var "Dash Cloak", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Mantis Claw", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Journal", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Super Dash", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Quake", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Pure Seed", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "King's Brand", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    on BIG → Big Get Flash
    on FINISHED → 
  }

  state Get {
    SetPlayerDataBool(boolName="hasPinGrub", value=true)
    SendMessage(gameObject=var "", delivery=SendMessage, options=1, functionCall=SetupMap())
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=collectors_map_prompt (sharedassets124.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="CollectorMap", everyFrame=false)
    on GET ITEM MSG END → Finish
  }

  state Big Get Flash {
    CreateObject(gameObject=Item Get Effect (sharedassets7.assets), spawnPoint=var "Hero", position=(0, -0.76, -1), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=shiny_item_pickup (resources.assets))
    SetSpriteRenderer(gameObject=var "Self", active=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 2")  // disabled
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="KINGS BRAND", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect StandToIdle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Get
  }

  state Slug Fling {
    FlingObject(flungObject=Self, speedMin=20, speedMax=20, angleMin=70, angleMax=70)
    on FINISHED → In Air
  }

  var Charm ID: int = 0
  var Trinket Num: int = 0
  var Charm: bool = false
  var Dash Cloak: bool = false
  var Exit Dream: bool = false
  var Fling On Start: bool = false
  var Journal: bool = false
  var King's Brand: bool = false
  var Mantis Claw: bool = false
  var Pure Seed: bool = false
  var Quake: bool = false
  var Show Charm Tute: bool = true
  var Super Dash: bool = false
  var PD Bool Name: string = ""

  var Speed: float = 0
  var Type: int = 0
  var Activated: bool = false
  var Fling L: bool = false
  var Slug Fling: bool = false
  var Item Name: string = ""
  var Return Door: string = "door_dreamReturn"
  var Return Scene: string = "Crossroads_10"
  var Inspect Region: gameObject = <null>
  var Msg: gameObject = <null>
  var Msg Icon: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Self: gameObject = <null>
  var Trail: gameObject = <null>
  var Tute Msg: gameObject = <null>
  var UI Msg: gameObject = <null>
}
