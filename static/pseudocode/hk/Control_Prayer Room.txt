# Control
# 1 location(s):
#   Crossroads_10_boss_defeated / Prayer Room

fsm Control {
  start Pause

  state Pause {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="falseKnightDefeated", isTrue=(none), isFalse=→"DISABLE")
    on DISABLE → Disable
  }

  state Disable {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Self: gameObject = <null>
}
