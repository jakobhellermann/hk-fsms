# Hit
# 1 location(s):
#   Deepnest_East_16 / Training Dummy

fsm Hit {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(collideTag="", sendEvent=→"NAIL HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    on NAIL HIT → Get Damager Parameters
  }

  state Get Damager Parameters {
    AudioPlayRandom(gameObject=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.8, pitchMax=1)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="PLAY", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    GetTag(gameObject=var "Damager", storeResult=var "Tag", everyFrame=false)
    StringCompare(stringVariable=var "Tag", compareTo="Hero Spell", equalEvent=→"HEAVY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    GetName(gameObject=var "Damager", storeName=var "Name", everyFrame=false)
    StringCompare(stringVariable=var "Name", compareTo="Great Slash", equalEvent=→"HEAVY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Name", compareTo="Dash Slash", equalEvent=→"HEAVY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on HEAVY → Heavy Dir
    on FINISHED → Light Dir
  }

  state Heavy Dir {
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 225], sendEvent=[→"HIT R", →"HIT L"], everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Heavy L
    on R → Heavy R
    on HIT L → 
    on HIT R → 
  }

  state Light Dir {
    FloatCompare(float1=var "Attack Direction", float2=0, tolerance=0, equal=→"HIT R", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Attack Direction", float2=180, tolerance=0, equal=→"HIT L", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Light L
    on R → Light R
    on HIT L → Recoil R
    on HIT R → Recoil L
  }

  state Heavy L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Heavy Hit L")
    on FINISHED → Summon?
  }

  state Recover {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Heavy R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Heavy Hit R")
    on FINISHED → Summon?
  }

  state Light L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Light Hit L")
    on FINISHED → Summon?
  }

  state Light R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Light Hit R")
    on FINISHED → Summon?
  }

  state Summon? {
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SUMMON", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Recoil R {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RecoilRight", parameters=[], storeResult=(unset var))
    on FINISHED → Light L
  }

  state Recoil L {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RecoilLeft", parameters=[], storeResult=(unset var))
    on FINISHED → Light R
  }

  var Hits: int = 25

  var Attack Direction: float = 0
  var Name: string = ""
  var Tag: string = ""
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
