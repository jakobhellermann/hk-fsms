// component disabled: this FSM does not run until something enables it
fsm Music Control {
  start Pause Frame

  state Init {
    on FINISHED → Delay
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Start {
    TransitionToAudioSnapshot(snapshot=var "Music Snapshot", transitionTime=var "Transition Time")
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent=var "Track Event", delay=0, everyFrame=false)
    on EXIT → Started
  }

  state Started {
    on EXIT → 
  }

  state Delay {
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start
  }

  var Delay: float = 0
  var Transition Time: float = 2
  var Track Event: string = "GREENPATH"
  var Music Snapshot: object = Normal (resources.assets)
}
