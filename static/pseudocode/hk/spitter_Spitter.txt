// component disabled: this FSM does not run until something enables it
fsm spitter {
  start Idle

  state Idle {
    SetAudioPitch(gameObject=Self, pitch=0.8, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    Tk2dPlayFrame(gameObject=Self, frame=2)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    IdleBuzz(gameObject=Self, waitMin=0.75, waitMax=1, speedMax=1.75, accelerationMax=15, roamingRange=1)
    CheckCanSeeHero(storeResult=var "Can See Hero", everyFrame=true)
    CheckAlertRangeByName(storeResult=var "In Alert Range", childName="Alert Range New", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Alert Range"], sendEvent=→"ALERT", storeResult=(unset), everyFrame=true)
    BoolTest(boolVariable=var "startAlert", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    on ALERT → Alert
  }

  state Distance Fly {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    Tk2dPlayFrame(gameObject=Self, frame=2)
    DistanceFly(gameObject=Self, target=var "Hero", distance=7, speedMax=4, acceleration=0.1, targetsHeight=false, height=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    WaitRandom(timeMin=1.5, timeMax=2.25, finishEvent=→"WAIT", realTime=false)
    SetBoolValue(boolVariable=var "Out Of Range", boolValue=var "Unalert Range", everyFrame=true)
    BoolFlipEveryFrame(boolVariable=var "Out Of Range", everyFrame=true)
    FloatAddV2(floatVariable=var "Range Out Timer", add=1, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=var "Out Of Range")
    SetFloatValueV2(floatVariable=var "Range Out Timer", floatValue=0, everyFrame=true, activeBool=var "Unalert Range")
    FloatCompare(float1=var "Range Out Timer", float2=8, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=true)
    on WAIT → Raycast
    on UNALERT → Unalert Frame
  }

  state Fire {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=spitter_spit (sharedassets32.assets), finishedEvent=(none))
    SpawnObjectFromGlobalPool(gameObject=Spitter Shot R (sharedassets32.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Shot")
    FireAtTarget(gameObject=var "Shot", target=var "Hero", speed=15, position=(0, 0, 0), spread=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"WAIT")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on WAIT → Fire Dribble
  }

  state Alert {
    SetAudioPitch(gameObject=Self, pitch=1.2, everyFrame=false)
    GetHero(storeResult=var "Hero")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Distance Fly
  }

  state Fire Anticipate {
    DistanceFly(gameObject=Self, target=var "Hero", distance=9, speedMax=2, acceleration=0.1, targetsHeight=false, height=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fire Long", animationTriggerEvent=→"WAIT", animationCompleteEvent=(none))
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=true)
    on WAIT → Fire
  }

  state Raycast {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "Hero Pos 2D", vector2Value=(unset), x=var "Hero X", y=var "Hero Y", everyFrame=false)
    SetVector2XY(vector2Variable=var "Self Pos 2D", vector2Value=(unset), x=var "Self X", y=var "Self Y", everyFrame=false)
    DistanceBetweenPoints2D(distanceResult=var "Distance", point1=var "Self Pos 2D", point2=var "Hero Pos 2D", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=14, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FALSE", everyFrame=false)
    SetFloatValue(floatVariable=var "Raycast X", floatValue=var "Hero X", everyFrame=false)
    SetFloatValue(floatVariable=var "Raycast Y", floatValue=var "Hero Y", everyFrame=false)
    FloatSubtract(floatVariable=var "Raycast X", subtract=var "Self X", everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Raycast Y", subtract=var "Self Y", everyFrame=false, perSecond=false)
    SetVector2XY(vector2Variable=var "Raycast Vector", vector2Value=(unset), x=var "Raycast X", y=var "Raycast Y", everyFrame=false)
    RayCast2d(fromGameObject=Self, fromPosition=(0, 0), direction=var "Raycast Vector", space=0, distance=var "Distance", minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Raycast Hit", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    NextFrameEvent(sendEvent=→"WAIT")
    on WAIT → Raycast Check
    on FALSE → Distance Fly
  }

  state Raycast Check {
    BoolTest(boolVariable=var "Raycast Hit", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Distance Fly
    on FALSE → Fly Back
  }

  state Fire Dribble {
    GetRotation(gameObject=var "Shot", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "AngleToHero", space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Dribble Spread Min", floatValue=var "AngleToHero", everyFrame=false)
    SetFloatValue(floatVariable=var "Dribble Spread Max", floatValue=var "AngleToHero", everyFrame=false)
    FloatAdd(floatVariable=var "Dribble Spread Max", add=25, everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Dribble Spread Min", subtract=25, everyFrame=false, perSecond=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=5, speedMin=10, speedMax=20, angleMin=var "Dribble Spread Min", angleMax=var "Dribble Spread Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    on WAIT → Distance Fly
  }

  state Fly Back {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    DistanceFly(gameObject=Self, target=var "Hero", distance=8.25, speedMax=4, acceleration=0.1, targetsHeight=false, height=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    on FINISHED → Fire Anticipate
  }

  state Distance Fly 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Tk2dPlayFrame(gameObject=Self, frame=2)
    RandomFloat(min=2.5, max=3, storeResult=var "Fire Timer")  // disabled
    DistanceFly(gameObject=Self, target=var "Hero", distance=7, speedMax=5, acceleration=0.1, targetsHeight=false, height=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"WAIT", realTime=false)
    Wait(time=var "Fire Timer", finishEvent=→"WAIT", realTime=false)  // disabled
    SetBoolValue(boolVariable=var "Out Of Range", boolValue=var "Unalert Range", everyFrame=true)
    BoolFlipEveryFrame(boolVariable=var "Out Of Range", everyFrame=true)
    FloatAddV2(floatVariable=var "Range Out Timer", add=1, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=var "Out Of Range")
    SetFloatValueV2(floatVariable=var "Range Out Timer", floatValue=0, everyFrame=true, activeBool=var "Unalert Range")
    FloatCompare(float1=var "Range Out Timer", float2=8, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=true)
    on WAIT → 
    on TOOK DAMAGE → 
    on UNALERT → 
  }

  state Unalert Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var startAlert: bool = true

  var AngleToHero: float = 0
  var Distance: float = 0
  var Dribble Spread Max: float = 0
  var Dribble Spread Min: float = 0
  var Fire Timer: float = 0
  var Hero X: float = 0
  var Hero Y: float = 0
  var Range Out Timer: float = 0
  var Raycast X: float = 0
  var Raycast Y: float = 0
  var Self X: float = 0
  var Self Y: float = 0
  var Can See Hero: bool = false
  var In Alert Range: bool = false
  var Out Of Range: bool = false
  var Raycast Hit: bool = false
  var Unalert Range: bool = false
  var Hero Pos 2D: vector2 = (0, 0)
  var Raycast Vector: vector2 = (0, 0)
  var Self Pos 2D: vector2 = (0, 0)
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Shot: gameObject = <null>
}
