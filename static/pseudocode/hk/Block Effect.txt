# Block Effect
# 1 location(s):
#   resources.assets / Orbit Shield/Shield

fsm Block Effect {
  start Idle
  on ORBIT SHIELD DOWN → Idle  // from any state

  state Idle {
    on BLOCK EFFECT → Block
  }

  state Block {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Block")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=sword_hit_reject)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }
}
