# FSM
# 2 location(s):
#   Ruins2_11 / Music Region
#   Ruins2_11_boss / Battle Scene

// uses template: destroy_if_playerdatabool
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var playerData bool: string = "collectorDefeated"
}
