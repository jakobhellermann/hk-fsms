// uses template: glow_response_control
fsm Control {
  start Pause
  on GLOW END → Glow End  // from any state

  state Init {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_17", isTrue=(none), isFalse=→"INACTIVE")  // disabled
    FindChild(gameObject=Self, childName="Fade Sprite", storeResult=var "Fade Sprites")
    FindChild(gameObject=Self, childName="Particle System", storeResult=var "Particle System")
    on FINISHED → Out
    on INACTIVE → Inactive
  }

  state Out {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Play Particle
  }

  state Enter {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=var "Audio Clip", pitchMin=var "Pitch Min", pitchMax=var "Pitch Max", volume=1, delay=0, storePlayer=<null>)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SetParticleEmission(gameObject=var "Particle System", emission=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FADE IN", delay=0, everyFrame=false)
    on EXIT → Exit
  }

  state Exit {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SetParticleEmission(gameObject=var "Particle System", emission=false)
    SendEventByName(eventTarget=GameObject(var "Fade Sprites"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    on ENTER → Enter
  }

  state Play Particle {
    PlayParticleEmitter(gameObject=var "Particle System", emit=0)
    on FINISHED → Enter
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inactive {
  }

  state Glow End {
    SetParticleEmission(gameObject=var "Particle System", emission=false)
    SendEventByName(eventTarget=GameObject(var "Fade Sprites"), sendEvent="FADE OUT", delay=0, everyFrame=false)
  }

  var Pitch Max: float = 1
  var Pitch Min: float = 1
  var Audio Clip: object = <null>

  var Fade Sprites: gameObject = <null>
  var Particle System: gameObject = <null>
}
