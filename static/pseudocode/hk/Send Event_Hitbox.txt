fsm Send Event {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"CONTACT", storeCollider=var "Collider")
    on CONTACT → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="DREAM IMPACT", delay=0, everyFrame=false)
    SendDreamImpact(target=var "Collider")
    on FINISHED → Idle
  }

  var Collider: gameObject = <null>
}
