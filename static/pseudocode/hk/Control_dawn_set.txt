# Control
# 4 location(s):
#   Dream_Guardian_Hegemol / dawn_set
#   Dream_Guardian_Lurien / dawn_set
#   Dream_Guardian_Monomon / dawn_set
#   Dream_Final_Boss / dawn_set

fsm Control {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="dawn_01", storeResult=var "dawn_01")
    FindChild(gameObject=Self, childName="dawn_02", storeResult=var "dawn_02")
    GetPlayerDataInt(gameObject=var "GameManager", intName="guardiansDefeated", storeValue=var "Guardians Defeated")
    IntSwitch(intVariable=var "Guardians Defeated", compareTo=[1, 2], sendEvent=[→"ONE KILLED", →"TWO KILLED"], everyFrame=false)
    on ONE KILLED → One Killed
    on TWO KILLED → Two Killed
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state One Killed {
    ActivateGameObject(gameObject=var "dawn_01", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Two Killed {
    ActivateGameObject(gameObject=var "dawn_02", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Guardians Defeated: int = 0
  var dawn_01: gameObject = <null>
  var dawn_02: gameObject = <null>
}
