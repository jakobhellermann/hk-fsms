# Control
# 1 location(s):
#   Mines_35 / Crystal Shaman

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Sprite Base", storeResult=var "Sprite Base")
    FindChild(gameObject=Self, childName="Death Glow", storeResult=var "Death Glow")
    FindChild(gameObject=Self, childName="Sprite Full", storeResult=var "Sprite Full")
    FindChild(gameObject=Self, childName="Orbs", storeResult=var "Orbs")
    FindChild(gameObject=Self, childName="Pieces", storeResult=var "Pieces")
    FindChild(gameObject=Self, childName="Pt Hit", storeResult=var "Pt Hit")
    FindChild(gameObject=Self, childName="Lines", storeResult=var "Lines")
    GetPlayerDataInt(gameObject=var "GameManager", intName="quakeLevel", storeValue=var "Quake Level")
    GetPosition(gameObject=var "Self", vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    IntCompare(integer1=var "Quake Level", integer2=2, equal=→"BROKEN", lessThan=(none), greaterThan=→"BROKEN", everyFrame=false)
    FindChild(gameObject=Self, childName="Black Wave", storeResult=var "Black Wave")
    on BROKEN → Broken
    on FINISHED → Idle
  }

  state Broken {
    SetSpriteRenderer(gameObject=var "Sprite Base", active=true)
    SetSpriteRenderer(gameObject=var "Sprite Full", active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on NAIL HIT → Hit
  }

  state Hit {
    SetParticleEmission(gameObject=var "Pt Hit", emission=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=crystal_break_2 (resources.assets), pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=8, equal=→"SHATTER", lessThan=(none), greaterThan=→"SHATTER", everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(-1.27, -1.27, 0), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(-1.27, -1.27, 0), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ObjectJitter(gameObject=Self, x=0.1, y=(unset), z=(unset), allowMovement=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hit Recover
    on SHATTER → Shatter
  }

  state Hit Recover {
    SetParticleEmission(gameObject=var "Pt Hit", emission=false)
    ObjectJitter(gameObject=Self, x=0.1, y=(unset), z=(unset), allowMovement=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on FINISHED → Idle
    on NAIL HIT → Hit
  }

  state Shatter {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=crystal_break_1 (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Death Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lines", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Black Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Pt Hit", emission=false)
    SetSpriteRenderer(gameObject=var "Sprite Base", active=true)
    SetSpriteRenderer(gameObject=var "Sprite Full", active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "Pieces", activate=true)
    FlingObjects(containerObject=var "Pieces", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=15, speedMax=22, angleMin=10, angleMax=170)
    on FINISHED → Land Hero
  }

  state Take Control {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=var "Hero Floor Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    on LAND → Land
  }

  state Land {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Dreamer Land")
  }

  state Land Hero {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LAND HERO", delay=0, everyFrame=false)
    on FINISHED → Orbs
  }

  state Orbs {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dark_spell_absorb (sharedassets28.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Orbs", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Rumble 2
  }

  state Regain Control {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
  }

  state Rumble 2 {
    SendEventByName(eventTarget=GameObject(var "Camera Parent"), sendEvent="MedRumble", delay=0, everyFrame=false)  // disabled
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get 2
  }

  state Get 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dark_spell_get (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Black Wave Default R (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Shadow", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get PlayerData 2
  }

  state Get PlayerData 2 {
    SetPlayerDataInt(intName="quakeLevel", value=2)
    on FINISHED → Call UI Msg 2
  }

  state Call UI Msg 2 {
    CreateObject(gameObject=UI Msg Get Item (sharedassets6.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "UI Msg", networkInstantiate=false, networkGroup=0)  // disabled
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=_0005_shadow_quake (sharedassets274.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Quake 2", everyFrame=false)
    on GET ITEM MSG END → Get Up
  }

  state Get Up {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="GetUpToIdle")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Regain Control
  }

  var Hero Floor Y: float = 48.51
  var Hero Y: float = 0
  var Hits: int = 0
  var Quake Level: int = 0
  var Start Pos: vector3 = (0, 0, 0)
  var Black Wave: gameObject = <null>
  var Camera Parent: gameObject = <null>
  var Cutscene Knight: gameObject = <null>
  var Death Glow: gameObject = <null>
  var Dream Dialogue: gameObject = <null>
  var Light 2: gameObject = <null>
  var Lines: gameObject = <null>
  var Main Camera: gameObject = <null>
  var Orbs: gameObject = <null>
  var Pieces: gameObject = <null>
  var Pt Hit: gameObject = <null>
  var Self: gameObject = <null>
  var Sprite Base: gameObject = <null>
  var Sprite Full: gameObject = <null>
  var UI Msg: gameObject = <null>
  var Wave: gameObject = <null>
}
