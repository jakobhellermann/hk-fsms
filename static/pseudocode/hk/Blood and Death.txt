# Blood and Death
# 1 location(s):
#   Room_Colosseum_Spectate / Spectate Effects

fsm Blood and Death {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Pause
  }

  state Pause {
    WaitRandom(timeMin=0.75, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Spawn
  }

  state Check Wounds {
    IntAdd(intVariable=var "Wounds", add=1, everyFrame=false)
    IntCompare(integer1=var "Wounds", integer2=3, equal=(none), lessThan=→"WOUND", greaterThan=(none), everyFrame=false)
    SendRandomEvent(events=[→"WOUND", →"KILL"], weights=[0.66, 0.34], delay=0)
    IntCompare(integer1=var "Wounds", integer2=8, equal=(none), lessThan=(none), greaterThan=→"KILL", everyFrame=false)
    on WOUND → Wound
    on KILL → Kill
  }

  state Wound {
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=<null>, position=var "Spawn Vector", spawnMin=5, spawnMax=10, speedMin=15, speedMax=20, angleMin=10, angleMax=80, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[enemy_damage (resources.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Pause
  }

  state Set Spawn {
    RandomFloat(min=var "Spawn X Min", max=var "Spawn X Max", storeResult=var "Spawn X")
    RandomFloat(min=var "Spawn Y Min", max=var "Spawn Y Max", storeResult=var "Spawn Y")
    SetVector3XYZ(vector3Variable=var "Spawn Vector", vector3Value=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), everyFrame=false)
    on FINISHED → Check Wounds
  }

  state Kill {
    SetIntValue(intVariable=var "Wounds", intValue=0, everyFrame=false)
    SendRandomEvent(events=[→"BLOOD", (none)], weights=[1, 0], delay=0)  // disabled
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[enemy_damage (resources.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[enemy_death_sword (resources.assets)], weights=[1], pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[crowd_gasp (sharedassets32.assets), crowd_cheer (sharedassets32.assets), crowd_cheer_2 (sharedassets35.assets), crowd_laugh_3_long (sharedassets32.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on BLOOD → Blood
    on FINISHED → Blood
  }

  state Blood {
    CreateObject(gameObject=Corpse Flukeman Bot (sharedassets35.assets), spawnPoint=<null>, position=var "Spawn Vector", rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CROWD SHORT CHEER", delay=0, everyFrame=false)
    on FINISHED → Pause
  }

  var Spawn X: float = 0
  var Spawn X Max: float = 112.5
  var Spawn X Min: float = 111
  var Spawn Y: float = 0
  var Spawn Y Max: float = 11.99
  var Spawn Y Min: float = 8.43
  var Wounds: int = 0
  var Spawn Vector: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
}
