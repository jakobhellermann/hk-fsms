# Set Slash Damage
# 2 location(s):
#   Knight_Pickup / Knight/Attacks
#   resources.assets / Knight:1/Attacks

fsm Set Slash Damage {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="AltSlash", storeResult=var "AltSlash")
    FindChild(gameObject=Self, childName="Cyclone Slash", storeResult=var "Cyclone Slash")
    FindChild(gameObject=var "Cyclone Slash", childName="Hits", storeResult=var "Cyclone Hits")
    FindChild(gameObject=var "Cyclone Hits", childName="Hit L", storeResult=var "Hit L")
    FindChild(gameObject=var "Cyclone Hits", childName="Hit R", storeResult=var "Hit R")
    FindChild(gameObject=Self, childName="DownSlash", storeResult=var "DownSlash")
    FindChild(gameObject=Self, childName="Great Slash", storeResult=var "Great Slash")
    FindChild(gameObject=Self, childName="Dash Slash", storeResult=var "Dash Slash")
    FindChild(gameObject=Self, childName="Slash", storeResult=var "Slash")
    FindChild(gameObject=Self, childName="UpSlash", storeResult=var "UpSlash")
    FindChild(gameObject=Self, childName="WallSlash", storeResult=var "WallSlash")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get Damage
  }

  state Set Damage {
    SetFsmInt(gameObject=var "DownSlash", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "AltSlash", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "Slash", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "UpSlash", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "WallSlash", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "Hit L", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "Hit R", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    IntOperator(integer1=var "Nail Damage", integer2=2, operation=Multiply, storeResult=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "Dash Slash", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    SetFsmInt(gameObject=var "Great Slash", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Nail Damage", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on UPDATE NAIL DAMAGE → Get Damage
    on CHARM INDICATOR CHECK → Get Damage
  }

  state Get Damage {
    GetPlayerDataInt(gameObject=var "GameManager", intName="nailDamage", storeValue=var "Nail Damage")  // disabled
    GetNailDamage(storeValue=var "Nail Damage")
    on FINISHED → Glass Attack Modifier
  }

  state Glass Attack Modifier {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_25", falseBool="brokenCharm_25", isTrue=(none), isFalse=→"FINISHED")
    ConvertIntToFloat(intVariable=var "Nail Damage", floatVariable=var "Damage Float", everyFrame=false)
    FloatMultiply(floatVariable=var "Damage Float", multiplyBy=1.5, everyFrame=false)
    ConvertFloatToInt(floatVariable=var "Damage Float", intVariable=var "Nail Damage", rounding=Nearest, everyFrame=false)
    on FINISHED → Set Beam Damage
  }

  state Set Beam Damage {
    ConvertIntToFloat(intVariable=var "Nail Damage", floatVariable=var "Damage Float", everyFrame=false)
    FloatOperator(float1=var "Damage Float", float2=0.5, operation=Multiply, storeResult=var "Damage Float", everyFrame=false)
    ConvertFloatToInt(floatVariable=var "Damage Float", intVariable=var "Beam Damage", rounding=Nearest, everyFrame=false)
    IntClamp(intVariable=var "Beam Damage", minValue=1, maxValue=90, everyFrame=false)
    SetPlayerDataInt(intName="beamDamage", value=var "Beam Damage")
    on FINISHED → Set Damage
  }

  var Damage Float: float = 0
  var Beam Damage: int = 0
  var Nail Damage: int = 0
  var AltSlash: gameObject = <null>
  var Cyclone Hits: gameObject = <null>
  var Cyclone Slash: gameObject = <null>
  var Dash Slash: gameObject = <null>
  var DownSlash: gameObject = <null>
  var Great Slash: gameObject = <null>
  var Hit L: gameObject = <null>
  var Hit R: gameObject = <null>
  var Slash: gameObject = <null>
  var UpSlash: gameObject = <null>
  var WallSlash: gameObject = <null>
}
