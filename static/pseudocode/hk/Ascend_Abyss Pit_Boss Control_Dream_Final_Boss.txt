# Ascend
# 1 location(s):
#   Dream_Final_Boss / Boss Control/Abyss Pit

fsm Ascend {
  start Idle

  state Idle {
    FindChild(gameObject=Self, childName="Shades", storeResult=var "Shades")
    on ASCEND → Ascend
  }

  state Ascend {
    FloatAdd(floatVariable=var "Hero Y", add=-3, everyFrame=false, perSecond=false)
    SetVector3XYZ(vector3Variable=var "Tween Pos", vector3Value=(unset), x=56.5, y=var "Hero Y", z=0, everyFrame=false)
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Tween Pos", time=4, delay=0, speed=(unset), space=0, easeType=easeInOutCubic, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on ASCEND → Reset
  }

  state Reset {
    on FINISHED → Ascend
  }

  var Hero Y: float = 0
  var Self Y: float = 0
  var Tween Pos: vector3 = (0, 0, 0)
  var Next Shade: gameObject = <null>
  var Shades: gameObject = <null>
}
