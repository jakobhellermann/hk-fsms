# Break
# 1 location(s):
#   Crossroads_10 / Fk Break Wall/Breakable

fsm Break {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Broken", storeResult=var "Broken")
    FindChild(gameObject=var "Parent", childName="Battle Gate 1", storeResult=var "Battle Gate")
    FindGameObject(objectName="CameraLockArea B", withTag="Untagged", store=var "CameraLockArea B")  // disabled
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Hit Left
  }

  state Hit Left {
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15)
    CreateObject(gameObject=Slash Impact (sharedassets39.assets), spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "Slash", networkInstantiate=false, networkGroup=0)
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(-1.12, 0, 0), spawnMin=3, spawnMax=5, speedMin=15, speedMax=20, angleMin=100, angleMax=140, originVariation=0)  // disabled
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Slash", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Slash", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    SpawnFromPool(pool=var "Pool S", adjustPosition=(-1.12, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=100, angleMax=140)  // disabled
    CreateObject(gameObject=Dust Hit Med (sharedassets6.assets), spawnPoint=var "Self", position=(-1, 0, 0), rotation=(180, 90, 270), storeObject=var "", networkInstantiate=false, networkGroup=0)  // disabled
    SpawnRandomObjects(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(2, -1, 0), spawnMin=3, spawnMax=5, speedMin=15, speedMax=20, angleMin=40, angleMax=80, originVariation=0)
    CreateObject(gameObject=Dust Hit Med (sharedassets6.assets), spawnPoint=var "Self", position=(-1, -1, 0), rotation=(0, 90, 270), storeObject=var "", networkInstantiate=false, networkGroup=0)
    IntOperator(integer1=var "HP", integer2=1, operation=Subtract, storeResult=var "HP", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilRight())
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(0, -1, 0), rotation=(unset), storeObject=var "")
    on FINISHED → Hit Recover
    on HIT → Break
  }

  state Hit Recover {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Break {
    ActivateGameObject(gameObject=var "Battle Gate", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CameraLockArea B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetPlayerDataBool(boolName="falseKnightWallBroken", value=true)
    ActivateGameObject(gameObject=var "Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlay(gameObject=var "", volume=1, oneShotClip=breakable_wall_death (sharedassets6.assets), finishedEvent=(none))
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med Down R (resources.assets), spawnPoint=var "Self", position=(0, 0, 0), rotation=(-72.5, -180, -180), storeObject=var "")
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(0, 0, 0), spawnMin=30, spawnMax=30, speedMin=0, speedMax=20, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Self", position=(0, 0, 0), spawnMin=30, spawnMax=30, speedMin=0, speedMax=20, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Small Transient (resources.assets), spawnPoint=var "Self", position=(0, 0, 0), spawnMin=40, spawnMax=40, speedMin=0, speedMax=30, angleMin=40, angleMax=140, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
  }

  var Hits: int = 4

  var Chooser: float = 0
  var Recoil Speed: float = 1
  var HP: int = 4
  var Midpoint: vector3 = (0, 0, 0)
  var Battle Gate: gameObject = <null>
  var Broken: gameObject = <null>
  var CameraLockArea B: gameObject = <null>
  var Parent: gameObject = <null>
  var Pool S: gameObject = <null>
  var Self: gameObject = <null>
  var Slash: gameObject = <null>
}
