    on EVADE RANGE → Slash?
    on TOOK DAMAGE → Startled?
  }

    on CANCEL → Slash Antic
  }

  }

  state Cooldown {

    Wait(time=0.5, finishEvent=→"WAIT", realTime=false)  // disabled
    WaitRandom(timeMin=0.1, timeMax=0.4, finishEvent=→"WAIT", realTime=false)
    on WAIT → Move Start
  }

  state Spawn Bullet L {
    FloatMultiply(floatVariable=var "Hero Distance", multiplyBy=-1, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Shot GladiatorSickle, spawnPoint=var "SpawnPoint", position=(unset), rotation=(unset), storeObject=var "Pickaxe")

  }

    FloatAddV2(floatVariable=var "Hero Distance", add=var "Distance Vary", everyFrame=false, perSecond=false, fixedUpdate=false, activeBool=(unset))

    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Terrain Detector", isTrue=→"TERRAIN DETECTOR", isFalse=(none), everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Evade Speed", floatValue=-13, everyFrame=false)
    FloatCompare(float1=var "Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Evade Speed", floatValue=13, everyFrame=false)
    on FINISHED → Evade Start Anim
    on TERRAIN DETECTOR → Attack Choice
  }

  state Evade {

    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP MINING", delay=0, everyFrame=false)
    SetCircleCollider(gameObject=var "Waker", active=true)
    BoolTest(boolVariable=var "Startled", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Startled", boolValue=true, everyFrame=false)

    FloatTestToBool(float1=var "Hero Distance", float2=var "Position", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Is Hero Right", everyFrame=false)
