# Death
# 3 location(s):
#   Ruins2_11_boss / Battle Scene/Jar Collector
#   GG_Collector / Battle Scene/Jar Collector
#   GG_Collector_V / Battle Scene/Jar Collector

fsm Death {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Death Wave", storeResult=var "Death Wave")
    FindChild(gameObject=Self, childName="Pt Dying", storeResult=var "Pt Dying")
    FindChild(gameObject=Self, childName="Death Blow", storeResult=var "Death Blow")
    FindChild(gameObject=Self, childName="Death Puddle", storeResult=var "Death Puddle")
    FindChild(gameObject=Self, childName="Pt Puddle", storeResult=var "Pt Puddle")
    on DEATH → Music
  }

  state Death Start {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Collector_death_start (sharedassets126.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_final_hit (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Stun Push Crt", floatValue=var "Stun Push", everyFrame=false)
    FloatMultiply(floatVariable=var "Stun Push Crt", multiplyBy=var "X Scale", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wound Air")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Stun Push Crt", y=20, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=2, y=2, z=(unset), everyFrame=false, lateUpdate=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(0))  // disabled
    SetDamageHeroAmount(target=Self, damageDealt=0)
    PlayParticleEmitter(gameObject=var "Death Wave", emit=0)
    on FINISHED → Death Air
  }

  state Death Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=→"LAND", leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=→"LAND", leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Start Effect
  }

  state Death Wound {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wound")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Pt Dying", emission=true)
    on TOOK DAMAGE → Death Hit
  }

  state Death Hit {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Collector_death_hit_01 (sharedassets126.assets), Collector_death_hit_02 (sharedassets126.assets), Collector_death_hit_03 (sharedassets126.assets), Collector_death_hit_04 (sharedassets126.assets), Collector_death_hit_05 (sharedassets126.assets), Collector_death_hit_06 (sharedassets126.assets), Collector_death_hit_07 (sharedassets126.assets)], weights=[1, 1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wound Hit")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SetParticleEmission(gameObject=var "Pt Dying", emission=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Hit Check
  }

  state Start Effect {
    AudioStop(gameObject=Self)
    SetAudioClip(gameObject=Self, audioClip=Collector_Stun_Loop (sharedassets126.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    PlayParticleEmitter(gameObject=var "Pt Dying", emit=0)
    on FINISHED → Death Wound
    on GG BOSS → Kill Enemies
  }

  state Hit Check {
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=15, equal=→"DEATH", lessThan=(none), greaterThan=→"DEATH", everyFrame=false)
    on DEATH → Death Antic
    on FINISHED → Death Wound
  }

  state Death Antic {
    AudioStop(gameObject=Self)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(0))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Death Wave", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Antic")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    on FINISHED → Blow
  }

  state Blow {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Collector_death_final (sharedassets126.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Death Blow", emit=0)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Ready
  }

  state Ready {
    StopParticleEmitter(gameObject=var "Death Blow")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Ready")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    on FINISHED → Set Data
  }

  state Fall {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    on FINISHED → Puddle
    on GG BOSS → GG End
  }

  state Puddle {
    PlayParticleEmitter(gameObject=var "Pt Puddle", emit=0)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Death Puddle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    iTweenScaleTo(gameObject=var "Death Puddle", id="", transformScale=var "", vectorScale=(1, 0.01, 1), time=3, delay=1, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → End
  }

  state End {
    StopParticleEmitter(gameObject=var "Pt Puddle")
    ActivateGameObject(gameObject=var "Death Puddle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE END", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["COLLECTOR"], storeResult=(unset var))
  }

  state Set Data {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    ShowGodfinderIcon(delay=0, unlockBossScene=Collector Boss Scene (resources.assets))
    SetPlayerDataBool(boolName="collectorDefeated", value=true)
    SetPlayerDataBool(boolName="killedJarCollector", value=true)
    SetPlayerDataBool(boolName="newDataJarCollector", value=true)
    SetPlayerDataInt(intName="killsJarCollector", value=0)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasJournal", isTrue=(none), isFalse=→"FINISHED")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DESTROY JOURNAL MSG", delay=0, everyFrame=false)
    CreateObject(gameObject=Journal Update Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Journal Msg", networkInstantiate=false, networkGroup=0)
    SetFsmBool(gameObject=var "Journal Msg", fsmName="Journal Msg", variableName="Full", setValue=true, everyFrame=false)
    on FINISHED → Fall
  }

  state GG End {
    SendHealthManagerDeathEvent(target=Self)
  }

  state Music {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="NONE", delay=0, everyFrame=false)
    ApplyMusicCue(musicCue=None (resources.assets), delayTime=0, transitionTime=0)
    on FINISHED → Death Start
  }

  state Kill Enemies {
    SendEventToRegister(eventName="KILL ALL ENEMIES")
    on FINISHED → Death Antic
  }

  var Stun Push: float = 10
  var Stun Push Crt: float = 0
  var X Scale: float = 0
  var Hits: int = 0
  var Death Blow: gameObject = <null>
  var Death Puddle: gameObject = <null>
  var Death Wave: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Journal Msg: gameObject = <null>
  var Pt Dying: gameObject = <null>
  var Pt Puddle: gameObject = <null>
  var Self: gameObject = <null>
}
