# Charm Equip Msg
# 1 location(s):
#   resources.assets / Bound Charm Msg

fsm Charm Equip Msg {
  start Button Info?
  on CHARM MSG → Destroy  // from any state
  on CLOSE → Down  // from any state
  on BENCH CLOSE → Bench Down  // from any state

  state Init {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHARM MSG", delay=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=2.77, z=-20, space=0, everyFrame=false, lateUpdate=false)
    FindChild(gameObject=Self, childName="Text", storeResult=var "Text")  // disabled
    GetLanguageString(sheetName="Prompts", convName="CHARM_REMINDER", storeValue=var "Msg String")  // disabled
    SetTextMeshProText(gameObject=var "Text", textString=var "Msg String")  // disabled
    on FINISHED → Up
    on INVENTORY CLOSED → Down
    on INVENTORY OPENED → Down
    on INV PANEL CHANGE → Down
  }

  state Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=4.5, finishEvent=→"FINISHED", realTime=true)
    ListenForRsDown(eventTarget=Self, wasPressed=→"FINISHED", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForRsUp(eventTarget=Self, wasPressed=→"FINISHED", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForDown(eventTarget=Self, wasPressed=→"FINISHED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForRight(eventTarget=Self, wasPressed=→"FINISHED", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForUp(eventTarget=Self, wasPressed=→"FINISHED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForLeft(eventTarget=Self, wasPressed=→"FINISHED", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"FINISHED", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on FINISHED → Down
    on INVENTORY CLOSED → Down
    on INVENTORY OPENED → Down
    on INV PANEL CHANGE → Down
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CHARM MSG DOWN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Bench Down {
    Wait(time=1, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Down
  }

  state Button Info? {
    BoolTest(boolVariable=var "Bench Sit", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Button Info", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Init
  }

  var Bench Sit: bool = false
  var Msg String: string = ""
  var Button Info: gameObject = <null>
  var Text: gameObject = <null>
}
