  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPreInstantiatedGameObject(target=Self, storeGameObject=var "Egg")
    SetPositionToObject(gameObject=var "Egg", targetObject=var "Self", xOffset=0, yOffset=0, zOffset=0)
    SetFsmInt(gameObject=var "Egg", fsmName="Shiny Control", variableName="Trinket Num", setValue=11, everyFrame=false)
    SetFsmBool(gameObject=var "Egg", fsmName="Shiny Control", variableName="Fling On Start", setValue=true, everyFrame=false)

    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle Min", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Angle Min", add=80, everyFrame=false, perSecond=false)
    FloatOperator(float1=var "Angle Min", float2=20, operation=Add, storeResult=var "Angle Max", everyFrame=false)
    on FINISHED → Spit
  }

  state Spit {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death")
    Wait(time=1.4, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Death", emit=0)
    on FINISHED → Burst
  }

  state Burst {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Egg", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Egg", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mimic_spider_transormation_stage_2, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    PlayParticleEmitter(gameObject=var "Pt Blow", emit=0)
    StopParticleEmitter(gameObject=var "Pt Death")
    SetParent(gameObject=var "Pt Blow", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Angle Max: float = 0
  var Angle Min: float = 0
  var Egg: gameObject = <null>

  var Pt Death: gameObject = Corpse Egg Sac/Pt Death

  var Spit Point: gameObject = <null>
}
