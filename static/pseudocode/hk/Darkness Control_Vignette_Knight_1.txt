# Darkness Control
# 1 location(s):
#   resources.assets / Knight:1/Vignette

fsm Darkness Control {
  start PAUSE
  on SCENE RESET NO LANTERN → Scene Reset 2  // from any state
  on DAMAGE → Damage  // from any state
  on DEATH → Death  // from any state
  on HERO RESPAWNED → Recover  // from any state
  on RESET → Pause  // from any state
  on SCENE RESET → Scene Reset  // from any state

  state Init {
    on FINISHED → Dark Lev Check
  }

  state Idle {
    SetPlayerDataInt(intName="previousDarkness", value=var "Darkness Level")
  }

  state Damage {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Damage Scale", time=0.25, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    ScaleTo(gameObject=Self, target=var "Damage Scale", duration=0.25, delay=0, curve=Linear)
    on FINISHED → Damage Pause
  }

  state Damage Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recover
  }

  state Recover {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Idle Scale", time=2, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    ScaleTo(gameObject=Self, target=var "Idle Scale", duration=2, delay=0, curve=Linear)
    on FINISHED → Idle
  }

  state Death {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Damage Scale", time=2, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    ScaleTo(gameObject=Self, target=var "Damage Scale", duration=2, delay=0, curve=Linear)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetSpriteRenderer(gameObject=Self, active=true)
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=Self, childName="Darkness Border", storeResult=var "Darkness Border")
    FindChild(gameObject=Self, childName="Darkness Plates", storeResult=var "Darkness Plates")
    ActivateGameObject(gameObject=var "Darkness Border", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetEventSender(sentByGameObject=var "Sender")
    on FINISHED → Init
  }

  state Dark Lev Check {
    ActivateGameObject(gameObject=var "Darkness Border", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    IntSwitch(intVariable=var "Darkness Level", compareTo=[0, -1], sendEvent=[→"NORMAL", →"DARK -1"], everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasLantern", isTrue=→"LANTERN", isFalse=(none))
    IntSwitch(intVariable=var "Darkness Level", compareTo=[1, 2], sendEvent=[→"DARK 1", →"DARK 2"], everyFrame=false)
    on DARK 1 → Dark 1
    on DARK 2 → Dark 2
    on NORMAL → Normal
    on LANTERN → Lantern
    on DARK -1 → Dark -1
  }

  state Normal {
    SetScale(gameObject=var "Lantern Glow", vector=(unset), x=0.1, y=0.1, z=0.1, everyFrame=false, lateUpdate=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=5.5, y=5.5, z=5.5, everyFrame=false)
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=4, y=4, z=4, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Border", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Set {
    SetScale(gameObject=Self, vector=var "Idle Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Lantern Glow", vector=var "Idle Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Dark 1 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=2.2, y=2.2, z=2.2, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=1.9, y=1.9, z=1.9, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Border", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Dark 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=0.8, y=0.8, z=0.8, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=0.8, y=0.8, z=0.8, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Border", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Darkness Plates", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Lantern {
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=3, y=3, z=3, everyFrame=false)
    ActivateGameObject(gameObject=var "Lantern Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=2.2, y=2.2, z=2.2, everyFrame=false)
    on FINISHED → Previously Dark?
  }

  state Previously Dark? {
    SetScale(gameObject=var "Lantern Glow", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="previousDarkness", storeValue=var "Previous Dark Level")
    IntCompare(integer1=var "Previous Dark Level", integer2=0, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    IntSwitch(intVariable=var "Darkness Level", compareTo=[1, 2], sendEvent=[→"DARK 1", →"DARK 2"], everyFrame=false)
    on FINISHED → Set
    on DARK 1 → Dark 1 Start
    on DARK 2 → Dark 2 Start
  }

  state Dark 1 Start {
    SetScale(gameObject=Self, vector=var "Idle Scale", x=2.2, y=2.2, z=2.2, everyFrame=false, lateUpdate=false)
    on FINISHED → Lantern Shrink
  }

  state Lantern Up {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Idle Scale", time=4, delay=0.5, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    ScaleTo(gameObject=Self, target=var "Idle Scale", duration=4, delay=0.5, curve=SinusoidalOut)
    iTweenScaleTo(gameObject=var "Lantern Glow", id="", transformScale=var "", vectorScale=(3, 3, 3), time=4, delay=0.5, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    ScaleTo(gameObject=var "Lantern Glow", target=(3, 3, 3), duration=4, delay=0.5, curve=SinusoidalOut)
    SendEventByName(eventTarget=GameObject(var "Darkness Plates"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Dark 2 Start {
    SetScale(gameObject=Self, vector=var "Idle Scale", x=1.1, y=1.1, z=1.1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Lantern Shrink
  }

  state Scene Reset {
    IntSwitch(intVariable=var "Darkness Level", compareTo=[0, -1], sendEvent=[→"NORMAL", →"DARK -1"], everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasLantern", isTrue=→"LANTERN", isFalse=(none))
    IntSwitch(intVariable=var "Darkness Level", compareTo=[1, 2], sendEvent=[→"DARK 1", →"DARK 2"], everyFrame=false)
    on NORMAL → Normal 2
    on DARK 1 → Dark 1 2
    on DARK 2 → Dark 2 2
    on LANTERN → Lantern 2
    on DARK -1 → Dark -1 2
  }

  state Normal 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=5.5, y=5.5, z=5.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=4, y=4, z=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Dark 1 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=2.2, y=2.2, z=2.2, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=1.9, y=1.9, z=1.9, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Dark 2 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=0.8, y=0.8, z=0.8, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=0.8, y=0.8, z=0.8, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Lantern 2 {
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=3, y=3, z=3, everyFrame=false)
    ActivateGameObject(gameObject=var "Lantern Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=2.2, y=2.2, z=2.2, everyFrame=false)
    on FINISHED → Lantern Up
  }

  state Lantern Shrink {
    SetScale(gameObject=var "Lantern Glow", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Lantern Up
  }

  state Dark -1 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=10.5, y=10.5, z=10.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=5, y=5, z=5, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Border", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Dark -1 2 {
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=10.5, y=10.5, z=10.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=5, y=5, z=5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Scene Reset 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    IntSwitch(intVariable=var "Darkness Level", compareTo=[0, -1], sendEvent=[→"NORMAL", →"DARK -1"], everyFrame=false)
    IntSwitch(intVariable=var "Darkness Level", compareTo=[1, 2], sendEvent=[→"DARK 1", →"DARK 2"], everyFrame=false)
    on NORMAL → Normal 2
    on DARK 1 → Dark 1 2
    on DARK 2 → Dark 2 2
    on DARK -1 → Dark -1 2
  }

  state Spawn Lantern Glow {
    CreateObject(gameObject=Lantern, spawnPoint=var "Hero", position=(0, -0.5, 0.006), rotation=(unset), storeObject=var "Lantern Glow", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Lantern Glow", parent=var "Hero", resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Pause
  }

  state PAUSE {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Spawn Lantern Glow
  }

  var Darkness Level: int = 0
  var Previous Dark Level: int = 0
  var Damage Scale: vector3 = (4, 4, 4)
  var Idle Scale: vector3 = (5.5, 5.5, 5.5)
  var Darkness Border: gameObject = <null>
  var Darkness Plates: gameObject = <null>
  var Lantern Glow: gameObject = <null>
  var Parent: gameObject = <null>
  var Sender: gameObject = <null>
}
