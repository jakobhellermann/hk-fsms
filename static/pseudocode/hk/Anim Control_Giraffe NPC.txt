# Anim Control
# 1 location(s):
#   Fungus2_34 / Giraffe NPC

fsm Anim Control {
  start Pause

  state Hero Away {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=32.74, tolerance=0, equal=(none), lessThan=→"CLOSE", greaterThan=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Eat")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EAT START", delay=0, everyFrame=false)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on CLOSE → Lower
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Hero Away
  }

  state Lower {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="EatToIdle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="EAT END", delay=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Whillo_alert (sharedassets196.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[moss_knight_burst_leaves (sharedassets34.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0.9, storePlayer=<null>)
    AudioStop(gameObject=Self)
    on FINISHED → Idle R
  }

  state Idle R {
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=32.74, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"AWAY", everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=25, tolerance=0, equal=(none), lessThan=→"TURN L", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=15, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UPPER", everyFrame=true)
    on AWAY → Upper
    on TURN L → Turn L
    on UPPER → Go Up
  }

  state Upper {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="IdleToEat", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[moss_knight_rustle (sharedassets137.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0.95, storePlayer=<null>)
    on FINISHED → Hero Away
  }

  state Turn L {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="LookLeft", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle L
  }

  state Idle L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Left")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero X", float2=26.2, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"TURN R", everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=15.5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"TURN R", everyFrame=true)
    on TURN R → Turn R
  }

  state Turn R {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="LookRight", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle R
  }

  state Go Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="IdleToEat", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Hero Up
  }

  state Hero Up {
    FloatCompare(float1=var "Hero Y", float2=14, tolerance=0, equal=(none), lessThan=→"LOWER", greaterThan=(none), everyFrame=true)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Eat")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="EAT START", delay=0, everyFrame=false)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    on LOWER → Lower
  }

  var Hero X: float = 0
  var Hero Y: float = 0
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
