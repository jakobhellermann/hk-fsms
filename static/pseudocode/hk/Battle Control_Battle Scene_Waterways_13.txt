# Battle Control
# 1 location(s):
#   Waterways_13 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Wave 2", storeResult=var "Wave 2")
    FindChild(gameObject=Self, childName="Wave 3", storeResult=var "Wave 3")
    FindChild(gameObject=Self, childName="Wave 4", storeResult=var "Wave 4")
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on ACTIVATE → Activated
    on HIT → Wave 1
  }

  state Activated {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
  }

  state Wave 1 {
    TransitionToAudioSnapshot(snapshot=Action (resources.assets), transitionTime=1)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Wave 1", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=var "W1 Sentry", vector=(unset), x=74.56, y=65.98, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "W1 FatSentry", vector=(unset), x=54, y=66.76, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Wave 1"), sendEvent="BATTLE SPAWN", delay=0, everyFrame=false)
    on END → Wave 2
  }

  state Wave 2 {
    ActivateAllChildren(gameObject=var "Wave 2", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=var "W2 FlySentry1", vector=(unset), x=57.6, y=77.1, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "W2 FlySentry2", vector=(unset), x=72.62, y=77.01, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on END → Wave 3
  }

  state Wave 3 {
    ActivateAllChildren(gameObject=var "Wave 3", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SetPosition(gameObject=var "W3 FlySentry", vector=(unset), x=72.4, y=77.05, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "W3 FatSentry", vector=(unset), x=56.16, y=66.79, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Wave 3"), sendEvent="BATTLE SPAWN", delay=0, everyFrame=false)
    on END → End Pause
  }

  state Wave 4 {
    ActivateAllChildren(gameObject=var "Wave 4", activate=true)
    SetIntValue(intVariable=var "Battle Enemies", intValue=2, everyFrame=false)
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    on END → End Pause
  }

  state End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=3)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var W1 FatSentry: gameObject = Battle Scene/Wave 1/Ruins Sentry Fat
  var W1 Sentry: gameObject = Battle Scene/Wave 1/Ruins Sentry 1
  var W2 FlySentry1: gameObject = Battle Scene/Wave 2/Ruins Flying Sentry
  var W2 FlySentry2: gameObject = Battle Scene/Wave 2/Ruins Flying Sentry (1)
  var W3 FatSentry: gameObject = Battle Scene/Wave 3/Ruins Sentry Fat
  var W3 FlySentry: gameObject = Battle Scene/Wave 3/Ruins Flying Sentry Javelin

  var Battle Enemies: int = 0
  var Activated: bool = false
  var CamLock: gameObject = <null>
  var Wave 1: gameObject = <null>
  var Wave 2: gameObject = <null>
  var Wave 3: gameObject = <null>
  var Wave 4: gameObject = <null>
}
