# Fury
# 1 location(s):
#   resources.assets / Knight:1/Charm Effects

fsm Fury {
  start Init Pause

  state Init {
    FindChild(gameObject=Self, childName="Fury", storeResult=var "Fury Object")
    FindChild(gameObject=Self, childName="Rage Burst Effect", storeResult=var "Rage Burst Effect")
    GetParent(gameObject=Self, storeResult=var "Knight")
    FindChild(gameObject=var "Knight", childName="Attacks", storeResult=var "Attacks")
    FindChild(gameObject=var "Attacks", childName="AltSlash", storeResult=var "Alt Slash")
    FindChild(gameObject=var "Attacks", childName="DownSlash", storeResult=var "Down Slash")
    FindChild(gameObject=var "Attacks", childName="UpSlash", storeResult=var "Up Slash")
    FindChild(gameObject=var "Attacks", childName="Slash", storeResult=var "Slash")
    FindChild(gameObject=var "Attacks", childName="WallSlash", storeResult=var "Wall Slash")
    FindChild(gameObject=var "Attacks", childName="Cyclone Slash", storeResult=var "Cyclone Slash")
    FindChild(gameObject=var "Attacks", childName="Dash Slash", storeResult=var "Dash Slash")
    FindChild(gameObject=var "Attacks", childName="Great Slash", storeResult=var "Great Slash")
    FindChild(gameObject=var "Cyclone Slash", childName="Hits/Hit L", storeResult=var "Cyclone Hit L")
    FindChild(gameObject=var "Cyclone Slash", childName="Hits/Hit R", storeResult=var "Cyclone Hit R")
    FindChild(gameObject=var "HUD Camera", childName="fury_effects_v2", storeResult=var "Fury Vignette")  // disabled
    on FINISHED → Idle
  }

  state Idle {
    on HERO DAMAGED → Pause
  }

  state Check HP {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_6", isTrue=(none), isFalse=→"CANCEL")
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue Health")
    IntTestToBool(int1=var "Blue Health", int2=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Has Blue Health", everyFrame=false)
    GetPlayerDataBool(boolName="equippedCharm_27", storeValue=var "Joni Equipped")
    BoolAllTrue(boolVariables=[var "Has Blue Health", var "Joni Equipped"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    IntCompare(integer1=var "HP", integer2=1, equal=→"FURY", lessThan=→"CANCEL", greaterThan=→"CANCEL", everyFrame=false)
    on FURY → Get Ref
    on CANCEL → Idle
  }

  state Activate {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Knight", audioClip=hero_fury_charm_burst, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Fury Object"), sendEvent="PLAY", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Fury Object", emit=0)
    SendMessage(gameObject=var "Alt Slash", delivery=SendMessage, options=1, functionCall=SetFury(true))
    SendMessage(gameObject=var "Down Slash", delivery=SendMessage, options=1, functionCall=SetFury(true))
    SendMessage(gameObject=var "Slash", delivery=SendMessage, options=1, functionCall=SetFury(true))
    SendMessage(gameObject=var "Up Slash", delivery=SendMessage, options=1, functionCall=SetFury(true))
    SendMessage(gameObject=var "Wall Slash", delivery=SendMessage, options=1, functionCall=SetFury(true))
    SetFsmFloat(gameObject=var "Down Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Alt Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Up Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1.75, everyFrame=false)
    SetFsmFloat(gameObject=var "Wall Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1.75, everyFrame=false)
    SetFsmBool(gameObject=var "Cyclone Hit L", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Cyclone Hit R", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dash Slash", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Great Slash", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    Tk2dSpriteSetColor(gameObject=var "Cyclone Slash", color=(1, 0.17647058, 0.17647058, 1), everyframe=false)
    Tk2dSpriteSetColor(gameObject=var "Dash Slash", color=(1, 0.17647058, 0.17647058, 1), everyframe=false)
    Tk2dSpriteSetColor(gameObject=var "Great Slash", color=(1, 0.17647058, 0.17647058, 1), everyframe=false)
    ActivateGameObject(gameObject=var "Rage Burst Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeGroupUp(target=var "Fury Vignette")
    on HERO DAMAGED → HP Pause
    on HERO HEALED → Recheck
    on HERO HEALED FULL → Deactivate
    on ADD BLUE HEALTH → Joni Check
    on ALL CHARMS END → Deactivate
    on FURY REFRESH → Recheck
  }

  state Deactivate {
    SendEventByName(eventTarget=GameObject(var "Fury Object"), sendEvent="STOP", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Fury Object")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="FURY END", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Alt Slash", delivery=SendMessage, options=1, functionCall=SetFury(false))
    SendMessage(gameObject=var "Wall Slash", delivery=SendMessage, options=1, functionCall=SetFury(false))
    SendMessage(gameObject=var "Down Slash", delivery=SendMessage, options=1, functionCall=SetFury(false))
    SendMessage(gameObject=var "Slash", delivery=SendMessage, options=1, functionCall=SetFury(false))
    SendMessage(gameObject=var "Up Slash", delivery=SendMessage, options=1, functionCall=SetFury(false))
    SetFsmFloat(gameObject=var "Down Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Wall Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Alt Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Up Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=var "Slash", fsmName="damages_enemy", variableName="Multiplier", setValue=1, everyFrame=false)
    SetFsmBool(gameObject=var "Dash Slash", fsmName="nailart_damage", variableName="Fury", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Cyclone Hit L", fsmName="nailart_damage", variableName="Fury", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Cyclone Hit R", fsmName="nailart_damage", variableName="Fury", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Great Slash", fsmName="nailart_damage", variableName="Fury", setValue=false, everyFrame=false)
    Tk2dSpriteSetColor(gameObject=var "Cyclone Slash", color=(1, 1, 1, 1), everyframe=false)
    Tk2dSpriteSetColor(gameObject=var "Dash Slash", color=(1, 1, 1, 1), everyframe=false)
    Tk2dSpriteSetColor(gameObject=var "Great Slash", color=(1, 1, 1, 1), everyframe=false)
    FadeGroupDown(target=var "Fury Vignette", fast=false)
    on FINISHED → Idle
  }

  state Stay Furied {
    SetFsmBool(gameObject=var "Cyclone Hit L", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Cyclone Hit R", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dash Slash", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Great Slash", fsmName="nailart_damage", variableName="Fury", setValue=true, everyFrame=false)
    on HERO DAMAGED → HP Pause
    on HERO HEALED → Recheck
    on HERO HEALED FULL → Deactivate
    on ADD BLUE HEALTH → Joni Check
    on ALL CHARMS END → Deactivate
  }

  state Recheck {
    GetPlayerDataInt(gameObject=var "GameManager", intName="health", storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=1, equal=→"RETURN", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Deactivate
    on RETURN → Stay Furied
  }

  state Joni Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_27", isTrue=→"CANCEL", isFalse=→"RETURN")
    on CANCEL → Deactivate
    on RETURN → Stay Furied
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check HP
  }

  state Init Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state HP Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Recheck
  }

  state Get Ref {
    GameObjectIsNull(gameObject=var "Fury Vignette", isNull=(none), isNotNull=→"FINISHED", storeResult=(unset), everyFrame=false)
    FindChild(gameObject=var "HUD Camera", childName="fury_effects_v2", storeResult=var "Fury Vignette")
    on FINISHED → Activate
  }

  var Blue Health: int = 0
  var HP: int = 0
  var Has Blue Health: bool = false
  var Joni Equipped: bool = false
  var Alt Slash: gameObject = <null>
  var Attacks: gameObject = <null>
  var Cyclone Hit L: gameObject = <null>
  var Cyclone Hit R: gameObject = <null>
  var Cyclone Slash: gameObject = <null>
  var Dash Slash: gameObject = <null>
  var Down Slash: gameObject = <null>
  var Fury Object: gameObject = <null>
  var Fury Vignette: gameObject = <null>
  var Great Slash: gameObject = <null>
  var Knight: gameObject = <null>
  var Rage Burst Effect: gameObject = <null>
  var Slash: gameObject = <null>
  var Up Slash: gameObject = <null>
  var Wall Slash: gameObject = <null>
}
