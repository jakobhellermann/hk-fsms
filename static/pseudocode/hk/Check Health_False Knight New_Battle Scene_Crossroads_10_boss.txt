# Check Health
# 1 location(s):
#   Crossroads_10_boss / Battle Scene/False Knight New

fsm Check Health {
  start Init
  on FALLEN → Inert  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get HP
  }

  state Check {
    GetFsmInt(gameObject=Self, fsmName="health_manager", variableName="HP", storeValue=var "HP", everyFrame=true)  // disabled
    IntCompare(integer1=var "HP", integer2=0, equal=→"STUN", lessThan=→"STUN", greaterThan=(none), everyFrame=true)  // disabled
    on ZERO HP → Stun
  }

  state Stun {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="FalseyControl"), sendEvent="STUN", delay=0, everyFrame=false)
    SetHP(target=Self, hp=var "Recover HP")
    on FINISHED → Check
  }

  state Inert {
  }

  state Get HP {
    GetHP(target=Self, storeValue=var "Recover HP")
    on FINISHED → Check
  }

  var HP: int = 0
  var Recover HP: int = 65
}
