# Control
# 8 location(s):
#   Fungus3_23 / shadow_gate/Dash Effect
#   Fungus3_25 / shadow_gate/Dash Effect
#   Fungus3_44 / shadow_gate/Dash Effect
#   Deepnest_44 / shadow_gate/Dash Effect
#   Deepnest_East_18 / shadow_gate/Dash Effect
#   Abyss_10 / higher_being/shadow_gate/Dash Effect
#   Abyss_15 / shadow_gate (1)/Dash Effect
#   Abyss_15 / shadow_gate (2)/Dash Effect

fsm Control {
  start Init

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    SetParticleEmission(gameObject=var "Pt Bounce", emission=false)
    on COLLIDE → Shadow Dashing?
  }

  state Effect {
    ActivateGameObject(gameObject=var "Black Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=shadow_gate_pass_through (sharedassets208.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    FloatSubtract(floatVariable=var "Hero Y", subtract=0.5, everyFrame=false, perSecond=false)
    SpawnObjectFromGlobalPool(gameObject=Black Wave Default R (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Wave")
    SetPosition(gameObject=var "Wave", vector=(unset), x=var "Self X", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Pt Bounce", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Pt Bounce 2", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Wave", vector=(unset), x=(unset), y=var "Hero Y", z=-0.1, space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt Bounce", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Bounce 2", emit=0)
    on FINISHED → Pause
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Bounce", storeResult=var "Pt Bounce")
    FindChild(gameObject=Self, childName="Pt Bounce 2", storeResult=var "Pt Bounce 2")
    FindChild(gameObject=Self, childName="Black Flash", storeResult=var "Black Flash")
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Pause {
    SetParticleEmission(gameObject=var "Pt Bounce", emission=true)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Shadow Dashing? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["shadowDashing"], storeResult=var "Shadow Dashing")
    BoolTest(boolVariable=var "Shadow Dashing", isTrue=→"FINISHED", isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Effect
    on CANCEL → Idle
  }

  var Hero Y: float = 0
  var Self X: float = 0
  var Shadow Dashing: bool = false
  var Black Flash: gameObject = <null>
  var Pt Bounce: gameObject = <null>
  var Pt Bounce 2: gameObject = <null>
  var Self: gameObject = <null>
  var Wave: gameObject = <null>
}
