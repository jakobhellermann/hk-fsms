# grass ball control
# 1 location(s):
#   sharedassets149.assets / Grass Ball

fsm grass ball control {
  start Init

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"TERRAIN", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Hero Spell", collideLayer=Attack(17), sendEvent=→"TERRAIN", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=Attack(17), sendEvent=→"TERRAIN", storeCollider=var "")
    on TERRAIN → Break
    on ORBIT SHIELD → Break
  }

  state Break {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=moss_knight_projectile_break (sharedassets140.assets), pitchMin=0.85, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    StopParticleEmitter(gameObject=var "Spore Trail")
    StopParticleEmitter(gameObject=var "Idle Grass B 2")
    StopParticleEmitter(gameObject=var "Idle Grass B 3")
    PlayParticleEmitter(gameObject=var "Explode", emit=0)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    on FINISHED → Recycle
  }

  state Init {
    SetScale(gameObject=Self, vector=(unset), x=0.5, y=0.5, z=(unset), everyFrame=false, lateUpdate=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(1, 1, 1), time=0.1, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Idle Grass B 2", emit=0)
    PlayParticleEmitter(gameObject=var "Idle Grass B 3", emit=0)
    PlayParticleEmitter(gameObject=var "Spore Trail", emit=0)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FINISHED → Idle
  }

  state Recycle {
    RecycleSelf()
  }

  var Explode: gameObject = Grass Ball/Explode
  var Idle Grass B 2: gameObject = Grass Ball/Idle Grass B 2
  var Idle Grass B 3: gameObject = Grass Ball/Idle Grass B 3
  var Self: gameObject = Grass Ball
  var Spore Trail: gameObject = Grass Ball/Spore Trail
}
