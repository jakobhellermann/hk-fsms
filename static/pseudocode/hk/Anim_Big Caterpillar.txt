# Anim
# 1 location(s):
#   Deepnest_East_04 / Big Caterpillar

fsm Anim {
  start Init

  state Sleep {
    BoolTest(boolVariable=var "Wake Range", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="To Sleep")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on WAKE → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStop(gameObject=Self)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Caterpillar_wake (sharedassets309.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Right Range", isTrue=→"RIGHT", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Sleep Range", isTrue=(none), isFalse=→"SLEEP", everyFrame=true)
    on RIGHT → Look R
    on SLEEP → Sleep
  }

  state Look R {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look Right", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Right Range", isTrue=(none), isFalse=→"LEFT", everyFrame=true)
    BoolTest(boolVariable=var "Sleep Range", isTrue=(none), isFalse=→"SLEEP", everyFrame=true)
    on LEFT → Look L
    on SLEEP → Sleep
  }

  state Look L {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look Left", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Right Range", isTrue=→"RIGHT", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Sleep Range", isTrue=(none), isFalse=→"SLEEP", everyFrame=true)
    on RIGHT → Look R
    on SLEEP → Sleep
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Sleep
  }

  var Right Range: bool = false
  var Sleep Range: bool = false
  var Wake Range: bool = false
  var Self: gameObject = <null>
}
