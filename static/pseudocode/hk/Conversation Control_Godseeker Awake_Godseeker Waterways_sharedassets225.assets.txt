# Conversation Control
# 1 location(s):
#   sharedassets225.assets / Godseeker Waterways/Godseeker Awake

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on AWAKE 1 → Awake 1
    on AWAKE 2 → Awake 2
    on AWAKE 3 → Awake 3
  }

  state End {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    ActivateGameObject(gameObject=var "AreaTitle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "AreaTitle", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "AreaTitle", fsmName="", variableName="Area Event", setValue="GODSEEKER", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="godseekerSpokenAwake", value=true)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=0.3)
    on FINISHED → Flower Repeat?
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Greet {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName=var "Spoken PD Bool", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "First Talk", "CP3"], storeResult=(unset var))
    AudioPlayRandom(gameObject=var "Self", audioClips=[GS_waterway_01, GS_waterway_03, GS_waterway_02], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on CONVO_FINISH → Talk Finish
  }

  state Awake 1 {
    SetStringValue(stringVariable=var "First Talk", stringValue="GODSEEKER_WATERWAYS_AWAKE_1", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat", stringValue="GODSEEKER_WATERWAYS_AWAKE_REPEAT", everyFrame=false)
    SetStringValue(stringVariable=var "Spoken PD Bool", stringValue="godseekerWaterwaysSpoken1", everyFrame=false)
    on FINISHED → Get Spoken
  }

  state Awake 2 {
    SetStringValue(stringVariable=var "First Talk", stringValue="GODSEEKER_WATERWAYS_AWAKE_2", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat", stringValue="GODSEEKER_WATERWAYS_AWAKE_REPEAT", everyFrame=false)
    SetStringValue(stringVariable=var "Spoken PD Bool", stringValue="godseekerWaterwaysSpoken2", everyFrame=false)
    on FINISHED → Get Spoken
  }

  state Awake 3 {
    SetStringValue(stringVariable=var "First Talk", stringValue="GODSEEKER_WATERWAYS_AWAKE_3", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat", stringValue="GODSEEKER_WATERWAYS_AWAKE_3_REPEAT", everyFrame=false)
    SetStringValue(stringVariable=var "Spoken PD Bool", stringValue="godseekerWaterwaysSpoken3", everyFrame=false)
    on FINISHED → Get Spoken
  }

  state Repeat? {
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"GREET", everyFrame=false)
    on GREET → Greet
    on REPEAT → Repeat
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Repeat", "CP3"], storeResult=(unset var))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=GS_waterway_07)
    on CONVO_FINISH → Flower?
  }

  state Godfinder? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasGodfinder", isTrue=→"FINISHED", isFalse=→"FALSE")
    on FALSE → No Godfinder
    on FINISHED → Repeat?
  }

  state No Godfinder {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GODSEEKER_WATERWAYS_AWAKE_NO_GODFINDER", "CP3"], storeResult=(unset var))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=GS_waterway_05)
    on CONVO_FINISH → Talk Finish
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["ELDERBUG_FLOWER", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 3
  }

  state Decline Pause {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Talk Finish
  }

  state Flower Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GODSEEKER_WATERWAYS_WHITEFLOWER_REPEAT", "CP3"], storeResult=(unset var))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=GS_waterway_07)
    on CONVO_FINISH → Talk Finish
  }

  state Flower Greet {
    SetPlayerDataBool(boolName="givenGodseekerFlower", value=true)
    SetPlayerDataBool(boolName="hasXunFlower", value=false)
    SetBoolValue(boolVariable=var "Flower Repeat", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=Godseeker Waterways/Godseeker Awake/Flower, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Godseeker Waterways/Godseeker Awake/Flower Appear, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GODSEEKER_WATERWAYS_WHITEFLOWER", "CP3"], storeResult=(unset var))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=GS_waterway_04)
    on CONVO_FINISH → Talk Finish
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up YN
  }

  state Box Up 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flower Greet
  }

  state Flower Repeat? {
    AudioStop(gameObject=Self)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    BoolTest(boolVariable=var "Flower Repeat", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Godfinder?
    on TRUE → Flower Repeat
  }

  state Get Spoken {
    GetPlayerDataBool(boolName=var "Spoken PD Bool", storeValue=var "Spoken")
    on FINISHED → Idle
  }

  state Flower? {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasXunFlower", falseBool="givenEmilitiaFlower", isTrue=→"FLOWER", isFalse=→"FINISHED")  // disabled
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasXunFlower", isTrue=(none), isFalse=→"CANCEL")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="givenGodseekerFlower", isTrue=→"CANCEL", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFlowerBroken", isTrue=→"CANCEL", isFalse=(none))
    on CANCEL → Talk Finish
    on FINISHED → Box Down 2
  }

  var Flower Repeat: bool = false
  var Spoken: bool = false
  var First Talk: string = ""
  var Repeat: string = "GODSEEKER_WATERWAYS_AWAKE_REPEAT"
  var Spoken PD Bool: string = ""
  var Self: gameObject = <null>
}
