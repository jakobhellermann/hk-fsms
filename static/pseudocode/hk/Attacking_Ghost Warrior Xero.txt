# Attacking
# 3 location(s):
#   RestingGrounds_02_boss / Warrior/Ghost Warrior Xero
#   GG_Ghost_Xero / Warrior/Ghost Warrior Xero
#   GG_Ghost_Xero_V / Warrior/Ghost Warrior Xero

fsm Attacking {
  start Init
  on ATTACK STOP → Attack Stop  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Xero_Cry (sharedassets238.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    FindGameObject(objectName="Sword 1", withTag="Untagged", store=var "Sword 1")
    FindGameObject(objectName="Sword 2", withTag="Untagged", store=var "Sword 2")
    FindChild(gameObject=Self, childName="Sword 3", storeResult=var "Sword 3")
    FindChild(gameObject=Self, childName="Sword 4", storeResult=var "Sword 4")
    on READY → Wait
  }

  state Wait {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"READY", realTime=false)
    BoolTest(boolVariable=var "Rage", isTrue=→"RAGE", isFalse=(none), everyFrame=false)
    on FINISHED → Check Hero Pos
    on RAGE → Wait Rage
  }

  state Antic {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Xero_Cast_01 (sharedassets238.assets), Xero_Cast_02 (sharedassets238.assets), Xero_Cast_03 (sharedassets238.assets), Xero_Cast_04 (sharedassets238.assets), Xero_Cast_05 (sharedassets238.assets), Xero_Cast_06 (sharedassets238.assets), Xero_Cast_07 (sharedassets238.assets), Xero_Cast_08 (sharedassets238.assets), Xero_Cast_09 (sharedassets238.assets)], weights=[1, 1, 1, 1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    BoolTestMulti(boolVariables=[var "Rage", var "Summoned Swords"], boolStates=[true, false], trueEvent=→"SUMMON", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Rage", isTrue=→"RAGE", isFalse=(none), everyFrame=false)
    GetFsmBool(gameObject=var "Sword 1", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    GetFsmBool(gameObject=var "Sword 2", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking 2", everyFrame=false)
    BoolAllTrue(boolVariables=[var "Sword Attacking", var "Sword Attacking 2"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK START", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choose
    on CANCEL → Wait
    on RAGE → Antic Rage
    on SUMMON → Summon Antic
  }

  state Recover {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK END", delay=0, everyFrame=false)
    on FINISHED → Wait
  }

  state Attack Stop {
    on ATTACK OK → Wait
  }

  state Choose {
    SendRandomEvent(events=[(none), (none)], weights=[1, 1], delay=0)
    on L → Attack 1
    on R → Attack 2
  }

  state Attack 1 {
    GetFsmBool(gameObject=var "Sword 1", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    BoolTest(boolVariable=var "Sword Attacking", isTrue=(none), isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sword 1"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Recover
    on R → Attack 2
  }

  state Attack 2 {
    GetFsmBool(gameObject=var "Sword 2", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    BoolTest(boolVariable=var "Sword Attacking", isTrue=(none), isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sword 2"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Recover
    on L → Attack 1
  }

  state Check Hero Pos {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=73.02, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=127.51, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    FloatCompare(float1=var "Hero Y", float2=7.8, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on FINISHED → Antic
    on CANCEL → Wait
  }

  state Wait Rage {
    WaitRandom(timeMin=0.75, timeMax=0.75, finishEvent=→"READY", realTime=false)
    on FINISHED → Check Hero Pos
  }

  state Antic Rage {
    GetFsmBool(gameObject=var "Sword 1", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    GetFsmBool(gameObject=var "Sword 2", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking 2", everyFrame=false)
    GetFsmBool(gameObject=var "Sword 3", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking 3", everyFrame=false)
    GetFsmBool(gameObject=var "Sword 4", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking 4", everyFrame=false)
    BoolAllTrue(boolVariables=[var "Sword Attacking", var "Sword Attacking 2", var "Sword Attacking 3", var "Sword Attacking 4"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK START", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choose Rage
    on CANCEL → Check Hero Pos
  }

  state Choose Rage {
    SendRandomEvent(events=[(none), (none), (none), (none)], weights=[0.25, 0.25, 0.25, 0.25], delay=0)
    on 1 → Attack 1r
    on 2 → Attack 2r
    on 3 → Attack 3
    on 4 → Attack 4
  }

  state Attack 3 {
    GetFsmBool(gameObject=var "Sword 3", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    BoolTest(boolVariable=var "Sword Attacking", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sword 3"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Recover
    on CANCEL → Attack 4
  }

  state Attack 4 {
    GetFsmBool(gameObject=var "Sword 4", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    BoolTest(boolVariable=var "Sword Attacking", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sword 4"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Recover
    on CANCEL → Attack 1r
  }

  state Attack 1r {
    GetFsmBool(gameObject=var "Sword 1", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    BoolTest(boolVariable=var "Sword Attacking", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sword 1"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Recover
    on CANCEL → Attack 2r
  }

  state Attack 2r {
    GetFsmBool(gameObject=var "Sword 2", fsmName="xero_nail", variableName="Attacking", storeValue=var "Sword Attacking", everyFrame=false)
    BoolTest(boolVariable=var "Sword Attacking", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sword 2"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Recover
    on CANCEL → Attack 3
  }

  state Summon Antic {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK START", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_enter (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "SFX Player")
    on FINISHED → Summon
  }

  state Summon {
    AudioStop(gameObject=var "Attack Pt")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_enter_pt_2 (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Sword 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sword 4", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Attack Pt", emit=0)
    SetBoolValue(boolVariable=var "Summoned Swords", boolValue=true, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recover
  }

  var Hero X: float = 0
  var Hero Y: float = 0
  var Rage: bool = false
  var Summoned Swords: bool = false
  var Sword Attacking: bool = false
  var Sword Attacking 2: bool = false
  var Sword Attacking 3: bool = false
  var Sword Attacking 4: bool = false
  var Attack Pt: gameObject = Warrior/Ghost Warrior Xero/Attack Pt
  var SFX Player: gameObject = <null>
  var Self: gameObject = <null>
  var Sword 1: gameObject = <null>
  var Sword 2: gameObject = <null>
  var Sword 3: gameObject = <null>
  var Sword 4: gameObject = <null>
}
