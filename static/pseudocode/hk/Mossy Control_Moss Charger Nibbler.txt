  start Start

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Get Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Direction
  }

  state Run L {
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AccelerateVelocity(gameObject=Self, xAccel=-0.5, yAccel=(unset), xMaxSpeed=10, yMaxSpeed=(unset))
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    RayCast2d(fromGameObject=Self, fromPosition=(unset), direction=(-1, 0), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"SUBMERGE", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=Self, fromPosition=(-3, 0), direction=(0, -1), space=1, distance=1.3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Ground Ahead", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    BoolTest(boolVariable=var "Ground Ahead", isTrue=(none), isFalse=→"SUBMERGE", everyFrame=true)
    Wait(time=1, finishEvent=→"SUBMERGE", realTime=false)
    on LEFT → Run R
    on SUBMERGE → On Ground?
  }

  state Run R {
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AccelerateVelocity(gameObject=Self, xAccel=0.5, yAccel=(unset), xMaxSpeed=10, yMaxSpeed=(unset))
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    RayCast2d(fromGameObject=Self, fromPosition=(unset), direction=(1, 0), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"SUBMERGE", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=Self, fromPosition=(3, 0), direction=(0, -1), space=1, distance=1.3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "Ground Ahead", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=0, repeatInterval=1, layerMask=[Terrain(8), Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)

    Wait(time=1, finishEvent=→"SUBMERGE", realTime=false)
    on RIGHT → Run L
    on SUBMERGE → On Ground?
  }

  state Direction {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="TurnRun")
    Tk2dPlayFrame(gameObject=Self, frame=2)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on LEFT → Run R
    on RIGHT → Run L
  }

  state In Air {
    SetFsmBool(gameObject=Self, fsmName="health_manager_enemy", variableName="Invincible", setValue=false, everyFrame=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"DOWN", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"DOWN", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetCollider(gameObject=Self, active=true)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetGravity2dScale(gameObject=Self, gravityScale=1.5)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun")
    StopParticleEmitter(gameObject=var "Grass Dig")
    on DOWN → Land
  }

  state Land {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Get Up
  }

  state Dig Start {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CANCEL RECOIL", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    Decelerate(gameObject=Self, deceleration=0.4)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Escape", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetFsmBool(gameObject=Self, fsmName="health_manager_enemy", variableName="Invincible", setValue=false, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(true))
    BoolTest(boolVariable=var "Dig Check", isTrue=(none), isFalse=→"FALL", everyFrame=true)  // disabled
    on FINISHED → Dig
  }

  state On Ground? {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=var "Grounded2", leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=var "Grounded", leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolAnyTrue(boolVariables=[var "Grounded", var "Grounded2"], sendEvent=→"DOWN", storeResult=(unset), everyFrame=true)
    on DOWN → Dig Start
    on FINISHED → Run L
  }

  state Dig {
    PlayParticleEmitter(gameObject=var "Grass Dig", emit=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[grub_burrow (sharedassets39.assets)], weights=[1], pitchMin=1, pitchMax=1)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    BoolTest(boolVariable=var "Dig Check", isTrue=(none), isFalse=→"FALL", everyFrame=true)
    on FINISHED → Destroy
  }

  state Start {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="INVULN", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "G Explode 1", emit=0)
    StopParticleEmitter(gameObject=var "Charge Grass")
    StopParticleEmitter(gameObject=var "Charge Grass")
    SetGravity2dScale(gameObject=Self, gravityScale=1.5)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun")
    GetFsmFloat(gameObject=Self, fsmName="health_manager_enemy", variableName="Attack Direction", storeValue=var "Attack Direction", everyFrame=false)
    RandomFloat(min=60, max=120, storeResult=var "Chooser")
    RandomFloat(min=28, max=38, storeResult=var "Random")
    SetVelocityAsAngle(gameObject=Self, angle=var "Chooser", speed=var "Random", everyFrame=false)
    RandomFloat(min=-100, max=-80, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    FindChild(gameObject=Self, childName="Grass Dig", storeResult=var "Grass Dig")
    on FINISHED → In Air
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Accel: float = 0.13

  var Attack Direction: float = 0

  var Chooser: float = 0
  var Current Charge Speed: float = 0
  var Emerge Speed: float = 0
  var Hero X: float = 0
  var Max Speed L: float = -6
  var Max Speed R: float = 6
  var Random: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Velocity: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var Looper: int = 0
  var Dig Check: bool = false
  var Ground Ahead: bool = false
  var Grounded: bool = false
  var Grounded2: bool = false
  var In Attack Range: bool = false
  var RayDown1: bool = false
  var left: bool = false
  var right: bool = false

}
