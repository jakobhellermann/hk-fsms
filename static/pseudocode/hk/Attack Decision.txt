fsm Attack Decision {
  start Init

  state Init {
    on FINISHED → Check
  }

  state Check {
    BoolTest(boolVariable=var "Attack Range", isTrue=→"ATTACK", isFalse=(none), everyFrame=true)
    SetBoolValue(boolVariable=var "Attack Range", boolValue=false, everyFrame=false)
    on ATTACK → Wait
  }

  state Wait {
    WaitRandom(timeMin=0.25, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recheck
  }

  state Recheck {
    BoolTest(boolVariable=var "Attack Range", isTrue=→"ATTACK", isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Check
    on ATTACK → Attack
  }

  state Attack {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Check
  }

  var Attack Range: bool = false
}
