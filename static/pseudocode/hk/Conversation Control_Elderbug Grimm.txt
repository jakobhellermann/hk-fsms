# Conversation Control
# 1 location(s):
#   Town / _NPCs/Elderbug Grimm

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue Normal")
    FindChild(gameObject=Self, childName="Dream Dialogue Flower", storeResult=var "Dream Dialogue Flower")
    FindChild(gameObject=Self, childName="Flower", storeResult=var "Flower")
    FindChild(gameObject=Self, childName="Flower Give", storeResult=var "Flower Give")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Grimm?
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ELDERBUG", everyFrame=false)
    on FINISHED → Convo
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Grimm? {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="troupeInTown", falseBool="elderbugConvoGrimm", isTrue=(none), isFalse=→"DISABLE")
    on FINISHED → Idle
    on DISABLE → Disable
  }

  state Disable {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Convo {
    SetPlayerDataBool(boolName="elderbugConvoGrimm", value=true)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Elderbug_01 (sharedassets7.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ELDERBUG_GRIMM", "CP2"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var Elderbug State: int = 0
  var History: int = 0
  var Dung Convo: bool = false
  var Dung Equipped: bool = false
  var Flower Broken: bool = false
  var Gave Flower: bool = false
  var Given Hint: bool = false
  var Had Flower Convo: bool = false
  var Have Flower: bool = false
  var Met Elderbug: bool = false
  var Offered Flower: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var Area Title: gameObject = <null>
  var Dream Dialogue Flower: gameObject = <null>
  var Dream Dialogue Normal: gameObject = <null>
  var Flower: gameObject = <null>
  var Flower Give: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
