# Battle Control
# 1 location(s):
#   Crossroads_10_boss / Battle Scene

fsm Battle Control {
  start Pause

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"START", storeCollider=var "")
    on START → Start
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "CameraLock 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DestroyObject(gameObject=var "Armour", delay=0, detachChildren=false)
    on FINISHED → Detect
    on ACTIVATED → Activate
  }

  state Start {
    SetIntValue(intVariable=var "Battle Enemies", intValue=1, everyFrame=false)
    DestroyObject(gameObject=var "Tension Region", delay=0, detachChildren=false)
    ActivateGameObject(gameObject=var "CameraLock 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CameraLock 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on END → End Wait
    on KILL ALL ENEMIES → Kill Zombies
  }

  state End Wait {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DestroyObject(gameObject=var "CameraLock 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "CameraLock 2", delay=0, detachChildren=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    DestroyObject(gameObject=var "Tension Region", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    DestroyObject(gameObject=var "CameraLock 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "CameraLock 2", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "False Knight", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Pre Battle Enemies", delay=0, detachChildren=false)
    SendEventByName(eventTarget=GameObject(var "Floor"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
  }

  state Kill Zombies {
    IntCompare(integer1=var "Battle Enemies", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    SetInvincible(target=var "Hornhead", Invincible=false, InvincibleFromDirection=0)
    TakeDamage(Target=var "Hornhead", AttackType=1, CircleDirection=false, DamageDealt=9999, Direction=90, IgnoreInvulnerable=true, MagnitudeMultiplier=1.5, MoveAngle=(unset), MoveDirection=(unset), Multiplier=(unset), SpecialType=(unset))
    SetInvincible(target=var "Runner", Invincible=false, InvincibleFromDirection=0)
    TakeDamage(Target=var "Runner", AttackType=1, CircleDirection=false, DamageDealt=9999, Direction=90, IgnoreInvulnerable=true, MagnitudeMultiplier=1.5, MoveAngle=(unset), MoveDirection=(unset), Multiplier=(unset), SpecialType=(unset))
    SetInvincible(target=var "Barger", Invincible=false, InvincibleFromDirection=0)
    TakeDamage(Target=var "Barger", AttackType=1, CircleDirection=false, DamageDealt=9999, Direction=90, IgnoreInvulnerable=true, MagnitudeMultiplier=1.5, MoveAngle=(unset), MoveDirection=(unset), Multiplier=(unset), SpecialType=(unset))
    on END → End Wait
  }

  var Armour: gameObject = Battle Scene/FK Armour
  var Barger: gameObject = Battle Scene/Pre Battle Enemies/Zombie Barger
  var CameraLock 1: gameObject = Battle Scene/CameraLockArea B
  var CameraLock 2: gameObject = Battle Scene/CameraLockArea B2
  var False Knight: gameObject = Battle Scene/False Knight New
  var Floor: gameObject = Battle Scene/FK Floor
  var Hornhead: gameObject = Battle Scene/Pre Battle Enemies/Zombie Hornhead
  var Pre Battle Enemies: gameObject = Battle Scene/Pre Battle Enemies
  var Runner: gameObject = Battle Scene/Pre Battle Enemies/Zombie Runner
  var Tension Region: gameObject = Battle Scene/Music Region B

  var Battle Enemies: int = 0
  var Activated: bool = false
}
