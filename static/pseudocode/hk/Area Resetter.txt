# Area Resetter
# 1 location(s):
#   Town / Area Resetter

fsm Area Resetter {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"TOUCH", storeCollider=var "")
    on TOUCH → Reset
  }

  state Reset {
    SetPlayerDataInt(intName="currentArea", value=-1)
  }
}
