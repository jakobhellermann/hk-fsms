# Deactivate
# 1 location(s):
#   GG_Atrium_Roof / Land of Storms Doors/GG_secret_door

fsm Deactivate {
  start Get Bindings

  state Get Bindings {
    GGGetTotalBindings(storeValue=var "Total")
    GGGetCompletedBindings(storeValue=var "Completed")
    IntCompare(integer1=var "Completed", integer2=var "Total", equal=(none), lessThan=→"DISABLE", greaterThan=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedGodseekerMask", isTrue=→"DISABLE", isFalse=(none))
    on FINISHED → Inert
    on DISABLE → Wait
  }

  state Inert {
  }

  state Disable {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Disable
  }

  var Completed: int = 0
  var Total: int = 0
}
