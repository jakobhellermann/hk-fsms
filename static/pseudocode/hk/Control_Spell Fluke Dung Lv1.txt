# Control
# 1 location(s):
#   resources.assets / Spell Fluke Dung Lv1

fsm Control {
  start Init
  on ENEMY HIT → Blow  // from any state

  state Init {
    ActivateGameObject(gameObject=var "Dung Cloud", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Detect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dung Air")
    SetMeshRenderer(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetPolygonCollider(gameObject=Self, active=true)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_19", isTrue=→"SPELL UP", isFalse=→"NORMAL")
    on NORMAL → Normal
    on SPELL UP → Spell Up
  }

  state Air {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spaz
  }

  state Flash {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Current Scale", yScale=var "Current Y Scale", zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Current Scale", multiplyBy=var "Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "Current Y Scale", multiplyBy=var "Scale", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Current Scale", y=var "Current Y Scale", z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Air
  }

  state Normal {
    RandomFloat(min=1, max=1, storeResult=var "Scale")
    CallMethodProper(gameObject=var "Dung Cloud", behaviour="DamageEffectTicker", methodName="SetDamageInterval", parameters=[0.1], storeResult=(unset var))
    on FINISHED → Flash
  }

  state Spell Up {
    RandomFloat(min=1.1, max=1.1, storeResult=var "Scale")
    CallMethodProper(gameObject=var "Dung Cloud", behaviour="DamageEffectTicker", methodName="SetDamageInterval", parameters=[0.075], storeResult=(unset var))
    on FINISHED → Flash
  }

  state Spaz {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dung Antic")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0)
    on FINISHED → Blow
  }

  state Blow {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClip=explosion_4_wet_original, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Enemy Detect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Dung Cloud", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Antic")
    ActivateGameObject(gameObject=var "Splat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Dung Cloud", activate=true)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Current Scale: float = 0
  var Current Y Scale: float = 0
  var Scale: float = 0
  var Damager: gameObject = Spell Fluke Dung Lv1/Damager
  var Dung Cloud: gameObject = Spell Fluke Dung Lv1/Knight Dung Cloud
  var Enemy Detect: gameObject = Spell Fluke Dung Lv1/Enemy Detect
  var Pt Antic: gameObject = Spell Fluke Dung Lv1/Pt Antic
  var Self: gameObject = <null>
  var Splat: gameObject = Spell Fluke Dung Lv1/Splat
}
