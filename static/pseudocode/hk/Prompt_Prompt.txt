# Prompt
# 1 location(s):
#   Crossroads_01 / Ascender Crossroads/Prompt

fsm Prompt {
  start Init

  state Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Blank")
    on UP → Go Up
  }

  state Go Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ascend Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Up
    on DOWN → Down
  }

  state Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ascend Idle")
    on DOWN → Down
  }

  state Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ascend Down")
    on UP → Go Up
  }
}
