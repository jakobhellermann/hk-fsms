# Conversation Control
# 1 location(s):
#   Room_Queen / Queen

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Look
    on GET ITEM MSG END → Collider On
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → NPC Anim
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="QUEEN", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Meet {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Queen_generic_01 (sharedassets348.assets), Queen_generic_02 (sharedassets348.assets), Queen_generic_03 (sharedassets348.assets), Queen_generic_04 (sharedassets348.assets), Queen_generic_05 (sharedassets348.assets), Queen_generic_06 (sharedassets348.assets)], weights=[1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetPlayerDataBool(boolName="metQueen", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUEEN_MEET", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Box Down 2
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state NPC Anim {
    BoolTest(boolVariable=var "Awake", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Awake", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wake")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="AWAKE", delay=0, everyFrame=false)
    Wait(time=1.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up
  }

  state Convo Choice {
    GetPlayerDataBool(boolName="killedNightmareGrimm", storeValue=var "Defeated Nightmare Grimm")
    GetPlayerDataBool(boolName="destroyedNightmareLantern", storeValue=var "Destroyed Nightmare Lantern")
    GetPlayerDataBool(boolName="queenConvo_grimm1", storeValue=var "Grimm Convo 1")
    GetPlayerDataBool(boolName="queenConvo_grimm2", storeValue=var "Grimm Convo 2")
    GetPlayerDataBool(boolName="equippedCharm_40", storeValue=var "Equipped Grimm")
    SetAudioVolume(gameObject=var "Voice Player", volume=0, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metQueen", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_40", falseBool="queenConvo_grimm1", isTrue=→"GRIMM 1", isFalse=(none))
    BoolTestMulti(boolVariables=[var "Equipped Grimm", var "Grimm Convo 1", var "Destroyed Nightmare Lantern"], boolStates=[true, false, false], trueEvent=→"GRIMM 1", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Defeated Nightmare Grimm", var "Equipped Grimm", var "Grimm Convo 2", var "Destroyed Nightmare Lantern"], boolStates=[true, true, false, false], trueEvent=→"GRIMM 2", falseEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_10", falseBool="queenDung1", isTrue=→"DUNG1", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_10", falseBool="queenDung2", isTrue=→"DUNG2", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasKingsBrand", falseBool="queenHornet", isTrue=→"HORNET", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="queenTalk1", isTrue=(none), isFalse=→"TALK1")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="queenTalk2", isTrue=(none), isFalse=→"TALK2")
    BoolTest(boolVariable=var "Spoken", isTrue=(none), isFalse=→"REMEET", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    IntSwitch(intVariable=var "Royal Charm State", compareTo=[3, 4], sendEvent=[→"KINGSOUL", →"SHADECHARM"], everyFrame=false)
    on MEET → Meet
    on FINISHED → Repeat
    on TALK1 → Talk 1
    on TALK2 → Talk 2
    on HORNET → Hornet
    on DUNG1 → Dung 1
    on DUNG2 → Dung 2
    on KINGSOUL → Kingsoul Repeat
    on REMEET → Remeet
    on SHADECHARM → Shadecharm Repeat
    on GRIMM 1 → Grimm 1
    on GRIMM 2 → Grimm 2
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Summon
  }

  state Summon {
    PlayParticleEmitter(gameObject=var "Pt Summon", emit=0)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[focus_health_charging (resources.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Appear
  }

  state Appear {

    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "White Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)

    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[shaman_spell (sharedassets43.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "")
    on FINISHED → Talk Finish
  }

  state Collider On {

    on FINISHED → Idle
  }

  state Repeat {

  }

  }

    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Queen_generic_01 (sharedassets348.assets), Queen_generic_02 (sharedassets348.assets), Queen_generic_03 (sharedassets348.assets), Queen_generic_04 (sharedassets348.assets), Queen_generic_05 (sharedassets348.assets), Queen_generic_06 (sharedassets348.assets)], weights=[1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")

    SetPlayerDataBool(boolName="givenWhiteLadyFlower", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUEEN_FLOWER", "CP3"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Decline Pause 2 {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Talk Finish
  }

  state Box Up YN 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text 2
  }

  state Send Text 2 {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=["ELDERBUG_FLOWER", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause 2
    on YES → Yes 2
  }

  state Yes 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 10
  }

  state Box Up 10 {

  state Box Down 3 {

  state Flower? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasXunFlower", isTrue=(none), isFalse=→"CANCEL")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="givenWhiteLadyFlower", isTrue=→"CANCEL", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunFlowerBroken", isTrue=→"CANCEL", isFalse=(none))

    on FINISHED → Box Down 3
  }

  var Royal Charm State: int = 0
  var Awake: bool = false
  var Defeated Nightmare Grimm: bool = false
