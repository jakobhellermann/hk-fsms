# Control
# 1 location(s):
#   Abyss_19 / IK Remains

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Chinese?
  }

  state Check {
    FindChild(gameObject=Self, childName="Base Glow", storeResult=var "Base Glow")
    FindChild(gameObject=Self, childName="Idle Pt", storeResult=var "Idle Pt")
    FindChild(gameObject=Self, childName="Dream Enter", storeResult=var "Dream Enter")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="infectedKnightDreamDefeated", isTrue=→"NO", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDreamNail", isTrue=→"GLOW", isFalse=→"NO")
    on GLOW → Glow
    on NO → Off
  }

  state Off {
    ActivateGameObject(gameObject=var "Dream Enter", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Glow {
    PlayParticleEmitter(gameObject=var "Idle Pt", emit=0)
    SendEventByName(eventTarget=GameObject(var "Base Glow"), sendEvent="UP", delay=0, everyFrame=false)
  }

  state Chinese? {
    CheckIsChineseBuild(trueEvent=(none), falseEvent=→"FINISHED")
    on FINISHED → Check
  }

  var Base Glow: gameObject = <null>
  var Dream Enter: gameObject = <null>
  var Idle Pt: gameObject = <null>
}
