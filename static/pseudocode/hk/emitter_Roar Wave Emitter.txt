# emitter
# 1 location(s):
#   sharedassets13.assets / Roar Wave Emitter

fsm emitter {
  start Init
  on ROAR WAVE END → Destroy  // from any state

  state Emit {
    DestroyObject(gameObject=var "Lines", delay=0, detachChildren=false)
    SetScale(gameObject=var "Wave 1", vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Wave 2", vector=(unset), x=5.5, y=5.5, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleBy(gameObject=var "Wave 1", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=loop, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Wave 2", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=loop, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on END → End
  }

  state Init {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CLOSE INVENTORY", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Rubble Fall", storeResult=var "Rubble")
    FindChild(gameObject=Self, childName="wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="wave 2", storeResult=var "Wave 2")
    FindChild(gameObject=Self, childName="lines", storeResult=var "Lines")
    SetMaterialColor(gameObject=var "Lines", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    SetMaterialColor(gameObject=var "Wave 1", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    SetMaterialColor(gameObject=var "Wave 2", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    SetParticleEmission(gameObject=var "Rubble", emission=false)
    on FINISHED → Pause
    on END → End
  }

  state Emit Start {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ROAR WAVE START", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="MedRumble", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Rubble", emission=true)
    SetMaterialColor(gameObject=var "Lines", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    SetMaterialColor(gameObject=var "Wave 1", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0.5372549), everyFrame=false)
    iTweenScaleBy(gameObject=var "Wave 1", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Lines", id="", vector=(6, 6, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish Lines
    on END → End
  }

  state Finish Lines {
    SetMaterialColor(gameObject=var "Wave 2", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0.5529412), everyFrame=false)
    SetScale(gameObject=var "Wave 1", vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleBy(gameObject=var "Wave 1", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Wave 2", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Lines", id="", vector=(6, 6, 0), time=0.15, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emit
    on END → End
  }

  state End {
    SetParticleEmission(gameObject=var "Rubble", emission=false)
    DestroyObject(gameObject=var "Wave 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Wave 2", delay=0, detachChildren=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    on FINISHED → Destroy
  }

  state Destroy {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="StopRumble", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Emit Start
    on END → End
  }

  var Lines: gameObject = <null>
  var Rubble: gameObject = <null>
  var Wave 1: gameObject = <null>
  var Wave 2: gameObject = <null>
}
