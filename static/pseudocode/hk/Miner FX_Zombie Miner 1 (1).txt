fsm Miner FX {
  start Init
  on STOP MINING → Stop Mining  // from any state

  state Idle {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Frame ID", everyframe=true)
    IntCompare(integer1=var "Frame ID", integer2=21, equal=→"PLAY", lessThan=(none), greaterThan=(none), everyFrame=true)
    on PLAY → Choice
  }

  state Play {
    SpawnRandomObjects(gameObject=Particle Rock Tiny Transient Plain (resources.assets), spawnPoint=var "Dust", position=(0, 0, -1), spawnMin=2, spawnMax=4, speedMin=5, speedMax=10, angleMin=110, angleMax=140, originVariation=0)  // disabled
    SetParticleEmissionRate(gameObject=var "Dust", emissionRate=80, everyFrame=false)  // disabled
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pause
  }

  state Pause {
    SetParticleEmissionRate(gameObject=var "Dust", emissionRate=0, everyFrame=false)  // disabled
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Idle
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dust", storeResult=var "Dust")
    on FINISHED → Idle
  }

  state Choice {
    RandomFloat(min=0.85, max=1.15, storeResult=var "Chooser")
    SendRandomEvent(events=[(none), (none)], weights=[1, 1], delay=0)
    on 1 → 1
    on 2 → 2
  }

  state 2 {
    AudioPlay(gameObject=var "Self", volume=var "Chooser", oneShotClip=myla_pickaxe_2 (sharedassets73.assets), finishedEvent=(none))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Play
  }

  state 1 {
    AudioPlay(gameObject=var "Self", volume=var "Chooser", oneShotClip=myla_pickaxe_1 (sharedassets73.assets), finishedEvent=(none))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Play
  }

  state Stop Mining {
  }

  var Chooser: float = 0
  var Frame ID: int = 0
  var Dust: gameObject = <null>
  var Self: gameObject = <null>
}
