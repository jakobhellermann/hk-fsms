# Conversation Control
# 1 location(s):
#   Ruins2_Watcher_Room / Telescope Inspect

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Face Left
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Cinematic", store=var "Cinematic Player")
    FindChild(gameObject=Self, childName="Telescope Fader", storeResult=var "Telescope Fader")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Anim End {
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
  }

  state Face Left {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Fade
  }

  state Fade {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=1.1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    on FINISHED → Scene Blanker
  }

  state Cinematic {
    SendMessage(gameObject=var "Cinematic Player", delivery=SendMessage, options=1, functionCall=TriggerStartVideo())
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Telescope Fader", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Can End
  }

  state Fade Back {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=1.1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER OFF", delay=0, everyFrame=false)
    on FINISHED → Talk Finish
  }

  state Can End {
    ListenForMenuSubmit(eventTarget=Self, wasPressed=→"END", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForMenuCancel(eventTarget=Self, wasPressed=→"END", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"END", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForCast(eventTarget=Self, wasPressed=→"END", wasReleased=(none), isPressed=(none), isNotPressed=(none), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"END", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on END → Stop
  }

  state Stop {
    SendMessage(gameObject=var "Cinematic Player", delivery=SendMessage, options=1, functionCall=TriggerStopVideo())
    on FINISHED → Wait
  }

  state Wait {
    Wait(time=0.55, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Back
  }

  state Scene Blanker {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER ON", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Cinematic
  }

  var Cinematic Player: gameObject = <null>
  var Self: gameObject = <null>
  var Telescope Fader: gameObject = <null>
}
