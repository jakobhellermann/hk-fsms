# FSM
# 1 location(s):
#   GG_Grimm / Grimm Scene

fsm FSM {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Grimm_teleport_effect", storeResult=var "Grimm_teleport_effect")
    FindChild(gameObject=Self, childName="Grimm_teleport_effect2", storeResult=var "Grimm_teleport_effect2")
    FindChild(gameObject=Self, childName="Godseeker Crowd", storeResult=var "Godseeker")
    on FINISHED → Battle Ready
  }

  state Battle Ready {
    ActivateGameObject(gameObject=var "Grimm NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Boss", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PreSpawnCorpse(target=var "Boss")
    on FINISHED → Appear Pause 4
  }

  state Battle Start {
    SetPlayerDataBool(boolName="isInvincible", value=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    SetGameObject(variable=var "Area Title2", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title2", fsmName="", variableName="Display Right", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title2", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title2", fsmName="", variableName="Area Event", setValue="GRIMM", everyFrame=false)
    ActivateGameObject(gameObject=var "Grimm NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Appear Pause 4 {
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tele SFX 4
  }

  state Grimm Appear 4 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")  // disabled
    SendEventByName(eventTarget=GameObject(var "Spotlight"), sendEvent="UP", delay=0, everyFrame=false)  // disabled
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")  // disabled
    ActivateGameObject(gameObject=var "Grimm NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Grimm_teleport_effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Grimm NPC", animLibName="", clipName="TeleToIdle")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Battle Start
  }

  state Tele SFX 4 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=grimm_explode_into_bats (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetFsmGameObject(gameObject=var "Godseeker", fsmName="Control", variableName="Target", setValue=var "Boss", everyFrame=false)
    on FINISHED → Grimm Appear 4
  }

  var Hero Y: float = 0
  var Area Title2: gameObject = <null>
  var Boss: gameObject = Grimm Scene/Grimm Boss
  var Godseeker: gameObject = <null>
  var Grimm NPC: gameObject = Grimm Scene/Grimm NPC
  var Grimm_teleport_effect: gameObject = Grimm Scene/Grimm_teleport_effect
  var Grimm_teleport_effect2: gameObject = Grimm Scene/Grimm_teleport_effect2
  var Self: gameObject = <null>
  var Spotlight: gameObject = Grimm Scene/Spotlight (sharedassets391.assets)
}
