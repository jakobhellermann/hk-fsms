#   sharedassets398.assets / White Defender:0/Slam Pillars/Dung Pillar (2)
#   sharedassets398.assets / White Defender:0/Slam Pillars/Dung Pillar (5)

// uses template: dung_pillar_ground
fsm Control {
  start Damage

  state Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dung Pillar Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Hit
  }

  state Hit {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dung Pillar Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetPolygonCollider(gameObject=Self, active=true)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Break {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dung Pillar Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false)
    PlayParticleEmitter(gameObject=var "Chunks", emit=0)
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false)
    on PILLAR UP → Delay
  }

  state Delay {
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
  }

  state Init {
    FindChild(gameObject=Self, childName="Chunks", storeResult=var "Chunks")
    IntAdd(intVariable=var "Damage", add=1, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=var "Damage")
    on FINISHED → Dormant
  }

  state Damage {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    GetPlayerDataInt(gameObject=var "GameManager", intName="whiteDefenderDefeats", storeValue=var "Damage")
    on FINISHED → Init
  }

  var Delay: float = 0.1

  var Damage: int = 0
  var Chunks: gameObject = <null>
}
