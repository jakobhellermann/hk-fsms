# Shroom Turret
# 1 location(s):
#   Fungus2_11 / Mushroom Turret

fsm Shroom Turret {
  start Init
  on HERO EXIT → Hero Away  // from any state

  state Init {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Fire Angle", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Fire Angle", add=90, everyFrame=false, perSecond=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Dormant
  }

  state Dormant {
    on HERO ENTER → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sleep to Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dung_defender_roll_dung_ball (sharedassets167.assets), pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → First Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 1")
    WaitRandom(timeMin=3.5, timeMax=4.25, finishEvent=→"ATTACK", realTime=false)
    on ATTACK → Attack Antic
  }

  state Attack Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Fire
  }

  state Fire {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Spore Bomb (sharedassets167.assets), spawnPoint=var "Spawn Point", position=(unset), rotation=(unset), storeObject=var "Projectile")
    SetVelocityAsAngle(gameObject=var "Projectile", angle=var "Fire Angle", speed=5.5, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Spore Cloud", emit=0)
    on FINISHED → Idle
  }

  state First Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 1")
    WaitRandom(timeMin=0.5, timeMax=0.5, finishEvent=→"ATTACK", realTime=false)
    on ATTACK → Attack Antic
  }

  state Hero Away {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 1")
    on HERO ENTER → First Idle
  }

  var Fire Angle: float = 0
  var Projectile: gameObject = <null>
  var Self: gameObject = <null>
  var Spawn Point: gameObject = Mushroom Turret/Spawn Point
  var Spore Cloud: gameObject = Mushroom Turret/Spore Cloud
}
