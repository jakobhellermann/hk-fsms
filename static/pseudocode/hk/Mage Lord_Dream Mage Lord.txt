# Mage Lord
# 2 location(s):
#   Dream_02_Mage_Lord / Dream Mage Lord
#   GG_Soul_Tyrant / Dream Mage Lord

fsm Mage Lord {
  start Pause
  on STUN → Stun Init  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Self Z", space=0, everyFrame=false)
    FindChild(gameObject=var "Silhouette", childName="White Flash", storeResult=var "Silh Flash")
    FindChild(gameObject=var "Silhouette", childName="Shot Charge", storeResult=var "Silh Charge")
    SetParent(gameObject=var "CamLock", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Sleep
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToIdle", resetFrame=true, everyFrame=true)
    FloatAdd(floatVariable=var "Idle Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Idle Timer", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on TELE AWAY → Tele Away
    on HIGH SPINNER → Tele Spinner X
    on CHARGE → Tele Charge
    on SHOOT → Tele Shoot
    on FINISHED → Attack Choice
    on TOOK DAMAGE → Damaged Idle
    on QUAKE → Tele Quake
  }

  state Teleport {
    SetVector3XYZ(vector3Variable=var "Teleport Point", vector3Value=(unset), x=var "Tele X", y=var "Tele Y", z=var "Self Z", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Tele Out Anim", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_appear (sharedassets34.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    GetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Teleport Point", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Tele Out Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Appear Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Tele Out Anim", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Tele Out Anim", vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Tele Line
  }

  state Tele Line {
    Vector3Lerp(fromVector=var "Teleport Point", toVector=var "Self Pos", amount=0.5, storeResult=var "Line Pos", everyFrame=false)
    GetAngleBetweenPoints(point1=var "Teleport Point", point2=var "Self Pos", storeAngle=var "Tele Angle", everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Final Tele Distance", point1=var "Self Pos", point2=var "Teleport Point", ignoreZ=false, everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Line Pos", addX=0, addY=0, addZ=0.1, everyFrame=false, perSecond=false)
    CreateObject(gameObject=Tele Line (sharedassets34.assets), spawnPoint=var "", position=var "Line Pos", rotation=(unset), storeObject=var "Tele Line", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Tele Line", vector=(unset), x=var "Final Tele Distance", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Tele Line", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tele Angle", space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Tele Line", emit=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele In", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → After Tele
  }

  state Tele Away {
    SetStringValue(stringVariable=var "Next Event", stringValue="IDLE", everyFrame=false)
    RandomFloat(min=var "Left X", max=var "Right X", storeResult=var "Tele X")
    RandomFloatEither(value1=var "Top Y", value2=var "Bot Y", storeResult=var "Tele Y")
    on FINISHED → Away Valid?
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Away Valid? {
    SetVector3XYZ(vector3Variable=var "Teleport Point", vector3Value=(unset), x=var "Tele X", y=var "Tele Y", z=var "Self Z", everyFrame=false)
    GetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Self Pos", point2=var "Teleport Point", ignoreZ=true, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=6, tolerance=0, equal=(none), lessThan=→"REDO", greaterThan=(none), everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Hero Pos", point2=var "Teleport Point", ignoreZ=true, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=9.5, tolerance=0, equal=(none), lessThan=→"REDO", greaterThan=(none), everyFrame=false)
    on REDO → Tele Away
    on FINISHED → Teleport
  }

  state Tele Top Y {
    SetFloatValue(floatVariable=var "Tele Y", floatValue=var "Top Y", everyFrame=false)
    on FINISHED → Teleport
  }

  state Tele Spinner X {
    SetStringValue(stringVariable=var "Next Event", stringValue="HIGH SPINNER", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFsmFloat(gameObject=var "Orb Spinner", fsmName="Spin Control", variableName="Spin Speed", setValue=75, everyFrame=false)
    SetFloatValue(floatVariable=var "Tele X", floatValue=var "Left X", everyFrame=false)
    SetBoolValue(boolVariable=var "High Spinner Right", boolValue=true, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "Hero Mid X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Tele X", floatValue=var "Right X", everyFrame=false)
    SetBoolValue(boolVariable=var "High Spinner Right", boolValue=false, everyFrame=false)
    SetFsmFloat(gameObject=var "Orb Spinner", fsmName="Spin Control", variableName="Spin Speed", setValue=-75, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Tele Top Y
  }

  state After Tele {
    SendEventByName(eventTarget=Self, sendEvent=var "Next Event", delay=0, everyFrame=false)
    on IDLE → Set Idle Timer
    on HIGH SPINNER → HS Summon
    on CHARGE → Charge Antic
    on SHOOT → Set Shots
  }

  state HS Summon {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Summon")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_lord_projectile_charge (sharedassets102.assets))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Soul_Master_Cast_03 (sharedassets102.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → HS Orb
  }

  state HS Orb {
    SendEventByName(eventTarget=GameObject(var "Orb Spinner"), sendEvent="SPINNER SUMMON", delay=0, everyFrame=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → HS Dir
  }

  state HS Dir {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="SummonToIdle")
    BoolTest(boolVariable=var "High Spinner Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → HS Right
    on LEFT → HS Left
  }

  state HS Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "High Spinner Speed", y=(unset), everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Right X", tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → HS Stop
  }

  state HS Left {
    SetFloatValue(floatVariable=var "Speed Crt", floatValue=var "High Spinner Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed Crt", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed Crt", y=(unset), everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Left X", tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → HS Stop
  }

  state HS Stop {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="TurnToIdle")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → HS Tele Out
  }

  state HS Tele Out {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Tele Out Anim", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Tele Out Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Appear Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Tele Out Anim", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(true))
    BoolTest(boolVariable=var "High Spinner Right", isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_appear (sharedassets34.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    on LEFT → HS Ret Left
    on RIGHT → Hs Ret Right
  }

  state HS Ret Left {
    AccelerateVelocity(gameObject=Self, xAccel=-0.6, yAccel=(unset), xMaxSpeed=30, yMaxSpeed=(unset))
    Wait(time=1.65, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_lord_projectile_shoot (sharedassets102.assets))
    on FINISHED → HS Dissipate
  }

  state Hs Ret Right {
    AccelerateVelocity(gameObject=Self, xAccel=0.6, yAccel=(unset), xMaxSpeed=30, yMaxSpeed=(unset))
    Wait(time=1.65, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_lord_projectile_shoot (sharedassets102.assets))
    on FINISHED → HS Dissipate
  }

  state Reactivate {
    SetMeshRenderer(gameObject=Self, active=true)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(false))
    SetStringValue(stringVariable=var "Next Event", stringValue="IDLE", everyFrame=false)
    ActivateGameObject(gameObject=var "Quake Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Head Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Tele Away
  }

  state HS Dissipate {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Orb Spinner"), sendEvent="DISSIPATE", delay=0, everyFrame=false)
    on FINISHED → Reactivate
  }

  state Tele Charge {
    SetStringValue(stringVariable=var "Next Event", stringValue="CHARGE", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Tele X", floatValue=var "Left X", everyFrame=false)
    SetBoolValue(boolVariable=var "High Spinner Right", boolValue=true, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "Hero Mid X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Tele X", floatValue=var "Right X", everyFrame=false)
    SetBoolValue(boolVariable=var "High Spinner Right", boolValue=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Tele Bot Y
  }

  state Charge Dir {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dash")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_lord_dash (sharedassets102.assets))
    BoolTest(boolVariable=var "High Spinner Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on LEFT → Charge Left
    on RIGHT → Charge Right
  }

  state Charge Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-19, y=(unset), everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Left X", tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Charge Stop
  }

  state Charge Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=19, y=(unset), everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Right X", tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Charge Stop
  }

  state Charge Stop {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SetStringValue(stringVariable=var "Next Event", stringValue="IDLE", everyFrame=false)
    on FINISHED → Tele Away
  }

  state Tele Bot Y {
    SetFloatValue(floatVariable=var "Tele Y", floatValue=var "Bot Y", everyFrame=false)
    on FINISHED → Teleport
  }

  state Set Shots {
    RandomInt(min=1, max=2, storeResult=var "Shots", inclusiveMax=true)
    on FINISHED → Shot Antic
  }

  state Shot Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Summon")
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Shot Charge", emit=0)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_lord_projectile_charge (sharedassets102.assets))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Soul_Master_Cast_01 (sharedassets102.assets), Soul_Master_Cast_02 (sharedassets102.assets), Soul_Master_Cast_03 (sharedassets102.assets), Soul_Master_Cast_04 (sharedassets102.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Shot
  }

  state Shot {
    StopParticleEmitter(gameObject=var "Shot Charge")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mage_lord_projectile_shoot (sharedassets102.assets))
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Mage Orb (sharedassets34.assets), spawnPoint=var "Fire Effect", position=(unset), rotation=(unset), storeObject=var "Projectile")
    SetIsKinematic2d(gameObject=var "Projectile", isKinematic=false)
    SendEventByName(eventTarget=GameObject(var "Projectile"), sendEvent="FIRE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Fire Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=-0.5, storeAngle=var "Angle", everyFrame=false)
    SetVelocityAsAngle(gameObject=var "Projectile", angle=var "Angle", speed=var "Shot Speed", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="SummonToIdle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Next?
  }

  state Tele Shoot {
    SetStringValue(stringVariable=var "Next Event", stringValue="SHOOT", everyFrame=false)
    RandomFloat(min=var "Left X", max=var "Right X", storeResult=var "Tele X")
    RandomFloatEither(value1=var "Top Y", value2=var "Bot Y", storeResult=var "Tele Y")
    on FINISHED → Shoot Valid?
  }

  state Attack Choice {
    SendRandomEventV2(events=[→"CHARGE", →"HIGH SPINNER", →"SHOOT", →"QUAKE"], weights=[0.15, 0.15, 0.35, 0.35], trackingInts=[var "Ct Charge", var "Ct HS", var "Ct Shoot", var "Ct Quake"], eventMax=[1, 1, 2, 2])  // disabled
    SendRandomEventV3(events=[→"CHARGE", →"HIGH SPINNER", →"SHOOT", →"QUAKE"], weights=[0.15, 0.15, 0.35, 0.35], trackingInts=[var "Ct Charge", var "Ct HS", var "Ct Shoot", var "Ct Quake"], eventMax=[1, 1, 2, 2], trackingIntsMissed=[var "Ms Charge", var "Ms HS", var "Ms Shoot", var "Ms Quake"], missedMax=[6, 6, 4, 4])
    on HIGH SPINNER → Tele Spinner X
    on CHARGE → Tele Charge
    on SHOOT → Tele Shoot
    on QUAKE → Tele Quake
  }

  state Next? {
    IntAdd(intVariable=var "Shots", add=-1, everyFrame=false)
    IntCompare(integer1=var "Shots", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"SHOOT", everyFrame=false)
    on SHOOT → Tele Shoot
    on FINISHED → Set Idle Timer
  }

  state Damaged Idle {
    SendRandomEvent(events=[→"FINISHED", →"TELE AWAY", →"ATTACK"], weights=[0, 0.5, 0.5], delay=0)
    on FINISHED → Idle
    on TELE AWAY → Damage Tele Away
    on ATTACK → Attack Choice
  }

  state Set Idle Timer {
    RandomFloat(min=0.5, max=0.5, storeResult=var "Idle Timer")
    on FINISHED → Idle
  }

  state Shoot Valid? {
    SetVector3XYZ(vector3Variable=var "Teleport Point", vector3Value=(unset), x=var "Tele X", y=var "Tele Y", z=var "Self Z", everyFrame=false)
    GetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Self Pos", point2=var "Teleport Point", ignoreZ=true, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=5, tolerance=0, equal=(none), lessThan=→"REDO", greaterThan=(none), everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Hero Pos", point2=var "Teleport Point", ignoreZ=true, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=8.5, tolerance=0, equal=(none), lessThan=→"REDO", greaterThan=(none), everyFrame=false)
    on REDO → Tele Shoot
    on FINISHED → Teleport
  }

  state Tele Quake {
    RandomFloat(min=0, max=100, storeResult=var "Random")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Tele X", y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero Y", float2=var "Knight Quake Y Max", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    SetFloatValue(floatVariable=var "Tele Y", floatValue=var "Quake Y", everyFrame=false)
    FloatClamp(floatVariable=var "Tele X", minValue=var "Left X", maxValue=var "Right X", everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_appear (sharedassets34.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    on CANCEL → Attack Choice
    on FINISHED → TeleportQ
  }

  state TeleportQ {
    SetVector3XYZ(vector3Variable=var "Teleport Point", vector3Value=(unset), x=var "Tele X", y=var "Tele Y", z=var "Self Z", everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Tele Out Anim", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Teleport Point", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Tele Out Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Appear Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Tele Out Anim", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Tele Out Anim", vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Tele Line Q
  }

  state Tele Line Q {
    Vector3Lerp(fromVector=var "Teleport Point", toVector=var "Self Pos", amount=0.5, storeResult=var "Line Pos", everyFrame=false)
    GetAngleBetweenPoints(point1=var "Teleport Point", point2=var "Self Pos", storeAngle=var "Tele Angle", everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Final Tele Distance", point1=var "Self Pos", point2=var "Teleport Point", ignoreZ=false, everyFrame=false)
