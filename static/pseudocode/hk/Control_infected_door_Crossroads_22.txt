# Control
# 1 location(s):
#   Crossroads_22 / infected_door

fsm Control {
  start Pause
  on BLOB OPEN Q → Destroy  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Idle {
    on BLOB OPEN → Open Rumble
  }

  state Open Rumble {
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=boss_gushing (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Audio Player")
    on FINISHED → Burst
  }

  state Init {
    FindChild(gameObject=Self, childName="Blobs", storeResult=var "Blobs")
    FindChild(gameObject=Self, childName="Globs", storeResult=var "Globs")
    FindChild(gameObject=Self, childName="Collider", storeResult=var "Collider")
    FindChild(gameObject=Self, childName="Rumble Spurt", storeResult=var "Rumble Spurt")
    FindChild(gameObject=Self, childName="Splat", storeResult=var "Splat")
    FindChild(gameObject=Self, childName="Pt Antic", storeResult=var "Pt Antic")
    FindChild(gameObject=Self, childName="Pt Burst", storeResult=var "Pt Burst")
    on FINISHED → Idle
  }

  state Burst {
    AudioStop(gameObject=var "Audio Player")
    PlayParticleEmitter(gameObject=var "Pt Burst", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=explosion_4_wet_original (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Blobs", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Splat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
  }

  var Audio Player: gameObject = <null>
  var Blobs: gameObject = <null>
  var Collider: gameObject = <null>
  var Globs: gameObject = <null>
  var Pt Antic: gameObject = <null>
  var Pt Burst: gameObject = <null>
  var Rumble Spurt: gameObject = <null>
  var Splat: gameObject = <null>
}
