# Weaverling Control
# 1 location(s):
#   resources.assets / Knight:1/Charm Effects

fsm Weaverling Control {
  start Init
  on LEVEL LOADED → Spawn Pause  // from any state
  on CHARM EQUIP CHECK → Check  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_39", isTrue=→"EQUIPPED", isFalse=→"UNEQUIPPED")
    on EQUIPPED → Wait Frame
    on UNEQUIPPED → Unequipped
  }

  state Unequipped {
  }

  state Spawn {
    SpawnObjectFromGlobalPool(gameObject=Weaverling, spawnPoint=var "Hero", position=(0, -0.5, 0), rotation=(unset), storeObject=var "Weaverling")
    SetAudioClip(gameObject=var "Weaverling", audioClip=spider_buddy_loop_1)
    SpawnObjectFromGlobalPool(gameObject=Weaverling, spawnPoint=var "Hero", position=(0, -0.5, 0), rotation=(unset), storeObject=var "Weaverling")
    SetAudioClip(gameObject=var "Weaverling", audioClip=spider_buddy_loop_2)
    SpawnObjectFromGlobalPool(gameObject=Weaverling, spawnPoint=var "Hero", position=(0, -0.5, 0), rotation=(unset), storeObject=var "Weaverling")
    SetAudioClip(gameObject=var "Weaverling", audioClip=spider_buddy_loop_3)
  }

  state Spawn Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Charms Allowed?
  }

  state Charms Allowed? {
    GetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", storeValue=var "No Charms", everyFrame=false)
    BoolTest(boolVariable=var "No Charms", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="IsGameplayScene", parameters=[], storeResult=var "Is Gameplay Scene")
    BoolTest(boolVariable=var "Is Gameplay Scene", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    GetTagCount(tag="Weaverling", storeResult=var "Weaverling Count")
    IntCompare(integer1=var "Weaverling Count", integer2=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Check
    on CANCEL → Unequipped
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Spawn
  }

  var Weaverling Count: int = 0
  var Is Gameplay Scene: bool = false
  var No Charms: bool = false
  var Weaverling: gameObject = <null>
}
