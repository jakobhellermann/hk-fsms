// uses template: activate_if_pd_bool
fsm Activate Infected {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetOwner(storeGameObject=var "Self")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PD Bool Name", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → True
    on FALSE → False
  }

  state True {
    BoolTest(boolVariable=var "Active If Bool True", isTrue=→"ACTIVE", isFalse=→"INACTIVE", everyFrame=false)
    on ACTIVE → Active
    on INACTIVE → Inactive
  }

  state False {
    BoolTest(boolVariable=var "Active If Bool True", isTrue=→"INACTIVE", isFalse=→"ACTIVE", everyFrame=false)
    on ACTIVE → Active
    on INACTIVE → Inactive
  }

  state Active {
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  state Inactive {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Active If Bool True: bool = false
  var PD Bool Name: string = "crossroadsInfected"

  var Self: gameObject = <null>
}
