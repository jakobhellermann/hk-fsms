# FSM
# 1 location(s):
#   Deepnest_East_03 / Col Drop Corpse 01

// uses template: corpse_dropper
fsm FSM {
  start Pause

  state Detect {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    PreBuildTK2DSprites(gameObject=var "Self", useChildren=true)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Activate
    on ACTIVATE → Activated
  }

  state Activate {
    ActivateGameObject(gameObject=var "Colosseum Drop Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Colosseum Drop Corpse", storeResult=var "Colosseum Drop Corpse")
    FindChild(gameObject=Self, childName="Dropped Corpse", storeResult=var "Dropped Corpse")
    on FINISHED → Detect
  }

  state Activated {
    ActivateGameObject(gameObject=var "Dropped Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
  }

  var Activated: bool = false
  var Colosseum Drop Corpse: gameObject = <null>
  var Dropped Corpse: gameObject = <null>
  var Self: gameObject = <null>
}
