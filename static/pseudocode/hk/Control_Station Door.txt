# Control
# 1 location(s):
#   Room_Town_Stag_Station / Station Door

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Open", storeResult=var "Open")
    FindChild(gameObject=Self, childName="Opened", storeResult=var "Opened")
    FindChild(gameObject=Self, childName="Roof Collider", storeResult=var "Roof Collider")
    FindChild(gameObject=Self, childName="Glow", storeResult=var "Glow")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="openedTownBuilding", isTrue=→"OPENED", isFalse=(none))
    on FINISHED → Idle
    on OPENED → Opened
  }

  state Idle {
    on ACTIVATE → Activate Pause
  }

  state Opened {
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Roof Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Glow"), sendEvent="UP INSTANT", delay=0, everyFrame=false)
  }

  state Activate Pause {
    SetPlayerDataBool(boolName="openedTownBuilding", value=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  state Activate {
    ActivateGameObject(gameObject=var "Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Roof Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Glow"), sendEvent="UP", delay=0, everyFrame=false)
  }

  var Glow: gameObject = <null>
  var Open: gameObject = <null>
  var Opened: gameObject = <null>
  var Roof Collider: gameObject = <null>
}
