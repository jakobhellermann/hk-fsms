# Control
# 1 location(s):
#   Crossroads_10 / Fk Break Wall

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check State
  }

  state Check State {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="falseKnightWallRepaired", isTrue=→"REPAIRED", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="shamanPillar", isTrue=→"REPAIRED", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasSpell", isTrue=→"REPAIRED", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="falseKnightWallBroken", isTrue=→"BROKEN", isFalse=(none))
    on REPAIRED → Repaired
    on FINISHED → Initial
    on BROKEN → Broken
  }

  state Repaired {
    ActivateGameObject(gameObject=var "Repaired", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Initial", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Initial {
    on FK RAGE COMPLETE → Breakable
  }

  state Breakable {
    ActivateGameObject(gameObject=var "Breakable", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Initial", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Broken {
    ActivateGameObject(gameObject=var "Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Initial", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK CLOSE", delay=0, everyFrame=false)  // disabled
    FindGameObject(objectName="Zombies", withTag="Untagged", store=var "Zombies")
    ActivateGameObject(gameObject=var "Zombies", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Battle Gate 1", storeResult=var "Battle Gate")
    ActivateGameObject(gameObject=var "Battle Gate", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Battle Gate: gameObject = <null>
  var Breakable: gameObject = Fk Break Wall/Breakable
  var Broken: gameObject = Fk Break Wall/Broken
  var Initial: gameObject = Fk Break Wall/Initial
  var Repaired: gameObject = Fk Break Wall/Repaired
  var Zombies: gameObject = <null>
}
