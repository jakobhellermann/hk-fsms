# Vignette Control
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health/Low Health Vignette
#   resources.assets / _GameCameras/HudCamera/Anchor TL/Hud Canvas Offset/Hud Canvas/Health/Low Health Vignette

fsm Vignette Control {
  start Init

  state Init {
    SetScale(gameObject=Self, vector=(unset), x=5.9, y=5.9, z=5.9, everyFrame=false, lateUpdate=false)
    on FINISHED → Down
    on UP → Up
    on DOWN → Down
  }

  state Down {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(6, 6, 6), time=1, delay=0, speed=(unset), easeType=easeInSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    FadeGroupDown(target=Self, fast=false)
    on UP → Up
  }

  state Up {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(4, 4, 4), time=1, delay=0, speed=(unset), easeType=spring, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    FadeGroupUp(target=Self)
    on DOWN → Down
    on FINISHED → Pulsing
  }

  state Pulsing {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(5.5, 5.5, 5.5), time=1, delay=0, speed=(unset), easeType=easeInOutSine, loopType=pingPong, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=7, finishEvent=→"DOWN", realTime=false)
    on DOWN → Down
  }

  var New Colour: color = (0, 0, 0, 1)
  var Prev Colour: color = (0, 0, 0, 1)
}
