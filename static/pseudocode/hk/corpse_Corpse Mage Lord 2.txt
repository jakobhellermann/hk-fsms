    FloatAdd(floatVariable=var "emissionSpeed", add=1, everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "emissionSpeed", minValue=0, maxValue=110, everyFrame=true)
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)

  }

    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")

    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"LEFT", negativeEvent=→"RIGHT", space=0)

    on RIGHT → R
  }

  state Ready {

    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    EaseFloat(fromValue=var "emissionRate", toValue=0, floatVariable=var "emissionRate", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)

    on FINISHED → Sting
  }

  state Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")  // disabled
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Steam
  }

  state L {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-5, y=10, everyFrame=false)
    on FINISHED → Chinese?
  }

  state R {
    SetVelocity2d(gameObject=Self, vector=(unset), x=5, y=10, everyFrame=false)
    on FINISHED → Chinese?
  }

    PlayParticleEmitter(gameObject=var "Corpse Steam", emit=0)
    on LAND → Land
  }

    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Soul Breath
  }

    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=→"FINISHED")
    on FINISHED → Quake Pickup

  }

  state Quake Pickup {
    PlayParticleEmitter(gameObject=var "Soul Breath", emit=0)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["SOUL_MASTER_DEFEAT"], storeResult=(unset var))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="QUAKE PICKUP START", delay=0, everyFrame=false)
  }

  state End GG Scene {
    EndGGBossScene()
  }

  state Music {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=2)
    on FINISHED → Init
  }

  state Sting {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Self", audioClip=Boss Defeat (sharedassets40.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)

    on LAND → Land Chinese
  }

  state Land Chinese {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Soul Breath
  }

  state Chinese? {
    CheckIsChineseBuild(trueEvent=→"TRUE", falseEvent=→"FALSE")
    on FALSE → In Air
    on TRUE → In Air Chinese
  }
