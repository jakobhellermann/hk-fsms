# corpse
# 1 location(s):
#   sharedassets50.assets / Corpse Blocker

fsm corpse {
  start Shaman Update

  state Flame Check {
    BoolTest(boolVariable=var "spellBurn", isTrue=→"TRUE", isFalse=(none), everyFrame=false)
    on TRUE → Start Flame
  }

  state Start Flame {
    PlayParticleEmitter(gameObject=var "Flame", emit=0)
  }

  state Shaman Update {
    GetOwner(storeGameObject=var "Self")
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman State")
    IntCompare(integer1=var "Shaman State", integer2=3, equal=→"UPDATE 1", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Shaman State", integer2=2, equal=→"UPDATE 2", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Flame Check 2
    on UPDATE 1 → Update 1
    on UPDATE 2 → Update 2
  }

  state Update 1 {
    SetPlayerDataInt(intName="shaman", value=4)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="UPDATE SHAMAN", delay=0, everyFrame=false)
    on FINISHED → Flame Check 2
  }

  state Update 2 {
    SetPlayerDataInt(intName="shaman", value=5)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="UPDATE SHAMAN", delay=0, everyFrame=false)
    on FINISHED → Flame Check 2
  }

  state Death Stun {
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stun End
  }

  state Stun End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death")  // disabled
    FindChild(gameObject=Self, childName="Stun Steam", storeResult=var "Stun Steam")
    StopParticleEmitter(gameObject=var "Stun Steam")
    CreateObject(gameObject=Death Wave Infected (sharedassets32.assets), spawnPoint=var "Self", position=(0, 0, -0.001), rotation=(unset), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=2, y=2, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Chinese?
  }

  state Flame Check 2 {
    BoolTest(boolVariable=var "spellBurn", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Blacken
    on FALSE → Death Stun
  }

  state Blacken {
    Tk2dSpriteSetColor(gameObject=Self, color=(0.19607843, 0.19607843, 0.19607843, 1), everyframe=false)
    on FINISHED → Death Stun
  }

  state Chinese? {
    CheckIsChineseBuild(trueEvent=(none), falseEvent=→"FINISHED")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Flame Check
  }

  var Shaman State: int = 0
  var spellBurn: bool = false
  var Velocity: vector2 = (0, 0)
  var Flame: gameObject = Corpse Blocker/Corpse Flame
  var Hit Effect: gameObject = <null>
  var Self: gameObject = <null>
  var Stun Steam: gameObject = <null>
}
