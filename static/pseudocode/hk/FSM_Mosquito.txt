// uses template: flyer_receive_direction_msg
fsm FSM {
  start Idle

  state Idle {
    on GO UP → Go Up
    on GO LEFT → Go Left
    on GO RIGHT → Go Right
    on GO DOWN → Go Down
  }

  state Go Right {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Velocity", minValue=0, maxValue=9999, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=false)
    AddForce2dV2(gameObject=Self, atPosition=(unset), vector=(unset), x=40, y=(unset), vector3=(unset), maxSpeed=(unset), maxSpeedX=10, maxSpeedY=(unset), everyFrame=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset Rotate?
  }

  state Reset Rotate? {
    BoolTest(boolVariable=var "Reset Rotation", isTrue=(none), isFalse=→"RETURN", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    on RETURN → Notify Self
    on FINISHED → Notify Self
  }

  state Go Left {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Velocity", minValue=-9999, maxValue=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=false)
    AddForce2dV2(gameObject=Self, atPosition=(unset), vector=(unset), x=-40, y=(unset), vector3=(unset), maxSpeed=(unset), maxSpeedX=10, maxSpeedY=(unset), everyFrame=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset Rotate?
  }

  state Go Down {
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Velocity", minValue=-9999, maxValue=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", everyFrame=false)
    AddForce2dV2(gameObject=Self, atPosition=(unset), vector=(unset), x=(unset), y=-40, vector3=(unset), maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=10, everyFrame=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset Rotate?
  }

  state Go Up {
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Velocity", minValue=0, maxValue=9999, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity", everyFrame=false)
    AddForce2dV2(gameObject=Self, atPosition=(unset), vector=(unset), x=(unset), y=40, vector3=(unset), maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=10, everyFrame=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset Rotate?
  }

  state Notify Self {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DIR MSG", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Reset Rotation: bool = true

  var Velocity: float = 0
}
