# Check for Blockers
# 1 location(s):
#   Fungus1_28 / CameraLockArea

fsm Check for Blockers {
  start Check

  state Check {
    FindGameObject(objectName="Blocker 1", withTag="Untagged", store=var "Blocker 1")
    FindGameObject(objectName="Blocker 2", withTag="Untagged", store=var "Blocker 2")
    GameObjectIsNull(gameObject=var "Blocker 1", isNull=(none), isNotNull=(none), storeResult=var "Blocker 1 Null", everyFrame=true)
    GameObjectIsNull(gameObject=var "Blocker 2", isNull=(none), isNotNull=(none), storeResult=var "Blocker 2 Null", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Blocker 1 Null", var "Blocker 2 Null"], sendEvent=→"DESTROY", storeResult=(unset), everyFrame=true)
    on DESTROY → Pause
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  var Blocker 1 Null: bool = false
  var Blocker 2 Null: bool = false
  var Blocker 1: gameObject = <null>
  var Blocker 2: gameObject = <null>
}
