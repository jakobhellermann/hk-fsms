# Conversation Control
# 1 location(s):
#   Room_nailmaster_02 / NM Parent/Modelling/NM Sheo NPC Modeller

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Stop Anim  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Convo Choice {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Nailmaster_Look_to_talk_03 (sharedassets13.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metNailmasterSheo", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="nailsmithSheo", falseBool="sheoConvoNailsmith", isTrue=→"NAILSMITH", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hasDashSlash", isTrue=(none), isFalse=→"REOFFER")
    BoolTest(boolVariable=var "Bowing", isTrue=→"BOWING", isFalse=(none), everyFrame=false)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasUpwardSlash", falseBool="sheoConvoOro", isTrue=→"ORO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasCyclone", falseBool="sheoConvoMato", isTrue=→"MATO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gotCharm_26", falseBool="sheoConvoSly", isTrue=→"SLY", isFalse=(none))
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"GREET", everyFrame=false)
    on MEET → 
    on BOWING → Bowing
    on GREET → Nailsmith
    on REPEAT → Repeat
    on ORO → Oro
    on MATO → Mato
    on SLY → Sly
    on REOFFER → 
    on NAILSMITH → Nailsmith
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Cinematic", store=var "Cinematic Player")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="LookUpEnd", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    BoolTest(boolVariable=var "Bowing", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NM Model Talk")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="NM_SHEO", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Greet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHEO_GREET", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    on CONVO_FINISH → Talk End
  }

  state Stop Anim {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Bowing {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHEO_BOW", "Nailmasters"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Talk End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NM Model Talk End")
    on FINISHED → Talk Finish
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHEO_REPEAT_MODEL", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    on CONVO_FINISH → Talk End
  }

  state Oro {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHEO_ORO", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="sheoConvoOro", value=true)
    on CONVO_FINISH → Talk End
  }

  state Mato {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHEO_MATO", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="sheoConvoMato", value=true)
    on CONVO_FINISH → Talk End
  }

  state Sly {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHEO_SLY", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="sheoConvoSly", value=true)
    on CONVO_FINISH → Talk End
  }

  state Nailsmith {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["SHEO_NAILSMITH", "Nailmasters"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetPlayerDataBool(boolName="sheoConvoNailsmith", value=true)
    on CONVO_FINISH → Talk End
  }

  var Bowing: bool = false
  var Equipped Sly Charm: bool = false
  var Got Dash Slash: bool = false
  var Got Great Slash: bool = false
  var Oro Convo: bool = false
  var Sheo Convo: bool = false
  var Sly Convo: bool = false
  var Spoken: bool = false
  var Visited Crossroads: bool = false
  var Area Title: gameObject = <null>
  var Arrow Prompt: gameObject = <null>
  var Cinematic Player: gameObject = <null>
  var Roar: gameObject = <null>
  var Self: gameObject = <null>
  var UI Msg: gameObject = <null>
  var Voice Player: gameObject = <null>
}
