# Rope Plat
# 1 location(s):
#   Ruins1_28 / Chain Platform

fsm Rope Plat {
  start Pause
  on ACIDLAND → Acid Land  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Vine", storeResult=var "Vine")
    FindChild(gameObject=Self, childName="Enemy Kill", storeResult=var "Enemy Kill")
    RayCast2dV2(fromGameObject=Self, fromPosition=(0, 1), direction=(0, 1), space=1, distance=100, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=var "Ray Distance", repeatInterval=1, layerMask=[], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)  // disabled
    FloatDivide(floatVariable=var "Ray Distance", divideBy=4.5, everyFrame=false)  // disabled
    SetScale(gameObject=var "Vine", vector=(unset), x=(unset), y=var "Ray Distance", z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  state Idle {
    on CUT → PlayerData?
  }

  state Fall {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=chain_cut (sharedassets30.assets))
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    DestroyObject(gameObject=var "Hero Land", delay=0, detachChildren=false)
    SetParticleEmissionRate(gameObject=var "Land Grass", emissionRate=0, everyFrame=false)
    AccelerateVelocity(gameObject=Self, xAccel=(unset), yAccel=-2, xMaxSpeed=(unset), yMaxSpeed=60)
    ActivateGameObject(gameObject=var "Enemy Kill", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on HARDLAND → Hard Land
    on ACIDLAND → Acid Land
  }

  state Hard Land {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=chain_plat_land (sharedassets46.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Enemy Kill", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Grass A", emit=0)
    PlayParticleEmitter(gameObject=var "Grass B", emit=0)
    on FINISHED → Kinemetise
  }

  state Kinemetise {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
  }

  state Acid Land {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Enemy Kill", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CreateObject(gameObject=Acid Splash (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=-8, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9)
    Wait(time=0.2, finishEvent=→"WAIT", realTime=false)
    on WAIT → Kinemetise
  }

  state PlayerData? {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    StringCompare(stringVariable=var "playerData bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "playerData bool", value=true)
    on FINISHED → Fall
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Activated Plat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Hero Land", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Plat Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vine", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vine Stub", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var playerData bool: string = ""

  var Ray Distance: float = 0
  var Activated: bool = false
  var Activated Plat: gameObject = Chain Platform/Activated Plat
  var Enemy Kill: gameObject = <null>
  var Grass A: gameObject = Chain Platform/Green Grass A
  var Grass B: gameObject = Chain Platform/Green Grass B
  var Hero Land: gameObject = Chain Platform/Hero Land
  var Land Grass: gameObject = Chain Platform/Land Grass
  var Plat Sprite: gameObject = Chain Platform/Plat Sprite
  var Self: gameObject = <null>
  var Slam Effect: gameObject = Chain Platform/Slam Effect
  var Vine: gameObject = Chain Platform/Vine
  var Vine Stub: gameObject = Chain Platform/Vine Stub
}
