# Spore Bomb
# 1 location(s):
#   sharedassets167.assets / Spore Bomb

fsm Spore Bomb {
  start Init

  state Wait {
    WaitRandom(timeMin=0.3, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"COLLIDE", storeCollider=var "")
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    on FINISHED → Chase
    on COLLIDE → Explode
  }

  state Chase {
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=var "Speed Max", accelerationForce=var "Accel Force", offsetX=0, offsetY=0)
    Wait(time=2.9, finishEvent=→"READY", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"COLLIDE", storeCollider=var "")
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer=(unset), sendEvent=→"COLLIDE", storeCollider=var "")
    on READY → Ready
    on COLLIDE → Explode
    on ORBIT SHIELD → Explode
  }

  state Ready {
    DecelerateV2(gameObject=Self, deceleration=0.95)
    SetParticleEmission(gameObject=var "Spore Fizzle", emission=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Explode")
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Explode
  }

  state Explode {
    AudioStop(gameObject=Self)
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Recycle M (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Explosion")
    SetParticleEmission(gameObject=var "Spore Fizzle", emission=false)
    SetParticleEmission(gameObject=var "Spore Cloud", emission=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Init {
    SetParticleEmission(gameObject=var "Spore Cloud", emission=true)
    SetParticleEmission(gameObject=var "Spore Fizzle", emission=false)
    GetOwner(storeGameObject=var "Self")
    SetFloatValue(floatVariable=var "Accel Force", floatValue=var "Accel Force M", everyFrame=false)
    SetFloatValue(floatVariable=var "Speed Max", floatValue=var "Speed Max M", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle 2")
    SetCircleCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Wait
  }

  state Recycle {
    RecycleSelf()
  }

  var Accel Force M: float = 8
  var Speed Max M: float = 5

  var Accel Force: float = 0
  var Speed Max: float = 0
  var Explosion: gameObject = <null>
  var Self: gameObject = <null>
  var Spore Cloud: gameObject = Spore Bomb/Spore Cloud
  var Spore Fizzle: gameObject = Spore Bomb/Spore Fizzle
}
