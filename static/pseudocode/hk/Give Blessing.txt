# Give Blessing
# 1 location(s):
#   Room_Charm_Shop / Charm Slug

fsm Give Blessing {
  start Inactive

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dialogue 1
  }

  state Dialogue 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_BLESSING_1", "Charm Slug"], storeResult=(unset var))
    on CONVO_FINISH → Box Down
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade
  }

  state Fade {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Scene Blanker
  }

  state Cinematic {
    FindGameObject(objectName="", withTag="Cinematic", store=var "Cinematic Player")
    SendMessage(gameObject=var "Cinematic Player", delivery=SendMessage, options=1, functionCall=TriggerStartVideo())
    AudioPlayerOneShotSingle(audioPlayer=Audio Player NoDestroy (resources.assets), spawnPoint=var "Hero", audioClip=Salubra_Kiss (sharedassets10.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CINEMATIC END → Cinematic End
  }

  state Cinematic End {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get Msg
  }

  state Fade Back {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER OFF", delay=0, everyFrame=false)
    on FINISHED → Box Up 2
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dialogue 2
  }

  state Dialogue 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["CHARMSLUG_BLESSING_2", "Charm Slug"], storeResult=(unset var))
    SetPlayerDataBool(boolName="metCharmSlug", value=true)
    on CONVO_FINISH → Box Down 2
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shop End")
    on FINISHED → End
  }

  state End {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["BLESSED"], storeResult=(unset var))
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOP REGION ACTIVE", delay=0, everyFrame=false)
  }

  state Pause {
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up
  }

  state Inactive {
    on SALUBRA BLESSING → Pause
  }

  state Get Msg {
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=blessing_prompt (sharedassets10.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Blessing", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="NONE", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal Flange (resources.assets), transitionTime=0)
    on GET ITEM MSG END → Fade Back
  }

  state Scene Blanker {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCENE BLANKER ON", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Cinematic
  }

  var Arrow Prompt: gameObject = <null>
  var Cinematic Player: gameObject = <null>
  var UI Msg: gameObject = <null>
}
