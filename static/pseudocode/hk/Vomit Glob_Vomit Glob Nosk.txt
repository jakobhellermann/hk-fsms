# Vomit Glob
# 1 location(s):
#   sharedassets290.assets / Vomit Glob Nosk

fsm Vomit Glob {
  start Init
  on LOW GRAV → Low Grav  // from any state

  state Init {
    RandomFloat(min=1, max=1.75, storeResult=var "Scale")
    RandomFloat(min=0.0041, max=0.00499, storeResult=var "PosZ")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "PosZ", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Puddle Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateGameObject(gameObject=var "Puddle Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHeroAmount(target=var "Puddle Box", damageDealt=1)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → In Air
  }

  state In Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
    on ORBIT SHIELD → Insta Disappear
  }

  state Land {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GLOB LAND", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    StopParticleEmitter(gameObject=var "Air Steam")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.25, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayParticleEmitter(gameObject=var "Steam", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Land")
    ActivateGameObject(gameObject=var "Puddle Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    WaitRandom(timeMin=1.75, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shrink
    on ORBIT SHIELD → Insta Disappear
  }

  state Shrink {
    SetDamageHeroAmount(target=var "Puddle Box", damageDealt=0)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.1, 0.1, 0.1), time=0.25, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Stop Steam
  }

  state Stop Steam {
    StopParticleEmitter(gameObject=var "Steam")
    Wait(time=1, finishEvent=→"LAND", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on LAND → Recycle
  }

  state Recycle {
    SetGravity2dScale(gameObject=Self, gravityScale=0.7)
    RecycleSelf()
  }

  state Insta Disappear {
    AudioPlaySimple(gameObject=Self, volume=0.7, oneShotClip=<null>)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    StopParticleEmitter(gameObject=var "Air Steam")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.25, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayParticleEmitter(gameObject=var "Steam", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Land")
    ActivateGameObject(gameObject=var "Puddle Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Shrink
  }

  state Low Grav {
    SetGravity2dScale(gameObject=Self, gravityScale=0.5)
    on FINISHED → Init
  }

  var PosZ: float = 0
  var Scale: float = 0
  var Air Steam: gameObject = Vomit Glob Nosk/Air Steam
  var Puddle Box: gameObject = Vomit Glob Nosk/Puddle Box
  var Steam: gameObject = Vomit Glob Nosk/Steam
}
