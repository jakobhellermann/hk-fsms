# Control
# 3 location(s):
#   Room_Colosseum_Bronze / Colosseum Manager/Walls/Colosseum Wall L
#   Room_Colosseum_Silver / Colosseum Manager/Walls/Colosseum Wall L
#   Room_Colosseum_Gold / Colosseum Manager/Walls/Colosseum Wall L

fsm Control {
  start Init

  state Idle {
    on MOVE → Start Shake
    on RESET → Reset
  }

  state Init {
    FindChild(gameObject=Self, childName="Wall", storeResult=var "Wall")
    FindChild(gameObject=var "Wall", childName="Whole Dust", storeResult=var "Dust")
    SetPosition(gameObject=var "Wall", vector=(unset), x=-5, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Move Out {
    SetVector3XYZ(vector3Variable=var "Tween Vector", vector3Value=(unset), x=var "Distance", y=0, z=0, everyFrame=false)  // disabled
    iTweenMoveTo(gameObject=var "Wall", id="", transformPosition=var "", vectorPosition=var "Tween Vector", time=(unset), delay=0, speed=10, space=1, easeType=easeInSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    SetVelocity2d(gameObject=var "Wall", vector=var "Out Speed", x=(unset), y=(unset), everyFrame=false)
    GetPosition(gameObject=var "Wall", vector=(unset), x=var "Wall Pos", y=(unset), z=(unset), space=1, everyFrame=true)
    FloatCompare(float1=var "Wall Pos", float2=var "Distance", tolerance=0, equal=→"ARRIVED", lessThan=(none), greaterThan=→"ARRIVED", everyFrame=true)
    AudioPlayRandom(gameObject=var "Self", audioClips=[col_moving_wall_move (sharedassets32.assets)], weights=[1], pitchMin=0.75, pitchMax=1.25)
    on ARRIVED → Shake
  }

  state Shake {
    SetVelocity2d(gameObject=var "Wall", vector=(unset), x=0, y=0, everyFrame=false)
    SetPosition(gameObject=var "Wall", vector=(unset), x=var "Distance", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=var "Wall", vector=var "Jitter End Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    ObjectJitter(gameObject=Self, x=0.075, y=0.075, z=(unset), allowMovement=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[col_moving_wall_impact_into_place (sharedassets32.assets)], weights=[1], pitchMin=0.75, pitchMax=1.25)
    on FINISHED → Shake End
  }

  state Shake End {
    SetPosition(gameObject=var "Wall", vector=var "Jitter End Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Check Dir {
    SetPosition(gameObject=var "Wall", vector=var "Jitter End Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    FloatCompare(float1=var "Wall Pos", float2=var "Distance", tolerance=0, equal=→"CANCEL", lessThan=→"OUT", greaterThan=→"IN", everyFrame=false)
    on CANCEL → Idle
    on IN → Move In
    on OUT → Move Out
  }

  state Move In {
    SetVector3XYZ(vector3Variable=var "Tween Vector", vector3Value=(unset), x=var "Distance", y=0, z=0, everyFrame=false)  // disabled
    iTweenMoveTo(gameObject=var "Wall", id="", transformPosition=var "", vectorPosition=var "Tween Vector", time=(unset), delay=0, speed=10, space=1, easeType=easeInSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    SetVelocity2d(gameObject=var "Wall", vector=var "In Speed", x=(unset), y=(unset), everyFrame=false)
    GetPosition(gameObject=var "Wall", vector=(unset), x=var "Wall Pos", y=(unset), z=(unset), space=1, everyFrame=true)
    FloatCompare(float1=var "Wall Pos", float2=var "Distance", tolerance=0, equal=→"ARRIVED", lessThan=→"ARRIVED", greaterThan=(none), everyFrame=true)
    AudioPlayRandom(gameObject=var "Self", audioClips=[col_moving_wall_move (sharedassets32.assets)], weights=[1], pitchMin=0.75, pitchMax=1.25)
    on ARRIVED → Shake
  }

  state Start Shake {
    GetPosition(gameObject=var "Wall", vector=var "Jitter End Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    ObjectJitter(gameObject=Self, x=0.075, y=0.075, z=(unset), allowMovement=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Dir
  }

  state Reset {
    SetFloatValue(floatVariable=var "Distance", floatValue=-5, everyFrame=false)
    on FINISHED → Start Shake
  }

  var Distance: float = 0
  var In Speed: vector2 = (-28, 0)
  var Out Speed: vector2 = (28, 0)

  var Wall Pos: float = 0
  var Jitter End Pos: vector3 = (0, 0, 0)
  var Tween Vector: vector3 = (0, 0, 0)
  var Dust: gameObject = <null>
  var Self: gameObject = <null>
  var Wall: gameObject = <null>
}
