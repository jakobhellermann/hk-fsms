# Pillar Control
# 1 location(s):
#   Crossroads_06 / _Scenery/Raising Pillar

fsm Pillar Control {
  start Pause Frame

  state Idle {
    on ACTIVATE → Shake
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Opened
  }

  state Opened {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pillar Opened")
    SetCollider(gameObject=Self, active=false)
  }

  state Shake {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pillar Shake")
    PlayParticleEmitter(gameObject=var "Shake Dust", emit=0)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    PlayParticleEmitter(gameObject=var "Bot Dust", emit=0)
    PlayParticleEmitter(gameObject=var "Shake Dust 2", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pillar Open")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    on FINISHED → Finish Opening
  }

  state Finish Opening {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pillar Opened")
    PlayParticleEmitter(gameObject=var "Top Dust", emit=0)
    PlayParticleEmitter(gameObject=var "Stream", emit=0)
    PlayParticleEmitter(gameObject=var "Rocks", emit=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=platform_dirt_fall_1 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Opened
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Bot Dust: gameObject = _Scenery/Raising Pillar/bot dust
  var Shake Dust: gameObject = _Scenery/Raising Pillar/shake_dust
  var Shake Dust 2: gameObject = _Scenery/Raising Pillar/shake_dust2
  var Top Dust: gameObject = _Scenery/Raising Pillar/top dust

  var Activated: bool = false
  var Rocks: gameObject = _Scenery/Raising Pillar/Rocks
  var Self: gameObject = <null>
  var Stream: gameObject = _Scenery/Raising Pillar/Stream
}
