// uses template: ui_list_button_listen
fsm ui_list_button_listen {
  start Init
  on UI ACTIVE → Detect  // from any state
  on UI INACTIVE → UI Inactive  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "Start Inactive", isTrue=→"INACTIVE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on INACTIVE → UI Inactive
  }

  state Detect {
    ListenForMenuActions(eventTarget=Self, submitPressed=→"SELECT PRESSED", cancelPressed=→"CANCEL PRESSED", ignoreAttack=false)
    on SELECT PRESSED → Select Pressed
    on CANCEL PRESSED → Cancel Pressed
  }

  state Select Pressed {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="SELECTION MADE", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Cancel Pressed {
    GetFsmInt(gameObject=var "Self", fsmName="ui_list", variableName="Cancel Item", storeValue=var "Cancel Item", everyFrame=false)
    SetFsmInt(gameObject=var "Self", fsmName="ui_list", variableName="Current Item", setValue=var "Cancel Item", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="SELECTION MADE CANCEL", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI, spawnPoint=var "Hero", audioClip=button, pitchMin=0.65, pitchMax=0.65, volume=0.6, delay=0, storePlayer=<null>)
    on FINISHED → Detect
  }

  state UI Inactive {
  }

  var Start Inactive: bool = false

  var Cancel Item: int = 0
  var Self: gameObject = <null>
}
