# Attach
# 1 location(s):
#   Dream_Abyss / Extra Darkness

fsm Attach {
  start Pause
  on SCENE START → State 1  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Attach
  }

  state Attach {
    SetParent(gameObject=Self, parent=var "Hero", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
  }

  state State 1 {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
