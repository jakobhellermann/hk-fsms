# Shield Attack
# 4 location(s):
#   Deepnest_East_10 / Warrior/Ghost Warrior Markoth
#   Deepnest_East_16 / Warrior/Ghost Warrior Markoth
#   GG_Ghost_Markoth / Warrior/Ghost Warrior Markoth
#   GG_Ghost_Markoth_V / Warrior/Ghost Warrior Markoth

fsm Shield Attack {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    CreateObject(gameObject=Markoth Shield (sharedassets314.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Shield 1", networkInstantiate=false, networkGroup=0)
    SetParent(gameObject=var "Shield 1", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Idle {
    BoolTestMulti(boolVariables=[var "Rage", var "Rage Shield"], boolStates=[true, false], trueEvent=→"RAGE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    WaitRandom(timeMin=10, timeMax=15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Ready
    on RAGE → Rage Pause
  }

  state Ready {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SelectRandomString(strings=["ATTACK CW", "ATTACK CCW"], weights=[1, 1], storeString=var "Event")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK STOP", delay=0, everyFrame=false)
    on FINISHED → Send Attack
  }

  state Send Attack {
    SendEventByName(eventTarget=GameObject(var "Shield 1"), sendEvent=var "Event", delay=0, everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[Markoth_cast_02 (sharedassets314.assets), Markoth_cast_03 (sharedassets314.assets), Markoth_cast_04 (sharedassets314.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on SHIELD END → Reset
  }

  state Reset {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK OK", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK END", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack End")
    on FINISHED → Idle
  }

  state Rage Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rage Anim
  }

  state Rage Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ATTACK STOP", delay=0, everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[<null>, <null>, <null>], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on FINISHED → Send Summon
  }

  state Send Summon {
    SendEventByName(eventTarget=GameObject(var "Shield 1"), sendEvent="SUMMON SHIELD", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Rage Shield", boolValue=true, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Attack Pt", emit=0)
    on FINISHED → Reset
  }

  var Rage: bool = false
  var Rage Shield: bool = false
  var Event: string = ""
  var Attack Pt: gameObject = Warrior/Ghost Warrior Markoth/Attack Pt
  var Self: gameObject = <null>
  var Shield 1: gameObject = <null>
}
