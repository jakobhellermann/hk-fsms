# Destroy If Defeated
# 1 location(s):
#   Ruins1_24_boss / Mage Lord

// uses template: destroy_if_playerdatabool
fsm Destroy If Defeated {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var playerData bool: string = "mageLordDefeated"
}
