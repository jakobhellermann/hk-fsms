# summon
# 2 location(s):
#   GG_False_Knight / Battle Scene/FK Barrel Summon
#   Crossroads_10_boss / Battle Scene/FK Barrel Summon

fsm summon {
  start Idle

  state Idle {
    on SUMMON → Determine Spawns
  }

  state Determine Spawns {
    RandomInt(min=6, max=8, storeResult=var "Spawns", inclusiveMax=true)  // disabled
    WaitRandom(timeMin=0.15, timeMax=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Spawns
  }

  state Spawn {
    RandomFloat(min=var "Summon Min", max=var "Summon Max", storeResult=var "Spawn X")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Spawn Pos", vector3Value=(unset), x=var "Spawn X", y=var "Self Y", z=0.006, everyFrame=false)
    CreateObject(gameObject=Falling Barrel (sharedassets48.assets), spawnPoint=<null>, position=var "Spawn Pos", rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)  // disabled
    SpawnObjectFromGlobalPool(gameObject=Falling Barrel (sharedassets48.assets), spawnPoint=<null>, position=var "Spawn Pos", rotation=(unset), storeObject=var "")
    IntOperator(integer1=var "Spawns", integer2=1, operation=Subtract, storeResult=var "Spawns", everyFrame=false)
    WaitRandom(timeMin=0.15, timeMax=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Spawns
  }

  state Check Spawns {
    IntCompare(integer1=var "Spawns", integer2=0, equal=→"ENDED", lessThan=→"ENDED", greaterThan=→"SUMMON", everyFrame=false)
    on SUMMON → Spawn
    on ENDED → Idle
  }

  var Self Y: float = 0
  var Spawn X: float = 0
  var Summon Max: float = 44.27
  var Summon Min: float = 13.18
  var Spawns: int = 0
  var Spawn Pos: vector3 = (0, 0, 0)
}
