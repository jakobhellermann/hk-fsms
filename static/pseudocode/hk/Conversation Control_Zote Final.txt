# Conversation Control
# 1 location(s):
#   Town / _NPCs/Zote Final Scene/Zote Final

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Box Up
  }

  state Convo Choice {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Zote_01 (sharedassets7.assets), Zote_02 (sharedassets7.assets), Zote_03 (sharedassets7.assets), Zote_04 (sharedassets7.assets), Zote_05 (sharedassets7.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")  // disabled
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ZOTE", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="zotePrecept", storeValue=var "Precept Int")
    IntCompare(integer1=var "Precept Int", integer2=58, equal=→"REPEAT", lessThan=(none), greaterThan=→"REPEAT", everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"PRECEPT", delay=0, everyFrame=false)
    on PRECEPT → Precept
    on REPEAT → Repeat
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)  // disabled
    AudioStop(gameObject=var "Voice Player")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Precept {
    ConvertIntToString(intVariable=var "Precept Int", stringVariable=var "Precept String", format="", everyFrame=false)
    BuildString(stringParts=["PRECEPT_", var "Precept String"], separator="", addToEnd=true, storeResult=var "Convo Name", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Convo Name", "Zote"], storeResult=(unset var))
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="zotePrecept")
    on CONVO_FINISH → Talk Finish
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["PRECEPT_1_R", "Zote"], storeResult=(unset var))
    SetPlayerDataInt(intName="zotePrecept", value=2)
    on CONVO_FINISH → Talk Finish
  }

  var NPC State: int = 0
  var Precept Int: int = 0
  var Spoken: bool = false
  var Convo Name: string = ""
  var Precept String: string = ""
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
