# Corpse
# 1 location(s):
#   sharedassets102.assets / Corpse Mage Lord 1

fsm Corpse {
  start Init

  state Pause {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    FloatSubtract(floatVariable=var "Wait Time", subtract=0.02, everyFrame=false, perSecond=false)
    FloatCompare(float1=var "Wait Time", float2=0.1, tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    on FINISHED → Get Dest
    on END → End
  }

  state Teleport {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_appear (sharedassets34.assets), pitchMin=0.8, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    GetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Teleport Point", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObjectFSM(<null>, fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Tele Out Corpse R, spawnPoint=<null>, position=var "Self Pos", rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Appear Flash R, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=White Flash R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Tele Line
  }

  state Tele Line {
    Vector3Lerp(fromVector=var "Teleport Point", toVector=var "Self Pos", amount=0.5, storeResult=var "Line Pos", everyFrame=false)
    GetAngleBetweenPoints(point1=var "Teleport Point", point2=var "Self Pos", storeAngle=var "Tele Angle", everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Final Tele Distance", point1=var "Self Pos", point2=var "Teleport Point", ignoreZ=false, everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Line Pos", addX=0, addY=0, addZ=0.1, everyFrame=false, perSecond=false)
    CreateObject(gameObject=Tele Line (sharedassets34.assets), spawnPoint=var "", position=var "Line Pos", rotation=(unset), storeObject=var "Tele Line", networkInstantiate=false, networkGroup=0)
    SetScale(gameObject=var "Tele Line", vector=(unset), x=var "Final Tele Distance", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Tele Line", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tele Angle", space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Tele Line", emit=0)
    SetVelocityAsAngle(gameObject=Self, angle=var "Vel Angle", speed=var "Vel Speed", everyFrame=false)  // disabled
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Angular Velocity", everyFrame=false)  // disabled
    on FINISHED → Pause
  }

  state Get Dest {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Cry")
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    StopParticleEmitter(gameObject=var "Weak Particles")
    PlayParticleEmitter(gameObject=var "Death Particles", emit=0)
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)  // disabled
    RandomFloat(min=var "X Min", max=var "X Max", storeResult=var "Tele X")
    RandomFloat(min=var "Y Min", max=var "Y Max", storeResult=var "Tele Y")
    RandomFloatEither(value1=-1, value2=1, storeResult=var "X Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=0.5, max=2, storeResult=var "Vel Speed")
    RandomFloat(min=0, max=360, storeResult=var "Vel Angle")
    RandomFloat(min=-80, max=80, storeResult=var "Angular Velocity")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Teleport Point", vector3Value=(unset), x=var "Tele X", y=var "Tele Y", z=var "Self Z", everyFrame=false)
    on FINISHED → Valid?
  }

  state Init {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DISSIPATE", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun Start 1")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Self Z", space=0, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=2)
    FindChild(gameObject=Self, childName="Hurt Loop", storeResult=var "Hurt Loop")
    on FINISHED → Godseeker Set
  }

  state First Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=-0.75, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Weak Particles", emit=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get Dest
  }

  state Stun Init {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Soul_Master_First_Death_Hit], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=(unset), yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun Start 1")  // disabled
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"LEFT", negativeEvent=→"RIGHT", space=0)
    on LEFT → Stun L
    on RIGHT → Stun R
  }

  state Stun L {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-2.5, y=3, everyFrame=false)
    on FINISHED → Stun Start 1
  }

  state Stun R {
    SetVelocity2d(gameObject=Self, vector=(unset), x=2.5, y=3, everyFrame=false)
    on FINISHED → Stun Start 1
  }

  state Stun Start 1 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Stun Start 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Stun Start 2
  }

  state Stun Start 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Stun Start 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.85)
    on FINISHED → Encountered?
  }

  state End {
    StopParticleEmitter(gameObject=var "Death Particles")
    AudioStop(gameObject=var "Hurt Loop")
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mage_lord_onscreen_appear, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPosition(gameObject=Self, vector=var "End Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Arrive Particles", emit=0)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "End Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=→"FINISHED")
    on FINISHED → Fake Quake
    on GG BOSS → Phase 2
  }

  state Valid? {
    GetPosition(gameObject=Self, vector=var "Current Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    DistanceBetweenPoints(distanceResult=var "Distance", point1=var "Current Pos", point2=var "Teleport Point", ignoreZ=true, everyFrame=false)
    FloatCompare(float1=var "Distance", float2=6, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on FINISHED → Teleport
    on CANCEL → Get Dest
  }

  state Encountered? {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[<null>], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")  // disabled
    AudioPlaySimple(gameObject=var "Hurt Loop", volume=0.5, oneShotClip=<null>)
    SetFloatValue(floatVariable=var "Subtracter", floatValue=0.02, everyFrame=false)
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="mageLordEncountered_2", isTrue=(none), isFalse=→"FINISHED")
    SetFloatValue(floatVariable=var "Subtracter", floatValue=0.055, everyFrame=false)
    SetFloatValue(floatVariable=var "Wait Time", floatValue=0.4, everyFrame=false)
    on FINISHED → Get Dest
  }

  state Quick Death? {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="mageLordEncountered_2", isTrue=→"QUICK", isFalse=(none))
    on FINISHED → Stun Init
    on QUICK → Quick Death
  }

  state Quick Death {
    CreateObject(gameObject=Corpse Dream Mage Lord 1, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Fake Quake {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="QUAKE FAKE APPEAR", delay=0, everyFrame=false)
  }

  state Phase 2 {
    SetFsmGameObject(gameObject=var "Godseeker", fsmName="Control", variableName="Target", setValue=var "Hero", everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="PHASE 2", delay=0, everyFrame=false)
  }

  state Godseeker Set {
    GGCheckIfBossScene(bossSceneEvent=(none), regularSceneEvent=→"FINISHED")
    FindGameObject(objectName="Godseeker Crowd", withTag="Untagged", store=var "Godseeker")
    SetFsmGameObject(gameObject=var "Godseeker", fsmName="Control", variableName="Target", setValue=var "Self", everyFrame=false)
    on FINISHED → Quick Death?
  }

  var Angular Velocity: float = 0
  var Distance: float = 0
  var Final Tele Distance: float = 0
  var Self Z: float = 0
  var Subtracter: float = 0
  var Tele Angle: float = 0
  var Tele X: float = 0
  var Tele Y: float = 0
  var Vel Angle: float = 0
  var Vel Speed: float = 0
  var Wait Time: float = 0.6
  var X Max: float = 33.8
  var X Min: float = 8.34
  var X Scale: float = 0
  var Y Max: float = 36.99
  var Y Min: float = 31.61
  var Current Pos: vector3 = (0, 0, 0)
  var End Pos: vector3 = (20.64, 33.39, 0.005)
  var Line Pos: vector3 = (0, 0, 0)
  var Self Pos: vector3 = (0, 0, 0)
  var Teleport Point: vector3 = (0, 0, 0)
  var Arrive Particles: gameObject = Corpse Mage Lord 1/Arrive Particles
  var Death Particles: gameObject = Corpse Mage Lord 1/Death Particles
  var End Flash: gameObject = Corpse Mage Lord 1/End Flash
  var Godseeker: gameObject = <null>

}
