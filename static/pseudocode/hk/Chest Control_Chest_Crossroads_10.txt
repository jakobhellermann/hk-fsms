# Chest Control
# 1 location(s):
#   Crossroads_10 / Chest

fsm Chest Control {
  start Pause Frame

  state Pause Frame {
    FindChild(gameObject=Self, childName="Item", storeResult=var "Item Holder")
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Bouncer", storeResult=var "Bouncer")
    on FINISHED → Init
  }

  state Init {
    ActivateAllChildren(gameObject=var "Item Holder", activate=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Opened", storeResult=var "Opened")
    FindChild(gameObject=Self, childName="Idle Glow", storeResult=var "Idle Glow")
    FindChild(gameObject=Self, childName="Open Effects", storeResult=var "Open Effects")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Geo Chest", isTrue=→"GEO", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activated
    on GEO → Pool Geo
    on FINISHED → Idle
  }

  state Activated {
    SetCollider(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    StopParticleEmitter(gameObject=var "Idle Glow")
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bouncer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Range?
  }

  state Pool Geo {
    FindChild(gameObject=Self, childName="Geo Pool", storeResult=var "Geo Pool")
    on FINISHED → Idle
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=chest_open (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=barrel_death_2 (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    StopParticleEmitter(gameObject=var "Idle Glow")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Open", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Spawn Items
  }

  state Spawn Items {
    ActivateGameObject(gameObject=var "Open Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlingObjectsFromGlobalPool(gameObject=Geo Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Small", spawnMax=var "Geo Small", speedMin=25, speedMax=38, angleMin=78, angleMax=102, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Geo Med (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Med", spawnMax=var "Geo Med", speedMin=25, speedMax=38, angleMin=78, angleMax=102, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Geo Large (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Large", spawnMax=var "Geo Large", speedMin=25, speedMax=38, angleMin=78, angleMax=102, originVariationX=1, originVariationY=1, FSM="", FSMEvent="")
    on FINISHED → Opened
  }

  state Opened {
    SetCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bouncer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Range? {
    FindChild(gameObject=Self, childName="Hero Region", storeResult=var "Hero Region")
    BoolTest(boolVariable=var "Hero Range", isTrue=→"NAIL HIT", isFalse=(none), everyFrame=false)
    GameObjectIsNull(gameObject=var "Hero Region", isNull=→"NAIL HIT", isNotNull=(none), storeResult=(unset), everyFrame=false)
    on NAIL HIT → Open
    on FINISHED → Idle
  }

  var Geo Large: int = 4
  var Geo Med: int = 10
  var Geo Small: int = 50
  var Geo Chest: bool = true

  var Activated: bool = false
  var Hero Range: bool = false
  var Bouncer: gameObject = <null>
  var Geo Pool: gameObject = <null>
  var Hero Region: gameObject = <null>
  var Idle Glow: gameObject = <null>
  var Item Holder: gameObject = <null>
  var Open Effects: gameObject = <null>
  var Opened: gameObject = <null>
  var Self: gameObject = <null>
}
