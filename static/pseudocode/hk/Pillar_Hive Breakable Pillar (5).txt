# Pillar
# 2 location(s):
#   Hive_03_c / Hive Breakable Pillar (5)
#   Hive_04 / Hive Breakable Pillar (5)

fsm Pillar {
  start Pause

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on ACTIVATE → Activate
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Remains", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    ReceivedDamage(collideTag="", sendEvent=→"HIT", fsmName="damages_enemy", storeGameObject=var "Damager", ignoreAcid=false, ignoreWater=false)
    SetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on HIT → Check
  }

  state Check {
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    IntCompare(integer1=var "Attack Type", integer2=2, equal=→"SPELL", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Attack Type", integer2=0, equal=→"NAIL", lessThan=→"OTHER", greaterThan=→"OTHER", everyFrame=false)
    on CANCEL → Cancel Frame
    on OTHER → Damage
    on NAIL → Attack FX Nail
    on SPELL → Destroy
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Damage {
    AudioPlayRandom(gameObject=var "Self", audioClips=[honey_pillar_hit_1 (sharedassets384.assets), honey_pillar_hit_2 (sharedassets384.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    SetPosition(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"DESTROY", lessThan=→"DESTROY", greaterThan=→"DAMAGED", everyFrame=false)
    on DAMAGED → Damage Effect
    on DESTROY → Destroy
  }

  state Damage Effect {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    ObjectJitter(gameObject=Self, x=0.25, y=(unset), z=(unset), allowMovement=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=15, speedMin=5, speedMax=20, angleMin=160, angleMax=200, originVariationX=0.5, originVariationY=2, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=15, speedMin=5, speedMax=20, angleMin=-20, angleMax=20, originVariationX=0.5, originVariationY=2, FSM="", FSMEvent="")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Idle
  }

  state Attack FX Nail {
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Nail Left
    on RIGHT → Nail Right
    on UP → Nail Up
    on DOWN → Nail Down
  }

  state Nail Left {
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=-1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Damage
  }

  state Nail Right {
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Damage
  }

  state Nail Up {
    RandomFloat(min=70, max=110, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Damage
  }

  state Nail Down {
    RandomFloat(min=250, max=290, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Damage
  }

  state Destroy {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=honey_pillar_final_break (sharedassets384.assets))
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Remains", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rubble", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=15, speedMin=5, speedMax=20, angleMin=160, angleMax=200, originVariationX=0.5, originVariationY=2, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Self", position=(unset), spawnMin=10, spawnMax=15, speedMin=5, speedMax=20, angleMin=-20, angleMax=20, originVariationX=0.5, originVariationY=2, FSM="", FSMEvent="")
    FlingObjects(containerObject=var "Rubble", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=0, speedMax=5, angleMin=0, angleMax=360)
  }

  var Attack Direction: float = 0
  var Chooser: float = 0
  var Attack Type: int = 0
  var Damage: int = 0
  var Hits: int = 5
  var Activated: bool = false
  var Self Pos: vector3 = (0, 0, 0)
  var Damager: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Remains: gameObject = Hive Breakable Pillar (5)/Remains
  var Rubble: gameObject = Hive Breakable Pillar (5)/Rubble
  var Self: gameObject = <null>
  var Sprite: gameObject = Hive Breakable Pillar (5)/Sprite
}
