# Bottle Control
# 1 location(s):
#   sharedassets33.assets / Grub Mimic Bottle Col

fsm Bottle Control {
  start Pause

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Shatter
  }

  state Shatter {
    DestroyObject(gameObject=var "Phys Box", delay=0, detachChildren=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[crystal_break_1 (resources.assets), crystal_break_2 (resources.assets)], weights=[1, 1], pitchMin=1.3, pitchMax=1.3, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Grub"), sendEvent="FREE", delay=0, everyFrame=false)
    on FINISHED → Destroy Self
  }

  state Destroy Self {
    SetCollider(gameObject=Self, active=false)
    SpawnRandomObjects(gameObject=Bottle Lid, spawnPoint=var "Phys Box", position=(0, 1.69, 0), spawnMin=1, spawnMax=1, speedMin=7, speedMax=8, angleMin=85, angleMax=95, originVariation=0)
    ActivateGameObject(gameObject=var "Debris", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObjects(containerObject=var "Debris", adjustPosition=(unset), randomisePosition=false, speedMin=10, speedMax=15, angleMin=40, angleMax=140)
    PlayParticleEmitter(gameObject=var "Pt Glass S", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Glass L", emit=0)
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state Pause {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Debris", storeResult=var "Debris")
    FindChild(gameObject=Self, childName="Grub Mimic Top", storeResult=var "Grub")
    FindChild(gameObject=Self, childName="Bottle Physical", storeResult=var "Phys Box")
    FindChild(gameObject=Self, childName="Pt Glass L", storeResult=var "Pt Glass L")
    FindChild(gameObject=Self, childName="Pt Glass S", storeResult=var "Pt Glass S")
    on FINISHED → Idle
    on ACTIVATE → 
  }

  var Grub: gameObject = Grub Mimic Bottle Col/Grub Mimic Top

  var Speed Max: float = 30
  var Speed Min: float = 15
  var Activated: bool = false
  var Debris: gameObject = Grub Mimic Bottle/Debris
  var Phys Box: gameObject = <null>
  var Pool: gameObject = <null>
  var Pt Glass L: gameObject = Grub Mimic Bottle/Pt Glass L
  var Pt Glass S: gameObject = Grub Mimic Bottle/Pt Glass S
  var Self: gameObject = <null>
}
