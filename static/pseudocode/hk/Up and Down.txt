# Up and Down
# 2 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Charms/Equipped Charms/Next Dot
#   resources.assets / _GameCameras/HudCamera/Inventory/Charms/Equipped Charms/Next Dot

fsm Up and Down {
  start Idle
  on NOTCH DEF UP → To Up  // from any state

  state Idle {
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite")
    on NOTCH DOWN → Down
    on NOTCH UP → Up
    on FINISHED → Up
  }

  state Down {
    SetSpriteRenderer(gameObject=var "Sprite", active=false)
    on NOTCH UP → Up
  }

  state Up {
    SetSpriteRenderer(gameObject=var "Sprite", active=true)
    SetScale(gameObject=var "Sprite", vector=(unset), x=0.1, y=0.1, z=0.1, everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=var "Sprite", id="", transformScale=var "", vectorScale=(2.494224, 2.494224, 2.494224), time=0.15, delay=0, speed=(unset), easeType=easeOutSine, loopType=none, startEvent=(none), finishEvent=(none), realTime=true, stopOnExit=true, loopDontFinish=true)
    on NOTCH DOWN → Down
  }

  state To Up {
    on FINISHED → Up
  }

  var Sprite: gameObject = <null>
}
