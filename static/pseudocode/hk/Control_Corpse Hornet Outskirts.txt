# Control
# 2 location(s):
#   Deepnest_East_Hornet_boss / Corpse Hornet Outskirts

fsm Control {
  start Set PD

  state Blow {
    GetOwner(storeGameObject=var "Self")
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Hornet_Fight_Death_01 (sharedassets105.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=FreezeMoment(0))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=boss_explode_clean (resources.assets))
    SendEventByName(eventTarget=BroadcastAll, sendEvent="HORNET KILLED", delay=0, everyFrame=false)
    FindGameObject(objectName="Needle Tink", withTag="Untagged", store=var "Needle Tink")
    DestroyObject(gameObject=var "Needle Tink", delay=0, detachChildren=false)
    SpawnRandomObjectsV2(gameObject=Slash Effect Ghost1 (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=8, spawnMax=8, speedMin=2, speedMax=35, angleMin=0, angleMax=360, originVariationX=0, originVariationY=0)
    SpawnRandomObjectsV2(gameObject=Slash Effect Ghost2 (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=2, spawnMax=3, speedMin=2, speedMax=35, angleMin=0, angleMax=360, originVariationX=0, originVariationY=0)
    CreateObject(gameObject=White Wave (sharedassets6.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(0))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    on FINISHED → Launch
  }

  state Launch {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Velocity", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=5, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=20, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → In Air
  }

  state In Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=true, lateUpdate=false)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wounded")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    CheckCurrentMapZone(mapZone="DREAM_WORLD", equalEvent=→"CANCEL", notEqualEvent=(none))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Can Talk
    on CANCEL → Don't Talk
  }

  state Set PD {
    SetPlayerDataBool(boolName="hornetOutskirtsDefeated", value=true)
    FindChild(gameObject=Self, childName="NPC", storeResult=var "NPC")
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["HORNET_2"], storeResult=(unset var))
    FindChild(gameObject=Self, childName="Enemy Dream", storeResult=var "Enemy Dream")
    on FINISHED → Limit Y
  }

  state Can Talk {
    ActivateGameObject(gameObject=var "NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on HORNET END CONVO → Pause frame
  }

  state Check Pos {
    SetCollider(gameObject=var "NPC", active=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Pos", float2=27.3, tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on L → L
    on R → R
  }

  state L {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVector3Value(vector3Variable=var "Leave Vector", vector3Value=(8, 8, 0), everyFrame=false)
    SetFloatValue(floatVariable=var "Rotation", floatValue=45, everyFrame=false)
    on FINISHED → Jump
  }

  state R {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetVector3Value(vector3Variable=var "Leave Vector", vector3Value=(-8, 8, 0), everyFrame=false)
    SetFloatValue(floatVariable=var "Rotation", floatValue=-45, everyFrame=false)
    on FINISHED → Jump
  }

  state Jump {
    Wait(time=0.415, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=Hornet_Fight_Yell_05 (sharedassets133.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Leave Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    Tk2dPlayAnimation(gameObject=var "Leave Anim", animLibName="", clipName="Jump Full")
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 6, 0), time=0.7, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=true)
    on FINISHED → Throw Start
  }

  state Throw Start {
    Tk2dPlayAnimationWithEvents(gameObject=var "Leave Anim", clipName="Throw Side Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Throw
  }

  state Throw {
    ActivateGameObject(gameObject=var "Thread", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Thread", clipName="Thread 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=var "Leave Anim", animLibName="", clipName="Throw Side")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_needle_thow_spin (sharedassets105.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Yank
  }

  state Yank {
    Tk2dPlayAnimation(gameObject=var "Leave Anim", animLibName="", clipName="Harpoon Side")
    SetParent(gameObject=var "Thread", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetParent(gameObject=var "Thread", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    iTweenMoveBy(gameObject=Self, id="", vector=var "Leave Vector", time=0.37, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hornet_dash (sharedassets34.assets), pitchMin=1.15, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  state End {
    SetParent(gameObject=var "Grass Escape", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetRotation(gameObject=var "Grass Escape", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=0, zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Leave Anim", active=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="HORNET LEAVE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Dream", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → 
  }

  state Pause frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Pos
  }

  state Limit Y {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Y Pos", minValue=29.2, maxValue=9999, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Blow
  }

  state Don't Talk {
  }

  var Rotation: float = 0
  var X Pos: float = 0
  var X Velocity: float = 0
  var Y Pos: float = 0
  var Leave Vector: vector3 = (0, 0, 0)
  var Enemy Dream: gameObject = <null>
  var Grass Escape: gameObject = Corpse Hornet Outskirts/Grass Escape
  var Leave Anim: gameObject = Corpse Hornet Outskirts/Leave Anim
  var NPC: gameObject = <null>
  var Needle Tink: gameObject = <null>
  var Self: gameObject = <null>
  var Thread: gameObject = Corpse Hornet Outskirts/Thread
}
