# destroy_if_playerdatabool
# 1 location(s):
#   Deepnest_East_Hornet_boss / Hornet Boss 2

// uses template: destroy_if_playerdatabool
fsm destroy_if_playerdatabool {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var playerData bool: string = "hornetOutskirtsDefeated"
}
