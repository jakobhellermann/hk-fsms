# Control
# 1 location(s):
#   Abyss_19 / infected_door

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="killedInfectedKnight", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
    on FINISHED → Init
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Idle {
    on IK GATE OPEN → Open Rumble
  }

  state Open Rumble {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    FlingObjectsFromGlobalPoolTime(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Rumble Spurt", position=(unset), frequency=0.1, spawnMin=3, spawnMax=4, speedMin=20, speedMax=30, angleMin=0, angleMax=45, originVariationX=0, originVariationY=1.75)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst
  }

  state Init {
    FindChild(gameObject=Self, childName="Blobs", storeResult=var "Blobs")
    FindChild(gameObject=Self, childName="Globs", storeResult=var "Globs")
    FindChild(gameObject=Self, childName="Collider", storeResult=var "Collider")
    FindChild(gameObject=Self, childName="Rumble Spurt", storeResult=var "Rumble Spurt")
    FindChild(gameObject=Self, childName="Splat", storeResult=var "Splat")
    on FINISHED → Idle
  }

  state Burst {
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=enemy_death_sword (resources.assets), pitchMin=0.75, pitchMax=0.75, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=explosion_4_wet (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Blobs", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Splat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Rumble Spurt", position=(unset), spawnMin=50, spawnMax=50, speedMin=5, speedMax=35, angleMin=45, angleMax=0, originVariationX=0, originVariationY=1.75, FSM="", FSMEvent="")
    ActivateAllChildren(gameObject=var "Globs", activate=true)
    FlingObjects(containerObject=var "Globs", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=5, speedMax=10, angleMin=45, angleMax=135)
  }

  var Blobs: gameObject = <null>
  var Collider: gameObject = <null>
  var Globs: gameObject = <null>
  var Rumble Spurt: gameObject = <null>
  var Splat: gameObject = <null>
}
