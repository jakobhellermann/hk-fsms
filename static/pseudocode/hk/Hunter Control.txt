# Hunter Control
# 1 location(s):
#   Fungus1_08 / Hunter Eyes

fsm Hunter Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Appear Region", isTrue=→"ENTER", isFalse=(none), everyFrame=true)
    on ENTER → Appear
  }

  state Appear {
    Tk2dPlayAnimation(gameObject=var "Eyes", animLibName="", clipName="Eyes Open")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="hunterRoared", isTrue=(none), isFalse=→"ROAR")
    Tk2dWatchAnimationEvents(gameObject=var "Eyes", animationTriggerEvent=(none), animationCompleteEvent=→"IDLE")
    on ROAR → Start Roar
    on IDLE → Appeared
  }

  state Start Roar {
    SetPlayerDataBool(boolName="hunterRoared", value=true)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SetParent(gameObject=var "Roar Emitter", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=mawlek_scream (sharedassets45.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End Effect
  }

  state End Roar {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Eyes", animLibName="", clipName="Eyes Idle")
  }

  state Appeared {
    Tk2dPlayAnimation(gameObject=var "Eyes", animLibName="", clipName="Eyes Idle")
  }

  state End Effect {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    on FINISHED → End Roar
  }

  var Appear Region: bool = false
  var Eyes: gameObject = Hunter Eyes/Eyes
  var Roar Emitter: gameObject = <null>
  var Self: gameObject = <null>
}
