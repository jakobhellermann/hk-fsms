# breakable_wall_v2
# 1 location(s):
#   Ruins_Bathhouse / Breakable Wall

// uses template: breakable_wall_v2
fsm breakable_wall_v2 {
  start Get Refs

  state Idle {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Wall Breaker", collideLayer="", sendEvent=→"WALL BREAKER", storeCollider=var "")
    on TAKE DAMAGE → Check If Nail
    on WALL BREAKER → PD Bool?
  }

  state Check If Nail {
    GetEventSender(sentByGameObject=var "Damager")
    GetFsmInt(gameObject=var "Damager", fsmName="damages_enemy", variableName="attackType", storeValue=var "Attack Type", everyFrame=false)
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    IntSwitch(intVariable=var "Attack Type", compareTo=[0, 2], sendEvent=[→"TRUE", →"SPELL"], everyFrame=false)
    on TRUE → Hit
    on FINISHED → Idle
    on SPELL → Spell Destroy
  }

  state Hit {
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Strike Effect")
    SetScale(gameObject=var "Strike Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Direction
  }

  state Initiate {
    FindChild(gameObject=var "Self", childName="Dust Break 1", storeResult=var "Dust Break 1")
    FindChild(gameObject=var "Self", childName="Dust Break 2", storeResult=var "Dust Break 2")
    SetParent(gameObject=var "Pool S", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Pool L", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activated
  }

  state Check Direction {
    RandomFloat(min=0.85, max=1.1, storeResult=var "Chooser")
    SetAudioPitch(gameObject=var "Self", pitch=var "Chooser", everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sharedassets6.assets), breakable_wall_hit_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.85, pitchMax=1.15)
    IntSwitch(intVariable=var "Facing", compareTo=[0, 1, 2, 3], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on RIGHT → Hit Right
    on UP → Hit Up
    on LEFT → Hit Left
    on DOWN → Hit Down
  }

  state Break {
    PlayParticleEmitter(gameObject=var "Dust Break 1", emit=0)
    PlayParticleEmitter(gameObject=var "Dust Break 2", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=breakable_wall_death (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (resources.assets), spawnPoint=var "Hero", audioClip=secret_discovered_temp (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetCollider(gameObject=var "Self", active=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UNCOVER", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="BREAK", delay=0, everyFrame=false)
    CreateObject(gameObject=Dust Break Wall, spawnPoint=var "Self", position=(unset), rotation=(-72.5, -180, -180), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetSpriteRenderer(gameObject=Self, active=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=0)
    PlayParticleEmitter(gameObject=var "Particles Large", emit=0)
    on FINISHED → Destroy
  }

  state Hit Right {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=-1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    SpawnFromPool(pool=var "Pool S", adjustPosition=(1.12, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=40, angleMax=80)
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed Neg", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Return Right
    on HIT → Pause Frame
  }

  state Return Right {
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit Left {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (resources.assets), spawnPoint=var "Self", position=(-1, 0, 0), rotation=(180, 90, 270), storeObject=var "")
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Return Left
    on HIT → Pause Frame
  }

  state Return Left {
    SetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Recoil Speed Neg", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit Down {
    SpawnFromPool(pool=var "Pool S", adjustPosition=(0, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=70, angleMax=110)  // disabled
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(270, 90, 270), storeObject=var "")
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Recoil Speed", everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Return Down
    on HIT → Pause Frame
  }

  state Return Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Recoil Speed Neg", everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit Up {
    SpawnFromPool(pool=var "Pool S", adjustPosition=(0, 0, 0), spawnMin=8, spawnMax=10, speedMin=15, speedMax=20, angleMin=70, angleMax=110)  // disabled
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(270, 90, 270), storeObject=var "")
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    PlayParticleEmitter(gameObject=var "Particles Small", emit=5)
    on FINISHED → Idle
    on HIT → Pause Frame
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    DestroyObject(gameObject=var "Camera Locks", delay=0, detachChildren=false)
    on FINISHED → PD Bool?
  }

  state Destroy {
    ActivateGameObject(gameObject=var "Camera Locks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  sequence state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Activated?
  }

  state Activated {
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false)
    DestroyObject(gameObject=var "Camera Locks", delay=0, detachChildren=false)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UNCOVER", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Masks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Activated? {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    StringCompare(stringVariable=var "PlayerData Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PlayerData Bool", isTrue=→"ACTIVATE", isFalse=(none))
    on ACTIVATE → Activated
    on FINISHED → Ruin Lift?
  }

  state Ruin Lift? {
    BoolTest(boolVariable=var "Ruin Lift", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CheckTargetDirection(gameObject=var "Self", target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"RUIN LIFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)  // disabled
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=34.5, tolerance=0, equal=(none), lessThan=→"RUIN LIFT", greaterThan=(none), everyFrame=false)
    on FINISHED → Initiate
    on RUIN LIFT → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=var "Masks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Get Refs {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Self", childName="Camera Locks", storeResult=var "Camera Locks")
    FindChild(gameObject=var "Self", childName="Masks", storeResult=var "Masks")
    FindChild(gameObject=Self, childName="Particle_rocks_small", storeResult=var "Particles Small")
    FindChild(gameObject=Self, childName="Particle_rocks_large", storeResult=var "Particles Large")
    on FINISHED → Pause
  }

  state PD Bool? {
    StringCompare(stringVariable=var "PlayerData Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "PlayerData Bool", value=true)
    on FINISHED → Break
  }

  state Spell Destroy {
    SpawnObjectFromGlobalPool(gameObject=Fireball Hit (resources.assets), spawnPoint=var "Self", position=var "Effect Origin", rotation=(unset), storeObject=var "Hit Effect")
    on FINISHED → PD Bool?
  }

  var Facing: int = 2
  var Hits: int = 4
  var Ruin Lift: bool = false
  var PlayerData Bool: string = "bathHouseWall"

  var Attack Direction: float = 0
  var Chooser: float = 0
  var Hero X: float = 0
  var Recoil Speed: float = 1
  var Recoil Speed Neg: float = -1
  var Start X: float = 0
  var Start Y: float = 0
  var Attack Type: int = 0
  var Activated: bool = false
  var Camera Locks: gameObject = <null>
  var Damager: gameObject = <null>
  var Dust Break 1: gameObject = <null>
  var Dust Break 2: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Masks: gameObject = <null>
  var Particles Large: gameObject = <null>
  var Particles Small: gameObject = <null>
  var Pool L: gameObject = <null>
  var Pool S: gameObject = <null>
  var Self: gameObject = <null>
  var Strike Effect: gameObject = <null>
}
