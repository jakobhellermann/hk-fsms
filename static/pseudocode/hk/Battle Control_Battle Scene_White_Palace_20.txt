# Battle Control
# 1 location(s):
#   White_Palace_20 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Wave 1", storeResult=var "Wave 1")
    FindChild(gameObject=Self, childName="Gate", storeResult=var "Gate")
    on FINISHED → Idle
  }

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="CameraLockAreaB", storeResult=var "CamLock")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ACTIVATE → Activate
    on ENTER → Start
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    ActivateGameObject(gameObject=var "Royal Gaurd", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Start {
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WAKE", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=HK Decline 5 (resources.assets), transitionTime=2)
    on ROYAL GUARD KILLED → Check Pause
  }

  state End Pause {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sting
  }

  state Open {
    SendEventByName(eventTarget=GameObject(var "Gate"), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Gate", active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=white_seal_1 (sharedassets19.assets), pitchMin=0.8, pitchMax=0.8, volume=1, delay=0, storePlayer=<null>)
  }

  state Check {
    GameObjectHasChildren(gameObject=var "Wave 1", trueEvent=→"FINISHED", falseEvent=→"END", storeResult=(unset), everyFrame=false)
    on FINISHED → Start
    on END → End Pause
  }

  state Check Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Sting {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=Final Boss transition and end sting (sharedassets383.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=1)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  var Activated: bool = false
  var CamLock: gameObject = <null>
  var Gate: gameObject = <null>
  var Royal Gaurd: gameObject = <null>
  var Self: gameObject = <null>
  var Wave 1: gameObject = <null>
}
