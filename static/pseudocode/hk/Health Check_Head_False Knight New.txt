# Health Check
# 2 location(s):
#   GG_False_Knight / Battle Scene/False Knight New/Head
#   Crossroads_10_boss / Battle Scene/False Knight New/Head

fsm Health Check {
  start Check 1

  state Check 1 {
    GetOwner(storeGameObject=var "Self")
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=ColorReturnNeutral())
    GetFsmInt(gameObject=Self, fsmName="health_manager", variableName="HP", storeValue=var "HP", everyFrame=true)  // disabled
    IntCompare(integer1=var "HP", integer2=0, equal=→"STUN", lessThan=→"STUN", greaterThan=(none), everyFrame=true)  // disabled
    on STUN → State 1
    on ZERO HP → State 1
  }

  state State 1 {
    SpawnBlood(spawnPoint=var "Self", position=(0, 0, 0), spawnMin=50, spawnMax=50, speedMin=5, speedMax=35, angleMin=20, angleMax=160, colorOverride=(unset))
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=ColorReturnNeutral())
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Idle")
    SetHP(target=Self, hp=40)
    SendEventByName(eventTarget=GameObject(var "Body"), sendEvent="STUN END", delay=0, everyFrame=false)
    on FINISHED → Check 1
  }

  var HP: int = 0
  var Body: gameObject = Battle Scene/False Knight New
  var Self: gameObject = <null>
}
