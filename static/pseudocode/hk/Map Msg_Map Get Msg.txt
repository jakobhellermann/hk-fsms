# Map Msg
# 1 location(s):
#   sharedassets64.assets / Map Get Msg

fsm Map Msg {
  start Init
  on DESTROY JOURNAL MSG → Destroy  // from any state

  state Init {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DESTROY JOURNAL MSG", delay=0, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Icon", storeResult=var "Icon")
    FindChild(gameObject=Self, childName="Text", storeResult=var "Text")
    SetPosition(gameObject=Self, vector=var "Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    GetLanguageString(sheetName="UI", convName=var "Convo", storeValue=var "Name")
    SetTextMeshProText(gameObject=var "Text", textString=var "Name")
    on FINISHED → Up
  }

  state Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=4.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroyObject(gameObject=var "Self", delay=0, detachChildren=false)
  }

  var Position: vector3 = (-11.91, -6.22, 0)

  var ID: int = 0
  var Full: bool = false
  var Anim Down: string = ""
  var Convo: string = ""
  var ID String: string = ""
  var Name: string = ""
  var Name Convo: string = ""
  var Text Str: string = ""
  var Icon: gameObject = <null>
  var Self: gameObject = <null>
  var Text: gameObject = <null>
  var Sprite: object = <null>
}
