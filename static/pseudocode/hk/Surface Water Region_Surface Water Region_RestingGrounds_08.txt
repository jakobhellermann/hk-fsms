fsm Surface Water Region {
  start Pause

  state Idle {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"HERO ENTER", storeCollider=var "")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Hero Surface Y", z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Hero Surface Y", add=var "Hero Offset", everyFrame=false, perSecond=false)
    on HERO ENTER → Big Splash?
  }

  state Out {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"HERO ENTER", storeCollider=var "")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="SURFACE EXIT", delay=0, everyFrame=false)
    on HERO ENTER → Big Splash?
  }

  state In {
    SendEventByName(eventTarget=GameObject(var "Hero Obj"), sendEvent="SURFACE ENTER", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="INVENTORY CANCEL", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Inventory"), sendEvent="INVENTORY CANCEL", delay=0, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"HERO EXIT", storeCollider=var "")
    on HERO EXIT → Splash Out effects
  }

  state Splash In Norm {
    SetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Surface Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    FlingObjectsFromGlobalPool(gameObject=var "Spatter Obj", spawnPoint=var "Hero", position=(0.2, -0.7, 0), spawnMin=6, spawnMax=8, speedMin=7, speedMax=15, angleMin=120, angleMax=145, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=var "Spatter Obj", spawnPoint=var "Hero", position=(-0.2, -0.7, 0), spawnMin=6, spawnMax=8, speedMin=7, speedMax=15, angleMin=35, angleMax=60, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=0.8, pitchMax=0.95, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Splash In Black (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Splash")
    SetScale(gameObject=var "Splash", vector=(unset), x=1.75, y=1.75, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=var "Splash", color=var "Colour", everyframe=false)
    on FINISHED → In
  }

  state Splash Out effects {
    SpawnObjectFromGlobalPool(gameObject=var "Dripper Obj", spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=1, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=var "Splash Out Obj", spawnPoint=var "Hero", position=(0, -1, 0), rotation=(unset), storeObject=var "Splash")
    on FINISHED → Out
  }

  state Big Splash? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["willHardLand"], storeResult=var "Will Hard Land")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["spellQuake"], storeResult=var "Quaking")
    BoolTest(boolVariable=var "Quaking", isTrue=→"BIG", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Will Hard Land", isTrue=→"BIG", isFalse=→"NORMAL", everyFrame=false)
    on NORMAL → Splash In Norm
    on BIG → Splash In Big
  }

  state Splash In Big {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Surface Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    FlingObjectsFromGlobalPool(gameObject=var "Spatter Obj", spawnPoint=var "Hero", position=(0, -0.7, 0), spawnMin=60, spawnMax=60, speedMin=10, speedMax=30, angleMin=80, angleMax=100, originVariationX=1, originVariationY=0, FSM="", FSMEvent="")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=water_splash (sharedassets31.assets), pitchMin=0.8, pitchMax=0.95, volume=1, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Splash In Black (resources.assets), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Splash")
    SetScale(gameObject=var "Splash", vector=(unset), x=2.5, y=2.5, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=var "Splash", color=var "Colour", everyframe=false)
    SpawnObjectFromGlobalPool(gameObject=var "Splash Out Obj", spawnPoint=var "Hero", position=(0, 1, 0), rotation=(unset), storeObject=var "Splash")
    SetScale(gameObject=var "Splash", vector=(unset), x=2, y=2, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → In
  }

  state Init {
    SetGameObject(variable=var "Hero Obj", gameObject=var "Hero", everyFrame=false)
    SetGameObject(variable=var "HUD Cam Obj", gameObject=var "HUD Camera", everyFrame=false)
    BoolTest(boolVariable=var "Black", isTrue=→"BLACK", isFalse=→"NORMAL", everyFrame=false)
    on BLACK → Black
    on NORMAL → Blue
  }

  state Black {
    SetColorValue(colorVariable=var "Colour", color=(0, 0, 0, 1), everyFrame=false)
    SetGameObject(variable=var "Splash Out Obj", gameObject=Splash Out Black (resources.assets), everyFrame=false)
    SetGameObject(variable=var "Spatter Obj", gameObject=Spatter Black (resources.assets), everyFrame=false)
    SetGameObject(variable=var "Dripper Obj", gameObject=Dripper Black (sharedassets50.assets), everyFrame=false)
    on FINISHED → Idle
  }

  state Blue {
    SetColorValue(colorVariable=var "Colour", color=(1, 1, 1, 1), everyFrame=false)
    SetGameObject(variable=var "Splash Out Obj", gameObject=Splash Out White (resources.assets), everyFrame=false)
    SetGameObject(variable=var "Spatter Obj", gameObject=Spatter White R (resources.assets), everyFrame=false)
    SetGameObject(variable=var "Dripper Obj", gameObject=Dripper White (sharedassets50.assets), everyFrame=false)
    on FINISHED → Idle
  }

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  var Hero Offset: float = 1.53
  var Black: bool = false

  var Hero Surface Y: float = 0
  var Quaking: bool = false
  var Will Hard Land: bool = false
  var Colour: color = (0, 0, 0, 1)
  var Dripper Obj: gameObject = <null>
  var HUD Cam Obj: gameObject = <null>
  var Hero Obj: gameObject = <null>
  var Spatter Obj: gameObject = <null>
  var Splash: gameObject = <null>
  var Splash Out Obj: gameObject = <null>
}
