# Scream Get
# 1 location(s):
#   Abyss_12 / Scream 2 Get

fsm Scream Get {
  start Pause
  on SCREAM GET → Get Pause  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="screamLevel", storeValue=var "Scream Level")
    IntCompare(integer1=var "Scream Level", integer2=2, equal=→"INERT", lessThan=(none), greaterThan=→"INERT", everyFrame=false)
    FindChild(gameObject=Self, childName="Particle Wave", storeResult=var "Particle Wave")
    FindChild(gameObject=Self, childName="Shade Particles 1", storeResult=var "Shade Particles 1")
    FindChild(gameObject=Self, childName="Shade Particles 2", storeResult=var "Shade Particles 2")
    FindChild(gameObject=Self, childName="Shade Particles 3", storeResult=var "Shade Particles 3")
    FindChild(gameObject=Self, childName="Shade Particles 4", storeResult=var "Shade Particles 4")
    FindChild(gameObject=Self, childName="Shade Particles 5", storeResult=var "Shade Particles 5")
    FindChild(gameObject=Self, childName="Shade Particles 6", storeResult=var "Shade Particles 6")
    FindChild(gameObject=Self, childName="Shade Particles 7", storeResult=var "Shade Particles 7")
    FindChild(gameObject=Self, childName="Shade Particles 8", storeResult=var "Shade Particles 8")
    FindChild(gameObject=Self, childName="Pt Pool", storeResult=var "Pt Pool")
    FindChild(gameObject=Self, childName="Orb Spawn", storeResult=var "Orb Spawn")
    FindChild(gameObject=Self, childName="Black Wave", storeResult=var "Black Wave")
    FindChild(gameObject=Self, childName="Cutscene Knight", storeResult=var "Cutscene Knight")
    on FINISHED → Idle
    on INERT → Inert
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    SetFsmBool(gameObject=var "Hero", fsmName="Spell Control", variableName="Scream 2 Zone", setValue=true, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
  }

  state Out {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SetFsmBool(gameObject=var "Hero", fsmName="Spell Control", variableName="Scream 2 Zone", setValue=false, everyFrame=false)
    on ENTER → In
  }

  state Get Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Spell Control", variableName="Scream 2 Zone", setValue=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    SetParent(gameObject=var "Cutscene Knight", parent=var "Hero", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Cutscene Knight", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    ActivateGameObject(gameObject=var "Cutscene Knight", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Cutscene Knight", animLibName="", clipName="Orb")
    on FINISHED → Move To
  }

  state Move To {
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    Vector3Compare(vector3Variable1=var "Hero Pos", vector3Variable2=var "Get Pos", tolerance=1, equal=→"FINISHED", notEqual=(none), everyFrame=false)
    iTweenMoveTo(gameObject=var "Hero", id="", transformPosition=var "", vectorPosition=(47.38, 14.41, 0.004), time=(unset), delay=0, speed=7, space=0, easeType=easeInOutSine, loopType=none, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Moved Pause
  }

  state Moved Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Cutscene Knight", animLibName="", clipName="Absorb")
    on FINISHED → Spill
  }

  state Spill {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, finishedEvent=(none))
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=1)
    PlayParticleEmitter(gameObject=var "Shade Particles 1", emit=0)
    PlayParticleEmitter(gameObject=var "Shade Particles 2", emit=0)
    PlayParticleEmitter(gameObject=var "Shade Particles 3", emit=0)
    PlayParticleEmitter(gameObject=var "Shade Particles 4", emit=0)
    PlayParticleEmitter(gameObject=var "Shade Particles 5", emit=0)
    PlayParticleEmitter(gameObject=var "Shade Particles 6", emit=0)
    PlayParticleEmitter(gameObject=var "Shade Particles 7", emit=0)
    PlayParticleEmitter(gameObject=var "Shade Particles 8", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Pool", emit=0)
    Wait(time=4.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Spill
  }

  state End Spill {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=1)  // disabled
    StopParticleEmitter(gameObject=var "Shade Particles 1")
    StopParticleEmitter(gameObject=var "Shade Particles 2")
    StopParticleEmitter(gameObject=var "Shade Particles 3")
    StopParticleEmitter(gameObject=var "Shade Particles 4")
    StopParticleEmitter(gameObject=var "Shade Particles 5")
    StopParticleEmitter(gameObject=var "Shade Particles 6")
    StopParticleEmitter(gameObject=var "Shade Particles 7")
    StopParticleEmitter(gameObject=var "Shade Particles 8")
    on FINISHED → Orb Spawn
  }

  state Orb Spawn {
    AudioStop(gameObject=Self)
    StopParticleEmitter(gameObject=var "Pt Pool")
    SpawnRandomObjectsV2(gameObject=Spell Get Orb Black (sharedassets274.assets), spawnPoint=var "Orb Spawn", position=(unset), spawnMin=100, spawnMax=100, speedMin=5, speedMax=15, angleMin=80, angleMax=100, originVariationX=7, originVariationY=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=dark_spell_absorb (sharedassets28.assets))
    on FINISHED → Getting
  }

  state Getting {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    Wait(time=2.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get
  }

  state Get {
    GetPosition(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=dark_spell_get (resources.assets))
    SetPosition(gameObject=var "Black Wave", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Particle Wave", vector=var "Hero Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Black Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Cutscene Knight", clipName="Return", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Particle Wave", emit=0)
    on FINISHED → Fall
  }

  state Fall {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=var "Hero", frame=9)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=10.5, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    ActivateGameObject(gameObject=var "Cutscene Knight", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Cutscene Knight", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Dreamer Land")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hornet_ground_land (sharedassets28.assets))
    on FINISHED → Ui Msg
  }

  state Ui Msg {
    SetPlayerDataInt(intName="screamLevel", value=2)
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (sharedassets6.assets), storeObject=var "UI Msg", sprite=_0007_shadow_scream (sharedassets338.assets))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Scream 2", everyFrame=false)
    on GET ITEM MSG END → Get Up
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End
  }

  state Inert {
  }

  state End {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="SCREAM GET END", delay=0, everyFrame=false)
  }

  var Hero Y: float = 0
  var Scream Level: int = 0
  var Get Pos: vector3 = (47.38, 14.41, 0.004)
  var Hero Pos: vector3 = (0, 0, 0)
  var Black Wave: gameObject = <null>
  var Cutscene Knight: gameObject = <null>
  var Orb Spawn: gameObject = <null>
  var Particle Wave: gameObject = <null>
  var Pt Pool: gameObject = <null>
  var Shade Particles 1: gameObject = <null>
  var Shade Particles 2: gameObject = <null>
  var Shade Particles 3: gameObject = <null>
  var Shade Particles 4: gameObject = <null>
  var Shade Particles 5: gameObject = <null>
  var Shade Particles 6: gameObject = <null>
  var Shade Particles 7: gameObject = <null>
  var Shade Particles 8: gameObject = <null>
  var UI Msg: gameObject = UI Msg Get Item(Clone):3
}
