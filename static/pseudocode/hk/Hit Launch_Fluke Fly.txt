fsm Hit Launch {
  start Idle

  state Idle {
    on HIT LEFT → Hit Left
    on HIT RIGHT → Hit Right
    on HIT UP → Hit Up
    on HIT DOWN → Hit Down
    on GO UP → Move Up
  }

  state Hit Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-25, y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Hit Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=25, y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Hit Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=25, everyFrame=false)
    on FINISHED → Idle
  }

  state Hit Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-25, everyFrame=false)
    on FINISHED → Idle
  }

  state Move Up {
    GetFsmBool(gameObject=Self, fsmName="Fluke Fly", variableName="In Range", storeValue=var "In Range", everyFrame=false)
    BoolTest(boolVariable=var "In Range", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Hit Up
    on FALSE → Hit Up 2
  }

  state Hit Up 2 {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0.5, everyFrame=false)
    on FINISHED → Idle
  }

  var In Range: bool = false
}
