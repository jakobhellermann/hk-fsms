# Cancel
# 3 location(s):
#   Dream_Guardian_Hegemol / Dreamer NPC/Cage
#   Dream_Guardian_Lurien / Dreamer NPC/Cage
#   Dream_Guardian_Monomon / Dreamer NPC/Cage

fsm Cancel {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="SLOPE CANCEL", delay=0, everyFrame=false)
    on FINISHED → Idle
  }
}
