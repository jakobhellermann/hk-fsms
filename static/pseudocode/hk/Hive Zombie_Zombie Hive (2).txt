# Hive Zombie
# 1 location(s):
#   Hive_02 / Zombie Hive (2)

fsm Hive Zombie {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Extra Tag", store=var "Cage")
    FindChild(gameObject=Self, childName="Ground Check", storeResult=var "Ground Check Obj")
    on FINISHED → Friendly?
  }

  state Idle {
    BoolTest(boolVariable=var "Alert Range", isTrue=→"ALERT", isFalse=(none), everyFrame=true)
    on ALERT → Startle
  }

  state Run {
    AudioPlayInState(gameObject=var "FootstepAudio", volume=1)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToRun", resetFrame=true, everyFrame=true)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Self X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Self X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Facing Right", greaterThanBool=(unset), everyFrame=false)
    SetWalkerFacing(walkRight=var "Facing Right", randomStartDir=false, target=Self, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    RunAway(gameObject=Self, target=var "Hero", speedMax=8, acceleration=0.4, animateTurnAndRun=false, runAnimation="", turnAnimation="")
    AudioPlayInState(gameObject=Self, volume=1)
    WaitRandom(timeMin=0.7, timeMax=1.4, finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "Terrain Back", isTrue=→"BACK", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Unalert Range", isTrue=(none), isFalse=→"UNALERT", everyFrame=true)
    RayCast2d(fromGameObject=var "Centre", fromPosition=(unset), direction=(-1, 0), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"FOREWARD", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Move Choice
    on BACK → Back High?
    on FOREWARD → Fore High?
    on UNALERT → Unalert
  }

  state Startle {
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Birth", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Royal_Zombie_death_01 (sharedassets84.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Run
  }

  state Move Choice {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SendRandomEventV2(events=[→"HATCH", →"JUMP"], weights=[0.7, 0.3], trackingInts=[var "Hatch Count", var "Jump Count"], eventMax=[3, 2])
    on HATCH → Hatched Amount
    on JUMP → Aim Either
  }

  state Hatch Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Birth", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Royal_Zombie_attack_01 (sharedassets84.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Check Dir
  }

  state Spot Choice {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hive_zombie_give_birth (sharedassets384.assets))
    SetBoolValue(boolVariable=var "Doubled", boolValue=false, everyFrame=false)
    SendRandomEvent(events=[→"SPOT 1", →"SPOT 2"], weights=[1, 1], delay=0)
    on SPOT 1 → Spot 1
    on SPOT 2 → Spot 2
  }

  state Check Dir {
    SetFloatValue(floatVariable=var "B1 Angle Min", floatValue=115, everyFrame=false)
    SetFloatValue(floatVariable=var "B1 Angle Max", floatValue=165, everyFrame=false)
    SetFloatValue(floatVariable=var "B2 Angle Min", floatValue=15, everyFrame=false)
    SetFloatValue(floatVariable=var "B2 Angle Max", floatValue=65, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Self X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignTest(floatValue=var "Self X Scale", isPositive=→"FINISHED", isNegative=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "B2 Angle Min", floatValue=115, everyFrame=false)
    SetFloatValue(floatVariable=var "B2 Angle Max", floatValue=165, everyFrame=false)
    SetFloatValue(floatVariable=var "B1 Angle Min", floatValue=15, everyFrame=false)
    SetFloatValue(floatVariable=var "B1 Angle Max", floatValue=65, everyFrame=false)
    on FINISHED → Spot Choice
  }

  state Spot 1 {
    FlingObjectsFromGlobalPool(gameObject=Spatter Honey (sharedassets384.assets), spawnPoint=var "Birth Spot 1", position=(unset), spawnMin=5, spawnMax=6, speedMin=15, speedMax=20, angleMin=var "B1 Angle Min", angleMax=var "B1 Angle Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    GetRandomChild(gameObject=var "Cage", storeResult=var "Hatchling")
    GameObjectIsNull(gameObject=var "Hatchling", isNull=→"CANCEL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hatchling"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "Hatchling", fsmName="Control", variableName="Owner", setValue=var "Self", everyFrame=false)
    GetPosition(gameObject=var "Birth Spot 1", vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Hatchling", vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    FlingObject(flungObject=var "Hatchling", speedMin=15, speedMax=20, angleMin=var "B1 Angle Min", angleMax=var "B1 Angle Max")
    IntAdd(intVariable=var "Spawned", add=1, everyFrame=false)
    on FINISHED → Anim End
    on CANCEL → Anim End
  }

  state Spot 2 {
    FlingObjectsFromGlobalPool(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Birth Spot 2", position=(unset), spawnMin=5, spawnMax=6, speedMin=15, speedMax=20, angleMin=var "B2 Angle Min", angleMax=var "B2 Angle Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    GetRandomChild(gameObject=var "Cage", storeResult=var "Hatchling")
    GameObjectIsNull(gameObject=var "Hatchling", isNull=→"CANCEL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hatchling"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "Hatchling", fsmName="Control", variableName="Owner", setValue=var "Self", everyFrame=false)
    GetPosition(gameObject=var "Birth Spot 2", vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Hatchling", vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    FlingObject(flungObject=var "Hatchling", speedMin=15, speedMax=20, angleMin=var "B2 Angle Min", angleMax=var "B2 Angle Max")
    IntAdd(intVariable=var "Spawned", add=1, everyFrame=false)
    on FINISHED → Anim End
    on CANCEL → Anim End
  }

  state Jump Antic {
    SetBoolValue(boolVariable=var "On Ground", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Ground Check Obj", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    BoolTest(boolVariable=var "On Ground", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Jump Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Jump Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Jump
    on CANCEL → Cancel Frame
  }

  state Jump {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_jump (sharedassets34.assets))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    RandomFloat(min=28, max=30, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Air
  }

  state In Air {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=(unset), everyFrame=true)
    on LAND → Land
  }

  state Aim Either {
    RayCast2d(fromGameObject=Self, fromPosition=(unset), direction=(0, 1), space=1, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=→"CANCEL", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    SendRandomEvent(events=[→"LEFT", →"RIGHT"], weights=[1, 1], delay=0)
    on LEFT → Aim Left
    on RIGHT → Aim Right
    on CANCEL → Hatched Amount
  }

  state Land {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hive_zombie_land (sharedassets384.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Hatch?
  }

  state Aim Left {
    RandomFloat(min=-8, max=-2, storeResult=var "Jump X")
    SetFloatValue(floatVariable=var "Jump Scale", floatValue=1, everyFrame=false)
    on FINISHED → Jump Antic
  }

  state Aim Right {
    RandomFloat(min=2, max=8, storeResult=var "Jump X")
    SetFloatValue(floatVariable=var "Jump Scale", floatValue=-1, everyFrame=false)
    on FINISHED → Jump Antic
  }

  state Back {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Self X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignTest(floatValue=var "Self X Scale", isPositive=→"LEFT", isNegative=→"RIGHT", everyFrame=false)
    on LEFT → Aim Left
    on RIGHT → Aim Right
  }

  state Foreward {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Self X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignTest(floatValue=var "Self X Scale", isPositive=→"RIGHT", isNegative=→"LEFT", everyFrame=false)
    on LEFT → Aim Left
    on RIGHT → Aim Right
  }

  state Hatched Amount {
    IntCompare(integer1=var "Spawned", integer2=var "Hatched Max", equal=→"CANCEL", lessThan=→"FINISHED", greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Hatch Antic
    on CANCEL → Run
  }

  state Hatch? {
    SendRandomEvent(events=[→"HATCH", →"CANCEL"], weights=[1, 1], delay=0)
    on HATCH → Hatched Amount
    on CANCEL → Run
  }

  state Anim End {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Run
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Run
  }

  state Back High? {
    BoolTest(boolVariable=var "Terrain Back High", isTrue=→"BACK", isFalse=→"FOREWARD", everyFrame=false)
    on FOREWARD → Foreward
    on BACK → Back
  }

  state Fore High? {
    BoolTest(boolVariable=var "Terrain Forward High", isTrue=→"FOREWARD", isFalse=→"BACK", everyFrame=false)
    on FOREWARD → Foreward
    on BACK → Back
  }

  state Unalert {
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    on FINISHED → Idle
  }

  state Friendly? {
    GetPlayerDataBool(boolName="equippedCharm_29", storeValue=var "Friendly")
    BoolTest(boolVariable=var "Friendly", isTrue=(none), isFalse=→"HOSTILE", everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on HOSTILE → Idle
    on TOOK DAMAGE → Startle
  }

  var B1 Angle Max: float = 0
  var B1 Angle Min: float = 0
  var B2 Angle Max: float = 0
  var B2 Angle Min: float = 0
  var Jump Scale: float = 0
  var Jump X: float = 0
  var Jump Y: float = 0
  var Self X Scale: float = 0
  var Spawn X: float = 0
  var Spawn Y: float = 0
  var Hatch Count: int = 0
  var Hatched Max: int = 5
  var Jump Count: int = 0
  var Spawned: int = 0
  var Alert Range: bool = false
  var Doubled: bool = false
  var Facing Right: bool = false
  var Friendly: bool = false
  var On Ground: bool = false
  var Terrain Back: bool = false
  var Terrain Back High: bool = false
  var Terrain Forward High: bool = false
  var Unalert Range: bool = false
  var Birth Spot 1: gameObject = Zombie Hive (2)/Birth Spot 1
  var Birth Spot 2: gameObject = Zombie Hive (2)/Birth Spot 2
  var Cage: gameObject = <null>
  var Centre: gameObject = Zombie Hive (2)/Centre
  var FootstepAudio: gameObject = Zombie Hive (2)/FootstepAudio
  var Ground Check Obj: gameObject = <null>
  var Hatchling: gameObject = <null>
  var Run Check Obj: gameObject = <null>
  var Self: gameObject = <null>
}
