# Stag Control
# 1 location(s):
#   Cliffs_03 / Stag

fsm Stag Control {
  start Pause
  on CONVO START → Convo  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetFsmBool(gameObject=Self, fsmName="npc_control", variableName="Can Talk", setValue=false, everyFrame=false)
    FindGameObject(objectName="", withTag="Player", store=var "Hero")
    FindGameObject(objectName="", withTag="Stag Grate", store=var "Grate")
    FindChild(gameObject=Self, childName="Travel Range", storeResult=var "Travel Range")
    FindChild(gameObject=Self, childName="Arrival Dust", storeResult=var "Arrival Dust")
    FindChild(gameObject=Self, childName="Arrival Anim", storeResult=var "Arrival Anim")
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    FindChild(gameObject=var "Self", childName="Prompt Marker Trv", storeResult=var "Prompt Marker")
    SetStringValue(stringVariable=var "Prompt Name", stringValue="Travel", everyFrame=false)
    on FINISHED → Reset HUD and Flower Check
  }

  state Check if Station Opened {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "Station Opened Bool", isTrue=→"OPENED", isFalse=→"CLOSED")
    on OPENED → Station Opened
    on CLOSED → Station Closed
  }

  state Station Closed {
    SetBoolValue(boolVariable=var "Station Opened", boolValue=false, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="npc_control", variableName="Can Talk", setValue=false, everyFrame=false)
    on FINISHED → Away
  }

  state Station Opened {
    DestroyObject(gameObject=var "Grate", delay=0, detachChildren=false)
    SetBoolValue(boolVariable=var "Station Opened", boolValue=true, everyFrame=false)
    on FINISHED → Check Position
  }

  state Check Position {
    GetPlayerDataInt(gameObject=var "GameManager", intName="stagPosition", storeValue=var "Current Position")
    IntCompare(integer1=var "Current Position", integer2=var "Station Position Number", equal=→"PRESENT", lessThan=→"AWAY", greaterThan=→"AWAY", everyFrame=false)
    on PRESENT → Activate
    on AWAY → Away
  }

  state Away {
    SetMeshRenderer(gameObject=Self, active=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Can Talk
  }

  state Opened? {
    SetPlayerDataInt(intName="stagPosition", value=var "Station Position Number")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "Station Opened Bool", isTrue=→"OPENED", isFalse=→"CLOSED")
    on OPENED → Arrive Pause
    on CLOSED → Open Grate
  }

  state Open Grate {
    SetPlayerDataBool(boolName=var "Station Opened Bool", value=true)
    SetBoolValue(boolVariable=var "Station Opened", boolValue=true, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Grate", clipName="Grate_disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Remove Grate
  }

  state Arrive Pause {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=stag_arrive (sharedassets9.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetIntValue(intVariable=var "Current Position", intValue=var "Station Position Number", everyFrame=false)
    SetPlayerDataInt(intName="stagPosition", value=var "Station Position Number")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Arrive Anim
  }

  state Arrive Anim {
    SetMeshRenderer(gameObject=var "Arrival Anim", active=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Arrival Anim", clipName="Arrive", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetMeshRenderer(gameObject=var "Arrival Dust", active=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Arrival Dust", clipName="Arrive_smoke", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Shaking
  }

  state Remove Grate {
    DestroyObject(gameObject=var "Grate", delay=0, detachChildren=false)
    on FINISHED → Arrive Pause
  }

  state Activate {
    SetMeshRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Arrival Dust", active=false)
    SetMeshRenderer(gameObject=var "Arrival Anim", active=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → Activate 2
  }

  state Idle {
    SetFsmBool(gameObject=Self, fsmName="npc_control", variableName="Can Talk", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)  // disabled
    HidePromptMarker(storedObject=var "Prompt")
    on IN RANGE → Show Prompt
  }

  state In Range {
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="UP", delay=0, everyFrame=false)  // disabled
    ListenForDown(eventTarget=Self, wasPressed=→"UP PRESSED", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)

    on OUT OF RANGE → Idle
    on UP PRESSED → Can Talk?
  }

  state Activate 2 {
    ActivateGameObject(gameObject=var "Travel Range", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Travel Range", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Met Before?
  }

  state Met Before? {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metStag", isTrue=→"FINISHED", isFalse=(none))
    SetFsmString(gameObject=var "Prompt", fsmName="Prompt Control", variableName="Prompt Name", setValue="Listen", everyFrame=false)  // disabled
    SetStringValue(stringVariable=var "Prompt Name", stringValue="Listen", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="CHANGE LABEL", delay=0, everyFrame=false)  // disabled
    on FINISHED → Start Audio
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Range
  }

  state Can Talk? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="CanInput", parameters=[], storeResult=var "Can Input")
    BoolTest(boolVariable=var "Can Input", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["onGround"], storeResult=var "On Ground")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["attacking"], storeResult=var "Attacking")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["upAttacking"], storeResult=var "Attacking Up")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["downAttacking"], storeResult=var "Attacking Down")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["dashing"], storeResult=var "Dashing")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetState", parameters=["backDashing"], storeResult=var "Backdashing")
    BoolTest(boolVariable=var "On Ground", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    BoolAnyTrue(boolVariables=[var "Attacking", var "Attacking Down", var "Attacking Up", var "Backdashing", var "Dashing"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Take Control
  }

  state Take Control {
    AudioStop(gameObject=Self)
    SetBoolValue(boolVariable=var "Talking", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Ranges Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var))
    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="DOWN", delay=0, everyFrame=false)  // disabled
    HidePromptMarker(storedObject=var "Prompt")
    SetStringValue(stringVariable=var "Prompt Name", stringValue="Travel", everyFrame=false)
    on FINISHED → Check Proximity
  }

  state Meeting {

    SendEventByName(eventTarget=GameObject(var "Prompt"), sendEvent="CHANGE LABEL", delay=0, everyFrame=false)
    on FINISHED → Open map
  }

  state Meeting End {
    SetFsmBool(gameObject=var "Self", fsmName="advance_conversation", variableName="Conversing", setValue=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Travel Range", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Talking", boolValue=false, everyFrame=false)

    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    on FINISHED → Start Audio
  }

  state Check Proximity {
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)

    SetFloatValue(floatVariable=var "Left Prox", floatValue=var "Self X", everyFrame=false)
    FloatSubtract(floatVariable=var "Left Prox", subtract=var "Move To Offset", everyFrame=false, perSecond=false)
    SetFloatValue(floatVariable=var "Right Prox", floatValue=var "Self X", everyFrame=false)
    FloatAdd(floatVariable=var "Right Prox", add=var "Move To Offset", everyFrame=false, perSecond=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Self X", upperValue=var "Right Prox", boolVariable=(unset), trueEvent=→"MOVE RIGHT", falseEvent=(none), everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Left Prox", upperValue=var "Self X", boolVariable=(unset), trueEvent=→"MOVE LEFT", falseEvent=(none), everyFrame=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Hero X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Hero Facing Right", greaterThanBool=var "Hero Facing Left", everyFrame=false)
    FloatTestToBool(float1=var "Hero X", float2=var "Self X", tolerance=0, equalBool=(unset), lessThanBool=var "Hero On Left", greaterThanBool=var "Hero On Right", everyFrame=false)
    BoolAllTrue(boolVariables=[var "Hero Facing Right", var "Hero On Right"], sendEvent=→"TURN LEFT", storeResult=(unset), everyFrame=false)
    BoolAllTrue(boolVariables=[var "Hero Facing Left", var "Hero On Left"], sendEvent=→"TURN RIGHT", storeResult=(unset), everyFrame=false)
    on TURN LEFT → Turn Hero Left
    on TURN RIGHT → Turn Hero Right
    on FINISHED → Convo?
  }
