# Conversation Control
# 1 location(s):
#   Room_spider_small / Brumm Torch NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REMOVE DREAM", delay=0, everyFrame=false)
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Active
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="IGOR", everyFrame=false)
    on FINISHED → Convo 1
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Convo 1 {
    GetPlayerDataInt(gameObject=var "GameManager", intName="flamesCollected", storeValue=var "Flames Collected")
    GetPlayerDataInt(gameObject=var "GameManager", intName="flamesRequired", storeValue=var "Flames Required")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotBrummsFlame", isTrue=→"REPEAT", isFalse=(none))
    IntCompare(integer1=var "Flames Collected", integer2=var "Flames Required", equal=→"NO FLAME", lessThan=(none), greaterThan=→"NO FLAME", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRUMM_DEEPNEST_1", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_talk_01 (sharedassets305.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down 3
    on REPEAT → Repeat
    on NO FLAME → NF Convo 1
  }

  state Check Active {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_40", isTrue=(none), isFalse=→"INACTIVE")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="defeatedNightmareGrimm", isTrue=→"INACTIVE", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotBrummsFlame", isTrue=→"INACTIVE", isFalse=(none))
    GetPlayerDataInt(gameObject=var "GameManager", intName="grimmChildLevel", storeValue=var "Grimmchild Level")
    IntCompare(integer1=var "Grimmchild Level", integer2=3, equal=(none), lessThan=→"INACTIVE", greaterThan=→"INACTIVE", everyFrame=false)
    on FINISHED → Idle
    on INACTIVE → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRUMM_DEEPNEST_REPEAT", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_talk_03 (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk Finish
  }

  state Convo 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRUMM_DEEPNEST_2", "CP2"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Torch Look")  // disabled
    on CONVO_FINISH → Convo 3
  }

  state Convo 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRUMM_DEEPNEST_3", "CP2"], storeResult=(unset var))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Torch Return")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BRUMM END", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_talk_02 (sharedassets305.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Box Down 2
  }

  state Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN NO HUD", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Give Antic
  }

  state Give Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Torch Give")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.06, everyFrame=false)
    SetPosition(gameObject=var "Flame", vector=(unset), x=-1.05, y=0.22, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=flamebearer_fire_spiral_prepare (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_flame_give_short (sharedassets305.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Give
  }

  state Give {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Torch End")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Flame", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Explode Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=flamebearer_fire_spiral_cast (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Flame Ring
  }

  state Flame Ring {
    FindGameObject(objectName="", withTag="Grimmchild", store=var "Grimmchild")
    SetParent(gameObject=var "Get Flame", parent=var "Grimmchild", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetPositionToObject(gameObject=var "Get Flame", targetObject=var "Grimmchild", xOffset=0, yOffset=0, zOffset=0)
    SetPosition(gameObject=var "Get Flame", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    PlayParticleEmitter(gameObject=var "Get Flame", emit=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=nightmare_lantern_powering_up (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Get
  }

  state Get {
    SendMessage(gameObject=var "Grimmchild", delivery=SendMessage, options=1, functionCall=FlashGrimmflame())
    SetParent(gameObject=var "Get Flash", parent=var "Grimmchild", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    ActivateGameObject(gameObject=var "Get Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPositionToObject(gameObject=var "Get Flash", targetObject=var "Grimmchild", xOffset=0, yOffset=0, zOffset=0)
    SetPosition(gameObject=var "Get Flash", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="flamesCollected")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Relic Get Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg")
    FindChild(gameObject=var "Msg", childName="Text", storeResult=var "Msg Text")
    FindChild(gameObject=var "Msg", childName="Icon", storeResult=var "Msg Icon")
    SetSpriteRendererSprite(gameObject=var "Msg Icon", sprite=charm_grimmkin_01_upgrade (resources.assets))
    GetLanguageString(sheetName="UI", convName="FLAME_COLLECTED", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Grimmchild", audioClip=nightmare_lantern_flame_burst (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=0.75, delay=0, storePlayer=<null>)
    SetPlayerDataBool(boolName="gotBrummsFlame", value=true)
    Wait(time=2.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 2
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo 4
  }

  state Convo 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRUMM_DEEPNEST_4", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_talk_03 (sharedassets236.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → Talk Finish
  }

  state Box Down 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN NO HUD", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BRUMM SHOULDER", delay=0, everyFrame=false)
    on FINISHED → Box Up 3
  }

  state Box Up 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Torch Look")
    on FINISHED → Convo 2
  }

  state NF Convo 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["BRUMM_DEEPNEST_NF_1", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Brumm_talk_01 (sharedassets305.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataBool(boolName="gotBrummsFlame", value=true)
    on CONVO_FINISH → Talk Finish
  }

  var Flames Collected: int = 0
  var Flames Required: int = 0
  var Grimmchild Level: int = 0
  var Activated: bool = false
  var Item Name: string = ""
  var Area Title: gameObject = <null>
  var Explode Effects: gameObject = Brumm Torch NPC/Explode Effects
  var Flame: gameObject = Brumm Torch NPC/Flame
  var Get Flame: gameObject = Brumm Torch NPC/Get Flame
  var Get Flash: gameObject = Brumm Torch NPC/Get Flash
  var Grimmchild: gameObject = <null>
  var Msg: gameObject = <null>
  var Msg Icon: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Self: gameObject = <null>
}
