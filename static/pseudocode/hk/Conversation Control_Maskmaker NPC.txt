# Conversation Control
# 1 location(s):
#   Room_Mask_Maker / Maskmaker NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Check Direction
  }

  state Mask1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_GREET", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="maskmakerMet", value=true)
    on CONVO_FINISH → Talk Finish
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Voice Loop", storeResult=var "Voice Loop")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpEnd")
    on FINISHED → Check Direction 2
  }

  state Check Direction {
    BoolTest(boolVariable=var "Unmasked", isTrue=(none), isFalse=→"MASKED", everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Talk R
    on LEFT → Talk L
    on MASKED → Hero Look
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk Right")
    on FINISHED → Hero Look
  }

  state Talk L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk Left")
    on FINISHED → Hero Look
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="MASKMAKER", everyFrame=false)
    on FINISHED → Unmasked?
  }

  state Check Direction 2 {
    BoolTest(boolVariable=var "Unmasked", isTrue=(none), isFalse=→"MASKED", everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Face Right
    on LEFT → Face Left
    on MASKED → End
  }

  state Face Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Return Right")
    on FINISHED → End
  }

  state Face Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Return Left")
    on FINISHED → End
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Unmasked? {
    BoolTest(boolVariable=var "Unmasked", isTrue=→"UNMASKED", isFalse=→"MASKED", everyFrame=false)
    on MASKED → Mask Choice
    on UNMASKED → Unmask Choice
  }

  state Mask Choice {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="maskmakerMet", isTrue=(none), isFalse=→"MASK1")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="maskmakerConvo1", isTrue=(none), isFalse=→"MASK2")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="maskmakerConvo2", isTrue=(none), isFalse=→"MASK3")
    on MASK1 → Mask1
    on MASK2 → Mask 2
    on MASK3 → Mask 3
    on FINISHED → Mask 4
  }

  state Mask 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_REPEAT", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="maskmakerConvo1", value=true)
    on CONVO_FINISH → Talk Finish
  }

  state Mask 3 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_REPEAT2", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="maskmakerConvo2", value=true)
    on CONVO_FINISH → Talk Finish
  }

  state Mask 4 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_REPEAT3", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Unmask Choice {
    AudioStop(gameObject=var "Voice Loop")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Mask_Maker_unmask_talk_01 (sharedassets21.assets), Mask_Maker_unmask_talk_02 (sharedassets21.assets), Mask_Maker_unmask_talk_03 (sharedassets21.assets), Mask_Maker_unmask_talk_04 (sharedassets21.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasShadowDash", falseBool="maskmakerShadowDash", isTrue=→"SHADOW DASH", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasKingsBrand", falseBool="maskmakerKingsBrand", isTrue=→"KINGS BRAND", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="maskmakerUnmasked1", isTrue=(none), isFalse=→"UNMASK 1")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="maskmakerUnmasked2", isTrue=(none), isFalse=→"UNMASK 2")
    on UNMASK 1 → Unmask 1
    on UNMASK 2 → Unmask 2
    on KINGS BRAND → Kings Brand
    on SHADOW DASH → Shadow Dash
    on FINISHED → Unmask Repeat
  }

  state Unmask 1 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_UNMASK", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="maskmakerUnmasked1", value=true)
    on CONVO_FINISH → Play Audio
  }

  state Unmask 2 {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_UNMASK3", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="maskmakerUnmasked2", value=true)
    on CONVO_FINISH → Play Audio
  }

  state Kings Brand {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_UNMASK4", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="maskmakerKingsBrand", value=true)
    on CONVO_FINISH → Play Audio
  }

  state Shadow Dash {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_UNMASK2", "Minor NPC"], storeResult=(unset var))
    SetPlayerDataBool(boolName="maskmakerShadowDash", value=true)
    on CONVO_FINISH → Play Audio
  }

  state Unmask Repeat {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["MASK_MAKER_UNMASK_REPEAT", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Play Audio
  }

  state Play Audio {
    AudioPlaySimple(gameObject=var "Voice Loop", volume=1, oneShotClip=<null>)
    on FINISHED → Talk Finish
  }

  var Unmasked: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Loop: gameObject = <null>
  var Voice Player: gameObject = <null>
}
