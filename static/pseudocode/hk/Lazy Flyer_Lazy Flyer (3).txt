# Lazy Flyer
# 1 location(s):
#   Fungus3_50 / Lazy Flyer (3)

fsm Lazy Flyer {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Left X", floatValue=var "Start X", everyFrame=false)
    SetFloatValue(floatVariable=var "Right X", floatValue=var "Start X", everyFrame=false)
    FloatAdd(floatVariable=var "Right X", add=var "Flight Distance", everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Left X", subtract=var "Flight Distance", everyFrame=false, perSecond=false)
    SendRandomEvent(events=[→"LEFT", →"RIGHT"], weights=[1, 1], delay=0)
    on LEFT → Left
    on RIGHT → Right
  }

  state Left {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=-3, max=-4, storeResult=var "Fly Speed")
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    on FINISHED → Fly
  }

  state Fly {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Fly Speed", y=(unset), everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect Turn
  }

  state Turn Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.95)
    on FINISHED → Turn
  }

  state Turn Pause {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Turn Anim
  }

  state Turn {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    on LEFT → Left
    on RIGHT → Right
  }

  state Right {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=3, max=4, storeResult=var "Fly Speed")
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    on FINISHED → Fly
  }

  state Detect Turn {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Left X", tolerance=0, equal=(none), lessThan=→"TURN", greaterThan=(none), everyFrame=true)
    FloatCompare(float1=var "Self X", float2=var "Right X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"TURN", everyFrame=true)
    on TURN → Turn Pause
  }

  var Flight Distance: float = 9

  var Fly Speed: float = 0
  var Left X: float = 0
  var Right X: float = 0
  var Self X: float = 0
  var Start X: float = 0
  var Facing Right: bool = false
}
