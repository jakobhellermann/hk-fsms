// component disabled: this FSM does not run until something enables it
fsm Hit Crystals {
  start Idle

  state Idle {
    SetParticleEmission(gameObject=var "Hit Crystals", emission=false)
    on BLOCKED HIT → Fire
  }

  state Fire {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmission(gameObject=var "Hit Crystals", emission=true)
    on FINISHED → Idle
  }

  var Hit Crystals: gameObject = Mines Crawler (1)/Hit Crystals Effect
}
