# enemy_dreamnail_reaction
# 1 location(s):
#   sharedassets33.assets / Grub Mimic Bottle Col/Grub Mimic Top

// uses template: enemy_dreamnail_reaction
fsm enemy_dreamnail_reaction {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Inactive", isTrue=→"INACTIVE", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on INACTIVE → Inactive
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"DREAM IMPACT", storeCollider=var "Collider")  // disabled
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer=Attack(17), sendEvent=→"DREAM IMPACT", storeCollider=var "Collider")  // disabled
    on DREAM IMPACT → Set SP
  }

  state Get Hit {
    SetFsmFloat(gameObject=Self, fsmName="recoil", variableName="Attack Magnitude", setValue=2, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashDreamImpact())
    SendEventByScale(gameObject=var "Hero", eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on L → Recoil L
    on R → Recoil R
  }

  state Recoil L {
    SetFsmFloat(gameObject=Self, fsmName="recoil", variableName="Attack Direction", setValue=180, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="RECOIL L", delay=0, everyFrame=false)
    on FINISHED → Recoil
  }

  state Recoil R {
    SetFsmFloat(gameObject=Self, fsmName="recoil", variableName="Attack Direction", setValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="RECOIL R", delay=0, everyFrame=false)
    on FINISHED → Recoil
  }

  state Recoil {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Get Soul {
    SendMessageV2(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AddMPCharge(var "MP Charge"), everyFrame=false)
    on FINISHED → Send Msg
  }

  state Set SP {
    SetIntValue(intVariable=var "MP Charge", intValue=33, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_30", isTrue=(none), isFalse=→"FINISHED")
    SetIntValue(intVariable=var "MP Charge", intValue=66, everyFrame=false)
    on FINISHED → Get Soul
  }

  state Send Msg {
    SetFsmInt(gameObject=var "Enemy Dream Msg", fsmName="Display", variableName="Convo Amount", setValue=var "Convo Amount", everyFrame=false)
    SetFsmString(gameObject=var "Enemy Dream Msg", fsmName="Display", variableName="Convo Title", setValue=var "Convo Title", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemy Dream Msg"), sendEvent="DISPLAY ENEMY DREAM", delay=0, everyFrame=false)
    on FINISHED → Get Hit
  }

  state Inactive {
    on DREAM REACTION ACTIVE → Idle
  }

  var Convo Amount: int = 3
  var Start Inactive: bool = false
  var Convo Title: string = "GRUB_MIMIC"

  var X Scale: float = 0
  var MP Charge: int = 0
  var Collider: gameObject = <null>
  var Self: gameObject = <null>
}
