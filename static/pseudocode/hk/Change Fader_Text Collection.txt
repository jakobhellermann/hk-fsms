# Change Fader

fsm Change Fader {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    on CHANGE UP → Delay
    on CHANGE DOWN → Down
  }

  state Up {
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))
    SetColorValue(colorVariable=var "New Colour", color=var "Prev Colour", everyFrame=false)
    EaseColor(fromValue=var "Prev Colour", toValue=var "Up Colour", colorVariable=var "New Colour", time=var "Up Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    SetTextMeshProColor(gameObject=Self, color=var "New Colour", everyFrame=true)
    on CHANGE DOWN → Down
  }

  state Down {
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Prev Colour", fail=(none))
    SetColorValue(colorVariable=var "New Colour", color=var "Prev Colour", everyFrame=false)
    GetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", fail=(none))
    EaseColor(fromValue=var "Prev Colour", toValue=var "Down Colour", colorVariable=var "New Colour", time=var "Down Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "New Colour", everyFrame=true)
    SetTextMeshProColor(gameObject=Self, color=var "New Colour", everyFrame=true)
    on CHANGE UP → Delay
  }

  state Delay {
    Wait(time=var "Up Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
    on CHANGE DOWN → Down
  }

  var Down Time: float = 0.15
  var Up Delay: float = 0
  var Up Time: float = 0.15
  var Down Colour: color = (1, 1, 1, 0)
  var Up Colour: color = (1, 1, 1, 1)

  var New Colour: color = (0, 0, 0, 1)
  var Prev Colour: color = (1, 1, 1, 1)
  var Self: gameObject = <null>
}
