# Tink
# 1 location(s):
#   Abyss_05 / Dusk Knight/Shield

fsm Tink {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Attack Pt", storeResult=var "Attack Pt")
    FindChild(gameObject=Self, childName="Dream Impact", storeResult=var "Dream Impact")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"DREAM HIT", storeCollider=var "")
    on DREAM HIT → Tink
  }

  state Tink {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Attack Pt", emission=true)
    ActivateGameObject(gameObject=var "Dream Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByScale(gameObject=var "Hero", eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on L → Recoil L
    on R → Recoil R
  }

  state End emission {
    SetParticleEmission(gameObject=var "Attack Pt", emission=false)
    on FINISHED → Idle
  }

  state Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End emission
  }

  state Recoil L {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilLeft())  // disabled
    Translate(gameObject=var "Hero", vector=(unset), x=-1.5, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Pause
  }

  state Recoil R {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilRight())  // disabled
    Translate(gameObject=var "Hero", vector=(unset), x=1.5, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Pause
  }

  var Convo Name: string = ""
  var Sheet Name: string = ""

  var Attack Pt: gameObject = Dream Dialogue/Active Pt (sharedassets6.assets)
  var Dream Impact: gameObject = <null>
  var Impact Pt: gameObject = Dream Dialogue/Impact Pt (sharedassets6.assets)
  var Self: gameObject = <null>
}
