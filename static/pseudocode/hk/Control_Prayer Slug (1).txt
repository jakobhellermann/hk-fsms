# Control
# 1 location(s):
#   Crossroads_10_boss_defeated / Prayer Room/Prayer Slug (1)

// component disabled: this FSM does not run until something enables it
fsm Control {
  start Init
  on DREAM WAKE → Startle  // from any state
  on ZERO HP → Death Clip  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Pause
  }

  state Idle Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Up")
    WaitRandom(timeMin=0, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on WAKE → Startle
    on FINISHED → To Down
  }

  state Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    AudioStop(gameObject=Self)
    SetAudioClip(gameObject=Self, audioClip=flip_hopper_crawl_loop (sharedassets49.assets))
    AudioPlayRandom(gameObject=var "Self", audioClips=[Maggot_Startle_01 (sharedassets49.assets), Maggot_Startle_02 (sharedassets49.assets), Maggot_Startle_03 (sharedassets49.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)  // disabled
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Maggot_Startle_01 (sharedassets49.assets), Maggot_Startle_02 (sharedassets49.assets), Maggot_Startle_03 (sharedassets49.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Direction
  }

  state Run R {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=3.5, max=5.5, storeResult=var "Max Speed R")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Velocity", add=var "Accel", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Velocity", minValue=var "Max Speed L", maxValue=var "Max Speed R", everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=true)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"LAND", bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on RIGHT → Run L
    on LAND → Bump R
  }

  state Direction {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    on LEFT → Run R
    on RIGHT → Run L
  }

  state Run L {
    RandomFloat(min=-5.5, max=-3.5, storeResult=var "Max Speed L")
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=true)
    FloatSubtract(floatVariable=var "Velocity", subtract=var "Accel", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Velocity", minValue=var "Max Speed L", maxValue=var "Max Speed R", everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=true)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=→"LAND", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on LEFT → Run R
    on LAND → Bump L
  }

  state Bump R {
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=health_bug_wall_hit (sharedassets6.assets), pitchMin=0.9, pitchMax=1.1, volume=0.5, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Maggot_bounce_01 (sharedassets49.assets), Maggot_bounce_02 (sharedassets49.assets), Maggot_bounce_03 (sharedassets49.assets), Maggot_bounce_04 (sharedassets49.assets), Maggot_bounce_05 (sharedassets49.assets), Maggot_bounce_06 (sharedassets49.assets), Maggot_bounce_07 (sharedassets49.assets)], weights=[1, 1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    RandomFloat(min=-2.5, max=-5, storeResult=var "Chooser")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Chooser", y=10, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Run R
  }

  state Bump L {
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=health_bug_wall_hit (sharedassets6.assets), pitchMin=0.9, pitchMax=1.1, volume=0.5, delay=0, storePlayer=<null>)
    RandomFloat(min=2.5, max=5, storeResult=var "Chooser")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Chooser", y=10, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Maggot_bounce_01 (sharedassets49.assets), Maggot_bounce_02 (sharedassets49.assets), Maggot_bounce_03 (sharedassets49.assets), Maggot_bounce_04 (sharedassets49.assets), Maggot_bounce_05 (sharedassets49.assets), Maggot_bounce_06 (sharedassets49.assets), Maggot_bounce_07 (sharedassets49.assets)], weights=[1, 1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bounce")  // disabled
    AudioPlayRandom(gameObject=var "Self", audioClips=[Maggot_bounce_01 (sharedassets49.assets), Maggot_bounce_02 (sharedassets49.assets), Maggot_bounce_03 (sharedassets49.assets), Maggot_bounce_04 (sharedassets49.assets), Maggot_bounce_05 (sharedassets49.assets), Maggot_bounce_06 (sharedassets49.assets), Maggot_bounce_07 (sharedassets49.assets)], weights=[1, 1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1)  // disabled
    on FINISHED → Run L
  }

  state To Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle ToDown")
    WaitRandom(timeMin=1.5, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on WAKE → Startle
    on FINISHED → To Up
  }

  state To Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle ToUp")
    WaitRandom(timeMin=1.5, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on WAKE → Startle
    on FINISHED → To Down
  }

  state Check Dream Return {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=10, tolerance=0, equal=(none), lessThan=→"DREAM", greaterThan=(none), everyFrame=false)
    on DREAM → Dream Return
    on FINISHED → Idle Up
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Dream Return
  }

  state Dream Return {
    FindChild(gameObject=Self, childName="Wake Region", storeResult=var "Wake Region")
    ActivateGameObject(gameObject=var "Wake Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)
    Translate(gameObject=Self, vector=(unset), x=2, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Idle Up
  }

  state Death Clip {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=var "Death Clip", pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
  }

  var Death Clip: object = Maggot_death_03 (sharedassets49.assets)

  var Accel: float = 0.3
  var Chooser: float = 0
  var Distance: float = 0
  var Max Speed L: float = -4.5
  var Max Speed R: float = 4.5
  var Velocity: float = 0
  var left: bool = false
  var right: bool = false
  var Wake Anim: string = ""
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
  var Wake Region: gameObject = <null>
}
