# crack control
# 1 location(s):
#   resources.assets / Hit Crack

fsm crack control {
  start Overcharm?

  state State 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hit Crack Appear")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SendRandomEvent(events=[→"FLIP", →"STAY"], weights=[1, 1], delay=0)
    on FLIP → Flip
    on STAY → Stay
  }

  state Flip {
    SetScale(gameObject=Self, vector=(unset), x=-1.90412, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=0, max=27, storeResult=var "Chooser")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)  // disabled
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hit Crack Appear", animationTriggerEvent=(none), animationCompleteEvent=→"CLIP DONE")
    on CLIP DONE → Recycle
  }

  state Stay {
    RandomFloat(min=-3, max=-23, storeResult=var "Chooser")
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hit Crack Appear", animationTriggerEvent=(none), animationCompleteEvent=→"CLIP DONE")
    SetScale(gameObject=Self, vector=(unset), x=1.90412, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on CLIP DONE → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Overcharm? {
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="overcharmed", isTrue=(none), isFalse=→"FINISHED")
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 0.5294118, 0.97385633, 1), everyframe=false)
    on FINISHED → State 1
  }

  var Chooser: float = 0
}
