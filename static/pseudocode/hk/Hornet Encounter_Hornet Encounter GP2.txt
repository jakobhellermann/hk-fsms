# Hornet Encounter
# 1 location(s):
#   Fungus1_10 / Hornet Encounter GP2

fsm Hornet Encounter {
  start Init

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="hornetGreenpath", storeValue=var "PlayerData")
    IntCompare(integer1=var "PlayerData", integer2=1, equal=(none), lessThan=(none), greaterThan=→"DESTROY", everyFrame=false)
    on FINISHED → Idle
    on DESTROY → Destroy
  }

  state Idle {
    BoolTest(boolVariable=var "Evade Range", isTrue=→"EVADE", isFalse=(none), everyFrame=true)
    on EVADE → Leave
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Leave {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetVelocity2d(gameObject=Self, vector=(unset), x=-10, y=(unset), everyFrame=false)
    Wait(time=5, finishEvent=→"DESTROY", realTime=false)
    SetPlayerDataInt(intName="hornetGreenpath", value=2)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on DESTROY → Destroy
  }

  var PlayerData: int = 0
  var Evade Range: bool = false
  var Turn Range: bool = false
  var CamLock: gameObject = <null>
  var Grass: gameObject = <null>
}
