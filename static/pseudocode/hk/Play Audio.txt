# Play Audio
# 1 location(s):
#   resources.assets / Hollow Shade:0

fsm Play Audio {
  start Pause

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Up
  }

  state Fade Up {
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=1)
  }
}
