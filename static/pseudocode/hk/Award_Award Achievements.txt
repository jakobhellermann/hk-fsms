# Award
# 1 location(s):
#   End_Game_Completion / Award Achievements

fsm Award {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Steel Soul
  }

  state 100% Completion {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="CountGameCompletion", parameters=[], storeResult=(unset var))
    GetPlayerDataFloat(gameObject=var "GameManager", floatName="completionPercentage", storeValue=var "Completion percentage Float")
    GetPlayerDataInt(gameObject=var "GameManager", intName="permadeathMode", storeValue=var "PermaDeath Mode")
    FloatCompare(float1=var "Completion percentage Float", float2=100, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["COMPLETION"], storeResult=(unset var))
    IntCompare(integer1=var "PermaDeath Mode", integer2=0, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["STEELSOUL_COMPLETION"], storeResult=(unset var))
    on FINISHED → Speedrun
  }

  state Steel Soul {
    GetPlayerDataInt(gameObject=var "GameManager", intName="permadeathMode", storeValue=var "PermaDeath Mode")
    IntCompare(integer1=var "PermaDeath Mode", integer2=0, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["STEELSOUL"], storeResult=(unset var))
    on FINISHED → 100% Completion
  }

  state Speedrun {
    GetPlayerDataFloat(gameObject=var "GameManager", floatName="playTime", storeValue=var "Play Time")
    FloatCompare(float1=var "Play Time", float2=36000, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["SPEEDRUN_1"], storeResult=(unset var))
    FloatCompare(float1=var "Play Time", float2=18000, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["SPEEDRUN_2"], storeResult=(unset var))
    on FINISHED → Speed Completion
  }

  state Speed Completion {
    FloatCompare(float1=var "Play Time", float2=72000, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    GetPlayerDataFloat(gameObject=var "GameManager", floatName="completionPercentage", storeValue=var "Completion percentage Float")
    FloatCompare(float1=var "Completion percentage Float", float2=100, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["SPEED_COMPLETION"], storeResult=(unset var))
    on FINISHED → 112% Completion
  }

  state Done {
  }

  state 112% Completion {
    FloatCompare(float1=var "Completion percentage Float", float2=112, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardAchievement", parameters=["COMPLETIONGG"], storeResult=(unset var))
    on FINISHED → Done
  }

  var Completion percentage Float: float = 0
  var Play Time: float = 0
  var PermaDeath Mode: int = 0
}
