# Mantis Gate
# 1 location(s):
#   Fungus2_14 / Mantis Floor Gate

fsm Mantis Gate {
  start Idle
  on ACTIVATE → Activate  // from any state
  on OPEN → Open  // from any state

  state Idle {
    GetOwner(storeGameObject=var "Self")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate Collider
  }

  state Activate {
    FindGameObject(objectName="Gate Mantis", withTag="Untagged", store=var "Gate Mantis")
    DestroyObject(gameObject=var "Gate Mantis", delay=0, detachChildren=false)
    ActivateGameObject(gameObject=var "Bits", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindGameObject(objectName="Mantis Gate F", withTag="Untagged", store=var "Mantis Gate F")
    SendEventByName(eventTarget=GameObject(var "Mantis Gate F"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
  }

  state Open {
    FindGameObject(objectName="Mantis Gate F", withTag="Untagged", store=var "Mantis Gate F")
    SetCollider(gameObject=Self, active=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Bits", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Rumble
  }

  state Activate Collider {
    SetCollider(gameObject=Self, active=true)
  }

  state Rumble {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=true, everyFrame=false)
    Wait(time=3.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MANTIS FLOOR RUMBLE", delay=0, everyFrame=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on FINISHED → Finish
  }

  state Finish {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="MANTIS FLOOR FINISH", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=misc_rumble_impact (sharedassets13.assets))
    on FINISHED → Open Door
  }

  state Open Door {
    FindGameObject(objectName="Mantis Gate F", withTag="Untagged", store=var "Mantis Gate F")
    SendEventByName(eventTarget=GameObject(var "Mantis Gate F"), sendEvent="OPEN", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Mantis Gate F", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)  // disabled
  }

  var Bits: gameObject = Mantis Floor Gate/Bits
  var Gate Mantis: gameObject = <null>
  var Mantis Gate F: gameObject = <null>
  var Self: gameObject = <null>
}
