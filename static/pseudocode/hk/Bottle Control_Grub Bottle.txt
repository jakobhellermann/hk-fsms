fsm Bottle Control {
  start Pause

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Shatter
  }

  state Return Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Shatter {
    BoolTest(boolVariable=var "Hero Range", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    IncrementPlayerDataInt(gameObject=var "GameManager", intName="grubsCollected")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=CheckGrubAchievements())
    AudioPlayRandom(gameObject=var "Self", audioClips=[crystal_break_1 (resources.assets), crystal_break_2 (resources.assets)], weights=[1, 1], pitchMin=1, pitchMax=1)
    ActivateGameObject(gameObject=var "Phys Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Grub"), sendEvent="FREE", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Debris", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObjects(containerObject=var "Debris", adjustPosition=(unset), randomisePosition=false, speedMin=10, speedMax=15, angleMin=40, angleMax=140)
    on FINISHED → Destroy Self
    on CANCEL → Return Pause
  }

  state Destroy Self {
    SetName(gameObject=Self, name="Collected")
    SetCollider(gameObject=Self, active=false)
    SpawnRandomObjects(gameObject=Bottle Lid (sharedassets33.assets), spawnPoint=var "Self", position=(0, 1.69, 0), spawnMin=1, spawnMax=1, speedMin=7, speedMax=8, angleMin=85, angleMax=95, originVariation=0)
    PlayParticleEmitter(gameObject=var "Pt Glass L", emit=0)
    PlayParticleEmitter(gameObject=var "Pt Glass M", emit=0)
    SetSpriteRenderer(gameObject=Self, active=false)
    on FINISHED → Set Map?
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GRUB SAVED", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false)
    DestroyAllChildren(gameObject=var "Self", disable=false)
    CreateObject(gameObject=Bottle Lid Grounded (sharedassets39.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SetName(gameObject=Self, name="Collected")
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state Set Map? {
    FindGameObject(objectName="Grub Bottle", withTag="Grub Bottle", store=var "Other Grub")
    GameObjectIsNull(gameObject=var "Other Grub", isNull=→"SET", isNotNull=(none), storeResult=(unset), everyFrame=false)
    on SET → Set
  }

  state Set {
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=AddToGrubList())
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    SetParent(gameObject=var "Pool", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    FindChild(gameObject=Self, childName="Grub", storeResult=var "Grub")
    FindChild(gameObject=Self, childName="Debris", storeResult=var "Debris")
    FindChild(gameObject=Self, childName="Bottle Physical", storeResult=var "Phys Box")
    on FINISHED → Idle
    on ACTIVATE → Activate
  }

  var Attack Type: int = 0
  var Activated: bool = false
  var Hero Range: bool = false
  var Damager: gameObject = <null>
  var Debris: gameObject = <null>
  var Grub: gameObject = <null>
  var Other Grub: gameObject = <null>
  var Phys Box: gameObject = <null>
  var Pool: gameObject = <null>
  var Pt Glass L: gameObject = Grub Bottle/Pt Glass L
  var Pt Glass M: gameObject = Grub Bottle/Pt Glass S
  var Self: gameObject = <null>
}
