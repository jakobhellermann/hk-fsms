# Mawlek Turret
# 2 location(s):
#   Abyss_20 / Mawlek Turret Ceiling
#   Abyss_20 / Mawlek Turret Ceiling (1)

// uses template: mawlek_turret
fsm Mawlek Turret {
  start Init

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sleep")
    on HERO ENTER → Active
  }

  state Active {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Active")
    RandomBool(storeResult=var "Fire Left")  // disabled
    BoolFlip(boolVariable=var "Fire Left")
    BoolTest(boolVariable=var "Fire Left", isTrue=→"FIRE LEFT", isFalse=→"FIRE RIGHT", everyFrame=false)
    on HERO EXIT → Idle
    on FIRE LEFT → Fire Left
    on FIRE RIGHT → Fire Right
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Spit Effect", storeResult=var "Spit Effect")
    SpawnObjectFromGlobalPool(gameObject=Shot Mawlek (sharedassets32.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "First Shot")
    ActivateGameObject(gameObject=var "First Shot", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Fire Left {
    AudioPlayRandom(gameObject=var "Self", audioClips=[mawlek_spit (sharedassets33.assets), mawlek_spit_b (sharedassets34.assets)], weights=[1, 1], pitchMin=1, pitchMax=1.15)
    FlingObjectsFromGlobalPool(gameObject=Shot Mawlek (sharedassets32.assets), spawnPoint=var "Self", position=var "Spawn Pos", spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=var "Angle Min L", angleMax=var "Angle Max L", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    SendEventByName(eventTarget=GameObject(var "Spit Effect"), sendEvent="PLAY", delay=0, everyFrame=false)
    WaitRandom(timeMin=0.25, timeMax=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Active
    on HERO EXIT → Idle
  }

  state Fire Right {
    AudioPlayRandom(gameObject=var "Self", audioClips=[mawlek_spit (sharedassets33.assets), mawlek_spit_b (sharedassets34.assets)], weights=[1, 1], pitchMin=1, pitchMax=1.15)
    FlingObjectsFromGlobalPool(gameObject=Shot Mawlek (sharedassets32.assets), spawnPoint=var "Self", position=var "Spawn Pos", spawnMin=1, spawnMax=1, speedMin=var "Shot Speed", speedMax=var "Shot Speed", angleMin=var "Angle Min R", angleMax=var "Angle Max R", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    SendEventByName(eventTarget=GameObject(var "Spit Effect"), sendEvent="PLAY", delay=0, everyFrame=false)
    WaitRandom(timeMin=0.25, timeMax=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Active
    on HERO EXIT → Idle
  }

  var Angle Max L: float = 350
  var Angle Max R: float = 270
  var Angle Min L: float = 270
  var Angle Min R: float = 190
  var Shot Speed: float = 10
  var Spit Angle Max: float = 240
  var Spit Angle Min: float = 300
  var Spawn Pos: vector3 = (0, -1.4, 0)

  var Fire Left: bool = false
  var First Shot: gameObject = <null>
  var Self: gameObject = <null>
  var Spit Effect: gameObject = <null>
}
