fsm Liquid Control {
  start Idle
  on CANT HEAL → Can't Heal  // from any state
  on CAN HEAL → Can Heal  // from any state

  state Idle {
  }

  state Can Heal {
    Tk2dSpriteGetColor(gameObject=Self, color=var "Prev Colour", everyframe=false)
    EaseColor(fromValue=var "Prev Colour", toValue=(1, 1, 1, 1), colorVariable=var "Colour", time=0.2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    on FINISHED → Idle
  }

  state Can't Heal {
    Tk2dSpriteGetColor(gameObject=Self, color=var "Prev Colour", everyframe=false)
    EaseColor(fromValue=var "Prev Colour", toValue=(0.43137255, 0.43137255, 0.43137255, 1), colorVariable=var "Colour", time=0.2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    on FINISHED → Idle
  }

  var Colour: color = (1, 1, 1, 1)
  var Prev Colour: color = (0, 0, 0, 1)
}
