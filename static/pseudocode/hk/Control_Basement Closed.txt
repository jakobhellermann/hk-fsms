# Control
# 1 location(s):
#   Room_shop / Basement Closed

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasAllNailArts", falseBool="gotSlyCharm", isTrue=→"OPENED", isFalse=(none))
    on OPENED → Inactive
  }

  state Inactive {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Self: gameObject = <null>
}
