# Activate
# 1 location(s):
#   Room_Mansion / Heart Piece Folder

fsm Activate {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="xunRewardGiven", isTrue=→"ACTIVATE", isFalse=(none))
    on ACTIVATE → Activate
  }

  state Activate {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=true)
    FindChild(gameObject=Self, childName="Heart Piece", storeResult=var "Heart Piece")
    SetFsmBool(gameObject=var "Heart Piece", fsmName="Heart Container Control", variableName="Appears", setValue=false, everyFrame=false)
  }

  var Heart Piece: gameObject = <null>
  var Self: gameObject = <null>
}
