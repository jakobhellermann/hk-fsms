# Anim
# 1 location(s):
#   Room_Colosseum_01 / Little Fool NPC

fsm Anim {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO R", leftEvent=→"HERO L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on HERO L → Idle L
    on HERO R → Idle R
  }

  state Turn L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Turn L")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO R", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on HERO R → Turn R
    on TALK → Talk L
  }

  state Turn R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Turn R")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"HERO L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on HERO L → Turn L
    on TALK → Talk R
  }

  state Talk L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk L")
    on TALK END → Idle L
  }

  state Idle L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle L")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO R", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on HERO R → Turn R
    on TALK → Talk L
  }

  state Idle R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle R")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"HERO L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=true)
    on HERO L → Turn L
    on TALK → Talk R
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk R")
    on TALK END → Idle R
  }
}
