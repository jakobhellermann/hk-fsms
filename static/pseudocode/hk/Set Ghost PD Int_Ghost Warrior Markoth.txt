# Set Ghost PD Int
# 2 location(s):
#   Deepnest_East_10 / Warrior/Ghost Warrior Markoth
#   Deepnest_East_16 / Warrior/Ghost Warrior Markoth

fsm Set Ghost PD Int {
  start Idle

  state Idle {
    on ZERO HP → Set
  }

  state Set {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="GHOST DEATH", delay=0, everyFrame=false)
    SetPlayerDataInt(intName=var "PD Int", value=1)
  }

  var PD Int: string = "markothDefeated"
}
