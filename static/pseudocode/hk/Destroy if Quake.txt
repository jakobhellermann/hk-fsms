# Destroy if Quake
# 1 location(s):
#   Ruins1_24 / Battle Gate (1)

fsm Destroy if Quake {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="quakeLevel", storeValue=var "Quake Level")
    IntCompare(integer1=var "Quake Level", integer2=1, equal=→"DESTROY", lessThan=(none), greaterThan=→"DESTROY", everyFrame=false)
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Quake Level: int = 0
}
