# Cancel
# 1 location(s):
#   Deepnest_East_12 / Hornet Encounter Outskirts

fsm Cancel {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"NAIL HIT", storeCollider=var "")
    on NAIL HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="SLOPE CANCEL", delay=0, everyFrame=false)
    on FINISHED → Idle
  }
}
