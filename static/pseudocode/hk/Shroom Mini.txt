fsm Shroom Mini {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Scale
  }

  state Anim Pick {
    SendRandomEvent(events=[(none), (none), (none)], weights=[1, 1, 1], delay=0)
    on 1 → 1
    on 2 → 2
    on 3 → 3
  }

  state 1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mini Rest 1")
    SetStringValue(stringVariable=var "Wake Anim", stringValue="Mini Rest to Wake 1", everyFrame=false)
    on FINISHED → Sleep
  }

  state 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mini Rest 2")
    SetStringValue(stringVariable=var "Wake Anim", stringValue="Mini Rest to Wake 2", everyFrame=false)
    on FINISHED → Sleep
  }

  state 3 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mini Rest 3")
    SetStringValue(stringVariable=var "Wake Anim", stringValue="Mini Rest to Wake 3", everyFrame=false)
    on FINISHED → Sleep
  }

  state Sleep {
    on WAKE → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Wake Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[ceiling_dropper_look_1 (sharedassets48.assets), ceiling_dropper_look_2 (sharedassets48.assets), ceiling_dropper_look_3 (sharedassets48.assets), ceiling_dropper_look_4 (sharedassets48.assets), ceiling_dropper_look_5 (sharedassets48.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1.1, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Startle
  }

  state Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Direction
  }

  state Run R {
    AudioPlayInState(gameObject=Self, volume=1)
    SetScale(gameObject=Self, vector=(unset), x=var "Neg Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Velocity", add=var "Accel", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Velocity", minValue=var "Max Speed L", maxValue=var "Max Speed R", everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=true)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"LAND", bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on RIGHT → Run L
    on LAND → Bump R
  }

  state Direction {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    on LEFT → Run R
    on RIGHT → Run L
  }

  state Run L {
    AudioPlayInState(gameObject=Self, volume=1)
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), space=0, everyFrame=true)
    FloatSubtract(floatVariable=var "Velocity", subtract=var "Accel", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Velocity", minValue=var "Max Speed L", maxValue=var "Max Speed R", everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity", y=(unset), everyFrame=true)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=var "right", leftBool=var "left", everyFrame=true)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=→"LAND", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LEFT → Run R
    on LAND → Bump L
  }

  state Bump R {
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=health_bug_wall_hit (sharedassets6.assets), pitchMin=0.9, pitchMax=1.1, volume=0.5, delay=0, storePlayer=<null>)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    RandomFloat(min=-2.5, max=-5, storeResult=var "Chooser")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Chooser", y=10, everyFrame=false)
    on FINISHED → Run R
  }

  state Bump L {
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=health_bug_wall_hit (sharedassets6.assets), pitchMin=0.9, pitchMax=1.1, volume=0.5, delay=0, storePlayer=<null>)
    RandomFloat(min=2.5, max=5, storeResult=var "Chooser")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Chooser", y=10, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Run L
  }

  state Scale {
    RandomFloat(min=0.75, max=1, storeResult=var "Scale")
    FloatOperator(float1=var "Scale", float2=-1, operation=Multiply, storeResult=var "Neg Scale", everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    SendRandomEvent(events=[→"NEG", →"FINISHED"], weights=[1, 1], delay=0)
    on FINISHED → Anim Pick
    on NEG → Neg Scale
  }

  state Neg Scale {
    SetScale(gameObject=Self, vector=(unset), x=var "Neg Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Anim Pick
  }

  var Accel: float = 0.13
  var Chooser: float = 0
  var Max Speed L: float = -4.5
  var Max Speed R: float = 4.5
  var Neg Scale: float = 0
  var Scale: float = 0
  var Velocity: float = 0
  var left: bool = false
  var right: bool = false
  var Wake Anim: string = ""
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
