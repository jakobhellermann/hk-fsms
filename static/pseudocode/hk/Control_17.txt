# Control
# 2 location(s):
#   Fungus2_32 / Ring Holder/17
#   GG_Ghost_Hu / Ring Holder/17

fsm Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Box Small", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Big", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Impact", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Up", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Check Pos
  }

  state Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ring Antic 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Antic 2
  }

  state Down {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ring Down")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-60, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=6.01, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    ActivateGameObject(gameObject=var "Box Up", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Big", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on LAND → Land
  }

  state Land {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ring Impact 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=5.72, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Small", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Big", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Land 2
  }

  state Reset {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=13, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Land 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ring Impact 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Box Impact", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Reset
  }

  state Antic 2 {
    ActivateGameObject(gameObject=var "Box Small", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Box Up", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ring Antic 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Down
  }

  state Check Pos {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X", float2=32.35, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Self X", float2=66.32, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Antic
    on CANCEL → Reset
  }

  var Self X: float = 0
  var Self Y: float = 0
  var Box Big: gameObject = Ring Holder/17/Box Big
  var Box Impact: gameObject = Ring Holder/17/Box Impact
  var Box Small: gameObject = Ring Holder/17/Box Small
  var Box Up: gameObject = Ring Holder/17/Box Up
}
