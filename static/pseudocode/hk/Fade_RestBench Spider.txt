# Fade
# 1 location(s):
#   Deepnest_Spider_Town / RestBench Spider

fsm Fade {
  start Idle

  state Idle {
    FindGameObject(objectName="Vignette", withTag="Vignette", store=var "Vignette")
    FindChild(gameObject=Self, childName="Fader 1", storeResult=var "Fader 1")
    FindChild(gameObject=Self, childName="Fader 2", storeResult=var "Fader 2")
    FindChild(gameObject=Self, childName="Knight Anim", storeResult=var "Knight Anim")
    FindChild(gameObject=Self, childName="Webbed", storeResult=var "Webbed")
    FindChild(gameObject=Self, childName="Webbed Knight", storeResult=var "Webbed Knight")
    FindChild(gameObject=var "Webbed Knight", childName="Webbing", storeResult=var "Webbing")
    on FIRST STRUGGLE → Wait
  }

  state Wait {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Fade
  }

  state Start Fade {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SetDarkness(var ""))
    SetFsmInt(gameObject=var "Vignette", fsmName="Darkness Control", variableName="Darkness Level", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Vignette"), sendEvent="SCENE RESET", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Fader 1"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=5.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish Fade
  }

  state Finish Fade {
    SendEventByName(eventTarget=GameObject(var "Fader 2"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=0.5)
    on FINISHED → Sound 1
  }

  state Capture End {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    SendEventByName(eventTarget=BroadcastAll, sendEvent="CAPTURE END", delay=0, everyFrame=false)
    SetSpriteRenderer(gameObject=var "Webbed", active=false)
    SendEventByName(eventTarget=GameObject(var "Webbing"), sendEvent="UP", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Webbed Knight", active=true)
    SetMeshRenderer(gameObject=var "Knight Anim", active=false)
    GetPosition(gameObject=var "Webbed Knight", vector=(unset), x=var "Webbed Pos X", y=var "Webbed Pos Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Hero", vector=(unset), x=var "Webbed Pos X", y=var "Webbed Pos Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=0.1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE IN", delay=0, everyFrame=false)
    on FINISHED → Fade Up
  }

  state Fade Up {
    SetFsmFloat(gameObject=var "HUD Blanker", fsmName="Blanker Control", variableName="Fade Time", setValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stuck
  }

  state Stuck {
    ListenForJump(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForAttack(eventTarget=Self, wasPressed=→"STRUGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForLeft(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForDown(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForUp(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), stateEntryOnly=false)
    ListenForRight(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    on STRUGGLE → Struggle
    on WIGGLE → Wiggle
  }

  state Struggle {
    IntAdd(intVariable=var "Struggles", add=1, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=enemy_damage (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    IntCompare(integer1=var "Struggles", integer2=4, equal=→"BREAK", lessThan=(none), greaterThan=→"BREAK", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Webbed Knight", clipName="Web Struggle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Stuck
    on BREAK → Break
  }

  state Break {
    Tk2dPlayAnimationWithEvents(gameObject=var "Webbed Knight", clipName="Web Cut", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=enemy_damage (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=mimic_spider_transormation_stage_5 (sharedassets290.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Webbed Knight"), sendEvent="DOWN", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Webbed Knight", activate=true)
    SetSpriteRenderer(gameObject=var "Fader 1", active=false)
    SetSpriteRenderer(gameObject=var "Fader 2", active=false)
    on FINISHED → Fall
  }

  state Fall {
    SetMeshRenderer(gameObject=var "Webbed Knight", active=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=var "Hero", frame=9)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=var "Hero Land Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="DJ Get Land")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hero_land_soft (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetBenchRespawn", parameters=["Death Respawn Marker", "Deepnest_Spider_Town", 0, true], storeResult=(unset var))
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SetMapZoneToSpecific("BEASTS_DEN"))
    SetPlayerDataBool(boolName="spiderCapture", value=true)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=SaveGame())
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=TimePasses())
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="Area Title Control", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="Area Title Control", variableName="Display Right", setValue=false, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="HEGEMOL_NEST", everyFrame=false)
    on FINISHED → Relinquish Control
  }

  state Relinquish Control {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var))
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var))
    CreateObject(gameObject=Area Title (sharedassets9.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Area Title", networkInstantiate=false, networkGroup=0)  // disabled
  }

  state Wiggle {
    Tk2dPlayAnimationWithEvents(gameObject=var "Webbed Knight", clipName="Web Struggle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=grass_cut_1 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Stuck
  }

  state Sound 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=enemy_death_sword (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=mimic_spider_transormation_stage_2 (sharedassets81.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hero_damage_less_harsh (sharedassets304.assets), pitchMin=1, pitchMax=1, volume=1, delay=1, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=mimic_spider_transormation_stage_2 (sharedassets81.assets), pitchMin=1, pitchMax=1, volume=1, delay=2, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=mimic_spider_transormation_stage_4 (sharedassets290.assets), pitchMin=1, pitchMax=1, volume=1, delay=3, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=boss_final_hit (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=4, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=mawlek_scream (sharedassets45.assets), pitchMin=1.25, pitchMax=1.25, volume=1, delay=4.5, storePlayer=<null>)
    Wait(time=6, finishEvent=→"LAND", realTime=false)
    on LAND → Capture End
  }

  var Hero Land Y: float = 19.5
  var Hero Y: float = 0
  var Webbed Pos X: float = 0
  var Webbed Pos Y: float = 0
  var HP: int = 0
  var Struggles: int = 0
  var Area Title: gameObject = <null>
  var Fader 1: gameObject = <null>
  var Fader 2: gameObject = <null>
  var Knight Anim: gameObject = <null>
  var Vignette: gameObject = <null>
  var Webbed: gameObject = <null>
  var Webbed Knight: gameObject = <null>
  var Webbing: gameObject = <null>
}
