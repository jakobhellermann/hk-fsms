# Conversation Control
# 1 location(s):
#   Room_Jinn / Jinn NPC/Talk NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    CreateObject(gameObject=Area Title (sharedassets9.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Area Title", networkInstantiate=false, networkGroup=0)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="JINN", everyFrame=false)
    on FINISHED → Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TALK END", delay=0, everyFrame=false)
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Generic {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["JINN_TALK_GENERIC", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Choice {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Jinn_generic_01 (sharedassets29.assets), Jinn_generic_02 (sharedassets29.assets), Jinn_generic_03 (sharedassets29.assets), Jinn_generic_04 (sharedassets29.assets), Jinn_generic_05 (sharedassets29.assets)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TALK", delay=0, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="royalCharmState", storeValue=var "Royal Charm State")
    GetPlayerDataInt(gameObject=var "GameManager", intName="jinnEggsSold", storeValue=var "Eggs Sold")
    IntTestToBool(int1=var "Eggs Sold", int2=2, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Sold 3 Eggs", everyFrame=false)
    IntTestToBool(int1=var "Royal Charm State", int2=4, equalBool=var "Got Shade Charm", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=false)
    GetPlayerDataBool(boolName="jinnConvo3", storeValue=var "Egg3 Convo")
    GetPlayerDataBool(boolName="jinnConvoShadeCharm", storeValue=var "Shade Convo")
    BoolTestMulti(boolVariables=[var "Shade Convo", var "Got Shade Charm"], boolStates=[false, true], trueEvent=→"SHADE CHARM", falseEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="hasKingsBrand", falseBool="jinnConvoKingBrand", isTrue=→"KINGS BRAND", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="jinnConvo1", isTrue=(none), isFalse=→"CONVO 1")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="jinnConvo2", isTrue=(none), isFalse=→"CONVO 2")
    BoolTestMulti(boolVariables=[var "Egg3 Convo", var "Sold 3 Eggs"], boolStates=[false, true], trueEvent=→"CONVO 3", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Generic
    on SHADE CHARM → Shade Charm
    on KINGS BRAND → Kings Brand
    on CONVO 1 → Convo1
    on CONVO 2 → Convo2
    on CONVO 3 → Convo3
  }

  state Shade Charm {
    SetPlayerDataBool(boolName="jinnConvoShadeCharm", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["JINN_SHADE_CHARM", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Kings Brand {
    SetPlayerDataBool(boolName="jinnConvoKingBrand", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["JINN_KING_BRAND", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Convo1 {
    SetPlayerDataBool(boolName="jinnConvo1", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["JINN_TALK_01", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Convo2 {
    SetPlayerDataBool(boolName="jinnConvo2", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["JINN_TALK_02", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Convo3 {
    SetPlayerDataBool(boolName="jinnConvo3", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["JINN_TALK_03", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Eggs Sold: int = 0
  var Royal Charm State: int = 0
  var Egg3 Convo: bool = false
  var Got Shade Charm: bool = false
  var Shade Convo: bool = false
  var Sold 3 Eggs: bool = false
  var Spoken: bool = false
  var Area Title: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Talk NPC: gameObject = <null>
}
