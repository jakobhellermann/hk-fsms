// uses template: destroy_if_herocloseatstart
fsm FSM {
  start Pause

  state Check {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance to Hero", everyFrame=true)
    FloatCompare(float1=var "Distance to Hero", float2=var "Distance", tolerance=0, equal=→"DESTROY", lessThan=→"DESTROY", greaterThan=(none), everyFrame=false)
    on DESTROY → Destroy
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Distance: float = 10

  var Distance to Hero: float = 0
}
