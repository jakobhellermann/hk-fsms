# Trigger Activate
# 1 location(s):
#   Crossroads_ShamanTemple / _Areas/Death Respawn Trigger 1

fsm Trigger Activate {
  start Idle

  state Idle {
    GetPlayerDataInt(gameObject=var "GameManager", intName="shaman", storeValue=var "Shaman")
    IntCompare(integer1=var "Shaman", integer2=5, equal=(none), lessThan=(none), greaterThan=→"ACTIVATE RESPAWN TRIGGER", everyFrame=false)
    on ACTIVATE RESPAWN TRIGGER → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true)
  }

  var Shaman: int = 0
}
