# Control
# 2 location(s):
#   Hive_05 / Battle Scene/Globs
#   GG_Hive_Knight / Battle Scene/Globs

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on FIRE → Set L
  }

  state Set L {
    GetRandomChild(gameObject=Self, storeResult=var "Glob L")
    SetPosition(gameObject=var "Glob L", vector=(unset), x=-7.57, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetParent(gameObject=var "Glob L", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Set M
  }

  state Set M {
    GetRandomChild(gameObject=Self, storeResult=var "Glob M")
    SetPosition(gameObject=var "Glob M", vector=(unset), x=-0.57, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetParent(gameObject=var "Glob M", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Set R
  }

  state Set R {
    GetRandomChild(gameObject=Self, storeResult=var "Glob R")
    SetPosition(gameObject=var "Glob R", vector=(unset), x=6.43, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetParent(gameObject=var "Glob R", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fire
  }

  state Fire {
    SendEventByName(eventTarget=GameObject(var "Glob L"), sendEvent="FIRE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Glob M"), sendEvent="FIRE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Glob R"), sendEvent="FIRE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Glob L: gameObject = <null>
  var Glob M: gameObject = <null>
  var Glob R: gameObject = <null>
}
