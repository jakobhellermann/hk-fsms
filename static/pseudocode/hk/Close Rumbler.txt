# Close Rumbler
# 1 location(s):
#   Mines_31 / Super Dash Get/Close Rumbler

fsm Close Rumbler {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Far Rumbler", storeResult=var "Far Rumbler")
    FindChild(gameObject=var "Parent", childName="Beam Parent", storeResult=var "Beam Parent")
    FindChild(gameObject=var "Parent", childName="Glow", storeResult=var "Glow")
    on ENTER → Close
  }

  state Close {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far Rumbler"), sendEvent="AWAY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Glow"), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    SetScale(gameObject=var "Glow", vector=(unset), x=1.238869, y=2.650868, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleTo(gameObject=var "Beam Parent", id="", transformScale=var "", vectorScale=(2.71, 2.01, 1), time=0.2, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on EXIT → Away
  }

  state Away {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Far Rumbler"), sendEvent="CLOSE", delay=0, everyFrame=false)
    SetScale(gameObject=var "Glow", vector=(unset), x=0.7882982, y=1.68676, z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Glow"), sendEvent="DOWN INSTANT", delay=0, everyFrame=false)
    iTweenScaleTo(gameObject=var "Beam Parent", id="", transformScale=var "", vectorScale=(0.57, 1.48, 1), time=1, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on ENTER → Close
  }

  var Beam Parent: gameObject = <null>
  var Far Rumbler: gameObject = <null>
  var Glow: gameObject = <null>
  var Parent: gameObject = <null>
}
