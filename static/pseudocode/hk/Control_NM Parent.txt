# Control
# 1 location(s):
#   Room_nailmaster_02 / NM Parent

fsm Control {
  start Pause

  state Pause {
    FindChild(gameObject=Self, childName="Modelling", storeResult=var "Modelling")
    FindChild(gameObject=Self, childName="Painting", storeResult=var "Painting")
    on FINISHED → Check Nailsmith
  }

  state Check Nailsmith {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="nailsmithSheo", isTrue=(none), isFalse=→"ALONE")
    SendRandomEvent(events=[→"MODELLING", →"PAINTING"], weights=[1, 1], delay=0)
    on ALONE → Painting
    on MODELLING → Modelling
    on PAINTING → Painting
  }

  state Painting {
  }

  state Modelling {
    ActivateGameObject(gameObject=var "Painting", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Modelling", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Modelling: gameObject = <null>
  var Painting: gameObject = <null>
}
