# Hit Around
# 1 location(s):
#   Ruins_Bathhouse / Banker Spa NPC

fsm Hit Around {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="Dream Dialogue", storeResult=var "Dream Dialogue")
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerTheft", storeValue=var "Banker Theft")
    IntCompare(integer1=var "Banker Theft", integer2=0, equal=→"INACTIVE", lessThan=→"INACTIVE", greaterThan=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Roll Collider", storeResult=var "Roll Collider")
    FindChild(gameObject=Self, childName="Sit Collider", storeResult=var "Sit Collider")
    FindChild(gameObject=Self, childName="NPC", storeResult=var "NPC")
    on FINISHED → Idle
    on INACTIVE → Inactive
  }

  state Idle {
    on TINK DOWN → Hit Down
    on TINK LEFT → Hit Left
    on TINK RIGHT → Hit Right
    on TINK UP → Hit Up
  }

  state Hit Right {
    SetIntValue(intVariable=var "Hit Direction", intValue=0, everyFrame=false)
    FlingObject(flungObject=Self, speedMin=20, speedMax=22, angleMin=50, angleMax=70)
    RandomFloat(min=-90, max=-135, storeResult=var "Angular Velocity")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Angular Velocity", everyFrame=false)
    on FINISHED → Cancel NPC
  }

  state Hit Up {
    SetIntValue(intVariable=var "Hit Direction", intValue=1, everyFrame=false)
    FlingObject(flungObject=Self, speedMin=28, speedMax=30, angleMin=80, angleMax=100)
    RandomFloat(min=-45, max=45, storeResult=var "Angular Velocity")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Angular Velocity", everyFrame=false)
    on FINISHED → Cancel NPC
  }

  state Hit Left {
    SetIntValue(intVariable=var "Hit Direction", intValue=2, everyFrame=false)
    FlingObject(flungObject=Self, speedMin=20, speedMax=22, angleMin=110, angleMax=130)
    RandomFloat(min=90, max=135, storeResult=var "Angular Velocity")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Angular Velocity", everyFrame=false)
    on FINISHED → Cancel NPC
  }

  state Hit Down {
    SetIntValue(intVariable=var "Hit Direction", intValue=3, everyFrame=false)
    on FINISHED → Cancel NPC
  }

  state Withdraw {
    BoolTest(boolVariable=var "Withdrawn", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spa Withdraw")
    SetPolygonCollider(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Roll Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Sit Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Withdrawn", boolValue=true, everyFrame=false)
    on FINISHED → Give Geo
  }

  state Withdrawn {
    Wait(time=4.5, finishEvent=→"EMERGE", realTime=false)
    on TINK DOWN → Hit Down
    on TINK LEFT → Hit Left
    on TINK RIGHT → Hit Right
    on TINK UP → Hit Up
    on EMERGE → Emerge
  }

  state Emerge {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Banker_Spa_yelp_02 (sharedassets87.assets), Banker_Spa_talk_nervous_04 (sharedassets87.assets), Banker_Spa_talk_nervous_02 (sharedassets87.assets)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[roller_curl (sharedassets32.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetBoolValue(boolVariable=var "Withdrawn", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spa Emerge")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=10, everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetPolygonCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Sit Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on TINK DOWN → Hit Down
    on TINK LEFT → Hit Left
    on TINK RIGHT → Hit Right
    on TINK UP → Hit Up
  }

  state Cancel NPC {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Banker_Spa_hit_01 (sharedassets87.assets), Banker_Spa_hit_02 (sharedassets87.assets), Banker_Spa_hit_03 (sharedassets87.assets), Banker_Spa_hit_04 (sharedassets87.assets), Banker_Spa_yelp_03 (sharedassets87.assets), Banker_Spa_yelp_04 (sharedassets87.assets)], weights=[1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[barrel_death_1 (sharedassets6.assets), barrel_death_2 (sharedassets6.assets)], weights=[1, 1], pitchMin=0.75, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    BoolTest(boolVariable=var "NPC Cancelled", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "NPC Cancelled", boolValue=true, everyFrame=false)
    SetCollider(gameObject=var "NPC", active=false)  // disabled
    ActivateGameObject(gameObject=var "NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dream Dialogue", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ARROW DOWN", delay=0, everyFrame=false)
    on FINISHED → Withdraw
  }

  state Give Geo {
    GetTagCount(tag="Geo", storeResult=var "Geo In Scene")
    IntCompare(integer1=var "Geo In Scene", integer2=80, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerBalance", storeValue=var "Balance")
    IntCompare(integer1=var "Balance", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    RandomInt(min=6, max=10, storeResult=var "Geo Amount", inclusiveMax=true)
    FlingObjectsFromGlobalPool(gameObject=Geo Small (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Amount", spawnMax=var "Geo Amount", speedMin=15, speedMax=30, angleMin=80, angleMax=100, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    IntOperator(integer1=var "Balance", integer2=var "Geo Amount", operation=Subtract, storeResult=var "Balance", everyFrame=false)
    RandomInt(min=1, max=5, storeResult=var "Geo Amount", inclusiveMax=true)
    FlingObjectsFromGlobalPool(gameObject=Geo Med (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Amount", spawnMax=var "Geo Amount", speedMin=15, speedMax=30, angleMin=80, angleMax=100, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    IntOperator(integer1=var "Geo Amount", integer2=5, operation=Multiply, storeResult=var "Geo Amount", everyFrame=false)
    IntOperator(integer1=var "Balance", integer2=var "Geo Amount", operation=Subtract, storeResult=var "Balance", everyFrame=false)
    RandomInt(min=0, max=2, storeResult=var "Geo Amount", inclusiveMax=true)
    FlingObjectsFromGlobalPool(gameObject=Geo Large (resources.assets), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Amount", spawnMax=var "Geo Amount", speedMin=15, speedMax=30, angleMin=80, angleMax=100, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    IntOperator(integer1=var "Geo Amount", integer2=25, operation=Multiply, storeResult=var "Geo Amount", everyFrame=false)
    IntOperator(integer1=var "Balance", integer2=var "Geo Amount", operation=Subtract, storeResult=var "Balance", everyFrame=false)
    SetPlayerDataInt(intName="bankerBalance", value=var "Balance")
    on FINISHED → Withdrawn
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Angular Velocity: float = 0
  var Balance: int = 0
  var Banker Theft: int = 0
  var Geo Amount: int = 0
  var Geo In Scene: int = 0
  var Hit Direction: int = 0
  var NPC Cancelled: bool = false
  var Withdrawn: bool = false
  var Dream Dialogue: gameObject = <null>
  var NPC: gameObject = <null>
  var Roll Collider: gameObject = <null>
  var Self: gameObject = <null>
  var Sit Collider: gameObject = <null>
}
