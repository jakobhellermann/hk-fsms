# Fade Down
# 2 location(s):
#   Ruins2_11 / Ambient Loops
#   Ruins2_11_b / Ambient Loops

fsm Fade Down {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Audio Loop Collector", storeResult=var "Loop Collector")
    FindChild(gameObject=Self, childName="Audio Loop Grubs", storeResult=var "Loop Grubs")
    on FINISHED → Idle
  }

  state Idle {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="collectorDefeated", isTrue=→"DISABLE", isFalse=(none))
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on DISABLE → Disable
    on ENTER → Fade
  }

  state Disable {
    AudioStop(gameObject=var "Loop Collector")
    AudioStop(gameObject=var "Loop Grubs")
  }

  state Fade {
    FadeAudio(gameObject=var "Loop Collector", startVolume=1, endVolume=0, time=1)
    FadeAudio(gameObject=var "Loop Grubs", startVolume=1, endVolume=0, time=1)
  }

  var Loop Collector: gameObject = <null>
  var Loop Grubs: gameObject = <null>
}
