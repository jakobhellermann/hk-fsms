# Cage Control
# 1 location(s):
#   Fungus2_15_boss / Mantis Battle/mantis_cage_down

fsm Cage Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=19, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FindChild(gameObject=Self, childName="Colliders", storeResult=var "Colliders")
    FindChild(gameObject=Self, childName="Slam Dust", storeResult=var "Slam Dust")
    FindChild(gameObject=Self, childName="Rise Dust", storeResult=var "Rise Dust")
    on FINISHED → Idle
  }

  state Idle {
    on MLORD CAGE DOWN → Descend
  }

  state Descend {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -19, 0), time=0.5, delay=0, speed=(unset), easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    ActivateGameObject(gameObject=var "Colliders", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Land
  }

  state Land {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_lords_gate_close (sharedassets180.assets))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    PlayParticleEmitter(gameObject=var "Slam Dust", emit=0)
    on MLORD CAGE UP → CamLocks Off
  }

  state Rise 1 {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 2, 0), time=2, delay=0, speed=(unset), easeType=easeOutCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Rise Dust", emit=0)
    on FINISHED → Rise 2
  }

  state Rise 2 {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, 19, 0), time=2, delay=0, speed=(unset), easeType=easeInCubic, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    ActivateGameObject(gameObject=var "Colliders", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  state CamLocks Off {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="CamLocks", storeResult=var "CamLock")
    ActivateAllChildren(gameObject=var "CamLock", activate=false)
    on FINISHED → Rise 1
  }

  var CamLock: gameObject = <null>
  var Colliders: gameObject = <null>
  var Hero: gameObject = <null>
  var Parent: gameObject = <null>
  var Rise Dust: gameObject = <null>
  var Self: gameObject = <null>
  var Slam Dust: gameObject = <null>
}
