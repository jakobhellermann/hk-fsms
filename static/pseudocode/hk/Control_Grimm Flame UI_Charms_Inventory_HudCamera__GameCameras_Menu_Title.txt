# Control
# 1 location(s):
#   Menu_Title / _GameCameras/HudCamera/Inventory/Charms/Grimm Flame UI

fsm Control {
  start Init
  on HIDE → Hide  // from any state
  on INVENTORY CLOSED → Hide  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPlayerDataInt(gameObject=var "GameManager", intName="flamesCollected", storeValue=var "Flames Collected")
    GetPlayerDataInt(gameObject=var "GameManager", intName="grimmChildLevel", storeValue=var "Grimmchild Level")
    on FINISHED → Hide
  }

  state Hide {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on SHOW → Level?
  }

  state Level? {
    IntSwitch(intVariable=var "Grimmchild Level", compareTo=[1, 2, 3, 4, 5], sendEvent=[→"1", →"1", →"1", →"4", →"5"], everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"1", delay=0, everyFrame=false)
    on 1 → Set 1
    on 2 → 
    on 3 → 
    on 4 → Hide
    on 5 → Hide
  }

  state Set 1 {
    ActivateGameObject(gameObject=var "Set 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    IntSwitch(intVariable=var "Flames Collected", compareTo=[1, 2, 3, 0], sendEvent=[→"1", →"2", →"3", →"0"], everyFrame=false)
    on 1 → S1 F 1
    on 2 → S1 F 2
    on 3 → S1 F 3
    on 0 → None
  }

  state S1 F 1 {
    SetSpriteRendererSprite(gameObject=var "S1 Flame1", sprite=Grimm_charm_flame_front (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame2", sprite=Grimm_charm_flame_backboard (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame3", sprite=Grimm_charm_flame_backboard (resources.assets))
    on FINISHED → 
  }

  state S1 F 2 {
    SetSpriteRendererSprite(gameObject=var "S1 Flame1", sprite=Grimm_charm_flame_front (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame2", sprite=Grimm_charm_flame_front (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame3", sprite=Grimm_charm_flame_backboard (resources.assets))
    on FINISHED → 
  }

  state S1 F 3 {
    SetSpriteRendererSprite(gameObject=var "S1 Flame1", sprite=Grimm_charm_flame_front (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame2", sprite=Grimm_charm_flame_front (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame3", sprite=Grimm_charm_flame_front (resources.assets))
    on FINISHED → 
  }

  state None {
    SetSpriteRendererSprite(gameObject=var "S1 Flame1", sprite=Grimm_charm_flame_backboard (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame2", sprite=Grimm_charm_flame_backboard (resources.assets))
    SetSpriteRendererSprite(gameObject=var "S1 Flame3", sprite=Grimm_charm_flame_backboard (resources.assets))
  }

  var S1 Flame1: gameObject = _GameCameras/HudCamera/Inventory/Charms/Grimm Flame UI/Set 1/Flame 1
  var S1 Flame2: gameObject = _GameCameras/HudCamera/Inventory/Charms/Grimm Flame UI/Set 1/Flame 2
  var S1 Flame3: gameObject = _GameCameras/HudCamera/Inventory/Charms/Grimm Flame UI/Set 1/Flame 3
  var Set 1: gameObject = _GameCameras/HudCamera/Inventory/Charms/Grimm Flame UI/Set 1

  var Flames Collected: int = 0
  var Grimmchild Level: int = 0
  var Self: gameObject = <null>
}
