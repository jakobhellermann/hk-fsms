# Shade Orbs
# 1 location(s):
#   resources.assets / Hollow Shade Death

fsm Shade Orbs {
  start Pause

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Orbs
  }

  state Orbs {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SpawnRandomObjectsOverTime(gameObject=Shade Orb, spawnPoint=var "Self", position=(unset), frequency=0.075, spawnMin=1, spawnMax=2, speedMin=0, speedMax=0, angleMin=0, angleMax=0, originVariation=1)
    on FINISHED → End
  }

  state End {
  }

  var Self: gameObject = <null>
}
