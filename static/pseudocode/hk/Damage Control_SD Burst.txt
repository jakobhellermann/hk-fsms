# Damage Control
# 2 location(s):
#   Knight_Pickup / Knight/Effects/SD Burst
#   resources.assets / Knight:1/Effects/SD Burst

fsm Damage Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=true)
    SendEventByScale(gameObject=var "Hero", eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on L → L
    on R → R
  }

  state L {
    SetFsmFloat(gameObject=Self, fsmName="damages_enemy", variableName="direction", setValue=180, everyFrame=false)
    on FINISHED → Active
  }

  state R {
    SetFsmFloat(gameObject=Self, fsmName="damages_enemy", variableName="direction", setValue=0, everyFrame=false)
    on FINISHED → Active
  }

  state Active {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Inactive
  }

  state Inactive {
    SetPolygonCollider(gameObject=Self, active=false)
  }
}
