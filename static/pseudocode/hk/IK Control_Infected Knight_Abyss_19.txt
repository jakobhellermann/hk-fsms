# IK Control
# 1 location(s):
#   Abyss_19 / Infected Knight

fsm IK Control {
  start Pause
  on STUN → Stun Start  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Stun Hitbox", storeResult=var "Stun Hitbox")
    GGCheckIfBossScene(bossSceneEvent=→"ACTIVE", regularSceneEvent=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="infectedKnightEncountered", isTrue=→"ACTIVE", isFalse=(none))
    FindChild(gameObject=Self, childName="Stun Hitbox", storeResult=var "Stun Hitbox")
    on FINISHED → Sleep
    on ACTIVE → Repause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetGravity2dScale(gameObject=Self, gravityScale=var "Gravity")
    on FINISHED → Init
  }

  state Sleep {
    on BATTLE START → Start Pause
  }

  state Start Pause {
    SetPlayerDataBool(boolName="infectedKnightEncountered", value=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Corpse Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Rumble Start
  }

  state Rumble Start {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    SetAudioClip(gameObject=Self, audioClip=brkn_wand_room_shake_loop (sharedassets343.assets))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Summon Grass", emit=0)
    on FINISHED → Spawning 1
  }

  state Spawning 1 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCUTTLER SPAWN START", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawning 2
  }

  state Spawning 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wake 1")
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawning 3
  }

  state Spawning 3 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wake 2")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawning 4
  }

  state Spawning 4 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wake 3")
    Wait(time=0, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Spawning
  }

  state Check Final {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Check {
    FindGameObject(objectName="Orange Scuttler IK(Clone)", withTag="Untagged", store=var "Scuttler Check")
    GameObjectIsNull(gameObject=var "Scuttler Check", isNull=→"ROAR", isNotNull=→"FINISHED", storeResult=(unset), everyFrame=false)
    on FINISHED → Check Final
    on ROAR → Roar Start
  }

  state Stop Spawning {
    AudioStop(gameObject=Self)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SCUTTLER SPAWN END", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Summon Grass")
    on FINISHED → Check Final
  }

  state Roar Start {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(false))
    ActivateGameObject(gameObject=var "Shadow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Music
  }

  state Roar {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar Loop")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hollow_knight_scream_v1 (sharedassets19.assets))
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    Wait(time=2.4, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Visited", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="INFECTED_KNIGHT", everyFrame=false)
    on FINISHED → Roar End
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="START SPAWN", delay=0, everyFrame=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    SetCollider(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → First Counter
  }

  state Idle {
    SetBoolValue(boolVariable=var "Downstabbing", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToWalk", resetFrame=true, everyFrame=true)
    RandomFloat(min=0.5, max=1.25, storeResult=var "Walk Speed")
    RandomlyFlipFloat(storeResult=var "Walk Speed")
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Walk Speed", y=(unset), everyFrame=false)
    FloatSubtract(floatVariable=var "Idle Counter", subtract=1, everyFrame=true, perSecond=true)
    BoolTest(boolVariable=var "Do Shake", isTrue=→"SHAKE", isFalse=(none), everyFrame=false)
    FloatCompare(float1=var "Idle Counter", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    BoolAllTrue(boolVariables=[var "Will Evade", var "Evade Range"], sendEvent=→"EVADE", storeResult=(unset), everyFrame=false)
    BoolAllTrue(boolVariables=[var "Will Ohead", var "Ohead Range"], sendEvent=→"OHEAD", storeResult=(unset), everyFrame=false)
    on DOWNSTAB → Set Dstab
    on JUMP → Jump Antic
    on DASH → Air Dash?
    on OHEAD → Ohead Antic
    on EVADE → Evade Antic
    on SHAKE → Jump Antic 2
    on FINISHED → Attack Choice
    on TOOK DAMAGE → Damage Response
  }

  state Set Dstab {
    SetBoolValue(boolVariable=var "Downstabbing", boolValue=true, everyFrame=false)
    on FINISHED → Jump Antic
  }

  state Jump {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_jump (sharedassets34.assets))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    RandomFloat(min=55, max=55, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Jump X", y=var "Jump Y", everyFrame=false)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → In Air
  }

  state In Air {
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatTestToBool(float1=var "Self Y", float2=var "Min Dstab Height", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Over Min Height", everyFrame=true)
    FloatTestToBool(float1=var "Self Y", float2=var "Air Dash Height", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Above Air Dash Height", everyFrame=true)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Velocity", space=0, everyFrame=true)
    BoolAllTrue(boolVariables=[var "Downstabbing", var "Dstab Range", var "Over Min Height"], sendEvent=→"DOWNSTAB", storeResult=(unset), everyFrame=true)
    BoolAllTrue(boolVariables=[var "Above Air Dash Height", var "Will Air Dash"], sendEvent=→"AIRDASH", storeResult=(unset), everyFrame=true)
    on LAND → Land
    on DOWNSTAB → Dstab Antic
    on AIRDASH → Set Height
  }

  state Jump Antic {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Air Dash", boolValue=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Jump Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    on FINISHED → Jump Check
  }

  state Land {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_land (sharedassets32.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → To Attack?
  }

  state Aim Jump {
    RandomFloat(min=var "Left X", max=var "Right X", storeResult=var "Jump X")
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatSubtract(floatVariable=var "Jump X", subtract=var "Self X", everyFrame=false, perSecond=false)
    FloatMultiply(floatVariable=var "Jump X", multiplyBy=1.25, everyFrame=false)
    on FINISHED → Jump
  }

  state Aim Dstab {
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Hero X", float2=var "Self X", operation=Subtract, storeResult=var "Jump X", everyFrame=false)
    FloatMultiply(floatVariable=var "Jump X", multiplyBy=2.5, everyFrame=false)
    on FINISHED → Jump
  }

  state Dstab Antic {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_down_stab_prepare (sharedassets343.assets))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Downstab Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    on FINISHED → Dstab Fall
  }

  state Dstab Fall {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_down_stab_dash (sharedassets209.assets))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Downstab")
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-40, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=var "Gravity")
    ActivateGameObject(gameObject=var "Dstab Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on LAND → Dstab Land
  }

  state Dstab Land {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_down_stab_impact (sharedassets209.assets))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Dstab Slam", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Downstab Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=IK Projectile DS (sharedassets343.assets), spawnPoint=var "Self", position=(0, -0.5, 0), rotation=(unset), storeObject=var "Projectile")
    SetVelocity2d(gameObject=var "Projectile", vector=(unset), x=10, y=(unset), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=IK Projectile DS (sharedassets343.assets), spawnPoint=var "Self", position=(0, -0.5, 0), rotation=(unset), storeObject=var "Projectile")
    SetVelocity2d(gameObject=var "Projectile", vector=(unset), x=19, y=(unset), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=IK Projectile DS (sharedassets343.assets), spawnPoint=var "Self", position=(0, -0.5, 0), rotation=(unset), storeObject=var "Projectile")
    SetVelocity2d(gameObject=var "Projectile", vector=(unset), x=-10, y=(unset), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=IK Projectile DS (sharedassets343.assets), spawnPoint=var "Self", position=(0, -0.5, 0), rotation=(unset), storeObject=var "Projectile")
    SetVelocity2d(gameObject=var "Projectile", vector=(unset), x=-19, y=(unset), everyFrame=false)
    on FINISHED → Dstab Recover
  }

  state Attack Choice {
    SendRandomEventV3(events=[→"DOWNSTAB", →"JUMP", →"DASH"], weights=[1, 0.5, 1], trackingInts=[var "Dstab Ct", var "Jump Ct", var "Dash Ct"], eventMax=[2, 1, 2], trackingIntsMissed=[var "DStab Ms", var "Jump Ms", var "Dash Ms"], missedMax=[3, 5, 3])
    on DOWNSTAB → Set Dstab
    on JUMP → Jump Antic
    on DASH → Air Dash?
  }

  state Jump Check {
    BoolTest(boolVariable=var "Downstabbing", isTrue=→"DOWNSTAB", isFalse=→"JUMP", everyFrame=false)
    on JUMP → Aim Jump
    on DOWNSTAB → Aim Dstab
  }

  state Dash Antic {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_horizontal_dash_prepare (sharedassets343.assets))
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Antic 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Dash Speed Crt", floatValue=var "Dash Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Dash Speed Crt", multiplyBy=var "X Scale", everyFrame=false)
    SetFloatValue(floatVariable=var "Dash Reverse Crt", floatValue=var "Dash Reverse", everyFrame=false)
    FloatMultiply(floatVariable=var "Dash Reverse Crt", multiplyBy=var "X Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "Dash Reverse Crt", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Dash Reverse Crt", y=0, everyFrame=true)
    on FINISHED → Dash Antic 2
  }

  state Dash Attack {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_horizontal_dash_dash (sharedassets243.assets))
    ActivateGameObject(gameObject=var "Dash Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Attack 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Dash Speed Crt", y=(unset), everyFrame=true)
    on FINISHED → Dash Attack 2
  }

  state Dash Recover {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=var "Gravity")
    BoolTest(boolVariable=var "Will Air Dash", isTrue=→"FALL", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Set Counter
    on FALL → Fall
  }

  state Dash Attack 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=brkn_wand_horizontal_sword (sharedassets343.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Attack 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Dash Speed Crt", y=(unset), everyFrame=true)
    ActivateGameObject(gameObject=var "Dash Slash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dash Slash Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Dash Attack 3
  }

  state Dash Attack 3 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Attack 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Dash Speed Crt", y=(unset), everyFrame=true)
    ActivateGameObject(gameObject=var "Dash Slash Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Dash Recover
  }

  state Dstab Recover {
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Set Counter
  }

  state Ohead Antic {
    SetBoolValue(boolVariable=var "Will Ohead", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Overhead Antic")
    Wait(time=0.65, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Ohead Slashing
  }

  state Ohead Slashing {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_sword (sharedassets343.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_sword (sharedassets343.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.25, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_sword (sharedassets343.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.5, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_sword (sharedassets343.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Overhead Slashing", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Overhead Slash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Overhead Slash", animLibName="", clipName="Overhead Slash")
    Tk2dPlayFrame(gameObject=var "Overhead Slash", frame=0)
    on FINISHED → Ohead Recover
  }

  state Ohead Recover {
    AudioStop(gameObject=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Overhead Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    ActivateGameObject(gameObject=var "Overhead Slash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set Counter
  }

  state Evade Antic {
    BoolTest(boolVariable=var "Evade Check", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)

    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Evade Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)

  state Evade Dash {

    on FINISHED → Shake End
  }

  state Dash Antic 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Antic 2", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)

  }

  state Dash Antic 3 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dash Antic 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Dash Attack
  }

  state Air Dash? {
    BoolTest(boolVariable=var "Dash Range", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "Will Air Dash", boolValue=false, everyFrame=false)
    SendRandomEvent(events=[(none), →"DASH"], weights=[0.3, 0.7], delay=0)
    SetBoolValue(boolVariable=var "Will Air Dash", boolValue=true, everyFrame=false)
    on DASH → Dash Antic
    on FINISHED → Jump Antic
