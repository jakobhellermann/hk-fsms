fsm Set Invincible {
  start Init

  state Init {
    on FINISHED → Shield Off
  }

  state Shield Off {
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=(unset))
    on SHIELD ON → Check Dir
  }

  state Check Dir {
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=(none), space=0)
    on R → R Shielded
    on L → L Shielded
  }

  state R Shielded {
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=(unset))
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Y", add=2.29, everyFrame=true, perSecond=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatTestToBool(float1=var "Hero X", float2=var "Self X", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "On Shielded Side", everyFrame=true)
    FloatTestToBool(float1=var "Hero Y", float2=var "Self Y", tolerance=0, equalBool=(unset), lessThanBool=var "Not Above", greaterThanBool=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Not Above", var "On Shielded Side"], boolStates=[true, true], trueEvent=(none), falseEvent=→"UNSHIELDED", storeResult=(unset), everyFrame=true)
    on UNSHIELDED → R Unshielded
    on SHIELD OFF → Shield Off
    on SHIELD REFRESH → Check Dir
  }

  state R Unshielded {
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=(unset))

    FloatTestToBool(float1=var "Hero Y", float2=var "Self Y", tolerance=0, equalBool=(unset), lessThanBool=var "Not Above", greaterThanBool=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Not Above", var "On Shielded Side"], boolStates=[true, true], trueEvent=→"SHIELDED", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on SHIELDED → R Shielded
    on SHIELD OFF → Shield Off
    on SHIELD REFRESH → Check Dir
  }

  state L Shielded {
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=(unset))
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=true)

    FloatTestToBool(float1=var "Hero X", float2=var "Self X", tolerance=0, equalBool=(unset), lessThanBool=var "On Shielded Side", greaterThanBool=(unset), everyFrame=true)
    FloatTestToBool(float1=var "Hero Y", float2=var "Self Y", tolerance=0, equalBool=(unset), lessThanBool=var "Not Above", greaterThanBool=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Not Above", var "On Shielded Side"], boolStates=[true, true], trueEvent=(none), falseEvent=→"UNSHIELDED", storeResult=(unset), everyFrame=true)
