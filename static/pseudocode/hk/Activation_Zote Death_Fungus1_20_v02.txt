# Activation
# 1 location(s):
#   Fungus1_20_v02 / Zote Death

fsm Activation {
  start Pause

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="zoteDeathPos", storeValue=var "Zote Death")
    IntCompare(integer1=var "Death Number", integer2=var "Zote Death", equal=(none), lessThan=→"DESTROY", greaterThan=→"DESTROY", everyFrame=false)
    on DESTROY → Destroy
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Death Number: int = 1

  var Zote Death: int = 0
}
