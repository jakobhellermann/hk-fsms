# Loop
# 2 location(s):
#   Dream_Mighty_Zote / Grey Prince/Loop Charge
#   GG_Grey_Prince_Zote / Grey Prince/Loop Charge

fsm Loop {
  start Init

  state Idle {
    on CHARGE START → Play
  }

  state Play {
    AudioPlayRandom(gameObject=var "Self", audioClips=[zombie_shield_sword_1 (sharedassets32.assets), zombie_shield_sword_2 (sharedassets32.assets), zombie_shield_sword_3 (sharedassets32.assets)], weights=[1, 1, 1], pitchMin=0.8, pitchMax=1.2)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on CHARGE END → Idle
    on FINISHED → Loop Back
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Loop Back {
    on FINISHED → Play
    on CHARGE END → Idle
  }

  var Self: gameObject = <null>
}
