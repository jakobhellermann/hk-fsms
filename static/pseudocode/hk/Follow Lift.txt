fsm Follow Lift {
  start Init

  state Init {
    FindGameObject(objectName="Mines Exit Lift", withTag="Untagged", store=var "Lift")
    on FINISHED → Follow
  }

  state Follow {
    GetPosition(gameObject=var "Lift", vector=(unset), x=(unset), y=var "Lift Y", z=(unset), space=0, everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Lift Y", z=(unset), space=1, everyFrame=true, lateUpdate=false)
  }

  var Lift Y: float = 0
  var Lift: gameObject = <null>
}
