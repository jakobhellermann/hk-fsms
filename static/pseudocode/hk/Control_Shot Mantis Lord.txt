# Control
# 4 location(s):
#   Fungus2_15_boss / Shot Mantis Lord
#   GG_Mantis_Lords / Shot Mantis Lord
#   GG_Mantis_Lords_V / Shot Mantis Lord
#   sharedassets180.assets / Shot Mantis Lord

fsm Control {
  start Pause
  on NARROW R → Narrow R  // from any state
  on NARROW L → Narrow L  // from any state
  on WIDE R → Wide R  // from any state
  on WIDE L → Wide L  // from any state

  state Check UD {
    BoolTest(boolVariable=var "Throw UD", isTrue=→"UD", isFalse=→"DU", everyFrame=false)
    on UD → Set UD
    on DU → Set DU
  }

  state Set DU {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=1.1, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-0.5, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt DU", emit=0)
    on FINISHED → Corner Shot?
  }

  state Check LR {
    BoolTest(boolVariable=var "Throw R", isTrue=(none), isFalse=(none), everyFrame=false)
    on L → Set L
    on R → Set R
  }

  state Set R {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=33, everyFrame=false)
    SetFloatValue(floatVariable=var "X Force", floatValue=-24, everyFrame=false)
    on FINISHED → Fly
  }

  state Set L {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-33, everyFrame=false)
    SetFloatValue(floatVariable=var "X Force", floatValue=24, everyFrame=false)
    on FINISHED → Fly
  }

  state Fly {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "X Force", y=(unset), vector3=(unset), everyFrame=true)
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
    on ORBIT SHIELD → Shield Break
  }

  state Recycle {
    SetBoolValue(boolVariable=var "No Dust", boolValue=false, everyFrame=false)
    RecycleSelf()
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mantis Air Projectile")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Check UD
  }

  state Set UD {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-1.1, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt UD", emit=0)
    on FINISHED → No Dust?
  }

  state Wide L {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=33, everyFrame=false)
    SetFloatValue(floatVariable=var "X Force", floatValue=-24, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-2.2, everyFrame=false)
    on FINISHED → High Pt
  }

  state Wide R {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-33, everyFrame=false)
    SetFloatValue(floatVariable=var "X Force", floatValue=24, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-2.2, everyFrame=false)
    on FINISHED → High Pt
  }

  state Narrow L {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=16.5, everyFrame=false)
    SetFloatValue(floatVariable=var "X Force", floatValue=-12, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-2.2, everyFrame=false)
    on FINISHED → High Pt
  }

  state Narrow R {
    SetFloatValue(floatVariable=var "X Velocity", floatValue=-16.5, everyFrame=false)
    SetFloatValue(floatVariable=var "X Force", floatValue=12, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-2.2, everyFrame=false)
    on FINISHED → High Pt
  }

  state Corner Shot? {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Y Pos", float2=19, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=2, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt DU")
    on FINISHED → No Dust?
  }

  state High Pt {
    PlayParticleEmitter(gameObject=var "Pt UD", emit=0)
    on FINISHED → No Dust? 2
  }

  state Shield Break {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Projectile Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=sword_hit_reject (resources.assets), pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    StopParticleEmitter(gameObject=var "Pt DU")
    StopParticleEmitter(gameObject=var "Pt UD")
    on FINISHED → Break Wait
  }

  state Break Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state No Dust? {
    BoolTest(boolVariable=var "No Dust", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt DU")
    StopParticleEmitter(gameObject=var "Pt UD")
    on FINISHED → Check LR
  }

  state No Dust? 2 {
    BoolTest(boolVariable=var "No Dust", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt DU")
    StopParticleEmitter(gameObject=var "Pt UD")
    on FINISHED → Fly
  }

  var X Force: float = 0
  var X Velocity: float = 0
  var Y Pos: float = 0
  var No Dust: bool = false
  var Throw R: bool = false
  var Throw UD: bool = false
  var Pt DU: gameObject = Shot Mantis Lord/Pt DU
  var Pt UD: gameObject = Shot Mantis Lord/Pt UD
}
