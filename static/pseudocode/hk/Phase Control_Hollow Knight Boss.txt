# Phase Control
# 1 location(s):
#   Room_Final_Boss_Core / Boss Control/Hollow Knight Boss

fsm Phase Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Boss Corpse", storeResult=var "Battle Corpse")
    on FINISHED → Idle 1
  }

  state Idle 1 {
    on TOOK DAMAGE → Pause 1
  }

  state Check 1 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Phase2 HP", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Set Phase 2
    on FINISHED → Idle 1
  }

  state Set Phase 2 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=2, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time Min", setValue=0.15, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time Max", setValue=0.3, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Evade Chance", setValue=60, everyFrame=false)
    on FINISHED → Idle 2
  }

  state Idle 2 {
    on TOOK DAMAGE → Pause 2
  }

  state Check 2 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Phase3 HP", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Set Phase 3
    on FINISHED → Idle 2
  }

  state Set Phase 3 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=3, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time Min", setValue=0, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time Max", setValue=0.25, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Evade Chance", setValue=15, everyFrame=false)
    on FINISHED → Idle 3
  }

  state Idle 3 {
    on TOOK DAMAGE → Pause 3
  }

  state Check 4 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=var "Phase4 HP", equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Set Phase 4
    on FINISHED → Music Idle 2
  }

  state Set Phase 4 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=4, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time Min", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Slash Speed", setValue=0, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time Max", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Evade Chance", setValue=-1, everyFrame=false)
    SetHP(target=Self, hp=250)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotShadeCharm", isTrue=→"HORNET READY", isFalse=(none))
    on FINISHED → Idle 4
    on HORNET READY → Waiting for Hornet Scene
  }

  state Pause 1 {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check 1
  }

  state Pause 2 {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check 2
  }

  state Pause 3 {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check 6
  }

  state Idle 4 {
    on TOOK DAMAGE → Pause 4
  }

  state Check 5 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Die
    on FINISHED → Idle 4
  }

  state Pause 4 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pause 6
  }

  state Die {
    SetParent(gameObject=var "Battle Corpse", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ALL CHARMS END", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="killedHollowKnight", value=true)
    SetPlayerDataInt(intName="killsHollowKnight", value=0)
    ActivateGameObject(gameObject=var "Battle Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=StoryRecord_defeated("Hollow Knight"))
  }

  state Check 6 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=275, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → HK DECLINE 2
    on FINISHED → Idle 3
  }

  state HK DECLINE 2 {
    TransitionToAudioSnapshot(snapshot=HK Decline 2 (resources.assets), transitionTime=2)
    on FINISHED → Music Idle 1
  }

  state Music Idle 1 {
    on TOOK DAMAGE → Pause 5
  }

  state Pause 5 {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check 7
  }

  state Check 7 {
    GetHP(target=Self, storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=150, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → HK DECLINE 3
    on FINISHED → Music Idle 1
  }

  state HK DECLINE 3 {
    TransitionToAudioSnapshot(snapshot=HK Decline 3 (resources.assets), transitionTime=2)
    on FINISHED → Music Idle 2
  }

  state Music Idle 2 {
    on TOOK DAMAGE → Check 4
  }

  state Pause 6 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check 5
  }

  state Waiting for Hornet Scene {
    on HORNET CANCELLED → Hornet Cancel
  }

  state Hornet Cancel {
    SetHP(target=Self, hp=250)
    on FINISHED → Idle 4
  }

  var Phase2 HP: int = 750
  var Phase3 HP: int = 400
  var Phase4 HP: int = 0

  var HP: int = 0
  var Royal Charm State: int = 0
  var Battle Corpse: gameObject = <null>
}
