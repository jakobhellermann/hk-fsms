# Control
# 1 location(s):
#   Crossroads_ShamanTemple / Battle Scene/Reminder Cast

fsm Control {
  start Pause
  on BLOCKER DAMAGED → Inert  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Wait
  }

  state Send {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REMINDER FIREBALL", delay=0, everyFrame=false)
  }

  state Wait {
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Soul?
  }

  state Inert {
  }

  state Soul? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="MPCharge", storeValue=var "MP")
    IntCompare(integer1=var "MP", integer2=33, equal=→"YES", lessThan=→"NO", greaterThan=→"YES", everyFrame=false)
    on YES → Send
    on NO → Rewait
  }

  state Rewait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Soul?
  }

  var MP: int = 0
  var Activated: bool = false
}
