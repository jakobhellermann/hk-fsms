fsm Charm Unequip {
  start Idle
  on CHARM EQUIP CHECK → Check  // from any state

  state Idle {
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_40", isTrue=→"EQUIPPED", isFalse=→"UNEQUIPPED")
    on EQUIPPED → Idle
    on UNEQUIPPED → Unequipped
  }

  state Unequipped {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DESPAWN", delay=0, everyFrame=false)
  }

  var Scene Name: string = ""
}
