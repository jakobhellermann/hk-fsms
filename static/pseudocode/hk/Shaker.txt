# Shaker
# 2 location(s):
#   Mines_19 / Shake Region
#   Mines_37 / Shake Region

fsm Shaker {
  start Idle

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Wait
  }

  state Wait {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    Wait(time=var "Interval", finishEvent=→"SHAKE", realTime=false)
    on EXIT → Idle
    on SHAKE → Shake
  }

  state Shake {
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")  // disabled
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent=var "Event", delay=0, everyFrame=false)
    on FINISHED → Wait
    on EXIT → Idle
  }

  var Interval: float = 0.8
  var Event: string = "EnemyKillShake"
}
