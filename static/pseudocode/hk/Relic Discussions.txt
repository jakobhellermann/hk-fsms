# Relic Discussions
# 1 location(s):
#   Ruins1_05b / Relic Dealer

fsm Relic Discussions {
  start Init
  on TALK FINISH → Anim End  // from any state
  on RESET CONVO → Idle  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on SOLD TRINKET 1 → Set Prefix 1
    on SOLD TRINKET 2 → Set Prefix 2
    on SOLD TRINKET 3 → Set Prefix 3
    on SOLD TRINKET 4 → Set Prefix 4
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    CreateObject(gameObject=Area Title (sharedassets9.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Title", networkInstantiate=false, networkGroup=0)  // disabled
    SetFsmBool(gameObject=var "Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)  // disabled
    SetFsmString(gameObject=var "Title", fsmName="", variableName="Area Event", setValue="RELICDEALER", everyFrame=false)  // disabled
    on FINISHED → Convo
  }

  state Anim End {
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)  // disabled
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SHOP OPEN AUTO", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Convo {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Convo Name", "Relic Dealer"], storeResult=(unset var))
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Relic_Dealer_02 (sharedassets93.assets), Relic_Dealer_01 (sharedassets10.assets), Relic_Dealer_03 (sharedassets10.assets), Relic_Dealer_05 (sharedassets93.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Set Prefix 1 {
    SetStringValue(stringVariable=var "Convo Prefix", stringValue="RELICDEALER_JOURNAL_", everyFrame=false)
    SetStringValue(stringVariable=var "PD Int", stringValue="soldTrinket1", everyFrame=false)
    SetIntValue(intVariable=var "Clamp Max", intValue=8, everyFrame=false)
    on FINISHED → Check Sold Amt
  }

  state Set Prefix 2 {
    SetStringValue(stringVariable=var "Convo Prefix", stringValue="RELICDEALER_SEAL_", everyFrame=false)
    SetStringValue(stringVariable=var "PD Int", stringValue="soldTrinket2", everyFrame=false)
    SetIntValue(intVariable=var "Clamp Max", intValue=7, everyFrame=false)
    on FINISHED → Check Sold Amt
  }

  state Set Prefix 3 {
    SetStringValue(stringVariable=var "Convo Prefix", stringValue="RELICDEALER_IDOL_", everyFrame=false)
    SetStringValue(stringVariable=var "PD Int", stringValue="soldTrinket3", everyFrame=false)
    SetIntValue(intVariable=var "Clamp Max", intValue=5, everyFrame=false)
    on FINISHED → Check Sold Amt
  }

  state Set Prefix 4 {
    SetStringValue(stringVariable=var "Convo Prefix", stringValue="RELICDEALER_EGG_", everyFrame=false)
    SetStringValue(stringVariable=var "PD Int", stringValue="soldTrinket4", everyFrame=false)
    SetIntValue(intVariable=var "Clamp Max", intValue=4, everyFrame=false)
    on FINISHED → Check Sold Amt
  }

  state Check Sold Amt {
    IncrementPlayerDataInt(gameObject=var "GameManager", intName=var "PD Int")
    GetPlayerDataInt(gameObject=var "GameManager", intName=var "PD Int", storeValue=var "Sold Amt")
    IntClamp(intVariable=var "Sold Amt", minValue=0, maxValue=var "Clamp Max", everyFrame=false)
    ConvertIntToString(intVariable=var "Sold Amt", stringVariable=var "Sold Amt String", format="", everyFrame=false)
    BuildString(stringParts=[var "Convo Prefix", var "Sold Amt String"], separator="", addToEnd=true, storeResult=var "Convo Name", everyFrame=false)
    on FINISHED → Pause
  }

  var Clamp Max: int = 0
  var Sold: int = 0
  var Sold Amt: int = 0
  var Convo Name: string = ""
  var Convo Prefix: string = ""
  var PD Int: string = ""
  var Sold Amt String: string = ""
  var Self: gameObject = <null>
  var Title: gameObject = <null>
  var Voice Player: gameObject = <null>
}
