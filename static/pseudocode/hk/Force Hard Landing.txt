# Force Hard Landing
# 1 location(s):
#   Crossroads_01 / Force Hard Landing

fsm Force Hard Landing {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    Wait(time=2, finishEvent=→"DEACTIVATE", realTime=false)
    on ENTER → Set
    on DEACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="ForceHardLanding", parameters=[], storeResult=(unset var))
    on FINISHED → Deactivate
  }
}
