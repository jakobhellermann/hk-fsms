# Shade Control
# 1 location(s):
#   resources.assets / Hollow Shade Depart

fsm Shade Control {
  start Leave Pause
  on HERO LEAVE → Leave Pause  // from any state

  state Leave Pause {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Killed", isTrue=→"KILLED", isFalse=(none), everyFrame=false)
    DestroyObject(gameObject=var "Slash", delay=0, detachChildren=false)  // disabled
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=var "Y Speed", space=0, everyFrame=true)
    FloatMultiplyV2(floatVariable=var "X Speed", multiplyBy=0.9, everyFrame=true, fixedUpdate=true)
    FloatMultiplyV2(floatVariable=var "Y Speed", multiplyBy=0.9, everyFrame=true, fixedUpdate=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=var "Y Speed", everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=false)
    Wait(time=2, finishEvent=→"RETURN", realTime=false)
    on RETURN → Depart
  }

  state Depart {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Depart", animationTriggerEvent=(none), animationCompleteEvent=(none))
    iTweenScaleTo(gameObject=var "Big Light", id="", transformScale=var "", vectorScale=(0.2, 0.2, 0.2), time=0.6, delay=0, speed=(unset), easeType=linear, loopType=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    ActivateGameObject(gameObject=var "Big Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    Wait(time=0.6, finishEvent=→"WAIT", realTime=false)
    on WAIT → Particles
  }

  state Particles {
    StopParticleEmitter(gameObject=var "Shade Particles")
    SetMeshRenderer(gameObject=Self, active=false)
    PlayParticleEmitter(gameObject=var "Depart Particles", emit=0)
    ActivateGameObject(gameObject=var "Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Big Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var X Speed: float = 0
  var Y Speed: float = 0
  var Killed: bool = false
  var Big Light: gameObject = <null>
  var Depart Particles: gameObject = <null>
  var Light: gameObject = <null>
  var Self: gameObject = <null>
  var Shade Particles: gameObject = <null>
  var Slash: gameObject = <null>
}
