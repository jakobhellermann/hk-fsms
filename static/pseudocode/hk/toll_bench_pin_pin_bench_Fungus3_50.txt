fsm toll_bench_pin {
  start Check

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "PlayerData Bool", isTrue=→"ACTIVATE", isFalse=→"INACTIVE")
    on ACTIVATE → Activate
    on INACTIVE → Inactive
  }

  state Activate {
    SetSpriteRendererSprite(gameObject=Self, sprite=pin_bench)
  }

  state Inactive {
    SetSpriteRendererSprite(gameObject=Self, sprite=pin_bench_grey)
  }

  var PlayerData Bool: string = "tollBenchQueensGardens"
}
