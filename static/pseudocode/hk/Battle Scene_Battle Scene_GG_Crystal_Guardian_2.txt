# Battle Scene
# 1 location(s):
#   GG_Crystal_Guardian_2 / Battle Scene

fsm Battle Scene {
  start Init

  state Init {
    FindGameObject(objectName="Dummy Bench", withTag="Untagged", store=var "Dummy Bench")  // disabled
    FindGameObject(objectName="RestBench", withTag="RespawnPoint", store=var "Rest Bench")  // disabled
    on FINISHED → Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
  }

  state Detect {
    ActivateGameObject(gameObject=var "Rest Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Dummy Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="killsMegaBeamMiner", storeValue=var "KillsMegaBeamMiner")  // disabled
    IntCompare(integer1=var "KillsMegaBeamMiner", integer2=0, equal=→"ACTIVATED", lessThan=→"ACTIVATED", greaterThan=(none), everyFrame=false)  // disabled
    on ACTIVATED → Activate
    on KILLED → Wait
  }

  state Activate {
    ActivateGameObject(gameObject=var "Rest Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Dummy Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
  }

  state Wait {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="StopRumble", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Open
  }

  state Open {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Rest Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Dummy Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
    FindGameObject(objectName="Cam Lock", withTag="Untagged", store=var "Cam Lock")
    DestroyObject(gameObject=var "Cam Lock", delay=0, detachChildren=false)
  }

  var KillsMegaBeamMiner: int = 0
  var Activated: bool = false
  var Cam Lock: gameObject = <null>
  var Dummy Bench: gameObject = <null>
  var Rest Bench: gameObject = <null>
}
