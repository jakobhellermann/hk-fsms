# Conversation Control
# 1 location(s):
#   Room_temple / Quirrel

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Stop Anim  // from any state

  state Idle {
    on CONVO START → Hero Look
  }

  state Convo Choice {
    GetPlayerDataBool(boolName="metQuirrel", storeValue=var "Met")
    BoolTest(boolVariable=var "Met", isTrue=(none), isFalse=→"MEET", everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"TEMPLE", delay=0, everyFrame=false)
    on MEET → Meet
    on TEMPLE → Egg Temple
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="LookUpEnd", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Hero Look {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")  // disabled
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Meet {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=false, everyFrame=false)
    SetPlayerDataBool(boolName="metQuirrel", value=true)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_MEET_TEMPLE", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Meet B
  }

  state Egg Temple {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="QUIRREL", everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="quirrelEggTemple", storeValue=var "Egg Temple")
    IntSwitch(intVariable=var "Egg Temple", compareTo=[0, 1, 2, 3, 4, 5], sendEvent=[→"EGG 1", →"EGG 2", →"EGG 3", →"EGG 4", →"EGG 5", →"EGG 6"], everyFrame=false)
    on EGG 1 → Egg Temple 1
    on EGG 2 → Egg Temple 2
    on EGG 3 → Egg Temple 3
    on EGG 4 → Egg Temple 4
    on FINISHED → Egg Temple 4
  }

  state Egg Temple 1 {
    SetPlayerDataInt(intName="quirrelEggTemple", value=1)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEMPLE_1", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Stop Anim {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Play Talk End", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    on FINISHED → Box Down
  }

  state Egg Temple 2 {
    SetPlayerDataInt(intName="quirrelEggTemple", value=2)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEMPLE_4", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Egg Temple 3 {
    SetPlayerDataInt(intName="quirrelEggTemple", value=3)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_01 (sharedassets19.assets), Quirrel_02 (sharedassets19.assets), Quirrel_03 (sharedassets19.assets), Quirrel_04 (sharedassets19.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEMPLE_5", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Egg Temple 4 {
    SetPlayerDataInt(intName="quirrelEggTemple", value=4)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Quirrel_05_hmmm (sharedassets19.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SetBoolValue(boolVariable=var "Play Talk End", boolValue=true, everyFrame=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")  // disabled
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_TEMPLE_6", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Meet B {
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="QUIRREL", everyFrame=false)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_MEET_TEMPLE_B", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Meet C
  }

  state Meet C {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["QUIRREL_MEET_TEMPLE_C", "Quirrel"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Self X Scale: float = 0
  var Egg Temple: int = 0
  var Elderbug State: int = 0
  var Met: bool = false
  var Play Talk End: bool = false
  var Reintro Played: bool = false
  var Visited Crossroads: bool = false
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
