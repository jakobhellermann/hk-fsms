# Control
# 2 location(s):
#   Fungus1_04_boss / Cutscene Dreamer
#   sharedassets133.assets / Cutscene Dreamer

fsm Control {
  start Init
  on FORM → To Form  // from any state
  on QUICK ORB → Orb  // from any state
  on ORB → To Orb  // from any state
  on QUICK OFF → Off  // from any state
  on QUICK FORM → Form  // from any state

  state Init {
    FindChild(gameObject=var "Sprite", childName="Idle Pt", storeResult=var "Idle Pt")
    BuildString(stringParts=[var "Name", " Idle"], separator="", addToEnd=true, storeResult=var "Idle Anim", everyFrame=false)
    BuildString(stringParts=[var "Name", " Cast"], separator="", addToEnd=true, storeResult=var "Cast Anim", everyFrame=false)
    on FINISHED → Off
  }

  state Off {
    SetSpriteRenderer(gameObject=var "Sprite", active=false)
    StopParticleEmitter(gameObject=var "Idle Pt")
    StopParticleEmitter(gameObject=var "Orb Pt")
    StopParticleEmitter(gameObject=var "Trail Pt")
  }

  state Orb {
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    StopParticleEmitter(gameObject=var "Idle Pt")
    PlayParticleEmitter(gameObject=var "Orb Pt", emit=0)
    PlayParticleEmitter(gameObject=var "Trail Pt", emit=0)
  }

  state To Orb {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Blast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Burst Pt", emit=0)
    on FINISHED → Orb
  }

  state To Form {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Blast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Burst Pt", emit=0)
    on FINISHED → Form
  }

  state Form {
    SetMeshRenderer(gameObject=var "Sprite", active=true)
    PlayParticleEmitter(gameObject=var "Idle Pt", emit=0)
    StopParticleEmitter(gameObject=var "Orb Pt")
    StopParticleEmitter(gameObject=var "Trail Pt")
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName=var "Idle Anim")
    on CAST → Cast
  }

  state Cast {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName=var "Cast Anim")
    on IDLE → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName=var "Idle Anim")
    on CAST → Cast
  }

  var Name: string = ""

  var Cast Anim: string = ""
  var Idle Anim: string = ""
  var Blast: gameObject = Cutscene Dreamer/Blast
  var Burst Pt: gameObject = Cutscene Dreamer/Burst Pt
  var Idle Pt: gameObject = <null>
  var Orb Pt: gameObject = Cutscene Dreamer/Orb Pt
  var Sprite: gameObject = Cutscene Dreamer/Sprite
  var Trail Pt: gameObject = Cutscene Dreamer/Trail Pt
  var White Flash: gameObject = Cutscene Dreamer/White Flash
}
