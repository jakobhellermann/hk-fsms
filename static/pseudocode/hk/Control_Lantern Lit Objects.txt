# Control
# 1 location(s):
#   Cliffs_06 / Nightmare Lantern/lantern_dream/Lantern Lit Objects

// uses template: deactivate_ifnot_playerdatabool
fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=(none), isFalse=→"DEACTIVATE")
    on DEACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var playerData bool: string = "nightmareLanternLit"
}
