# Control
# 1 location(s):
#   Cutscene_Boss_Door / Mask Break Cutscene

fsm Control {
  start Check

  state Check {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="maskToBreak", storeValue=var "Mask To Break")
    IntSwitch(intVariable=var "Mask To Break", compareTo=[0, 1, 2], sendEvent=[→"LURIEN", →"HEGEMOL", →"MONOMON"], everyFrame=false)
    on HEGEMOL → Beast
    on LURIEN → Lurien
    on MONOMON → Monomon
  }

  state Beast {
    SetPlayerDataBool(boolName="maskBrokenHegemol", value=true)
    ActivateGameObject(gameObject=var "Mask Break Beast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Shake Pause
  }

  state Lurien {
    ActivateGameObject(gameObject=var "Mask Break Watcher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="maskBrokenLurien", value=true)
    on FINISHED → Shake Pause
  }

  state Monomon {
    ActivateGameObject(gameObject=var "Mask Break Teacher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="maskBrokenMonomon", value=true)
    on FINISHED → Shake Pause
  }

  state State 1 {
  }

  state Shake {
    Wait(time=4.8, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    on FINISHED → Fade
  }

  state Return {
    SetFsmBool(gameObject=var "MainCamera", fsmName="CameraFade", variableName="No Fade", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="ProxyFSM", variableName="No Charms", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Return", variableName="Dream Returning", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    GetPlayerDataString(gameObject=var "GameManager", stringName="dreamReturnScene", storeValue=var "Return Scene")
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ChangeToScene", parameters=[var "Return Scene", "door_dreamReturn", 0], storeResult=(unset var))  // disabled
    BeginSceneTransition(sceneName=var "Return Scene", entryGateName="door_dreamReturn", entryDelay=0, visualization=Dream, preventCameraFadeOut=true)
  }

  state Shake Pause {
    Wait(time=5.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  state Fade {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=3.1, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=3)
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=3)
    on FINISHED → Return
  }

  var Mask Break Beast: gameObject = Mask Break Cutscene/Break Masks/Mask Break Beast
  var Mask Break Teacher: gameObject = Mask Break Cutscene/Break Masks/Mask Break Teacher
  var Mask Break Watcher: gameObject = Mask Break Cutscene/Break Masks/Mask Break Watcher

  var Mask To Break: int = 0
  var Return Scene: string = ""
}
