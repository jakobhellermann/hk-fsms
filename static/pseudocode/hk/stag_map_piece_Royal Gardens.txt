// uses template: stag_map_piece
fsm stag_map_piece {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Always Active", isTrue=→"OPENED", isFalse=(none), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "Playerdata Bool", isTrue=→"OPENED", isFalse=→"UNOPENED")
    on UNOPENED → Unopened
    on OPENED → Opened
  }

  state Unopened {
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state Opened {
    SetSpriteRenderer(gameObject=Self, active=true)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Colour", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    on STAG MENU DOWN → Close
  }

  state Close {
    SetSpriteRenderer(gameObject=Self, active=true)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 1), everyFrame=false)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
  }

  var Always Active: bool = false
  var Playerdata Bool: string = "openedRoyalGardens"

  var Colour: color = (1, 1, 1, 0)
  var Self: gameObject = <null>
}
