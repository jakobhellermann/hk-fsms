# Gate Control
# 1 location(s):
#   Fungus2_23 / Waterways Gate

// uses template: destroy_if_playerdatabool
fsm Gate Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "playerData bool", isTrue=→"DESTROY", isFalse=(none))
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var playerData bool: string = "waterwaysGate"
}
