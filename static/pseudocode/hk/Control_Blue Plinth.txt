# Control
# 1 location(s):
#   Abyss_06_Core / Blue Plinth

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue Health")
    IntCompare(integer1=var "Blue Health", integer2=0, equal=(none), lessThan=(none), greaterThan=→"READY", everyFrame=false)
    on READY → Ready
  }

  state Init {
    FindChild(gameObject=Self, childName="level1", storeResult=var "level1")
    FindChild(gameObject=Self, childName="level2", storeResult=var "level2")
    FindChild(gameObject=Self, childName="White Flash", storeResult=var "White Flash")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gotCharm_9", isTrue=→"GOT CHARM", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="blueVineDoor", isTrue=→"END", isFalse=(none))
    on FINISHED → Check
    on END → Completed
    on GOT CHARM → Got Charm
  }

  state Ready {
    BoolTest(boolVariable=var "Hero Range", isTrue=→"ENTER", isFalse=(none), everyFrame=true)
    on ENTER → Start Pause
  }

  state Glow Start {
    SendEventByName(eventTarget=GameObject(var "level1"), sendEvent="UP", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "White Flash", audioClip=white_seal_1 (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Wait For Enter
  }

  state Blue Health? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue Health")
    IntCompare(integer1=var "Blue Health", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Glow Start
    on CANCEL → Not Enough
  }

  state Light Next? {
    GetPlayerDataInt(gameObject=var "GameManager", intName="healthBlue", storeValue=var "Blue Health")
    IntCompare(integer1=var "Blue Health", integer2=var "Current Number", equal=→"LIGHT", lessThan=→"END", greaterThan=→"LIGHT", everyFrame=false)
    on LIGHT → Send Event
    on END → Not Enough
  }

  state Not Enough {
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LIGHT", delay=0, everyFrame=false)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Increment
  }

  state Wait For Enter {
    BoolTest(boolVariable=var "Hero Range", isTrue=→"ENTER", isFalse=(none), everyFrame=true)
    on ENTER → Light Next?
  }

  state Increment {
    IntCompare(integer1=var "Current Number", integer2=var "Lights Total", equal=→"COMPLETE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntAdd(intVariable=var "Current Number", add=1, everyFrame=false)
    on FINISHED → Light Next?
    on COMPLETE → Complete Pause
  }

  state Complete Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Final Glow
  }

  state Start Pause {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Blue Health?
  }

  state Open Door {
    SetPlayerDataBool(boolName="blueVineDoor", value=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="OPEN VINE DOOR", delay=0, everyFrame=false)
  }

  state Final Glow {
    SendEventByName(eventTarget=GameObject(var "level2"), sendEvent="UP", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "White Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open Door
  }

  state Completed {
    SendEventByName(eventTarget=GameObject(var "level2"), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LIGHT ALL", delay=0, everyFrame=false)
  }

  state Got Charm {
  }

  var Lights Total: int = 14

  var Blue Health: int = 0
  var Current Number: int = 1
  var Hero Range: bool = false
  var White Flash: gameObject = <null>
  var level1: gameObject = <null>
  var level2: gameObject = <null>
  var Vines Grey: gameObject = <null>
}
