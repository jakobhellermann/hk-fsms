# Control
# 1 location(s):
#   Crossroads_10 / Fk Break Wall/Escape Checker

fsm Control {
  start Pause

  state Init {
    FindGameObject(objectName="False Knight New", withTag="Untagged", store=var "False Knight")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="falseKnightWallRepaired", isTrue=→"DISABLE", isFalse=→"FINISHED")
    on FINISHED → Idle
    on DISABLE → Disable
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Set
  }

  state Set {
    SetFsmBool(gameObject=var "False Knight", fsmName="FalseyControl", variableName="Hero Escaped", setValue=true, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Disable {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var False Knight: gameObject = <null>
}
