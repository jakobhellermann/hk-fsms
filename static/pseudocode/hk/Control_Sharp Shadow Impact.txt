# Control
# 1 location(s):
#   resources.assets / Sharp Shadow Impact

fsm Control {
  start Activate

  state Activate {
    GetOwner(storeGameObject=var "Self")
    RandomFloat(min=-15, max=15, storeResult=var "Rotation")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Rotation: float = 0
  var Self: gameObject = <null>
}
