// uses template: spatter_health
fsm Spatter Healer {
  start Init

  state Fall {
    GetOwner(storeGameObject=var "Self")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    RandomFloat(min=1.3, max=1.6, storeResult=var "Randomiser")
    FindChild(gameObject=Self, childName="Health Get Effect", storeResult=var "Get Effect")
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=1.4, stretchMinX=0.6, stretchMaxY=1.75, scaleModifier=var "Randomiser", everyFrame=true)
    WaitRandom(timeMin=0.25, timeMax=0.45, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Decel
  }

  state Attract {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=1.4, stretchMinX=0.6, stretchMaxY=1.75, scaleModifier=var "Randomiser", everyFrame=true)
    FireAtTarget(gameObject=Self, target=var "Hero", speed=var "Accel", position=(0, -0.5, 0), spread=0, everyFrame=true)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=0, tolerance=0.6, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=true)
    FloatAdd(floatVariable=var "Accel", add=0.85, everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Accel", minValue=0, maxValue=30, everyFrame=true)
    on FINISHED → Get
  }

  state Get {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[health_bug_collect_1, health_bug_collect_2, health_bug_collect_3], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=flashHealBlue())
    RandomFloat(min=0, max=360, storeResult=var "Chooser")
    ActivateGameObject(gameObject=var "Get Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetRotation(gameObject=var "Get Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=2, max=2.5, storeResult=var "Chooser")
    SetScale(gameObject=var "Get Effect", vector=(unset), x=var "Chooser", y=var "Chooser", z=(unset), everyFrame=false, lateUpdate=false)
    SetParent(gameObject=var "Get Effect", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Decel {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    FaceAngle(gameObject=Self, angleOffset=0, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=1.4, stretchMinX=0.6, stretchMaxY=1.75, scaleModifier=var "Randomiser", everyFrame=true)
    DecelerateV2(gameObject=Self, deceleration=0.75)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=true)
    FloatCompare(float1=var "Speed", float2=0.3, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Attract
  }

  state Init {
    BoolTest(boolVariable=var "Init", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Init", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Fall
  }

  var Accel: float = 0.1
  var Chooser: float = 0
  var Distance: float = 0
  var Randomiser: float = 0
  var Speed: float = 0
  var Init: bool = false
  var Get Effect: gameObject = Spatter Blue Heal/Health Get Effect
  var Self: gameObject = <null>
}
