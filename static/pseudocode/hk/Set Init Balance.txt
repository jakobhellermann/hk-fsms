# Set Init Balance
# 2 location(s):
#   Fungus3_35 / Bank Menu/Txt Balance Amount
#   sharedassets218.assets / Bank Menu/Txt Balance Amount

fsm Set Init Balance {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set
  }

  state Set {
    GetPlayerDataInt(gameObject=var "GameManager", intName="bankerBalance", storeValue=var "Balance")
    ConvertIntToString(intVariable=var "Balance", stringVariable=var "Balance Str", format="", everyFrame=false)
    SetTextMeshProText(gameObject=Self, textString=var "Balance Str")
  }

  var Balance: int = 0
  var Balance Str: string = ""
}
