# Remind
# 1 location(s):
#   Tutorial_01 / Interact Reminder

fsm Remind {
  start Pause

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ENTER → State 1
    on ACTIVATE → Activate
  }

  state State 1 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="REMINDER INTERACT", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  state Pause {
    GetPlayerDataBool(boolName="visitedDirtmouth", storeValue=var "Visited Dirtmouth")
    BoolTest(boolVariable=var "Visited Dirtmouth", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on ACTIVATE → Visited Dirtmouth
  }

  state Activate {
  }

  state Visited Dirtmouth {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Activated: bool = false
  var Visited Dirtmouth: bool = false
}
