fsm Blow Fly {
  start Init
  on SLEEP → Sleep  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Sleep
  }

  state Idle {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=2, speedMax=2, accelerationMin=5, accelerationMax=15, roamingRangeX=4.5, roamingRangeY=4.5, manualStartPos=var "Start Pos")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on COLLISION → Play Anim
  }

  state Collision Side {
    ReflectAngle(angle=var "Angle", reflectHorizontally=true, reflectVertically=false, storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=5, everyFrame=false)
    on FINISHED → Wait
  }

  state Play Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Collision Side
  }

  state Wait {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    IdleBuzz(gameObject=Self, waitMin=5, waitMax=10, speedMax=20, accelerationMax=10, roamingRange=12)
    on FINISHED → Idle
  }

  state Sleep {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    AudioStop(gameObject=Self)
    on WAKE → Idle
  }

  var Angle: float = 0
  var Start Pos: vector3 = (0, 0, 0)
  var Other: gameObject = <null>
}
