# Guard
# 1 location(s):
#   White_Palace_11 / Royal Gaurd

fsm Guard {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Rest
  }

  state Rest {
    BoolTest(boolVariable=var "Hero In Range", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    on WAKE → Wake
  }

  state Wake {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_ghost_appear (resources.assets), pitchMin=0.8, pitchMax=0.8, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=kingsguard_intro_stomp (sharedassets366.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_shield_sentry_shield_move (sharedassets34.assets))
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Evade Range", isTrue=→"EVADE", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Slash Range", isTrue=→"SLASH", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Throw Range", isTrue=→"THROW", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    Wait(time=1, finishEvent=→"CHASE", realTime=false)
    on EVADE → Evade Check
    on THROW → Choose Throw
    on SLASH → Choose Slash
    on CHASE → Chase
  }

  state Wait {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    WaitRandom(timeMin=0.2, timeMax=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Throw Start {
    SetBoolValue(boolVariable=var "Will Throw", boolValue=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Throw", animationTriggerEvent=→"FINISHED", animationCompleteEvent=→"FINISHED")
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=-0.5, storeAngle=var "Fire Angle", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_shield_sentry_shield_move (sharedassets34.assets))
    on FINISHED → Throw
  }

  state Throw {
    CreateObject(gameObject=Shot Kings Guard (sharedassets366.assets), spawnPoint=var "Self", position=(0, -1.5, 0), rotation=(unset), storeObject=var "Boomerang", networkInstantiate=false, networkGroup=0)
    SetFsmGameObject(gameObject=var "Boomerang", fsmName="Spin", variableName="Parent", setValue=var "Self", everyFrame=false)
    SetFsmFloat(gameObject=var "Boomerang", fsmName="Spin", variableName="Fire Angle", setValue=var "Fire Angle", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=mantis_heavy_flyer_projectile_throw (sharedassets34.assets))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=true)
    on CAUGHT → Throw Catch
  }

  state Throw Catch {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Throw Catch", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hornet_needle_catch (sharedassets133.assets))
    on FINISHED → Wait
  }

  state Slash Antic {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_shield_sentry_prepare (sharedassets34.assets))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Facing", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Facing", multiplyBy=-1, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Slash Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Slash Step
  }

  state Slash1 Attack {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Slash", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Slash 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_shield_sentry_sword_1 (sharedassets34.assets))
    on FINISHED → Slash 2
  }

  state Slash1 Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Slash Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Slash Step {
    SetFloatValue(floatVariable=var "CurrentSlashSpeed", floatValue=var "SlashStepSpeed", everyFrame=false)
    FloatMultiplyV2(floatVariable=var "CurrentSlashSpeed", multiplyBy=var "Facing", everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "CurrentSlashSpeed", y=(unset), everyFrame=false)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slash1 Attack
  }

  state Evade Check {
    BoolTest(boolVariable=var "Evade Check Range", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Evade")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on LEFT → Evade Left
    on RIGHT → Evade Right
    on FINISHED → Choose Evade
  }

  state Evade Left {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Evade Speed", floatValue=22, everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge1 Speed", floatValue=-13, everyFrame=false)
    on FINISHED → Evade 2
  }

  state Evade Right {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Evade Speed", floatValue=-22, everyFrame=false)
    SetFloatValue(floatVariable=var "Lunge1 Speed", floatValue=13, everyFrame=false)
    on FINISHED → Evade 2
  }

  state Evade 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=zombie_shield_raise (sharedassets32.assets))
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Evade Speed", y=(unset), everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Evade End
  }

  state Evade End {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hero_land_hard (sharedassets76.assets))  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hero_land_hard (sharedassets76.assets), pitchMin=1.2, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Choose Evade
  }

  state Charge {
    AudioPlayInState(gameObject=Self, volume=1)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Facing", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetFloatValue(floatVariable=var "Speed", floatValue=var "RunSpeed", everyFrame=false)
    FloatMultiplyV2(floatVariable=var "Speed", multiplyBy=var "Facing", everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=true)
    Wait(time=1, finishEvent=→"THROW", realTime=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"FINISHED", bottomHitEvent=(none), leftHitEvent=→"FINISHED", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolTest(boolVariable=var "Ground Detected", isTrue=(none), isFalse=→"FINISHED", everyFrame=true)  // disabled
    BoolTest(boolVariable=var "Throw Range", isTrue=→"THROW", isFalse=(none), everyFrame=true)  // disabled
    BoolTest(boolVariable=var "Slash Range", isTrue=→"SLASH", isFalse=(none), everyFrame=true)
    on THROW → Throw Start
    on SLASH → Slash Antic
    on FINISHED → Wait
  }

  state Choose Throw {
    SendRandomEvent(events=[→"THROW", →"CHARGE"], weights=[0.75, 0.25], delay=0)  // disabled
    SendRandomEventV2(events=[→"THROW", →"CHARGE"], weights=[1, 1], trackingInts=[var "Ct Throw", var "Ct Chase"], eventMax=[2, 1])
    on THROW → Throw Start
    on CHARGE → Charge
  }

  state Choose Move {
    SendRandomEvent(events=[→"EVADE", →"MOVE"], weights=[1, 1], delay=0)
    on EVADE → Evade Check
    on MOVE → Charge
  }

  state Choose Evade {
    BoolTest(boolVariable=var "Will Throw", isTrue=→"THROW", isFalse=(none), everyFrame=false)
    SendRandomEvent(events=[→"FINISHED", →"THROW", →"SLASH"], weights=[0.4, 0.2, 0.2], delay=0)
    on FINISHED → Wait
    on THROW → Throw Start
    on SLASH → Slash Antic
  }

  state Choose Slash {
    SendRandomEvent(events=[→"SLASH", →"THROW"], weights=[0.65, 0.35], delay=0)
    on SLASH → Slash Antic
    on THROW → Set Throw
  }

  state Choose Evade 2 {
    SendRandomEvent(events=[→"FINISHED", →"EVADE"], weights=[1, 1], delay=0)  // disabled
    on FINISHED → Wait
    on EVADE → Evade Check
  }

  state Set Throw {
    SetBoolValue(boolVariable=var "Will Throw", boolValue=true, everyFrame=false)
    on FINISHED → Evade Check
  }

  state Slash 2 {
    ActivateGameObject(gameObject=var "Slash 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ruin_shield_sentry_sword_2 (sharedassets34.assets))
    on FINISHED → Slash1 Recover
  }

  state Chase {
    AudioPlayInState(gameObject=Self, volume=1)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Facing", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetFloatValue(floatVariable=var "Speed", floatValue=var "RunSpeed", everyFrame=false)
    FloatMultiplyV2(floatVariable=var "Speed", multiplyBy=var "Facing", everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=true)
    Wait(time=2, finishEvent=→"THROW", realTime=false)
    CheckCollisionSide(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"FINISHED", bottomHitEvent=(none), leftHitEvent=→"FINISHED", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    BoolTest(boolVariable=var "Ground Detected", isTrue=(none), isFalse=→"FINISHED", everyFrame=true)  // disabled
    BoolTest(boolVariable=var "Slash Range", isTrue=→"SLASH", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Throw Range", isTrue=→"THROW", isFalse=(none), everyFrame=true)
    on SLASH → Choose Slash
    on THROW → Choose Throw
  }

  var CurrentSlashSpeed: float = 0
  var Evade Speed: float = 0
  var Facing: float = 0
  var Fire Angle: float = 0
  var Lunge1 Speed: float = 0
  var RunSpeed: float = -10
  var SlashStepSpeed: float = 20
  var Speed: float = 0
  var Ct Chase: int = 0
  var Ct Throw: int = 0
  var Evade Check Range: bool = false
  var Evade Range: bool = false
  var Ground Detected: bool = false
  var Hero In Range: bool = false
  var Slash Range: bool = false
  var Throw Range: bool = false
  var Will Throw: bool = false
  var Boomerang: gameObject = <null>
  var Self: gameObject = <null>
  var Slash 1: gameObject = Royal Gaurd/Slash 1
  var Slash 2: gameObject = Royal Gaurd/Slash 2
}
