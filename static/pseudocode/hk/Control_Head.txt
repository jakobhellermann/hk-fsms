# Control
# 1 location(s):
#   sharedassets481.assets / Zote Crew Death/Head

fsm Control {
  start State 3

  state State 1 {
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    FlingObject(flungObject=Self, speedMin=20, speedMax=25, angleMin=var "Min", angleMax=var "Max")
    Wait(time=1.6, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=true)
    on FINISHED → State 2
  }

  state State 2 {
    SetCollider(gameObject=Self, active=false)
  }

  state State 3 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on L → L
    on R → R
  }

  state R {
    SetFloatValue(floatVariable=var "Min", floatValue=100, everyFrame=false)
    SetFloatValue(floatVariable=var "Max", floatValue=130, everyFrame=false)
    on FINISHED → State 1
  }

  state L {
    SetFloatValue(floatVariable=var "Min", floatValue=50, everyFrame=false)
    SetFloatValue(floatVariable=var "Max", floatValue=80, everyFrame=false)
    on FINISHED → State 1
  }

  var Max: float = 0
  var Min: float = 0
}
