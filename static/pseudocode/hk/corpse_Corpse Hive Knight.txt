fsm corpse {
  start Music

  state Init {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)

    SendEventToRegisterV2(EventName="BOSS DEFEATED", ExcludeTarget=Self)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on LAND → Pause
    on FINISHED → Pause
  }

  state Steam {
    FlingObjectsFromGlobalPoolTime(gameObject=Spatter Orange (resources.assets), spawnPoint=var "Self", position=(0, 0, 0.002), frequency=0.1, spawnMin=12, spawnMax=15, speedMin=24, speedMax=30, angleMin=0, angleMax=360, originVariationX=1, originVariationY=1)

    CreateObject(gameObject=Death Puff Boss (sharedassets32.assets), spawnPoint=var "Self", position=(0, 0, -5), rotation=(unset), storeObject=var "Smoker", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    FloatAdd(floatVariable=var "emissionRate", add=5, everyFrame=true, perSecond=false)
    FloatAdd(floatVariable=var "emissionSpeed", add=1, everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "emissionSpeed", minValue=0, maxValue=110, everyFrame=true)
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Ready
  }

  state Blow {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=boss_explode (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="StopRumble", delay=0, everyFrame=false)
    CreateObject(gameObject=Death Explode Boss (sharedassets32.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetParticleEmission(gameObject=var "Smoker", emission=false)
    PlayParticleEmitter(gameObject=var "Corpse Steam", emit=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out 2", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Appear Pause
  }

  state Ready {
    SetParticleEmissionRate(gameObject=var "Smoker", emissionRate=var "emissionRate", everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Smoker", emissionSpeed=var "emissionSpeed", everyFrame=true)
    EaseFloat(fromValue=var "emissionRate", toValue=0, floatVariable=var "emissionRate", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)

    on FINISHED → Blow
  }

  state Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun Air")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Steam
  }

  state BG Open {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BATTLE END", delay=0, everyFrame=false)
  }

  state Appear Pause {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=2.75, finishEvent=→"FINISHED", realTime=false)
    GGCheckIfBossScene(bossSceneEvent=→"GG BOSS", regularSceneEvent=(none))
    on FINISHED → Chinese Ver?
    on GG BOSS → End
  }

  state Tele {
    SetPosition(gameObject=Self, vector=(unset), x=72.2, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death In", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)

  }

    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hornet_ground_land (sharedassets28.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=1.5, finishEvent=→"LAND", realTime=false)
    on LAND → Collapse
  }

  state Collapse {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Collapse")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=brkn_wand_land (sharedassets32.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.09, storePlayer=<null>)
    Wait(time=2, finishEvent=→"LAND", realTime=false)
    on LAND → BG Open
  }

  state Music {
    GGCheckIfBossScene(bossSceneEvent=→"FINISHED", regularSceneEvent=(none))
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=2)
    on FINISHED → Init
  }

  state End {
  }

  state Chinese Ver? {
    CheckIsChineseBuild(trueEvent=→"CANCEL", falseEvent=→"FINISHED")
    on FINISHED → Tele
    on CANCEL → BG Open
  }

  var emissionRate: float = 50
  var emissionSpeed: float = 5
  var Corpse Steam: gameObject = Corpse Infected Knight/Corpse Steam (sharedassets343.assets)
  var Self: gameObject = <null>
  var Smoker: gameObject = <null>
}
