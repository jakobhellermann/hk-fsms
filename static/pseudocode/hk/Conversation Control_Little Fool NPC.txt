# Conversation Control
# 1 location(s):
#   Room_Colosseum_01 / Little Fool NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state
  on GATE OPEN → Gate Open  // from any state

  state Idle {
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="CamLock Bronze", storeResult=var "CamLock Bronze")
    FindChild(gameObject=Self, childName="CamLock Silver", storeResult=var "CamLock Silver")
    FindChild(gameObject=Self, childName="CamLock Gold", storeResult=var "CamLock Gold")
    on FINISHED → Idle
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="LITTLE_FOOL", everyFrame=false)
    on FINISHED → Opened?
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    AudioStop(gameObject=var "Voice Player")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Convo Choice {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="colosseumBronzeOpened", isTrue=(none), isFalse=→"OFFER BRONZE")
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="colosseumBronzeCompleted", falseBool="colosseumSilverOpened", isTrue=→"OFFER SILVER", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="colosseumSilverCompleted", falseBool="colosseumGoldOpened", isTrue=→"OFFER GOLD", isFalse=(none))
    SendEventByName(eventTarget=Self, sendEvent="READY", delay=0, everyFrame=false)
    on OFFER BRONZE → Offer Bronze
    on READY → Ready
    on OFFER SILVER → Offer Silver
    on OFFER GOLD → Offer Gold
  }

  state Meet? {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK", delay=0, everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="littleFoolMet", isTrue=→"NEXT", isFalse=(none))
    SetPlayerDataBool(boolName="littleFoolMet", value=true)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_MEET", "Minor NPC"], storeResult=(unset var))
    on NEXT → Ran Away
    on CONVO_FINISH → Convo Choice
  }

  state Greet? {
    BoolTest(boolVariable=var "Spoken", isTrue=→"NEXT", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_GREET", "Minor NPC"], storeResult=(unset var))
    on NEXT → Convo Choice
    on CONVO_FINISH → Convo Choice
  }

  state Offer Bronze {
    GetLanguageString(sheetName="Prices", convName="COLOSSEUM_BRONZE", storeValue=var "Price String")
    ConvertStringToInt(stringVariable=var "Price String", intVariable=var "Unlock Price", everyFrame=false)
    SetStringValue(stringVariable=var "Unlock Bool", stringValue="colosseumBronzeOpened", everyFrame=false)
    SetStringValue(stringVariable=var "Unlock Event", stringValue="UNLOCK BRONZE", everyFrame=false)
    SetStringValue(stringVariable=var "Prompt Convo", stringValue="TRIAL_UNLOCK_BRONZE", everyFrame=false)
    SetGameObject(variable=var "CamLock", gameObject=var "CamLock Bronze", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_OFFER_1", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Dial Box Down
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=var "Unlock Price", everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Prompt Convo", "Prompts"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pan
  }

  state Dial Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up YN
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enough Geo?
  }

  state Decline Pause {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    on FINISHED → Box Up 2
  }

  state Enough Geo? {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Little_Fool_talk_05 (sharedassets30.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK", delay=0, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="geo", storeValue=var "Player Geo")
    IntCompare(integer1=var "Player Geo", integer2=var "Unlock Price", equal=→"ENOUGH", lessThan=→"NOT ENOUGH", greaterThan=→"ENOUGH", everyFrame=false)
    on ENOUGH → Decline
    on NOT ENOUGH → Not Enough
  }

  state Unlocked {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Little_Fool_talk_01 (sharedassets30.assets), Little_Fool_talk_02 (sharedassets30.assets), Little_Fool_talk_03 (sharedassets30.assets), Little_Fool_talk_04 (sharedassets30.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_PAID", "Minor NPC"], storeResult=(unset var))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK", delay=0, everyFrame=false)
    on CONVO_FINISH → Talk Finish
  }

  state Box Up 3 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Unlocked
  }

  state Decline {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_REFUSE", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Not Enough {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_NOTENOUGH", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Open Event {
    SetPlayerDataBool(boolName=var "Unlock Bool", value=true)
    SendEventByName(eventTarget=BroadcastAll, sendEvent=var "Unlock Event", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up 3
  }

  state Pan {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Zote Event
  }

  state Gate Open {
    SetBoolValue(boolVariable=var "Gate Open", boolValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  state Gate Opened {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_CHALLENGE", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  state Opened? {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Little_Fool_talk_01 (sharedassets30.assets), Little_Fool_talk_02 (sharedassets30.assets), Little_Fool_talk_03 (sharedassets30.assets), Little_Fool_talk_04 (sharedassets30.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    BoolTest(boolVariable=var "Gate Open", isTrue=→"GATE OPENED", isFalse=(none), everyFrame=false)
    on FINISHED → Meet?
    on GATE OPENED → Gate Opened
  }

  state Ready {
    AudioStop(gameObject=var "Voice Player")
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClips=[Little_Fool_talk_01 (sharedassets30.assets)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="colosseumGoldCompleted", isTrue=→"CHAMP", isFalse=(none))
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_MARK", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
    on CHAMP → Ready Champ
  }

  state Offer Silver {
    GetLanguageString(sheetName="Prices", convName="COLOSSEUM_SILVER", storeValue=var "Price String")
    SetBoolValue(boolVariable=var "Zote Event", boolValue=true, everyFrame=false)
    ConvertStringToInt(stringVariable=var "Price String", intVariable=var "Unlock Price", everyFrame=false)
    SetStringValue(stringVariable=var "Unlock Bool", stringValue="colosseumSilverOpened", everyFrame=false)
    SetStringValue(stringVariable=var "Unlock Event", stringValue="UNLOCK SILVER", everyFrame=false)
    SetStringValue(stringVariable=var "Prompt Convo", stringValue="TRIAL_UNLOCK_SILVER", everyFrame=false)
    SetGameObject(variable=var "CamLock", gameObject=var "CamLock Silver", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_OFFER_2", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Dial Box Down
  }

  state Offer Gold {
    GetLanguageString(sheetName="Prices", convName="COLOSSEUM_GOLD", storeValue=var "Price String")
    ConvertStringToInt(stringVariable=var "Price String", intVariable=var "Unlock Price", everyFrame=false)
    SetStringValue(stringVariable=var "Unlock Bool", stringValue="colosseumGoldOpened", everyFrame=false)
    SetStringValue(stringVariable=var "Unlock Event", stringValue="UNLOCK GOLD", everyFrame=false)
    SetStringValue(stringVariable=var "Prompt Convo", stringValue="TRIAL_UNLOCK_GOLD", everyFrame=false)
    SetGameObject(variable=var "CamLock", gameObject=var "CamLock Gold", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_OFFER_3", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Dial Box Down
  }

  state Ran Away {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="ranAway", isTrue=(none), isFalse=→"NEXT")
    SetPlayerDataBool(boolName="ranAway", value=false)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "DialogueText", fsmName="Dialogue Page Control", variableName="Use Stop", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_COWARD", "Minor NPC"], storeResult=(unset var))
    on NEXT → Greet?
    on CONVO_FINISH → Convo Choice
  }

  state Zote Event {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="zoteRescuedDeepnest", isTrue=→"FINISHED", isFalse=→"FINISHED")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="zoteTrappedDeepnest", isTrue=→"KILL", isFalse=→"SAVE")  // disabled
    on FINISHED → Open Event
    on KILL → Kill Zote
    on SAVE → Save Zote
  }

  state Kill Zote {
    SetPlayerDataBool(boolName="zoteDead", value=true)
    SetPlayerDataInt(intName="zoteDeathPos", value=2)
    on FINISHED → Open Event
  }

  state Save Zote {
    SetPlayerDataBool(boolName="zoteRescuedDeepnest", value=true)
    on FINISHED → Open Event
  }

  state Ready Champ {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["LITTLE_FOOL_CHAMP", "Minor NPC"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Player Geo: int = 0
  var Unlock Price: int = 0
  var Gate Open: bool = false
  var Spoken: bool = false
  var Zote Event: bool = false
  var Price String: string = ""
  var Prompt Convo: string = ""
  var Unlock Bool: string = ""
  var Unlock Event: string = ""
  var Area Title: gameObject = <null>
  var CamLock: gameObject = <null>
  var CamLock Bronze: gameObject = <null>
  var CamLock Gold: gameObject = <null>
  var CamLock Self: gameObject = <null>
  var CamLock Silver: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
