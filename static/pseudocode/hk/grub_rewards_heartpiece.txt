# grub_rewards_heartpiece
# 1 location(s):
#   Crossroads_38 / Grub King/Rewards Parent/Reward 5

// uses template: grub_reward_heartpiece
fsm grub_rewards_heartpiece {
  start Pause

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=var "Loose Rewards", childName=var "Piece Name", storeResult=var "Piece")
    FindGameObject(objectName=var "Piece Name", withTag="Heart Piece", store=var "Piece")  // disabled
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Piece", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → Idle
    on ACTIVATED → Activated
  }

  state Idle {
    on GIVE → Give
  }

  state Give {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Piece", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmGameObject(gameObject=var "Piece", fsmName="Heart Container Control", variableName="Appear Source", setValue=var "Self", everyFrame=false)
    SetFsmBool(gameObject=var "Piece", fsmName="Heart Container Control", variableName="Appears", setValue=true, everyFrame=false)
  }

  state Pause {
    FindGameObject(objectName="Loose Rewards", withTag="Extra Tag", store=var "Loose Rewards")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activated {
    ActivateGameObject(gameObject=var "Piece", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Piece Name: string = "Heart Piece"

  var Activated: bool = false
  var Loose Rewards: gameObject = <null>
  var Piece: gameObject = <null>
  var Self: gameObject = <null>
}
