# collide_with_wall
# 1 location(s):
#   sharedassets73.assets / Shot PickAxe

fsm collide_with_wall {
  start Pause

  state In Air {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="HeroBox", collideLayer="", sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"COLLIDE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=Attack(17), sendEvent=→"NAIL", storeCollider=var "Nail Object")
    on COLLIDE → Impact
    on NAIL → Nail Hit
    on ORBIT SHIELD → Impact
  }

  state Impact {
    AudioStop(gameObject=Self)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=barrel_death_2 (sharedassets6.assets))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pickaxe Shatter", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Spin {
    AddTorque2d(gameObject=Self, forceMode=0, torque=var "Torque", everyFrame=false)
    on FINISHED → In Air
  }

  state Recycle {
    RecycleSelf()
  }

  state Flip? {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Torque", floatValue=700, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatCompare(float1=var "X Speed", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Torque", floatValue=-700, everyFrame=false)
    on FINISHED → Spin
  }

  state Pause {
    SetAudioPitch(gameObject=Self, pitch=1, everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-102.8539, space=0, everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack Projectile")
    ActivateGameObject(gameObject=var "Impact", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    SetPolygonCollider(gameObject=Self, active=true)
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=0, everyFrame=false)
    on FINISHED → Flip?
  }

  state Nail Hit {
    GetFsmFloat(gameObject=var "Nail Object", fsmName="damages_enemy", variableName="direction", storeValue=var "Direction", everyFrame=false)
    SetAudioPitch(gameObject=Self, pitch=1.2, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Direction", speed=30, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SetFsmInt(gameObject=Self, fsmName="damages_enemy", variableName="damageDealt", setValue=10, everyFrame=false)
    on FINISHED → In Air
  }

  var Direction: float = 0
  var Torque: float = 0
  var X Speed: float = 0
  var Impact: gameObject = Shot PickAxe/Strike Nail
  var Nail Object: gameObject = <null>
}
