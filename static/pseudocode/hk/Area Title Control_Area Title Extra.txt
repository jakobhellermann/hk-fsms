# Area Title Control
# 1 location(s):
#   sharedassets9.assets / Area Title Extra

fsm Area Title Control {
  start Pause
  on LEVEL LOADED → Destroy  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Title Small", storeResult=var "Small Title Folder")  // disabled
    FindChild(gameObject=var "Small Title Folder", childName="Title Small Main", storeResult=var "Small Title Main")  // disabled
    FindChild(gameObject=var "Small Title Folder", childName="Title Small Sub", storeResult=var "Small Title Sub")  // disabled
    FindChild(gameObject=var "Small Title Folder", childName="Title Small Super", storeResult=var "Small Title Sup")  // disabled
    GetPosition(gameObject=var "Small Title Folder", vector=(unset), x=var "Small Title X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Small Title X Pos R", floatValue=13, everyFrame=false)
    FloatMultiply(floatVariable=var "Small Title X Pos R", multiplyBy=-1, everyFrame=false)  // disabled
    FindChild(gameObject=Self, childName="Title Large", storeResult=var "Large Title Folder")  // disabled
    FindChild(gameObject=var "Large Title Folder", childName="Title Large Main", storeResult=var "Large Title Main")  // disabled
    FindChild(gameObject=var "Large Title Folder", childName="Title Large Sub", storeResult=var "Large Title Sub")  // disabled
    FindChild(gameObject=var "Large Title Folder", childName="Title Large Super", storeResult=var "Large Title Super")  // disabled
    SendEventByName(eventTarget=Self, sendEvent=var "Area Event", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=Self, sendEvent="OTHER", delay=0, everyFrame=false)
    on CROSSROADS → Crossroads
    on DIRTMOUTH → Dirtmouth
    on EGG TEMPLE → Egg Temple
    on OTHER → Other
  }

  state Crossroads {
    GetLanguageString(sheetName="Titles", convName="CROSSROADS_MAIN", storeValue=var "Title Main")
    GetLanguageString(sheetName="Titles", convName="CROSSROADS_SUB", storeValue=var "Title Sub")
    GetLanguageString(sheetName="Titles", convName="CROSSROADS_SUPER", storeValue=var "Title Sup")
    on FINISHED → Reset Sub & Sup
  }

  state Visited Check {
    BoolTest(boolVariable=var "NPC Title", isTrue=→"VISITED", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Visited", isTrue=→"VISITED", isFalse=→"UNVISITED", everyFrame=false)
    on UNVISITED → Set Text Large
    on VISITED → Set Text Small
  }

  state Fleurs Up {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BIG TITLE START", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Fleur Top", animLibName="", clipName="Top Appear")
    Tk2dPlayAnimation(gameObject=var "Fleur Bot", animLibName="", clipName="Bottom Appear")
    Wait(time=var "Unvisited Wait", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Titles Down
  }

  state Titles Up {
    SendEventByName(eventTarget=GameObject(var "Large Title Folder"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fleurs Up
    on NPC CONVO START → Fleurs Up
  }

  state Titles Down {
    Tk2dPlayAnimation(gameObject=var "Fleur Top", animLibName="", clipName="Top Disappear")  // disabled
    Tk2dPlayAnimation(gameObject=var "Fleur Bot", animLibName="", clipName="Bottom Disappear")  // disabled
    SendEventByName(eventTarget=GameObject(var "Large Title Folder"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Fleur Bot"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Fleur Top"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Big Title End
  }

  state Done {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="TITLE DISPLAYED", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)
  }

  state Pause {
    SetParent(gameObject=Self, parent=var "HUD Camera", resetLocalPosition=true, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=0, y=6.1, z=38, space=1, everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=var "Fleur Top", animLibName="", clipName="Idle")
    Tk2dPlayAnimation(gameObject=var "Fleur Bot", animLibName="", clipName="Idle")
    on FINISHED → Init
  }

  state Visited Appear {
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Colour", time=1.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)  // disabled
    Tk2dSpriteSetColor(gameObject=var "Title Small", color=var "Colour", everyframe=true)  // disabled
    SendEventByName(eventTarget=GameObject(var "Small Title Folder"), sendEvent="UP", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "NPC Title", isTrue=→"NPC", isFalse=(none), everyFrame=false)
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Visited Disappear
    on NPC CONVO START → Quick Disappear
    on NPC → NPC Wait for Event
  }

  state Visited Disappear {
    Tk2dSpriteSetColor(gameObject=var "Title Small", color=var "Colour", everyframe=true)  // disabled
    SendEventByName(eventTarget=GameObject(var "Small Title Folder"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Done
  }

  state Dirtmouth {
    GetLanguageString(sheetName="Titles", convName="DIRTMOUTH_MAIN", storeValue=var "Title Main")
    GetLanguageString(sheetName="Titles", convName="DIRTMOUTH_SUB", storeValue=var "Title Sub")
    GetLanguageString(sheetName="Titles", convName="DIRTMOUTH_SUPER", storeValue=var "Title Sup")
    on FINISHED → Reset Sub & Sup
  }

  state Big Title End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BIG TITLE END", delay=0, everyFrame=false)
    on FINISHED → Done
  }

  state Set Text Small {
    SetTextMeshProText(gameObject=var "Small Title Main", textString=var "Title Main")
    SetTextMeshProText(gameObject=var "Small Title Sub", textString=var "Title Sub")
    SetTextMeshProText(gameObject=var "Small Title Sup", textString=var "Title Sup")
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=-13.68, y=-12.41, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Pause Frame
  }

  state Super only 2 {
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=(unset), y=-13.15, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Visited Appear
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Margin
  }

  state Set Margin {
    GetMeshRendererBounds(gameObject=var "Small Title Main", width=var "Width Main", height=(unset), widthMax=200, heightMax=(unset))
    FloatMultiply(floatVariable=var "Width Main", multiplyBy=0.5, everyFrame=false)
    GetMeshRendererBounds(gameObject=var "Small Title Sub", width=var "Width Sub", height=(unset), widthMax=200, heightMax=(unset))
    FloatMultiply(floatVariable=var "Width Sub", multiplyBy=0.5, everyFrame=false)
    GetMeshRendererBounds(gameObject=var "Small Title Sup", width=var "Width Super", height=(unset), widthMax=200, heightMax=(unset))
    FloatMultiply(floatVariable=var "Width Super", multiplyBy=0.5, everyFrame=false)
    SetFloatToHighest(floatVariable=var "Longest Width", value1=var "Width Main", value2=var "Width Sub", everyFrame=false)
    SetFloatToHighest(floatVariable=var "Longest Width", value1=var "Longest Width", value2=var "Width Super", everyFrame=false)
    BoolTest(boolVariable=var "Display Right", isTrue=→"RIGHT", isFalse=(none), everyFrame=false)
    FloatAdd(floatVariable=var "Longest Width", add=var "Margin X", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Longest Width", add=var "Small Title X Pos", everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=var "Longest Width", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Sub or Super Active? 2
    on RIGHT → Set Margin R
  }

  state Set Text Large {
    SetTextMeshProText(gameObject=var "Large Title Main", textString=var "Title Main")
    SetTextMeshProText(gameObject=var "Large Title Sub", textString=var "Title Sub")
    SetTextMeshProText(gameObject=var "Large Title Super", textString=var "Title Sup")
    on FINISHED → Pause Frame 2
  }

  state Check Sub {
    StringCompare(stringVariable=var "Title Sub", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Subscript", boolValue=true, everyFrame=false)
    on FINISHED → Check Super
  }

  state Reset Sub & Sup {
    SetBoolValue(boolVariable=var "Title Has Subscript", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Superscript", boolValue=false, everyFrame=false)
    on FINISHED → Check Sub
  }

  state Check Super {
    StringCompare(stringVariable=var "Title Sup", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Superscript", boolValue=true, everyFrame=false)
    on FINISHED → Visited Check
  }

  state Pause Frame 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Sub or Super Active?
  }

  state Sub or Super Active? {
    BoolAllTrue(boolVariables=[var "Title Has Subscript", var "Title Has Superscript"], sendEvent=→"BOTH", storeResult=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Subscript", isTrue=→"SUB", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Superscript", isTrue=→"SUP", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="NEITHER", delay=0, everyFrame=false)
    on BOTH → Titles Up
    on NEITHER → Neither
    on SUB → Sub Only
    on SUP → Sup Only
  }

  state Sub Only {
    SetPosition(gameObject=var "Fleur Bot", vector=(unset), x=(unset), y=-5.24, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Large Title Folder", vector=(unset), x=(unset), y=-2.24, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Titles Up
  }

  state Neither {
    SetPosition(gameObject=var "Fleur Bot", vector=(unset), x=(unset), y=-4.38, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Large Title Folder", vector=(unset), x=(unset), y=-2.24, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Titles Up
  }

  state Sup Only {
    SetPosition(gameObject=var "Fleur Bot", vector=(unset), x=(unset), y=-5.24, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Large Title Folder", vector=(unset), x=(unset), y=-3.03, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Titles Up
  }

  state Egg Temple {
    GetLanguageString(sheetName="Titles", convName="EGGTEMPLE_MAIN", storeValue=var "Title Main")
    GetLanguageString(sheetName="Titles", convName="EGGTEMPLE_SUB", storeValue=var "Title Sub")
    GetLanguageString(sheetName="Titles", convName="EGGTEMPLE_SUPER", storeValue=var "Title Sup")
    on FINISHED → Reset Sub & Sup
  }

  state NPC Wait for Event {
    on NPC TITLE DOWN → Quick Disappear
    on NPC CONVO START → Quick Disappear
  }

  state Other {
    SetStringValue(stringVariable=var "Main String", stringValue=var "Area Event", everyFrame=false)
    SetStringValue(stringVariable=var "Sub String", stringValue=var "Area Event", everyFrame=false)
    SetStringValue(stringVariable=var "Super String", stringValue=var "Area Event", everyFrame=false)
    BuildString(stringParts=[var "Main String", "_MAIN"], separator="", addToEnd=true, storeResult=var "Main String", everyFrame=false)
    BuildString(stringParts=[var "Sub String", "_SUB"], separator="", addToEnd=true, storeResult=var "Sub String", everyFrame=false)
    BuildString(stringParts=[var "Super String", "_SUPER"], separator="", addToEnd=true, storeResult=var "Super String", everyFrame=false)
    GetLanguageString(sheetName="Titles", convName=var "Main String", storeValue=var "Title Main")
    GetLanguageString(sheetName="Titles", convName=var "Sub String", storeValue=var "Title Sub")
    GetLanguageString(sheetName="Titles", convName=var "Super String", storeValue=var "Title Sup")
    on FINISHED → Reset Sub & Sup
  }

  state Sub or Super Active? 2 {
    BoolAllTrue(boolVariables=[var "Title Has Subscript", var "Title Has Superscript"], sendEvent=→"BOTH", storeResult=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Subscript", isTrue=→"SUB", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Superscript", isTrue=→"SUP", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="NEITHER", delay=0, everyFrame=false)
    on BOTH → Visited Appear
    on NEITHER → Super only 2
    on SUB → Visited Appear
    on SUP → Super only 2
  }

  state Quick Disappear {
    SetFsmFloat(gameObject=var "Small Title Main", fsmName="color_fader", variableName="Down Time", setValue=0.5, everyFrame=false)
    SetFsmFloat(gameObject=var "Small Title Sub", fsmName="color_fader", variableName="Down Time", setValue=0.5, everyFrame=false)
    SetFsmFloat(gameObject=var "Small Title Sup", fsmName="color_fader", variableName="Down Time", setValue=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Small Title Folder"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Done
  }

  state Set Margin R old {
    CallMethodProper(gameObject=var "Small Title Main", behaviour="GetTMProLeftVertex", methodName="GetLeftmostVertex", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Small Title Sub", behaviour="GetTMProLeftVertex", methodName="GetLeftmostVertex", parameters=[], storeResult=(unset var))
    CallMethodProper(gameObject=var "Small Title Sup", behaviour="GetTMProLeftVertex", methodName="GetLeftmostVertex", parameters=[], storeResult=(unset var))
    SetFloatToHighest(floatVariable=var "Longest Width", value1=var "Width Main", value2=var "Width Sub", everyFrame=false)
    SetFloatToHighest(floatVariable=var "Longest Width", value1=var "Longest Width", value2=var "Width Super", everyFrame=false)
    FloatAdd(floatVariable=var "Longest Width", add=var "Margin X", everyFrame=false, perSecond=false)  // disabled
    FloatAdd(floatVariable=var "Longest Width", add=var "Small Title X Pos", everyFrame=false, perSecond=false)  // disabled
    SetFloatValue(floatVariable=var "Right X", floatValue=14.5, everyFrame=false)
    FloatSubtract(floatVariable=var "Right X", subtract=var "Longest Width", everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=var "Right X", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Sub or Super Active? 2
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Set Margin R {
    FloatSubtract(floatVariable=var "Longest Width", subtract=var "Margin X", everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Small Title X Pos R", subtract=var "Longest Width", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Longest Width", add=var "Small Title X Pos R", everyFrame=false, perSecond=false)  // disabled
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=var "Small Title X Pos R", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Sub or Super Active? 2
  }

  var Fleur Bot: gameObject = <null>
  var Fleur Top: gameObject = <null>
  var Large Title Folder: gameObject = Area Title Extra/Title Large
  var Large Title Main: gameObject = Area Title Extra/Title Large/Title Large Main
  var Large Title Sub: gameObject = Area Title Extra/Title Large/Title Large Sub
  var Large Title Super: gameObject = Area Title Extra/Title Large/Title Large Super
  var Small Title Folder: gameObject = Area Title Extra/Title Small
  var Small Title Main: gameObject = Area Title Extra/Title Small/Title Small Main
  var Small Title Sub: gameObject = Area Title Extra/Title Small/Title Small Sub
  var Small Title Sup: gameObject = Area Title Extra/Title Small/Title Small Super
  var Title Small: gameObject = Area Title Extra/Title Small

  var Longest Width: float = 0
  var Margin X: float = 0
  var Right X: float = 0
  var Scaler: float = 0
  var Small Title X Pos: float = 0
  var Small Title X Pos R: float = 0
  var Unvisited Wait: float = 4.75
  var Width Main: float = 0
  var Width Sub: float = 0
  var Width Super: float = 0
  var Display Right: bool = false
  var NPC Title: bool = false
  var Title Has Subscript: bool = false
  var Title Has Superscript: bool = false
  var Visited: bool = false
  var Area Event: string = ""
  var Main String: string = ""
  var Sub String: string = ""
  var Super String: string = ""
  var Title Main: string = ""
  var Title Sub: string = ""
  var Title Sup: string = ""
  var Colour: color = (1, 1, 1, 0)
  var Title: gameObject = Area Title Extra/Title Large
}
