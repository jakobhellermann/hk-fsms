# Shade Control
# 1 location(s):
#   resources.assets / Hollow Shade Death

fsm Shade Control {
  start Death Start

  state Death Start {
    GetOwner(storeGameObject=var "Self")
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=EndSoulLimiter())
    GetPlayerDataInt(gameObject=var "GameManager", intName="geoPool", storeValue=var "Geo")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddGeoQuietly", parameters=[var "Geo"], storeResult=(unset var))
    SetBoolValue(boolVariable=var "Killed", boolValue=true, everyFrame=false)
    AudioStop(gameObject=Self)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClip=boss_explode_clean, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor, spawnPoint=var "Self", audioClip=hero_void_scream_spell, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPlayerDataInt(intName="geoPool", value=0)
    SetPlayerDataString(gameObject=var "GameManager", stringName="shadeScene", value="None")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetParticleEmissionRate(gameObject=var "Shade Particles", emissionRate=20, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Death Wave", emit=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Start")
    SetCollider(gameObject=Self, active=false)
    SpawnRandomObjectsV2(gameObject=Slash Effect GhostDark1, spawnPoint=var "Self", position=(unset), spawnMin=4, spawnMax=5, speedMin=20, speedMax=35, angleMin=0, angleMax=360, originVariationX=0, originVariationY=0)
    SpawnRandomObjectsV2(gameObject=Slash Effect GhostDark2, spawnPoint=var "Self", position=(unset), spawnMin=4, spawnMax=5, speedMin=20, speedMax=35, angleMin=0, angleMax=360, originVariationX=0, originVariationY=0)
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=0, allowMovement=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=StoryRecord_defeatedShade())
    on FINISHED → Blow
  }

  state Blow {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death")
    ActivateGameObject(gameObject=var "Death Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Death Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Camera", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state End {
    StopParticleEmitter(gameObject=var "Shade Particles")
    DestroyObject(gameObject=var "Light", delay=0, detachChildren=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dissipate
  }

  state Give Geo {
    FindChild(gameObject=var "HUD Canvas", childName="Soul Orb", storeResult=var "Soul Orb")
    SendEventByName(eventTarget=GameObject(var "Soul Orb"), sendEvent="SOUL LIMITER DOWN", delay=0, everyFrame=false)
    IntCompare(integer1=var "Geo", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddGeoToCounter", parameters=[var "Geo"], storeResult=(unset var))
    on FINISHED → Destroy
  }

  state Destroy {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="HOLLOW SHADE KILLED", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)
  }

  state Dissipate {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Death Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Give Geo
  }

  var Geo: int = 250
  var SP: int = 1

  var Distance: float = 0
  var Hero Y: float = 0
  var Scale: float = 0
  var Shade Y: float = 0
  var X Speed: float = 0
  var Y Speed: float = 0
  var Amount: int = 0
  var Fireball Level: int = 120
  var Can See Hero: bool = false
  var In Alert Range: bool = false
  var In Slash Range: bool = false
  var Killed: bool = false
  var Same Y: bool = false
  var Alerter: gameObject = <null>
  var Camera: gameObject = <null>
  var Death Glow: gameObject = Hollow Shade Death/Death Glow
  var Death Sprite: gameObject = Hollow Shade Death/Death Sprite
  var Death Wave: gameObject = Hollow Shade Death/Particle Wave
  var Depart Particles: gameObject = Hollow Shade Death/Depart Particles
  var Effect: gameObject = <null>
  var Light: gameObject = Hollow Shade Death/light_effect_v02
  var Self: gameObject = <null>
  var Shade Particles: gameObject = Hollow Shade Death/Shade Particles
  var Slash: gameObject = <null>
  var Soul Orb: gameObject = <null>
}
