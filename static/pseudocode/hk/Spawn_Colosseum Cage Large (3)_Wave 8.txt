# Spawn
# 1 location(s):
#   Room_Colosseum_Gold / Colosseum Manager/Waves/Wave 8/Colosseum Cage Large (3)

fsm Spawn {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="Colosseum Manager", store=var "Colosseum Manager")
    FindChild(gameObject=Self, childName="Anim", storeResult=var "Anim")
    SetSpriteRenderer(gameObject=Self, active=false)
    BoolTest(boolVariable=var "Summoner", isTrue=→"SUMMON", isFalse=(none), everyFrame=false)
    CreateObject(gameObject=var "Corpse to Instantiate", spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Enemy", networkInstantiate=false, networkGroup=0)
    ActivateGameObject(gameObject=var "Enemy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PreSpawnCorpse(target=var "Enemy")
    PreBuildTK2DSprites(gameObject=var "Enemy", useChildren=true)
    SetBattleScene(target=var "Enemy", battleScene=var "Colosseum Manager")
    SetGeoDrop(target=var "Enemy", smallGeo=0, mediumGeo=0, largeGeo=0)
    on SUMMON → Summon Init
    on FINISHED → Summon Idle
  }

  state Anim {
    ActivateGameObject(gameObject=var "Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="AUDIO CAGE UP", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="AUDIO CAGE UP")
    on FINISHED → Spawn
  }

  state Spawn {
    BoolTest(boolVariable=var "Summoner", isTrue=→"SUMMON", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Strike", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="AUDIO CAGE OPEN")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="AUDIO CAGE OPEN", delay=0, everyFrame=false)  // disabled
    BoolTest(boolVariable=var "Stunner", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetSendKilledToObject(target=var "Enemy", killedObject=var "Self")
    SetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="KILLED GameObject", setValue=var "Self", everyFrame=false)  // disabled
    on FINISHED → Pause
    on SUMMON → Summon
  }

  state End {
    ActivateGameObject(gameObject=var "Anim", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on KILLED → Stun
  }

  state Spawn Pause {
    Wait(time=var "Spawn Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Anim
  }

  state Pause {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
    on KILLED → Stun
  }

  state Stun {
    CreateObject(gameObject=Stun Effect (resources.assets), spawnPoint=var "Enemy", position=(unset), rotation=(unset), storeObject=var "", networkInstantiate=false, networkGroup=0)
  }

  state Summon {
    SendEventToRegister(eventName="AUDIO CAGE OPEN")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="AUDIO CAGE OPEN", delay=0, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Strike", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Corpse to Instantiate", vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state Summon Init {
    SetFsmGameObject(gameObject=var "Corpse to Instantiate", fsmName="health_manager_enemy", variableName="Battle Scene", setValue=var "Colosseum Manager", everyFrame=false)  // disabled
    SetFsmInt(gameObject=var "Corpse to Instantiate", fsmName="health_manager_enemy", variableName="Geo Small", setValue=0, everyFrame=false)  // disabled
    SetFsmInt(gameObject=var "Corpse to Instantiate", fsmName="health_manager_enemy", variableName="Geo Medium", setValue=0, everyFrame=false)  // disabled
    SetFsmInt(gameObject=var "Corpse to Instantiate", fsmName="health_manager_enemy", variableName="Geo Large", setValue=0, everyFrame=false)  // disabled
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", z=(unset), space=0, everyFrame=false)
    on SPAWN → Spawn Pause
  }

  state Create Corpse {
    GetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="Corpse", storeValue=var "Enemy Type", everyFrame=false)  // disabled
    SetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="Corpse", setValue=<null>, everyFrame=false)  // disabled
    CreateObject(gameObject=var "Enemy Type", spawnPoint=var "Enemy", position=(unset), rotation=(unset), storeObject=var "Corpse Instance", networkInstantiate=false, networkGroup=0)  // disabled
    ActivateGameObject(gameObject=var "Corpse Instance", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetParent(gameObject=var "Corpse Instance", parent=var "Enemy", resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetFsmGameObject(gameObject=var "Enemy", fsmName="health_manager_enemy", variableName="Corpse Instance", setValue=var "Corpse Instance", everyFrame=false)  // disabled
    on FINISHED → Summon Idle
  }

  state Summon Idle {
    on SPAWN → Spawn Pause
  }

  var Spawn Pause: float = 0
  var Stunner: bool = false
  var Summoner: bool = false
  var Corpse to Instantiate: gameObject = Colosseum_Worm (sharedassets33.assets)
  var Enemy Type: gameObject = <null>

  var Self X: float = 0
  var Self Y: float = 0
  var Anim: gameObject = Colosseum Manager/Waves/Wave 8/Colosseum Cage Large (3)/Anim
  var Colosseum Manager: gameObject = <null>
  var Corpse Instance: gameObject = <null>
  var Enemy: gameObject = <null>
  var Self: gameObject = <null>
  var Strike: gameObject = Colosseum Manager/Waves/Wave 8/Colosseum Cage Large (3)/Strike Nail
}
