# Conversation Control
# 1 location(s):
#   Town / _NPCs/Zote Town

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Box Down  // from any state

  state Idle {
    on CONVO START → Check Direction
  }

  state Convo Choice {
    SetAudioVolume(gameObject=var "Voice Player", volume=0, everyFrame=false)
    GetPlayerDataInt(gameObject=var "GameManager", intName="zoteTownConvo", storeValue=var "Convo")
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="ZOTE", everyFrame=false)
    IntSwitch(intVariable=var "Convo", compareTo=[0, 1, 2], sendEvent=[(none), (none), →"REPEAT"], everyFrame=false)
    on 1 → 1
    on 2 → 2
    on REPEAT → Repeat
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Check Active
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    DestroyObject(gameObject=var "Voice Player", delay=0, detachChildren=false)
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    on FINISHED → Check Direction 2
  }

  state Check Direction {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Talk R
    on LEFT → Talk Back
  }

  state Talk R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk L")
    on FINISHED → Box Up
  }

  state Talk Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk R")
    on FINISHED → Box Up
  }

  state Box Up {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Check Direction 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"RIGHT", leftEvent=→"LEFT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), everyFrame=false)
    on RIGHT → Face Right
    on LEFT → Face Left
  }

  state Face Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Right")
    on FINISHED → End
  }

  state Face Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Left")
    on FINISHED → End
  }

  state 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_05 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_TOWN_1", "Zote"], storeResult=(unset var))
    SetPlayerDataInt(intName="zoteTownConvo", value=1)
    on CONVO_FINISH → Talk Finish
  }

  state Check Active {
    GetPlayerDataInt(gameObject=var "GameManager", intName="zote", storeValue=var "NPC State")  // disabled
    IntSwitch(intVariable=var "NPC State", compareTo=[0], sendEvent=[→"DESTROY"], everyFrame=false)  // disabled
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="zoteRescuedBuzzer", isTrue=(none), isFalse=→"DESTROY")
    on FINISHED → Idle
    on DESTROY → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_03 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_TOWN_2", "Zote"], storeResult=(unset var))
    SetPlayerDataInt(intName="zoteTownConvo", value=2)
    on CONVO_FINISH → Talk Finish
  }

  state Repeat {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Zote_05 (sharedassets7.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ZOTE_TOWN_REPEAT", "Zote"], storeResult=(unset var))
    on CONVO_FINISH → Talk Finish
  }

  var Convo: int = 0
  var NPC State: int = 0
  var Area Title: gameObject = <null>
  var Self: gameObject = <null>
  var Voice Player: gameObject = <null>
}
