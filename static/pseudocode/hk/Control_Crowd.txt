# Control
# 1 location(s):
#   Grimm_Nightmare / Grimm Control/Crowd

fsm Control {
  start Pause

  state Pause {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  state Idle {
    FadeGroupUp(target=Self)
    on CROWD GASP → Gasp
  }

  state Gasp {
    FadeGroupDown(target=Self, fast=false)
    on CROWD IDLE → Idle
  }

  state Init {
    FadeGroupUp(target=Self)
    on CROWD GASP → Gasp
  }

  var Start Frame: int = 0
}
