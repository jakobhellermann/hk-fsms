# Control
# 2 location(s):
#   sharedassets392.assets / Firebat Pillar
#   sharedassets393.assets / Firebat Pillar NM

fsm Control {
  start Damage

  state Damage {
    SetCollider(gameObject=Self, active=true)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Collider Off
  }

  state Collider Off {
    SetCollider(gameObject=Self, active=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }
}
