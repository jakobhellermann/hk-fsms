# Control
# 1 location(s):
#   sharedassets409.assets / HK Plume

// uses template: hk_plume
fsm Control {
  start Init

  state Init {
    SetCollider(gameObject=Self, active=false)
    BoolTest(boolVariable=var "Auto", isTrue=→"AUTO", isFalse=(none), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on AUTO → Auto
    on FINISHED → Antic
  }

  state Auto {
    SetBoolValue(boolVariable=var "Auto", boolValue=false, everyFrame=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Plume Antic")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Plume1
  }

  state Plume1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Plume Burst", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Plume 2
  }

  state Antic {
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=1, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Plume Antic")
    on PLUMES UP → Scale Up
    on PLUME CANCEL → Auto?
    on HK STUN → Auto?
  }

  state Plume 2 {
    SetBoxCollider2DSize(gameObject1=Self, width=0.42, height=1.35, offsetX=0.11, offsetY=0.7)
    SetCollider(gameObject=Self, active=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Plume 3
  }

  state Plume 3 {
    SetBoxCollider2DSize(gameObject1=Self, width=0.65, height=3.395807, offsetX=0, offsetY=1.702904)
    SetCollider(gameObject=Self, active=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    PlayParticleEmitter(gameObject=var "Embers", emit=0)
    PlayParticleEmitter(gameObject=var "Embers B", emit=0)
    on FINISHED → Plume 4
  }

  state Plume 4 {
    SetBoxCollider2DSize(gameObject1=Self, width=0.65, height=5.27, offsetX=0, offsetY=2.64)
    SetCollider(gameObject=Self, active=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Collider Off
  }

  state Collider Off {
    SetCollider(gameObject=Self, active=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End
  }

  state Auto? {
    BoolTest(boolVariable=var "Auto", isTrue=→"AUTO", isFalse=→"FINISHED", everyFrame=false)
    on AUTO → Deactivate
    on FINISHED → Recycle
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Recycle {
    RecycleSelf()
  }

  state Scale Up {
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=2.4, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Plume1
  }

  state End {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Auto?
  }

  var Auto: bool = false
  var Embers: gameObject = HK Plume/Embers
  var Embers B: gameObject = HK Plume/Embers B
}
