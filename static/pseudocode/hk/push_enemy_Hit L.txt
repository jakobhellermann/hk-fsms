// uses template: push_enemy
fsm push_enemy {
  start Detect

  state Detect {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"HIT", storeCollider=var "Collider")
    on HIT → Send Event
  }

  state Send Event {
    BeginRecoil(target=var "Collider", attackDirection=var "Attack Direction", attackType=var "Attack Type", attackMagnitude=var "Magnitude")
    on FINISHED → Detect
  }

  var Attack Direction: float = 180

  var Magnitude: float = 2
  var Attack Type: int = 1
  var Collider: gameObject = <null>
}
