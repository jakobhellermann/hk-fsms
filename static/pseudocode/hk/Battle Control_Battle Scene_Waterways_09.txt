# Battle Control
# 1 location(s):
#   Waterways_09 / Battle Scene

fsm Battle Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Inactive
  }

  state Inactive {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activate
    on ENTER → Battle Start
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
  }

  state Battle Start {
    TransitionToAudioSnapshot(snapshot=Action (resources.assets), transitionTime=1)
    on FINISHED → Idle
  }

  state Check {
    GetChildCount(gameObject=Self, storeResult=var "Children")
    IntCompare(integer1=var "Children", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on END → End Pause
  }

  state Idle {
    Wait(time=1.5, finishEvent=→"CHECK", realTime=false)
    on CHECK → Check
  }

  state End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → End
  }

  state End {
    TransitionToAudioSnapshot(snapshot=Silent (resources.assets), transitionTime=3)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  var Children: int = 0
  var Activated: bool = false
  var Self: gameObject = <null>
}
