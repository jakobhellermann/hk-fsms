# Conversation Control
# 1 location(s):
#   Grimm_Divine / Divine NPC

fsm Conversation Control {
  start Init
  on RESET CONVO → Idle  // from any state
  on TALK FINISH → Anim End  // from any state
  on NO ANIM FINISH → Box Down  // from any state

  state Idle {
    AudioPlayInState(gameObject=Self, volume=1)
    on CONVO START → Hero Anim
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="Charm Holder", withTag="Extra Tag", store=var "Charm Holder")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pooed Heart?
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CONVO END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="RESET CONVO", delay=0, everyFrame=false)
  }

  state Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="NPC TITLE DOWN", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUpToIdle")
    on FINISHED → End
  }

  state Box Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk")
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SetGameObject(variable=var "Area Title", gameObject=var "AreaTitle", everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="NPC Title", setValue=true, everyFrame=false)
    SetFsmString(gameObject=var "Area Title", fsmName="", variableName="Area Event", setValue="DIVINE", everyFrame=false)
    SetFsmBool(gameObject=var "Area Title", fsmName="", variableName="Display Right", setValue=true, everyFrame=false)
    on FINISHED → Choice
  }

  state Anim End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Talk End")
    on FINISHED → Box Down
  }

  state Talk Finish {
    SendEventByName(eventTarget=Self, sendEvent="TALK FINISH", delay=0, everyFrame=false)
  }

  state Hero Anim {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="LookUp")
    on FINISHED → Box Up
  }

  state Greet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DIVINE_CONVO_1", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Divine_talk_05 (sharedassets390.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Choice {
    AudioStop(gameObject=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="divineFinalConvo", isTrue=→"FINAL REPEAT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="legEaterLeft", isTrue=→"FINAL", isFalse=(none))
    PlayerDataBoolAllTrue(gameObject=var "GameManager", stringVariables=["fragileHealth_unbreakable", "fragileGreed_unbreakable", "fragileStrength_unbreakable"], trueEvent=→"ALL EATEN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="metDivine", isTrue=(none), isFalse=→"MEET")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gaveFragileHeart", isTrue=→"SWALLOWED HEART", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gaveFragileGreed", isTrue=→"SWALLOWED GREED", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="gaveFragileStrength", isTrue=→"SWALLOWED STRENGTH", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_23", falseBool="fragileHealth_unbreakable", isTrue=→"EQUIPPED HEART", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_24", falseBool="fragileGreed_unbreakable", isTrue=→"EQUIPPED GEO", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="equippedCharm_25", falseBool="fragileStrength_unbreakable", isTrue=→"EQUIPPED STRENGTH", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_10", isTrue=→"DUNG", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gotCharm_23", falseBool="fragileHealth_unbreakable", isTrue=→"HAS CHARM", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gotCharm_24", falseBool="fragileGreed_unbreakable", isTrue=→"HAS CHARM", isFalse=(none))
    PlayerDataBoolTrueAndFalse(gameObject=var "GameManager", trueBool="gotCharm_25", falseBool="fragileStrength_unbreakable", isTrue=→"HAS CHARM", isFalse=(none))
    on MEET → Meet
    on FINISHED → Greet
    on HAS CHARM → Has Charm
    on EQUIPPED HEART → Set Heart
    on EQUIPPED GEO → Set Greed
    on EQUIPPED STRENGTH → Set Strength
    on SWALLOWED HEART → Swallowed Heart
    on SWALLOWED GREED → Swallowed Greed
    on SWALLOWED STRENGTH → Swallowed Strength
    on ALL EATEN → All Eaten
    on FINAL → Final
    on FINAL REPEAT → Final Repeat
    on DUNG → Dung
  }

  state Meet {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DIVINE_MEET", "CP2"], storeResult=(unset var))
    SetPlayerDataBool(boolName="metDivine", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Divine_talk_01 (sharedassets390.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Has Charm {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DIVINE_HAS_CHARM", "CP2"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Divine_talk_02 (sharedassets390.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on CONVO_FINISH → Talk Finish
  }

  state Set Heart {
    SetIntValue(intVariable=var "Current Charm", intValue=1, everyFrame=false)
    SetStringValue(stringVariable=var "Prompt Convo", stringValue="PROMPT_FRAGILE_HEART", everyFrame=false)
    on FINISHED → Request Charm
  }

  state Request Charm {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Divine_talk_02 (sharedassets390.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="fragileHealth_unbreakable", isTrue=→"REPEAT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="fragileGreed_unbreakable", isTrue=→"REPEAT", isFalse=(none))
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="fragileStrength_unbreakable", isTrue=→"REPEAT", isFalse=(none))
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["DIVINE_CHARM_EQUIPPED", "CP2"], storeResult=(unset var))
    on CONVO_FINISH → Dial Box Down
    on REPEAT → Request Charm 2
  }

  state Set Greed {
    SetIntValue(intVariable=var "Current Charm", intValue=2, everyFrame=false)
    SetStringValue(stringVariable=var "Prompt Convo", stringValue="PROMPT_FRAGILE_GREED", everyFrame=false)
    on FINISHED → Request Charm
  }

  state Set Strength {
    SetIntValue(intVariable=var "Current Charm", intValue=3, everyFrame=false)
    SetStringValue(stringVariable=var "Prompt Convo", stringValue="PROMPT_FRAGILE_STRENGTH", everyFrame=false)
    on FINISHED → Request Charm
  }

  state Box Up YN {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP YN", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Text
  }

  state Send Text {
    SetFsmInt(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Toll Cost", setValue=var "Something Cost", everyFrame=false)
    SetFsmGameObject(gameObject=var "DialogueTextYN", fsmName="Dialogue Page Control", variableName="Requester", setValue=var "Self", everyFrame=false)
    CallMethodProper(gameObject=var "DialogueTextYN", behaviour="DialogueBox", methodName="StartConversation", parameters=[var "Prompt Convo", "CP2"], storeResult=(unset var))
    on NO → Decline Pause
    on YES → Yes
  }

  state Yes {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN YN", delay=0, everyFrame=false)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Take Choice
  }

  state Dial Box Down {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN NO HUD", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Box Up YN
  }

  state Box Up 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)

  }

  state Decline Pause {

  }

  }

  state Dial Box Down 2 {
    SendEventByName(eventTarget=GameObject(var "DialogueManager"), sendEvent="BOX DOWN NO HUD", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Eat
  }

  state Eat {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Eat")
    Wait(time=1.5, finishEvent=→"WAIT", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Divine_eat_01 (sharedassets390.assets), pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=var "Voice Player")

  }

  state Eat Return {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Eat Return")
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on FINISHED → Swallow
    on WAIT → 
  }

  state Swallow {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Poo")  // disabled
    Wait(time=1.65, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Poo", animationTriggerEvent=(none), animationCompleteEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=Divine_post_eat_sigh (sharedassets390.assets), pitchMin=1, pitchMax=1, volume=1, delay=0.4, storePlayer=var "Voice Player")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=hatcher_give_birth (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=var "Voice Player")
    on FINISHED → Box Up 4
  }

  state Take Choice {
    SpawnObjectFromGlobalPool(gameObject=Relic Get Msg (resources.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Msg")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=ruin_sentry_sword_1 (sharedassets33.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=shiny_item_pickup (resources.assets), pitchMin=0.8, pitchMax=0.8, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Grab")
    FindChild(gameObject=var "Msg", childName="Text", storeResult=var "Msg Text")
    FindChild(gameObject=var "Msg", childName="Icon", storeResult=var "Msg Icon")
    GetLanguageString(sheetName="CP2", convName="UI_CHARM_LOST", storeValue=var "Item Name")
    SetTextMeshProText(gameObject=var "Msg Text", textString=var "Item Name")
    IntSwitch(intVariable=var "Current Charm", compareTo=[1, 2, 3], sendEvent=[→"HEART", →"GREED", →"STRENGTH"], everyFrame=false)
    on HEART → Take Heart
    on GREED → Take Greed
    on STRENGTH → Take Strength
    on FINISHED → Take Heart
  }

  state Take Heart {
    SetSpriteRendererSprite(gameObject=var "Msg Icon", sprite=_0002_charm_glass_heal (resources.assets))
    SetPlayerDataBool(boolName="gaveFragileHeart", value=true)
    SetPlayerDataBool(boolName="gotCharm_23", value=false)
    SetPlayerDataBool(boolName="equippedCharm_23", value=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=UnequipCharm(23))
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmSlotsFilled", storeValue=var "Charm Slots Filled")
    GetPlayerDataInt(gameObject=var "GameManager", intName="charmCost_23", storeValue=var "Charm Cost")
    IntOperator(integer1=var "Charm Slots Filled", integer2=var "Charm Cost", operation=Subtract, storeResult=var "Charm Slots Filled", everyFrame=false)
    SetPlayerDataInt(intName="charmSlotsFilled", value=var "Charm Slots Filled")
    on FINISHED → Grab Pause
  }

  state Grab Pause {
