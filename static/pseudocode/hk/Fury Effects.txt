fsm Fury Effects {
  start Init

  state Init {
    GetChild(gameObject=Self, childName="Fury Particle", withTag="Untagged", storeResult=var "Fury Particle")
    GetChild(gameObject=Self, childName="Idle", withTag="Untagged", storeResult=var "Idle")
    on FINISHED → Idle
  }

  state Idle {
    StopParticleEmitter(gameObject=var "Fury Particle")
    SendMessage(gameObject=var "Idle", delivery=SendMessage, options=1, functionCall=CancelFlash())
    on FURY START → Fury
  }

  state Fury {
    PlayParticleEmitter(gameObject=var "Fury Particle", emit=0)
    SendMessage(gameObject=var "Idle", delivery=SendMessage, options=1, functionCall=FlashingFury())
    on FURY END → Idle
  }

  var Fury Particle: gameObject = <null>
  var Idle: gameObject = <null>
}
