# Battle Start
# 1 location(s):
#   Room_Final_Boss_Core / Boss Control

fsm Battle Start {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Imprisoned Knight", storeResult=var "Imprisoned Knight")
    FindChild(gameObject=Self, childName="Chain Burst", storeResult=var "Chain Burst")
    FindChild(gameObject=Self, childName="Dust Hit Plume", storeResult=var "Dust Hit Plume")
    FindChild(gameObject=Self, childName="Nail", storeResult=var "Nail")
    FindChild(gameObject=Self, childName="Final Chains", storeResult=var "Final Chains")
    FindChild(gameObject=Self, childName="CamLock", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Init Roar Point", storeResult=var "Init Roar Point")
    FindChild(gameObject=Self, childName="Mist Burst", storeResult=var "Mist Burst")
    FindChild(gameObject=Self, childName="Mist Leak", storeResult=var "Mist Leak")
    FindChild(gameObject=Self, childName="Title", storeResult=var "Title")
    FindChild(gameObject=Self, childName="Hollow Knight Boss", storeResult=var "Hollow Knight Boss")
    FindChild(gameObject=Self, childName="Shield Hit", storeResult=var "Shield Hit")
    FindChild(gameObject=Self, childName="Shield Shing", storeResult=var "Shield Shing")
    FindChild(gameObject=Self, childName="Shield Shatter", storeResult=var "Shield Shatter")
    PreSpawnCorpse(target=var "Hollow Knight Boss")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="unchainedHollowKnight", isTrue=→"REVISIT", isFalse=(none))
    on FINISHED → Idle
    on REVISIT → Revisit
  }

  state Idle {
    on CHAIN BREAK → Check
  }

  state Check {
    IntOperator(integer1=var "Chains", integer2=1, operation=Subtract, storeResult=var "Chains", everyFrame=false)
    IntCompare(integer1=var "Chains", integer2=0, equal=→"ALL BROKEN", lessThan=→"ALL BROKEN", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on ALL BROKEN → Free Pause
  }

  state Free Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Struggle
  }

  state Struggle {
    SendEventByName(eventTarget=GameObject(var "Imprisoned Knight"), sendEvent="END", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hollow_knight_rumble_pre_chain_break (sharedassets225.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="SUSPENSE FINAL", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Imprisoned Knight", animLibName="", clipName="Rest Shake")
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Imprisoned Knight", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shield Shing", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shield Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Break Antic
  }

  state Break {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=boss_explode_clean (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=chain_cut (sharedassets30.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Chain Burst", emit=0)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Final Chains", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shield Shatter", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fall
  }

  state Break Antic {
    Tk2dPlayAnimationWithEvents(gameObject=var "Imprisoned Knight", clipName="Rest Break", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingBig", setValue=false, everyFrame=false)
    on FINISHED → Break
  }

  state Fall {
    Tk2dPlayAnimation(gameObject=var "Imprisoned Knight", animLibName="", clipName="Rest Fall")
    iTweenMoveBy(gameObject=var "Imprisoned Knight", id="", vector=(0, -5.155516, 0), time=1, delay=0, speed=30, easeType=linear, loopType=none, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"LAND", realTime=false, stopOnExit=true, loopDontFinish=true)
    on LAND → Land
  }

  state Land {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Imprisoned Knight", animLibName="", clipName="Rest Land")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=false_knight_land_1st_time (sharedassets6.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Dust Hit Plume", emit=0)
    ActivateGameObject(gameObject=var "Nail", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar Antic
  }

  state Roar Antic {
    Tk2dPlayAnimationWithEvents(gameObject=var "Imprisoned Knight", clipName="Roar Init", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Roar
  }

  state Roar {
    CreateObject(gameObject=Roar Wave Emitter (sharedassets13.assets), spawnPoint=var "Init Roar Point", position=(unset), rotation=(unset), storeObject=var "Roar Emitter", networkInstantiate=false, networkGroup=0)
    TransitionToAudioSnapshot(snapshot=Normal (resources.assets), transitionTime=0)
    ApplyMusicCue(musicCue=HKBattle (sharedassets409.assets), delayTime=0, transitionTime=0)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Imprisoned Knight", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=hollow_knight_scream_v1 (sharedassets19.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Particle Burst
  }

  state Particle Burst {
    PlayParticleEmitter(gameObject=var "Mist Burst", emit=0)
    PlayParticleEmitter(gameObject=var "Mist Leak", emit=0)
    SendEventByName(eventTarget=GameObject(var "Title"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=var "Title Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Particle End
  }

  state Particle End {
    SendEventByName(eventTarget=GameObject(var "Title"), sendEvent="DOWN", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Mist Leak")
    SendEventByName(eventTarget=GameObject(var "Roar Emitter"), sendEvent="END", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Imprisoned Knight", clipName="RoarToIdle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fight Start
  }

  state Fight Start {
    SetPlayerDataBool(boolName="unchainedHollowKnight", value=true)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Imprisoned Knight", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hollow Knight Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Revisit {
    ActivateGameObject(gameObject=var "Final Chains", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Imprisoned Knight"), sendEvent="END", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Nail", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Imprisoned Knight", vector=(unset), x=(unset), y=9.3, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Imprisoned Knight", animLibName="", clipName="Revisit Ready")
    SetFloatValue(floatVariable=var "Title Time", floatValue=3, everyFrame=false)
    ActivateGameObject(gameObject=var "Shield Shing", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shield Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Roar Antic
  }

  var Title Time: float = 4
  var Chains: int = 4
  var CamLock: gameObject = <null>
  var Chain Burst: gameObject = <null>
  var Dust Hit Plume: gameObject = <null>
  var Final Chains: gameObject = <null>
  var Hollow Knight Boss: gameObject = <null>
  var Imprisoned Knight: gameObject = <null>
  var Init Roar Point: gameObject = <null>
  var Mist Burst: gameObject = <null>
  var Mist Leak: gameObject = <null>
  var Nail: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Shield Hit: gameObject = <null>
  var Shield Shatter: gameObject = <null>
  var Shield Shing: gameObject = <null>
  var Title: gameObject = <null>
}
