# Change if Weaversong equipped
# 1 location(s):
#   Deepnest_41 / Happy Spider NPC/Dream Dialogue

fsm Change if Weaversong equipped {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_39", isTrue=→"CHANGE", isFalse=(none))
    on CHANGE → Set
  }

  state Set {
    SetFsmString(gameObject=Self, fsmName="npc_dream_dialogue", variableName="Convo Name", setValue="MIDWIFE_WEAVERSONG_DREAM", everyFrame=false)
    SetFsmString(gameObject=Self, fsmName="npc_dream_dialogue", variableName="Sheet Name", setValue="CP2", everyFrame=false)
  }
}
