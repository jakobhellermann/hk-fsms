fsm Rotater {
  start Init

  state Init {
    FindGameObject(objectName="Ruins_gate_main", withTag="Untagged", store=var "Gate")
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=var "Init Angle", zAngle=(unset), space=0, everyFrame=false)
    on FINISHED → Rotate
  }

  state Rotate {
    GetPosition(gameObject=var "Gate", vector=(unset), x=(unset), y=var "Gate Y", z=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Gate Y", float2=var "Init Angle", operation=Add, storeResult=var "Rotate Angle", everyFrame=true)
    FloatMultiply(floatVariable=var "Rotate Angle", multiplyBy=var "Multiplier", everyFrame=true)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Angle", space=0, everyFrame=true, lateUpdate=false)
  }

  var Multiplier: float = -50

  var Gate Y: float = 0
  var Init Angle: float = 0
  var Rotate Angle: float = 0
  var Gate: gameObject = <null>
}
