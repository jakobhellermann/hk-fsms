# Detect Quake
# 2 location(s):
#   Mines_01 / _Scenery/mine_1_quake_floor
#   Mines_35 / _Scenery/mine_1_quake_floor

fsm Detect Quake {
  start Pause

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Detect
    on ACTIVATE → Activate !!!
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"CONTACT", storeCollider=var "Hero Obj")
    on CONTACT → Check Quake
  }

  state Check Quake {
    CallMethodProper(gameObject=var "Hero Obj", behaviour="HeroController", methodName="GetState", parameters=["spellQuake"], storeResult=var "Quaking")
    BoolTest(boolVariable=var "Quaking", isTrue=→"QUAKE", isFalse=→"CANCEL", everyFrame=false)
    on QUAKE → Hit Pause
    on CANCEL → Detect
  }

  state Quake Hit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=breakable_wall_hit_2 (sharedassets6.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BREAK 1", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Quaked Floor", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust Break 1", emit=0)
    ActivateGameObject(gameObject=var "Cracks 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.85, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break 2
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activate !!! {
    DestroySelf(detachChildren=false)
  }

  state Hit Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Quake Hit
  }

  state Break 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BREAK 2", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=breakable_wall_hit_1 (sharedassets6.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Hero", audioClip=break_wall_after_tutorial_area (sharedassets6.assets), pitchMin=0.85, pitchMax=0.85, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Quaked Floor", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DestroyObject(gameObject=var "Floor 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Floor 2", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Floor 3", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Bot 1", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Bot 2", delay=0, detachChildren=false)
    DestroyObject(gameObject=var "Bot 3", delay=0, detachChildren=false)
    PlayParticleEmitter(gameObject=var "Dust Break 2", emit=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Small (resources.assets), spawnPoint=var "Effect Centre", position=(unset), spawnMin=70, spawnMax=70, speedMin=0, speedMax=10, angleMin=85, angleMax=95, originVariationX=6, originVariationY=2, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Particle Rock Large (resources.assets), spawnPoint=var "Effect Centre", position=(unset), spawnMin=40, spawnMax=40, speedMin=0, speedMax=10, angleMin=85, angleMax=95, originVariationX=6, originVariationY=2, FSM="", FSMEvent="")
    ActivateGameObject(gameObject=var "Cracks 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Quaking: bool = true
  var Bot 1: gameObject = _Scenery/mine_1_quake_floor/Bot 1
  var Bot 2: gameObject = _Scenery/mine_1_quake_floor/Bot 2
  var Bot 3: gameObject = _Scenery/mine_1_quake_floor/Bot 3
  var Cracks 1: gameObject = _Scenery/mine_1_quake_floor/Cracks 1
  var Cracks 2: gameObject = _Scenery/mine_1_quake_floor/Cracks 2
  var Dust Break 1: gameObject = _Scenery/mine_1_quake_floor/Dust Break 1
  var Dust Break 2: gameObject = _Scenery/mine_1_quake_floor/Dust Break 2
  var Effect Centre: gameObject = _Scenery/mine_1_quake_floor/Effect Centre
  var Floor 1: gameObject = _Scenery/mine_1_quake_floor/Loose Floor
  var Floor 2: gameObject = _Scenery/mine_1_quake_floor/Loose Floor 1
  var Floor 3: gameObject = _Scenery/mine_1_quake_floor/Loose Floor 2
  var Hero Obj: gameObject = <null>
  var Quaked Floor: gameObject = _Scenery/mine_1_quake_floor/Quaked Floor
}
