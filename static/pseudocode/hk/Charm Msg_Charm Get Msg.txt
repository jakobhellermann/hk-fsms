# Charm Msg
# 1 location(s):
#   sharedassets6.assets / Charm Get Msg

fsm Charm Msg {
  start Init
  on DESTROY JOURNAL MSG → Recycle  // from any state

  state Init {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DESTROY JOURNAL MSG", delay=0, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    SetPosition(gameObject=Self, vector=var "Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Charm Icons", behaviour="CharmIconList", methodName="GetSprite", parameters=[var "ID"], storeResult=var "Sprite")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=var "Sprite")
    ConvertIntToString(intVariable=var "ID", stringVariable=var "ID String", format="", everyFrame=false)
    BuildString(stringParts=["CHARM_NAME_", var "ID String"], separator="", addToEnd=true, storeResult=var "Name Convo", everyFrame=false)
    GetLanguageString(sheetName="UI", convName=var "Name Convo", storeValue=var "Name")
    SetTextMeshProText(gameObject=var "Text", textString=var "Name")
    on FINISHED → Up
  }

  state Up {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=4.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Position: vector3 = (-11.91, -6.22, 0)

  var ID: int = 0
  var ID String: string = ""
  var Name: string = ""
  var Name Convo: string = ""
  var Icon: gameObject = Charm Get Msg/Icon
  var Self: gameObject = <null>
  var Text: gameObject = Charm Get Msg/Text
  var Sprite: object = <null>
}
