# Control
# 1 location(s):
#   White_Palace_11 / Palace Gate

fsm Control {
  start Pause

  state Pause {
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Closed", storeResult=var "Closed")
    FindChild(gameObject=Self, childName="door2", storeResult=var "Door")
    FindChild(gameObject=Self, childName="Open", storeResult=var "Open")
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="duskKnightDefeated", isTrue=→"OPEN", isFalse=(none))
    on OPEN → Opened
    on FINISHED → Idle
  }

  state Idle {
    ActivateGameObject(gameObject=var "Door", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Door", active=false)
    on ROYAL GUARD KILLED → Open Pause
  }

  state Opened {
    ActivateGameObject(gameObject=var "Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Door", active=true)
    FindChild(gameObject=Self, childName="Open Haze", storeResult=var "Open Haze")
    ActivateGameObject(gameObject=var "Open Haze", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Open Pause {
    SetPlayerDataBool(boolName="duskKnightDefeated", value=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  state Open {
    ActivateGameObject(gameObject=var "Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Door", active=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Door", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1.2, pitchMax=1.2, volume=1, delay=0, storePlayer=<null>)
  }

  var Closed: gameObject = <null>
  var Door: gameObject = <null>
  var Open: gameObject = <null>
  var Open Haze: gameObject = <null>
  var Self: gameObject = <null>
}
