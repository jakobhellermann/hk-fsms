# FSM
# 1 location(s):
#   GG_Land_of_Storms / Enviro Region Simple (1)

// uses template: enviro_region
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetPlayerDataInt(gameObject=var "GameManager", intName="environmentTypeDefault", storeValue=var "Enviro Default")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter
  }

  state Enter {
    SetPlayerDataInt(intName="environmentType", value=var "Enviro Type")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=checkEnvironment())
    Trigger2dEvent(trigger=OnTriggerExit2D, collideTag="", collideLayer="", sendEvent=→"EXIT", storeCollider=var "")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ENVIRO UPDATE", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="ENVIRO UPDATE")
    on EXIT → Exit
  }

  state Exit {
    SetPlayerDataInt(intName="environmentType", value=var "Enviro Default")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=checkEnvironment())
    Trigger2dEvent(trigger=OnTriggerStay2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ENVIRO UPDATE", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="ENVIRO UPDATE")
    on ENTER → Enter
  }

  var Enviro Type: int = 4

  var Enviro Default: int = 0
}
