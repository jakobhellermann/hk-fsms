fsm Spin Control {
  start Init

  state Spin {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=180, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GameObjectIsNull(gameObject=var "Parent", isNull=→"PARENT NULL", isNotNull=(none), storeResult=(unset), everyFrame=true)
    on STOP → Idle
    on PARENT NULL → Mage Killed
  }

  state Idle {
    GameObjectIsNull(gameObject=var "Parent", isNull=→"PARENT NULL", isNotNull=(none), storeResult=(unset), everyFrame=true)
    on SPIN → Spin
    on PARENT NULL → Mage Killed
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Mage Killed {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DESTROY", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    DestroyAllChildren(gameObject=var "Self", disable=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Active: bool = false
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
