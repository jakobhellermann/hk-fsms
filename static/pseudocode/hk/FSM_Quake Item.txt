// uses template: spawn_white_wave
fsm FSM {
  start Init
  on END → End  // from any state

  state Wait {
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Spawn {
    SpawnObjectFromGlobalPool(gameObject=White Wave R (resources.assets), spawnPoint=var "Self", position=var "Offset", rotation=(unset), storeObject=var "")
    on FINISHED → Wait
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Wait
  }

  state End {
  }

  var Offset: vector3 = (0, -0.5, 0)

  var Self: gameObject = <null>
}
