# Set Sharp Shadow Damage
# 2 location(s):
#   Knight_Pickup / Knight/Attacks
#   resources.assets / Knight:1/Attacks

fsm Set Sharp Shadow Damage {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Sharp Shadow", storeResult=var "Sharp Shadow")
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataInt(gameObject=var "GameManager", intName="nailDamage", storeValue=var "nailDamage")
    ConvertIntToFloat(intVariable=var "nailDamage", floatVariable=var "Nail Damage Float", everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName="equippedCharm_31", isTrue=→"MASTER", isFalse=→"BASE")
    on BASE → Base
    on MASTER → Master
  }

  state Base {
    SetFsmInt(gameObject=var "Sharp Shadow", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Base", everyFrame=false)  // disabled
    on FINISHED → Set
  }

  state Master {
    SetFsmInt(gameObject=var "Sharp Shadow", fsmName="damages_enemy", variableName="damageDealt", setValue=var "Dashmaster", everyFrame=false)  // disabled
    FloatMultiplyV2(floatVariable=var "Nail Damage Float", multiplyBy=1.5, everyFrame=false, fixedUpdate=false)
    on FINISHED → Set
  }

  state Idle {
    on CHARM INDICATOR CHECK → Check
  }

  state Set {
    ConvertFloatToInt(floatVariable=var "Nail Damage Float", intVariable=var "nailDamage", rounding=Nearest, everyFrame=false)
    SetFsmInt(gameObject=var "Sharp Shadow", fsmName="damages_enemy", variableName="damageDealt", setValue=var "nailDamage", everyFrame=false)
    on FINISHED → Idle
  }

  var Base: int = 5
  var Dashmaster: int = 10

  var Nail Damage Float: float = 0
  var nailDamage: int = 0
  var Sharp Shadow: gameObject = <null>
}
