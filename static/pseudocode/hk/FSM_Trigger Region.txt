# FSM
# 1 location(s):
#   GG_Unlock_Wastes / Sequence/Trigger Region

fsm FSM {
  start Idle

  state Idle {
    on TRIGGER ENTER 2D → Anim
  }

  state Anim {
    ActivateGameObject(gameObject=var "Animator", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar Lock", variableName="Roar Object", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    Wait(time=var "Anim Wait Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade
  }

  state Fade {
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=1.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=var "Fade Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Godseeker Dialogue
  }

  state Godseeker Dialogue {
    SetTextMeshProColor(gameObject=var "DialogueText", color=(0, 0, 0, 1), everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=false, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=true, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GG_UNLOCK_FINAL_TALK", "CP3"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Hero", audioClip=GS_standard_06 (sharedassets500.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on CONVO_FINISH → To Atrium
  }

  state To Atrium {
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    SetTextMeshProColor(gameObject=var "DialogueText", color=(1, 1, 1, 1), everyFrame=false)
    SetPlayerDataBool(boolName="seenGGWastes", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    BeginSceneTransition(sceneName="GG_Atrium", entryGateName="door_dreamEnter", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  var Anim Wait Time: float = 1.5
  var Fade Time: float = 2
  var Animator: gameObject = Sequence/Hallownest_call_event

  var Self: gameObject = <null>
}
