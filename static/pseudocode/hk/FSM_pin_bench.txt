// uses template: pin_activation
fsm FSM {
  start Check

  state Check {
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "Pin Type Bool", isTrue=(none), isFalse=→"INACTIVE")
    StringCompare(stringVariable=var "Specific Bool", compareTo="", equalEvent=→"ACTIVATE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(gameObject=var "GameManager", boolName=var "Specific Bool", isTrue=→"ACTIVATE", isFalse=→"INACTIVE")
    on ACTIVATE → Activate
    on INACTIVE → Inactive
  }

  state Activate {
    SetSpriteRenderer(gameObject=Self, active=true)
  }

  state Inactive {
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  var Pin Type Bool: string = "hasPinBench"
  var Specific Bool: string = ""
}
