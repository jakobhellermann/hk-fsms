# Activate
# 1 location(s):
#   RestingGrounds_04 / Binding Shield Activate

fsm Activate {
  start Detect

  state Detect {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"ENTER", storeCollider=var "")
    GetOwner(storeGameObject=var "Self")
    on ENTER → Activate
  }

  state Activate {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=dream_ghost_appear (resources.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on SHIELD DOWN → Shield Down
  }

  state Shield Down {
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Self: gameObject = <null>
}
