# Toll Gate
# 2 location(s):
#   Ruins1_05c / Ruins Gate 3c
#   Ruins1_05c / Ruins Gate 3a

fsm Toll Gate {
  start Idle

  state Idle {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Dust", storeResult=var "Dust")
    SetParticleEmission(gameObject=var "Dust", emission=false)  // disabled
    on OPEN → Audio
    on ACTIVATE → Destroy Self
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)  // disabled
    PlayParticleEmitter(gameObject=var "Dust", emit=0)
    SetParticleEmission(gameObject=var "Dust", emission=true)  // disabled
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Anim
  }

  state Destroy Self {
    SetParticleEmission(gameObject=var "Dust", emission=false)
    DestroySelf(detachChildren=true)
  }

  state Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Gate Open")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy Self
  }

  state Audio {
    BoolTest(boolVariable=var "No Audio", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (resources.assets), spawnPoint=var "Self", audioClip=switch_gate_gate (sharedassets39.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Open
  }

  var No Audio: bool = true

  var Dust: gameObject = <null>
  var Self: gameObject = <null>
}
