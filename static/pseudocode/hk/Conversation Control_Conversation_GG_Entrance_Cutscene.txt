# Conversation Control
# 1 location(s):
#   GG_Entrance_Cutscene / Conversation

fsm Conversation Control {
  start Wait

  state Init {
    GetOwner(storeGameObject=var "Self")
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)  // disabled
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE OUT", delay=0, everyFrame=false)
    Wait(time=6.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "GG_Entrance_cutscene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set Appearance
  }

  state Talk Finish {
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Show HUD
  }

  state Greet Normal {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["ATRIUM_ENTER_TEXT", "CP3"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=GS_standard_01 (sharedassets480.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=misc_rumble_impact (sharedassets13.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    on CONVO_FINISH → Reset Appearance
  }

  state Set Appearance {
    SetTextMeshProColor(gameObject=var "DialogueText", color=(0, 0, 0, 1), everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=false, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=true, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    GGCheckIsBossRushMode(trueEvent=→"GG MODE", falseEvent=→"FINISHED")
    on FINISHED → Greet Normal
    on GG MODE → Greet GG Mode
  }

  state Reset Appearance {
    SetTextMeshProColor(gameObject=var "DialogueText", color=(1, 1, 1, 1), everyFrame=false)
    SetTextMeshProAlignment(gameObject=var "DialogueText", topLeft=true, topRight=false, topCentre=false, topJustified=false, centreLeft=false, centreRight=false, centreCentre=false, centreJustified=false, bottomLeft=false, bottomRight=false, bottomCentre=false, bottomJustified=false)
    on FINISHED → End Pause
  }

  state Show HUD {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Change Scene
  }

  state Change Scene {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="EnterWithoutInput", parameters=[true], storeResult=(unset var))
    BeginSceneTransition(sceneName="GG_Atrium", entryGateName="door_dreamEnter", entryDelay=0, visualization=GodsAndGlory, preventCameraFadeOut=true)
  }

  state Pause {
    FindChild(gameObject=Self, childName="GG_Entrance_cutscene", storeResult=var "GG_Entrance_cutscene")
    FindChild(gameObject=Self, childName="Audio", storeResult=var "Audio")
    TransitionToAudioSnapshot(snapshot=Off (resources.assets), transitionTime=0.001)  // disabled
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=0.001)  // disabled
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Atmos
  }

  state Atmos {
    TransitionToAudioSnapshot(snapshot=On (resources.assets), transitionTime=4)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Audio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    on FINISHED → Init
  }

  state End Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Off (resources.assets), transitionTime=1)
    TransitionToAudioSnapshot(snapshot=at None (resources.assets), transitionTime=1)
    on FINISHED → Talk Finish
  }

  state GG Mode Fade {
    GGCheckIsBossRushMode(trueEvent=(none), falseEvent=→"FINISHED")
    SendEventToRegister(eventName="HIDE START BLANKER")
    SetFsmFloat(gameObject=var "HUD Blanker White", fsmName="Blanker Control", variableName="Fade Time", setValue=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Blanker White"), sendEvent="FADE IN", delay=0, everyFrame=false)
    SetRespawningHero(value=false)
    on FINISHED → Pause
  }

  state Wait {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → GG Mode Fade
  }

  state Greet GG Mode {
    CallMethodProper(gameObject=var "DialogueText", behaviour="DialogueBox", methodName="StartConversation", parameters=["GODSEEKER_GGMODE_INTRO", "CP3"], storeResult=(unset var))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=GS_standard_01 (sharedassets480.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (resources.assets), spawnPoint=var "Self", audioClip=misc_rumble_impact (sharedassets13.assets), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioStop(gameObject=Self)
    on CONVO_FINISH → Reset Appearance
  }

  var Audio: gameObject = <null>
  var GG_Entrance_cutscene: gameObject = <null>
  var Self: gameObject = <null>
}
