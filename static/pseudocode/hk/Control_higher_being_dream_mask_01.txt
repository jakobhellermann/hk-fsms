# Control
# 1 location(s):
#   Abyss_10 / Shade_Beast_Dream_Stuff/higher_being_dream_mask_01

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="Dream Attack", collideLayer="", sendEvent=→"HIT", storeCollider=var "")
    on HIT → Flash
  }

  state Flash {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UP INSTANT", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Dream Impact (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    NextFrameEvent(sendEvent=→"FINISHED")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RecoilLeftLong())
    on FINISHED → Flash 2
  }

  state Flash 2 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
}
