# Return from boss
# 1 location(s):
#   GG_Workshop / GG_Statue_Oblobbles/door_dreamReturnGG

// uses template: gg_statue_dream_return
fsm Return from boss {
  start Pause

  state Pause {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Door Entry
  }

  state Door Entry {
    GetName(gameObject=var "Self", storeName=var "Door Entry", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetEntryGateName", parameters=[], storeResult=var "Door Entered")
    StringCompare(stringVariable=var "Door Entry", compareTo=var "Door Entered", equalEvent=(none), notEqualEvent=→"INACTIVE", storeResult=(unset), everyFrame=false)
    SetPlayerDataString(gameObject=var "GameManager", stringName="bossReturnEntryGate", value="")
    on INACTIVE → Not Returning
    on FINISHED → Wait
  }

  state Not Returning {
  }

  state Transition Wait {
    CreateObject(gameObject=gg_battle_transitions (sharedassets425.assets), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Transitions", networkInstantiate=false, networkGroup=0)
    SendEventByName(eventTarget=GameObject(var "Transitions"), sendEvent="GG TRANSITION OUT INSTANT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Transition In
  }

  state Wait {
    ActivateGameObject(gameObject=var "Inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="ClearMPSendEvents", parameters=[], storeResult=(unset var))
    SendEventToRegister(eventName="K HATCHLING END")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Transition Wait
  }

  state Gotten Up {
    ActivateGameObject(gameObject=var "Inspect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Get Up Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gotten Up
  }

  state Transition In {
    SendEventByName(eventTarget=GameObject(var "Transitions"), sendEvent="GG TRANSITION IN STATUE", delay=0, everyFrame=false)
    on DREAM WAKE → Get Up Wait
  }

  var Inspect: gameObject = GG_Statue_Oblobbles/Inspect

  var Door Entered: string = ""
  var Door Entry: string = ""
  var Self: gameObject = <null>
  var Transitions: gameObject = <null>
}
