# Text Control
# 1 location(s):
#   Cinematic_MrMushroom / ToBeContinued

fsm Text Control {
  start Wait

  state Idle {
    ActivateGameObject(gameObject=var "TextMesh", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CINEMATIC END → Show Text
  }

  state Show Text {
    ActivateGameObject(gameObject=var "TextMesh", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FADE OUT", realTime=false)
    on FADE OUT → Fade
  }

  state Wait {
    NextFrameEvent(sendEvent=→"BEGIN")
    on BEGIN → Idle
  }

  state Fade {
    SendEventByName(eventTarget=GameObjectFSM(var "MainCamera", fsm="CameraFade"), sendEvent="START FADE", delay=0, everyFrame=false)
    Wait(time=2.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Load Next Level
  }

  state Load Next Level {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="LoadScene", parameters=["End_Game_Completion"], storeResult=(unset var))
  }

  var TextMesh: gameObject = ToBeContinued/TextMeshPro
}
