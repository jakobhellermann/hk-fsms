#   Cliffs_06 / PS4 Banishment Achievement Fixer

fsm PS4 Map Achievement Fixer {
  start Check Platform

  state Check Platform {
    SwitchOnPlatform(Standalone=(none), Switch=(none), PS4=→"PS4", XB1=(none), Other=(none))
    on PS4 → Wait for Touch
  }

  state Run {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="CheckBanishmentAchievement", parameters=[], storeResult=(unset var))
  }

  state Wait for Touch {
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"TRIGGER ENTER 2D", storeCollider=var "")
    on TRIGGER ENTER 2D → Run
  }
}
