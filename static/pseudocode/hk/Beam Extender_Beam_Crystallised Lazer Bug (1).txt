fsm Beam Extender {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Beam Impact", storeResult=var "Beam Impact")
    on START → Extend
  }

  state Extend {
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(1, 0), space=1, distance=100, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Hit Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=var "Ray Distance", repeatInterval=1, layerMask=[Terrain(8), Orbit Shield(31)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    FloatDivide(floatVariable=var "Ray Distance", divideBy=1.7, everyFrame=true)
    SetScale(gameObject=Self, vector=(unset), x=var "Ray Distance", y=(unset), z=(unset), everyFrame=true, lateUpdate=false)
    GetVector2XY(vector2Variable=var "Hit Point", storeX=var "Hit Point X", storeY=var "Hit Point Y", everyFrame=true)
    SetVector3XYZ(vector3Variable=var "Impact Pos", vector3Value=(unset), x=var "Hit Point X", y=var "Hit Point Y", z=0.001, everyFrame=true)
    SetPosition(gameObject=var "Beam Impact", vector=var "Impact Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true)
    SetRotation(gameObject=var "Beam Impact", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true, lateUpdate=false)
    on STOP → Init
  }

  var Hit Point X: float = 0
  var Hit Point Y: float = 0
  var Ray Distance: float = 0
  var Rotation: float = 0
  var Hit Point: vector2 = (0, 0)
  var Impact Pos: vector3 = (0, 0, 0)
  var Beam Impact: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
