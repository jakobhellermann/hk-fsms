fsm Control Mesh Renderer {
  start Hide
  on HIDE BOUND CHARMS → Hide  // from any state
  on SHOW BOUND CHARMS → Show  // from any state

  state Hide {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Show {
    SetMeshRenderer(gameObject=Self, active=true)
  }
}
