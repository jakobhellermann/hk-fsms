// uses template: grass_control
fsm grass_control {
  start Pause
  on SHAKE ALL GRASS → Push  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetPosition(gameObject=var "Self", vector=(unset), x=(unset), y=(unset), z=var "z", space=0, everyFrame=false)
    FloatCompare(float1=var "z", float2=1.8, tolerance=0, equal=→"INERT", lessThan=(none), greaterThan=→"INERT", everyFrame=false)
    FloatCompare(float1=var "z", float2=-1.8, tolerance=0, equal=→"INERT", lessThan=→"INERT", greaterThan=(none), everyFrame=false)
    on FINISHED → Setup Type
    on INERT → Inert
  }

  state Idle {
    RandomFloat(min=0, max=1, storeResult=var "StartTime")
    AnimatorPlay(gameObject=var "Self", stateName="Idle", layer=0, normalizedTime=var "StartTime", everyFrame=false)
    Trigger2dEvent(trigger=OnTriggerEnter2D, collideTag="", collideLayer="", sendEvent=→"PUSH", storeCollider=var "Pusher")
    on TAKE DAMAGE → Check Direction
    on SHARP SHADOW → Choose
    on PUSH → Should Push?
  }

  state Cut {
    AudioPlayRandom(gameObject=var "Self", audioClips=[grass_cut_5 (sharedassets6.assets), grass_cut_2 (sharedassets6.assets), grass_cut_3 (sharedassets6.assets), grass_cut_4 (sharedassets6.assets)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1)
    AnimatorPlay(gameObject=var "Self", stateName="Dead", layer=0, normalizedTime=0, everyFrame=false)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
  }

  state Push {
    AnimatorPlay(gameObject=var "Self", stateName="Push", layer=0, normalizedTime=0, everyFrame=false)
    on TAKE DAMAGE → Check Direction
    on PUSH FINISHED → Idle
    on SHARP SHADOW → Choose
  }

  state Check Direction {
    GetEventSender(sentByGameObject=var "Damager")
    GetFsmFloat(gameObject=var "Damager", fsmName="damages_enemy", variableName="direction", storeValue=var "Attack Direction", everyFrame=false)
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Left
    on RIGHT → Right
    on UP → Choose
    on DOWN → Choose
  }

  state Left {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=-0.6, y=0.6, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Type
  }

  state Right {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hit Effect")
    RandomFloat(min=340, max=380, storeResult=var "Chooser")
    SetRotation(gameObject=var "Hit Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chooser", space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Hit Effect", vector=(unset), x=0.6, y=0.6, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Check Type
  }

  state Choose {
    RandomBool(storeResult=var "Chooser Bool")
    BoolTest(boolVariable=(unset), isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    on LEFT → Left
    on RIGHT → Right
  }

  state Inert {
  }

  state Check Type {
    BoolTest(boolVariable=var "Infected", isTrue=→"INFECTED", isFalse=(none), everyFrame=false)
    IntSwitch(intVariable=var "Type", compareTo=[0, 1, 2, 3, 4], sendEvent=[→"WHITE", →"GREEN", →"RAG", →"CHILD TYPE", →"SIMPLE TYPE"], everyFrame=false)
    on WHITE → White
    on GREEN → Grass
    on SIMPLE TYPE → Cut
    on RAG → Play Child Particle
    on CHILD TYPE → Play Child Particle
    on INFECTED → Infected
  }

  state White {
    SetPosition(gameObject=var "Hit Effect", vector=(unset), x=(unset), y=(unset), z=0, space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Particle 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Particle 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SpawnObjectFromGlobalPool(gameObject=White Grass A (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(-90, -90, -0.01), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=White Grass B (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(-90, -90, -0.01), storeObject=var "")
    on FINISHED → Cut
  }

  state Grass {
    CreateObject(gameObject=Green Grass A (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(-90, -90, -0.01), storeObject=var "Hit Effect", networkInstantiate=false, networkGroup=0)
    SetPosition(gameObject=var "Hit Effect", vector=(unset), x=(unset), y=(unset), z=0, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Hit Effect", vector=(unset), x=(unset), y=(unset), z=0, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Cut
  }

  state Check Damage {
    GetFsmInt(gameObject=var "Pusher", fsmName="damages_enemy", variableName="damageDealt", storeValue=var "Damage", everyFrame=false)
    IntCompare(integer1=var "Damage", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Cancel Frame
    on FINISHED → Check Direction
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Setup Type {
    IntSwitch(intVariable=var "Type", compareTo=[0, 1, 2, 3, 4], sendEvent=[→"WHITE", →"GREEN", →"RAG", →"CHILD TYPE", →"SIMPLE TYPE"], everyFrame=false)
    on WHITE → Setup White
    on GREEN → Setup Green
    on SIMPLE TYPE → Idle
    on RAG → Setup Child Type
    on CHILD TYPE → Setup Child Type
  }

  state Setup White {
    FindGameObject(objectName="", withTag="Infected Flag", store=var "Infected Flag")
    GameObjectIsNull(gameObject=var "Infected Flag", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Infected", boolValue=true, everyFrame=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=(1, 0.54901963, 0.21176471, 1), everyFrame=false)
    on FINISHED → Idle
  }

  state Setup Green {
    on FINISHED → Idle
  }

  state Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  state Play Child Particle {
    PlayParticleEmitter(gameObject=var "Child Particle", emit=0)
    on FINISHED → Play Cut Sound
  }

  state Setup Child Type {
    GetChildNum(gameObject=Self, childIndex=0, store=var "Child Particle")
    on FINISHED → Idle
  }

  state Play Cut Sound {
    AudioPlayRandom(gameObject=var "Self", audioClips=[grass_cut_5 (sharedassets6.assets)], weights=[1], pitchMin=1, pitchMax=1)
    AnimatorPlay(gameObject=var "Self", stateName="Dead", layer=0, normalizedTime=0, everyFrame=false)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Child Particles
  }

  state Stop Child Particles {
    StopParticleEmitter(gameObject=var "Child Particle")
  }

  state Should Push? {
    BoolTest(boolVariable=var "No Push Anim", isTrue=→"CANCEL", isFalse=→"PUSH", everyFrame=false)
    on PUSH → SFX
    on CANCEL → Idle
  }

  state Infected {
    SetPosition(gameObject=var "Hit Effect", vector=(unset), x=(unset), y=(unset), z=0, space=1, everyFrame=false, lateUpdate=false)
    SpawnObjectFromGlobalPool(gameObject=Infected Grass B (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(-90, -90, -0.01), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Infected Grass A (resources.assets), spawnPoint=var "Self", position=(unset), rotation=(-90, -90, -0.01), storeObject=var "")
    on FINISHED → Cut
  }

  state SFX {
    AudioPlayRandom(gameObject=var "Self", audioClips=[grass_move_1 (sharedassets6.assets), grass_move_2 (sharedassets6.assets), grass_move_3 (sharedassets6.assets), grass_move_4 (sharedassets6.assets), grass_move_5 (sharedassets6.assets), grass_move_6 (sharedassets6.assets)], weights=[1, 1, 1, 1, 1, 1], pitchMin=1, pitchMax=1)
    on FINISHED → Push
  }

  var Type: int = 1
  var No Push Anim: bool = false

  var Attack Direction: float = 0
  var Chooser: float = 0
  var StartTime: float = 0
  var z: float = 0
  var Damage: int = 0
  var Chooser Bool: bool = false
  var Infected: bool = false
  var Sound: string = ""
  var Child Particle: gameObject = <null>
  var Damager: gameObject = <null>
  var Hit Effect: gameObject = <null>
  var Infected Flag: gameObject = <null>
  var Particle 1: gameObject = <null>
  var Particle 2: gameObject = <null>
  var Pusher: gameObject = <null>
  var Self: gameObject = <null>
}
