# Harpoon Evade
# 1 location(s):
#   Memory_Coral_Tower / Battle Scenes/Battle Scene Chamber 3/Wave 8/Coral Hunter

fsm Harpoon Evade {
  start Init

  state Init {
    AddEventRegister(target=Self, eventName="THROWING HARPOON")
    on FINISHED → Idle
  }

  state Idle {
    on THROWING HARPOON → Evade Check
  }

  state Evade Check {
    CheckFacingTarget(facingObject=var "Hero", target=var "Self", spriteFacesRight=false, everyFrame=false, facingEvent=(none), notFacingEvent=→"CANCEL", facingBool=(unset), notFacingBool=(unset))
    GetYDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false, allowNegatives=false)
    FloatCompare(float1=var "Distance", float2=3, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=10, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → Idle
    on FINISHED → Send Evade Event
  }

  state Send Evade Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EVADE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Distance: float = 0
  var Self: gameObject = Battle Scenes/Battle Scene Chamber 3/Wave 8/Coral Hunter
}
