# Rest Animation
# 1 location(s):
#   Song_Enclave / Black Thread States/Normal World/Enclave States/States/Level 2/Enclave Simple NPC Short Horned

// component disabled: this FSM does not run until something enables it
fsm Rest Animation {
  start Idle
  on DEACTIVATED → Deactivated  // from any state

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Facing
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "FacingLeft"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Resting", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    on CANCEL → Get Facing 2
  }

  state Rest to Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest to Idle Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Get Facing {
    LookAnimNpcGetFacingLeft(FacingLeft=→"LEFT", FacingRight=(none), StoreValue=(unset), EveryFrame=false, Target=Self)
    on FINISHED → Large Rest to Idle Right
    on LEFT → Large Idle Left to Resting
  }

  state Large Idle Left to Resting {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Idle Left to Resting", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Large Rest to Idle Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest to Idle Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Get Facing 2 {
    LookAnimNpcGetFacingLeft(FacingLeft=→"LEFT", FacingRight=(none), StoreValue=var "FacingLeft", EveryFrame=false, Target=Self)
    on FINISHED → Rest to Idle
    on LEFT → Large Rest to Idle Left
  }

  state Large Rest to Idle Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest to Idle Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Deactivated {
    on FINISHED → Idle
  }

  var FacingLeft: bool = false
  var Resting: bool = false
}
