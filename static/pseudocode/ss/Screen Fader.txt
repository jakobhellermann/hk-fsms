# Screen Fader
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game
#   Menu_Title / _GameCameras/HudCamera/In-game

fsm Screen Fader {
  start Init
  on CINEMATIC SKIP FADE → Cinematic Skip Fade  // from any state
  on SCENE FADE OUT INSTANT → Scene Fade Out Instant  // from any state
  on SCENE FADE TO MENU → Scene Fade To Menu  // from any state
  on DEATH RESPAWN → Death Respawn  // from any state
  on SCENE FADE IN → Scene Fade In  // from any state
  on SCENE FADE OUT → Scene Fade Out  // from any state
  on CUSTOM FADE → Custom Fade  // from any state
  on HAZARD FADE → Hazard Fade  // from any state
  on HAZARD RESPAWN → Hazard Respawn  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Screen Fader", storeResult=var "Screen Fader")
    on FINISHED → Idle
  }

  state Idle {
  }

  state Scene Fade Out {
    DebugLogConsole(logLevel=0, text="Scene Fade Out", sendToUnityLog=true)
    SetSpriteRenderer(gameObject=var "Screen Fader", active=true)
    GetConstantsValue(variableName="SCENE_TRANSITION_WAIT", storeValue=var "Scene Transition Wait")
    GetConstantsValue(variableName="SCENE_TRANSITION_FADE", storeValue=var "Scene Fade Time")
    FloatOperator(float1=var "Scene Transition Wait", float2=var "Scene Fade Time", operation=Subtract, storeResult=var "Scene Fade Delay", everyFrame=false)
    FloatSubtract(floatVariable=var "Scene Fade Time", subtract=0.01, everyFrame=false, perSecond=false)
    EaseSpriteColor(target=var "Screen Fader", fromValue=(0, 0, 0, 0), toValue=(0, 0, 0, 1), disableSpriteRendererIfTransparent=false, time=var "Scene Fade Time", speed=(unset), delay=var "Scene Fade Delay", easeType=21, reverse=false, finishEvent=(none), realTime=false)
  }

  state Scene Fade In {
    DebugLogConsole(logLevel=0, text="Scene Fade In", sendToUnityLog=true)
    GetConstantsValue(variableName="SCENE_TRANSITION_FADE", storeValue=var "Scene Fade Time")
    FloatSubtract(floatVariable=var "Scene Fade Time", subtract=0.01, everyFrame=false, perSecond=false)
    EaseSpriteColor(target=var "Screen Fader", fromValue=(0, 0, 0, 1), toValue=(0, 0, 0, 0), disableSpriteRendererIfTransparent=false, time=var "Scene Fade Time", speed=(unset), delay=0.01, easeType=21, reverse=false, finishEvent=(none), realTime=false)
  }

  state Custom Fade {
    DebugLogConsole(logLevel=0, text="Custom Fade", sendToUnityLog=true)
    FloatCompare(float1=var "Duration", float2=0, tolerance=0, equal=→"INSTANT", lessThan=→"INSTANT", greaterThan=(none), everyFrame=false)
    EaseSpriteColor(target=var "Screen Fader", fromValue=var "Start Colour", toValue=var "End Colour", disableSpriteRendererIfTransparent=false, time=var "Duration", speed=(unset), delay=var "Delay", easeType=21, reverse=false, finishEvent=(none), realTime=false)
    on INSTANT → Instant
  }

  state Death Respawn {
    DebugLogConsole(logLevel=0, text="Death Respawn", sendToUnityLog=true)
    EaseSpriteColor(target=var "Screen Fader", fromValue=(0, 0, 0, 1), toValue=(0, 0, 0, 0), disableSpriteRendererIfTransparent=false, time=0.5, speed=(unset), delay=0.1, easeType=21, reverse=false, finishEvent=(none), realTime=false)
  }

  state Hazard Fade {
    DebugLogConsole(logLevel=0, text="Hazard Fade", sendToUnityLog=true)
    EaseSpriteColor(target=var "Screen Fader", fromValue=(0, 0, 0, 0), toValue=(0, 0, 0, 1), disableSpriteRendererIfTransparent=false, time=0.55, speed=(unset), delay=0.1, easeType=21, reverse=false, finishEvent=(none), realTime=false)
  }

  state Hazard Respawn {
    DebugLogConsole(logLevel=0, text="Hazard Respawn", sendToUnityLog=true)
    EaseSpriteColor(target=var "Screen Fader", fromValue=(0, 0, 0, 1), toValue=(0, 0, 0, 0), disableSpriteRendererIfTransparent=false, time=0.5, speed=(unset), delay=0.1, easeType=21, reverse=false, finishEvent=(none), realTime=false)
  }

  state Instant {
    SetSpriteRendererColor(TintColor=var "End Colour", Target=var "Screen Fader")
    SetSpriteRenderer(gameObject=var "Screen Fader", active=true)
  }

  state Scene Fade To Menu {
    DebugLogConsole(logLevel=0, text="Scene Fade To Menu", sendToUnityLog=true)
    SetSpriteRenderer(gameObject=var "Screen Fader", active=true)
    EaseSpriteColor(target=var "Screen Fader", fromValue=(0, 0, 0, 0), toValue=(0, 0, 0, 1), disableSpriteRendererIfTransparent=false, time=0.33, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=true)
  }

  state Scene Fade Out Instant {
    DebugLogConsole(logLevel=0, text="Fade Out Instant", sendToUnityLog=true)
    SetSpriteRenderer(gameObject=var "Screen Fader", active=true)
    SetSpriteRendererColor(TintColor=(0, 0, 0, 1), Target=var "Screen Fader")
  }

  state Cinematic Skip Fade {
    DebugLogConsole(logLevel=0, text="Cinematic Skip Fade", sendToUnityLog=true)
    SetSpriteRenderer(gameObject=var "Screen Fader", active=true)
    GetConstantsValue(variableName="CINEMATIC_SKIP_FADE_TIME", storeValue=var "Scene Fade Time")
    FloatSubtract(floatVariable=var "Scene Fade Time", subtract=0.01, everyFrame=false, perSecond=false)
    EaseSpriteColor(target=var "Screen Fader", fromValue=(0, 0, 0, 0), toValue=(0, 0, 0, 1), disableSpriteRendererIfTransparent=false, time=var "Scene Fade Time", speed=(unset), delay=var "Scene Fade Delay", easeType=21, reverse=false, finishEvent=(none), realTime=false)
  }

  var Delay: float = 0
  var Duration: float = 0
  var Scene Transition Wait: float = 0
  var Scene Fade Delay: float = 0
  var Scene Fade Time: float = 0
  var End Colour: color = (0, 0, 0, 1)
  var Start Colour: color = (0, 0, 0, 1)
  var Screen Fader: gameObject = _GameCameras/HudCamera/In-game/Screen Fader
}
