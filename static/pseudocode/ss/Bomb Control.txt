# Bomb Control
# 1 location(s):
#   herodynamic_assets_all.bundle / Flea Brew Bottle

fsm Bomb Control {
  start Init
  on DISABLE → Report Broken  // from any state
  on LAVA → Explode  // from any state

  state Init {
    SetCircleCollider(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetSpriteRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Explode Particles", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Started", boolValue=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Rotation
  }

  state Idle {
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"SMASH", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Explosion", sendEvent=→"SMASH", storeCollider=var "")
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Special Attack", sendEvent=→"SMASH", storeCollider=var "")
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Nail Attack", sendEvent=→"SMASH", storeCollider=var "Slash Object")
    on SMASH → Explode
  }

  state Explode {
    SetBoolValue(boolVariable=var "Started", boolValue=false, everyFrame=false)
    ToolItemStatesLiquidReportBottleBroken(Tool=Flea Brew (dataassets_assets_assets/dataassets/tools/toolitems.bundle))
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Explode Particles", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=globe_break (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Rotation {
    SetBoolValue(boolVariable=var "Started", boolValue=true, everyFrame=false)
    PlayAudioEvent(audioClip=trobbio_inair_throw (sfxstatic_assets_trobbio.bundle), pitchMin=1.15, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=16.5, space=0, everyFrame=false, lateUpdate=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Speed", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-16.5, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Report Broken {
    BoolTest(boolVariable=var "Started", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    ToolItemStatesLiquidReportBottleBroken(Tool=Flea Brew (dataassets_assets_assets/dataassets/tools/toolitems.bundle))
    on FINISHED → State 2
  }

  state State 2 {
  }

  var X Speed: float = 0
  var Started: bool = false
  var Explode Particles: gameObject = Flea Brew Bottle/Glass Burst Pt
  var Slash Object: gameObject = <null>
  var Break Effect: gameObject = Flea Brew Bottle/Break Effect
}
