# Control
# 1 location(s):
#   localpoolprefabs_assets_areashellwood.bundle / Pollen Shot FQ

fsm Control {
  start First

  state Pause {
    PlayParticleEmitter(gameObject=var "Pt Pollen", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetAudioSource(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Pop Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Max
  }

  state Fly {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=true)
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    RandomFloat(min=10, max=20, storeResult=var "Angle Add")
    RandomlyFlipFloat(storeResult=var "Angle Add")
    RandomFloat(min=0.5, max=0.75, storeResult=var "Ease Time")
    FloatOperator(float1=var "Angle", float2=var "Angle Add", operation=Add, storeResult=var "Angle Target", everyFrame=false)
    EaseFloat(fromValue=var "Angle", toValue=var "Angle Target", floatVariable=var "New Angle", time=var "Ease Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"SHIFT", realTime=false)
    ClampAngle(angleVariable=var "New Angle", minValue=var "Angle Min", maxValue=var "Angle Max", everyFrame=true)
    SetVelocityAsAngle(gameObject=Self, angle=var "New Angle", speed=10, everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"HIT HERO", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "")
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"BREAK", sendEventSpikes=→"BREAK", sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    FloatAdd(floatVariable=var "Lifetimer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Lifetimer", float2=0, tolerance=0, equal=(none), lessThan=→"BREAK", greaterThan=(none), everyFrame=true)
    on SHIFT → Fly
    on BREAK → Break
    on PROJECTILE BREAK → Break
    on MULTI HIT CONNECT → Multhit Shift
  }

  state Set Max {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    FloatOperator(float1=var "Angle", float2=25, operation=Add, storeResult=var "Angle Max", everyFrame=false)
    FloatOperator(float1=var "Angle", float2=25, operation=Subtract, storeResult=var "Angle Min", everyFrame=false)
    SetFloatValue(floatVariable=var "Lifetimer", floatValue=3, everyFrame=false)
    on FINISHED → Fly
  }

  state Break {
    SetAudioSource(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    StopParticleEmitter(gameObject=var "Pt Pollen")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pollen Pop", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Pop Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEventRandom(audioClips=[bloom_shooter_proj_explode_1 (sfxstatic_assets_areashellwood.bundle), bloom_shooter_proj_explode_2 (sfxstatic_assets_areashellwood.bundle), bloom_shooter_proj_explode_3 (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.8, pitchMax=1.2, volume=0.7, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Rec Pause
  }

  state Break Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Break
  }

  state Rec Pause {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Multhit Shift {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Translate", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Translate", multiplyBy=0.25, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=var "X Translate", y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Break
  }

  state First {
    BoolTest(boolVariable=var "init", isTrue=→"SKIP", isFalse=(none), everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Pollen", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetAudioSource(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Pop Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "init", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Disable Pop
    on SKIP → Pause
  }

  state Disable Pop {
    ActivateGameObject(gameObject=var "Pop Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set Max
  }

  var Angle: float = 0
  var Angle Add: float = 0
  var Angle Max: float = 0
  var Angle Min: float = 0
  var Angle Target: float = 0
  var Ease Time: float = 0
  var New Angle: float = 0
  var Lifetimer: float = 0
  var X Translate: float = 0
  var init: bool = false
  var Pt Break: gameObject = Pollen Shot FQ/Pt Break
  var Pt Pollen: gameObject = Pollen Shot FQ/Pt Pollen
  var Pop Damager: gameObject = Pollen Shot FQ/Pop Damager
}
