# Control
# 1 location(s):
#   Bone_East_17b / Battle Music

fsm Control {
  start Init

  state Init {
    on ANT ALERT → Increment
    on ANT KILLED → Decrement
  }

  state Increment {
    IntAdd(intVariable=var "Active", add=1, everyFrame=false)
    on FINISHED → Check
  }

  state Decrement {
    IntAdd(intVariable=var "Active", add=-1, everyFrame=false)
    on FINISHED → Check
  }

  state Check {
    IntCompare(integer1=var "Active", integer2=0, equal=→"SUB", lessThan=→"SUB", greaterThan=→"ACTION", everyFrame=false)
    on SUB → Sub
    on ACTION → Action
  }

  state Sub {
    TransitionToAudioSnapshot(snapshot=Sub Area (audioobjects_assets_all.bundle), transitionTime=5)
    on ANT ALERT → Increment
    on ANT KILLED → Decrement
  }

  state Action {
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
    on ANT ALERT → Increment
    on ANT KILLED → Decrement
  }

  var Active: int = 0
}
