# Phase Control
# 1 location(s):
#   Bone_East_08_boss_golem / Boss Scene/song_golem/Song_Butt/SG_waist/Torso/SG_head

fsm Phase Control {
  start Init
  on ZERO HP → Zero HP  // from any state

  state Init {
    GetGrandParent(gameObject=Self, storeResult=var "Waist")
    GetGrandParent(gameObject=var "Waist", storeResult=var "Golem")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Phase 1
  }

  state Phase 1 {
    on TOOK DAMAGE → HP Check 1
  }

  state HP Check 1 {
    CompareHP(enemy=var "Self", integer2=451, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Stun 1
    on FINISHED → Phase 1
  }

  state Phase 2 {
    on TOOK DAMAGE → HP Check 2
  }

  state HP Check 2 {
    CompareHP(enemy=var "Self", integer2=326, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Stun 2
    on FINISHED → Phase 2
  }

  state Phase 3 {
    on TOOK DAMAGE → HP Check 3
  }

  state HP Check 3 {
    CompareHP(enemy=var "Self", integer2=201, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=false)
    on NEXT → Stun 3
    on FINISHED → Phase 3
  }

  state Phase 4 {
  }

  state Zero HP {
    SendEventByName(eventTarget=GameObject(var "Golem"), sendEvent="ZERO HP", delay=0, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetIsDead(target=Self, setValue=true)
  }

  state Stun 1 {
    SendEventByName(eventTarget=GameObject(var "Golem"), sendEvent="STUN", delay=0, everyFrame=false)
    SetFsmInt(gameObject=var "Golem", fsmName="Control", variableName="Phase", setValue=2, everyFrame=false)
    on FINISHED → Phase 2
  }

  state Stun 2 {
    SendEventByName(eventTarget=GameObject(var "Golem"), sendEvent="STUN", delay=0, everyFrame=false)
    SetFsmInt(gameObject=var "Golem", fsmName="Control", variableName="Phase", setValue=3, everyFrame=false)
    on FINISHED → Phase 3
  }

  state Stun 3 {
    SendEventToRegister(eventName="FINAL PHASE")  // disabled
    SendEventByName(eventTarget=GameObject(var "Golem"), sendEvent="STUN", delay=0, everyFrame=false)
    SetFsmInt(gameObject=var "Golem", fsmName="Control", variableName="Phase", setValue=4, everyFrame=false)
    on FINISHED → Phase 4
  }

  var Golem: gameObject = <null>
  var Self: gameObject = <null>
  var Waist: gameObject = <null>
}
