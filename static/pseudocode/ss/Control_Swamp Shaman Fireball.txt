# Control
# 1 location(s):
#   localpoolprefabs_assets_areaswamp.bundle / Swamp Shaman Fireball

fsm Control {
  start Init

  state Init {
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Fireball Hit standard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fireball Hit wall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fly
  }

  state Fly {
    PlayParticleEmitterInState(gameObject=var "Pt Fly")
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Wait(time=1, finishEvent=→"DEALT DAMAGE", realTime=false)
    on DEALT DAMAGE → Hero Burst
    on COLLIDE → Wall Burst
  }

  state Hero Burst {
    ActivateGameObject(gameObject=var "Fireball Hit standard", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Burst
  }

  state Burst {
    PlayAudioEvent(audioClip=hero_quake_spell_impact (sfxstatic_assets_areaswamp.bundle), pitchMin=1.1, pitchMax=1.3, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    on FINISHED → Recycle
  }

  state Wall Burst {
    ActivateGameObject(gameObject=var "Fireball Hit wall", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Burst
  }

  state Recycle {
    ActivateGameObject(gameObject=var "Fireball Hit standard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fireball Hit wall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RecycleSelf()
  }

  var Pt Fly: gameObject = Swamp Shaman Fireball/Pt Fly
  var Fireball Hit standard: gameObject = Swamp Shaman Fireball/Fireball Hit standard
  var Fireball Hit wall: gameObject = Swamp Shaman Fireball/Fireball Hit wall
  var Pt Break: gameObject = Swamp Shaman Fireball/Pt Break
}
