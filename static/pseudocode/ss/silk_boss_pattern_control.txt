# silk_boss_pattern_control
# 2 location(s):
#   Cradle_03 / Boss Scene/Strand Patterns/Pattern 1
#   Cradle_03 / Boss Scene/Strand Patterns/Pattern 2

// uses template: silk_boss_pattern_control
fsm silk_boss_pattern_control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=Self, storeResult=var "BG Fader")
    GetPosition(gameObject=Self, vector=var "Init Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on ATTACK → Random Shift
  }

  state Web Appear {
    PlayAudioEvent(audioClip=silk_boss_web_attack_buildup (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="APPEAR", delay=0, everyFrame=false)
    SetSpriteRenderer(gameObject=var "BG Fader", active=true)
    EaseSpriteColor(target=var "BG Fader", fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.4392157), disableSpriteRendererIfTransparent=false, time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Web Burst Start
    on SILK STUNNED → Cancel
  }

  state Web Burst Start {
    AudioPlayerOneShotSingleV2(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClip=silk_boss_web_attack_burst (sfxstatic_assets_areacradle.bundle), vibrationDataAsset=<null>, playVibration=false, pitchMin=1, pitchMax=1, volume=1, delay=0.5, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BURST", delay=0, everyFrame=false)
    EaseSpriteColor(target=var "BG Fader", fromValue=(1, 1, 1, 0.441), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.25, speed=(unset), delay=0.75, easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reparent
    on SILK STUNNED → Cancel
  }

  state Random Shift {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    RandomlyFlipScale(gameObject=Self)
    RandomlyFlipYScale(gameObject=Self)
    SetPosition(gameObject=Self, vector=var "Init Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=Self, translateMin=(-3, -2, 0), translateMax=(3, 0, 0), space=0)
    on FINISHED → Web Appear
  }

  state Reparent {
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Cancel {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CANCEL", delay=0, everyFrame=false)
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SetSpriteRenderer(gameObject=var "BG Fader", active=false)
    on FINISHED → Idle
  }

  var X Pos: float = 0
  var Y Pos: float = 0
  var Init Pos: vector3 = (0, 0, 0)
  var web_strand_single: gameObject = <null>
  var BG Fader: gameObject = <null>
  var Parent: gameObject = <null>
}
