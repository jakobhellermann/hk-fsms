# Animation
# 1 location(s):
#   Belltown / Town States/Spinner Defeated/Bagpipers/Leader

// uses template: bagpiper_animation
fsm Animation {
  start Init
  on SING → Sing  // from any state
  on RESPONSE SHORT → Set Response Short  // from any state
  on RESPONSE LONG → Set Response Long  // from any state

  state Play Idle {
    BoolTest(boolVariable=var "Is Talking", isTrue=→"TALK", isFalse=(none), everyFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Play Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    RandomInt(min=0, max=5, storeResult=var "Frame Offset", inclusiveMax=true, noRepeat=false)
    Tk2dPlayFrame(gameObject=Self, frame=var "Frame Offset")
    on TALK → Talking
  }

  sequence state Sing {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Sing Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on SING END → Sing End
  }

  sequence state Sing End {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Sing End Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Play Idle
  }

  sequence state Response {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Sing Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=var "Response Duration", finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName=var "Sing End Anim", AnimationCompleteEvent=(none))
    on FINISHED → Play Idle
  }

  state Init {
    on FINISHED → Play Idle
  }

  state Set Response Long {
    SetFloatValue(floatVariable=var "Response Duration", floatValue=1, everyFrame=false)
    on FINISHED → Response
  }

  state Set Response Short {
    SetFloatValue(floatVariable=var "Response Duration", floatValue=0.5, everyFrame=false)
    on FINISHED → Response
  }

  state Talking {
    BoolTest(boolVariable=var "Is Talking", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="L Talk", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on CANCEL → Talk End
  }

  state Talk End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="L Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Play Idle
  }

  var Play Anim: string = "L Play"
  var Sing Anim: string = "L Sing"
  var Sing End Anim: string = "L Sing End"

  var Response Duration: float = 0
  var Frame Offset: int = 0
  var Is Talking: bool = false
}
