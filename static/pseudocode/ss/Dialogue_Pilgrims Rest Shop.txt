# Dialogue
# 1 location(s):
#   Bone_East_10_Room / Black Thread States/Normal World/Pilgrims Rest Shop

fsm Dialogue {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Shop Camlock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Post Ruckus?
  }

  state Has Met? {
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=false, Target=Self)
    PlayerDataVariableTest(VariableName="MetPilgrimsRestShop", ExpectedValue=true, IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on FALSE → Either Talk
    on TRUE → Idle
  }

  state Either Talk {
    SetBoolValue(boolVariable=var "Play Voice At Shop Open", boolValue=false, everyFrame=false)
    SetProperty(targetProperty=PlayMakerNPC.InteractLabel on Black Thread States/Normal World/Pilgrims Rest Shop/Shop Region@PlayMakerNPC, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.8, MaxReactDelay=2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT TALK → Set Control Talk
    on INTERACT SHOP → Set Control Shop
    on SING → Sing
    on STARTLE → Startle
  }

  state Meet Dialogue {
    SetPlayerDataVariable(VariableName="MetPilgrimsRestShop", SetValue=true)
    SetProperty(targetProperty=PlayMakerNPC.InteractLabel on Black Thread States/Normal World/Pilgrims Rest Shop/Shop Region@PlayMakerNPC, everyFrame=false)
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_MEET", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    on CONVO_END → Delivery? 2
  }

  state End Current {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=var "Current Controller", UseChildren=false)
    on FINISHED → Idle
  }

  state Set Control Talk {
    SetGameObject(variable=var "Current Controller", gameObject=var "Talk Controller", everyFrame=false)
    on FINISHED → Post Ruckus? 2
  }

  state Set Control Shop {
    SetGameObject(variable=var "Current Controller", gameObject=var "Shop Controller", everyFrame=false)
    on FINISHED → Post Ruckus? 2
  }

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.8, MaxReactDelay=2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on INTERACT TALK → Set Control Talk 2
    on INTERACT SHOP → Set Control Shop 2
    on SING → Sing
    on STARTLE → Startle
  }

  state Set Control Talk 2 {
    SetGameObject(variable=var "Current Controller", gameObject=var "Talk Controller", everyFrame=false)
    on FINISHED → Delivery?
  }

  state Set Control Shop 2 {
    SetBoolValue(boolVariable=var "Play Voice At Shop Open", boolValue=true, everyFrame=false)
    SetGameObject(variable=var "Current Controller", gameObject=var "Shop Controller", everyFrame=false)
    on FINISHED → Delivery? 2
  }

  state Shop? {
    EndDialogue(ReturnControl=false, ReturnHUD=true, Target=var "Current Controller", UseChildren=false)
    GetShopObject(StoreObject=var "Shop Object", Target=Self)
    ShopHasStock(Target=Self, trueEvent=(none), falseEvent=→"SHOP NO STOCK", storeValue=(unset))
    on SHOP NO STOCK → No Stock
    on FINISHED → Should Play Voice?
  }

  state No Stock {
    DebugLogConsole(logLevel=2, text="Empty shop behaviour not implemented, it was assumed this shop would never be empty!", sendToUnityLog=true)
    SetBoolValue(boolVariable=var "Play Voice At Shop Open", boolValue=false, everyFrame=false)
    on CONVO_END → Shop End
  }

  state Shop Up {
    ActivateGameObject(gameObject=var "Shop Camlock", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Shop Object"), sendEvent="SHOP UP", delay=0, everyFrame=false)
    on SHOP CLOSED → Purchased?
    on SHOP NO STOCK → Purchased?
  }

  state Purchased? {
    ShopWasItemPurchased(Target=Self, trueEvent=→"TRUE", falseEvent=→"FALSE", storeValue=(unset))
    on TRUE → Purchased.
    on FALSE → Not purchased
  }

  state Purchased. {
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_PURCHASED", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    on CONVO_END → Shop End
  }

  state Not purchased {
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_CANCEL", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    on CONVO_END → Shop End
  }

  state Shop End {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=var "Current Controller", UseChildren=false)
    on FINISHED → Idle
  }

  state Talk {
    GetPlayerDataVariable(VariableName="PilgrimsRestShopIdleTalkState", StoreValue=var "Idle Talk State")
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    IntSwitch(intVariable=var "Idle Talk State", compareTo=[0, 1], sendEvent=[→"1", →"2"], everyFrame=false)
    on 1 → Talk 1
    on 2 → Talk 2
    on FINISHED → Talk Repeat
  }

  state Talk 1 {
    SetPlayerDataVariable(VariableName="PilgrimsRestShopIdleTalkState", SetValue=1)
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_TALK1", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    on CONVO_END → End Current
  }

  state Talk 2 {
    SetPlayerDataVariable(VariableName="PilgrimsRestShopIdleTalkState", SetValue=2)
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_TALK2", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    on CONVO_END → End Current
  }

  state Talk Repeat {
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_TALK_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    on CONVO_END → End Current
  }

  state Sing {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.7, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SetBoolValue(boolVariable=var "Did Sing", boolValue=true, everyFrame=false)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Mort_sing (sfxdynamic_assets_areabone.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Restart Idle
  }

  state Post Ruckus {
    SetPlayerDataBool(boolName="pilgrimRestMerchant_RhinoRuckusConvo", value=true)
    SetPlayerDataVariable(VariableName="MetPilgrimsRestShop", SetValue=true)
    SetProperty(targetProperty=PlayMakerNPC.InteractLabel on Black Thread States/Normal World/Pilgrims Rest Shop/Shop Region@PlayMakerNPC, everyFrame=false)
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_RUCKUS", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    on CONVO_END → End Current
  }

  state Post Ruckus? {
    PlayerDataBoolTrueAndFalse(trueBool="didRhinoRuckus", falseBool="pilgrimRestMerchant_RhinoRuckusConvo", isTrue=→"TRUE", isFalse=→"FALSE")
    on FALSE → Has Met?
    on TRUE → Either Talk
  }

  state Post Ruckus? 2 {
    PlayerDataBoolTrueAndFalse(trueBool="didRhinoRuckus", falseBool="pilgrimRestMerchant_RhinoRuckusConvo", isTrue=→"TRUE", isFalse=→"FALSE")
    on FALSE → Meet Dialogue
    on TRUE → Post Ruckus
  }

  state Talk Delivery {
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_TALK_DELIVERY", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    on CONVO_END → Delivery Quest End
  }

  sequence state Delivery Quest End {
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=var "Current Controller", UseChildren=false)
    EndQuest(ConsumeCurrency=true, Quest=Courier Delivery Pilgrims Rest (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    on FINISHED → Talk Delivery Post
  }

  state Delivery? {
    CheckQuestStateV2(NotTrackedEvent=(none), IncompleteEvent=→"DELIVERY", CompletedEvent=(none), Quest=Courier Delivery Pilgrims Rest (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    on DELIVERY → Talk Delivery
    on FINISHED → Talk
  }

  state Delivery? 2 {
    CheckQuestStateV2(NotTrackedEvent=(none), IncompleteEvent=→"DELIVERY", CompletedEvent=(none), Quest=Courier Delivery Pilgrims Rest (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    on DELIVERY → Talk Delivery
    on FINISHED → Remeet Dialogue
  }

  state Talk Delivery Post {
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_TALK_DELIVERY_POST", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    on CONVO_END → Delivery Quest Reward
  }

  sequence state Delivery Quest Reward {
    EndDialogue(ReturnControl=false, ReturnHUD=true, Target=var "Current Controller", UseChildren=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    GetQuestRewardV2(StoreReward=var "Reward", StoreAmount=var "Reward Amount", Quest=Courier Delivery Pilgrims Rest (dataassets_assets_assets/dataassets/questsystem/quests.bundle))

    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=var "Current Controller", UseChildren=false)
    on FINISHED → Idle
  }

  state Remeet Dialogue {
    BoolTest(boolVariable=var "Spoken", isTrue=→"SKIP", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    RunDialogue(Sheet=var "Translation Sheet", Key="PILGRIM_REST_SHOP_REMEET", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Current Controller")
    SetBoolValue(boolVariable=var "Play Voice At Shop Open", boolValue=false, everyFrame=false)
    on SKIP → Shop?
    on CONVO_END → Shop?
  }

  state Startle {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flinch", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Mort_startle (sfxdynamic_assets_areabone.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)

  }

    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Right")
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    on FINISHED → Post Ruckus?
  }

  state Should Play Voice? {
    BoolTest(boolVariable=var "Play Voice At Shop Open", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Play Voice At Shop Open", boolValue=false, everyFrame=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Mort_talk (sfxdynamic_assets_areabone.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Shop Up
  }

  state Set Play Voice {
    SetBoolValue(boolVariable=var "Play Voice At Shop Open", boolValue=true, everyFrame=false)
    on FINISHED → Shop?
  }

  var Reward Amount: int = 0

  var Reward: object = <null>
}
