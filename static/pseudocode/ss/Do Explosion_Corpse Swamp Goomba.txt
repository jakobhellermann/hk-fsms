# Do Explosion
# 1 location(s):
#   enemycorpses_assets_areaswamp.bundle / Corpse Swamp Goomba

fsm Do Explosion {
  start Fly Out

  state Antic {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=swamp_bounce_pod_antic_loop (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Looping Audio")
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    on FINISHED → Slow Down
  }

  state Explode {
    AudioStop(gameObject=var "Looping Audio", fadeTime=0)
    PlayAudioEvent(audioClip=swamp_bounce_pod_explode (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Explode", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Swamp Bounce Pod Explosion (localpoolprefabs_assets_areaswamp.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Fly Out {
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=var "Audio Loop Voice", volume=1, oneShotClip=<null>)
    ClampSpeed(gameObject=Self, speedMin=(unset), speedMax=var "Max Speed", everyFrame=true)
    on FINISHED → Antic
  }

  state Slow Down {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    DecelerateV2(gameObject=Self, deceleration=0.87, brakeOnExit=false)
    Wait(time=0.45, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air 2")
    on FINISHED → Explode
  }

  var Max Speed: float = 15
  var Self: gameObject = Corpse Swamp Goomba
  var Looping Audio: gameObject = <null>
  var Antic Effects: gameObject = Corpse Swamp Goomba/Antic Effects
  var Audio Loop Voice: gameObject = Corpse Swamp Goomba/Audio Loop Voice
}
