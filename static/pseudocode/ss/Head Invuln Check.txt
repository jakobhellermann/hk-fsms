// component disabled: this FSM does not run until something enables it
fsm Head Invuln Check {
  start Idle
  on STOP HEAD CHECK → Stop  // from any state

  state Idle {
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0, resetOnStateExit=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Head Invuln Range", storeResult=false, sendEvent="START INVULN", outOfRangeEvent="", everyFrame=true)
    on START INVULN → Head Invuln
  }

  state Stop {
  }

  state Head Invuln {
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=4, resetOnStateExit=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Head Invuln Range", storeResult=false, sendEvent="", outOfRangeEvent="STOP INVULN", everyFrame=true)
    on STOP INVULN → Idle
  }
}
