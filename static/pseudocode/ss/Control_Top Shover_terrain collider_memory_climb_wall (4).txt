fsm Control {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In Range
  }

  state In Range {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Y Velocity", space=0, everyFrame=true)
    GetHeroCState(VariableName="onGround", StoreValue=var "On Ground", EveryFrame=true)
    BoolTest(boolVariable=var "On Ground", isTrue=→"GROUNDED", isFalse=(none), everyFrame=true)
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Y Velocity", space=0, everyFrame=true)  // disabled
    FloatCompare(float1=var "Y Velocity", float2=0, tolerance=0.1, equal=→"GROUNDED", lessThan=(none), greaterThan=(none), everyFrame=true)  // disabled
    on EXIT → Idle
    on GROUNDED → Shove
  }

  state Shove {
    Translate(gameObject=var "Hero", vector=(unset), x=var "X Translate", y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → In Range
    on EXIT → Idle
  }

  var X Translate: float = 0.2

  var Y Velocity: float = 0
  var On Ground: bool = false
}
