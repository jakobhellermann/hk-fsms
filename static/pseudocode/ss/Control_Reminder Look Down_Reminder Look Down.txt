# Control
# 1 location(s):
#   Tut_01 / Reminder Look Down/Reminder Look Down

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait For SF
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    on ACTIVATE → Activated
    on ENTER → Send
  }

  state Activated {
  }

  state Send {
    SendEventToRegister(eventName="REMINDER LOOK DOWN")
    CheckTargetDirection(gameObject=var "MainCamera", target=var "Hero", aboveEvent=→"END", belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=4, reverseIfNegativeScale=false, everyFrame=true)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=→"END", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=var "End if Hero Below By", reverseIfNegativeScale=false, everyFrame=true)
    on END → End Pause
    on LEAVING SCENE → End
    on END 2 → State 1
  }

  state End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventToRegister(eventName="REMINDER LOOK DOWN END")
  }

  state Wait For SF {
    BoolTest(boolVariable=var "No Silkfly", isTrue=→"SILKFLY LEAVE", isFalse=(none), everyFrame=false)
    on SILKFLY LEAVE → Idle Pause
  }

  state Idle Pause {
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state State 1 {
  }

  var End if Hero Below By: float = -3
  var No Silkfly: bool = false

  var Y Distance: float = 0
  var Activated: bool = false
}
