# Control
# 1 location(s):
#   Belltown_Shrine / Black Thread States Thread Only Variant/Normal World/Boss Scene

fsm Control {
  start Init
  on SPINNER SEQUENCE END → Set Sequence Completed  // from any state

  state Init {
    FindChild(gameObject=Self, childName="CamLock B", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Intro Spinner", storeResult=var "Intro Spinner")
    FindChild(gameObject=Self, childName="Spinner Boss", storeResult=var "Spinner Boss")
    FindChild(gameObject=Self, childName="Mask", storeResult=var "Mask")
    FindChild(gameObject=Self, childName="Speak Region", storeResult=var "Speak Region")
    FindChild(gameObject=Self, childName="Gates", storeResult=var "Gates")
    FindChild(gameObject=Self, childName="Audio Loop Strum", storeResult=var "Audio Loop Strum")
    FindNamedChild(gameObject=Self, storeResult=var "Challenge Region")
    on FINISHED → Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check State
  }

  state Check State {
    PlayerDataBoolTest(boolName="spinnerDefeated", isTrue=→"DEFEATED", isFalse=(none))
    PlayerDataBoolTest(boolName="encounteredSpinner", isTrue=→"ENCOUNTERED", isFalse=→"UNENCOUNTERED")
    on UNENCOUNTERED → State 2
    on ENCOUNTERED → State 1
    on DEFEATED → Defeated
  }

  state Unencountered {
    AudioPlaySimple(gameObject=var "Audio Loop Strum", volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Intro Spinner", volume=1, oneShotClip=<null>)
    CheckYPosition(gameObject=var "Hero", compareTo=9.8, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    ActivateInteractible(Target=var "Speak Region", Activate=false, AllowQueueing=false, UseChildren=false)
    on NEXT → Speak Pause
  }

  state Encountered {
    CheckYPosition(gameObject=var "Hero", compareTo=9.8, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    ActivateInteractible(Target=var "Speak Region", Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateGameObject(gameObject=var "Challenge Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Loop Strum", volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Intro Spinner", volume=1, oneShotClip=<null>)
    on NEXT → Spinner Look
  }

  state Defeated {
    ActivateGameObject(gameObject=var "Mask", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Intro Spinner", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Speak Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="bellShrineBellhart", isTrue=→"DID BELLSHRINE", isFalse=(none))
    on DID BELLSHRINE → Bellshrine Activated
  }

  state Speak Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Speak Ready
  }

  state Speak Ready {
    ActivateInteractible(Target=var "Speak Region", Activate=true, AllowQueueing=false, UseChildren=false)
    on SPOKEN → Spinner Tele Out
  }

  state Spinner Tele Out {
    Tk2dPlayAnimationWithEvents(gameObject=var "Intro Spinner", clipName="Tele Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPlayerDataBool(boolName="encounteredSpinner", value=true)
    PlayAudioEventRandom(audioClips=[widow_insectoid_teleport_1 (sfxstatic_assets_areabell.bundle), widow_insectoid_teleport_2 (sfxstatic_assets_areabell.bundle), widow_insectoid_teleport_3 (sfxstatic_assets_areabell.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Intro Spinner", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Spinner Away
  }

  state Spinner Away {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Intro Spinner", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Battle Start
  }

  state Battle Start {
    SendEventByName(eventTarget=GameObject(var "Spinner Boss"), sendEvent="BATTLE START", delay=0, everyFrame=false)
  }

  state Defeat {
    SetCollider(gameObject=<null>, active=false, resetOnExit=false)
    on SPINNER SEQUENCE END → Sequence End
  }

  state Spinner Look {
    Tk2dPlayAnimation(gameObject=var "Intro Spinner", animLibName="", clipName="Intro Look")
    ActivateGameObject(gameObject=var "Speak Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=var "Intro Spinner", startVolume=1, endVolume=0, time=0.25)
    SendEventToRegister(eventName="TENSION STOP")
    AudioStopV2(gameObject=var "Audio Loop Strum", fadeTime=0, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Intro Spinner", fadeTime=0, cancelOnEarlyExit=false)
    PlayRandomAudioClipTableV3(Table=Widow_attack (sfxdynamic_assets_areabell.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    on FINISHED → Spinner Tele Out
  }

  state Has Needolin? {
    PlayerDataVariableTest(VariableName="hasNeedolin", ExpectedValue=true, IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on FALSE → Start Needolin Quest
    on TRUE → Already Has Needolin
  }

  state Already Has Needolin {
  }

  sequence state Start Needolin Quest {
    Wait(time=0.3, finishEvent=(none), realTime=false)
    BeginQuest(Quest=<null>)
  }

  state Sequence End {
    SetPlayerDataBool(boolName="spinnerDefeated", value=true)
    AwardAchievementProgress(Key="DEFEATED_WIDOW", CurrentValue=(unset), MaxValue=(unset))
    EndQuest(ConsumeCurrency=false, Quest=The Threadspun Town (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    ActivateGameObject(gameObject=var "Mask", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
    on FINISHED → Has Needolin?
  }

  state Set Sequence Completed {
    EndQuest(ConsumeCurrency=false, Quest=The Threadspun Town (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Bellshrine Activated {
    ActivateGameObject(gameObject=var "Gates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state State 1 {
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=true)
    on FINISHED → Encountered
  }

  state State 2 {
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=true)
    on FINISHED → Unencountered
  }

  var CamLock: gameObject = <null>
  var Gates: gameObject = <null>
  var Intro Spinner: gameObject = <null>
  var Mask: gameObject = <null>
  var Speak Region: gameObject = <null>
  var Spinner Boss: gameObject = <null>
  var Challenge Region: gameObject = <null>
  var Audio Loop Tension: gameObject = Black Thread States Thread Only Variant/Normal World/Boss Scene/Audio Loop Tension
  var Self: gameObject = <null>
  var Audio Loop Strum: gameObject = <null>
}
