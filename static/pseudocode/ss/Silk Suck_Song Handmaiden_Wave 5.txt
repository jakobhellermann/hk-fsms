# Silk Suck
# 1 location(s):
#   Song_04 / Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden

fsm Silk Suck {
  start Pause
  on STOP SUCK → Stop  // from any state
  on ZERO HP → Stop  // from any state
  on DISABLE → Stop Suck 2  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on START SUCK → Deparent
  }

  state Stop {
    FloatAdd(floatVariable=var "Tick Timer", add=0, everyFrame=false, perSecond=false)
    RemoveUsingSilk(Amount=1, UsingType=Drain, DidAddTracker=var "Is Sucking")
    ActivateGameObject(gameObject=var "Silk Suck Threads", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Range Out {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Suck Range", storeResult=false, sendEvent="IN", outOfRangeEvent="", everyFrame=true)
    FloatAdd(floatVariable=var "Tick Timer", add=-1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Tick Timer", minValue=-100, maxValue=100, everyFrame=false)
    Wait(time=2, finishEvent=→"OUT", realTime=false)
    ActivateGameObject(gameObject=var "Silk Suck Threads", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on IN → Start Suck
    on OUT → No Silk
  }

  state Range In {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Suck Range", storeResult=false, sendEvent="", outOfRangeEvent="OUT", everyFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt SuckGas")
    PlayParticleEmitterInState(gameObject=var "Pt SuckStrand")
    PlayParticleEmitterInState(gameObject=var "Pt SuckPoint")
    SetPositionToObject2D(gameObject=var "Pt SuckGas", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    SetPositionToObject2D(gameObject=var "Pt SuckStrand", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    GetAngleToTarget2D(gameObject=var "Pt SuckGas", target=var "Spin Point", offsetX=0, offsetY=0, storeAngle=var "Suck Angle", pause=0, everyFrame=true)
    FloatAdd(floatVariable=var "Suck Angle", add=-90, everyFrame=true, perSecond=false)
    SetRotation(gameObject=var "Pt SuckGas", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Suck Angle", space=0, everyFrame=true, lateUpdate=false)
    SetRotation(gameObject=var "Pt SuckStrand", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Suck Angle", space=0, everyFrame=true, lateUpdate=false)
    GetDistance(gameObject=var "Pt SuckGas", target=var "Spin Point", storeResult=var "Suck Distance", everyFrame=true)
    FloatMultiply(floatVariable=var "Suck Distance", multiplyBy=5, everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Pt SuckGas", emissionSpeed=var "Suck Distance", everyFrame=true)
    FloatMultiply(floatVariable=var "Suck Distance", multiplyBy=0.5, everyFrame=true)
    SetParticleEmissionSpeed(gameObject=var "Pt SuckStrand", emissionSpeed=var "Suck Distance", everyFrame=true)
    Wait(time=0.25, finishEvent=→"FLASH", realTime=false)
    FloatAdd(floatVariable=var "Tick Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Tick Timer", float2=var "Tick Time", tolerance=0, equal=→"TICK", lessThan=(none), greaterThan=→"TICK", everyFrame=true)
    SetPositionToObject2D(gameObject=var "Silk Suck Threads", targetObject=var "Silk Suck Threads Origin", xOffset=0, yOffset=0, everyFrame=true)
    GetAngleToTarget2D(gameObject=var "Silk Suck Threads", target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Suck Thread Angle", pause=0, everyFrame=true)
    SetRotation(gameObject=var "Silk Suck Threads", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Suck Thread Angle", space=0, everyFrame=true, lateUpdate=false)
    GetDistanceV2(gameObject=var "Silk Suck Threads", target=var "Hero", targetOffsetX=0, targetOffsetY=0, storeResult=var "Suck Thread Scale", everyFrame=true)
    FloatDivide(floatVariable=var "Suck Thread Scale", divideBy=6, everyFrame=true)
    SetScale(gameObject=var "Silk Suck Threads", vector=(unset), x=var "Suck Thread Scale", y=(unset), z=(unset), everyFrame=true, lateUpdate=false)
    ActivateGameObjectDelay(gameObject=var "Silk Suck Threads", activate=true, resetOnExit=false, delay=0.01)
    on OUT → Stop Suck
    on FLASH → Flash Hero
    on TICK → Tick
  }

  state Flash Hero {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=flashArmoured())
    on FINISHED → Range In
  }

  state Tick {
    FloatSubtract(floatVariable=var "Tick Timer", subtract=var "Tick Time", everyFrame=false, perSecond=false)
    PlayerdataIntCompare(playerdataInt="silk", compareTo=0, equal=→"NO SILK", lessThan=→"NO SILK", greaterThan=(none))
    PlayAudioEvent(audioClip=hornet_silk_loss_like_from_handmaiden (sfxstatic_assets_silkshared.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    TakeSilk(target=var "Hero", amount=1)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SUCK TICK", delay=0, everyFrame=false)
    SetPositionToObject2D(gameObject=var "Burst Anim", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Burst Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Range In
    on NO SILK → No Silk
  }

  state No Silk {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="NO SILK", delay=0, everyFrame=false)
    on FINISHED → Stop
  }

  state Deparent {
    BoolTest(boolVariable=var "Did Deparent", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetParent(gameObject=var "Pt SuckGas", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Pt SuckStrand", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Silk Suck Threads", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Burst Anim", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetBoolValue(boolVariable=var "Did Deparent", boolValue=false, everyFrame=false)
    on FINISHED → Range Out
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Start Suck {
    AddUsingSilk(Amount=1, UsingType=Drain, DidAddTracker=var "Is Sucking")
    on FINISHED → Range In
  }

  state Stop Suck {
    RemoveUsingSilk(Amount=1, UsingType=Drain, DidAddTracker=var "Is Sucking")
    on FINISHED → Range Out
  }

  state Stop Suck 2 {
    RemoveUsingSilk(Amount=1, UsingType=Drain, DidAddTracker=var "Is Sucking")
    on FINISHED → 
  }

  var Suck Angle: float = 0
  var Suck Distance: float = 0
  var Tick Time: float = 1.1
  var Tick Timer: float = 0
  var Suck Thread Angle: float = 0
  var Suck Thread Scale: float = 0
  var Did Deparent: bool = false
  var Is Sucking: bool = false
  var Pt SuckGas: gameObject = Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden/Pt SuckGas
  var Pt SuckPoint: gameObject = Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden/Pt SuckPoint
  var Pt SuckStrand: gameObject = Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden/Pt SuckStrand
  var Spin Point: gameObject = Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden/Spin Point
  var Silk Suck Threads: gameObject = Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden/Silk Suck Threads
  var Silk Suck Threads Origin: gameObject = Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden/Silk Suck Threads Origin
  var Burst Anim: gameObject = Black Thread States/Normal World/Scene States/Battle Scene/Wave 5/Song Handmaiden/Burst Anim
}
