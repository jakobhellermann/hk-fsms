# Control
# 1 location(s):
#   Organ_01 / Boss Scene

fsm Control {
  start Pause

  state Pause {
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Phantom", storeResult=var "Phantom")
    FindChild(gameObject=Self, childName="Organ Phantom", storeResult=var "Organ Phantom")
    FindChild(gameObject=Self, childName="CamLock Boss", storeResult=var "CamLock Boss")
    FindChild(gameObject=Self, childName="Gates", storeResult=var "Gates")
    FindChild(gameObject=Self, childName="Fog Column BG", storeResult=var "Fog Column BG")
    FindChild(gameObject=Self, childName="Fog Column FG", storeResult=var "Fog Column FG")
    FindNamedChild(gameObject=Self, storeResult=var "Fog Column Antic")
    FindNamedChild(gameObject=Self, storeResult=var "Fog Damager")
    FindChild(gameObject=Self, childName="Final Fog", storeResult=var "Final Fog")
    PlayerDataBoolTest(boolName="defeatedPhantom", isTrue=→"DEFEATED", isFalse=(none))
    ActivateGameObject(gameObject=var "Thread Memory", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendTrigger2DEvent(eventTarget=Self, trigger=OnTriggerEnter2D, collideTag="", collideLayer=(unset), sendEvent=→"ENTER", storeCollider=var "")
    on DEFEATED → Defeated
    on ENTER → Enter
  }

  state Defeated {
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Final Fog", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Organ Phantom", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Phantom", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Enter {
    SendEventToRegister(eventName="EXHAUST END")
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=2)
    on FINISHED → Organ Hit
  }

  state Organ Hit {
    Tk2dPlayAnimationWithEvents(gameObject=var "Organ Phantom", clipName="Organ Hit", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0.1)
    on FINISHED → Organ Note
  }

  state Organ Note {
    Tk2dWatchAnimationEvents(gameObject=var "Organ Phantom", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=true, everyFrame=false)
    Wait(time=0.7, finishEvent=→"NEXT", realTime=false)
    ApplyMusicCue(musicCue=Phantom (audiocuesdynamic_assets_areaorgan.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.1)
    on NEXT → BG Fog
  }

  state FG Column {
    StopParticleEmittersInChildren(gameObject=var "Fog Column BG")
    ActivateGameObject(gameObject=var "Organ Phantom", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fog Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Fog Column FG", resetTimeIfPlaying=false, stopOnStateExit=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=phantom_organ_air_blow_at_start (sfxstatic_assets_areaorgan.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Start Battle
  }

  state Start Battle {
    SetCollider(gameObject=var "Fog Damager", active=false, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Phantom"), sendEvent="PHANTOM START", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=false, bossTitle="PHANTOM")
    on BATTLE END → End
  }

  sequence state End {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Boss", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state BG Fog {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Fog Column BG", resetTimeIfPlaying=false, stopOnStateExit=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Fog Column BG", audioClip=phantom_organ_air_blow_at_start (sfxstatic_assets_areaorgan.bundle), pitchMin=0.9, pitchMax=0.9, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → FG Antic
  }

  state FG Antic {
    PlayParticleEmitterInState(gameObject=var "Fog Column Antic")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → FG Column
  }

  state End Note {
    PlayAudioEvent(audioClip=phantom_ending_note (sfxstatic_assets_areaorgan.bundle), pitchMin=1, pitchMax=1, volume=0.8, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Organ Hit
  }

  var CamLock Boss: gameObject = <null>
  var Fog Column BG: gameObject = <null>
  var Fog Column FG: gameObject = <null>
  var Gates: gameObject = <null>
  var Organ Phantom: gameObject = <null>
  var Phantom: gameObject = <null>
  var Fog Column Antic: gameObject = <null>
  var Fog Damager: gameObject = <null>
  var Thread Memory: gameObject = Boss Scene/thread_memory_lace
  var Final Fog: gameObject = <null>
}
