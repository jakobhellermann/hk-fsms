# volume_curve
# 3 location(s):
#   Chapel_Wanderer / Audio Loop DustFall
#   Slab_15 / Breakable Wall Waterways/Camera Locks/Audio Loop DustFall
#   Slab_18 / Crumbly Roof/Audio Loop DustFall

fsm volume_curve {
  start Up

  state Up {
    RandomFloat(min=0.6, max=0.8, storeResult=var "Target")
    RandomFloat(min=1, max=2, storeResult=var "Time")
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=var "Target", time=var "Time")
    Wait(time=var "Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pause
  }

  state Down {
    RandomFloat(min=0.1, max=0.4, storeResult=var "Target")
    RandomFloat(min=1, max=2, storeResult=var "Time")
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=var "Target", time=var "Time")
    Wait(time=var "Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 2
  }

  state Pause {
    WaitRandom(timeMin=0.35, timeMax=0.75, finishEvent=(none), realTime=false)
    on FINISHED → Down
  }

  state State 2 {
    WaitRandom(timeMin=0.35, timeMax=0.75, finishEvent=(none), realTime=false)
    on FINISHED → Up
  }

  var Target: float = 0
  var Time: float = 0
}
