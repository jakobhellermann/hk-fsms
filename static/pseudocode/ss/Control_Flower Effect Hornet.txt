# Control
# 1 location(s):
#   Song_Tower_Destroyed / Flower Effect Hornet

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Walk")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Land")
    on FINISHED → Wait for Hero In Position
  }

  state Out {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    PlayParticleEmitter(gameObject=var "Pt Land", emit=0, resetIfPlaying=false)  // disabled
    on ENTER → In
  }

  state In {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    PlayParticleEmitterInState(gameObject=var "Pt Walk")
    PlayParticleEmitter(gameObject=var "Pt Land", emit=0, resetIfPlaying=false)
    on EXIT → Out
  }

  state Wait for Hero In Position {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on ENTER → Waiting In
    on FINISHED → Out
  }

  state Waiting In {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    PlayParticleEmitterInState(gameObject=var "Pt Walk")
    on EXIT → Wait for Hero In Position
  }

  var Pt Walk: gameObject = <null>
  var Pt Land: gameObject = <null>
}
