# Detect Hit
# 1 location(s):
#   Bone_East_LavaChallenge / explode_floor_battle/explode rock

fsm Detect Hit {
  start Idle

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"HIT", storeCollider=var "Collider")
    on HIT → Check
  }

  state Check {
    CheckTrackTriggerCount(target=explode_floor_battle/explode rock/Hero Detector, count=0, test=LessThanOrEqual, everyFrame=false, successEvent=→"FINISHED")
    GetPosition(gameObject=var "Collider", vector=(unset), x=var "Atk Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Atk Pos", lowerValue=var "Atk Min X", upperValue=var "Atk Max X", boolVariable=(unset), trueEvent=(none), falseEvent=→"FINISHED", everyFrame=false)
    GetPosition(gameObject=var "Collider", vector=(unset), x=(unset), y=var "Atk Pos", z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Atk Pos", lowerValue=var "Atk Min Y", upperValue=var "Atk Max Y", boolVariable=(unset), trueEvent=→"VALID", falseEvent=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on VALID → Send
  }

  state Send {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="STRIKE", delay=0, everyFrame=false)
  }

  var Atk Max X: float = 9999
  var Atk Max Y: float = 9999
  var Atk Min X: float = 0
  var Atk Min Y: float = 0

  var Atk Pos: float = 0
  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
}
