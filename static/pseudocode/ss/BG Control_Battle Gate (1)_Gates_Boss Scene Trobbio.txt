# BG Control
# 1 location(s):
#   Library_13 / Grand Stage Scene/Boss Scene Trobbio/Gates/Battle Gate (1)

fsm BG Control {
  start Opened
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Opened")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    FindNamedChild(gameObject=Self, storeResult=var "Wall Collider")
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=gate_slam (sharedeffects_assets_all.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.5, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="BG Close 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Close 2
  }

  state Close 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Close 2")
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BG OPEN → Open
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=gate_open (sharedeffects_assets_all.bundle), pitchMin=1, pitchMax=1, volume=0.85, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Plat Stream Dust", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="BG Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Finish Open
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Closed")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BG OPEN → Open
    on FINISHED → Double Close
  }

  state Double Close {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Closed")
    on BG OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BG Opened")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Finish Open {
    SetMeshRenderer(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  var Start Closed: bool = false

  var Plat Dust: gameObject = Grand Stage Scene/Boss Scene Trobbio/Gates/Battle Gate (1)/Plat Dust
  var Plat Stream Dust: gameObject = Grand Stage Scene/Boss Scene Trobbio/Gates/Battle Gate (1)/Plat Stream Dust
  var Raise Dust: gameObject = <null>
  var Self: gameObject = <null>
  var Slam Effect: gameObject = Grand Stage Scene/Boss Scene Trobbio/Gates/Battle Gate (1)/Close Effect
  var Wall Collider: gameObject = Grand Stage Scene/Boss Scene Trobbio/Gates/Battle Gate (1)/Wall Collider
}
