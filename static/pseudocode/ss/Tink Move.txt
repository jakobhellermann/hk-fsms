  start Idle

  state Idle {
    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
    on TINK UP → Tink Up
    on TINK DOWN → Tink Down
  }

  state Tink Left {
    SetVelocity2d(gameObject=Self, vector=(-5, 2), x=(unset), y=(unset), everyFrame=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=20, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Idle
  }

  state Tink Right {
    SetVelocity2d(gameObject=Self, vector=(5, 2), x=(unset), y=(unset), everyFrame=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-20, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Idle
  }

  state Tink Up {
    SetVelocity2d(gameObject=Self, vector=(0, 9), x=(unset), y=(unset), everyFrame=false)
    RandomFloat(min=-5, max=5, storeResult=var "Rotate")
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Idle
  }

  state Tink Down {
    SetVelocity2d(gameObject=Self, vector=(0, 6), x=(unset), y=(unset), everyFrame=false)
    RandomFloat(min=-5, max=5, storeResult=var "Rotate")
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Idle
  }

  var Rotate: float = 0
}
