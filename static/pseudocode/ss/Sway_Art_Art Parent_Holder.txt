  start Pause
  on LAND → Centre  // from any state
  on STOP MOVE → Stop Move  // from any state

  state Init {
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=0, space=1, everyFrame=false, lateUpdate=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Sway L
    on R → Sway R
  }

  state Sway L {
    GetAngleToTarget2D(gameObject=var "chain", target=var "Chain Top", offsetX=0, offsetY=0, storeAngle=var "Chain Angle", pause=0, everyFrame=true)

    RandomFloat(min=-0.09, max=-0.125, storeResult=var "Sway X")
    RandomFloat(min=2.25, max=3, storeResult=var "Time")
    EaseFloat(fromValue=var "Start X", toValue=var "Sway X", floatVariable=var "Tween X", time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Tween X", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)
    on FINISHED → Sway R
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink U
    on TINK DOWN → Tink D
  }

  state Sway R {
    GetAngleToTarget2D(gameObject=var "chain", target=var "Chain Top", offsetX=0, offsetY=0, storeAngle=var "Chain Angle", pause=0, everyFrame=true)
    FloatAdd(floatVariable=var "Chain Angle", add=-90, everyFrame=true, perSecond=false)
    SetRotation(gameObject=var "chain", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Chain Angle", space=0, everyFrame=true, lateUpdate=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=(unset), z=(unset), space=1, everyFrame=false)
    RandomFloat(min=0.09, max=0.125, storeResult=var "Sway X")
    RandomFloat(min=2.25, max=3, storeResult=var "Time")
    EaseFloat(fromValue=var "Start X", toValue=var "Sway X", floatVariable=var "Tween X", time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Tween X", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)
    on FINISHED → Sway L
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink U
    on TINK DOWN → Tink D
  }

  state Centre {
    AudioStop(gameObject=var "Chain Audio Player", fadeTime=0)
    BoolTest(boolVariable=var "Dont Move", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetAngleToTarget2D(gameObject=var "chain", target=var "Chain Top", offsetX=0, offsetY=0, storeAngle=var "Chain Angle", pause=0, everyFrame=true)
