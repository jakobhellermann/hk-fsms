# Threads
# 1 location(s):
#   Ward_09 / Black Thread Enemies/Enemy Folder/Song Pilgrim 02 (1)/Attack Threads

fsm Threads {
  start Init
  on DISABLE → Disable  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Hero Damager", storeResult=var "Damager")
    on FINISHED → Inert
  }

  state Inert {
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on START ATTACK → Start Attack
  }

  state Start Attack {
    SetGravity2dScale(gameObject=var "Parent", gravityScale=0)
    PlayAudioEvent(audioClip=song_pilgrim_02_antic (sfxstatic_assets_pilgrimslower.bundle), pitchMin=0.9, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    PlayAudioEvent(audioClip=scholar_antic (sfxstatic_assets_arealibrary.bundle), pitchMin=0.9, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Threads Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=var "Parent", deceleration=0.85, brakeOnExit=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=var "Grounded", GroundedEvent=(none), NotGroundedEvent=(none), EveryFrame=false)
    SetVelocity2dBool(gameObject=Self, activeBool=var "Grounded", vector=(unset), x=(unset), y=5, everyFrame=false)
    on FINISHED → Attack Loop
  }

  state Attack Loop {
    PlayAudioEvent(audioClip=song_pilgrim_02_arms_attack (sfxstatic_assets_pilgrimslower.bundle), pitchMin=0.9, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    SetDamageHero(Target=var "Parent", Enabled=false)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Threads Loop", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=var "Attack Loop Time", finishEvent=→"END ATTACK", realTime=false)
    SetVelocity2d(gameObject=var "Parent", vector=(0, 5), x=(unset), y=(unset), everyFrame=false)
    DecelerateV2(gameObject=var "Parent", deceleration=0.85, brakeOnExit=false)
    on END ATTACK → End Attack
    on MULTI HIT CONNECT → Multihitting
  }

  state End Attack {
    SendEventByNameV2(eventTarget=GameObject(var "Parent"), sendEvent="END ATTACK", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Threads Disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHero(Target=var "Parent", Enabled=true)
    SetGravity2dScale(gameObject=var "Parent", gravityScale=1)
    on FINISHED → Inert
  }

  state Disable {
    AudioStop(gameObject=var "Audio Player", fadeTime=0)
  }

  state Multihitting {
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Multihit
  }

  state End Multihit {
    SendEventByNameV2(eventTarget=GameObject(var "Parent"), sendEvent="END ATTACK", delay=0, everyFrame=false)
    SetGravity2dScale(gameObject=var "Parent", gravityScale=1)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Threads Multihit End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHero(Target=var "Parent", Enabled=true)
    on FINISHED → Inert
  }

  var Attack Loop Time: float = 0.5
  var Parent: gameObject = Black Thread Enemies/Enemy Folder/Song Pilgrim 02 (1)

  var Grounded: bool = false
  var Audio Player: gameObject = <null>
  var Damager: gameObject = <null>
}
