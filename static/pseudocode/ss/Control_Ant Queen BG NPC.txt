# Control
# 1 location(s):
#   Ant_Queen / Memory Group/here/Ant Queen BG NPC

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Deep_Memory_appear")
    ActivateGameObject(gameObject=var "Deep_Memory_appear", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=var "Deep_Memory_appear", childName="Appear Effects", storeResult=var "Deep Memory Appear Effects")
    ActivateGameObject(gameObject=var "Deep Memory Appear Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=var "Parent", storeResult=var "Deep memory zone effect")
    ActivateGameObject(gameObject=var "Deep memory zone effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=var "Parent", childName="thread_memory", storeResult=var "thread_memory")
    SetFsmBool(gameObject=var "thread_memory", fsmName="FSM", variableName="Allow Memory Enter", setValue=false, everyFrame=false)
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="door_wakeOnGround", equalEvent=→"MEMORY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Entry Gate", compareTo="Death Respawn Marker", equalEvent=→"MEMORY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on MEMORY → Memory Return
    on FINISHED → Did Deep Melody Convo?
  }

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=2, MaxReactDelay=2, None=(none), ActiveInner=(none), ActiveOuter=→"SING", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT → Has Deep Melody?
    on SING → Fade Breath
  }

  state Has Deep Melody? {
    PlayerDataBoolTest(boolName="hasNeedolinMemoryPowerup", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → Melody
    on FALSE → Met?
  }

  state Did Deep Melody Convo? {
    PlayerDataBoolTest(boolName="antQueenNPC_deepMelodyConvo", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → Idle Active
    on FALSE → Idle
  }

  state Met? {
    PlayerDataBoolTest(boolName="metAntQueenNPC", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → Repeat
    on FALSE → Meet
  }

  state End Dialogue {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    BoolTest(boolVariable=var "Activate Memory", isTrue=→"MEMORY", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on MEMORY → Activate Memory
  }

  state Repeat {
    RunDialogue(Sheet="Wilds", Key="ANT_QUEEN_BG_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TextAlignmentOptions(0), OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Meet {
    SetPlayerDataBool(boolName="metAntQueenNPC", value=true)
    RunDialogue(Sheet="Wilds", Key="ANT_QUEEN_BG_MEET", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Melody Repeat {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SetBoolValue(boolVariable=var "Activate Memory", boolValue=true, everyFrame=false)
    RunDialogue(Sheet="Wilds", Key="ANT_QUEEN_HASMELODY_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TextAlignmentOptions(0), OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Melody {
    SetPlayerDataBool(boolName="antQueenNPC_deepMelodyConvo", value=true)
    PlayerDataBoolTest(boolName="metAntQueenNPC", isTrue=→"CANCEL", isFalse=(none))
    SetPlayerDataBool(boolName="metAntQueenNPC", value=true)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SetBoolValue(boolVariable=var "Activate Memory", boolValue=true, everyFrame=false)
    GetPlayerDataVariable(VariableName="HasWhiteFlower", StoreValue=var "Has White FLower")
    ConvertBoolToString(boolVariable=var "Has White FLower", stringVariable=var "Convo Key", falseString="ANT_QUEEN_HASMELODY", trueString="ANT_QUEEN_HASMELODY_FLOWER_OWNED", everyFrame=false)
    RunDialogue(Sheet="Wilds", Key=var "Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CANCEL → Melody Repeat
    on CONVO_END → End Dialogue
  }

  sequence state Activate Memory {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Deep_Memory_appear", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Deep Memory Appear Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Deep memory zone effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "thread_memory", fsmName="FSM", variableName="Allow Memory Enter", setValue=true, everyFrame=false)
    on FINISHED → Wait Sing
  }

  state Memory Return {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateGameObject(gameObject=var "Deep_Memory_appear", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=var "Deep_Memory_appear", stateName="Deep_Memory_Appear_Clover", layer=0, normalizedTime=1, everyFrame=false)
    StopParticleEmittersInChildrenV2(Target=var "Deep_Memory_appear", StopBehaviour=0)
    ActivateGameObject(gameObject=var "Deep memory zone effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "thread_memory", fsmName="FSM", variableName="Allow Memory Enter", setValue=true, everyFrame=false)
  }

  state Idle Active {
    ActivateGameObject(gameObject=var "Deep_Memory_appear", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Deep memory zone effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=var "Deep_Memory_appear", stateName="Deep_Memory_Appear_Clover", layer=0, normalizedTime=1, everyFrame=false)
    StopParticleEmittersInChildrenV2(Target=var "Deep_Memory_appear", StopBehaviour=0)
    SetFsmBool(gameObject=var "thread_memory", fsmName="FSM", variableName="Allow Memory Enter", setValue=true, everyFrame=false)
    on INTERACT → Melody Repeat
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=var "Sing Table", noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    on CANCEL → Sing End
  }

  state Sing 2 {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=var "Sing Table", noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=true, altThreadSpawnPoint=<null>)
    on CANCEL → Wait Sing
    on NEEDOLIN LOCK → Wait Collapse
    on THREAD MEMORY COLLAPSE → Fade Sing
  }

  state Wait Sing {
    AudioStop(gameObject=var "Audio Loop Voice", fadeTime=0)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.5, MaxReactDelay=1.5, None=(none), ActiveInner=(none), ActiveOuter=→"SING", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SING → Sing 2
  }

  state Wait Collapse {
    on THREAD MEMORY COLLAPSE → Fade Sing
  }

  state Fade Sing {
    AudioStop(gameObject=var "Audio Loop Voice", fadeTime=2)
  }

  sequence state Sing End {
    FadeAudioV2(Target=var "Audio Loop Voice", StartVolume=1, EndVolume=0, Time=0.5, SetOnExit=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    SetRandomAudioClipFromTable(Table=Karmelita_Old_breathing_loop (sfxdynamic_assets_areaantareamemory.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  state Fade Breath {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0.25, cancelOnEarlyExit=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sing
  }

  var Activate Memory: bool = false
  var Has White FLower: bool = false
  var Entry Gate: string = ""
  var Convo Key: string = ""
  var Deep Memory Appear Effects: gameObject = <null>
  var Deep memory zone effect: gameObject = <null>
  var Deep_Memory_appear: gameObject = <null>
  var Parent: gameObject = <null>
  var thread_memory: gameObject = <null>
  var Audio Loop Voice: gameObject = Memory Group/here/Ant Queen BG NPC/Audio Loop Voice
  var Sing Table: object = Karmelita_Old_sing (sfxdynamic_assets_areaantareamemory.bundle)
}
