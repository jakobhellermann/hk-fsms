fsm Collect {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Get Shine")
    RandomInt(min=10, max=20, storeResult=var "Rosary Stash", inclusiveMax=true, noRepeat=false)  // disabled
    SetIntValue(intVariable=var "Rosary Stash", intValue=5, everyFrame=false)
    RandomInt(min=30, max=35, storeResult=var "Rosary Stash Goal", inclusiveMax=true, noRepeat=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Geo", collideLayer=Particle(18), sendEvent=→"COLLECT", storeCollider=var "Rosary")
    on COLLECT → Collect
  }

  state Collect {
    ActivateGameObject(gameObject=var "Get Shine", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Get Shine", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Rosary", delivery=SendMessage, options=1, functionCall=ForceRecycle())
    IntAdd(intVariable=var "Rosary Stash", add=1, everyFrame=false)
    IntClamp(intVariable=var "Rosary Stash", minValue=0, maxValue=50, everyFrame=false)
    SetGeoDrop(target=var "Parent", smallGeo=var "Rosary Stash", mediumGeo=(unset), largeGeo=(unset))
    PlayAudioEventRandom(audioClips=[rosary_collect_1 (sfxstatic_assets_currency.bundle), rosary_collect_2 (sfxstatic_assets_currency.bundle), rosary_collect_3 (sfxstatic_assets_currency.bundle), rosary_collect_4 (sfxstatic_assets_currency.bundle), rosary_collect_5 (sfxstatic_assets_currency.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Ready to Leave?
  }

  state Ready to Leave? {
    IntCompare(integer1=var "Rosary Stash", integer2=var "Rosary Stash Goal", equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetFsmBool(gameObject=var "Parent", fsmName="Control", variableName="Ready To Go", setValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  var Rosary Stash: int = 0
  var Rosary Stash Goal: int = 0
  var Parent: gameObject = <null>
  var Get Shine: gameObject = <null>
  var Rosary: gameObject = <null>
}
