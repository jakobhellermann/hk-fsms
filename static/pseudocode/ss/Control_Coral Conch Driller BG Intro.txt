# Control
# 4 location(s):
#   Coral_27 / Battle Scene/Roar Drillers/Coral Conch Driller BG Intro
#   Coral_27 / Battle Scene/Roar Drillers/Coral Conch Driller BG Intro (1)
#   Coral_27 / Battle Scene/Roar Drillers/Coral Conch Driller BG Intro (3)
#   Coral_27 / Battle Scene/Roar Drillers/Coral Conch Driller BG Intro (2)

fsm Control {
  start Dormant

  state Dormant {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetMeshRenderer(gameObject=Self, active=false)
    on DRILL INTRO ROAR → Appear Pause
  }

  state Appear Pause {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    FloatTestToBool(float1=var "Rotation", float2=10, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Going Up", everyFrame=false)
    WaitRandom(timeMin=0.45, timeMax=0.65, finishEvent=→"FINISHED", realTime=false)
    ConvertBoolToFloat(boolVariable=var "Going Up", floatVariable=var "Leave Pos", falseValue=8, trueValue=23, everyFrame=false)
    on FINISHED → Drill In
  }

  state Drill In {
    SetMeshRenderer(gameObject=Self, active=true)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="EMERGE", delay=0, everyFrame=false)
    RandomFloat(min=15, max=17, storeResult=var "Arrive Pos")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    AnimateYPositionTo(GameObject=Self, ToValue=var "Arrive Pos", localSpace=true, time=0.3, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Drill End
  }

  state Drill End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Drill Attack End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Leave Pause
  }

  state Leave Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=false, pauseBetweenTurns=0.5)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    RandomFloat(min=0, max=1, storeResult=var "Direction")
    SetVelocityAsAngle(gameObject=Self, angle=var "Direction", speed=0.5, everyFrame=false)
    on FINISHED → Leave Antic
  }

  state Leave Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Drill Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    RotateTo(gameObject=Self, targetAngle=var "Rotation", speed=1500)
    on FINISHED → Leave
  }

  state Leave {
    AnimateYPositionTo(GameObject=Self, ToValue=var "Leave Pos", localSpace=true, time=0.5, speed=(unset), delay=(unset), easeType=3, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    Tk2dStopAnimation(gameObject=Self)
    SetMeshRenderer(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="IMPACT", delay=0, everyFrame=false)
  }

  var Arrive Pos: float = 0
  var Rotation: float = 0
  var Direction: float = 0
  var Leave Pos: float = 0
  var Going Up: bool = false
  var Parent: gameObject = <null>
}
