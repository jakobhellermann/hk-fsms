fsm Water Skim Effects {
  start Init
  on EFFECT STOP → Effect Stop  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    RayCast2dV2(fromGameObject=Self, fromPosition=(0, 2), direction=(0, -1), space=0, distance=3.5, minDepth=(unset), maxDepth=(unset), hitEvent=→"SPARK", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Soft Terrain(25)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on SPARK → Spark
  }

  state Spark {
    CheckSceneName(sceneName="Greymoor_05", equalEvent=→"EFFECT STOP", notEqualEvent=(none))
    RayCast2dV2(fromGameObject=Self, fromPosition=(0, 2), direction=(0, -1), space=0, distance=3.5, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"END", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Hit Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Soft Terrain(25)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    SetPosition2D(GameObject=var "Pt Skim", Vector=var "Hit Point", X=0, Y=0, Space=0, EveryFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Skim")
    AudioPlayInState(gameObject=var "Audio Loop WaterSkim", volume=1)
    on END → Idle
  }

  state Effect Stop {
  }

  var Hit Point: vector2 = (0, 0)
  var Pt Skim: gameObject = Vampire Gnat Sickle/Pt Skim
  var Audio Loop WaterSkim: gameObject = Vampire Gnat Sickle/Audio Loop WaterSkim
}
