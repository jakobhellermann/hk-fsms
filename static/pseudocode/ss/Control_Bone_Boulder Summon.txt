# Control
# 1 location(s):
#   Chapel_Wanderer / Pilgrim Trap Boulder/Bone_Boulder Summon

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="boulder", storeResult=var "boulder")
    FindChild(gameObject=Self, childName="boulder/Sprites", storeResult=var "Sprites")
    FindChild(gameObject=Self, childName="boulder/break_effects", storeResult=var "break_effects")
    GetRandomChild(gameObject=var "Sprites", storeResult=var "Sprite Crt")
    ActivateAllChildren(gameObject=var "Sprites", activate=false)
    ActivateGameObject(gameObject=var "Sprite Crt", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Drop Antic
  }

  state Drop Antic {
    AudioStop(gameObject=Self, fadeTime=0)
    PlayAudioEvent(audioClip=bone_boulder_shake_n_break (sfxstatic_assets_areabone.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    WaitRandom(timeMin=0.75, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=false)
    AnimatePositionBy(gameObject=var "boulder", shiftBy=(0, -0.4, 0), time=0.35, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    ObjectJitterLocal(gameObject=var "Sprite Crt", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=0)
    SpawnObjectFromGlobalPool(gameObject=Radial Burst Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Sprite Crt", position=(0, -1, 0.001), rotation=(unset), storeObject=var "")
    on FINISHED → Drop
  }

  state Drop {
    AccelerateToY(gameObject=Self, accelerationFactor=4, targetSpeed=-40)
    ActivateGameObjectDelay(gameObject=var "Terrain Detector", activate=true, resetOnExit=false, delay=0.2)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"LAND", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    SetPolygonCollider(gameObject=var "Sprite Crt", active=true, resetOnExit=false)
    SetDamageHeroAmount(target=var "Sprite Crt", damageDealt=2)
    SetDamageEnemyAmount(target=var "Sprite Crt", damageDealt=80)
    on LAND → Break
  }

  state Break {
    PlayAudioEvent(audioClip=bone_boulder_ground_impact (sfxstatic_assets_areabone.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite Crt", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "break_effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Impact Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait Break Effects
  }

  state Recycle {
    SetPolygonCollider(gameObject=var "Sprite Crt", active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "break_effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHeroAmount(target=var "Sprite Crt", damageDealt=0)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait Break Effects {
    GameObjectIsNull(gameObject=var "break_effects", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    WaitParticleSystemsInChildren(target=var "break_effects", stop=false, stopLoops=true, withChildren=true, timeOut=12)
    on FINISHED → Recycle
  }

  var Impact Damager: gameObject = Pilgrim Trap Boulder/Bone_Boulder Summon/Impact Damager
  var Pt Antic: gameObject = Pilgrim Trap Boulder/Bone_Boulder Summon/Pt Antic
  var Sprite Crt: gameObject = <null>
  var Sprites: gameObject = <null>
  var Terrain Detector: gameObject = Pilgrim Trap Boulder/Bone_Boulder Summon/Terrain Detector
  var boulder: gameObject = <null>
  var break_effects: gameObject = <null>
}
