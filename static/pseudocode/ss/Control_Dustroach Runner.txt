# Control
# 3 location(s):
#   Dust_03 / Battle Control/Battle Scene/Wave 7/Dustroach Runner
#   Dust_03 / Battle Control/Battle Scene/Wave 3/Dustroach Runner
#   Dust_03 / Battle Control/Battle Scene/Wave 5/Dustroach Runner

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on BATTLE START → Dir Check
  }

  state Dir Check {
    CheckXPosition(gameObject=var "Hero", compareTo=var "Mid X", compareToOffset=0, tolerance=0, equal=→"R", equalBool=(unset), lessThan=→"R", lessThanBool=(unset), greaterThan=→"L", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    on L → Run L
    on R → Run R
  }

  state Run L {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Hero X", add=30, everyFrame=false, perSecond=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Translate", floatValue=-34, everyFrame=false)
    on FINISHED → Run
  }

  state Run R {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Hero X", add=-30, everyFrame=false, perSecond=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Translate", floatValue=34, everyFrame=false)
    on FINISHED → Run
  }

  state Run {
    Translate(gameObject=Self, vector=(unset), x=var "Translate", y=(unset), z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    SetMeshRenderer(gameObject=Self, active=true)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  var Mid X: float = 71.7

  var Hero X: float = 0
  var Translate: float = 0
}
