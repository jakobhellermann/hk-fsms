# Rubble Timer
# 1 location(s):
#   Cradle_03 / Boss Scene/Rubble Fields

fsm Rubble Timer {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    SetBoolValue(boolVariable=var "Rubble Active", boolValue=false, everyFrame=false)
    on ATTACK → Rubbling
  }

  state Rubbling {
    SetBoolValue(boolVariable=var "Rubble Active", boolValue=true, everyFrame=false)
    Wait(time=9, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Rubble Active: bool = false
}
