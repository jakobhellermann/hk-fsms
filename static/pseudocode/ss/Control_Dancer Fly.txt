# Control
# 1 location(s):
#   Slab_16 / Event Control/Battle Cloakless Start Scene/Dancer Fly

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    on FINISHED → Set Taunt Type
  }

  state Dance To L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dance")
    SetVelocity2d(gameObject=Self, vector=(unset), x=-0.8, y=(unset), everyFrame=false)
    CheckXPosition(gameObject=Self, compareTo=24.75, compareToOffset=0, tolerance=0, equal=→"FINISHED", equalBool=(unset), lessThan=→"FINISHED", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=SF_dress_dance (sfxdynamic_assets_areaslab.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Set Pokes
  }

  state Set Pokes {
    SetBoolValue(boolVariable=var "Do Laughter", boolValue=true, everyFrame=false)
    RandomInt(min=3, max=4, storeResult=var "Pokes", inclusiveMax=true, noRepeat=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=var "Taunt Table", pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Poke
  }

  state Poke {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dance Poke", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Audio Loop Voice", audioClips=[hornet_wanderer_sword_1 (sfxstatic_assets_shared.bundle), hornet_wanderer_sword_2 (sfxstatic_assets_shared.bundle), hornet_wanderer_sword_3 (sfxstatic_assets_shared.bundle), hornet_wanderer_sword_4 (sfxstatic_assets_shared.bundle)], weights=[1, 1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=0.45, delay=0.08, storePlayer=<null>)
    on FINISHED → Laughter?
  }

  state Repoke? {
    IntAdd(intVariable=var "Pokes", add=-1, everyFrame=false)
    IntCompare(integer1=var "Pokes", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"REPEAT", everyFrame=false)
    on REPEAT → Poke
    on FINISHED → Turn
  }

  state Turn {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dance Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Turn Poke
  }

  state Turn Poke {
    SendRandomEvent(events=[→"FINISHED", (none)], weights=[0.5, 0.5], delay=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dance Poke", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Move
  }

  state Move {
    BoolFlip(boolVariable=var "Going R")
    BoolTest(boolVariable=var "Going R", isTrue=→"R", isFalse=→"L", everyFrame=false)
    on R → Dance To R
    on L → Dance To L
  }

  state Dance To R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dance")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0.8, y=(unset), everyFrame=false)
    CheckXPosition(gameObject=Self, compareTo=26.25, compareToOffset=0, tolerance=0, equal=→"FINISHED", equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"FINISHED", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=SF_dress_dance (sfxdynamic_assets_areaslab.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Set Pokes
  }

  state Laughter? {
    BoolTest(boolVariable=var "Do Laughter", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Do Laughter", boolValue=false, everyFrame=false)
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.5, 0.5], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    SendEventToRegister(eventName="CROWD CHUCKLE")
    PlayRandomAudioClipTable(Table=SF_crowd_laughs (sfxdynamic_assets_areaslab.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SetBoolValue(boolVariable=var "Do Laughter", boolValue=false, everyFrame=false)
    on FINISHED → Repoke?
  }

  state Set Taunt Type {
    PlayerDataBoolTest(boolName="slabCaptor_heardChallengeRings", isTrue=→"POSHANKA", isFalse=(none))
    PlayerDataBoolTest(boolName="slabCaptor_heardChallenge", isTrue=→"GARAMA", isFalse=(none))
    on GARAMA → Garama
    on POSHANKA → Poshanka
    on FINISHED → Standard
  }

  state Standard {
    SetObjectValue(objectVariable=var "Taunt Table", objectValue=SF_dress_poke (sfxdynamic_assets_areaslab.bundle), everyFrame=false)
    on FINISHED → Dance To L
  }

  state Garama {
    SetObjectValue(objectVariable=var "Taunt Table", objectValue=SF_dress_poke_garama (sfxdynamic_assets_areaslab.bundle), everyFrame=false)
    on FINISHED → Dance To L
  }

  state Poshanka {
    SetObjectValue(objectVariable=var "Taunt Table", objectValue=SF_dress_poke_poshanka (sfxdynamic_assets_areaslab.bundle), everyFrame=false)
    on FINISHED → Dance To L
  }

  var Pokes: int = 0
  var Do Laughter: bool = false
  var Going R: bool = false
  var Audio Loop Voice: gameObject = <null>
  var Taunt Table: object = <null>
}
