# Mantle
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet

// uses template: hornet_mantle
fsm Mantle {
  start Init
  on HERO DAMAGED → Cancel All  // from any state
  on LEAVING SCENE → Cancel All  // from any state
  on FSM CANCEL → Cancel All  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Effects", storeResult=var "Effects")
    FindChild(gameObject=var "Effects", childName="clamber_impact_effects", storeResult=var "clamber_impact_effects")
    FindGameObject(objectName="", withTag="CameraTarget", store=var "CamTarget")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    SetStringValue(stringVariable=var "Send Event", stringValue="", everyFrame=false)
    SetBoolValue(boolVariable=var "In Mantle", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "R1", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "R2", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "R3", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "R4", boolValue=false, everyFrame=false)
    on AIR MANTLE → Facing
  }

  state Facing {
    GetFsmBool(gameObject=Self, fsmName="Harpoon Dash", variableName="Harpoon Dashing", storeValue=var "Harpoon Dashing", everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"L", negativeEvent=→"R", space=0)
    on L → Set L
    on R → Set R
  }

  state Check Forward {
    CheckIsClamberBlocked(trueEvent=→"CANCEL", falseEvent=(none), storeValue=(unset))
    RayCast2d(fromGameObject=Self, fromPosition=(0, 0.67), direction=var "Ray1 Direction", space=1, distance=0.75, minDepth=(unset), maxDepth=(unset), hitEvent=→"CANCEL 1", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=Self, fromPosition=var "R2 Pos", direction=(0, -1), space=1, distance=2.26, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "R1", storeHitObject=var "", storeHitPoint=var "Point 1", storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=Self, fromPosition=var "R3 Pos", direction=(0, -1), space=0, distance=2.26, minDepth=(unset), maxDepth=(unset), hitEvent=(none), storeDidHit=var "R2", storeHitObject=var "", storeHitPoint=var "Point 2", storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.015686274, 0.89138603, 1), debug=true)
    Vector2AddXY(vector2Variable=var "Point 1", addX=0, addY=0.1, everyFrame=false, perSecond=false)
    Vector2AddXY(vector2Variable=var "Point 2", addX=0, addY=0.1, everyFrame=false, perSecond=false)
    RayCast2d(fromGameObject=<null>, fromPosition=var "Point 1", direction=(0, 1), space=0, distance=2.16, minDepth=(unset), maxDepth=(unset), hitEvent=→"CANCEL 2", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, debugColor=(0, 0.92156863, 0.015686275, 1), debug=true)
    RayCast2d(fromGameObject=<null>, fromPosition=var "Point 2", direction=(0, 1), space=0, distance=2.16, minDepth=(unset), maxDepth=(unset), hitEvent=→"CANCEL 4", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, debugColor=(0.3754698, 0.015686274, 1, 1), debug=true)
    BoolTest(boolVariable=var "R1", isTrue=(none), isFalse=→"CANCEL 3", everyFrame=false)
    BoolTest(boolVariable=var "R2", isTrue=(none), isFalse=→"CANCEL 5", everyFrame=false)
    GetVector2XY(vector2Variable=var "Point 1", storeX=(unset), storeY=var "Y1", everyFrame=false)
    GetVector2XY(vector2Variable=var "Point 2", storeX=(unset), storeY=var "Y2", everyFrame=false)
    FloatCompare(float1=var "Y1", float2=var "Y2", tolerance=0.1, equal=→"NEXT", lessThan=→"CANCEL 6", greaterThan=→"CANCEL 6", everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"NEXT", delay=0, everyFrame=false)
    on NEXT → Take Control
    on CANCEL 1 → Test Break
    on CANCEL 2 → Test Break
    on CANCEL 3 → Test Break
    on CANCEL 4 → Test Break
    on CANCEL 5 → Test Break
    on CANCEL 6 → Test Break
    on CANCEL → Test Break
  }

  state Take Control {
    SetHeroCState(VariableName="mantling", Value=true, EveryFrame=false, SetOppositeOnExit=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="ResetHardLandingTimer", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="START MANTLE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPRINT CANCEL", delay=0, everyFrame=false)
    SendEventToRegister(eventName="HERO MANTLE")
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetAllowNailChargingWhileRelinquished", parameters=[true], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=ResetAirMoves())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=ResetHardLandingTimer())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="DetachHeroLight", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=0, functionCall=ShortDetach())
    SetBoolValue(boolVariable=var "In Mantle", boolValue=true, everyFrame=false)
    SetHeroCState(VariableName="isInCancelableFSMMove", Value=true, EveryFrame=false, SetOppositeOnExit=false)  // disabled
    SetTransitionBlocked(SetTime=0, SetIsBlocked=true)
    on FINISHED → Cling
  }

  state Cling {
    SetBoolValue(boolVariable=var "Attack Queued", boolValue=false, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y1", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mantle Cling", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandom(gameObject=var "Self", audioClips=[hornet_edge_grab_v2_1 (herosfxstatic_assets_all.bundle), hornet_edge_grab_v2_2 (sfxstatic_assets_shared.bundle), hornet_edge_grab_v2_3 (sfxstatic_assets_shared.bundle), hornet_edge_grab_v2_4 (sfxstatic_assets_shared.bundle)], weights=[0.25, 0.25, 0.25, 0.25], pitchMin=1, pitchMax=1)
    PlayVibrationV2(vibrationData={lowFidelityVibration=LowFidelityVibrations(0), highFidelityVibration=<null>, gamepadVibration=<null>, ps5Vibration=<null>}, vibrationDataAsset=hornet_edge_grab (vibrationstatic_assets_vibrationdataasset.bundle), motors=None, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)
    QueueJump(queueBool=var "Jump Queue")
    ListenForDash(eventTarget=Self, wasPressed=→"DO SPRINT", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=var "Pressed R", pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=var "Pressed L", pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    BoolNoneTrue(boolVariables=[var "Pressed R", var "Pressed L"], sendEvent=(none), storeResult=var "Can Drop", everyFrame=true)
    BoolTest(boolVariable=var "Harpoon Dashing", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "clamber_impact_effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ListenForAttackV2(EventTarget=Self, WasPressed=(none), WasReleased=(none), IsPressed=(none), IsNotPressed=(none), queueBool=var "Attack Queued", DelayBeforeActive=0, IsActive=true)
    on FINISHED → Vault
    on DO SPRINT → Queue Sprint
  }

  state Vault {
    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=0, functionCall=EndFallStick())
    SetHeroCState(VariableName="isInCancelableFSMMove", Value=true, EveryFrame=false, SetOppositeOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mantle Vault", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=Grunt Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Translate(gameObject=Self, vector=(unset), x=(unset), y=1.6, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="ReattachHeroLight", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=true)
    BoolTest(boolVariable=var "Jump Queue", isTrue=→"JUMP", isFalse=(none), everyFrame=false)
    ListenForJump(eventTarget=Self, wasPressed=→"JUMP", wasReleased=(none), isPressed=(none), isNotPressed=(none), queueBool=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"ATTACK", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    ListenForAttackV2(EventTarget=Self, WasPressed=(none), WasReleased=(none), IsPressed=→"ATTACK", IsNotPressed=(none), queueBool=(unset), DelayBeforeActive=0, IsActive=var "Attack Queued")
    ListenForToolThrow(EventTarget=Self, WasPressed=→"TOOLTHROW", WasReleased=(none), IsPressed=(none), IsNotPressed=(none), RequireToolToThrow=true, IsActive=(unset))
    on FINISHED → Land
    on JUMP → Set Jump Cancel Anim
    on ATTACK → Attack Cancel
    on TOOLTHROW → Toolthrow Cancel
  }

  state Land {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="PlaySoftLandingEffect", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetBackOnGround", parameters=[], storeResult=(unset var), EveryFrame=false)
    ListenForDash(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"DO SPRINT", isNotPressed=(none), delayBeforeActive=0)  // disabled
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mantle Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), everyFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    ListenForJump(eventTarget=Self, wasPressed=→"JUMP", wasReleased=(none), isPressed=(none), isNotPressed=(none), queueBool=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"ATTACK", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    GetPlayerDataBool(boolName="hasDash", storeValue=var "Can Dash")
    ListenForDashV2(EventTarget=Self, WasPressed=→"DO SPRINT", WasReleased=(none), IsPressed=(none), IsNotPressed=(none), DelayBeforeActive=0, IsActive=var "Can Dash")
    ListenForToolThrow(EventTarget=Self, WasPressed=→"TOOLTHROW", WasReleased=(none), IsPressed=(none), IsNotPressed=(none), RequireToolToThrow=true, IsActive=(unset))
    on FINISHED → Sprint?
    on JUMP → Jump Cancel
    on ATTACK → Attack Cancel
    on TOOLTHROW → Toolthrow Cancel
    on DO SPRINT → Sprint
  }

  state End {
    SetTransitionBlocked(SetTime=0, SetIsBlocked=false)
    SetHeroCState(VariableName="isInCancelableFSMMove", Value=false, EveryFrame=false, SetOppositeOnExit=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RegainControl())
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetAllowNailChargingWhileRelinquished", parameters=[false], storeResult=(unset var), EveryFrame=false)
    SetHeroCState(VariableName="mantling", Value=false, EveryFrame=false, SetOppositeOnExit=false)
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"RUNNING", isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"RUNNING", isNotPressed=→"FINISHED", isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    on FINISHED → Start Animation Idle
    on RUNNING → Start Animation Run
  }

  state Set L {
    SetFloatValue(floatVariable=var "Speed", floatValue=-5, everyFrame=false)
    SetVector2XY(vector2Variable=var "Ray1 Direction", vector2Value=(unset), x=-1, y=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "R2 Pos", vector2Value=(unset), x=-0.77, y=0.67, everyFrame=false)
    SetVector2XY(vector2Variable=var "R3 Pos", vector2Value=(unset), x=-0.37, y=0.67, everyFrame=false)
    on FINISHED → Check Ledge
  }

  state Set R {
    SetFloatValue(floatVariable=var "Speed", floatValue=5, everyFrame=false)
    SetVector2XY(vector2Variable=var "Ray1 Direction", vector2Value=(unset), x=1, y=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "R2 Pos", vector2Value=(unset), x=0.77, y=0.67, everyFrame=false)
    SetVector2XY(vector2Variable=var "R3 Pos", vector2Value=(unset), x=0.37, y=0.67, everyFrame=false)
    on FINISHED → Check Ledge
  }

  state Cancel All {
    BoolTest(boolVariable=var "In Mantle", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetHeroCState(VariableName="isInCancelableFSMMove", Value=false, EveryFrame=false, SetOppositeOnExit=false)
    SetHeroCState(VariableName="mantling", Value=false, EveryFrame=false, SetOppositeOnExit=false)
    SetTransitionBlocked(SetTime=0, SetIsBlocked=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="ReattachHeroLight", parameters=[], storeResult=(unset var), EveryFrame=false)
    StringCompare(stringVariable=var "Send Event", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent=var "Send Event", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Test Break {
    on FINISHED → Idle
  }

  state Jump Cancel {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelDownSpikeBounces())
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetStartFromMantle())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetStartWithFullJump())
    SendEventByNameUpwards(Target=var "Clambered Collider", EventName="HERO CLAMBERED JUMP")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → End
  }

  state Sprint? {
    GetPlayerDataBool(boolName="hasDash", storeValue=var "Can Dash")
    ListenForDashV2(EventTarget=Self, WasPressed=(none), WasReleased=(none), IsPressed=→"DO SPRINT", IsNotPressed=(none), DelayBeforeActive=0, IsActive=var "Can Dash")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Start From Mantle
    on DO SPRINT → Sprint
  }

  state Attack Cancel {
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetStartWithAttack())
    SetBoolValue(boolVariable=var "Attack Queued", boolValue=false, everyFrame=false)
    on FINISHED → End
  }

  state Sprint {
    SetStringValue(stringVariable=var "Send Event", stringValue="MANTLE SPRINT", everyFrame=false)
    on FINISHED → Cancel All
  }

  state Start Animation Idle {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    on FINISHED → Idle
  }

  state Start Animation Run {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartAnimationControlRunning())
    on FINISHED → Idle
  }

  state Start From Mantle {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetStartFromMantle())
    on FINISHED → End
  }

  state Set Jump Cancel Anim {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelDownSpikeBounces())
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetPlayMantleCancel())
    PlayAudioEvent(audioClip=hornet_sprint_jump_2nd_spin (herosfxstatic_assets_all.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayVibrationV2(vibrationData={lowFidelityVibration=LowFidelityVibrations(0), highFidelityVibration=<null>, gamepadVibration=<null>, ps5Vibration=<null>}, vibrationDataAsset=hornet_sprint_jump_2nd_spin (vibrationstatic_assets_vibrationdataasset.bundle), motors=None, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)
    on FINISHED → Jump Cancel
  }

  state Sprint Cancel {
    SetBoolValue(boolVariable=var "Queued Sprint", boolValue=false, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.5, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetStartWithDash())
    on FINISHED → End
  }

  state Queue Sprint {
    SetBoolValue(boolVariable=var "Queued Sprint", boolValue=true, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Vault
  }

  state Toolthrow Cancel {
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetStartWithToolThrow())
    on FINISHED → End
  }

  state Check Ledge {
    CheckHeroClamberLedge(StoreY=var "Y1", StoreCollider=var "Clambered Collider", trueEvent=→"FINISHED", falseEvent=→"CANCEL", storeValue=(unset))
    on CANCEL → Test Break
    on FINISHED → Take Control
  }

  state Drop {
    SetTransitionBlocked(SetTime=0, SetIsBlocked=false)
    SetHeroCState(VariableName="isInCancelableFSMMove", Value=false, EveryFrame=false, SetOppositeOnExit=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RegainControl())
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetAllowNailChargingWhileRelinquished", parameters=[false], storeResult=(unset var), EveryFrame=false)
    SetHeroCState(VariableName="mantling", Value=false, EveryFrame=false, SetOppositeOnExit=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    on FINISHED → Idle
  }

  var Land Duration: float = 0
  var Speed: float = 0
  var Y1: float = 0
  var Y2: float = 0
  var In Mantle: bool = false
  var Jump Queue: bool = false
  var Queued Sprint: bool = false
  var R1: bool = false
  var R2: bool = false
  var R3: bool = false
  var R4: bool = false
  var Can Dash: bool = false
  var Pressed R: bool = false
  var Can Drop: bool = false
  var Pressed L: bool = false
  var Harpoon Dashing: bool = false
  var Attack Queued: bool = false
  var Send Event: string = ""
  var Point 1: vector2 = (0, 0)
  var Point 2: vector2 = (0, 0)
  var R2 Pos: vector2 = (0, 0)
  var R3 Pos: vector2 = (0, 0)
  var Ray1 Direction: vector2 = (0, 0)
  var Effects: gameObject = <null>
  var Self: gameObject = <null>
  var clamber_impact_effects: gameObject = <null>
  var CamTarget: gameObject = <null>
  var Clambered Collider: gameObject = <null>
}
