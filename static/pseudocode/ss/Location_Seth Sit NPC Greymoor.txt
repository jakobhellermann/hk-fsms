# Location
# 1 location(s):
#   Greymoor_02 / Seth Sit NPC Greymoor

// uses template: seth_location_evaluate
fsm Location {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Is Current Location?
  }

  state Is Current Location? {
    CheckStaticBool(variableName="SethKeepLocation", trueEvent=→"TRUE", falseEvent=(none), EveryFrame=false)
    GetPlayerDataVariable(VariableName="SethNpcLocation", StoreValue=var "Current Location")
    EnumCompare(enumVariable=var "Current Location", compareTo=var "This Location", equalEvent=→"TRUE", notEqualEvent=→"FALSE", storeResult=(unset), everyFrame=false)
    on FALSE → Disable
    on TRUE → Wait For Talk
  }

  state Disable {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait For Talk {
    SendEventToRegister(eventName="SETH PRESENT")
    SetStaticVariable(variableName="SethKeepLocation", setValue=true, sceneTransitionsLimit=3)
    on TALKED → Set New Location
  }

  state Set New Location {
    EnumCompare(enumVariable=var "This Location", compareTo=Fleatopia, equalEvent=→"CANCEL", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    IntOperator(integer1=var "This Location", integer2=1, operation=Add, storeResult=var "New Location Int", everyFrame=false)
    ConvertIntToEnum(IntValue=var "New Location Int", EnumVariable=var "New Location", EveryFrame=false)
    SetPlayerDataVariable(VariableName="SethNpcLocation", SetValue=var "New Location")
    on CANCEL → Fleatopia
  }

  state Fleatopia {
  }

  var This Location: enum = SethNpcLocations(1)

  var New Location Int: int = 0
  var Current Location: enum = SethNpcLocations(0)
  var New Location: enum = SethNpcLocations(0)
}
