# Music Control
# 1 location(s):
#   Cog_Dancers_boss / Dancer Control

fsm Music Control {
  start Init
  on START P4 → P4 Music  // from any state
  on START P3 → P3 Music  // from any state
  on STOP → Stop  // from any state
  on START P2 → P2 Music  // from any state

  state Init {
    on FINISHED → Dormant
  }

  state Dormant {
    on START P1 → P1 Music
  }

  state P1 Music {
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.1)
    SetAudioClip(gameObject=Self, audioClip=H181ClockworkDancers1stPhase (sfxstatic_assets_areacog.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }

  state Stop {
    AudioStop(gameObject=Self, fadeTime=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=H181ClockworkDancersSting (sfxstatic_assets_areacog.bundle))
  }

  state P2 Music {
    SetAudioClip(gameObject=Self, audioClip=H181ClockworkDancers2ndPhase (sfxstatic_assets_areacog.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }

  state P3 Music {
    SetAudioClip(gameObject=Self, audioClip=H181ClockworkDancers3rdPhase (sfxstatic_assets_areacog.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }

  state P4 Music {
    SetAudioClip(gameObject=Self, audioClip=H181ClockworkDancers4thPhase (sfxstatic_assets_areacog.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
  }
}
