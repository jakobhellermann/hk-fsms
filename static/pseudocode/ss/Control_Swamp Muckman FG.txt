# Control
# 6 location(s):
#   Shadow_03 / Swamp Muckman FG
#   Shadow_04b / Swamp Muckman FG
#   Shadow_05 / Swamp Muckman FG

#   Shadow_12 / Swamp Muckman FG

fsm Control {
  start Dormant

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="AMBUSH", outOfRangeEvent="", everyFrame=true)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Floor Y", space=0, everyFrame=false)
    on AMBUSH → Voice
  }

  state Hop {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hop")
    RandomFloat(min=14, max=15, storeResult=var "X Speed")
    RandomFloat(min=28, max=30, storeResult=var "Y Speed")
    SetVelocityByScale(gameObject=Self, speed=var "X Speed", ySpeed=var "Y Speed", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Floor Y", Space=0, EveryFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayAudioEventRandom(audioClips=[swamp_muckman_jump_1 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_jump_2 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_jump_3 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_jump_4 (sfxstatic_assets_areaswamp.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Hop Type
  }

  state Air {
    CheckYPosition(gameObject=Self, compareTo=var "Floor Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAND → Land
  }

  state Hop Type {
    IntAdd(intVariable=var "Hops", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hops", integer2=0, equal=→"FINAL", lessThan=→"FINAL", greaterThan=(none), everyFrame=false)
    on FINISHED → Air
    on FINAL → Air Final
  }

  state Land {
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Floor Y", Space=0, EveryFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hop Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    PlayAudioEventRandom(audioClips=[swamp_muckman_land_1 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_land_2 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_land_3 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_land_4 (sfxstatic_assets_areaswamp.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTable(gameObject=Swamp Muckman FG/Audio Player Muckmen crowd calls, audioClipTable=Stilkin_call_group (sfxdynamic_assets_areaswamp.bundle), pitchOffset=0, stopPreviousSound=false, forcePlay=false)
    on FINISHED → Hop
  }

  state Air Final {
    Wait(time=1.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    PlayAudioEventRandom(audioClips=[swamp_muckman_land_1 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_land_2 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_land_3 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_land_4 (sfxstatic_assets_areaswamp.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Voice {
    PlayAudioEventRandom(audioClips=[Stilkin_call_08 (sfxstatic_assets_areaswamp.bundle), Stilkin_call_09 (sfxstatic_assets_areaswamp.bundle), Stilkin_call_11 (sfxstatic_assets_areaswamp.bundle), Stilkin_call_12 (sfxstatic_assets_areaswamp.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Hop
  }

  var Y Speed: float = 0
  var X Speed: float = 0
  var Floor Y: float = 0
  var Hops: int = 3
  var Activated: bool = false
}
