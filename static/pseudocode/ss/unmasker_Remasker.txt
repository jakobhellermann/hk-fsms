// uses template: remasker
fsm unmasker {
  start Pause
  on HERO LEAVE → Hero Leave  // from any state

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="Player", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Play Sound
  }

  state Fade Out {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="Player", sendEvent=→"COVER", storeCollider=var "")
    on COVER → Fade In
  }

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Fade In {
    iTweenFadeTo(gameObject=Self, id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="Player", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Fade Out
  }

  state Hero Leave {
  }

  state Play Sound {
    BoolTest(boolVariable=var "Activated", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Play Sound", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClip=secret_discovered_temp (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Fade Out
  }

  var Fade Time: float = 0.5
  var Play Sound: bool = false

  var Activated: bool = false
}
