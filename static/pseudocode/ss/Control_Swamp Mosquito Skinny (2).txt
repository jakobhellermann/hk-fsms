# Control
# 2 location(s):
#   Aqueduct_01 / Swamp Mosquito Skinny (2)
#   Aqueduct_03 / Swamp Mosquito Skinny (2)

fsm Control {
  start Pause
  on GO UP → Go Up  // from any state

  state Init State {
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")  // disabled
    StringCompare(stringVariable=var "Clip", compareTo="Wall Cling", equalEvent=(none), notEqualEvent=(none), storeResult=(unset), everyFrame=false)  // disabled
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(-1, 0), space=0, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"WALL L", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Wall Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(1, 0), space=0, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"WALL R", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Wall Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    ActivateGameObject(gameObject=var "Terrain Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on WALL L → Stick to Wall L
    on FINISHED → Start Idle
    on WALL R → Stick to Wall R
  }

  state Start Idle {
    SetPositionToObject2D(gameObject=var "Idle Point", targetObject=var "Self", xOffset=0, yOffset=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    RandomFloat(min=0.85, max=1.15, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    FloatOperator(float1=var "Pitch", float2=0.3, operation=Add, storeResult=var "Pitch High", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Idle Dart
  }

  state Idle Dart {
    GetAngleToTarget2D(gameObject=Self, target=var "Idle Point", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    FloatAddRandom(floatVariable=var "Angle", addMin=-25, addMax=25)
    RandomFloat(min=2, max=3, storeResult=var "Speed")
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.975, brakeOnExit=false)
    WaitRandom(timeMin=1.5, timeMax=2.5, finishEvent=→"FINISHED", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"ALERT", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Idle Dart
    on ALERT → Startle
    on TOOK DAMAGE → Startle
  }

  state Wall Cling {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Terrain Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=0, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=(none), ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on TOOK DAMAGE → Wall Wake
    on ALERT → Wall Wake
  }

  state Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    PlayAudioEvent(audioClip=swamp_mosquito_notice_hero (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Shift Antic
  }

  state Wall Wake {
    SetBoolValue(boolVariable=var "Aim High", boolValue=true, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Wake Range")
    ActivateGameObject(gameObject=var "Wake Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    SetVelocityByScale(gameObject=Self, speed=-10, ySpeed=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    PlayAudioEvent(audioClip=swamp_mosquito_notice_hero (sfxstatic_assets_areaswamp.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    RandomFloat(min=0.85, max=1.15, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Shift Antic
  }

  state Start at Node? {
    SetParent(gameObject=var "Fly Target", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FindGameObject(objectName="", withTag="Orb Target", store=var "Node Folder")
    SetParent(gameObject=var "Idle Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    CheckXPosition(gameObject=Self, compareTo=0, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"NODE", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    on FINISHED → Init State
    on NODE → Get Node
  }

  state Node Check {
    SetPosition2D(GameObject=Self, Vector=var "Spawn Point", X=0, Y=0, Space=0, EveryFrame=false)
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(-1, 0), space=0, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"WALL L", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Wall Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(1, 0), space=0, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"WALL R", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Wall Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on WALL L → Stick to Wall L
    on WALL R → Stick to Wall R
    on FINISHED → Start Idle
  }

  state Stick to Wall L {
    SetPosition2D(GameObject=Self, Vector=var "Wall Point", X=0, Y=0, Space=0, EveryFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1.2, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wall Cling")
    Translate(gameObject=Self, vector=(unset), x=0.65, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Wall Cling
  }

  state Stick to Wall R {
    SetPosition2D(GameObject=Self, Vector=var "Wall Point", X=0, Y=0, Space=0, EveryFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1.2, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wall Cling")
    Translate(gameObject=Self, vector=(unset), x=-0.65, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Wall Cling
  }

  state Go Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=5, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    Decelerate(gameObject=Self, deceleration=0.9)
    on FINISHED → Attack Choice
  }

  state Get Node {
    GetRandomChild(gameObject=var "Node Folder", storeResult=var "My Node")
    SetParent(gameObject=var "My Node", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetPosition2D(gameObject=var "My Node", vector=var "Spawn Point", x=(unset), y=(unset), space=0, everyFrame=false)
    SendRandomEvent(events=[→"FINISHED", (none)], weights=[0.5, 0.5], delay=0)
    FindNamedChild(gameObject=var "My Node", storeResult=var "Alt Point")
    GetPosition2D(gameObject=var "Alt Point", vector=var "Spawn Point", x=(unset), y=(unset), space=0, everyFrame=false)
    on FINISHED → Node Check
  }

  state Choose Drift Dir {
    PlayAudioEventRandom(audioClips=[coral_flyer_in_air_dash_1 (sfxstatic_assets_areacoral.bundle), coral_flyer_in_air_dash_2 (sfxstatic_assets_areacoral.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Target L
    on R → Target R
  }

  state Target R {
    SetPositionToObject2D(gameObject=var "Fly Target", targetObject=var "Hero", xOffset=8, yOffset=0, everyFrame=false)
    BoolTest(boolVariable=var "Aim High", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetPositionToObject2D(gameObject=var "Fly Target", targetObject=var "Hero", xOffset=-8, yOffset=0, everyFrame=false)
    on FINISHED → Shift Target
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Start at Node?
  }

  state Shift Target {
    BoolTest(boolVariable=var "Aim High", isTrue=→"AIM HIGH", isFalse=(none), everyFrame=false)
    TranslateRandom(gameObject=var "Fly Target", translateMin=(-2, 0, 0), translateMax=(2, 6, 0), space=1)
    on FINISHED → Move Choice
    on AIM HIGH → Aim High
  }

  state Target L {
    SetPositionToObject2D(gameObject=var "Fly Target", targetObject=var "Hero", xOffset=-8, yOffset=0, everyFrame=false)
    BoolTest(boolVariable=var "Aim High", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetPositionToObject2D(gameObject=var "Fly Target", targetObject=var "Hero", xOffset=8, yOffset=0, everyFrame=false)
    on FINISHED → Shift Target
  }

  state Shift Antic {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shift Fly", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Choose Drift Dir
  }

  state Move Choice {
    GetDistanceV2(gameObject=Self, target=var "Fly Target", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=8, tolerance=0, equal=→"LONG", lessThan=→"SHORT", greaterThan=→"LONG", everyFrame=false)
    on SHORT → Short Fly
    on LONG → Long Fly
  }

  state Short Fly {
    PlayAudioEvent(audioClip=<null>, pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetDistanceV2(gameObject=Self, target=var "Fly Target", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=false)
    FloatOperator(float1=var "Distance", float2=8, operation=Multiply, storeResult=var "Speed", everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    FireAtTarget(gameObject=Self, target=var "Fly Target", speed=var "Speed", position=(0, 0, 0), spread=0, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    WaitRandom(timeMin=0.4, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fly End
  }

  state Fly End {
    IntAdd(intVariable=var "Unalert Count", add=1, everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Unalert Range", storeResult=var "In Unalert Range", sendEvent="", outOfRangeEvent="", everyFrame=false)
    SetIntValueBool(intVariable=var "Unalert Count", testBool=var "In Unalert Range", trueValue=0, falseValue=(unset))
    IntCompare(integer1=var "Unalert Count", integer2=3, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=false)
    on FINISHED → Attack Choice
    on UNALERT → Start Idle
  }

  state Long Fly {
    FireAtTarget(gameObject=Self, target=var "Fly Target", speed=30, position=(0, 0, 0), spread=0, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.92, brakeOnExit=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    WaitRandom(timeMin=0.7, timeMax=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fly End
  }

  state Shoot Fly {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shoot Fly")
    ActivateGameObject(gameObject=var "Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DO SHOOT", delay=0, everyFrame=false)
    GetDistanceV2(gameObject=Self, target=var "Fly Target", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=false)
    FloatOperator(float1=var "Distance", float2=10, operation=Multiply, storeResult=var "Speed", everyFrame=false)
    FloatClamp(floatVariable=var "Speed", minValue=(unset), maxValue=60, everyFrame=false)
    FireAtTarget(gameObject=Self, target=var "Fly Target", speed=var "Speed", position=(0, 0, 0), spread=0, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.875, brakeOnExit=false)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fly End
  }

  state Attack Choice {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"ATTACK", delay=0, everyFrame=false)  // disabled
    GetDistanceV2(gameObject=Self, target=var "Hero", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=18, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SendRandomEventV4(events=[→"FINISHED", →"ATTACK"], weights=[0.5, 0.5], eventMax=[2, 1], missedMax=[1, 2], activeBool=(unset))
    on ATTACK → Shift Antic 2
    on FINISHED → Shift Antic
    on SING → Sing
  }

  state Choose Drift Dir 2 {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    PlayAudioEventRandom(audioClips=[coral_flyer_in_air_dash_1 (sfxstatic_assets_areacoral.bundle), coral_flyer_in_air_dash_2 (sfxstatic_assets_areacoral.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Target L 2
    on R → Target R 2
  }

  state Target R 2 {
    SetPositionToObject2D(gameObject=var "Fly Target", targetObject=var "Hero", xOffset=10, yOffset=4, everyFrame=false)
    TranslateRandom(gameObject=var "Fly Target", translateMin=(-1, -1.5, 0), translateMax=(1, 1.5, 0), space=0)
    on FINISHED → Shoot Fly
  }

  state Target L 2 {
    SetPositionToObject2D(gameObject=var "Fly Target", targetObject=var "Hero", xOffset=-10, yOffset=3, everyFrame=false)
    TranslateRandom(gameObject=var "Fly Target", translateMin=(-1, -1.5, 0), translateMax=(1, 1.5, 0), space=0)
    on FINISHED → Shoot Fly
  }

  state Shift Antic 2 {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shift Fly", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Choose Drift Dir 2
  }

  state Aim High {
    TranslateRandom(gameObject=var "Fly Target", translateMin=(-2, 8, 0), translateMax=(2, 8, 0), space=1)
    SetBoolValue(boolVariable=var "Aim High", boolValue=false, everyFrame=false)
    on FINISHED → Move Choice
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"SING DURATION END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=mosquito_sing (sfxdynamic_assets_areaswamp.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Attack Choice
  }

  var Angle: float = 0
  var Speed: float = 0
  var Distance: float = 0
  var Pitch: float = 0
  var Pitch Crt: float = 0
  var Pitch High: float = 0
  var Unalert Count: int = 0
  var In Unalert Range: bool = false
  var Aim High: bool = false
  var Clip: string = ""
  var Wall Point: vector2 = (0, 0)
  var Spawn Point: vector2 = (0, 0)
  var Terrain Box: gameObject = Swamp Mosquito Skinny (2)/Terrain Box
  var Idle Point: gameObject = Swamp Mosquito Skinny (2)/Idle Point
  var Self: gameObject = Swamp Mosquito Skinny (2)
  var Wake Range: gameObject = <null>
  var Node Folder: gameObject = <null>
  var My Node: gameObject = <null>
  var Alt Point: gameObject = <null>
  var Fly Target: gameObject = Swamp Mosquito Skinny (2)/Fly Target
  var Glow: gameObject = Swamp Mosquito Skinny (2)/Glow
  var Audio Loop Voice: gameObject = Swamp Mosquito Skinny (2)/Audio Loop Voice
}
