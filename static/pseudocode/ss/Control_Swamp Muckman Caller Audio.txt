# Control
# 4 location(s):
#   Shadow_03 / Swamp Muckman Caller Audio
#   Shadow_04b / Swamp Muckman Caller Audio
#   Shadow_10 / Swamp Muckman Caller Audio
#   Shadow_12 / Swamp Muckman Caller Audio

fsm Control {
  start Start Pause

  state Pause {
    WaitRandom(timeMin=1, timeMax=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Pos
  }

  state Play Voice {
    GetRandomAudioClipFromTable(Table=Stilkin_call (sfxdynamic_assets_areaswamp.bundle), ForcePlay=false, StoreClip=var "Clip")
    AudioPlaySimple(gameObject=var "Audio Player", volume=(unset), oneShotClip=var "Clip")
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Pause
  }

  state Sound Type {
    SendRandomEventV4(events=[→"VOICE", →"ACTION"], weights=[1, 1], eventMax=[3, 2], missedMax=[2, 3], activeBool=(unset))
    on VOICE → Play Voice
    on ACTION → Play Action
  }

  state Play Action {
    PlayAudioEventRandom(audioClips=[swamp_muckman_jump_1 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_jump_2 (sfxstatic_assets_areaswamp.bundle), swamp_muckman_turn_to_dive_under (sfxstatic_assets_areaswamp.bundle), swamp_muckman_water_enter_option (sfxstatic_assets_areaswamp.bundle)], pitchMin=1, pitchMax=1, volume=0.6, audioPlayerPrefab=var "", spawnPoint=var "Audio Player", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Pause
  }

  state Set Pos {
    SetPositionToObject2D(gameObject=Self, targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    RandomFloat(min=8, max=25, storeResult=var "X Offset")
    RandomlyFlipFloat(storeResult=var "X Offset")
    RandomFloat(min=-10, max=10, storeResult=var "Y Offset")
    SetPosition2d(gameObject=var "Audio Player", vector=(unset), x=var "X Offset", y=var "Y Offset", space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Sound Type
  }

  state Start Pause {
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Pause
  }

  var X Offset: float = 0
  var Volume: float = 0
  var Pitch: float = 0
  var Wait Min: float = 0.5
  var Wait Max: float = 2
  var Y Offset: float = 0
  var Audio Player: gameObject = Swamp Muckman Caller Audio/Audio Player
  var Clip: object = <null>
}
