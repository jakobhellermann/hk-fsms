fsm Control {
  start Pause

  state Check Hero Out {
    SetBoxColliderTrigger(gameObject=Self, trigger=true)
    CheckTrackTriggerCount(target=Self, count=0, test=Equal, everyFrame=true, successEvent=→"HARDEN")
    on HARDEN → Set Solid
  }

  state Set Solid {
    SetBoxColliderTrigger(gameObject=Self, trigger=false)
    EnableFsmSelf(SetEnabled=false)
  }

  state Pause {
    SetBoxColliderTrigger(gameObject=Self, trigger=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Hero Out
  }
}
