# FSM
# 1 location(s):
#   Bone_East_08_boss_beastfly / Boss Scene Beastfly/Beastfly States/Active/Bone Flyer Giant

fsm FSM {
  start Init

  state Init {
    on START DEATH CONTROL → In Killable Range
  }

  state In Killable Range {
    SetSpecialDeath(target=Self, hasSpecialDeath=false)
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=var "Self X", y=(unset), space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Self X", lowerValue=var "X Min", upperValue=var "X Max", boolVariable=(unset), trueEvent=(none), falseEvent=→"OUT", everyFrame=true)
    on OUT → Out Killable Range
  }

  state Out Killable Range {
    SetSpecialDeath(target=Self, hasSpecialDeath=true)
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=var "Self X", y=(unset), space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Self X", lowerValue=var "X Min", upperValue=var "X Max", boolVariable=(unset), trueEvent=→"IN", falseEvent=(none), everyFrame=true)
    on IN → In Killable Range
  }

  var Self X: float = 0
  var X Min: float = 70.6
  var X Max: float = 91.05
}
