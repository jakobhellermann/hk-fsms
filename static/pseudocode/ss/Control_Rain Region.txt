fsm Control {
  start Init

  state Init {
    on FINISHED → Inactive
  }

  state Inactive {
    SendTrigger2DEvent(eventTarget=Self, trigger=OnTriggerStay2D, collideTag="", collideLayer=(unset), sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Active
  }

  state Active {
    SendTrigger2DEvent(eventTarget=Self, trigger=OnTriggerExit2D, collideTag="", collideLayer=(unset), sendEvent=→"EXIT", storeCollider=var "")
    Wait(time=1, finishEvent=→"CLEAN", realTime=false)
    PlayParticleEmitterInState(gameObject=var "Pt Rain")
    SetPositionToObject2D(gameObject=var "Rain Holder", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    on EXIT → Inactive
    on CLEAN → Clean
  }

  state Clean {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=ReduceOdours(5))
    on FINISHED → Active
  }

  var Pt Rain: gameObject = Rain Region/Rain Holder/Pt Rain
  var Rain Holder: gameObject = Rain Region/Rain Holder
}
