# Control
# 1 location(s):
#   Library_04 / Acolyte Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "First Cluster")  // disabled
    FindNamedChild(gameObject=Self, storeResult=var "Return")  // disabled
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="scholarAcolytesReleased", isTrue=→"SEEN", isFalse=→"END")
    on SEEN → Seen
    on END → Unreleased
  }

  state Seen {
    RandomInt(min=7, max=8, storeResult=var "Spawns", inclusiveMax=true, noRepeat=false)
    on FINISHED → Activate
  }

  state Activate {
    GetRandomChild(gameObject=Self, storeResult=var "Acolyte")
    ActivateGameObject(gameObject=var "Acolyte", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Acolyte", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    IntAdd(intVariable=var "Spawns", add=-1, everyFrame=false)
    IntCompare(integer1=var "Spawns", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"LOOP", everyFrame=false)
    on END → End
    on LOOP → Activate
  }

  state End {
  }

  state Unreleased {
  }

  var Spawns: int = 0
  var First Cluster: gameObject = <null>
  var Return: gameObject = <null>
  var Acolyte: gameObject = <null>
}
