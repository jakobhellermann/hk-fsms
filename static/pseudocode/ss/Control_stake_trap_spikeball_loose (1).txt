# Control
# 1 location(s):
#   Shadow_11 / stake_trap_spikeball_loose Scene/stake_trap_spikeball_loose (1)

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on ROLL → Antic
  }

  state Roll Pause {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Wait(time=var "Roll Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roll Out
  }

  state Antic {
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=30)
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Roll Pause
  }

  state Roll Out {
    PlayParticleEmitter(gameObject=var "Pt Trail", emit=0, resetIfPlaying=false)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=3.5, y=(unset), vector3=(unset), everyFrame=true)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Velocity X", float2=-20, operation=Multiply, storeResult=var "Spin", everyFrame=true)
    FloatClamp(floatVariable=var "Spin", minValue=-200, maxValue=200, everyFrame=true)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Spin", everyFrame=true)
    CheckYPosition(gameObject=Self, compareTo=var "Falling Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    RandomFloat(min=1.2, max=1.5, storeResult=var "Pitch")
    SetAudioPitch(gameObject=var "Audio Loop Roll", pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Loop Roll", volume=(unset), oneShotClip=<null>)
    on NEXT → Falling
  }

  state Falling {
    CheckYPosition(gameObject=Self, compareTo=var "End Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    FadeAudio(gameObject=var "Audio Loop Roll", startVolume=1, endVolume=0, time=0.5)
    on NEXT → Smash
  }

  state Smash {
    ActivateGameObject(gameObject=var "Destroy Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bouncer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetCircleCollider(gameObject=Self, active=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    Wait(time=2, finishEvent=(none), realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=<null>, cancelOnExit=false, DoFreeze=false, Delay=0)
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(25, 18), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    PlayAudioEvent(audioClip=bone_boulder_ground_impact (sfxstatic_assets_areabone.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.5, audioPlayerPrefab=Audio Player Actor Distant V2@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=shellwood_twig_wall_break (sfxstatic_assets_areashellwood.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.5, audioPlayerPrefab=Audio Player Actor Distant V2@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Roll Pause: float = 0.25

  var End Y: float = 46
  var Falling Y: float = 101
  var Spin: float = 0
  var Velocity X: float = 0
  var Pitch: float = 0
  var Activated: bool = false
  var Audio Loop Roll: gameObject = stake_trap_spikeball_loose Scene/stake_trap_spikeball_loose (1)/Audio Loop Roll
  var Pt Antic: gameObject = stake_trap_spikeball_loose Scene/stake_trap_spikeball_loose (1)/Pt Antic
  var Pt Trail: gameObject = stake_trap_spikeball_loose Scene/stake_trap_spikeball_loose (1)/Pt Trail
  var Destroy Effects: gameObject = stake_trap_spikeball_loose Scene/stake_trap_spikeball_loose (1)/Destroy Effects
  var Bouncer: gameObject = stake_trap_spikeball_loose Scene/stake_trap_spikeball_loose (1)/Bouncer
  var Terrain Detector: gameObject = stake_trap_spikeball_loose Scene/stake_trap_spikeball_loose (1)/Terrain Detector
}
