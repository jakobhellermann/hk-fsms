// uses template: detect_taunt
fsm detect_taunt {
  start Init

  state Init {
    AddEventRegister(target=Self, eventName="DID TAUNT")
    on FINISHED → Idle
  }

  state Idle {
    on DID TAUNT → Check
  }

  state Check {
    GetDistanceV2(gameObject=Self, target=var "Hero", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=var "Range", tolerance=0, equal=→"DID TAUNT", lessThan=→"FINISHED", greaterThan=→"DID TAUNT", everyFrame=false)
    on DID TAUNT → Send Msg
    on FINISHED → Idle
  }

  state Send Msg {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TAUNT RECEIVED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Range: float = 0

  var Distance: float = 0
}
