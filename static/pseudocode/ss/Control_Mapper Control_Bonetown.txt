# Control
# 1 location(s):
#   Bonetown / Mapper Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindNamedChild(gameObject=Self, storeResult=var "Shakra Away")
    FindNamedChild(gameObject=Self, storeResult=var "Shakra Resting")
    PlayerDataBoolTest(boolName="defeatedBellBeast", isTrue=(none), isFalse=→"NONE")
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"NONE", isFalse=(none))
    PlayerDataBoolTest(boolName="spinnerDefeated", isTrue=→"NONE", isFalse=(none))
    PlayerDataBoolTest(boolName="skullKingInvaded", isTrue=→"NONE", isFalse=(none))
    PlayerDataBoolTest(boolName="skullKingWillInvade", isTrue=→"NONE", isFalse=(none))
    GetPlayerDataBool(boolName="visitedDeepDocks", storeValue=var "Visited Docks")
    GetPlayerDataBool(boolName="visitedShellwood", storeValue=var "Visited Shellwood")
    BoolNoneTrue(boolVariables=[var "Visited Docks", var "Visited Shellwood"], sendEvent=→"RESTING", storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName="mapperAway", isTrue=→"AWAY", isFalse=→"RESTING")
    on NONE → None
    on AWAY → Away
    on RESTING → Resting
  }

  state None {
  }

  state Away {
    ActivateGameObject(gameObject=var "Shakra Away", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shakra Resting", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Resting {
    ActivateGameObject(gameObject=var "Shakra Away", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shakra Resting", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Visited Greymoor: bool = false
  var Visited Shellwood: bool = false
  var Visited Docks: bool = false
  var Shakra Away: gameObject = <null>
  var Shakra Resting: gameObject = <null>
}
