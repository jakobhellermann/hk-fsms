fsm Control {
  start State 1

  state State 1 {
    AudioStop(gameObject=Self, fadeTime=0)
    on STEAM DOWN ON → State 2
  }

  state State 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on STEAM DOWN OFF → State 3
  }

  state State 3 {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 1
  }
}
