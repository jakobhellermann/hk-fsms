# Detect Grab
# 5 location(s):
#   Aqueduct_01 / Swamp Barnacle (1)/Tendril
#   Aqueduct_02 / Swamp Barnacle (1)/Tendril
#   Aqueduct_03 / Swamp Barnacle (1)/Tendril
#   Aqueduct_07 / Swamp Barnacle (1)/Tendril
#   Aqueduct_08 / Swamp Barnacle (1)/Tendril

fsm Detect Grab {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventV2(gameObject=Self, trigger=0, collideTag="", collideLayer=Hero Box(20), sendEvent=→"HERO ENTER", storeCollider=var "")
    Trigger2dEventV2(gameObject=Self, trigger=1, collideTag="", collideLayer=Hero Box(20), sendEvent=→"HERO ENTER", storeCollider=var "")
    Trigger2dEventV2(gameObject=Self, trigger=0, collideTag="", collideLayer=Interactive Object(19), sendEvent=→"INTERACTIVE", storeCollider=var "Colliding Object")
    on HERO ENTER → Check
    on INTERACTIVE → Check Interactive
    on FRIGHTENED → Frightened
  }

  state Check {
    CanHeroBeGrabbed(eventTarget=Self, canTakeDmgEvent=→"GRAB", cannotTakeDmgEvent=→"CANCEL")  // disabled
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="CanBeBarnacleGrabbed", parameters=[], storeResult=var "Can Be Grabbed", EveryFrame=false)
    BoolTest(boolVariable=var "Can Be Grabbed", isTrue=→"GRAB", isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Idle
    on GRAB → Send Grab Event
  }

  state Send Grab Event {
    SendEventByName(eventTarget=GameObjectFSM(var "Parent", fsm="Control"), sendEvent="GRAB HERO", delay=0, everyFrame=false)
    on TENDRIL RESET → Idle
  }

  state Check Interactive {
    HasComponent(gameObject=var "Colliding Object", component="CustomTag", removeOnExit=false, trueEvent=(none), falseEvent=→"FINISHED", store=(unset), everyFrame=false)
    CallMethodProper(gameObject=var "Colliding Object", behaviour="CustomTag", methodName="GetCustomTagAsString", parameters=[], storeResult=var "Custom Tag", EveryFrame=false)
    StringSwitch(stringVariable=var "Custom Tag", compareTo=["Bomb", "DustBomb", "LightningBolaBall", "LightningBola", "SkinnyMosquito", "CogworkHatchling"], sendEvent=[→"BOMB", →"DUST BOMB", →"LIGHTNING BOLA BALL", →"LIGHTNING BOLA", →"MOSQUITO", →"HATCHLING"], everyFrame=false)
    on FINISHED → Idle
    on BOMB → Bomb
    on DUST BOMB → Dust Bomb
    on LIGHTNING BOLA BALL → Bola Ball
    on LIGHTNING BOLA → Lightning Bola
    on MOSQUITO → Mosquito
    on HATCHLING → Hatchling
  }

  state Bomb {
    GetPosition2D(gameObject=var "Colliding Object", vector=var "Grabbed Object Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetFsmVector2(gameObject=var "Parent", fsmName="Control", variableName="Grabbed Object Pos", setValue=var "Grabbed Object Pos", everyFrame=false)
    GetParent(gameObject=var "Colliding Object", storeResult=var "Parent Object")
    RecycleObject(gameObject=var "Parent Object")
    SendEventByName(eventTarget=GameObjectFSM(var "Parent", fsm="Control"), sendEvent="GRAB BOMB", delay=0, everyFrame=false)
    on TENDRIL RESET → Idle
  }

  state Dust Bomb {
    GetPosition2D(gameObject=var "Colliding Object", vector=var "Grabbed Object Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetFsmVector2(gameObject=var "Parent", fsmName="Control", variableName="Grabbed Object Pos", setValue=var "Grabbed Object Pos", everyFrame=false)
    GetParent(gameObject=var "Colliding Object", storeResult=var "Parent Object")
    RecycleObject(gameObject=var "Parent Object")
    SendEventByName(eventTarget=GameObjectFSM(var "Parent", fsm="Control"), sendEvent="GRAB DUST BOMB", delay=0, everyFrame=false)
    on TENDRIL RESET → Idle
  }

  state Bola Ball {
    GetPosition2D(gameObject=var "Colliding Object", vector=var "Grabbed Object Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetFsmVector2(gameObject=var "Parent", fsmName="Control", variableName="Grabbed Object Pos", setValue=var "Grabbed Object Pos", everyFrame=false)
    GetParent(gameObject=var "Colliding Object", storeResult=var "Parent Object")
    RecycleObject(gameObject=var "Parent Object")
    SendEventByName(eventTarget=GameObjectFSM(var "Parent", fsm="Control"), sendEvent="LIGHTNING BOLA BALL", delay=0, everyFrame=false)
    on TENDRIL RESET → Idle
  }

  state Lightning Bola {
    GetPosition2D(gameObject=var "Colliding Object", vector=var "Grabbed Object Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    GetParent(gameObject=var "Colliding Object", storeResult=var "Parent Object")
    GetVelocity2d(gameObject=var "Parent Object", vector=var "Grabbed Object Velocity", x=(unset), y=(unset), space=0, everyFrame=false)
    SetFsmVector2(gameObject=var "Parent", fsmName="Control", variableName="Grabbed Object Pos", setValue=var "Grabbed Object Pos", everyFrame=false)
    SetFsmVector2(gameObject=var "Parent", fsmName="Control", variableName="Grabbed Object Velocity", setValue=var "Grabbed Object Velocity", everyFrame=false)
    RecycleObject(gameObject=var "Parent Object")
    SendEventByName(eventTarget=GameObjectFSM(var "Parent", fsm="Control"), sendEvent="LIGHTNING BOLA", delay=0, everyFrame=false)
    on TENDRIL RESET → Idle
  }

  state Mosquito {
    GetPosition2D(gameObject=var "Colliding Object", vector=var "Grabbed Object Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    GetParent(gameObject=var "Colliding Object", storeResult=var "Parent Object")
    SetFsmVector2(gameObject=var "Parent", fsmName="Control", variableName="Grabbed Object Pos", setValue=var "Grabbed Object Pos", everyFrame=false)
    ActivateGameObject(gameObject=var "Parent Object", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckIsBlackThreaded(Target=var "Colliding Object", TrueEvent=(none), FalseEvent=(none), StoreValue=var "Enemy Is Black Threaded", EveryFrame=false)
    ConvertBoolToString(boolVariable=var "Enemy Is Black Threaded", stringVariable=var "Grab Event", falseString="GRAB MOSQUITO", trueString="GRAB MOSQUITO BLACK", everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Parent", fsm="Control"), sendEvent=var "Grab Event", delay=0, everyFrame=false)
    on TENDRIL RESET → Idle
  }

  state Hatchling {
    GetPosition2D(gameObject=var "Colliding Object", vector=var "Grabbed Object Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    GetParent(gameObject=var "Colliding Object", storeResult=var "Parent Object")
    SetFsmVector2(gameObject=var "Parent", fsmName="Control", variableName="Grabbed Object Pos", setValue=var "Grabbed Object Pos", everyFrame=false)
    RecycleObject(gameObject=var "Parent Object")
    SendEventByName(eventTarget=GameObjectFSM(var "Parent", fsm="Control"), sendEvent="GRAB HATCHLING", delay=0, everyFrame=false)
    on TENDRIL RESET → Idle
  }

  state Set Colliders {
    SetParent(gameObject=var "Tendril Collider Bottom", parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    GetPosition2d(gameObject=var "Tendril Collider Bottom", vector_2d=(unset), x=(unset), y=var "Collider Bot Y", space=1, everyFrame=false)
    FloatOperator(float1=var "Collider Bot Y", float2=-1, operation=Multiply, storeResult=var "Collider Height", everyFrame=false)
    FloatOperator(float1=var "Collider Bot Y", float2=0.5, operation=Multiply, storeResult=var "Mid Point Y", everyFrame=false)
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=(unset), y=var "Self Y", space=1, everyFrame=false)
    FloatOperator(float1=var "Mid Point Y", float2=var "Self Y", operation=Subtract, storeResult=var "Collider Offset", everyFrame=false)
    SetBoxCollider2DSize(gameObject=Self, width=0.3124999, height=var "Collider Height", offsetX=0, offsetY=var "Collider Offset")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Attack Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Frightened {
    on TENDRIL RESET → Idle
  }

  var Collider Bot Y: float = 0
  var Collider Height: float = 0
  var Collider Offset: float = 0
  var Collider Top Y: float = 0
  var Self Y: float = 0
  var Collider Height Half: float = 0
  var Mid Point Y: float = 0
  var Can Be Grabbed: bool = false
  var Enemy Is Black Threaded: bool = false
  var Custom Tag: string = ""
  var Grab Event: string = ""
  var Grabbed Object Pos: vector2 = (0, 0)
  var Grabbed Object Velocity: vector2 = (0, 0)
  var Parent: gameObject = Swamp Barnacle (1)
  var Colliding Object: gameObject = <null>
  var Parent Object: gameObject = <null>
  var Tendril Collider Top: gameObject = <null>
  var Tendril Collider Bottom: gameObject = Swamp Barnacle (1)/Tendril/Tendril Collider Bottom
  var Attack Detector: gameObject = Swamp Barnacle (1)/Tendril/Attack Detector
}
