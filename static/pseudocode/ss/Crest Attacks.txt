# Crest Attacks

// uses template: crest_attacks
fsm Crest Attacks {
  start Init
  on FSM CANCEL → Cancel  // from any state
  on HERO DAMAGED → Cancel  // from any state
  on CANCEL → Idle  // from any state
  on LEVEL LOADED → Cancel  // from any state
  on CREST ATTACK CANCEL → Cancel  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Attacks/Scythe/DownSlash New", storeResult=var "Rpr DownSlash")
    FindChild(gameObject=Self, childName="Attacks/Witch/DownSlash New", storeResult=var "Witch Downslash")
    FindChild(gameObject=Self, childName="Attacks/Witch/Lash Damager", storeResult=var "Lash Damager")
    FindChild(gameObject=Self, childName="Attacks/Witch/Followup Slash", storeResult=var "Followup Slash")
    FindChild(gameObject=Self, childName="Attacks/Witch/Lash Checker", storeResult=var "Lash Checker")
    FindChild(gameObject=Self, childName="Attacks/Witch/Tink Checker", storeResult=var "Tink Checker")
    FindChild(gameObject=Self, childName="Attacks/Warrior/SpinSlash", storeResult=var "SpinSlash")
    FindChild(gameObject=Self, childName="Attacks/Warrior/SpinSlashRage", storeResult=var "SpinSlashRage")
    FindChild(gameObject=Self, childName="Attacks/Warrior/SpinSlash Grind Dummy", storeResult=var "SpinSlash Grind Dummy")
    FindChild(gameObject=Self, childName="Attacks/Shaman/DownSlash", storeResult=var "Shaman Downslash")
    FindChild(gameObject=Self, childName="Attacks/Toolmaster/DownSlash New", storeResult=var "Toolmaster Downslash")
    FindChild(gameObject=Self, childName="Attacks/Toolmaster/DownSlash New Charged", storeResult=var "Toolmaster Downslash Charged")
    FindChild(gameObject=Self, childName="Attacks/Toolmaster/Drill Checker", storeResult=var "Drill Checker")
    FindChild(gameObject=Self, childName="Attacks/Toolmaster/DownSlash BouncePodDummy", storeResult=var "DownSlash BouncePodDummy")
    FindChild(gameObject=Self, childName="Effects/Downspike Burst", storeResult=var "Downspike Burst")
    FindNamedChild(gameObject=Self, storeResult=var "HeroBox")
    on FINISHED → Idle
  }

  state Idle {
    SetBoolValue(boolVariable=var "In Crest Attack", boolValue=false, everyFrame=false)
    on RPR DOWNSLASH → Rpr Downslash Antic
    on WITCH DOWNSLASH → Witch Downslash Antic
    on WARRIOR DOWNSLASH → SpinBall Antic
    on SHAMAN DOWNSLASH → Shaman Downslash Antic
    on TOOLMASTER DOWNSLASH → ToolM ChargeStart
  }

  state Cancel {
    BoolTest(boolVariable=var "In Crest Attack", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetBoolValue(boolVariable=var "In Crest Attack", boolValue=false, everyFrame=false)
    AudioStop(gameObject=var "Spawned Sound", fadeTime=0)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downAttacking", false], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downTravelling", false], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=var "Current SpinSlash", delivery=SendMessage, options=1, functionCall=CancelAttack())
    SendMessage(gameObject=var "Current SpinSlash", delivery=SendMessage, options=1, functionCall=FullCancelAttack())
    SendMessage(gameObject=var "Attack Target", delivery=SendMessage, options=1, functionCall=CancelLagHitsForSource(var "Witch Downslash"))
    SendMessage(gameObject=var "Current DrillSlash", delivery=SendMessage, options=1, functionCall=CancelAttack())
    SetGameObject(variable=var "Current SpinSlash", gameObject=<null>, everyFrame=false)
    ActivateGameObject(gameObject=var "Followup Slash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "HIdden Renderer", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetBoolValue(boolVariable=var "HIdden Renderer", boolValue=false, everyFrame=false)
    on FINISHED → Re-enable animation
    on CANCEL → Idle
  }

  state Rpr Downslash Antic {
    SetBoolValue(boolVariable=var "In Crest Attack", boolValue=true, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControlNotVelocity())
    HeroControllerMethods(method=StopAnimationControl, parameters=[], everyFrame=true, storeValue=(unused), isTrue=(none), isFalse=(none))
    SetBoolValue(boolVariable=var "Disabled Animation", boolValue=true, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Scythe Anim", clipName="v3 Down Slash Antic")
    Tk2dWatchAnimationEventsV3(gameObject=Self, eventInfo="", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED", sendTriggerEventIfAnimationChanged=false, sendCompleteEventIfAnimationChanged=false, animationInterruptedEvent=→"STOP")
    CallMethodProper(gameObject=var "HeroBox", behaviour="HeroBox", methodName="HeroBoxReaperDownSlash", parameters=[], storeResult=(unset var), EveryFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    DecelerateXY(gameObject=Self, decelerationX=(unset), decelerationY=0.8, brakeOnExit=false)  // disabled
    UmbrellaMotionX(gameObject=Self, accelerationFactor=3, decelerationFactor=1, maxSpeed=12)  // disabled
    on FINISHED → Rpr Downslash
    on STOP → Rpr Downslash Antic
  }

  state Rpr Downslash {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downAttacking", true], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Scythe Anim", clipName="v3 Down Slash")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SendMessage(gameObject=var "Rpr DownSlash", delivery=SendMessage, options=1, functionCall=StartSlash())
    DecelerateV2(gameObject=Self, deceleration=0.7, brakeOnExit=false)
    SetVelocityByScale(gameObject=Self, speed=-30, ySpeed=-30, everyFrame=false)
    on FINISHED → Rpr Downslash Fall
    on ATTACK LANDED → Rpr Downslash Bounce
    on BOUNCE TINKED → Rpr Downslash Bounce
    on THUNKED DOWN → Rpr Downslash Thunk
    on BOUNCE CANCEL → Rpr Downslash Bounce
    on LEAVING SCENE → Rpr Downslash End
  }

  state Rpr Downslash End {
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downAttacking", false], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="FinishDownspike", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on FINISHED → End
  }

  state End {
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downAttacking", false], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RegainControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=StartAnimationControlToIdle())
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downTravelling", false], storeResult=(unset var), EveryFrame=false)
    SetBoolValue(boolVariable=var "Disabled Animation", boolValue=false, everyFrame=false)
    SetGameObject(variable=var "Attack Target", gameObject=<null>, everyFrame=false)
    SetBoolValue(boolVariable=var "In Crest Attack", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Rpr Downslash Bounce {
    GetEventSender(sentByGameObject=var "Event Sender")
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetStartWithDownSpikeBounce", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downAttacking", false], storeResult=(unset var), EveryFrame=false)
    on FINISHED → End
  }

  state Rpr Downslash Thunk {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=28, everyFrame=false)  // disabled
    DecelerateV2(gameObject=Self, deceleration=0.715, brakeOnExit=false)
    on FINISHED → Rpr Downslash Fall
    on ATTACK LANDED → Rpr Downslash Bounce
    on BOUNCE TINKED → Rpr Downslash Bounce
    on BOUNCE CANCEL → Rpr Downslash Bounce
    on LEAVING SCENE → Rpr Downslash End
  }

  state Witch Downslash Antic {
    SetBoolValue(boolVariable=var "In Crest Attack", boolValue=true, everyFrame=false)
    SetGameObject(variable=var "Attack Target", gameObject=<null>, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControlNotVelocity())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControl())  // disabled
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SetBoolValue(boolVariable=var "Disabled Animation", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Witch Anim", clipName="DownSpike Antic")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.98, brakeOnExit=false)
    UmbrellaMotionXV2(gameObject=Self, accelerationFactor=3, decelerationFactor=0, maxSpeed=25, active=true)  // disabled
    SetFloatValue(floatVariable=var "Attack Target Y", floatValue=0, everyFrame=false)
    on FINISHED → Witch Downslash
  }

  state Witch Downslash {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downAttacking", true], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Witch Anim", clipName="DownSpike")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=var "Witch Downslash", delivery=SendMessage, options=1, functionCall=StartSlash())
    DecelerateV2(gameObject=Self, deceleration=0.78, brakeOnExit=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=var "Velocity Y", space=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Velocity Add", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Velocity Add", multiplyBy=-5, everyFrame=false)
    FloatAdd(floatVariable=var "Velocity X", add=var "Velocity Add", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Velocity Y", add=-5, everyFrame=false, perSecond=false)  // disabled
    FloatClamp(floatVariable=var "Velocity X", minValue=-25, maxValue=25, everyFrame=false)
    FloatClamp(floatVariable=var "Velocity Y", minValue=-12, maxValue=12, everyFrame=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=-10, everyFrame=false)
    SetVelocityByScale(gameObject=Self, speed=-5, ySpeed=-10, everyFrame=false)  // disabled
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    SendMessageDelay(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(true), delay=0.1)
    on FINISHED → Witch Downslash End
    on BOUNCE TINKED → Witch Tink
    on BOUNCE CANCEL → Witch Downslash Bounce
    on LEAVING SCENE → Witch Downslash End
    on HIT ATTACK TARGET → Set Rotation Offset
    on ATTACK LANDED → Set Rotation Offset
  }

  state Witch Downslash End {
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downAttacking", false], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="FinishDownspike", parameters=[true], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → End
  }

  state Witch Downslash Bounce {
    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Witch Anim", clipName="Downspike Followup")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="StartDownspikeInvulnerability", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetStartWithDownSpikeBounce", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → End
  }

  state Witch Lash To {
    Trigger2dEventV2(gameObject=var "Lash Damager", trigger=0, collideTag="", collideLayer=Enemies(11), sendEvent=→"LASH LANDED", storeCollider=var "")
    Trigger2dEventV2(gameObject=var "Lash Damager", trigger=1, collideTag="", collideLayer=Enemies(11), sendEvent=→"LASH LANDED", storeCollider=var "")
    Wait(time=0.08, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Lash Damager", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Witch Downslash End
    on LASH LANDED → Witch Step
  }

  state Revenge Antic {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControlNotVelocity())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Witch Anim", clipName="Revenge Slash")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))

    on FINISHED → Revenge Slash
  }

  state Revenge Slash {

    on FINISHED → End
  }

  state SpinBall Antic {
    SetBoolValue(boolVariable=var "In Crest Attack", boolValue=true, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControlNotVelocity())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SetBoolValue(boolVariable=var "Disabled Animation", boolValue=true, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Warrior Anim", clipName="SpinBall Antic")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocityByScale(gameObject=Self, speed=-10, ySpeed=5, everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=(unset), decelerationY=0.9, brakeOnExit=false)
    SetBoolValue(boolVariable=var "SpinBall Bouncing Up", boolValue=false, everyFrame=false)
    UmbrellaMotionX(gameObject=Self, accelerationFactor=1.5, decelerationFactor=0, maxSpeed=10)
    on FINISHED → SpinBall Launch
  }

  state SpinBall Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    UmbrellaMotionXV2(gameObject=Self, accelerationFactor=3, decelerationFactor=0, maxSpeed=25, active=var "Can Drift")
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=(unset), y=-90, vector3=(unset), everyFrame=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → SpinBall End
    on ATTACK LANDED → SpinBall Grind
    on LAND → SpinBall Land
    on BOUNCE TINKED → SpinBall Tink
    on BOUNCE CANCEL → SpinBall Bounce
    on LEAVING SCENE → SpinBall End
    on WALL L → SpinBall Wall L
    on WALL R → SpinBall Wall R
  }

  state SpinBall End {
    SendMessage(gameObject=var "Current SpinSlash", delivery=SendMessage, options=1, functionCall=CancelAttack())
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="FinishDownspike", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on FINISHED → End
  }

  state SpinBall Grind {
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="StartDownspikeInvulnerabilityLong", parameters=[], storeResult=(unset var), EveryFrame=false)

    Tk2dPlayAnimation(gameObject=Self, animLibName="Hornet CrestWeapon Warrior Anim", clipName=var "SpinBall Grind Anim")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    GetScale(gameObject=var "Current SpinSlash", vector=var "SpinSlash Scale", xScale=(unset), yScale=(unset), zScale=(unset), space=0, everyFrame=false)

    HeroGetWarriorCrestState(IsInRageMode=var "In Rage Mode")
    ConvertBoolToString(boolVariable=var "In Rage Mode", stringVariable=var "SpinBall Grind Anim", falseString="SpinBall Grind Effect", trueString="SpinBall Grind Rage Effect", everyFrame=false)

    on BOUNCE CANCEL → SpinBall Bounce
    on FINISHED → SpinBall Bounce
    on LEAVING SCENE → SpinBall End
  }

  state SpinBall Bounce {
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downTravelling", false], storeResult=(unset var), EveryFrame=false)
    BoolTest(boolVariable=var "SpinBall Bouncing Up", isTrue=→"NO BOUNCE", isFalse=(none), everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetStartWithDownSpikeBounce", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=var "Current SpinSlash", delivery=SendMessage, options=1, functionCall=CancelAttack())
    on FINISHED → End
    on NO BOUNCE → SpinBall Stop
  }

  state SpinBall Land {

    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["downTravelling", false], storeResult=(unset var), EveryFrame=false)

    HeroBoxControlV2(setOnExit=Off, heroBoxState=Normal)
    on FINISHED → End
  }
