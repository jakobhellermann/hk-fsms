fsm Detect {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Special Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"HIT", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="DESTROY", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  var Parent: gameObject = <null>
}
