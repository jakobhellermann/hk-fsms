fsm Grab Cooldown Timer {
  start Idle

  state Idle {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Friend Grab Cooldown", setValue=false, everyFrame=false)
    on HORNET GRABBED → Cooldown
  }

  state Cooldown {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Friend Grab Cooldown", setValue=true, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }
}
