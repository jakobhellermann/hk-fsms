# Control
# 1 location(s):
#   Memory_Needolin / Memory Weaver Scuttle (16)

fsm Control {
  start Pause

  state Init {
    SendRandomEvent(events=[→"HOOK", →"POINT", →"SPINE"], weights=[1, 1, 1], delay=0)
    on HOOK → Hook
    on POINT → Point
    on SPINE → Spine
  }

  state Hook {
    SetStringValue(stringVariable=var "Horns Idle Anim", stringValue="Horns Hook Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Horns Retreat Anim", stringValue="Horns Hook Retreat Antic", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    RandomInt(min=0, max=5, storeResult=var "Start Frame", inclusiveMax=true, noRepeat=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Scuttle Weaver Idle")
    Tk2dPlayFrame(gameObject=Self, frame=var "Start Frame")
    Tk2dPlayAnimation(gameObject=var "Horns", animLibName="", clipName=var "Horns Idle Anim")
    Tk2dPlayFrame(gameObject=var "Horns", frame=var "Start Frame")
    Trigger2dEvent(gameObject=var "Retreat Range", trigger=0, collideTag="", sendEvent=→"RETREAT", storeCollider=var "")
    on RETREAT → Retreat Antic
    on PROSTRATE RISE STAND → Retreat Pause
  }

  state Point {
    SetStringValue(stringVariable=var "Horns Idle Anim", stringValue="Horns Point Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Horns Retreat Anim", stringValue="Horns Point Retreat Antic", everyFrame=false)
    on FINISHED → Idle
  }

  state Spine {
    SetStringValue(stringVariable=var "Horns Idle Anim", stringValue="Horns Spine Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Horns Retreat Anim", stringValue="Horns Spine Retreat Antic", everyFrame=false)
    on FINISHED → Idle
  }

  state Retreat Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Scuttle Weaver Retreat Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimation(gameObject=var "Horns", animLibName="", clipName=var "Horns Retreat Anim")
    PlayAudioEventRandom(audioClips=[widow_insectoid_teleport_1 (sfxstatic_assets_areabell.bundle), widow_insectoid_teleport_2 (sfxstatic_assets_areabell.bundle), widow_insectoid_teleport_3 (sfxstatic_assets_areabell.bundle)], pitchMin=1, pitchMax=1.35, volume=0.25, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayRandomAudioClipTableV3(Table=Memory_Weaver_leave (sfxdynamic_assets_areacoralareamemory.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    on FINISHED → Retreat
  }

  state Retreat {
    ActivateGameObject(gameObject=var "Horns", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Scuttle Weaver Retreat")
    AccelerateToXByScale(gameObject=Self, accelerationFactor=1.5, targetSpeed=28)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEventRandom(audioClips=[widow_insectoid_walk_backwards_on_ground (sfxstatic_assets_areabell.bundle), widow_insectoid_walk_sideways_on_web (sfxstatic_assets_areabell.bundle)], pitchMin=0.85, pitchMax=1.35, volume=0.25, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Disappear
  }

  state Disappear {
    Tk2dSpriteTweenColor(gameObject=Self, color=(0, 0, 0, 0), duration=0.3)
    PlayParticleEmitter(gameObject=var "Pt Disappear", emit=0, resetIfPlaying=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Shuffle Z
  }

  state Shuffle Z {
    RandomFloat(min=-0.001, max=0.001, storeResult=var "Z Translate")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Translate", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Init
  }

  state Retreat Pause {
    WaitRandom(timeMin=0.3, timeMax=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Retreat Antic
  }

  var Z Translate: float = 0
  var Start Frame: int = 0
  var Horns Idle Anim: string = ""
  var Horns Retreat Anim: string = ""
  var Horns: gameObject = Memory Weaver Scuttle (16)/Horns
  var Retreat Range: gameObject = Memory Weaver Scuttle (16)/Retreat Range
  var Pt Disappear: gameObject = Memory Weaver Scuttle (16)/Pt Disappear
}
