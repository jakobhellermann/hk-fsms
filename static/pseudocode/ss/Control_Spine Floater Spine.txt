# Control
# 1 location(s):
#   localpoolprefabs_assets_areawilds.bundle / Spine Floater Spine

fsm Control {
  start Init

  state Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="S Start")
    SetGameObject(variable=var "Damaged By", gameObject=<null>, everyFrame=false)
    SetLayer(gameObject=Self, layer=Attack(17))
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Spark", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Hit Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHero(Target=Self, Enabled=false)
    on FINISHED → Idle
  }

  state Idle {
    on ANTIC → Antic
  }

  state Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="S Antic")
    Wait(time=0.75, finishEvent=→"FIRE", realTime=false)
    on FIRE → Fire
  }

  state Fire {
    ActivateGameObject(gameObject=var "Hit Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObjectDelay(gameObject=var "Hit Detector", activate=true, resetOnExit=false, delay=0.1)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SetDamageHero(Target=Self, Enabled=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="S Fly")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Angle", add=-90, everyFrame=false, perSecond=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=25, everyFrame=false)
    Wait(time=5, finishEvent=→"END", realTime=false)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"THUNK", storeCollider=var "", storeForce=(unset))
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"SMASH", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damaged By", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on THUNK → Thunk
    on END → Break
    on SMASH → Spawn Pickup
    on PROJECTILE BREAK → Spawn Pickup
    on HERO DAMAGED → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Thunk {
    PlayAudioEventRandom(audioClips=[spine_floater_projectile_impact_1 (sfxstatic_assets_shared.bundle), spine_floater_projectile_impact_2 (sfxstatic_assets_shared.bundle), spine_floater_projectile_impact_3 (sfxstatic_assets_shared.bundle), spine_floater_projectile_impact_4 (sfxstatic_assets_shared.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="S Impact")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=8, finishEvent=→"END", realTime=false)
    SetLayer(gameObject=Self, layer=Interactive Object(19))
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"SMASH", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on END → Break Antic
    on SMASH → Spawn Pickup
  }

  state Break Antic {
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"SMASH", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Break
    on SMASH → Spawn Pickup
  }

  state Break {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Spark", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Hit Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHero(Target=Self, Enabled=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEventRandom(audioClips=[understore_spanner_break (sfxstatic_assets_shared.bundle)], pitchMin=1.2, pitchMax=1.45, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Recycle
  }

  state Spawn Pickup {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    GetDamageEnemyAttackType(Target=var "Damaged By", StoreAttackType=var "Damaged By Type")
    EnumSwitch(enumVariable=var "Damaged By Type", compareTo=[AttackTypes(8), AttackTypes(17)], sendEvent=[→"FINISHED", →"FINISHED"], everyFrame=false)
    CheckQuestState(NotTrackedEvent=→"FINISHED", TrackedEvent=(none), CompletedEvent=→"FINISHED", Quest=Brolly Get (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    SpawnObjectFromGlobalPool(gameObject=Spine Pickup, spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Pickup")
    FlingObject(flungObject=var "Pickup", speedMin=15, speedMax=20, angleMin=68, angleMax=112)
    RandomFloat(min=0.003, max=0.0039, storeResult=var "Pickup Z Pos")
    SetPosition(gameObject=var "Pickup", vector=(unset), x=(unset), y=(unset), z=var "Pickup Z Pos", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Break
  }

  var Angle: float = 0
  var Pickup Z Pos: float = 0
  var Pt Shatter: gameObject = Spine Floater Spine/Pt Shatter
  var Self: gameObject = <null>
  var Spark: gameObject = Spine Floater Spine/Spark
  var Terrain Detector: gameObject = Spine Floater Spine/Terrain Detector
  var Thunk Effect: gameObject = Spine Floater Spine/Thunk Effect
  var Hit Detector: gameObject = Spine Floater Spine/Hit Detector
  var Pickup: gameObject = <null>
  var Damaged By: gameObject = <null>
  var Damaged By Type: enum = AttackTypes(0)
}
