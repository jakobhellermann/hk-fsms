// uses template: hornet_superjump
fsm Superjump {
  start Init
  on HERO DAMAGED → Cancel  // from any state
  on FSM CANCEL → Cancel  // from any state
  on SUPERJUMP HIT ROOF → Hit Roof Soft  // from any state
  on LEAVING SCENE → Leaving Scene  // from any state
  on PRE ENTER SUPERJUMPING → Pre Entered Jumping  // from any state

  state Inactive {
    on DO MOVE → Enough Silk?
  }

  state Regain Control To Idle {
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["superDashOnWall", false], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["superDashing", false], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=RegainControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=StartAnimationControlToIdle())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    on FINISHED → Reset Effects
  }

  state Cancel {
    RemoveUsingSilk(Amount=1, UsingType=Normal, DidAddTracker=var "Did Add Using Silk")
    ActivateGameObject(gameObject=var "Extra Ground Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Extra Throw Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thread", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStopV2(gameObject=var "Charge Audio", fadeTime=0.4, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Superjump Audio Loop", fadeTime=0, cancelOnEarlyExit=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["superDashing", false], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["freezeCharge", false], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Camera Target", behaviour="CameraTarget", methodName="SetSuperJump", parameters=[false, var "Stick Needle Y"], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Did Start Flash", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CancelFlashByID(Target=Self, ID=var "Sprite Flash ID")
    on FINISHED → Cancel Rumbling Focus
  }

  state Relinquish Control {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())  // disabled
    on FINISHED → Start Delay
  }

  state Ground Charge {

    ListenForSuperdash(eventTarget=Self, wasPressed=(none), wasReleased=→"BUTTON UP", isPressed=(none), isNotPressed=(none))

    on BUTTON UP → Charge Cancel Ground
    on WAIT → Ground Charged
  }

  }
