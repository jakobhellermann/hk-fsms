fsm Control {
  start Init

  state Init {
    on FINISHED → Rest
  }

  state Rest {
    on LOOK START → Start Dir
  }

  state Look L {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Angle", add=90, everyFrame=true, perSecond=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Target Angle", pause=0, everyFrame=true)
    GetDirectionBetweenAngles(startAngle=var "Self Angle", targetAngle=var "Target Angle", storeDirection=var "Clockwise", storeDistance=var "Angle Distance", everyFrame=true)
    BoolTest(boolVariable=var "Clockwise", isTrue=(none), isFalse=→"R", everyFrame=true)
    FloatAdd(floatVariable=var "Rotate Speed", add=600, everyFrame=true, perSecond=true)
    FloatOperator(float1=var "Angle Distance", float2=-4, operation=Multiply, storeResult=var "Angle Clamp Min", everyFrame=true)
    FloatOperator(float1=var "Angle Distance", float2=4, operation=Multiply, storeResult=var "Angle Clamp Max", everyFrame=true)
    FloatClamp(floatVariable=var "Angle Clamp Min", minValue=-9999, maxValue=-50, everyFrame=true)  // disabled
    FloatClamp(floatVariable=var "Angle Clamp Max", minValue=50, maxValue=9999, everyFrame=true)  // disabled
    FloatClamp(floatVariable=var "Rotate Speed", minValue=var "Angle Clamp Min", maxValue=var "Angle Clamp Max", everyFrame=true)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on LOOK STOP → Get Reset Angle
    on R → Look R
  }

  state Reset {
    AnimateRotationTo(gameObject=Self, fromValue=var "Self Angle", toValue=0, worldSpace=false, negativeSpace=false, time=0.35, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rest
  }

  state Start Dir {
    SetFloatValue(floatVariable=var "Rotate Speed", floatValue=0, everyFrame=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Self Angle", add=90, everyFrame=false, perSecond=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Target Angle", pause=0, everyFrame=false)
    FloatOperator(float1=360, float2=var "Self Angle", operation=Subtract, storeResult=var "Delta", everyFrame=false)  // disabled
    GetDirectionBetweenAngles(startAngle=var "Self Angle", targetAngle=var "Target Angle", storeDirection=var "Clockwise", storeDistance=0, everyFrame=false)
    BoolTest(boolVariable=var "Clockwise", isTrue=→"L", isFalse=→"R", everyFrame=false)
    on L → Look L
    on R → Look R
  }

  state Look R {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Angle", add=90, everyFrame=true, perSecond=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Target Angle", pause=0, everyFrame=true)
    GetDirectionBetweenAngles(startAngle=var "Self Angle", targetAngle=var "Target Angle", storeDirection=var "Clockwise", storeDistance=var "Angle Distance", everyFrame=true)
    BoolTest(boolVariable=var "Clockwise", isTrue=→"L", isFalse=(none), everyFrame=true)
    FloatAdd(floatVariable=var "Rotate Speed", add=-600, everyFrame=true, perSecond=true)
    FloatOperator(float1=var "Angle Distance", float2=-4, operation=Multiply, storeResult=var "Angle Clamp Min", everyFrame=true)
    FloatOperator(float1=var "Angle Distance", float2=4, operation=Multiply, storeResult=var "Angle Clamp Max", everyFrame=true)
    FloatClamp(floatVariable=var "Angle Clamp Min", minValue=-9999, maxValue=-50, everyFrame=true)  // disabled
    FloatClamp(floatVariable=var "Angle Clamp Max", minValue=50, maxValue=9999, everyFrame=true)  // disabled
    FloatClamp(floatVariable=var "Rotate Speed", minValue=var "Angle Clamp Min", maxValue=var "Angle Clamp Max", everyFrame=true)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on LOOK STOP → Get Reset Angle
    on L → Look L
  }

  state Get Reset Angle {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=1, everyFrame=false)
    FloatCompare(float1=var "Self Angle", float2=180, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatAdd(floatVariable=var "Self Angle", add=-360, everyFrame=false, perSecond=false)
    on FINISHED → Reset
  }

  var Angle: float = 0
  var Angle Clamp Max: float = 0
  var Angle Clamp Min: float = 0
  var Angle Distance: float = 0

}
