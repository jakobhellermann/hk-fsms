# Call Out
# 1 location(s):
#   Coral_24 / Flea Rescue Generic

fsm Call Out {
  start Init
  on RESCUE → Save  // from any state
  on BREAK → Save  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check State
  }

  state Save {
    SavedItemGetV2(Item=var "Quest Target", Amount=1, ShowPopup=false)
    ActivateGameObject(gameObject=var "Flea Distressed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flea Rescue Activation", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on NEXT → 
  }

  state Check State {
    CheckQuestPdSceneBool(QuestTarget=var "Quest Target", ExpectedValue=true, trueEvent=→"ACTIVATE", falseEvent=(none), storeValue=(unset))
    on ACTIVATE → Rescued
    on FINISHED → Idle
  }

  state Rescued {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
  }

  var Flea Distressed: gameObject = Flea Rescue Generic/Flea Distressed
  var Flea Rescue Activation: gameObject = Flea Rescue Generic/Flea Rescue Activation
  var Quest Target: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
