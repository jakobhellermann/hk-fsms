// uses template: corpse_chunk_flinger
fsm corpse_chunk_flinger {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fling
  }

  state Fling {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    FloatOperator(float1=var "Angle", float2=-35, operation=Add, storeResult=var "Angle Min", everyFrame=false)
    FloatOperator(float1=var "Angle", float2=35, operation=Add, storeResult=var "Angle Max", everyFrame=false)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=false)
    FloatOperator(float1=var "Speed", float2=0.8, operation=Multiply, storeResult=var "Speed Min", everyFrame=false)
    FloatOperator(float1=var "Speed", float2=1.2, operation=Multiply, storeResult=var "Speed Max", everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Chunks")
    SetTransformParent(gameObject=var "Chunks", parent=<null>, worldPositionStays=true)
    FlingObjects(containerObject=var "Chunks", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=var "Speed Min", speedMax=var "Speed Max", angleMin=var "Angle Min", angleMax=var "Angle Max", space=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
  }

  var Angle: float = 0
  var Speed: float = 0
  var Angle Min: float = 0
  var Angle Max: float = 0
  var Speed Min: float = 0
  var Speed Max: float = 0
  var Chunks: gameObject = <null>
}
