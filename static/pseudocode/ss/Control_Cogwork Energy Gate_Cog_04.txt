# Control
# 2 location(s):
#   Cog_04 / Cogwork Energy Gate
#   Cog_08 / Cogwork Energy Gate

fsm Control {
  start Init
  on ACTIVATE → Activate  // from any state
  on COMPLETE → Complete  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Gate Sequence 1")
    FindNamedChild(gameObject=Self, storeResult=var "Gate Sequence 2")
    FindNamedChild(gameObject=Self, storeResult=var "Active Object")
    FindChild(gameObject=var "Gate Sequence 1", childName="Hero Crusher", storeResult=var "Hero Crusher 2")
    on FINISHED → Idle
  }

  state Idle {
    SetPosition(gameObject=var "Gate Mid", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Gate FG", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Gate BG", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Gate Sequence 1", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on START → Operate Gate
  }

  state Activate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Operate Gate {
    FloatOperator(float1=var "Energy", float2=var "Division Factor", operation=Divide, storeResult=var "Gate Mid Y", everyFrame=false)
    FloatOperator(float1=var "Energy", float2=var "Division Factor", operation=Divide, storeResult=var "Gate BG Y", everyFrame=false)
    FloatOperator(float1=var "Energy", float2=var "Division Factor", operation=Divide, storeResult=var "Gate FG Y", everyFrame=false)
    AnimateYPositionTo(GameObject=var "Gate Mid", ToValue=var "Gate Mid Y", localSpace=true, time=0.7, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    AnimateYPositionTo(GameObject=var "Gate FG", ToValue=var "Gate FG Y", localSpace=true, time=0.4, speed=(unset), delay=(unset), easeType=19, reverse=false, finishEvent=(none), realTime=false)
    AnimateYPositionTo(GameObject=var "Gate BG", ToValue=var "Gate BG Y", localSpace=true, time=0.4, speed=(unset), delay=(unset), easeType=10, reverse=false, finishEvent=(none), realTime=false)
    FloatOperator(float1=var "Gate Mid Y", float2=-1, operation=Multiply, storeResult=var "Gate Low Y", everyFrame=false)
    AnimateYPositionTo(GameObject=var "Gate Sequence 1", ToValue=var "Gate Low Y", localSpace=true, time=0.7, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    on START → Operate Gate
    on DROP → Drop
  }

  state Drop {
    FloatOperator(float1=var "Energy", float2=var "Division Factor", operation=Divide, storeResult=var "Gate Mid Y", everyFrame=true)
    FloatOperator(float1=var "Energy", float2=var "Division Factor", operation=Divide, storeResult=var "Gate BG Y", everyFrame=true)
    FloatOperator(float1=var "Energy", float2=var "Division Factor", operation=Divide, storeResult=var "Gate FG Y", everyFrame=true)
    FloatOperator(float1=var "Gate Mid Y", float2=-1, operation=Multiply, storeResult=var "Gate Low Y", everyFrame=true)
    SetPosition(gameObject=var "Gate Mid", vector=(unset), x=(unset), y=var "Gate Mid Y", z=(unset), space=1, everyFrame=true, lateUpdate=false)
    SetPosition(gameObject=var "Gate FG", vector=(unset), x=(unset), y=var "Gate FG Y", z=(unset), space=1, everyFrame=true, lateUpdate=false)
    SetPosition(gameObject=var "Gate BG", vector=(unset), x=(unset), y=var "Gate BG Y", z=(unset), space=1, everyFrame=true, lateUpdate=false)
    SetPosition(gameObject=var "Gate Sequence 1", vector=(unset), x=(unset), y=var "Gate Low Y", z=(unset), space=1, everyFrame=true, lateUpdate=false)
    FloatCompare(float1=var "Energy", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    ActivateGameObject(gameObject=var "Hero Crusher", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Hero Crusher 2", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Idle
    on START → Operate Gate
  }

  state Complete {
    ActivateGameObject(gameObject=var "Active Object", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AnimateYPositionTo(GameObject=var "Gate Mid", ToValue=6, localSpace=true, time=0.7, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    AnimateYPositionTo(GameObject=var "Gate FG", ToValue=6, localSpace=true, time=0.4, speed=(unset), delay=(unset), easeType=19, reverse=false, finishEvent=(none), realTime=false)
    AnimateYPositionTo(GameObject=var "Gate BG", ToValue=6, localSpace=true, time=0.5, speed=(unset), delay=(unset), easeType=1, reverse=false, finishEvent=(none), realTime=false)
    AnimateYPositionTo(GameObject=var "Gate Sequence 1", ToValue=-6, localSpace=true, time=0.5, speed=(unset), delay=(unset), easeType=1, reverse=false, finishEvent=(none), realTime=false)
  }

  var Energy: float = 0
  var Gate Mid Y: float = 0
  var Gate BG Y: float = 0
  var Gate FG Y: float = 0
  var Division Factor: float = 90
  var Gate Low Y: float = 0
  var Terrain Collider: gameObject = Cogwork Energy Gate/Terrain Collider
  var Gate Mid: gameObject = Cogwork Energy Gate/Gate Mid
  var Gate FG: gameObject = Cogwork Energy Gate/Gate FG
  var Gate BG: gameObject = Cogwork Energy Gate/Gate BG
  var Hero Crusher: gameObject = Cogwork Energy Gate/Gate Mid/Hero Crusher
  var Gate Sequence 2: gameObject = <null>
  var Gate Sequence 1: gameObject = <null>
  var Hero Crusher 2: gameObject = <null>
  var Active Object: gameObject = <null>
}
