# Control
# 1 location(s):
#   Cog_07 / Black Thread States/Normal World/Repairable Scene

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Repairing Scene")
    ActivateGameObject(gameObject=var "Repairing Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="cog7_automatonDestroyed", isTrue=→"DESTROYED", isFalse=(none))
    PlayerDataBoolTest(boolName="cog7_automatonRepairing", isTrue=→"REPAIRING", isFalse=(none))
    on REPAIRING → Activate Repairing Scene
    on DESTROYED → Destroyed
  }

  state Activate Repairing Scene {
    ActivateGameObject(gameObject=var "Repairing Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on LEAVING SCENE → Check Repair
  }

  state Check Repair {
    FindNamedChild(gameObject=var "Repairing Scene", storeResult=var "Song Automaton Fly Repair 1")
    FindNamedChild(gameObject=var "Repairing Scene", storeResult=var "Song Automaton Fly Repair 2")
    GameObjectIsNull(gameObject=var "Song Automaton Fly Repair 1", isNull=→"FAIL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GameObjectIsNull(gameObject=var "Song Automaton Fly Repair 2", isNull=→"FAIL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetFsmBool(gameObject=var "Song Automaton Fly Repair", fsmName="Control", variableName="Alerted", storeValue=var "Fly Alerted", everyFrame=false)
    BoolTest(boolVariable=var "Fly Alerted", isTrue=→"FAIL", isFalse=(none), everyFrame=false)
    GetFsmBool(gameObject=var "Song Automaton Fly Repair 2", fsmName="Control", variableName="Alerted", storeValue=var "Fly Alerted", everyFrame=false)
    BoolTest(boolVariable=var "Fly Alerted", isTrue=→"FAIL", isFalse=→"REPAIR COMPLETE", everyFrame=false)
    on FAIL → Repair Fail
    on REPAIR COMPLETE → Complete
  }

  state Repair Fail {
    SetPlayerDataBool(boolName="cog7_automatonRepairingComplete", value=false)
  }

  state Complete {
    SetPlayerDataBool(boolName="cog7_automatonRepairingComplete", value=true)
  }

  state Destroyed {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Automaton Defeated: bool = false
  var Automaton 2 Defeated: bool = false
  var Automaton Shield Defeated: bool = false
  var Fly Alerted: bool = false
  var Repairing Scene: gameObject = <null>
  var Song Automaton Fly Repair: gameObject = <null>
  var Song Automaton Fly Repair 2: gameObject = <null>
  var Song Automaton Fly Repair 1: gameObject = <null>
}
