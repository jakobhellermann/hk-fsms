# Phase Control
# 1 location(s):
#   Cradle_03 / Boss Scene/Silk Boss

fsm Phase Control {
  start Init
  on MID STAGGER → Mid Stagger  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Rubble Fields")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Spike Floors")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Death Sequence")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Death Sequence Cursed")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Enviro Region NoEffect")
    FindNamedChild(gameObject=var "Rubble Fields", storeResult=var "Rubble Field L")
    FindNamedChild(gameObject=var "Rubble Fields", storeResult=var "Rubble Field M")
    FindNamedChild(gameObject=var "Rubble Fields", storeResult=var "Rubble Field R")
    on FINISHED → Set P1
  }

  state Set P1 {
    SetFsmFloat(gameObject=Self, fsmName="Attack Control", variableName="Attack Time", setValue=0.8, everyFrame=false)
    SetHP(target=Self, hp=var "P1 HP")
    on FINISHED → P1
  }

  state P1 {
    on TOOK DAMAGE → HP Check 1
  }

  state HP Check 1 {
    CheckIsInEditor(trueEvent=→"TEST END", falseEvent=(none), storeValue=(unset))
    CompareHP(enemy=var "Self", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P1
    on NEXT → Set P2
  }

  state Set P2 {
    SetFsmFloat(gameObject=Self, fsmName="Attack Control", variableName="Attack Time", setValue=0.5, everyFrame=false)
    AddHP(target=Self, amount=var "P2 HP", healToMax=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Can Web Strand Attack", setValue=true, everyFrame=false)
    on FINISHED → P2
  }

  state HP Check 2 {
    CompareHP(enemy=var "Self", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P2
    on NEXT → Set P3
  }

  state Set P3 {
    SetFsmFloat(gameObject=Self, fsmName="Attack Control", variableName="Attack Time", setValue=0, everyFrame=false)
    AddHP(target=Self, amount=var "P3 HP", healToMax=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Do Phase Roar", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Can Claw Combo", setValue=true, everyFrame=false)
    on ROARING → Set P3 Web Strand
  }

  state P2 {
    on TOOK DAMAGE → HP Check 2
  }

  state P3 {
    on TOOK DAMAGE → HP Check 3
  }

  state Set P3 Web Strand {
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Double Web Strand", setValue=true, everyFrame=false)
    on FINISHED → P3
  }

  state HP Check 3 {
    CompareHP(enemy=var "Self", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P3
    on NEXT → Stagger Hit
  }

  state Stagger Hit {
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL RESET", delay=0, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    PlayAudioEventRandom(audioClips=[silk_short_scream_A (sfxstatic_assets_areacradle.bundle), silk_short_scream_B (sfxstatic_assets_areacradle.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stagger Hit")
    CreateObject(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SendEventToRegister(eventName="ATTACK CLEAR")
    SendEventToRegister(eventName="SILK STAGGERED")
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="MOVE STOP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Attack Control"), sendEvent="ATTACK STOP", delay=0, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0.25)
    on FINISHED → Stagger Fall
  }

  state Stagger Fall {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    PlayAudioEventRandom(audioClips=[silk_rise_up_close (sfxstatic_assets_areacradle.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stagger Fall")
    AccelerateToY(gameObject=Self, accelerationFactor=1, targetSpeed=-30)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stagger Pause
  }

  state Stagger Pause {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → BG Break Sequence
  }

  state BG Break Sequence {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rubble M
  }

  state Rubble M {
    SendEventByName(eventTarget=GameObject(var "Rubble Field M"), sendEvent="ATTACK LONG", delay=0, everyFrame=false)
    Wait(time=6.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rubble Sides
  }

  state Rubble Sides {
    SendEventByName(eventTarget=GameObject(var "Rubble Fields"), sendEvent="ATTACK", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Rubble Field L"), sendEvent="ATTACK LONG", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Rubble Field R"), sendEvent="ATTACK LONG", delay=0, everyFrame=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set P4
  }

  state Set P4 {
    SetLayer(gameObject=Self, layer=Enemies(11))
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Can Rubble Attack", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="RERISE", delay=0, everyFrame=false)
    AddHP(target=Self, amount=var "P4 HP", healToMax=false)
    SetFsmFloat(gameObject=Self, fsmName="Attack Control", variableName="Attack Time", setValue=0.5, everyFrame=false)
    on FINISHED → P4
  }

  state P4 {
    on TOOK DAMAGE → HP Check 4
  }

  state HP Check 4 {
    CompareHP(enemy=var "Self", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P4
    on NEXT → Set P5
  }

  state Set P5 {
    AddHP(target=Self, amount=var "P5 HP", healToMax=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Can Spike Pull", setValue=true, everyFrame=false)
    on FINISHED → P5
  }

  state P5 {
    on TOOK DAMAGE → HP Check 5
  }

  state HP Check 5 {
    CompareHP(enemy=var "Self", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P5
    on NEXT → Set P6
  }

  state Set P6 {
    SetFsmFloat(gameObject=Self, fsmName="Attack Control", variableName="Attack Time", setValue=0.1, everyFrame=false)
    AddHP(target=Self, amount=var "P6 HP", healToMax=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Do Phase Roar", setValue=true, everyFrame=false)
    on FINISHED → P6
  }

  state P6 {
    on TOOK DAMAGE → HP Check 6
  }

  state HP Check 6 {
    CompareHP(enemy=var "Self", integer2=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P6
    on NEXT → Death Hit
  }

  state Death Hit {
    SetPlayerDataBool(boolName="disableInventory", value=true)
    RecordJournalKillV2(Record=Silk Boss (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle), ShowPopup=false)
    PlayAudioEvent(audioClip=silk_short_scream_A (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Enviro Region NoEffect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Hit")
    CreateObject(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SendEventToRegister(eventName="ATTACK CLEAR")
    SendEventToRegister(eventName="SILK STAGGERED")
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="MOVE STOP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Attack Control"), sendEvent="ATTACK STOP", delay=0, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetEnviroRegionProperties(Target=Self, EnviroType=NoEffect, Priority=10)
    CallStaticMethod(className="DeliveryQuestItem", methodName="BreakAll", parameters=[], storeResult=(unset var), everyFrame=false)
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    SendEventByName(eventTarget=GameObject(var "Physics Hair"), sendEvent="DEATH PT1", delay=0, everyFrame=false)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0, resetOnStateExit=false)
    on FINISHED → Death Type
  }

  state Start Death Sequence {
    SendEventByName(eventTarget=GameObject(var "Death Sequence"), sendEvent="DEATH START", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Disappear
  }

  state Disappear {
    SendEventToRegister(eventName="CLEAR EFFECTS")
    SetMeshRenderer(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
  }

  state Death Type {
    CheckIfCrestEquipped(Crest=Cursed (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"CURSED", falseEvent=(none), storeValue=(unset))
    on FINISHED → Start Death Sequence
    on CURSED → Start Death Sequence Curse
  }

  state Start Death Sequence Curse {
    SendEventByName(eventTarget=GameObject(var "Death Sequence Cursed"), sendEvent="DEATH START", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Disappear
  }

  state Mid Stagger {
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Double Web Strand", setValue=true, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Attack Control", variableName="Attack Time", setValue=0, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Can Claw Combo", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Control", variableName="Can Web Strand Attack", setValue=true, everyFrame=false)
    on FINISHED → Stagger Hit
  }

  var P1 HP: int = 100
  var P2 HP: int = 192
  var P3 HP: int = 240
  var P4 HP: int = 110
  var P5 HP: int = 242
  var P6 HP: int = 340
  var Physics Hair: gameObject = Boss Scene/Silk_Hair

  var Self: gameObject = Boss Scene/Silk Boss
  var Boss Scene: gameObject = <null>
  var Rubble Field L: gameObject = <null>
  var Rubble Field M: gameObject = <null>
  var Rubble Field R: gameObject = <null>
  var Rubble Fields: gameObject = <null>
  var Spike Floors: gameObject = <null>
  var Death Sequence: gameObject = <null>
  var Enviro Region NoEffect: gameObject = <null>
  var Death Sequence Cursed: gameObject = <null>
}
