# Shake Control
# 2 location(s):
#   Dust_11 / Steel Soul States/Regular/NPC Control/silk_farm_centre
#   Dust_11 / Steel Soul States/Regular/NPC Control/silk_farm_lvl2

// uses template: silk_farm_shake
fsm Shake Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Shake")
    on FINISHED → Idle
  }

  state Idle {
    on SHAKE START → Shake
    on TURN → Turn Shake
  }

  state Shake {
    SendMessageV2(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartJitter(), everyFrame=false)
    PlayParticleEmitterInState(gameObject=var "Pt Shake")
    on SHAKE END → End
  }

  state End {
    SendMessageV2(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StopJitter(), everyFrame=false)
    on FINISHED → Idle
  }

  state Turn Shake {
    SendMessageV2(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartJitter(), everyFrame=false)
    PlayParticleEmitterInState(gameObject=var "Pt Shake")  // disabled
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  var Pt Shake: gameObject = <null>
}
