fsm Hero Turn Sfx {
  start Idle

  state Idle {
    on HERO SIT START → Frame Check
  }

  state Frame Check {
    Tk2dSpriteGetId(gameObject=var "Hero", spriteID=var "Sprite ID", everyframe=true)
    IntCompare(integer1=var "Sprite ID", integer2=103, equal=→"PLAY SFX", lessThan=(none), greaterThan=(none), everyFrame=true)
    IntCompare(integer1=var "Sprite ID", integer2=107, equal=→"PLAY SFX", lessThan=(none), greaterThan=(none), everyFrame=true)
    on HERO SIT END → Idle
    on PLAY SFX → Play SFX
  }

  state Play SFX {
    PlayRandomAudioClipTable(Table=hornet_run_start (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Wait(time=0.15, finishEvent=(none), realTime=false)
    on HERO SIT END → Idle
    on FINISHED → Frame Check
  }

  var Sprite ID: int = 0
}
