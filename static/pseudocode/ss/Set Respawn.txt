# Set Respawn
# 1 location(s):
#   Coral_Judge_Arena / Caravan_Set/Caravan

fsm Set Respawn {
  start Wait

  state Wait {
    on CARAVAN TRAVEL START → Check Left Behind
  }

  state Left Behind {
    SetDeathRespawnMarker(Target=door_caravanTravelEnd/Death Respawn Marker)
  }

  state Check Left Behind {
    GetFsmBool(gameObject=Self, fsmName="Quest End Sequence", variableName="Do Travel", storeValue=var "Do Travel", everyFrame=false)
    BoolTest(boolVariable=var "Do Travel", isTrue=(none), isFalse=→"LEFT BEHIND", everyFrame=false)
    on LEFT BEHIND → Left Behind
  }

  var Do Travel: bool = false
}
