# Control
# 1 location(s):
#   Hang_14 / Handmaiden Scene/Hanging Handmaidens/Song Handmaiden (4)/Rest Pivot

fsm Control {
  start Pause

  state Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choose
  }

  state Choose {
    SendRandomEvent(events=[→"L", →"R"], weights=[0.5, 0.5], delay=0)
    on L → Start L
    on R → Start R
  }

  state Rotate L {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start", space=0, everyFrame=false)  // disabled
    FloatAdd(floatVariable=var "Start", add=-360, everyFrame=false, perSecond=false)  // disabled
    RandomFloat(min=0.9, max=1.1, storeResult=var "Start")
    RandomFloat(min=2.75, max=3.5, storeResult=var "Time")
    AnimateRotationTo(gameObject=Self, fromValue=var "Start", toValue=var "End", worldSpace=true, negativeSpace=false, time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)  // disabled
    EaseFloat(fromValue=var "End", toValue=var "Start", floatVariable=var "Tween", time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tween", space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → Audio R
  }

  state Rotate R {
    RandomFloat(min=-0.9, max=-1.1, storeResult=var "End")
    RandomFloat(min=2.75, max=3.5, storeResult=var "Time")
    EaseFloat(fromValue=var "Start", toValue=var "End", floatVariable=var "Tween", time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tween", space=0, everyFrame=true, lateUpdate=false)
    AnimateRotationTo(gameObject=Self, fromValue=var "Start", toValue=var "End", worldSpace=true, negativeSpace=false, time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Audio L
  }

  state Audio R {
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.5, 0.5], eventMax=[1, 2], missedMax=[2, 1], activeBool=(unset))
    PlayAudioEventRandom(audioClips=[hanging_by_thread_01 (sfxstatic_assets_silkshared.bundle), hanging_by_thread_02 (sfxstatic_assets_silkshared.bundle), hanging_by_thread_03 (sfxstatic_assets_silkshared.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Rotate R
  }

  state Audio L {
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.5, 0.5], eventMax=[1, 2], missedMax=[2, 1], activeBool=(unset))
    PlayAudioEventRandom(audioClips=[hanging_by_thread_01 (sfxstatic_assets_silkshared.bundle), hanging_by_thread_02 (sfxstatic_assets_silkshared.bundle), hanging_by_thread_03 (sfxstatic_assets_silkshared.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Rotate L
  }

  state Start R {
    on FINISHED → Rotate R
  }

  state Start L {
    on FINISHED → Rotate L
  }

  var Start: float = 0
  var End: float = 0
  var Time: float = 0
  var Tween: float = 0
  var Handmaiden: gameObject = Handmaiden Scene/Hanging Handmaidens/Song Handmaiden (4)
}
