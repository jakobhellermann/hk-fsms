fsm Hit Hero {
  start Idle

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"HIT", storeCollider=var "")
    on HIT → Hit
  }

  state Hit {
    SendEventByName(eventTarget=GameObject(var "Phantom"), sendEvent="HIT HERO", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Phantom: gameObject = Boss Scene/Phantom
}
