# Detect Silk Dash
# 1 location(s):
#   Crawl_03b / Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm:0

fsm Detect Silk Dash {
  start Idle

  state Idle {
    on SILK DASH HIT → Stagger
  }

  state Stagger {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVE STOP", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stagger")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AddHP(target=Self, amount=9999, healToMax=false)
    on FINISHED → Death
  }

  state Death {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Silk Dash Death", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="ZERO HP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Damager Name: string = ""
  var Damager: gameObject = <null>
  var Self: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm:0
}
