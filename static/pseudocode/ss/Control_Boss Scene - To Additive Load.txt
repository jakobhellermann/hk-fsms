# Control
# 1 location(s):
#   Hang_17b / Boss Scene - To Additive Load

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Gates")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Boss")
    on FINISHED → Idle
  }

  state Idle {
    on ARENA START → Arena Start
    on BATTLE START → Skip Arena
  }

  state Battle Start {
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=true, bossTitle="SONG_KNIGHT")  // disabled
    ApplyMusicCue(musicCue=Grand Forum Battle (audiocuesdynamic_assets_areahangarealibrary.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.1)
    on ARENA END → Battle End
  }

  sequence state Battle End {
    ActivateGameObject(gameObject=var "CamLock Boss", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Arena Start {
    ActivateGameObject(gameObject=var "CamLock Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on BATTLE START → Battle Start
  }

  state Skip Arena {
    ActivateGameObject(gameObject=var "CamLock Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on FINISHED → Battle Start
  }

  var Gates: gameObject = <null>
  var CamLock Boss: gameObject = <null>
}
