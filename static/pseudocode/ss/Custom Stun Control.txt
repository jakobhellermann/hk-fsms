# Custom Stun Control
# 1 location(s):
#   Ward_02_boss / Boss Scene/Conductor Boss

fsm Custom Stun Control {
  start Init

  state Init {
    GetHP(target=Self, storeValue=var "HP")
    MultiplyIntByFloat(integer=var "HP", multiplyFloat=0.75, storeResult=var "P2 HP", everyFrame=false, forceRoundUp=false)
    MultiplyIntByFloat(integer=var "HP", multiplyFloat=0.4, storeResult=var "P3 HP", everyFrame=false, forceRoundUp=false)
    on FINISHED → P1
  }

  state P1 {
    on TOOK DAMAGE → P1 Stun?
  }

  state P1 Stun? {
    CompareHP(enemy=var "Self", integer2=var "P2 HP", equal=→"STUN", lessThan=→"STUN", greaterThan=(none), everyFrame=false)
    on FINISHED → P1
    on STUN → P1 Stun
  }

  state P1 Stun {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="STUN", delay=0, everyFrame=false)
    ApplyMusicCue(musicCue=Creepy Boss Main (audiocuesdynamic_assets_areabellareacoralareashellwoodareaward.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.1)
    on STUN END → P2
  }

  state P2 {
    on TOOK DAMAGE → P2 Stun?
  }

  state P2 Stun? {
    CompareHP(enemy=var "Self", integer2=var "P3 HP", equal=→"STUN", lessThan=→"STUN", greaterThan=(none), everyFrame=false)
    on FINISHED → P2
    on STUN → P2 Stun
  }

  state P2 Stun {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="STUN", delay=0, everyFrame=false)
    on STUN END → P3
  }

  state P3 {
    on TOOK DAMAGE → Death?
  }

  state Death? {
    CompareHP(enemy=var "Self", integer2=0, equal=→"DIE", lessThan=→"DIE", greaterThan=(none), everyFrame=false)
    on FINISHED → P3
    on DIE → Die
  }

  state Die {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="DIE", delay=0, everyFrame=false)
  }

  var HP: int = 0
  var P2 HP: int = 0
  var P3 HP: int = 0
  var Self: gameObject = Boss Scene/Conductor Boss
}
