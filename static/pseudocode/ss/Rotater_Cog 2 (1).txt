fsm Rotater {
  start Init

  state Init {
    FindGameObject(objectName="Chain Plat Lever", withTag="Untagged", store=var "Gate")  // disabled
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=var "Init Angle", zAngle=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Gate", vector=(unset), x=(unset), y=var "Gate Y", z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Gate Y", float2=var "Init Angle", operation=Add, storeResult=var "Rotate Angle", everyFrame=false)
    FloatMultiply(floatVariable=var "Rotate Angle", multiplyBy=var "Multiplier", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Angle", space=0, everyFrame=false, lateUpdate=false)
    on START → Get Ref
  }

  state Rotate {
    GetPosition(gameObject=var "Gate", vector=(unset), x=(unset), y=var "Gate Y", z=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Gate Y", float2=var "Init Angle", operation=Add, storeResult=var "Rotate Angle", everyFrame=true)
    FloatMultiply(floatVariable=var "Rotate Angle", multiplyBy=var "Multiplier", everyFrame=true)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Angle", space=0, everyFrame=true, lateUpdate=false)
  }

  state Get Ref {
    FindGameObject(objectName="Chain Plat Lever", withTag="Untagged", store=var "Gate")  // disabled
    on FINISHED → Rotate
  }

  var Multiplier: float = 25
  var Gate: gameObject = <null>

  var Gate Y: float = 0
  var Init Angle: float = 0
  var Rotate Angle: float = 0
}
