# Control
# 1 location(s):
#   Slab_10b / Boss Scene/Blasts/First Weaver Blast (2)

fsm Control {
  start Init

  state Idle {
    ActivateGameObject(gameObject=var "Blast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BLAST → Wait
  }

  state Wait {
    WaitRandom(timeMin=0, timeMax=0.65, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Pos X
  }

  state Set Pos X {
    RandomFloat(min=0.9, max=1.2, storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=var "X Min", max=var "X Max", storeResult=var "Spawn X")
    BoolTest(boolVariable=var "Low", isTrue=→"LOW", isFalse=→"HIGH", everyFrame=false)
    on LOW → Pos Low
    on HIGH → Pos High
  }

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=(unset), z=(unset), space=0, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Blast Pos 1")
    FindNamedChild(gameObject=Self, storeResult=var "Blast Pos 2")
    FindNamedChild(gameObject=Self, storeResult=var "Blast Pos 3")
    FloatOperator(float1=var "Start X", float2=-1.5, operation=Add, storeResult=var "X Min", everyFrame=false)
    FloatOperator(float1=var "Start X", float2=1.5, operation=Add, storeResult=var "X Max", everyFrame=false)
    ActivateGameObject(gameObject=var "Blast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Blast {
    ActivateGameObject(gameObject=var "Blast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.95, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Sound
  }

  state End {
    ActivateGameObject(gameObject=var "Blast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolFlip(boolVariable=var "Low")
    on FINISHED → Idle
  }

  state Sound {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Blast", audioClip=explosion_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
    on BLAST → Quick Reblast
  }

  state Pos Low {
    RandomFloat(min=11.11, max=12.11, storeResult=var "Spawn Y")
    SetPosition(gameObject=Self, vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Multiblast
  }

  state Pos High {
    RandomFloat(min=15, max=17.5, storeResult=var "Spawn Y")
    SetPosition(gameObject=Self, vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Multiblast
  }

  state Quick Reblast {
    ActivateGameObject(gameObject=var "Blast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait
  }

  state Multiblast {
    SetRandomRotation(gameObject=Self, x=(unset), y=(unset), z=true)
    RandomVector3(Start=(-1, -1, 0), End=(1, 1, 0), StoreResult=var "Shift", EveryFrame=false)
    SpawnObjectFromGlobalPool(gameObject=First Weaver Bomb Blast (localpoolprefabs_assets_areaslab.bundle), spawnPoint=var "Blast Pos 1", position=var "Shift", rotation=(0, 0, 0), storeObject=var "Next Blast")
    SetFsmFloat(gameObject=var "Next Blast", fsmName="Control", variableName="Wait Time", setValue=0, everyFrame=false)
    RandomVector3(Start=(-1, -1, 0), End=(1, 1, 0), StoreResult=var "Shift", EveryFrame=false)
    SpawnObjectFromGlobalPool(gameObject=First Weaver Bomb Blast (localpoolprefabs_assets_areaslab.bundle), spawnPoint=var "Blast Pos 2", position=var "Shift", rotation=(0, 0, 0), storeObject=var "Next Blast")
    SetFsmFloat(gameObject=var "Next Blast", fsmName="Control", variableName="Wait Time", setValue=0.2, everyFrame=false)
    RandomVector3(Start=(-1, -1, 0), End=(1, 1, 0), StoreResult=var "Shift", EveryFrame=false)
    SpawnObjectFromGlobalPool(gameObject=First Weaver Bomb Blast (localpoolprefabs_assets_areaslab.bundle), spawnPoint=var "Blast Pos 3", position=var "Shift", rotation=(0, 0, 0), storeObject=var "Next Blast")
    SetFsmFloat(gameObject=var "Next Blast", fsmName="Control", variableName="Wait Time", setValue=0.4, everyFrame=false)
    on FINISHED → End
  }

  var Low: bool = true

  var Spawn X: float = 0
  var Spawn Y: float = 0
  var Start X: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var Scale: float = 0
  var Shift: vector3 = (0, 0, 0)
  var Blast: gameObject = <null>
  var Blast Pos 1: gameObject = Boss Scene/Blasts/First Weaver Blast (2)/Blast Pos 1
  var Blast Pos 2: gameObject = Boss Scene/Blasts/First Weaver Blast (2)/Blast Pos 2
  var Blast Pos 3: gameObject = Boss Scene/Blasts/First Weaver Blast (2)/Blast Pos 3
  var Next Blast: gameObject = <null>
}
