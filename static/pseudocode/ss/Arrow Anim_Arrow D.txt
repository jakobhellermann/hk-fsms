# Arrow Anim
# 1 location(s):
#   shopui_assets_all.bundle / Shop Menu/Item List Group/Arrow D

fsm Arrow Anim {
  start Init

  state Init {
    SetVector3Value(vector3Variable=var "Move Pos", vector3Value=var "Init Pos", everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Move Pos", addX=0, addY=-0.15, addZ=0, everyFrame=false, perSecond=false)
    on FINISHED → Idle
  }

  state Idle {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Init Pos", time=0.1, delay=0, speed=(unset), space=1, easeType=13, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=false)
    on MOVE → Up
  }

  state Up {
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Move Pos", time=0.05, delay=0, speed=(unset), space=1, easeType=21, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Idle
    on MOVE → Repeat
  }

  state Repeat {
    on FINISHED → Up
  }

  var Init Pos: vector3 = (-6.83, -0.92, -1)
  var Move Pos: vector3 = (-6.83, -0.92, -1)
}
