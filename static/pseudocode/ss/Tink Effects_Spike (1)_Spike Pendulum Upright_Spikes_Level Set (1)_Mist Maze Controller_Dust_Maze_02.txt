// component disabled: this FSM does not run until something enables it
fsm Tink Effects {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Plat Dust")
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    on TINK UP → Tink Up
    on TINK RIGHT → Tink R
    on TINK LEFT → Tink L
    on TINK DOWN R → Tink R
    on TINK DOWN L → Tink L
  }

  state Tink Shake {
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    ObjectJitterLocal(gameObject=Self, x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=0)
    on FINISHED → Idle
  }

  state Tink Up {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=Self, vector=(unset), x=0.3, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=true)
    on FINISHED → Tink Shake
  }

  state Tink R {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)  // disabled
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.3, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)  // disabled
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=true)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK R", delay=0, everyFrame=false)
    on FINISHED → Tink Shake
  }

  state Tink L {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)  // disabled
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.3, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)  // disabled
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=true)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK L", delay=0, everyFrame=false)
    on FINISHED → Tink Shake
  }

  var Event: string = ""
  var Start Pos: vector3 = (0, 0, 0)
  var Plat Dust: gameObject = <null>
  var Parent: gameObject = <null>
}
