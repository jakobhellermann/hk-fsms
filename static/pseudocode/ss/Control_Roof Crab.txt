# Control
# 1 location(s):

fsm Control {
  start Init
  on ZERO HP → Death  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindAlertRange(target=Self, storeResult=var "Unalert Range", childName="Unalert Range")
    SetParent(gameObject=var "Unalert Range Obj", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Idle X", y=(unset), space=0, everyFrame=false)
    on FINISHED → Dormant
  }

  state Dormant {
    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=→"WAKE", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)

    on WAKE → Chase
  }

  state Chase {

    on ATTACK → Attack Antic
    on ANGRY → Angry
  }

  state Shooting {
    PlayAudioEvent(audioClip=roof_crab_vomit_attack (sfxstatic_assets_areacrawl.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Current Clip")

    PlayParticleEmitterChildren(gameObject=var "Pt Spit Current", resetTimeIfPlaying=false, stopOnStateExit=true)

    ParticleSystemPlay(gameObject=var "Pt ChargeDust", withChildren=false, cacheChildren=false, stopOnExit=false, destroyOnFinish=false)
    on FINISHED → Shoot End Audio
  }

  state Shoot End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=(unset), brakeOnExit=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingSmall", setValue=false, everyFrame=false)
    on FINISHED → Shoot Recover
  }

  state Shoot Recover {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Idle")
    ChaseObjectGround(gameObject=Self, target=var "Hero", xOffset=0, speedMax=20, acceleration=1, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=2, snapTo=false, snapSpeedOnly=false, onlyOnStateEntry=false)

    on SING → Sing Antic
  }

  state Attack Antic {

    AudioStopV2(gameObject=var "Audio Loop RunVoice", fadeTime=0, cancelOnEarlyExit=false)
    CallMethodProper(gameObject=Self, behaviour="BlackThreadState", methodName="CheckIsBlackThreaded", parameters=[], storeResult=var "Is Black Threaded", EveryFrame=false)
    BoolTestToGameObject(Test=var "Is Black Threaded", ExpectedValue=true, TrueGameObject=var "Pt BlackSpit Antic", FalseGameObject=var "Pt Spit Antic", StoreResult=var "Pt Spit Current", EveryFrame=false)
    PlayAudioEvent(audioClip=roof_crab_vomit_antic (sfxstatic_assets_areacrawl.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Current Clip")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=(unset), brakeOnExit=false)
    PlayParticleEmitterInState(gameObject=var "Pt Spit Current")
    Tk2dPlayAnimation(gameObject=var "Body", animLibName="", clipName="Body Move")
    AudioStopV2(gameObject=Self, fadeTime=0.15, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Audio Loop Walk", fadeTime=0.15, cancelOnEarlyExit=false)
    on FINISHED → Shooting
  }

  state Angry {
    ChaseObjectGround(gameObject=Self, target=var "Hero", xOffset=0, speedMax=20, acceleration=1, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=2, snapTo=false, snapSpeedOnly=false, onlyOnStateEntry=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Angry Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=→"UNALERT", OutOfRangeDelay=0, everyFrame=true)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.1)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=var "Performance Radius", MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ATTACK → Sing Catch
    on SING → Sing Antic
    on UNALERT → Attack Antic
  }

  state Return To Idle? {
    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=→"UNALERT", OutOfRangeDelay=0, everyFrame=false)
    on FINISHED → Chase
    on UNALERT → Return Dir
  }

  state Return Dir {
    CheckXPosition(gameObject=Self, compareTo=var "Idle X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"R", lessThanBool=(unset), greaterThan=→"L", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    on L → Return L
    on R → Return R
  }

  state Return L {
    AccelerateToX(gameObject=Self, accelerationFactor=1, targetSpeed=-20)
    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=→"WAKE", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    CheckXPosition(gameObject=Self, compareTo=var "Idle X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"FINISHED", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.1)
    on WAKE → Chase
    on TOOK DAMAGE → Chase
    on FINISHED → To Dormant
  }

  state Return R {
    AccelerateToX(gameObject=Self, accelerationFactor=1, targetSpeed=20)
    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=→"WAKE", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    CheckXPosition(gameObject=Self, compareTo=var "Idle X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"FINISHED", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))

    on WAKE → Chase
    on TOOK DAMAGE → Chase
    on FINISHED → To Dormant
  }

  state Death {
    SetParent(gameObject=var "Pt ChargeDust", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Pt Spit", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Pt BlackSpit", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)

  }

  state Sing Antic {
    AudioStopV2(gameObject=var "Audio Loop RunVoice", fadeTime=0, cancelOnEarlyExit=false)
    PlayerDataBoolTest(boolName="roofCrabEncountered", isTrue=(none), isFalse=→"ROAR")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    AudioStopV2(gameObject=Self, fadeTime=0.15, cancelOnEarlyExit=false)
    StopParticleEmitter(gameObject=var "Pt ChargeDust")
    ActivateGameObject(gameObject=var "Audio Loop RunVoice", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Sing
    on ROAR → Roar Antic
  }

  state Sing {
    AudioPlaySimpleV2(gameObject=var "Audio Loop Walk", volume=1, startTime=0, oneShotClip=<null>)

    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.15)
    on CANCEL → Shoot End
    on SING DURATION END → Shoot End
  }

  state Sing Catch {
    CheckHeroPerformanceRegionV2(Target=Self, Radius=var "Performance Radius", MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)

    on FINISHED → Meet?
  }

    on ROAR → Roar Antic
    on FINISHED → Attack Antic
  }

    ActivateGameObject(gameObject=var "Audio Loop RunVoice", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roar Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    StopParticleEmitter(gameObject=var "Pt ChargeDust")
    AudioStopV2(gameObject=var "Audio Loop RunVoice", fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → Roar
  }

  state Roar {
    SetPlayerDataBool(boolName="roofCrabEncountered", value=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Sing")

  }

  }
