# Control
# 1 location(s):
#   Shadow_20 / Collectable Mandrake Scene/Witch_Cluster_Vine (1)

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on ACTIVATED → Activated
    on FINISHED → Idle
  }

  state Activated {
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="BROKEN", delay=0, everyFrame=false)
    SetParent(gameObject=var "Stumps", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Stumps", activate=var "Leave Stumps", recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Nail Attack", sendEvent=→"NAIL HIT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"NAIL HIT", sendEventHeavy=→"DESTROY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on NAIL HIT → Strike
    on DESTROY → Heavy Break
  }

  state Strike {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Hit Cancel Range", storeResult=false, sendEvent="CANCEL", outOfRangeEvent="", everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Hit", resetTimeIfPlaying=false, stopOnStateExit=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayAudioEventRandom(audioClips=[shellwood_twig_wall_hit_1 (sfxstatic_assets_areashellwood.bundle), shellwood_twig_wall_hit_2 (sfxstatic_assets_areashellwood.bundle), shellwood_twig_wall_hit_3 (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    IntSwitch(intVariable=var "Hits", compareTo=[1, 2, 3, 4], sendEvent=[→"1", →"2", →"3", →"4"], everyFrame=false)
    on 1 → Hit 1
    on 2 → Hit 2
    on 3 → Hit 3
    on 4 → PD Bool
    on FINISHED → PD Bool
    on CANCEL → Idle
  }

  state Hit 2 {
    ActivateGameObject(gameObject=var "Thick", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayFrame(gameObject=var "Standard", frame=0)
    Tk2dPlayFrame(gameObject=var "Standard 2", frame=0)
    Tk2dPlayFrame(gameObject=var "Twirl", frame=0)
    on FINISHED → Recover
  }

  state Recover {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit 3 {
    ActivateGameObject(gameObject=var "Standard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayFrame(gameObject=var "Standard 2", frame=0)
    Tk2dPlayFrame(gameObject=var "Twirl", frame=0)
    on FINISHED → Recover
  }

  state Hit 1 {
    Tk2dPlayFrame(gameObject=var "Thick", frame=0)
    Tk2dPlayFrame(gameObject=var "Standard", frame=0)
    Tk2dPlayFrame(gameObject=var "Standard 2", frame=0)
    Tk2dPlayFrame(gameObject=var "Twirl", frame=0)
    on FINISHED → Recover
  }

  state Hit 4 {
    ActivateGameObject(gameObject=var "Standard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Standard 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Twirl", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thick", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Destroy Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Attached Object", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="BREAK", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Hero Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Stumps", activate=var "Leave Stumps", recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Extra Vines", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Mask", ToAlpha=0, FadeTime=0.2)
    PlayAudioEvent(audioClip=shellwood_twig_wall_break (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
  }

  state PD Bool {
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "PD Bool", value=true)
    on FINISHED → Hit 4
  }

  state Heavy Break {
    SetIntValue(intVariable=var "Hits", intValue=4, everyFrame=false)
    on FINISHED → Strike
  }

  var Leave Stumps: bool = true
  var PD Bool: string = ""
  var Attached Object: gameObject = <null>
  var Event Receiver: gameObject = Collectable Mandrake Scene/Crying Audio Control

  var Hits: int = 0
  var Activated: bool = false
  var Destroy Effects: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Destroy Effects
  var Extra Vines: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Extra Vines
  var Hero Blocker: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Hero Blocker
  var Mask: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Mask
  var Self: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)
  var Standard: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Standard
  var Standard 2: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Standard 2
  var Stumps: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Stumps
  var Thick: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Thick
  var Twirl: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Twirl
  var Pt Hit: gameObject = Collectable Mandrake Scene/Witch_Cluster_Vine (1)/Pt Hit
}
