# Attack Control
# 1 location(s):
#   Cradle_03 / Boss Scene/Silk Boss

fsm Attack Control {
  start Init
  on PREPARE DASH → Prepare Dash  // from any state
  on HORNET CAUGHT → Catch  // from any state
  on STUN → Uncatch Hornet?  // from any state
  on ATTACK STOP → Uncatch Hornet?  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Strand Patterns")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Beast Slash Fader")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "beast_slash_sequence")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "beast_slash_sequence_fast")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Web Strand Catch Effect")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Rubble Fields")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Spike Floors")
    FindNamedChild(gameObject=var "Rubble Fields", storeResult=var "Rubble Field L")
    FindNamedChild(gameObject=var "Rubble Fields", storeResult=var "Rubble Field M")
    FindNamedChild(gameObject=var "Rubble Fields", storeResult=var "Rubble Field R")
    on FINISHED → Dormant
  }

  state Idle {
    Wait(time=var "Attack Time", finishEvent=→"ATTACK", realTime=false)
    on ATTACK → Rubble Attack?
  }

  state Hand Ptn Choice {
    SendEvent(eventTarget=Self, sendEvent=→"STOMP SWIPE STOMP", delay=0, everyFrame=false)  // disabled
    SendRandomEventV4(events=[→"PINCER STOMP", →"STOMP SWIPE", →"SWIPE STOMP", →"PINCER SWIPE", →"SWIPE STOMP SWIPE", →"STOMP SWIPE STOMP"], weights=[1, 1, 1, 1, 1, 1], eventMax=[1, 1, 1, 1, 1, 1], missedMax=[7, 7, 7, 7, 3, 3], activeBool=var "Can Claw Combo")
    SendRandomEventV4(events=[→"PINCER STOMP", →"STOMP SWIPE", →"SWIPE STOMP", →"PINCER SWIPE"], weights=[1, 1, 1, 1], eventMax=[1, 1, 1, 1], missedMax=[5, 5, 5, 5], activeBool=(unset))
    on STOMP SWIPE → Stomp Swipe
    on SWIPE STOMP → Swipe Stomp
    on SWIPE CLAW → Swipe Claw
    on CLAW SWIPE → Claw Swipe
    on PINCER CLAW → Pincer Claw
    on PINCER STOMP → Pincer Stomp
    on PINCER SWIPE → Pincer Swipe
    on SWIPE STOMP SWIPE → Swp Stomp Swp
    on STOMP SWIPE STOMP → Stomp Swp Stomp
  }

  state Stomp Swipe {
    SendEventByName(eventTarget=GameObject(var "Primary Hand"), sendEvent="STOMP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="PREPARE", delay=0, everyFrame=false)
    RandomFloat(min=0.5, max=1, storeResult=var "Attack Delay")
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="SWIPE", delay=var "Attack Delay", everyFrame=false)
    on FINISHED → Wait For Hands Ready
  }

  state Set Primary Hand {
    SetBoolValue(boolVariable=var "Has Done Hand Attack", boolValue=true, everyFrame=false)
    SetGameObject(variable=var "Primary Hand", gameObject=var "Hand L", everyFrame=false)
    SetGameObject(variable=var "Secondary Hand", gameObject=var "Hand R", everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"FINISHED", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    SetGameObject(variable=var "Secondary Hand", gameObject=var "Hand L", everyFrame=false)
    SetGameObject(variable=var "Primary Hand", gameObject=var "Hand R", everyFrame=false)
    on FINISHED → Hand Ptn Choice
  }

  state Wait For Hands Ready {
    GetFsmBool(gameObject=var "Hand L", fsmName="Control", variableName="Ready", storeValue=var "Ready Hand L", everyFrame=true)
    GetFsmBool(gameObject=var "Hand R", fsmName="Control", variableName="Ready", storeValue=var "Ready Hand R", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Ready Hand L", var "Ready Hand R"], sendEvent=→"FINISHED", storeResult=(unset), everyFrame=true)
    on FINISHED → Idle
  }

  state Swipe Stomp {
    SendEventByName(eventTarget=GameObject(var "Primary Hand"), sendEvent="SWIPE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="PREPARE", delay=0, everyFrame=false)
    RandomFloat(min=0.25, max=1, storeResult=var "Attack Delay")
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="STOMP", delay=var "Attack Delay", everyFrame=false)
    on FINISHED → Wait For Hands Ready
  }

  state Swipe Claw {
    SendEventByName(eventTarget=GameObject(var "Primary Hand"), sendEvent="SWIPE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="PREPARE", delay=0, everyFrame=false)
    RandomFloat(min=0.75, max=1.2, storeResult=var "Attack Delay")
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="CLAW", delay=var "Attack Delay", everyFrame=false)
    on FINISHED → Wait For Hands Ready
  }

  state Claw Swipe {
    SendEventByName(eventTarget=GameObject(var "Primary Hand"), sendEvent="CLAW", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="PREPARE", delay=0, everyFrame=false)
    RandomFloat(min=0.75, max=1.2, storeResult=var "Attack Delay")
    SendEventByName(eventTarget=GameObject(var "Secondary Hand"), sendEvent="SWIPE", delay=var "Attack Delay", everyFrame=false)
    on FINISHED → Wait For Hands Ready
  }

  state Pincer Claw {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=33, upperValue=48, boolVariable=(unset), trueEvent=(none), falseEvent=→"RANGE OUT", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand R"), sendEvent="CLAW FAR L", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand L"), sendEvent="CLAW FAR R", delay=0, everyFrame=false)
    on FINISHED → Wait For Hands Ready
    on RANGE OUT → Hand Ptn Choice
  }

  state Pincer Stomp {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=35, upperValue=46, boolVariable=(unset), trueEvent=(none), falseEvent=→"RANGE OUT", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand L"), sendEvent="STOMP FAR L", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand R"), sendEvent="STOMP FAR R", delay=0, everyFrame=false)
    on FINISHED → Wait For Hands Ready
    on RANGE OUT → Hand Ptn Choice
  }

  state Pincer Swipe {
    SendEvent(eventTarget=Self, sendEvent=→"HIGH", delay=0, everyFrame=false)
    SendRandomEvent(events=[→"HIGH", →"LOW"], weights=[1, 1], delay=0)  // disabled
    on HIGH → Pincer Swipe H
    on LOW → 
  }

  state Pincer Swipe H {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=37, upperValue=44, boolVariable=(unset), trueEvent=(none), falseEvent=→"RANGE OUT", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand L"), sendEvent="PINCER HIGH R", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand R"), sendEvent="PINCER HIGH L", delay=0, everyFrame=false)
    on FINISHED → Wait For Hands Ready
    on RANGE OUT → Hand Ptn Choice
  }

  state Prepare Dash {
    GetFsmBool(gameObject=var "Hand L", fsmName="Control", variableName="Ready", storeValue=var "Ready Hand L", everyFrame=true)
    GetFsmBool(gameObject=var "Hand R", fsmName="Control", variableName="Ready", storeValue=var "Ready Hand R", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Ready Hand L", var "Ready Hand R"], sendEvent=→"FINISHED", storeResult=(unset), everyFrame=true)
    on FINISHED → Dash Ready
  }

  state Dash Ready {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="READY", delay=0, everyFrame=false)
    on DASH ATTACK → Dash Attack Dir
  }

  state Dash Attack Dir {
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"L", negativeEvent=→"R", space=0)
    on L → Dash L
    on R → Dash R
  }

  state Dash L {
    SendEventByName(eventTarget=GameObject(var "Hand L"), sendEvent="STOMP DASH L", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand R"), sendEvent="STOMP DASH L", delay=0, everyFrame=false)
    on FINISHED → Wait For Hands Ready
  }

  state Dash R {
    SendEventByName(eventTarget=GameObject(var "Hand L"), sendEvent="STOMP DASH R", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hand R"), sendEvent="STOMP DASH R", delay=0, everyFrame=false)
    on FINISHED → Wait For Hands Ready
  }

  state Attack Choice {
    BoolTest(boolVariable=var "Has Done Hand Attack", isTrue=(none), isFalse=→"HAND ATTACK", everyFrame=false)
    BoolTestMulti(boolVariables=[var "Can Web Strand Attack", var "Did Web Strand Attack"], boolStates=[true, false], trueEvent=→"WEB ATTACK", falseEvent=(none), storeResult=(unset), everyFrame=false)
    SendRandomEventV4(events=[→"HAND ATTACK", →"DASH ATTACK", →"WEB ATTACK"], weights=[0.6, 0.25, 0.15], eventMax=[3, 1, 1], missedMax=[3, 4, 5], activeBool=var "Can Web Strand Attack")
    SendRandomEventV4(events=[→"HAND ATTACK", →"DASH ATTACK"], weights=[0.65, 0.35], eventMax=[2, 1], missedMax=[1, 2], activeBool=(unset))
    SendRandomEventV4(events=[→"HAND ATTACK", →"WEB ATTACK"], weights=[0.8, 0.2], eventMax=[4, 1], missedMax=[1, 4], activeBool=(unset))  // disabled
    on WEB ATTACK → Web Prepare
    on HAND ATTACK → Set Primary Hand
    on DASH ATTACK → DashAttack Prepare
  }

  state Web Prepare {
    SetBoolValue(boolVariable=var "Did Web Strand Attack", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Attack Prepare", setValue=true, everyFrame=false)
    on ATTACK PREPARED → Web Cast
  }

  state Web Cast {
    PlayAudioEventRandom(audioClips=[silk_short_scream_A (sfxstatic_assets_areacradle.bundle), silk_short_scream_B (sfxstatic_assets_areacradle.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Physics Hair"), sendEvent="ROAR", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Cast", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Web Lift
  }

  state Activate Strands {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL STOP", delay=0, everyFrame=false)
    GetRandomChild(gameObject=var "Strand Patterns", storeResult=var "Web Pattern")
    SendEventByName(eventTarget=GameObject(var "Web Pattern"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Double Web?
  }

  state Web Recover {
    SendEventByName(eventTarget=GameObject(var "Physics Hair"), sendEvent="IDLE", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetVelocity2d(gameObject=Self, vector=(0, -7), x=(unset), y=(unset), everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    on FINISHED → Move Restart
  }

  state Web Lift {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=8, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.92, brakeOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    CheckYPositionV2(gameObject=Self, compareTo=140.6, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=(none), greaterThanBool=var "At Y Max", everyFrame=true, space=0, activeBool=(unset))
    SetVelocity2dBool(gameObject=Self, activeBool=var "At Y Max", vector=(unset), x=(unset), y=0, everyFrame=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL STOP", delay=0, everyFrame=false)
    on FINISHED → Activate Strands
  }

  state Catch {
    SetBoolValue(boolVariable=var "Hornet Caught", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="HERO GRAB BREAK", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL STOP", delay=0, everyFrame=false)
    PlayRandomAudioClipTable(Table=hornet_wound_heavy (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayAudioEvent(audioClip=grab_damage (sfxstatic_assets_areamemory.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=silk_heart_get_pt2_1st_flash (sfxstatic_assets_silkshared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=silk_prop_kill (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventToRegister(eventName="ATTACK CLEAR")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    HeroControllerMethods(method=CancelAttack, parameters=[true], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AffectedByGravity", parameters=[false], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetPlayerDataBool(boolName="isInvincible", value=true)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SetPositionToObject2D(gameObject=var "Web Strand Catch Effect", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    MatchScaleSign(Target=var "Web Strand Catch Effect", MatchTo=var "Hero", EveryFrame=false)
    MatchScaleSign(Target=var "beast_slash_sequence", MatchTo=var "Hero", EveryFrame=false)
    ActivateGameObject(gameObject=var "Web Strand Catch Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fade Over
  }

  state Fade Over {
    EaseSpriteColor(target=var "Beast Slash Fader", fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), disableSpriteRendererIfTransparent=false, time=0.15, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shift Hero
  }

  state Shift Hero {
    GetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Y", storeValue=var "Hero Y", everyFrame=false)
    FloatAdd(floatVariable=var "Hero Y", add=-1.5, everyFrame=false, perSecond=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Hero X", minValue=33, maxValue=47, everyFrame=false)
    SetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Web Strand Catch Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0, resetOnStateExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SendEventToRegister(eventName="BLADES AWAY")
    SendEventToRegister(eventName="WEB AWAY")
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="BEAST SLASH", delay=0, everyFrame=false)
    on FINISHED → Do Slash
  }

  state Do Slash {
    BoolTest(boolVariable=var "Did Beast Slash", isTrue=→"QUICK", isFalse=(none), everyFrame=false)  // disabled
    SetBoolValueAtTime(BoolVariable=var "Did Beast Slash", BoolValue=true, Time=0, SetOppositeOnStateEntry=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    SetPositionToObject2D(gameObject=var "beast_slash_sequence", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    EaseSpriteColor(target=var "Beast Slash Fader", fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.15, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "beast_slash_sequence", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slash End
    on QUICK → Quick Slash
  }

  state Slash End {
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SetBoolValue(boolVariable=var "Hornet Caught", boolValue=false, everyFrame=false)
    PlayRandomAudioClipTable(Table=hornet_wound_heavy (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVE START", delay=0, everyFrame=false)
    SendEventToRegister(eventName="BLADES RETURN")
    SetPlayerDataBool(boolName="isInvincible", value=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0, resetOnStateExit=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL START", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on TOOK DAMAGE → Idle
  }

  state Move Restart {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVE START", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL START", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Quick Slash {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    SetPositionToObject2D(gameObject=var "beast_slash_sequence_fast", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    EaseSpriteColor(target=var "Beast Slash Fader", fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.15, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "beast_slash_sequence_fast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1.9, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slash End
  }

  state Attack Stop {
    ActivateGameObject(gameObject=var "beast_slash_sequence", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "beast_slash_sequence_fast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "DashSlash Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteColor(gameObject=var "Beast Slash Fader", color=(1, 1, 1, 0), red=(unset), green=(unset), blue=(unset), alpha=(unset), resetOnExit=false, everyFrame=false)
    on STUN END → Idle
    on ATTACK START → Idle
  }

  state Rubble Attack? {
    CheckHeroPerformanceRegionV2(Target=Self, Radius=8, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=true, ActiveBool=var "Can Sing React", StoreState=None, EveryFrame=true)
    BoolTest(boolVariable=var "Do Phase Roar", isTrue=→"ROAR", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Can Rubble Attack", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    BoolTestMulti(boolVariables=[var "Can Spike Pull", var "Did Spike Pull"], boolStates=[true, false], trueEvent=→"SPIKE LIFT", falseEvent=(none), storeResult=(unset), everyFrame=false)
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.5, 0.5], eventMax=[1, 1], missedMax=[1, 1], activeBool=(unset))
    BoolTest(boolVariable=var "Can Spike Pull", isTrue=(none), isFalse=→"RUBBLE PULL", everyFrame=false)
    SendRandomEventV4(events=[→"RUBBLE PULL", →"SPIKE LIFT"], weights=[0.5, 0.5], eventMax=[1, 2], missedMax=[2, 1], activeBool=(unset))
    on RUBBLE PULL → Rubble Pull Pos
    on FINISHED → Attack Choice
    on SPIKE LIFT → Spike Lift Type
    on ROAR → Roar Prepare
    on SING → Sing Prepare
  }

  state Rubble Pull Pos {
    CheckXPosition(gameObject=var "Hero", compareTo=40, compareToOffset=0, tolerance=0, equal=→"HERO L", equalBool=(unset), lessThan=→"HERO L", lessThanBool=(unset), greaterThan=→"HERO R", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    on HERO L → Rubble Hero L
    on HERO R → Rubble Hero R
  }

  state Rubble Pull Recover {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Primary Hand
  }

  state Rubble Hero L {
    SendRandomEventV4(events=[→"RUBBLE L", →"RUBBLE M", →"RUBBLE PINCER"], weights=[0.25, 0.5, 0.25], eventMax=[1, 1, 1], missedMax=[2, 2, 2], activeBool=(unset))
    on RUBBLE L → Rubble L
    on RUBBLE M → Rubble M
    on RUBBLE PINCER → Rubble P
  }

  state Rubble Hero R {
    SendRandomEventV4(events=[→"RUBBLE R", →"RUBBLE M", →"RUBBLE PINCER"], weights=[0.25, 0.5, 0.25], eventMax=[1, 1, 1], missedMax=[2, 2, 2], activeBool=(unset))
    on RUBBLE R → Rubble R
    on RUBBLE M → Rubble M
    on RUBBLE PINCER → Rubble P
  }

  state Rubble L {
    SendEventByName(eventTarget=GameObject(var "Rubble Field L"), sendEvent="ATTACK", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Rubble Fields"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Rubble Pull Recover
