# FSM
# 1 location(s):
#   Demo End / Demo End Sequence

fsm FSM {
  start Pause

  state Fade In {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=3)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait
  }

  state Wait {
    Wait(time=7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → To Menu
  }

  state To Menu {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="ReturnToMainMenuNoSave", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  state Pause {
    SendEventByNameV2(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    SetInputHandlerSkipMode(SkipMode=NOT_SKIPPABLE)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fade In
  }
}
