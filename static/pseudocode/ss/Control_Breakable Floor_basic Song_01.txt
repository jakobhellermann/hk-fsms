# Control
# 2 location(s):
#   Song_01 / Breakable Floor_basic Song_01
#   Tube_Hub / Breakable Floor_basic

// uses template: breakable_floor
fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "Bouncer")
    FindNamedChild(gameObject=Self, storeResult=var "Break Effects")
    FindNamedChild(gameObject=Self, storeResult=var "CamLocks")
    FindNamedChild(gameObject=Self, storeResult=var "Hero Detect")
    FindNamedChild(gameObject=Self, storeResult=var "Jitter Bits")
    FindNamedChild(gameObject=Self, storeResult=var "Masks")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Hit")
    FindNamedChild(gameObject=Self, storeResult=var "Terrain Box")
    FindNamedChild(gameObject=var "Jitter Bits", storeResult=var "Jitter Floor")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → PD Activation
  }

  state Activated? {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATED → Activated
  }

  state Activated {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEvent(gameObject=var "Hero Detect", trigger=0, collideTag="", sendEvent=→"STEP", storeCollider=var "")
    on HIT → Check
    on STEP → Step On
  }

  state Check {
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=→"HIT", everyFrame=false)
    on HIT → Hit
    on BREAK → Set PD
  }

  state Hit {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=var "Hit Sfx", pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    AnimatePositionBy(gameObject=var "Jitter Bits", shiftBy=(0, -0.2, 0), time=0.1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    AnimatePositionBy(gameObject=var "Jitter Bits", shiftBy=(0, 0.2, 0), time=0.1, speed=(unset), delay=0.1, easeType=12, reverse=false, finishEvent=(none), realTime=false)
    ObjectJitterLocal(gameObject=var "Jitter Floor", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=0)
    SetParticleEmissionChildren(Target=var "Pt Hit", SetEmission=true, EveryFrame=false, stopOnExit=true)
    on FINISHED → Hit End
  }

  state Hit End {
    SetPosition(gameObject=var "Jitter Bits", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Break {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    FadeGroupDown(target=var "Masks", fast=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=var "Hit Sfx", pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=var "Break Sfx", pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Jitter Bits", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Step On {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEvent(gameObject=var "Hero Detect", trigger=2, collideTag="", sendEvent=→"LEAVE", storeCollider=var "")
    AnimatePositionTo(gameObject=var "Jitter Bits", toValue=(0, -0.08, 0), localSpace=true, time=0.1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Jitter Bits"), sendEvent="ENTER", delay=0, everyFrame=false)
    on HIT → Check
    on LEAVE → Step Off
  }

  state PD Activation {
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName=var "PD Bool", isTrue=→"FINISHED", isFalse=(none))
    on ACTIVATED → Activated
    on FINISHED → Activated?
  }

  state Set PD {
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "PD Bool", value=true)
    on FINISHED → Break
  }

  state Step Off {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")
    AnimatePositionTo(gameObject=var "Jitter Bits", toValue=(0, 0, 0), localSpace=true, time=0.1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)  // disabled
    Trigger2dEvent(gameObject=var "Hero Detect", trigger=0, collideTag="", sendEvent=→"STEP", storeCollider=var "")
    SendEventByName(eventTarget=GameObject(var "Jitter Bits"), sendEvent="EXIT", delay=0, everyFrame=false)
    on FINISHED → Idle
    on HIT → Check
    on STEP → Step On
  }

  var Break Sfx: object = breakable_wall_death (sfxstatic_assets_shared.bundle)
  var Hit Sfx: object = <null>

  var Hits: int = 1
  var Activated: bool = false
  var PD Bool: string = ""
  var Bouncer: gameObject = <null>
  var Break Effects: gameObject = <null>
  var CamLocks: gameObject = <null>
  var Hero Detect: gameObject = <null>
  var Jitter Bits: gameObject = <null>
  var Jitter Floor: gameObject = <null>
  var Masks: gameObject = <null>
  var Pt Hit: gameObject = <null>
  var Self: gameObject = <null>
  var Terrain Box: gameObject = <null>
}
