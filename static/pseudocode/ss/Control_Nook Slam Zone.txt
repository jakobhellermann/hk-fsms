# Control
# 1 location(s):
#   Bone_15 / Boss Scene/Nook Slam Zone

fsm Control {
  start Idle

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer=(unset), sendEvent=→"ENTER", storeCollider=var "Skull King")
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="Boss", collideLayer=(unset), sendEvent=→"EXIT", storeCollider=var "Skull King")
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SetFsmBool(gameObject=var "Skull King", fsmName="Control", variableName="Nook Slam Zone", setValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  state Exit {
    SetFsmBool(gameObject=var "Skull King", fsmName="Control", variableName="Nook Slam Zone", setValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Skull King: gameObject = <null>
}
