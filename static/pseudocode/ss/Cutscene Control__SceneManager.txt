# Cutscene Control
# 1 location(s):
#   Room_Caravan_Interior_Travel / _SceneManager

// uses template: flea_caravan_travel_cutscene_control
fsm Cutscene Control {
  start Start

  state Start {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    SendEventByNameV2(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    GetPlayerDataVariable(VariableName="CaravanTroupeLocation", StoreValue=var "Troupe Location")
    GetCaravanLocationScene(Location=var "Troupe Location", StoreSceneName=var "Target Scene", InvalidEvent=(none))
    on FINISHED → Wait
  }

  state Transition Scene {
    CallStaticMethod(className="DeliveryQuestItem", methodName="BreakAllNoEffects", parameters=[], storeResult=(unset var), everyFrame=false)
    BeginSceneTransition(sceneName=var "Target Scene", entryGateName="door_caravanTravelEnd", entryDelay=0, visualization=Default, preventCameraFadeOut=true)
  }

  sequence state Fade Out {
    SetInputHandlerSkipMode(SkipMode=NOT_SKIPPABLE)
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 1), duration=1.5)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=1.5)
    TransitionToAudioSnapshot(snapshot=Off (audioobjects_assets_all.bundle), transitionTime=1.5)
    TransitionToAudioSnapshot(snapshot=at None (audioobjects_assets_all.bundle), transitionTime=1.5)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    PlayAudioEvent(audioClip=flea_caravan_arrive_2d (sfxstatic_assets_areadocks.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=6.5, finishEvent=(none), realTime=false)
    on FINISHED → Transition Scene
  }

  sequence state Wait {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=1.5)
    StartPreloadingScene(SceneName=var "Target Scene", LoadMode=LoadSceneMode(1))
    Wait(time=1.5, finishEvent=(none), realTime=false)
    SetInputHandlerSkipMode(SkipMode=SKIP_PROMPT)
    AddEventRegister(target=Self, eventName="CUSTOM CUTSCENE SKIP")
    Wait(time=6, finishEvent=(none), realTime=false)
    on FINISHED → Fade Out
    on CUSTOM CUTSCENE SKIP → Fade Out
  }

  var Target Scene: string = ""
  var Troupe Location: enum = CaravanTroupeLocations(0)
}
