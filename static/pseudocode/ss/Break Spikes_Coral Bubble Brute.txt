fsm Break Spikes {
  start Inactive
  on BREAK END → Inactive  // from any state

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"ENTER", storeCollider=var "Collider")
    on ENTER → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="CORAL BREAK", delay=0, everyFrame=false)  // disabled
    SendEventByNameUpwards(Target=var "Collider", EventName="TEMP BREAK")
    on FINISHED → Idle
  }

  state Inactive {
    on BREAK START → Idle
  }

  var Collider: gameObject = <null>
}
