# Control
# 1 location(s):
#   Bone_05_boss / Boss Scene/Thick Silk Vines Beast

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on DESTROY → Destroy
  }

  state Destroy {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(4))
    ActivateGameObject(gameObject=var "Wall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Thick Vines"), sendEvent="DESTROY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Beast"), sendEvent="FREE", delay=0, everyFrame=false)
  }

  var Gate 1: gameObject = <null>
  var Gate 2: gameObject = <null>

  var Beast: gameObject = Boss Scene/Thick Silk Vines Beast/Beast
  var Thick Vines: gameObject = Boss Scene/Thick Silk Vines Beast/Thick Vines
  var Wall: gameObject = Boss Scene/Thick Silk Vines Beast/Wall
}
