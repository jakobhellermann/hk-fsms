# Dialogue
# 1 location(s):
#   Shellgrave / Event Control/Plinney Rescue Scene/Plinney Outside

fsm Dialogue {
  start Init

  state Init {
    on FINISHED → Scared
  }

  state Scared {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Scared", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=1, finishEvent=→"TURN", realTime=false)
    SetRandomAudioClipFromTable(Table=Plinney_scared_loop (sfxdynamic_assets_areamoss.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    on BATTLE END → Wait
    on TURN → Scared L
  }

  state Stand {
    PlayRandomAudioClipTable(Table=small_movement_shell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    ActivateGameObject(gameObject=var "Pt Emerge", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=hornet_land_grass_1 (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Scared To Idle Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.005, space=0, everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Plinney_rise_from_scared (sfxdynamic_assets_areabellareamoss.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Activate
  }

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeAudioV2(Target=var "Audio Loop Voice", StartVolume=1, EndVolume=0, Time=0.5, SetOnExit=false)
    on FINISHED → Stand
  }

  state Start NPC {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=1.9, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT → Talked?
    on SING → Sing
    on STARTLE → Startle
  }

  state Sing {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=false, Target=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Plinney_Sing (sfxdynamic_assets_areabellareamoss.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Activate
  }

  state Thank Dialogue {
    SetIntValue(intVariable=var "Talk State", intValue=1, everyFrame=false)
    RunDialogueV4(CustomText="", Sheet="Belltown", Key="PINSMITH_SAVED_THANK_1", TextColor=(unset), NpcVoiceTableOverride=Plinney_laugh (sfxdynamic_assets_areamoss.bundle), OverrideContinue=true, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Clap")
    AudioPlayInState(gameObject=var "Audio Loop Clap", volume=0.15)
    on CONVO_END → Thank Dialogue 2
  }

  state Talked? {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="PINSMITH")
    IntSwitch(intVariable=var "Talk State", compareTo=[0, 1], sendEvent=[→"0", →"1"], everyFrame=false)
    on 0 → Thank Dialogue
    on 1 → Talk Dlg
    on FINISHED → Repeat Dialogue
  }

  state End Dialogue {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Right")
    on FINISHED → Reactivate
  }

  state Repeat Dialogue {
    RunDialogue(Sheet="Belltown", Key="PINSMITH_SAVED_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Reactivate {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    on FINISHED → Start NPC
  }

  state Activate {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=false, Target=Self)
    on FINISHED → Start NPC
  }

  state Scared L {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Scared LookL")
    WaitRandom(timeMin=1, timeMax=4, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"FLINCH", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on BATTLE END → Wait
    on FINISHED → Scared R
    on FLINCH → Flinch Anim
    on HERO DAMAGED → Flinch Anim
    on PILGRIM AMBUSH → Flinch Anim
  }

  state Scared R {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Scared LookR")
    WaitRandom(timeMin=1, timeMax=4, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"FLINCH", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on BATTLE END → Wait
    on FINISHED → Scared L
    on FLINCH → Flinch Anim
    on HERO DAMAGED → Flinch Anim
    on PILGRIM AMBUSH → Flinch Anim
  }

  state Flinch Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Scared Flinch", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on BATTLE END → Flinch Anim End
    on FINISHED → Scared R
  }

  state Flinch Anim End {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Talk Dlg {
    SetIntValue(intVariable=var "Talk State", intValue=2, everyFrame=false)
    RunDialogue(Sheet="Belltown", Key="PINSMITH_SAVED_TALK", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state End Thank Dialogue {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Reactivate
  }

  state Thank Dialogue 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Clap End")
    PlayRandomAudioClipTable(Table=small_movement_shell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    RunDialogue(Sheet="Belltown", Key="PINSMITH_SAVED_THANK_2", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Thank Dialogue
  }

  state Startle {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=false, Target=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.4, finishEvent=→"CANCEL", realTime=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Plinney_Startle (sfxdynamic_assets_areamoss.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on CANCEL → Sing End
  }

  var Talk State: int = 0
  var Audio Loop Clap: gameObject = Event Control/Plinney Rescue Scene/Plinney Outside/Audio Loop Clap
  var Audio Loop Voice: gameObject = Event Control/Plinney Rescue Scene/Plinney Outside/Audio Loop Voice
  var Pt Emerge: gameObject = Event Control/Plinney Rescue Scene/Plinney Outside/Pt Emerge
}
