# Control
# 1 location(s):
#   Dock_09 / Boss Scene/Slide Floors/Dock Slide Floor

fsm Control {
  start Idle

  state Idle {
    on OPEN → State 1
  }

  state Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    EaseFloat(fromValue=0, toValue=5.6, floatVariable=var "X Pos", time=0.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=song_trap_door_open (sfxstatic_assets_areasong.bundle))
    on FINISHED → Opened
  }

  state Opened {
    ActivateGameObject(gameObject=var "Art", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CLOSE → Close
  }

  state Close {
    ActivateGameObject(gameObject=var "Art", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    EaseFloat(fromValue=5.6, toValue=0, floatVariable=var "X Pos", time=0.5, speed=(unset), delay=(unset), easeType=23, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=song_trap_door_open (sfxstatic_assets_areasong.bundle))
  }

  state Antic {
    ObjectJitter(gameObject=var "Art", x=0.05, y=0.05, z=(unset), allowMovement=false, limitFps=30)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=dock_sealed_gate_initialise_open (sfxstatic_assets_areadocks.bundle))
    on FINISHED → Open
  }

  state State 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  var X Pos: float = 0
  var Art: gameObject = Boss Scene/Slide Floors/Dock Slide Floor/Art
}
