// uses template: inert_sign_hit
fsm Hit Animate {
  start Init

  state Idle {
    on TINK UP → Choose Dir
    on TINK DOWN → Choose Dir
    on TINK LEFT → Hit L
    on TINK RIGHT → Hit R
  }

  state Hit L {
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=var "Noise Radius")
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=3, worldSpace=false, time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Settle
  }

  state Choose Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HIT_LEFT", leftEvent=→"HIT_RIGHT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=true, everyFrame=false)
    on HIT_LEFT → Hit L
    on HIT_RIGHT → Hit R
  }

  state Hit R {
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=var "Noise Radius")
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=-3, worldSpace=false, time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Settle
  }

  state Settle {
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=0, worldSpace=false, time=0.25, speed=(unset), delay=(unset), easeType=22, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on TINK DOWN → Choose Dir
    on TINK LEFT → Hit L
    on TINK RIGHT → Hit R
    on TINK UP → Choose Dir
  }

  state Init {
    on FINISHED → Idle
  }

  var Noise Radius: float = 10
}
