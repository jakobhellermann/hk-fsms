# Control
# 2 location(s):
#   Weave_03 / water_components_moss (1)/Tiny Dragonfly (6)
#   Weave_03 / water_components_moss (1)/Tiny Dragonfly (5)

fsm Control {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "xPos", y=var "yMin", z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "yMin", float2=var "Y Range", operation=Add, storeResult=var "yMax", everyFrame=false)
    FloatOperator(float1=var "xPos", float2=var "X Range", operation=Subtract, storeResult=var "xMin", everyFrame=false)
    FloatOperator(float1=var "xPos", float2=var "X Range", operation=Add, storeResult=var "xMax", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    FlingObject(flungObject=Self, speedMin=0.05, speedMax=0.2, angleMin=0, angleMax=360)
    WaitRandom(timeMin=1.7, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=false, StoreState=None, EveryFrame=true)
    on FINISHED → Choice
    on NEEDOLIN → Needolin Pause
  }

  state Choice {
    SendEvent(eventTarget=Self, sendEvent=→"HOR", delay=0, everyFrame=false)  // disabled
    SendRandomEventV2(events=[→"HOR", →"VERT"], weights=[0.5, 0.5], trackingInts=[(unset), (unset)], eventMax=[3, 3])
    on VERT → Vert Choice
    on HOR → Hor Choice
  }

  state Vert Choice {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "yPos", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "yPos", float2=var "yMin", tolerance=0, equal=(none), lessThan=→"U", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "yPos", float2=var "yMax", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"D", everyFrame=false)
    BoolTest(boolVariable=var "isDown", isTrue=→"U", isFalse=→"D", everyFrame=false)
    on U → Up
    on D → Down
  }

  state Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=10, everyFrame=false)
    SetBoolValue(boolVariable=var "isDown", boolValue=false, everyFrame=false)
    on FINISHED → Vert
  }

  state Vert {
    Decelerate(gameObject=Self, deceleration=0.85)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-10, everyFrame=false)
    SetBoolValue(boolVariable=var "isDown", boolValue=true, everyFrame=false)
    on FINISHED → Vert
  }

  state Hor Choice {
    GetPosition(gameObject=Self, vector=(unset), x=var "xPos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "xPos", float2=var "xMin", tolerance=0, equal=(none), lessThan=→"R", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "xPos", float2=var "xMax", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"L", everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[0.5, 0.5], delay=0)
    on L → L
    on R → R
  }

  state L {
    FlingObject(flungObject=Self, speedMin=17, speedMax=19, angleMin=175, angleMax=185)
    on FINISHED → Hor
  }

  state Hor {
    Decelerate(gameObject=Self, deceleration=0.95)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    on FINISHED → Idle
  }

  state R {
    FlingObject(flungObject=Self, speedMin=17, speedMax=19, angleMin=-5, angleMax=5)
    on FINISHED → Hor
  }

  state Needolin Pause {
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=false, StoreState=None, EveryFrame=true)
    on FINISHED → Needolin Decel
    on END → Choice
  }

  state Needolin Decel {
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=false, StoreState=None, EveryFrame=true)
    on FINISHED → Pulse Set
    on END → Choice
  }

  state Pulse Up {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=false, StoreState=None, EveryFrame=true)
    EaseFloat(fromValue=var "Pulse Down Y", toValue=var "Pulse Up Y", floatVariable=var "Pulse Y", time=1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Pulse Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on END → Choice
    on FINISHED → Pulse Down
  }

  state Pulse Set {
    RandomFloat(min=1, max=2, storeResult=var "Random")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Pulse Down Y", z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Pulse Down Y", float2=var "Random", operation=Add, storeResult=var "Pulse Up Y", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Pulse Up
  }

  state Pulse Down {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=false, StoreState=None, EveryFrame=true)
    EaseFloat(fromValue=var "Pulse Up Y", toValue=var "Pulse Down Y", floatVariable=var "Pulse Y", time=1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Pulse Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on END → Choice
    on FINISHED → Pulse Up
  }

  var X Range: float = 5
  var Y Range: float = 2

  var Pulse Down Y: float = 0
  var Pulse Up Y: float = 1.5
  var Pulse Y: float = 0
  var Random: float = 0
  var xMax: float = 0
  var xMin: float = 0
  var xPos: float = 0
  var yMax: float = 0
  var yMin: float = 0
  var yPos: float = 0
  var Ct Horizontal: int = 0
  var Ct Vert: int = 0
  var isDown: bool = true
}
