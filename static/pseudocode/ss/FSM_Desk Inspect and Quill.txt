# FSM
# 1 location(s):
#   Library_13b / Desk Inspect and Quill

fsm FSM {
  start Wait

  sequence state Wait {
    NextFrameEvent(sendEvent=(none))
    GetPlayerDataBool(boolName="blackThreadWorld", storeValue=var "Is Act 3")
    GetQuestState(IsAccepted=var "Tormented Trobbio Quest Accepted", IsCompleted=(unset), Quest=Tormented Trobbio (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    BoolTestMulti(boolVariables=[var "Is Act 3", var "Tormented Trobbio Quest Accepted"], boolStates=[true, false], trueEvent=→"SHOW NONE", falseEvent=(none), storeResult=(unset), everyFrame=false)  // disabled
    GetPlayerDataInt(intName="QuillState", storeValue=var "Quill State")
    IntSwitch(intVariable=var "Quill State", compareTo=[0, 1, 2, 3], sendEvent=[→"SHOW COLLECTABLE QUILL", →"SHOW NON COLLECTABLE QUILL", →"ACT 2 COLLECTED", →"ACT 3 COLLECTED"], everyFrame=false)
    PlayerDataBoolTest(boolName="hasQuill", isTrue=→"SHOW NON COLLECTABLE QUILL", isFalse=(none))
    on SHOW COLLECTABLE QUILL → State
    on SHOW NON COLLECTABLE QUILL → Show Non-Collectable Quill
    on ACT 2 COLLECTED → Hide Red Quill
    on ACT 3 COLLECTED → Hide Purple Quill
    on SHOW NONE → Show None
  }

  state Show Collectable Quill {
    ActivateGameObject(gameObject=var "desk_inspect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Non-Interactive Quills", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on QUILL COLLECTED → Check Act
  }

  state Show Non-Collectable Quill {
    ActivateGameObject(gameObject=var "desk_inspect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Non-Interactive Quills", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    EnableFsmSelf(SetEnabled=false)
  }

  state Hide Red Quill {
    ActivateGameObject(gameObject=var "non-interactive quill red", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Show Non-Collectable Quill
  }

  state Hide Purple Quill {
    ActivateGameObject(gameObject=var "non-interactive quill purple", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Hide Red Quill
  }

  state Check Act {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=(none), isFalse=→"ACT 2 COLLECTED")
    on ACT 2 COLLECTED → Collected Act 2
    on FINISHED → Collected Act 3
  }

  state Collected Act 2 {
    SetPlayerDataInt(intName="QuillState", value=2)
    on FINISHED → Enable Desk
  }

  state Collected Act 3 {
    SetPlayerDataInt(intName="QuillState", value=3)
    on FINISHED → Enable Desk
  }

  state Enable Desk {
    SetPlayerDataBool(boolName="hasQuill", value=true)
    ActivateGameObject(gameObject=var "desk_inspect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    EnableFsmSelf(SetEnabled=false)
  }

  state Show None {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  sequence state State {
    Comment(comment="Added check so that older versions of player data without quill state will still work correctly")
    PlayerDataBoolTest(boolName="hasQuill", isTrue=→"SHOW NON COLLECTABLE QUILL", isFalse=(none))
    on FINISHED → Show Collectable Quill
    on SHOW NON COLLECTABLE QUILL → Show Non-Collectable Quill
  }

  var Quill State: int = 0
  var Is Act 3: bool = false
  var Tormented Trobbio Quest Accepted: bool = false
  var desk_inspect: object = Desk Inspect and Quill/Group/desk_inspect
  var Non-Interactive Quills: object = Desk Inspect and Quill/Group/Non-Interactive Quills
  var Collectable Item Pickup: object = Desk Inspect and Quill/Interactive Quills
  var non-interactive quill red: object = Desk Inspect and Quill/Group/Non-Interactive Quills/non-interactive quill red
  var non-interactive quill purple: object = Desk Inspect and Quill/Group/Non-Interactive Quills/non-interactive quill purple
}
