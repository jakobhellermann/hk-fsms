# First Challenge
# 1 location(s):
#   Cradle_03 / Boss Scene/Intro Sequence

fsm First Challenge {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Challenge Glows")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "CamLock Open")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "CamLock Intro")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "CamLock IntroCocoon")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "silk_cocoon_core")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Silk Boss")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Battle Gate")
    GetPlayerDataBool(boolName="encounteredSilk", storeValue=var "Encountered Silk")
    on FINISHED → Idle
  }

  state Idle {
    on CHALLENGE START → Challenge Cam
  }

  state Challenge Cam {
    ActivateGameObject(gameObject=var "CamLock Open", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock IntroCocoon", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Battle Gate", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=1)
    on CHALLENGE → Wait For Beat End
  }

  state Wait For Beat End {
    CheckIsInEditor(trueEvent=→"QUICK START", falseEvent=(none), storeValue=(unset))  // disabled
    SendEventByName(eventTarget=GameObject(var "silk_cocoon_core"), sendEvent="STOP", delay=0.25, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst Anim
    on QUICK START → Quick Start
  }

  state Burst Anim {
    AnimatorPlay(gameObject=var "silk_cocoon_core", stateName="Silk_Cocoon_Intro_Burst", layer=0, normalizedTime=0, everyFrame=false)
    ListenForAnimationEvent(Target=var "silk_cocoon_core", Response=→"FINISHED")
    FadeNestedFadeGroup(Target=var "Challenge Glows", ToAlpha=0, FadeTime=0.2)
    on FINISHED → Ready Wait
  }

  state Intro Shake {
    PlayAudioEvent(audioClip=silk_rumble_on_black_before_rise (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Encountered Silk", time=2, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    on FINISHED → Hero Facing
  }

  state Silk Enter {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventByName(eventTarget=GameObject(var "Silk Boss"), sendEvent="BEGIN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock IntroCocoon", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Hero Facing {
    GetXDistance(gameObject=var "Hero", target=var "Silk Boss", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=2, tolerance=0, equal=(none), lessThan=→"STRAIGHT BACK", greaterThan=(none), everyFrame=false)
    CheckTargetDirection(gameObject=var "Hero", target=var "Silk Boss", aboveEvent=(none), belowEvent=(none), rightEvent=→"HERO L", leftEvent=→"HERO R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on HERO L → Hero L
    on HERO R → Hero R
    on STRAIGHT BACK → Straight Back
  }

  state Hero L {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Taunt Back Up")
    on FINISHED → Silk Enter
  }

  state Hero R {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Taunt Back Up")
    on FINISHED → Silk Enter
  }

  state Quick Start {
    FadeNestedFadeGroup(Target=var "Challenge Glows", ToAlpha=0, FadeTime=0.2)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventByName(eventTarget=GameObject(var "Silk Boss"), sendEvent="READY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Silk Boss"), sendEvent="BEGIN", delay=0.25, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Open", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock IntroCocoon", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "silk_cocoon_core", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Battle Gate", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Ready Wait {
    SendEventByName(eventTarget=GameObject(var "Silk Boss"), sendEvent="READY", delay=0, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Encountered Silk", time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Intro Shake
  }

  state Straight Back {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Taunt Straight Back Q")
    on FINISHED → Silk Enter
  }

  var Distance: float = 0
  var Loop Count Init: int = 0
  var Encountered Silk: bool = false
  var Boss Scene: gameObject = <null>
  var CamLock Intro: gameObject = <null>
  var CamLock Open: gameObject = <null>
  var CamLock IntroCocoon: gameObject = <null>
  var silk_cocoon_core: gameObject = <null>
  var Silk Boss: gameObject = <null>
  var Battle Gate: gameObject = <null>
  var Challenge Glows: gameObject = <null>
}
