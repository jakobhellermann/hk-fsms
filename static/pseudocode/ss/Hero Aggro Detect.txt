fsm Hero Aggro Detect {
  start Idle

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Close Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"ALERT", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    AddEventRegister(target=Self, eventName="BATS AGGRO")
    on TOOK DAMAGE → Alert
    on ALERT → Alert
    on BATS AGGRO → Become Aggro
  }

  state Alert {
    SendEventToRegister(eventName="BATS AGGRO")
    on FINISHED → Become Aggro
  }

  state Become Aggro {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Targeting NPC", setValue=false, everyFrame=false)
    SetFsmGameObject(gameObject=Self, fsmName="Control", variableName="Current Target", setValue=var "Hero", everyFrame=false)
  }
}
