# Music End
# 1 location(s):
#   Dock_09 / Boss Scene

fsm Music End {
  start Idle

  state Idle {
    on MUSIC UPDATE → Check Pause
  }

  state Check {
    BoolAllTrue(boolVariables=[var "Slasher Defeated", var "Thrower Defeated"], sendEvent=→"END", storeResult=(unset), everyFrame=false)
    on FINISHED → Idle
    on END → State 1
  }

  state Check Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state State 1 {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
  }

  var Slasher Defeated: bool = false
  var Thrower Defeated: bool = false
}
