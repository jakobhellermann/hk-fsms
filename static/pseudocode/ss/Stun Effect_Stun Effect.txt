# Stun Effect
# 1 location(s):
#   globalpoolprefabs_assets_all.bundle / Stun Effect

fsm Stun Effect {
  start Effects

  state Effects {
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(4))
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    DestroySelf(detachChildren=false)  // disabled
    RecycleSelf()
  }
}
