# Anim
# 1 location(s):
#   Ward_09 / Sherma Rescue Scene/Activation Folder/Sherma Rescue NPC

fsm Anim {
  start Init
  on FLINCH → Flinch  // from any state

  state Init {
    on FINISHED → Scared F
  }

  state Scared F {
    SetBoolValue(boolVariable=var "Facing F", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Scared F")
    WaitRandom(timeMin=0.5, timeMax=2, finishEvent=→"CHIME", realTime=false)
    SetStringValue(stringVariable=var "Chime Anim", stringValue="Scared F Chime", everyFrame=false)
    WaitBool(boolTest=var "Battle Started", time=0.1, finishEvent=→"CHIME", realTime=false)
    WaitBool(boolTest=var "Singing", time=0.2, finishEvent=→"CHIME", realTime=false)
    on CHIME → Turn?
  }

  state Turn? {
    BoolTest(boolVariable=var "Stop Chime", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing F", var "Speaking"], boolStates=[true, true], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing F", var "Speaking"], boolStates=[false, true], trueEvent=→"TURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    SendRandomEventV4(events=[→"TURN", (none)], weights=[0.35, 0.65], eventMax=[1, 3], missedMax=[3, 2], activeBool=(unset))
    on TURN → Turn Anim
    on FINISHED → Chime Antic
    on CANCEL → Scared F
  }

  state Turn Anim {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Scared Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"FLINCH", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    on FINISHED → Turn To
  }

  state Turn To {
    BoolTest(boolVariable=var "Facing F", isTrue=→"TURN BACK", isFalse=→"TURN FORWARD", everyFrame=false)
    on TURN FORWARD → Scared F
    on TURN BACK → Scared B
  }

  state Scared B {
    SetBoolValue(boolVariable=var "Facing F", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Scared B")
    WaitRandom(timeMin=0.5, timeMax=2, finishEvent=→"CHIME", realTime=false)
    SetStringValue(stringVariable=var "Chime Anim", stringValue="Scared B Chime", everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"FLINCH", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    BoolTest(boolVariable=var "Speaking", isTrue=→"TURN", isFalse=(none), everyFrame=false)
    WaitBool(boolTest=var "Battle Started", time=0.1, finishEvent=→"CHIME", realTime=false)
    WaitBool(boolTest=var "Speaking", time=0.25, finishEvent=→"TURN", realTime=false)
    WaitBool(boolTest=var "Singing", time=0.2, finishEvent=→"CHIME", realTime=false)
    on CHIME → Turn?
    on TURN → Turn Anim
  }

  sequence state Chime Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Chime Anim", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"FLINCH", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Chime
  }

  state Chime {
    AudioPlayRandom(gameObject=var "Self", audioClips=[sherma_bell_ring_single_1 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_2 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_3 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_4 (sfxstatic_assets_sherma.bundle)], weights=[1, 1, 1, 1], pitchMin=1, pitchMax=1)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"FLINCH", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    on FINISHED → Return To Idle
  }

  state Return To Idle {
    BoolTest(boolVariable=var "Facing F", isTrue=→"TURN FORWARD", isFalse=→"TURN BACK", everyFrame=false)
    on TURN FORWARD → Scared F
    on TURN BACK → Scared B
  }

  state Flinch {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Scared Flinch", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Scared F
  }

  var Facing F: bool = false
  var Speaking: bool = false
  var Stop Chime: bool = false
  var Battle Started: bool = false
  var Singing: bool = false
  var Chime Anim: string = ""
  var Self: gameObject = Sherma Rescue Scene/Activation Folder/Sherma Rescue NPC
}
