# FSM
# 1 location(s):
#   Coral_Judge_Arena / Caravan_Set/Caravan/Active/door2

fsm FSM {
  start Inert

  state Inert {
    on START ENTRY → Set Door Open
  }

  state Set Door Open {
    AnimatorPlay(gameObject=var "Door Animator", stateName="Open", layer=0, normalizedTime=1, everyFrame=false)
    on FINISH ENTRY → Close Door
  }

  state Close Door {
    AnimatorPlay(gameObject=var "Door Animator", stateName="Close", layer=(unset), normalizedTime=(unset), everyFrame=false)
  }

  var Door Animator: gameObject = Caravan_Set/Caravan/Active/Caravan_Spa/caravan_spa_door
}
