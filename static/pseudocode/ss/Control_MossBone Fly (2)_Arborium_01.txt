# Control
# 3 location(s):
#   Arborium_01 / MossBone Fly (2)
#   Tut_01 / MossBone Fly (2)
#   Weave_03 / MossBone Fly (2)

fsm Control {
  start Initiate
  on EXTRACT → Extract  // from any state

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=true, ActiveBool=var "Can Hear", StoreState=None, EveryFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=(none), ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=true, ActiveBool=var "Can Hear", StoreState=None, EveryFrame=true)
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName="Fly")
    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=1, speedMax=2, accelerationMin=5, accelerationMax=15, roamingRangeX=1.5, roamingRangeY=0.75, manualStartPos=(unset))
    FaceDirection(gameObject=var "Self", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckCanSeeHero(storeResult=var "Can See Hero", sendEvent="", eventTarget=Self, everyFrame=true)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=var "In Alert Range", InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Alert Range"], sendEvent=→"ALERT", storeResult=(unset), everyFrame=true)
    on ALERT → Startle
    on TOOK DAMAGE → Startle
    on NEEDOLIN → Pray
    on DEAF → 
  }

  state Initiate {
    SetBoolValue(boolVariable=var "Can Hear", boolValue=var "z_Deaf", everyFrame=false)
    BoolFlip(boolVariable=var "Can Hear")
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    FindAlertRange(target=Self, storeResult=var "Alert Range", childName="Alert Range")
    FindAlertRange(target=Self, storeResult=var "Attack Range", childName="Attack Range")
    FindChild(gameObject=Self, childName="Enemy Blocker", storeResult=var "Enemy Blocker")
    FindChild(gameObject=Self, childName="Land Effect", storeResult=var "Land Effect")
    BoolTest(boolVariable=var "Start Alert", isTrue=→"ALERT", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ALERT → Get Above
  }

  state Get Above {
    SetFloatValue(floatVariable=var "Range Out Timer", floatValue=0, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimation(gameObject=var "Self", animLibName="", clipName="Chase")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToChase", resetFrame=true, everyFrame=true)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=5.5, accelerationForce=18, offsetX=0, offsetY=6)
    CheckCanSeeHero(storeResult=var "Can See Hero", sendEvent="", eventTarget=Self, everyFrame=true)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=var "In Alert Range", InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    CheckAlertRange(alertRange=var "Attack Range", storeResult=var "In Attack Range", InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Attack Range"], sendEvent=→"ATTACK", storeResult=(unset), everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Attack Range"], sendEvent=(none), storeResult=var "Can Not See Hero", everyFrame=true)
    BoolFlipEveryFrame(boolVariable=var "Can Not See Hero", everyFrame=true)
    FloatAddV2(floatVariable=var "Range Out Timer", add=1, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=var "Can Not See Hero")
    FloatCompare(float1=var "Range Out Timer", float2=8, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"RANGE OUT", everyFrame=true)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK", rightHitEvent=→"BONK", bottomHitEvent=(none), leftHitEvent=→"BONK", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK", rightHitEvent=→"BONK", bottomHitEvent=(none), leftHitEvent=→"BONK", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on ATTACK → Attack Antic
    on RANGE OUT → Range Out
    on BONK → Bonk?
    on NEEDOLIN → Pray
  }

  state Attack Antic {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=mossbone_fly_antic (sfxstatic_assets_areatutorial.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[mossbone_fly_antic (sfxstatic_assets_areatutorial.bundle), mossbone_fly_antic_2 (sfxstatic_assets_areamoss.bundle), mossbone_fly_antic_3 (sfxstatic_assets_areamoss.bundle)], weights=[1, 1, 1], pitchMin=0.9, pitchMax=1.1, stopPreviousSound=true)
    SetAudioSource(gameObject=Self, active=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Antic")
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.8, brakeOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=12, everyFrame=false)
    on FINISHED → Drill
  }

  state Drill {
    SetVelocity2d(gameObject=Self, vector=(0, -20), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Drill")
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Wait(time=0.6, finishEvent=→"END", realTime=false)
    ActivateGameObject(gameObject=var "Enemy Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=mossbone_fly_down_attack (sfxstatic_assets_areatutorial.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on COLLIDE → Drill Collide
    on END → Drill End
  }

  state Drill End {
    SetAudioSource(gameObject=Self, active=true)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Drill End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Get Above
  }

  state Drill Collide {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Drill Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=mossbone_fly_ground_hit (sfxstatic_assets_areatutorial.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 14), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Blocker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Land Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=(unset), decelerationY=0.85, brakeOnExit=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Drill Recover
  }

  state Startle {
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[mossbone_fly_alert (sfxstatic_assets_areatutorial.bundle), mossbone_fly_alert_2 (sfxstatic_assets_areatutorial.bundle), mossbone_fly_alert_3 (sfxstatic_assets_areatutorial.bundle), mossbone_fly_alert_4 (sfxstatic_assets_areatutorial.bundle)], weights=[1, 1, 1, 1], pitchMin=0.9, pitchMax=1.1, stopPreviousSound=true)
    on FINISHED → Get Above
  }

  state Drill Recover {
    SetAudioSource(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=1, everyFrame=false)  // disabled
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Enemy Blocker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=(unset), decelerationY=0.85, brakeOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=(none), ActiveOuter=→"NEEDOLIN", IgnoreNeedolinRange=false, useActiveBool=true, ActiveBool=var "Can Hear", StoreState=None, EveryFrame=true)
    on FINISHED → Get Above
    on NEEDOLIN → Pray
  }

  state Range Out {
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Chase End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Still In Range? {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    CheckCanSeeHero(storeResult=var "Can See Hero", sendEvent="", eventTarget=Self, everyFrame=true)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=var "In Alert Range", InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    BoolAllTrue(boolVariables=[var "Can See Hero", var "In Alert Range"], sendEvent=→"ALERT", storeResult=(unset), everyFrame=true)
    on ALERT → Get Above
    on FINISHED → Idle
  }

  state Bonk? {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Hero Y", float2=var "Self Y", operation=Subtract, storeResult=var "Difference", everyFrame=false)
    FloatCompare(float1=var "Difference", float2=-2, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-4, everyFrame=false)
    on FINISHED → Bonk Frame
  }

  state Pray {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Antic")
    IdleBuzzV3(gameObject=Self, waitMin=0.5, waitMax=0.75, speedMax=2, accelerationMin=5, accelerationMax=10, roamingRangeX=0.25, roamingRangeY=0.25, manualStartPos=(unset))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=mossbone_fly_sing_loop (sfxdynamic_assets_areamoss.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on END → Pray End
    on SING DURATION END → Pray End
  }

  state Pray End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="AnticEnd", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Get Above
  }

  state Bonk Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get Above
  }

  state Extract {
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Extract")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Offset", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Offset", multiplyBy=0.25, everyFrame=false)
    FindChild(gameObject=var "Hero", childName="Tool Effects/Extract Point", storeResult=var "Extract Point")
    SetPositionToObject(gameObject=Self, targetObject=var "Extract Point", xOffset=var "Offset", yOffset=0.25, zOffset=(unset), overrideZ=0.003, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashingMossExtract())
    on EXTRACT FINISH → Kill Dir
  }

  state Extract Kill {
    InstaDeath(target=Self, direction=var "Kill Angle")
  }

  state Kill Dir {
    SetFloatValue(floatVariable=var "Kill Angle", floatValue=0, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)
    SetFloatValue(floatVariable=var "Kill Angle", floatValue=180, everyFrame=false)
    on FINISHED → Extract Kill
  }

  var Start Alert: bool = false
  var z_Deaf: bool = false

  var Difference: float = 0
  var Hero Y: float = 0
  var Range Out Timer: float = 0
  var Self Y: float = 0
  var Kill Angle: float = 0
  var Offset: float = 0
  var Rotation: float = 0
  var Z Pos: float = 0
  var Can Not See Hero: bool = false
  var Can See Hero: bool = false
  var In Alert Range: bool = false
  var In Attack Range: bool = false
  var Can Hear: bool = false
  var Enemy Blocker: gameObject = <null>
  var Land Effect: gameObject = <null>
  var Self: gameObject = <null>
  var Extract Point: gameObject = <null>
  var Hero: gameObject = <null>
  var Audio Loop Voice: gameObject = MossBone Fly (2)/Audio Loop Voice
  var Alert Range: object = <null>
  var Attack Range: object = <null>
}
