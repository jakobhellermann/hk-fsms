# Map Msg
# 1 location(s):
#   localpoolprefabs_assets_shared.bundle / Map Update Msg

fsm Map Msg {
  start Init
  on INVENTORY OPENED → Destroy  // from any state
  on SCENE TRANSITION BEGAN → Destroy  // from any state

  state Start {
    SetPosition(gameObject=Self, vector=var "Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    BoolTest(boolVariable=var "Map Shop", isTrue=→"MAP SHOP", isFalse=(none), everyFrame=false)
    on FINISHED → Set
    on MAP SHOP → Map Shop Pos
  }

  state Set {
    GetLanguageString(sheetName="UI", convName="MAP_UPDATED", storeValue=var "Text Str")
    SetTextMeshProText(gameObject=var "Text", textString=var "Text Str")
    SetStringValue(stringVariable=var "Anim Down", stringValue="Entry Full Down", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Check Quill
  }

  state Up {
    FadeNestedFadeGroupV3(Target=Self, ToAlpha=1, FadeTime=0.3, Curve=curve[(time=0, value=0, inSlope=0, outSlope=1), (time=1, value=1, inSlope=1, outSlope=0)], Realtime=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Animation", clipName="Map Writing", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Map Complete
  }

  state Down {
    FadeNestedFadeGroupV3(Target=Self, ToAlpha=0, FadeTime=1, Curve=curve[(time=0, value=0, inSlope=0, outSlope=1), (time=1, value=1, inSlope=1, outSlope=0)], Realtime=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Destroy
  }

  state Destroy {
    DestroyObject(gameObject=var "Self", delay=0, detachChildren=false)
  }

  state Unseen {
    SetPlayerDataVariable(VariableName="HasSeenMapUpdated", SetValue=true)
    ActivateGameObject(gameObject=var "Backboard Large", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard Small", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Up
  }

  state Map Complete {
    Tk2dPlayAnimation(gameObject=var "Animation", animLibName="", clipName="Map Complete")
    on FINISHED → Down
  }

  state Map Shop Pos {
    SetPosition(gameObject=Self, vector=(unset), x=-10.1, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Set
  }

  state Seen? {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ui_map_update (sfxstatic_assets_shared.bundle))
    PlayerDataVariableTest(VariableName="HasSeenMapUpdated", ExpectedValue=true, IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on FALSE → Unseen
    on TRUE → Seen
  }

  state Seen {
    ActivateGameObject(gameObject=var "Backboard Large", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard Small", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=(unset), X=0, Y=(unset), Space=1, EveryFrame=false)
    on FINISHED → Up
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Animation", storeResult=var "Animation")
    FindChild(gameObject=Self, childName="Text", storeResult=var "Text")
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0)
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start
  }

  state Check Quill {
    PlayerDataBoolTest(boolName="hasQuill", isTrue=→"TRUE", isFalse=→"FALSE")
    on TRUE → Seen?
    on FALSE → Seen? 2
  }

  state Unseen 2 {
    SetPlayerDataVariable(VariableName="HasSeenMapMarkerUpdated", SetValue=true)
    ActivateGameObject(gameObject=var "Backboard Large", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard Small", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Up 2
  }

  state Seen? 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=ui_map_update_pins_only (sfxstatic_assets_shared.bundle))
    PlayerDataVariableTest(VariableName="HasSeenMapMarkerUpdated", ExpectedValue=true, IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on FALSE → Unseen 2
    on TRUE → Seen 2
  }

  state Seen 2 {
    ActivateGameObject(gameObject=var "Backboard Large", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Backboard Small", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Text", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=(unset), X=0, Y=(unset), Space=1, EveryFrame=false)
    on FINISHED → Up 2
  }

  state Up 2 {
    FadeNestedFadeGroupV3(Target=Self, ToAlpha=1, FadeTime=0.3, Curve=curve[(time=0, value=0, inSlope=0, outSlope=1), (time=1, value=1, inSlope=1, outSlope=0)], Realtime=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Animation", clipName="Map Writing Marker", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Map Complete 2
  }

  state Map Complete 2 {
    Tk2dPlayAnimation(gameObject=var "Animation", animLibName="", clipName="Map Writing Marker Complete")
    on FINISHED → Down
  }

  var Position: vector3 = (-1.29, -6.83, 0)

  var Delay: float = 0
  var Map Shop: bool = false
  var Anim Down: string = ""
  var Text Str: string = ""
  var Animation: gameObject = <null>
  var Backboard Large: gameObject = Map Update Msg/Backboard Large
  var Self: gameObject = <null>
  var Text: gameObject = <null>
  var Backboard Small: gameObject = Map Update Msg/Backboard Small
}
