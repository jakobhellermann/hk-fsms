# Control
# 1 location(s):
#   Coral_27 / Battle Scene/Charge Drillers

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "Driller A")
    FindNamedChild(gameObject=Self, storeResult=var "Driller B")
    FindNamedChild(gameObject=Self, storeResult=var "Driller C")
    on FINISHED → Idle
  }

  state Idle {
    on CHARGE L → Set Charge L
    on CHARGE R → Set Charge R
  }

  state Set Charge L {
    SetStringValue(stringVariable=var "Charge Event", stringValue="CHARGE L", everyFrame=false)
    on FINISHED → Set First
  }

  state Set Charge R {
    SetStringValue(stringVariable=var "Charge Event", stringValue="CHARGE R", everyFrame=false)
    on FINISHED → Set First
  }

  state Set First {
    GetRandomChild(gameObject=Self, storeResult=var "Driller Crt")
    SetFsmFloat(gameObject=var "Driller Crt", fsmName="Control", variableName="Charge Pause", setValue=0, everyFrame=false)
    SetParent(gameObject=var "Driller Crt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Set Second
  }

  state Set Second {
    GetRandomChild(gameObject=Self, storeResult=var "Driller Crt")
    SetFsmFloat(gameObject=var "Driller Crt", fsmName="Control", variableName="Charge Pause", setValue=0.6, everyFrame=false)
    SetParent(gameObject=var "Driller Crt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Set Third
  }

  state Set Third {
    GetRandomChild(gameObject=Self, storeResult=var "Driller Crt")
    SetFsmFloat(gameObject=var "Driller Crt", fsmName="Control", variableName="Charge Pause", setValue=1.2, everyFrame=false)
    SetParent(gameObject=var "Driller Crt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Attack
  }

  state Attack {
    SetParent(gameObject=var "Driller A", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Driller B", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Driller C", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent=var "Charge Event", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Charge Event: string = ""
  var Driller A: gameObject = <null>
  var Driller B: gameObject = <null>
  var Driller C: gameObject = <null>
  var Driller Crt: gameObject = <null>
  var Self: gameObject = <null>
}
