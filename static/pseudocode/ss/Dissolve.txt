# Dissolve
# 1 location(s):
#   herodynamic_assets_all.bundle / Hornet Cocoon Corpse/Core

fsm Dissolve {
  start Delay

  state Delay {
    FadeNestedFadeGroup(Target=var "Hornet Core", ToAlpha=1, FadeTime=0)
    ActivateGameObject(gameObject=var "Dissolve Particles", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dissolve Start
  }

  state Dissolve Start {
    ActivateGameObject(gameObject=var "Dissolve Particles", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Out
  }

  state Fade Out {
    FadeNestedFadeGroup(Target=var "Hornet Core", ToAlpha=0, FadeTime=0.1)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    CallMethodProper(gameObject=Self, behaviour="JitterSelf", methodName="StopJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  var Hornet Core: gameObject = Hornet Cocoon Corpse/Core/Hornet
  var Dissolve Particles: gameObject = Hornet Cocoon Corpse/Core/Pt Spider Fall
}
