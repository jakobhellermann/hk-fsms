# Control
# 1 location(s):
#   Clover_10 / Boss Scene/Clover Dancer Sickle (2)

fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Flash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fly Out
  }

  state Fly Out {
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AccelerateToXByScale(gameObject=Self, accelerationFactor=0.5, targetSpeed=40)
    TiltBySpeedY(gameObject=Self, tiltFactor=3, tiltMax=45, rotationSpeed=50)
    DecelerateXY(gameObject=Self, decelerationX=(unset), decelerationY=0.9, brakeOnExit=false)  // disabled
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
    on MULTI HIT CONNECT → Multi Slash
  }

  state Recycle {
    RecycleSelf()
  }

  state Multi Slash {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Cyclone Projectile Slash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    GetPosition2D(gameObject=var "Hero", vector=var "Here Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    AnimateRigidBody2DPositionTo(GameObject=Self, ToValue=var "Here Pos", time=0.1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Recycle
  }

  var Here Pos: vector2 = (0, 0)
  var Flash: gameObject = Boss Scene/Clover Dancer Sickle (2)/Flash
}
