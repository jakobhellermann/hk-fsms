# Control

fsm Control {
  start Activate Entry CamLock?

  state Crash Pause {
    ActivateGameObject(gameObject=var "Floor Unbroken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Idle Crows", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Broken", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Entry Crows", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=true)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetCState", parameters=["transitioning", false], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetCState", parameters=["inUpdraft", false], storeResult=(unset var), EveryFrame=false)
    SetPosition(gameObject=var "Hero", vector=(unset), x=68, y=2, z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → Crash In
  }

  state Crash In {
    PlayAudioEvent(audioClip=rock_roller_explode (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=rock_roller_explode (sfxstatic_assets_areabone.bundle), pitchMin=0.75, pitchMax=0.75, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=rock_roller_explode (sfxstatic_assets_areabone.bundle), pitchMin=0.5, pitchMax=0.5, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Floor Unbroken", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Updraft Shoot")
    SetVelocity2d(gameObject=var "Hero", vector=(0, 18), x=(unset), y=(unset), everyFrame=true)
    Wait(time=0.4, finishEvent=→"NEXT", realTime=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    on NEXT → Flip
  }

  state Flip {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="DownSpikeBounce 1")
    ActivateGameObject(gameObject=var "CamLock Entry", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=false)
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=-8.5, y=(unset), everyFrame=true)
    CheckYPosition(gameObject=var "Hero", compareTo=5.7, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    SendEventToRegister(eventName="FLY AWAY")
    on LAND → Land
  }

  state Land {
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="HardLand Greymoor", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Resume
  }

  state Resume {
    SetPlayerDataBool(boolName="crashedIntoGreymoor", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroAnimationController", methodName="ResetPlays", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControlToIdle", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetCollider(gameObject=var "bot1", active=true, resetOnExit=false)
    on FINISHED → Area Title
  }

  state Area Title {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Area Title Controller", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Music
  }

  state Music {
    ApplyMusicCue(musicCue=Greymoor (audiocuesdynamic_assets_areagreymoor.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal Greymoor (audioobjects_assets_all.bundle), transitionTime=1)
  }

  state Broken {
    SetCollider(gameObject=var "bot1", active=true, resetOnExit=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Floor Broken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Idle Crows", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Area Title Controller", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Floor Unbroken", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Music
  }

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Pause 2
  }

  state Check State {
    FindNamedChild(gameObject=Self, storeResult=var "Idle Crows")
    FindNamedChild(gameObject=Self, storeResult=var "Break Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Entry Crows")
    FindNamedChild(gameObject=Self, storeResult=var "Floor Broken")
    FindNamedChild(gameObject=Self, storeResult=var "Floor Unbroken")
    FindNamedChild(gameObject=Self, storeResult=var "Area Title Controller")
    FindNamedChild(gameObject=Self, storeResult=var "bot1")
    PlayerDataBoolTest(boolName="crashedIntoGreymoor", isTrue=→"BROKEN", isFalse=(none))
    PlayerDataBoolTest(boolName="crashingIntoGreymoor", isTrue=→"CRASHING IN", isFalse=(none))
    SendEvent(eventTarget=Self, sendEvent=→"UNBROKEN", delay=0, everyFrame=false)
    on CRASHING IN → Crash Pause
    on BROKEN → Broken
    on UNBROKEN → Unbroken
  }

  state Unbroken {
    ActivateGameObject(gameObject=var "Floor Unbroken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Idle Crows", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "bot1", active=true, resetOnExit=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Area Title Controller", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Music
  }

  state Pause 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check State
  }

  state Land To idle {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Run To Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    on FINISHED → Resume
  }

  state Activate Entry CamLock? {
    ActivateGameObject(gameObject=var "CamLock Entry", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Entry")
    PlayerDataBoolTest(boolName="crashedIntoGreymoor", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(boolName="crashingIntoGreymoor", isTrue=(none), isFalse=→"FINISHED")
    ActivateGameObject(gameObject=var "CamLock Entry", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Pause
  }

  var Break Effects: gameObject = <null>
  var Crows: gameObject = <null>
  var Planks: gameObject = <null>

  var Audio Point: gameObject = Floor Control Scene/Break Audio Point
  var CamLock Entry: gameObject = <null>
}
