    on EXTEND → Extend
  }

  }

  state End {

  }

  state Extend {
    Tk2dPlayFrame(gameObject=var "Ball", frame=0)  // disabled

    on EXTEND → Extend
    on END → Wait
  }

  var Scale: float = 0
  var Ball: gameObject = Hero_Hornet/Special Attacks/Sphere Ball/Ball
}
