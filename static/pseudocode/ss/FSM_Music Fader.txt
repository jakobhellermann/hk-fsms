# FSM
# 2 location(s):
#   Dust_05 / Music Fader
#   Shadow_04 / Music Fader

fsm FSM {
  start Wait

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Fade
  }

  state Fade {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
  }
}
