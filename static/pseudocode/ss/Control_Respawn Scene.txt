# Control
# 1 location(s):
#   Ward_02 / Boss Scene Parent/Respawn Scene

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Husks")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slasher 1")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slammer 1")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slasher 2")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slammer 2")
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check Valid
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Respawn Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Spawn Pause
  }

  state Spawn Pause {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Respawn Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    on FINISHED → Spawn Choice
    on EXIT → Idle
  }

  state Spawn Choice {
    SendRandomEvent(events=[→"BOTH", →"2 SLAMMER", →"2 SLASHER"], weights=[0.5, 0.25, 0.25], delay=0)
    on BOTH → Both
    on 2 SLAMMER → 2 Slammer
    on 2 SLASHER → 2 Slasher
  }

  state Battle {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on HUSK KILLED → Remaining Count 2
    on FINISHED → Remaining Count 2
  }

  state Remaining Count 2 {
    GetChildCount(gameObject=var "Husks", storeResult=var "Husks Remaining")
    IntCompare(integer1=var "Husks Remaining", integer2=4, equal=→"NEXT", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Battle
    on NEXT → Spawn Pause
  }

  state 2 Slammer {
    SendEventByName(eventTarget=GameObject(var "Slammer 1"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Slammer 2"), sendEvent="SPAWN", delay=1, everyFrame=false)
    on FINISHED → Battle
  }

  state 2 Slasher {
    SendEventByName(eventTarget=GameObject(var "Slasher 1"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Slasher 2"), sendEvent="SPAWN", delay=1, everyFrame=false)
    on FINISHED → Battle
  }

  state Both {
    SendEventByName(eventTarget=GameObject(var "Slasher 1"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Slammer 1"), sendEvent="SPAWN", delay=1, everyFrame=false)
    on FINISHED → Battle
  }

  state Check Valid {
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="top1", equalEvent=→"VALID", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Entry Gate", compareTo="right1", equalEvent=→"VALID", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Entry Gate", compareTo="bot1", equalEvent=→"VALID", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on VALID → Idle
  }

  var Husks Remaining: int = 0
  var Entry Gate: string = ""
  var Husks: gameObject = <null>
  var Slammer 1: gameObject = <null>
  var Slammer 2: gameObject = <null>
  var Slasher 1: gameObject = <null>
  var Slasher 2: gameObject = <null>
  var Floor Shuffler: gameObject = <null>
}
