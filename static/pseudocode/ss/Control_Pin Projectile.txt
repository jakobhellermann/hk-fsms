# Control
# 1 location(s):
#   localpoolprefabs_assets_areagreymoor.bundle / Pin Projectile

fsm Control {
  start First
  on DEALT DAMAGE → Dealt Damage  // from any state

  state Init {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pin Fly")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dmg Receiver", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Fly
  }

  state Fly {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"HIT TERRAIN", storeCollider=var "")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"PROJECTILE BREAK", sendEventSpikes=→"PROJECTILE BREAK", sendEventLava=→"PROJECTILE BREAK", sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    ActivateGameObject(gameObject=var "Dmg Receiver", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on HIT TERRAIN → Stick
    on PARRIED → Check
    on PROJECTILE BREAK → Break
  }

  state Stick {
    PlayAudioEvent(audioClip=straight_pin_wall_hit (herosfxstatic_assets_all.bundle), pitchMin=0.9, pitchMax=1.25, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pin Thunk")
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    WaitRandom(timeMin=5, timeMax=6, finishEvent=→"WAIT", realTime=false)
    on WAIT → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Deactivate
  }

  state Deflect {
    FloatAdd(floatVariable=var "Angle", add=var "Angle Mod", everyFrame=false, perSecond=false)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=2, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on FINISHED → Fly
  }

  state Check {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    FloatCompare(float1=var "Angle", float2=0, tolerance=1, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Angle", float2=180, tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on L → Deflect L
    on R → Deflect R
    on FINISHED → Deflect Other
  }

  state Deflect L {
    RandomFloat(min=-5, max=-10, storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Deflect R {
    RandomFloat(min=5, max=10, storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Deflect Other {
    RandomFloat(min=5, max=10, storeResult=var "Angle Mod")
    RandomlyFlipFloat(storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Dealt Damage {
    on FINISHED → Deactivate
  }

  state Deactivate {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Recycle
  }

  state First {
    BoolTest(boolVariable=var "Init", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pin Fly")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=false)
    SetBoolValue(boolVariable=var "Init", boolValue=true, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Break Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dmg Receiver", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  var Angle: float = 0
  var Angle Mod: float = 0
  var Speed: float = 0
  var Init: bool = false
  var Break Effect: gameObject = Pin Projectile/Break Effect
  var Pt Break: gameObject = Pin Projectile/Pt Break
  var Thunk Effect: gameObject = Pin Projectile/Thunk Effect
  var Dmg Receiver: gameObject = Pin Projectile/Dmg Receiver
}
