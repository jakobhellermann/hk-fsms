# Phase Control
# 1 location(s):
#   Crawl_10 / Area_States/Infected/Blue Assistant

fsm Phase Control {
  start Init
  on TOOK DAMAGE → Check Init Max  // from any state
  on TOOK TAG DAMAGE → Check Init Max  // from any state
  on CHECK → Check Init Max  // from any state

  state Init {
    GetHP(target=Self, storeValue=var "Init HP")
    MultiplyIntByFloat(integer=var "Init HP", multiplyFloat=0.95, storeResult=var "P2 HP", everyFrame=false, forceRoundUp=false)
    MultiplyIntByFloat(integer=var "Init HP", multiplyFloat=0.75, storeResult=var "P3 HP", everyFrame=false, forceRoundUp=false)
    MultiplyIntByFloat(integer=var "Init HP", multiplyFloat=0.5, storeResult=var "P4 HP", everyFrame=false, forceRoundUp=false)
    MultiplyIntByFloat(integer=var "Init HP", multiplyFloat=0.25, storeResult=var "P5 HP", everyFrame=false, forceRoundUp=false)
    on FINISHED → Check Init Max
  }

  state P 1 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=1, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time", setValue=1, everyFrame=false)
    Wait(time=0.5, finishEvent=→"CHECK", realTime=false)
  }

  state Check {
    CompareHP(enemy=var "Self", integer2=var "P2 HP", equal=(none), lessThan=(none), greaterThan=→"P1", everyFrame=false)
    CompareHP(enemy=var "Self", integer2=var "P3 HP", equal=(none), lessThan=(none), greaterThan=→"P2", everyFrame=false)
    CompareHP(enemy=var "Self", integer2=var "P4 HP", equal=(none), lessThan=(none), greaterThan=→"P3", everyFrame=false)
    CompareHP(enemy=var "Self", integer2=var "P5 HP", equal=→"P4", lessThan=→"P5", greaterThan=→"P4", everyFrame=false)
    on P1 → P 1
    on P2 → P 2
    on P3 → P 3
    on P4 → P 4
    on P5 → P 5
  }

  state P 2 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=2, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time", setValue=1, everyFrame=false)
    Wait(time=0.5, finishEvent=→"CHECK", realTime=false)
  }

  state P 3 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=3, everyFrame=false)
    Wait(time=0.5, finishEvent=→"CHECK", realTime=false)
  }

  state P 4 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=4, everyFrame=false)
    Wait(time=0.5, finishEvent=→"CHECK", realTime=false)
  }

  state P 5 {
    SetFsmInt(gameObject=Self, fsmName="Control", variableName="Phase", setValue=5, everyFrame=false)
    Wait(time=0.5, finishEvent=→"CHECK", realTime=false)
  }

  state Check Init Max {
    GetPlayerDataInt(intName="BlueAssistantBloodCount", storeValue=var "Blood Count")
    IntSwitch(intVariable=var "Blood Count", compareTo=[0, 1, 2, 3, 4], sendEvent=[→"0", →"1", →"2", →"3", →"4"], everyFrame=false)
    on 0 → Blood Count 0
    on 1 → Blood Count 1
    on 2 → Blood Count 2
    on 3 → Blood Count 3
    on 4 → Check
  }

  state Blood Count 0 {
    SetIntValue(intVariable=var "New Max", intValue=1, everyFrame=false)
    on FINISHED → Set Max
  }

  state Set Max {
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=UpdateMaxHP(var "New Max"))
    GetHP(target=Self, storeValue=var "HP")
    IntClamp(intVariable=var "HP", minValue=0, maxValue=var "New Max", everyFrame=false)
    SetHP(target=Self, hp=var "HP")
    on FINISHED → Check
  }

  state Blood Count 1 {
    SetIntValue(intVariable=var "New Max", intValue=250, everyFrame=false)
    on FINISHED → Set Max
  }

  state Blood Count 2 {
    SetIntValue(intVariable=var "New Max", intValue=500, everyFrame=false)
    on FINISHED → Set Max
  }

  state Blood Count 3 {
    SetIntValue(intVariable=var "New Max", intValue=750, everyFrame=false)
    on FINISHED → Set Max
  }

  var Init HP: int = 0
  var P2 HP: int = 0
  var P3 HP: int = 0
  var P4 HP: int = 0
  var P5 HP: int = 0
  var Blood Count: int = 0
  var New Max: int = 0
  var HP: int = 0
  var Self: gameObject = Area_States/Infected/Blue Assistant
}
