# Hit Response
# 1 location(s):
#   Greymoor_03 / Weathervane States/Post Quest/Scarecraw

fsm Hit Response {
  start State 1

  state State 1 {
    on HIT_DOWN → Random
    on HIT_UP → Random
    on HIT_LEFT → Hit Left
    on HIT_RIGHT → Hit Right
  }

  state Hit Left {
    AnimatorPlay(gameObject=Self, stateName="Hit Left", layer=0, normalizedTime=0, everyFrame=false)
    AnimatorPlay(gameObject=var "Sprite", stateName="Hit", layer=0, normalizedTime=0, everyFrame=false)
    on FINISHED → State 1
  }

  state Hit Right {
    AnimatorPlay(gameObject=Self, stateName="Hit Right", layer=0, normalizedTime=0, everyFrame=false)
    AnimatorPlay(gameObject=var "Sprite", stateName="Hit", layer=0, normalizedTime=0, everyFrame=false)
    on FINISHED → State 1
  }

  state Random {
    SendRandomEventFair(Events=[{SendEvent=→"HIT_LEFT", Probability=1}, {SendEvent=→"HIT_RIGHT", Probability=1}], TrackingArray=var "Tracking Array", MissedMultiplier=2)
    on HIT_LEFT → Hit Left
    on HIT_RIGHT → Hit Right
  }

  var Sprite: gameObject = Weathervane States/Post Quest/Scarecraw/Sprite
  var Tracking Array: array = var "Tracking Array"
}
