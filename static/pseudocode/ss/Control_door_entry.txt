# Control

fsm Control {
  start Inactive

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Hornet Fall Dummy")
    FindNamedChild(gameObject=Self, storeResult=var "Landing Splash")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "intro_darkness")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Arena Roof")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "CamLocks")
    FindNamedChild(gameObject=var "CamLocks", storeResult=var "1 - Enter")
    FindNamedChild(gameObject=var "CamLocks", storeResult=var "2 - Battle")
    ActivateAllChildren(gameObject=var "CamLocks", activate=false)
    ActivateGameObject(gameObject=var "1 - Enter", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=FaceRight())
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SetPosition2d(gameObject=var "Hero", vector=(unset), x=29, y=55, space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "intro_darkness", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetDarknessLevel(SetLevel=1, StoreCurrentLevel=(unset))
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetHeroAffectedByGravity(SetTime=0, CanSet=(unset), SetAffectedByGravity=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fall In
  }

  state Wait For Entered Scene {
    SetPlayerDataBool(boolName="disableInventory", value=true)
    on FINISH ENTRY → Silk Darkness?
  }

  state Fall In {
    TransitionToAudioSnapshot(snapshot=On (audioobjects_assets_all.bundle), transitionTime=0.25)
    TransitionToAudioSnapshot(snapshot=at Layer 1 (audioobjects_assets_all.bundle), transitionTime=0.75)
    ScreenFaderAlpha(EndAlpha=0, Duration=0.5)
    ActivateGameObject(gameObject=var "Hornet Fall Dummy", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    SetPositionToObject2D(gameObject=var "Hornet Fall Dummy", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    SetHeroAffectedByGravity(SetTime=0, CanSet=(unset), SetAffectedByGravity=true)
    CheckIsCharacterGrounded(Target=var "Hero", RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    Wait(time=2, finishEvent=→"LAND", realTime=false)
    AddForce2d(gameObject=var "Hero", forceMode=0, atPosition=(unset), vector=(unset), x=(unset), y=30, vector3=(unset), everyFrame=true)
    AudioPlayInState(gameObject=Self, volume=0)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=0.5, time=2)
    on LAND → Land
  }

  state Inactive {
    on START ENTRY → Wait For Entered Scene
  }

  state Land {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=hornet_land_hard new (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=enemy_splash_in_water (sfxstatic_assets_water.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SpawnObjectFromGlobalPool(gameObject=Hard Land Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="HardLand Greymoor", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "1 - Enter", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "2 - Battle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Landing Splash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Arena Roof", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → Return Control
  }

  state Return Control {
    PlayerDataBoolTest(boolName="EncounteredLostLace", isTrue=→"ENCOUNTERED", isFalse=(none))
    ResetCutsceneBools()
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on ENCOUNTERED → Encountered
  }

  state Encountered {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventToRegister(eventName="HORNET ENTERED")
    SetPlayerDataBool(boolName="disableInventory", value=false)
  }

  state Silk Darkness? {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    PlayerDataBoolTest(boolName="EncounteredLostLace", isTrue=→"FINISHED", isFalse=(none))
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "darkness_silk_cover")
    ActivateGameObject(gameObject=var "darkness_silk_cover", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Init
  }

  var 1 - Enter: gameObject = <null>
  var Boss Scene: gameObject = <null>
  var CamLocks: gameObject = <null>
  var Hornet Fall Dummy: gameObject = <null>
  var darkness_silk_cover: gameObject = <null>
  var intro_darkness: gameObject = <null>
  var 2 - Battle: gameObject = <null>
  var Landing Splash: gameObject = <null>
  var Arena Roof: gameObject = <null>
}
