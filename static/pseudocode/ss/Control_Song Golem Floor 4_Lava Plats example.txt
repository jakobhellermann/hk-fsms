# Control
# 1 location(s):
#   Bone_East_08_boss_beastfly / Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4

fsm Control {
  start Init
  on AWAY → Away  // from any state
  on BREAK AWAY → Break Away Antic  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=0, everyFrame=false)
    on FINISHED → Beastfly Target Init
  }

  state Idle {
    SetPosition(gameObject=var "part 1", vector=(unset), x=0, y=-0.428, z=0, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "part 2", vector=(unset), x=0.013, y=-0.39, z=0.254, space=1, everyFrame=false, lateUpdate=false)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Player(9), sendEvent=→"STEP", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))

    on BOB → Bob
    on SWIPE BOB → Swipe Dust
    on LAVA DEATH END → Invincible
    on PLAT BREAK → Beastfly Target Unset
    on STEP → Step
    on BIG BOMB EXPLOSION → Beastfly Target Unset
  }

  state Break {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "part 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "part 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=6.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reform Antic
    on LAVA DEATH END → Quick Reform
    on GOLEM DEATH → Quick Reform
    on GOLEM STUN → Quick Reform
  }

  state Bob {
    PlayAudioEventRandom(audioClips=[lavaplatform_land_1 (sfxstatic_assets_areabone.bundle), lavaplatform_land_2 (sfxstatic_assets_areabone.bundle), lavaplatform_land_3 (sfxstatic_assets_areabone.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    SetPosition(gameObject=var "part 2", vector=(unset), x=0.013, y=-0.39, z=0.254, space=1, everyFrame=false, lateUpdate=false)
    iTweenPunchPosition(gameObject=var "part 1", id="", vector=(0, -0.5, 0), time=3, delay=0, loopType=0, space=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)

    PlayParticleEmitter(gameObject=var "Pt Bob", emit=0, resetIfPlaying=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on PLAT BREAK → Beastfly Target Unset
  }

  state Reform Antic {
    PlayAudioEvent(audioClip=lavaplatform_reform (sfxstatic_assets_areabone.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitterInState(gameObject=var "Pt Bubble")
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Beastfly Target Set
    on PLAT BREAK → Beastfly Target Unset
    on LAVA DEATH END → Quick Reform
    on GOLEM DEATH → Quick Reform
    on GOLEM STUN → Quick Reform
  }

  state Reform {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetPosition(gameObject=var "part 1", vector=(unset), x=0, y=-0.428, z=0, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "part 2", vector=(unset), x=0.013, y=-0.39, z=0.254, space=1, everyFrame=false, lateUpdate=false)

    PlayParticleEmitter(gameObject=var "Pt Emerge", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "part 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "part 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    iTweenPunchPosition(gameObject=var "part 1", id="", vector=(0, 0.5, 0), time=3, delay=0, loopType=0, space=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)

    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on PLAT BREAK → Beastfly Target Unset
    on LAVA DEATH END → Quick Reform
    on GOLEM DEATH → Quick Reform
    on GOLEM STUN → Quick Reform
  }

  state Swipe Dust {
    PlayParticleEmitterChildren(gameObject=var "Swipe Fx", resetTimeIfPlaying=false, stopOnStateExit=false)
    on FINISHED → Bob
    on PLAT BREAK → Beastfly Target Unset
  }

  state Invincible {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Quick Reform {
    SetPosition(gameObject=var "part 1", vector=(unset), x=0, y=-0.428, z=0, space=1, everyFrame=false, lateUpdate=false)

    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    PlayParticleEmitter(gameObject=var "Pt Bob", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Emerge", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "part 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "part 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    iTweenPunchPosition(gameObject=var "part 1", id="", vector=(0, 0.5, 0), time=1, delay=0, loopType=0, space=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenPunchPosition(gameObject=var "part 2", id="", vector=(0, 0.5, 0), time=1, delay=0, loopType=0, space=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Invincible
  }

  state Away {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -4, 0), time=1, speed=(unset), delay=(unset), easeType=24, reverse=false, finishEvent=(none), realTime=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on RETURN → Return Pause
  }

  state Return Pause {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "part 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "part 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=var "Return Pause", finishEvent=→"FINISHED", realTime=false)

  }

  state Step {
    PlayAudioEventRandom(audioClips=[lavaplatform_land_1 (sfxstatic_assets_areabone.bundle), lavaplatform_land_2 (sfxstatic_assets_areabone.bundle), lavaplatform_land_3 (sfxstatic_assets_areabone.bundle)], pitchMin=0.85, pitchMax=1.15, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetPosition(gameObject=var "part 1", vector=(unset), x=0, y=-0.428, z=0, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "part 2", vector=(unset), x=0.013, y=-0.39, z=0.254, space=1, everyFrame=false, lateUpdate=false)
    iTweenPunchPosition(gameObject=var "part 1", id="", vector=(0, -0.1, 0), time=2, delay=0, loopType=0, space=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)

    PlayParticleEmitter(gameObject=var "Dust", emit=0, resetIfPlaying=false)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Player(9), sendEvent=→"STEP", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    on FINISHED → Idle
    on STEP → Step
    on PLAT BREAK → Beastfly Target Unset
    on BOB → Bob
    on SWIPE BOB → Swipe Dust
    on LAVA DEATH END → Invincible
  }

  state Break Away {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "part 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "part 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=breakable_wall_death (sfxstatic_assets_shared.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

  state Break Away Antic {
    ObjectJitterLocal(gameObject=var "part 1", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=false, limitFps=0)
    ObjectJitterLocal(gameObject=var "part 2", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=false, limitFps=0)
    on BREAK → Break Away
  }

  state Beastfly Target Init {
    BoolTest(boolVariable=var "Beastfly Target", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetTag(gameObject=Self, tag="Target")
    on FINISHED → Idle
  }

  state Beastfly Target Unset {
    BoolTest(boolVariable=var "Beastfly Target", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetTag(gameObject=Self, tag="Untagged")
    on FINISHED → Break
  }

  state Beastfly Target Set {
    BoolTest(boolVariable=var "Beastfly Target", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetTag(gameObject=Self, tag="Target")
    on FINISHED → Reform
  }

  var Return Pause: float = 0
  var Beastfly Target: bool = false

  var Start Y: float = 0
  var Pt Bob: gameObject = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/Pt Bob
  var Break Effects: object = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/Break Effects
  var Dust: object = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/Dust
  var Pt Bubble: object = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/Pt Bubble
  var Pt Emerge: object = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/Pt Emerge
  var Swipe Fx: object = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/Swipe Fx
  var part 1: object = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/part 1
  var part 2: object = Boss Scene Beastfly/Lava Plats example/Song Golem Floor 4/part 2
}
