# Hive Health Regen
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Health
#   Menu_Title / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Health

fsm Hive Health Regen {
  start Init
  on CHARM INDICATOR CHECK → Check  // from any state
  on HERO LEAVE → Check  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Hive Recovery Blob", storeResult=var "Recovery Blob")
    StopParticleEmittersInChildrenV2(Target=var "Lifeblood Drips", StopBehaviour=0)
    on FINISHED → Pause
  }

  state Idle {
    PlayParticleEmitter(gameObject=var "Lifeblood Drips", emit=(unset), resetIfPlaying=false)
    on DAMAGE TAKEN → Start Recovery
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    SetMeshRenderer(gameObject=var "Recovery Blob", active=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="get_IsInLifebloodState", parameters=[], storeResult=var "Has Regen", EveryFrame=false)
    BoolTest(boolVariable=var "Has Regen", isTrue=→"HIVE", isFalse=→"NORMAL", everyFrame=false)
    on NORMAL → Inert
    on HIVE → Idle
  }

  state Inert {
    StopParticleEmittersInChildrenV2(Target=var "Lifeblood Drips", StopBehaviour=1)
  }

  state Reset Timer {
    SetFloatValue(floatVariable=var "Timer", floatValue=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Recovery Blob", active=true)
    Tk2dPlayAnimation(gameObject=var "Recovery Blob", animLibName="", clipName="Hive Health Recover1 B")
    Tk2dPlayFrame(gameObject=var "Recovery Blob", frame=0)
    GetPlayerDataInt(intName="health", storeValue=var "HP")
    ConvertIntToFloat(intVariable=var "HP", floatVariable=var "Blob X", everyFrame=false)
    FloatSubtract(floatVariable=var "Blob X", subtract=1, everyFrame=false, perSecond=false)  // disabled
    FloatMultiply(floatVariable=var "Blob X", multiplyBy=0.94, everyFrame=false)
    FloatAdd(floatVariable=var "Blob X", add=-10.32, everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Recovery Blob", vector=(unset), x=var "Blob X", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Recover 1
  }

  state Start Recovery {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset Timer
  }

  state Recover 1 {
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=var "Recover Time", tolerance=0, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on DAMAGE TAKEN → Start Recovery
    on NEXT → Recover 2
    on HERO HEALED → Again?
    on HERO HEALED FULL → Again?
  }

  state Recover 2 {
    SetFloatValue(floatVariable=var "Timer", floatValue=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Recovery Blob", animLibName="", clipName="Hive Health Recover2 B")
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=var "Recover Time", tolerance=0, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on DAMAGE TAKEN → Start Recovery
    on NEXT → Recover
    on HERO HEALED → Again?
    on HERO HEALED FULL → Again?
  }

  state Recover {
    SetMeshRenderer(gameObject=var "Recovery Blob", active=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddHealth", parameters=[1], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Again?
  }

  state Again? {
    SetMeshRenderer(gameObject=var "Recovery Blob", active=false)
    GetPlayerDataVariable(VariableName="health", StoreValue=var "HP")
    GetPlayerDataVariable(VariableName="CurrentMaxHealth", StoreValue=var "Max HP")
    IntCompare(integer1=var "HP", integer2=var "Max HP", equal=→"FALSE", lessThan=→"TRUE", greaterThan=→"FALSE", everyFrame=false)
    on FALSE → Idle
    on TRUE → Start Recovery
  }

  var Blob X: float = 0
  var Recover Time: float = 3
  var Timer: float = 0
  var HP: int = 0
  var Max HP: int = 0
  var Has Regen: bool = false
  var Recovery Blob: gameObject = <null>
  var Lifeblood Drips: gameObject = _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Blue_Health_Overblue_HUD_drips
}
