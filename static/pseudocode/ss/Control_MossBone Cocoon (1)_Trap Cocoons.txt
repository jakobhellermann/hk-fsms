# Control

fsm Control {
  start Init
  on EXTRACT → Extract  // from any state

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=2, MaxReactDelay=2.5, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on COCOON DESTROY → Get Silent
    on NEEDOLIN → Choose Dir
  }

  state Die {
    InstaDeath(target=Self, direction=0)
  }

  state Sway L {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.15, MaxReactDelay=0.3, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)

    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on END → Sway End
    on FINISHED → Sway R
    on COCOON DESTROY → Die
  }

  state Init {
    BoolTest(boolVariable=var "No Needolin React", isTrue=→"NO NEEDOLIN", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on NO NEEDOLIN → Inert
  }

  state Choose Dir {
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Sway L
    on R → Sway R
  }

  state Sway End {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    RotateTo(gameObject=var "Sprite", targetAngle=0, speed=3)  // disabled
    AnimateRotationToV2(gameObject=var "Sprite", fromValue=(unset), toValue=0, worldSpace=false, time=1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Idle
    on COCOON DESTROY → Die
  }

  state Sway R {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.25, MaxReactDelay=0.6, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    RotateTo(gameObject=var "Sprite", targetAngle=3, speed=3)  // disabled
    AnimateRotationToV2(gameObject=var "Sprite", fromValue=(unset), toValue=3, worldSpace=false, time=1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on END → Sway End
    on FINISHED → Sway L
    on COCOON DESTROY → Die
  }

  state Inert {
  }

  state Get Silent {
    SetCorpsePrefab(target=Self, corpsePrefab=<null>)  // disabled
    FindChild(gameObject=Self, childName="Corpse Cocoon(Clone)", storeResult=var "Corpse")  // disabled
    SetAudioClip(gameObject=var "Corpse", audioClip=<null>, autoPlay=false, stopOnExit=false)  // disabled
    on FINISHED → Die
  }

  state Extract {
    PlayAudioEvent(audioClip=enemy_death_sword_old (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    ActivateGameObject(gameObject=var "Cords", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on EXTRACT FINISH → Extract Kill
  }

  state Extract Kill {
    InstaDeath(target=Self, direction=0)
  }

  var No Needolin React: bool = false

  var Corpse: gameObject = <null>
  var Sprite: gameObject = Black Thread States/Normal World/Battle Scene/Trap Cocoons/MossBone Cocoon (1)/Sprite
  var Extract Point: gameObject = <null>
  var Cords: gameObject = Black Thread States/Normal World/Battle Scene/Trap Cocoons/MossBone Cocoon (1)/Cords
}
