# Control

fsm Control {
  start Pause

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Idle Tone")
    FindNamedChild(gameObject=var "Mask Half Back", storeResult=var "bg_glow")
    FindNamedChild(gameObject=var "Mask Half Front", storeResult=var "front_glow")
    FindNamedChild(gameObject=var "Mask Half Front", storeResult=var "Pt Slide")
    FindNamedChild(gameObject=var "song_gate_effects", storeResult=var "loom_floor_large_glow")
    FindNamedChild(gameObject=var "song_gate_effects", storeResult=var "mid_glow")
    FindNamedChild(gameObject=var "song_gate_effects", storeResult=var "mid_glow small")
    FindNamedChild(gameObject=var "song_gate_effects", storeResult=var "Crest_Shrine_light_spill")

    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Leave Range", storeResult=false, sendEvent="EMERGING", outOfRangeEvent="", everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Loop Idle Tone", volume=1, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Terrain Block", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on EMERGING → Hero Emerging
    on FINISHED → Idle
  }

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Init
  }

  state Idle {
    AudioStop(gameObject=Self, fadeTime=0)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.25, MaxReactDelay=0.25, None=(none), ActiveInner=→"NEEDOLIN START", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on NEEDOLIN START → Playing
  }

  state Hero Emerging {
    Translate(gameObject=var "Mask Half Front", vector=(unset), x=0.2, y=-9, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Mask Half Back", vector=(unset), x=-0.2, y=-9, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    TransitionToAudioSnapshot(snapshot=var "Audio Snapshot Open", transitionTime=0.5)
    on FINISHED → Wait to Close
  }

  state Playing {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=var "")
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.5)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"NEEDOLIN STOP", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=3, finishEvent=→"OPEN", realTime=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumStart())
    FadeColorFader(target=var "bg_glow", fadeType=UP, useChildren=false)
    FadeColorFader(target=var "front_glow", fadeType=UP, useChildren=false)
    FadeColorFader(target=var "loom_floor_large_glow", fadeType=UP, useChildren=false)
    FadeColorFader(target=var "mid_glow small", fadeType=UP, useChildren=false)
    on NEEDOLIN STOP → Needolin Stop
    on OPEN → Crack
  }

  state Needolin Stop {
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    FadeColorFader(target=var "bg_glow", fadeType=DOWN, useChildren=false)
    FadeColorFader(target=var "front_glow", fadeType=DOWN, useChildren=false)
    FadeColorFader(target=var "loom_floor_large_glow", fadeType=DOWN, useChildren=false)
    FadeColorFader(target=var "mid_glow small", fadeType=DOWN, useChildren=false)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Crack {
    AudioStopV2(gameObject=var "Audio Loop Idle Tone", fadeTime=0.25, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=weaver_gate_open_start (sfxstatic_assets_areaweaver.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Translate(gameObject=var "Mask Half Front", vector=(unset), x=0.1, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Mask Half Back", vector=(unset), x=-0.1, y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Crack", emit=0, resetIfPlaying=false)
    FadeColorFader(target=var "bg_glow", fadeType=DOWN, useChildren=false)
    FadeColorFader(target=var "front_glow", fadeType=DOWN, useChildren=false)
    ActivateGameObject(gameObject=var "mid_glow small", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "mid_glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Crest_Shrine_light_spill", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    ActivateGameObject(gameObject=var "Big Crest", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Big Crest", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="WEAVER GATE OPEN")
    on FINISHED → Slide
  }

  state Slide {

    TransitionToAudioSnapshot(snapshot=var "Audio Snapshot Open", transitionTime=0.5)
    on FINISHED → Slam
  }

  state Slam {
    PlayAudioEvent(audioClip=weaver_gate_open_end (sfxstatic_assets_areaweaver.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Terrain Block", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=var "Mask Half Back", active=false)
    SetSpriteRenderer(gameObject=var "Mask Half Front", active=false)
    PlayParticleEmitter(gameObject=var "Pt Slam", emit=0, resetIfPlaying=false)
    FadeColorFader(target=var "loom_floor_large_glow", fadeType=DOWN, useChildren=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="WEAVER GATE SLAM")
    on FINISHED → Wait to Close
  }

  state Slide Up {
    SetSpriteRenderer(gameObject=var "Mask Half Back", active=true)
    SetSpriteRenderer(gameObject=var "Mask Half Front", active=true)
    PlayAudioEvent(audioClip=weaver_gate_close_start (sfxstatic_assets_areaweaver.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

  }

    PlayAudioEvent(audioClip=weaver_gate_close_end (sfxstatic_assets_areaweaver.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)

    ActivateGameObject(gameObject=var "Big Crest", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled

    on CLOSE → Slide Up
  }
