# Check Height
# 2 location(s):
#   Cog_Dancers_boss / Dancer Control/Dancer A
#   Cog_Dancers_boss / Dancer Control/Dancer B

fsm Check Height {
  start Idle

  state Idle {
    on CHECK HEIGHT → Check
  }

  state Check {
    SetBoolValue(boolVariable=var "Top Row", boolValue=false, everyFrame=false)
    CheckYPosition(gameObject=Self, compareTo=var "Top Row Pos", compareToOffset=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"TOP ROW", everyFrame=false, space=0, activeBool=(unset))
    on FINISHED → Idle
    on TOP ROW → Top Row
  }

  state Top Row {
    SetBoolValue(boolVariable=var "Top Row", boolValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  var Top Row Pos: float = 10

  var Top Row: bool = false
}
