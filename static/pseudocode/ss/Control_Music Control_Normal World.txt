# Control
# 1 location(s):
#   Bone_East_10_Room / Black Thread States/Normal World/Music Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="rhinoRuckus", isTrue=→"RUCKUS", isFalse=→"FINISHED")
    on RUCKUS → State 1
    on FINISHED → Normal
  }

  state State 1 {
    on RUCKUS END → State 2
  }

  sequence state Normal {
    Wait(time=2, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=Shrine (audiocuesdynamic_assets_areabellareabone.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.5)
  }

  state State 2 {
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 3
  }

  state State 3 {
    ApplyMusicCue(musicCue=Shrine (audiocuesdynamic_assets_areabellareabone.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=3)
    SendEventToRegister(eventName="RUCKUS END")
  }
}
