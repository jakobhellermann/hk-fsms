# Control
# 1 location(s):

  on FLING UP → Fling Up  // from any state

    SetGravity2dScale(gameObject=Self, gravityScale=0)
    RandomFloat(min=0.8, max=1, storeResult=var "Scale")

    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Get Spin
  }

  state Launch {
    ArrayGetRandom(array=var "Sprites", storeValue=var "Sprite Crt", index=(unset), noRepeat=false, everyFrame=false)
    SetSpriteRendererSprite(gameObject=var "Sprite Object", sprite=var "Sprite Crt")
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Pt Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Launch Air
  }

  state Bounce {

    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=0.4, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=37, everyFrame=false)
    on FINISHED → Air
  }

  state Recycle {
    RecycleSelf()
  }

  state Get Spin {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "X Velocity", float2=-10, operation=Multiply, storeResult=var "Spin", everyFrame=false)
    on FINISHED → Launch
  }

  state Air {
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=-0.1, tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    Rotate(gameObject=var "Sprite Object", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckXPosition(gameObject=Self, compareTo=var "Wall L X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"WALL L", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    CheckXPosition(gameObject=Self, compareTo=var "Wall R X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"WALL R", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on END → End Pause
    on WALL L → Wall L
    on WALL R → Wall R
    on TINK UP → Tink Up
    on TINK DOWN → Tink Down
    on TINK RIGHT → Tink Right
    on TINK LEFT → Tink Left
  }

  state Wall L {
    SetPosition(gameObject=Self, vector=(unset), x=var "Wall L X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)

    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    FloatOperator(float1=var "X Velocity", float2=-20, operation=Multiply, storeResult=var "Spin", everyFrame=false)

    AudioPlayRandom(gameObject=var "Self", audioClips=[bell_ball_bounce_1 (sfxstatic_assets_areabell.bundle), bell_ball_bounce_2 (sfxstatic_assets_areabell.bundle)], weights=[0.5, 0.5], pitchMin=1.4, pitchMax=1.5)

    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false, space=0, activeBool=(unset))
    on FINISHED → Air
  }

  state Wall R {
    SetPosition(gameObject=Self, vector=(unset), x=var "Wall R X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=-0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    FloatOperator(float1=var "X Velocity", float2=-20, operation=Multiply, storeResult=var "Spin", everyFrame=false)
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false, space=0, activeBool=(unset))
    AudioPlayRandom(gameObject=var "Self", audioClips=[bell_ball_bounce_1 (sfxstatic_assets_areabell.bundle), bell_ball_bounce_2 (sfxstatic_assets_areabell.bundle)], weights=[0.5, 0.5], pitchMin=1.4, pitchMax=1.5)
    ActivateGameObject(gameObject=var "Slam Effect Wall R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Air
  }

  state End Pause {
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=-0.1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"END", everyFrame=true, space=0, activeBool=(unset))

    on END → Air
    on FINISHED → Recycle
  }

  state Tink Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=25, everyFrame=false)
    on FINISHED → Air
  }

  state Tink Up Slow {
    SetGravity2dScale(gameObject=Self, gravityScale=1.1)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=0.4, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    on FINISHED → Tink Up
  }

  state Launch Air {
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    Rotate(gameObject=var "Sprite Object", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on TINK UP → Tink Up Slow
    on LAND → Bounce
    on TINK LEFT → Launch L
    on TINK RIGHT → Launch R
  }

  state Launch L {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Velocity", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=-1, everyFrame=false)
    on FINISHED → Launch L Min
  }

  state Launch R {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "X Velocity", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=-1, everyFrame=false)
    on FINISHED → Launch R Min
  }

  }

  state Tink Right {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)

    RandomFloat(min=0.8, max=1, storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=var "Scale", everyFrame=false, lateUpdate=false)
    GetOwner(storeGameObject=var "Self")

    SetSpriteRendererSprite(gameObject=var "Sprite Object", sprite=var "Sprite Crt")
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Pt Dust", emit=0, resetIfPlaying=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1.1)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "X Velocity", float2=-100, operation=Multiply, storeResult=var "Spin", everyFrame=false)

    on TINK DOWN → Tink Down
  }

  state Tink R Min {
    FloatClamp(floatVariable=var "X Velocity", minValue=15, maxValue=100, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    on FINISHED → Air
  }

  state Tink L Min {
    FloatClamp(floatVariable=var "X Velocity", minValue=-100, maxValue=-15, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    on FINISHED → Air
  }

  state Launch L Min {

    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    on FINISHED → Launch Air
  }

  state Launch R Min {
    FloatClamp(floatVariable=var "X Velocity", minValue=15, maxValue=100, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)

  }

  var Ground Y: float = 7.5
  var Wall L X: float = 37.7
  var Wall R X: float = 67.3
  var X Velocity: float = 0
  var Spin: float = 0
  var Scale: float = 0
  var Pt Dust: gameObject = Spinner AtkBell/Pt Dust
  var Pt Rock: gameObject = Spinner AtkBell/Pt Rock
  var Self: gameObject = <null>

  var Slam Effect Wall R: gameObject = Spinner AtkBell/Slam Effect Wall R

}
