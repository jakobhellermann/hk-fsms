# Control Effects
# 1 location(s):
#   Coral_03 / lift mechanism cogs

fsm Control Effects {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on OPEN → Antic
  }

  state Antic {
    PlayAudioEvent(audioClip=dock_chain_rock_plat_start_move (sfxstatic_assets_areadocks.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    on FINISHED → Move
  }

  state Move {
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StopJitter())
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartAnimation())
    Wait(time=3.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StopAtCurrentPoint())
  }
}
