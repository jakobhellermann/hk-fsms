# Control
# 1 location(s):
#   Bone_01 / Battle Scene/End Scene/Crumble Gate (1)

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Static Sprites", storeResult=var "Static Sprites")
    on FINISHED → Idle
  }

  state Idle {
    on BREAK → Antic
  }

  state Antic {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterInState(gameObject=var "Antic Dust")
    ObjectJitter(gameObject=var "Sprite 1", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Sprite 2", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Sprite 3", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=magnet_platform_v2_antic (sfxstatic_assets_areabone.bundle))
    on FINISHED → Break
  }

  state Break {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite 3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Static Sprites", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=magnet_platform_v2_break (sfxstatic_assets_areabone.bundle))
  }

  var Antic Dust: gameObject = Battle Scene/End Scene/Crumble Gate (1)/Antic Dust
  var Break Effects: gameObject = Battle Scene/End Scene/Crumble Gate (1)/Break Effects
  var Sprite 1: gameObject = Battle Scene/End Scene/Crumble Gate (1)/Sprites/floor2 (2)
  var Sprite 2: gameObject = Battle Scene/End Scene/Crumble Gate (1)/Sprites/floor2 (3)
  var Sprite 3: gameObject = Battle Scene/End Scene/Crumble Gate (1)/Sprites/GameObject
  var Static Sprites: gameObject = <null>
}
