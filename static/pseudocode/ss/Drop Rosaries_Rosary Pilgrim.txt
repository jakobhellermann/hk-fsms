# Drop Rosaries
# 1 location(s):
#   Shadow_23 / Rosary Pilgrim

fsm Drop Rosaries {
  start Init

  state Idle {
    on TOOK DAMAGE → Fling Body
    on WHIP F → Fling Whip F
    on WHIP B → Fling Whip B
  }

  state Do Fling {
    IntCompare(integer1=var "Value", integer2=20, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Geo Small (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Fling Point", position=(unset), spawnMin=1, spawnMax=1, speedMin=30, speedMax=40, angleMin=var "Angle Min", angleMax=var "Angle Max", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    IntOperator(integer1=var "Value", integer2=1, operation=Subtract, storeResult=var "Value", everyFrame=false)
    SetGeoDrop(target=Self, smallGeo=var "Value", mediumGeo=(unset), largeGeo=(unset))
    on END → Stop Dropping
    on FINISHED → Idle
  }

  state Init {
    SetGeoDrop(target=Self, smallGeo=var "Value", mediumGeo=(unset), largeGeo=(unset))
    on FINISHED → Idle
  }

  state Stop Dropping {
  }

  state Set Death Rosaries {
    SetGeoDrop(target=Self, smallGeo=var "Value", mediumGeo=(unset), largeGeo=(unset))
  }

  state Fling Body {
    SendRandomEventV4(events=[→"CANCEL", (none)], weights=[0.65, 0.35], eventMax=[3, 1], missedMax=[1, 3], activeBool=(unset))
    SetGameObject(variable=var "Fling Point", gameObject=var "Self", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Min", floatValue=40, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=140, everyFrame=false)
    on FINISHED → Do Fling
    on CANCEL → Idle
  }

  state Fling Whip F {
    SendRandomEventV4(events=[→"CANCEL", (none)], weights=[0.9, 0.1], eventMax=[9, 1], missedMax=[1, 9], activeBool=(unset))
    SetGameObject(variable=var "Fling Point", gameObject=var "Whip Point F", everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FLING L", negativeEvent=→"FLING R", space=0)
    on CANCEL → Idle
    on FLING L → Fling L
    on FLING R → Fling R
  }

  state Fling Whip B {
    SendRandomEventV4(events=[→"CANCEL", (none)], weights=[0.9, 0.1], eventMax=[9, 1], missedMax=[1, 9], activeBool=(unset))
    SetGameObject(variable=var "Fling Point", gameObject=var "Whip Point B", everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FLING R", negativeEvent=→"FLING L", space=0)
    on CANCEL → Idle
    on FLING L → Fling L
    on FLING R → Fling R
  }

  state Fling L {
    SetFloatValue(floatVariable=var "Angle Min", floatValue=110, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=140, everyFrame=false)
    on FINISHED → Do Fling
  }

  state Fling R {
    SetFloatValue(floatVariable=var "Angle Min", floatValue=40, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Max", floatValue=70, everyFrame=false)
    on FINISHED → Do Fling
  }

  var Value: int = 75

  var Angle Min: float = 0
  var Angle Max: float = 0
  var Amount: int = 0
  var Self: gameObject = Rosary Pilgrim
  var Fling Point: gameObject = <null>
  var Whip Point F: gameObject = Rosary Pilgrim/Whip Point F
  var Whip Point B: gameObject = Rosary Pilgrim/Whip Point B
}
