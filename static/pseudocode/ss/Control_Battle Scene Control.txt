# Control
# 1 location(s):
#   Library_02 / GameObject/Battle Scene Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Battle Scene Acolyte")
    FindNamedChild(gameObject=Self, storeResult=var "Return Acolytes")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="completedLibraryEntryBattle", isTrue=(none), isFalse=→"FIRST BATTLE")
    PlayerDataBoolTest(boolName="scholarAcolytesInLibrary_02", isTrue=(none), isFalse=→"NONE")
    PlayerDataBoolTest(boolName="completedLibraryAcolyteBattle", isTrue=→"ACOLYTE RETURN", isFalse=→"SECOND BATTLE")
    on FIRST BATTLE → First Battle
    on NONE → None
    on SECOND BATTLE → Second Battle
    on ACOLYTE RETURN → Acolyte Return
  }

  state First Battle {
    ActivateGameObject(gameObject=var "Battle Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Battle Scene Acolyte", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return Acolytes", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state None {
    ActivateGameObject(gameObject=var "Battle Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Battle Scene Acolyte", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return Acolytes", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Second Battle {
    ActivateGameObject(gameObject=var "Battle Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Battle Scene Acolyte", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return Acolytes", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Acolyte Return {
    ActivateGameObject(gameObject=var "Battle Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Battle Scene Acolyte", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return Acolytes", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Battle Scene: gameObject = <null>
  var Battle Scene Acolyte: gameObject = <null>
  var Return Acolytes: gameObject = <null>
}
