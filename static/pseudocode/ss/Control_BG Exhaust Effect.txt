# Control
# 1 location(s):
#   Dust_09 / BG Exhaust Effect

fsm Control {
  start Init

  state Init {
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=2, finishEvent=(none), realTime=false)
    on FINISHED → Play
  }

  state Play {
    PlayParticleEmitter(gameObject=var "Pt Dust", emit=0, resetIfPlaying=false)
    Wait(time=2, finishEvent=(none), realTime=false)
    RandomFloat(min=0.85, max=1.15, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=organ_steam_attack 1 (sfxstatic_assets_areaorgan.bundle))
    RandomFloat(min=0.75, max=1.25, storeResult=var "Pitch")  // disabled
    SetAudioPitch(gameObject=var "Organ Player", pitch=var "Pitch", everyFrame=false)  // disabled
    AudioPlaySimple(gameObject=var "Organ Player", volume=1, oneShotClip=organ_note (sfxstatic_assets_areaorgan.bundle))  // disabled
    on FINISHED → Pause
  }

  var Pitch: float = 0
  var Pt Dust: gameObject = BG Exhaust Effect/Pt Dust (1)
  var Organ Player: gameObject = BG Exhaust Effect/Organ Player
}
