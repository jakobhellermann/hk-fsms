# Control
# 1 location(s):
#   Ant_04 / Hunter Sickle Trap (3)

fsm Control {
  start Init

  state Init {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Rotation", space=1, everyFrame=false)
    FloatOperator(float1=var "Start Rotation", float2=5, operation=Subtract, storeResult=var "Antic Rotation", everyFrame=false)
    FloatOperator(float1=var "Start Rotation", float2=5, operation=Add, storeResult=var "Tink Rotation", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Tink React", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on ACTIVATE → Pre Delay
    on TINK → Tink Up
  }

  state Swipe {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sickle Blur")
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    EaseFloat(fromValue=var "Antic Rotation", toValue=var "Target Angle", floatVariable=var "Angle", time=0.25, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=1, everyFrame=true, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Slam
  }

  state Slam {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Slam", resetTimeIfPlaying=false, stopOnStateExit=true)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="SICKLE SLAM")
    on FINISHED → Reset
  }

  state Reset {
    BoolTest(boolVariable=var "No Reset", isTrue=→"END", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Quick Reset", isTrue=→"QUICK", isFalse=(none), everyFrame=false)
    EaseFloat(fromValue=var "Antic Rotation", toValue=var "Start Rotation", floatVariable=var "Angle", time=0.5, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=1, everyFrame=true, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sickle")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Idle
    on END → End
    on QUICK → Quick Reset
  }

  state Antic {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=<null>, WaitForEndOfClip=false, finishedEvent=(none))
    Wait(time=var "OG Delay", finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=false)
    EaseFloat(fromValue=var "Start Rotation", toValue=var "Antic Rotation", floatVariable=var "Angle", time=0.25, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=(none), realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=1, everyFrame=true, lateUpdate=false)
    on FINISHED → Swipe
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pre Delay {
    FloatOperator(float1=var "Delay", float2=var "OG Delay", operation=Subtract, storeResult=var "New Delay", everyFrame=false)
    Wait(time=var "New Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  state Tink Up {
    EaseFloat(fromValue=var "Start Rotation", toValue=var "Tink Rotation", floatVariable=var "Angle", time=0.1, speed=(unset), delay=(unset), easeType=4, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=1, everyFrame=true, lateUpdate=false)
    on FINISHED → Tink Down
    on ACTIVATE → Pre Delay
  }

  state Tink Down {
    EaseFloat(fromValue=var "Tink Rotation", toValue=var "Start Rotation", floatVariable=var "Angle", time=0.25, speed=(unset), delay=(unset), easeType=22, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=1, everyFrame=true, lateUpdate=false)
    on FINISHED → Idle
    on ACTIVATE → Pre Delay
  }

  state Quick Reset {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Rotation", space=1, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sickle")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Idle
  }

  var Delay: float = 0.6
  var Target Angle: float = -4
  var No Reset: bool = false
  var Quick Reset: bool = false

  var Angle: float = 0
  var Start Rotation: float = 0
  var Antic Rotation: float = 0
  var OG Delay: float = 0.4
  var New Delay: float = 0
  var Tink Rotation: float = 0
  var Pt Antic: gameObject = Hunter Sickle Trap (3)/Pt Antic
  var Pt Slam: gameObject = Hunter Sickle Trap (3)/Pt Slam
  var Slam Effect: gameObject = Hunter Sickle Trap (3)/Slam Effect
  var Tink React: gameObject = Hunter Sickle Trap (3)/Tink React
}
