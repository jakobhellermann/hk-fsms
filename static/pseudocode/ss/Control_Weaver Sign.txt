# Control
# 1 location(s):
#   Weave_02 / Group:2/Weaver Sign

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on FINISHED → Inactive
    on ACTIVATED → Activated
  }

  state Activated {
    SendMessage(gameObject=var "Sign Art", delivery=SendMessage, options=1, functionCall=UnpauseFade())
    PlayParticleEmitter(gameObject=var "Pt Wisp", emit=0, resetIfPlaying=false)
  }

  state Inactive {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    SetSpriteRenderer(gameObject=var "Sign Art", active=false)
    on ENTER → Activate
  }

  sequence state Activate {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Wisp", emit=0, resetIfPlaying=false)
    SetSpriteRenderer(gameObject=var "Sign Art", active=true)
    EaseSpriteColor(target=var "Sign Art", fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.29803923), disableSpriteRendererIfTransparent=false, time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseSpriteColor(target=var "Sign Art", fromValue=(1, 1, 1, 0.29803923), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseSpriteColor(target=var "Sign Art", fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.29803923), disableSpriteRendererIfTransparent=false, time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseSpriteColor(target=var "Sign Art", fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseSpriteColor(target=var "Sign Art", fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.29803923), disableSpriteRendererIfTransparent=false, time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseSpriteColor(target=var "Sign Art", fromValue=(1, 1, 1, 0.297), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseSpriteColor(target=var "Sign Art", fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.497), disableSpriteRendererIfTransparent=false, time=0.75, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Sign Start
  }

  state Sign Start {
    SendMessage(gameObject=var "Sign Art", delivery=SendMessage, options=1, functionCall=UnpauseFade())
  }

  var Activated: bool = false
  var Sign Art: gameObject = Group:2/Weaver Sign/Sign Art
  var Pt Wisp: gameObject = Group:2/Weaver Sign/Pt Wisp
}
