# Control
# 1 location(s):
#   Song_17 / March Group Control

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "March Group L")
    FindNamedChild(gameObject=Self, storeResult=var "March Group R")
    FindNamedChild(gameObject=Self, storeResult=var "Handmaiden Meet")
    on FINISHED → Wait For Hero Pos
  }

  state Dir {
    PlayerDataBoolTest(boolName="marionettesMet", isTrue=(none), isFalse=→"HANDMAIDEN")  // disabled
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → March L
    on R → March R
  }

  state March L {
    PlayerDataBoolTest(boolName="song_17_clearedOut", isTrue=→"NONE", isFalse=(none))
    ActivateGameObject(gameObject=var "March Group L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "March Group R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Handmaiden Meet", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on NONE → None
  }

  state March R {
    PlayerDataBoolTest(boolName="song_17_clearedOut", isTrue=→"NONE", isFalse=(none))
    ActivateGameObject(gameObject=var "March Group R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "March Group L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Handmaiden Meet", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on NONE → None
  }

  state Handmaiden {
    ActivateGameObject(gameObject=var "March Group R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "March Group L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Handmaiden Meet", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="song_17_clearedOut", value=true)
  }

  state None {
    ActivateGameObject(gameObject=var "March Group R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "March Group L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Handmaiden Meet", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait For Hero Pos {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Dir
  }

  var March Group L: gameObject = <null>
  var March Group R: gameObject = <null>
  var Handmaiden Meet: gameObject = <null>
}
