# needolin_capture_timer
# 2 location(s):
#   Cog_Dancers_boss / Dancer Control
#   Coral_11 / Black Thread States Thread Only Variant/Normal World/Boss Scene

// uses template: needolin_capture_timer
fsm needolin_capture_timer {
  start Idle
  on NEEDOLIN CAPTURE RESET → Idle  // from any state
  on NEEDOLIN CAPTURE COOLDOWN → Cooldown  // from any state

  state Idle {
    SetFloatValue(floatVariable=var "Needolin Timer", floatValue=0, everyFrame=false)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=0, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SHIFT", ActiveOuter=(none), IgnoreNeedolinRange=true, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SHIFT → Needolin
  }

  state Needolin {
    CheckHeroPerformanceRegionV2(Target=Self, Radius=0, MinReactDelay=0, MaxReactDelay=0, None=→"SHIFT", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FloatAdd(floatVariable=var "Needolin Timer", add=1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Needolin Timer", minValue=0, maxValue=100, everyFrame=true)
    on SHIFT → Idle
  }

  state Cooldown {
    Wait(time=var "Cooldown Time", finishEvent=(none), realTime=false)
    SetFloatValue(floatVariable=var "Needolin Timer", floatValue=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Needolin Timer: float = 0
  var Cooldown Time: float = 5
}
