# Hero Damaged Effects
# 2 location(s):
#   coremanagers_assets__gamemanager.bundle / _GameManager/AudioManager
#   Menu_Title / _GameManager/AudioManager

fsm Hero Damaged Effects {
  start Idle
  on DAMAGE LOWPASS → To Lowpass  // from any state

  state Idle {
    EnableFsmSelf(SetEnabled=false)
  }

  state To Lowpass {
    GetPlayerDataInt(intName="health", storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Damaged (audioobjects_assets_all.bundle), transitionTime=0.05)
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Back
    on CANCEL → Idle
  }

  state Fade Back {
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=2)
    EnableFsmSelf(SetEnabled=false)
  }

  var HP: int = 0
}
