// uses template: ward_corpse_projectile
fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Corpse Folder")
    FindNamedChild(gameObject=Self, storeResult=var "Slam Effect")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Ground")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Fly")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Slam")
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    on FINISHED → Dormant
  }

  state Dormant {
    SetPosition2d(gameObject=Self, vector=(unset), x=-100, y=-10, space=0, everyFrame=false, lateUpdate=false)
    PlayRandomAudioClipTableV3(Table=Dreg_Husk_death_launch (sfxdynamic_assets_areaward.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    SetParent(gameObject=Self, parent=var "Corpse Folder", resetLocalPosition=false, resetLocalRotation=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    StopParticleEmitter(gameObject=var "")
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    on LAUNCH → Launch
  }

  state Launch {
    AddHP(target=Self, amount=0, healToMax=true)
    SetLayer(gameObject=Self, layer=Enemies(11))
    PlayAudioEvent(audioClip=slab_fly_large_spit (sfxstatic_assets_areaslab.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetMeshRenderer(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Pt Fly", emit=0, resetIfPlaying=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spawn Roll")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetGravity2dScale(gameObject=Self, gravityScale=var "Gravity")
    on WALL → Impact
    on TOOK HEAVY DAMAGE → Heavy Damage
  }

  state Impact {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.01, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayAudioEvent(audioClip=enemy_death_sword_old (sfxstatic_assets_shared.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=false_knight_land_1st_time (sfxstatic_assets_enemyheavy.bundle), pitchMin=1.2, pitchMax=1.5, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    SetVelocityByScale(gameObject=Self, speed=8, ySpeed=10, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    PlayParticleEmitter(gameObject=var "Pt Slam", emit=0, resetIfPlaying=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.6886792, 0.6886792, 0.6886792, 1), everyframe=false)
    PlayRandomAudioClipTableV3(Table=Dreg_Husk_death_land (sfxdynamic_assets_areaward.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    on LAND → Land
  }

  state Land {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Ground", emit=0, resetIfPlaying=false)
    on FINISHED → Dormant
  }

  state Heavy Damage {
    PlayAudioEvent(audioClip=enemy_death_sword_old (sfxstatic_assets_shared.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.01, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayAudioEvent(audioClip=false_knight_land_1st_time (sfxstatic_assets_enemyheavy.bundle), pitchMin=1.2, pitchMax=1.5, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    SetVelocityByScale(gameObject=Self, speed=0, ySpeed=-10, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    PlayParticleEmitter(gameObject=var "Pt Slam", emit=0, resetIfPlaying=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.6901961, 0.6901961, 0.6901961, 1), everyframe=false)
    on LAND → Land
  }

  var Ground Y: float = 5.9
  var Gravity: float = 0.4
  var Slam Effect: gameObject = <null>
  var Pt Ground: gameObject = <null>
  var Corpse Folder: gameObject = <null>
  var Pt Fly: gameObject = <null>
  var Pt Slam: gameObject = <null>
}
