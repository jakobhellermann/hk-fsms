# FSM
# 1 location(s):
#   Dock_10 / Sauna Control

fsm FSM {
  start Init

  state Init {
    FadeNestedFadeGroup(Target=var "Fog Group", ToAlpha=var "Fog Alpha Min", FadeTime=0)
    StopParticleEmitter(gameObject=var "Drip Particles")
    on FINISHED → Idle
  }

  state Idle {
    WaitRandom(timeMin=15, timeMax=20, finishEvent=(none), realTime=false)
    FadeNestedFadeGroup(Target=var "Fog Group", ToAlpha=var "Fog Alpha Min", FadeTime=10)
    AudioStopV2(gameObject=var "Audio Loop Steam", fadeTime=1, cancelOnEarlyExit=false)
    on FINISHED → Drip Sequence
    on TEST → Drip Sequence
  }

  sequence state Drip Sequence {
    AudioPlaySimple(gameObject=var "Audio Loop Water", volume=1, oneShotClip=<null>)
    PlayParticleEmitter(gameObject=var "Drip Particles", emit=0, resetIfPlaying=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    AudioPlaySimple(gameObject=var "Audio Loop Steam", volume=1, oneShotClip=<null>)
    Wait(time=1, finishEvent=(none), realTime=false)
    FadeNestedFadeGroup(Target=var "Fog Group", ToAlpha=var "Fog Alpha Max", FadeTime=5)
    Wait(time=5, finishEvent=(none), realTime=false)
    AudioStopV2(gameObject=var "Audio Loop Water", fadeTime=0.5, cancelOnEarlyExit=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  var Fog Alpha Min: float = 0.55
  var Fog Alpha Max: float = 1

  var Drip Particles: gameObject = Sauna Control/Drips
  var Fog Group: gameObject = Sauna Control/Fog Quads
  var Audio Loop Water: gameObject = Sauna Control/Audio Loop Water
  var Audio Loop Steam: gameObject = Sauna Control/Audio Steam
}
