# Control
# 1 location(s):
#   Dust_Maze_01 / Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Sprung", isTrue=→"SPRUNG", isFalse=(none), everyFrame=false)
    on ACTIVATED → Activated
    on FINISHED → Idle
    on SPRUNG → Already Sprung
  }

  state Activated {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RandomFloatEither(value1=-100, value2=100, storeResult=var "Tilt")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Tilt", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Snapped")
    ActivateGameObject(gameObject=var "Phys", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RandomFloat(min=0.007, max=0.0075, storeResult=var "zPos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "zPos", space=0, everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.75686276, 0.75686276, 0.75686276, 1), everyframe=false)
    on FINISHED → Sprung
  }

  state Idle {
    SetBoolValue(boolVariable=var "Grabbing Hero", boolValue=false, everyFrame=false)
    WaitRandom(timeMin=3, timeMax=5, finishEvent=→"GLEAM", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"SPRING", storeCollider=var "Collider")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Corpse(26), sendEvent=→"SPRING", storeCollider=var "Collider")
    CheckTrackTriggerCountV2(Target=Self, Count=0, Test=MoreThan, EveryFrame=true, SetBool=(unset), SuccessEvent=→"SPRING", FailEvent=(none))
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"SPRING", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Enemy Attack(22), sendEvent=→"SPRING", storeCollider=var "Collider")
    on GLEAM → Gleam
    on SPRING → Mite?
    on TINK → Antic
  }

  state Gleam {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Gleam")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Idle
  }

  state Antic {
    PlayAudioEvent(audioClip=mite_trap_trap (sfxstatic_assets_areabellwayareadustmazeareagreymoorareasong.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Step")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero?
  }

  state Snap 1 {
    ActivateGameObject(gameObject=var "Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=15, everyFrame=false)
    ActivateGameObject(gameObject=var "Damager 1", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Damager 1", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    InstaDeath(target=var "Mite", direction=0)
    on FINISHED → Snap 2
  }

  state Snap 2 {
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Damager 2", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Snap", emit=0, resetIfPlaying=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Snap 3
  }

  state Snap 3 {
    ActivateGameObject(gameObject=var "Damager 3", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    RandomFloatEither(value1=-100, value2=100, storeResult=var "Tilt")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Tilt", everyFrame=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=var "Phys", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RandomFloat(min=0.007, max=0.0075, storeResult=var "zPos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "zPos", space=0, everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.75686276, 0.75686276, 0.75686276, 1), everyframe=false)
    BoolTest(boolVariable=var "Grabbing Hero", isTrue=→"HERO", isFalse=(none), everyFrame=false)
    on HERO → Hero End
    on FINISHED → Sprung
  }

  state Sprung {
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK DOWN → Tink D
    on TINK UP → Tink U
  }

  state Tink L {
    RandomFloat(min=115, max=125, storeResult=var "Angle")
    SetFloatValue(floatVariable=var "Speed", floatValue=5, everyFrame=false)
    SetFloatValue(floatVariable=var "Angular Velocity", floatValue=0, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tinked {
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=DoSpin())  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Bounce Anim")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Angular Velocity", everyFrame=false)
    on FINISHED → Sprung
  }

  state Tink R {
    RandomFloat(min=55, max=65, storeResult=var "Angle")
    SetFloatValue(floatVariable=var "Speed", floatValue=5, everyFrame=false)
    SetFloatValue(floatVariable=var "Angular Velocity", floatValue=0, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tink D {
    RandomFloat(min=260, max=280, storeResult=var "Angle")
    SetFloatValue(floatVariable=var "Speed", floatValue=15, everyFrame=false)
    RandomFloat(min=-20, max=20, storeResult=var "Angular Velocity")
    on FINISHED → Tinked
  }

  state Tink U {
    RandomFloat(min=80, max=100, storeResult=var "Angle")
    SetFloatValue(floatVariable=var "Speed", floatValue=15, everyFrame=false)
    RandomFloat(min=-35, max=35, storeResult=var "Angular Velocity")
    on FINISHED → Tinked
  }

  state Mite? {
    GetTag(gameObject=var "Collider", storeResult=var "Tag", everyFrame=false)
    StringCompare(stringVariable=var "Tag", compareTo="Wraith", equalEvent=→"CANCEL", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Tag", compareTo="Target", equalEvent=→"MITE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Antic
    on MITE → Snap Mite
    on CANCEL → Cancel
  }

  state Snap Mite {
    SetStringValue(stringVariable=var "Anim", stringValue="Snap Mite", everyFrame=false)
    SetStringValue(stringVariable=var "Bounce Anim", stringValue="Bounce Mite", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="TRAP", delay=0, everyFrame=false)
    SetGameObject(variable=var "Mite", gameObject=var "Collider", everyFrame=false)
    SetBoolValue(boolVariable=var "Grabbing Mite", boolValue=false, everyFrame=false)
    on FINISHED → Antic Q
  }

  state Already Sprung {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Snap Mite")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RandomFloatEither(value1=-100, value2=100, storeResult=var "Tilt")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Tilt", everyFrame=false)
    ActivateGameObject(gameObject=var "Phys", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetStringValue(stringVariable=var "Bounce Anim", stringValue="Bounce Mite", everyFrame=false)
    RandomFloat(min=0.007, max=0.0075, storeResult=var "zPos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "zPos", space=0, everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.75686276, 0.75686276, 0.75686276, 1), everyframe=false)
    on FINISHED → Sprung
  }

  state Cancel {
    SetGameObject(variable=var "Collider", gameObject=<null>, everyFrame=false)
    SetStringValue(stringVariable=var "Tag", stringValue="", everyFrame=false)
    on FINISHED → Idle
  }

  state Hero? {
    BoolTest(boolVariable=var "Grabbing Mite", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    CheckTrackTriggerCountV2(Target=Self, Count=0, Test=MoreThan, EveryFrame=false, SetBool=(unset), SuccessEvent=→"HERO", FailEvent=(none))
    on HERO → Hero
    on FINISHED → Snap 1
  }

  state Hero {
    SetBoolValue(boolVariable=var "Grabbing Hero", boolValue=true, everyFrame=false)
    RunFSM(fsmTemplateControl=template=-5200671031615995940 in[Bypass Checks<-false, Dont Deal Damage<-false, No Hero Invuln<-false, Grab Effect Point<-var "Grab Effect Point"], finishEvent=(none), everyFrame=false)
    SetStringValue(stringVariable=var "Anim", stringValue="Snap Hero", everyFrame=false)
    on FINISHED → Snap 1
    on CANCEL → No Hero
  }

  state Hero End {
    tk2dPlayAnimAfterPreviousComplete(Target=Self, AnimName="Snap Hero Loop", StoreDidPlay=(unset))
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero End Release
  }

  state Hero End Release {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Snapped", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Grab Point", xOffset=0, yOffset=0, everyFrame=false)
    RunFSM(fsmTemplateControl=template=-7631507063665673418, finishEvent=(none), everyFrame=false)
    on FINISHED → Sprung
  }

  state No Hero {
    SetBoolValue(boolVariable=var "Grabbing Hero", boolValue=false, everyFrame=false)
    on FINISHED → Snap 1
  }

  state Antic Q {
    PlayAudioEvent(audioClip=mite_trap_trap (sfxstatic_assets_areabellwayareadustmazeareagreymoorareasong.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Step")
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hero?
  }

  var Sprung: bool = false

  var Angle: float = 0
  var Speed: float = 0
  var Tilt: float = 0
  var zPos: float = 0
  var Angular Velocity: float = 0
  var Activated: bool = false
  var Grabbing Hero: bool = false
  var Grabbing Mite: bool = false
  var Anim: string = "Snap"
  var Bounce Anim: string = "Bounce"
  var Tag: string = ""
  var Collider: gameObject = <null>
  var Damager 1: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Damager 1
  var Damager 2: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Damager 2
  var Damager 3: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Damager 3
  var Detector: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Detector
  var Enemy Damager 1: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Enemy Damager 1
  var Mite: gameObject = <null>
  var Phys: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Phys
  var Pt Snap: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Pt Snap
  var Slam Effect: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Slam Effect
  var Grab Effect Point: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Grab Effect Point
  var Grab Point: gameObject = Mist Maze Controller/Trap Sets/Trap Set (3)/Mite Trap/Grab Point
}
