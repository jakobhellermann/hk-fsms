# Slash Dust Effects
# 1 location(s):
#   localpoolprefabs_assets_areaaqueduct.bundle / Vampire Gnat

fsm Slash Dust Effects {
  start Init
  on SLASH STOP → Effect Stop  // from any state

  state Init {
    on FINISHED → Effect Stop
  }

  state Idle {
    RayCast2dV2(fromGameObject=var "SlashDust Pt", fromPosition=(unset), direction=(0, -1), space=0, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=→"PLAY", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on PLAY → Play
  }

  state Play {
    RayCast2dV2(fromGameObject=var "SlashDust Pt", fromPosition=(unset), direction=(0, -1), space=0, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"END", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Hit Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Pt SlashDust", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetPosition2D(GameObject=var "Pt SlashDust", Vector=var "Hit Point", X=0, Y=0, Space=0, EveryFrame=true)
    PlayParticleEmitterChildren(gameObject=var "Pt SlashDust", resetTimeIfPlaying=false, stopOnStateExit=true)
    on END → Idle
  }

  state Effect Stop {
    on SLASH START → Idle
  }

  var X Scale: float = 0
  var Hit Point: vector2 = (0, 0)
  var Pt SlashDust: gameObject = Vampire Gnat/Pt SlashDust
  var SlashDust Pt: gameObject = Vampire Gnat/SlashDust Pt
  var Self: gameObject = Vampire Gnat
}
