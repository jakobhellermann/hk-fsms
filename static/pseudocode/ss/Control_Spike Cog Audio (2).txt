fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on TICK → Distance Check
    on TICK HALF → Distance Check
  }

  state Distance Check {
    GetDistance(gameObject=Self, target=var "MainCamera", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=var "Sound Distance", tolerance=0, equal=(none), lessThan=→"SFX", greaterThan=(none), everyFrame=false)
    on SFX → Sfx
    on FINISHED → Idle
  }

  state Sfx {
    PlayRandomAudioClipTable(Table=Pendulum Cog (herodynamic_assets_all.bundle), AudioPlayerPrefab=Audio Player Actor VeryShortRange@AudioSource (audioobjects_assets_all.bundle), SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Idle
  }

  var Distance: float = 0
  var Sound Distance: float = 43
}
