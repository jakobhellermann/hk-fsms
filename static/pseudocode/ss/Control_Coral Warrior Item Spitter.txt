# Control
# 1 location(s):
#   Coral_39 / Coral Warrior Item Spitter

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Centipede Botherer")
    FindNamedChild(gameObject=Self, storeResult=var "Pit Pickup")
    on FINISHED → Idle
  }

  state Idle {
    on WARRIOR EATEN → Pause
  }

  state Pause {
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Bother Centipedes
  }

  state Bother Centipedes {
    ActivateGameObject(gameObject=var "Centipede Botherer", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Spit Item
  }

  state Spit Item {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=mini_mawlek_spit (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Pit Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Centipede Botherer: gameObject = <null>
  var Pit Pickup: gameObject = <null>
}
