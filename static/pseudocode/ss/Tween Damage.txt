// component disabled: this FSM does not run until something enables it
fsm Tween Damage {
  start Normal Dmg

  state Normal Dmg {
    CallMethodProper(gameObject=Self, behaviour="DamageEnemies", methodName="SetNailDamageMultiplier", parameters=[0.7], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Low Dmg
  }

  state Low Dmg {
    CallMethodProper(gameObject=Self, behaviour="DamageEnemies", methodName="SetNailDamageMultiplier", parameters=[0.25], storeResult=(unset var), EveryFrame=false)
  }
}
