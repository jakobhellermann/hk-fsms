# FSM
# 1 location(s):
#   Under_03d / Black Thread States/Normal World/Hero Treadmill/needle behaviour

fsm FSM {
  start Init

  state Below {
    on NEEDLE ABOVE → Above
  }

  state Above {
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on NEEDLE BELOW → Below
    on FINISHED → Spit Antic
  }

  state Spit Antic {
    ObjectJitterLocal(gameObject=Self, x=0.01, y=0.01, z=(unset), resetPosOnStateExit=true, limitFps=0)
    WaitRandom(timeMin=0.7, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on NEEDLE BELOW → Below
    on FINISHED → Spit
  }

  state Spit {
    SendEventToRegister(eventName="WORK COMPLETE")
    CallMethodProper(gameObject=Black Thread States/Normal World/Hero Treadmill, behaviour="HeroTreadmill", methodName="DropNeedle", parameters=[], storeResult=(unset var), EveryFrame=false)
    IntOperator(integer1=var "Value", integer2=1, operation=Add, storeResult=var "Value", everyFrame=false)
    BoolTest(boolVariable=var "Given Rosary", isTrue=(none), isFalse=→"ROSARY", everyFrame=false)
    IntCompare(integer1=var "Value", integer2=var "Max Payout", equal=(none), lessThan=(none), greaterThan=→"FAIL", everyFrame=false)
    SendRandomEventFair(Events=[{SendEvent=→"ROSARY", Probability=0.8333333}, {SendEvent=→"FAIL", Probability=0.16666667}], TrackingArray=var "Spit Event Tracking", MissedMultiplier=2)
    on ROSARY → Spawn Rosary
    on FAIL → Fail
  }

  state Spawn Rosary {
    SetBoolValue(boolVariable=var "Given Rosary", boolValue=true, everyFrame=false)
    SpawnProjectileV2(SpawnPoint=Black Thread States/Normal World/Hero Treadmill/needle behaviour/Rosary Fire Point, Prefab=Geo Small Magnet Range Variant (localpoolprefabs_assets_areaunderstore.bundle), Position=(unset), Rotation=(unset), FlipScaleFireLeft=false, Speed=18, MinAngle=130, MaxAngle=140, Space=1, StoreSpawned=var "")
    CallMethodProper(gameObject=Self, behaviour="JitterSelfForTime", methodName="StartTimedJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Shot Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shot Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    PlayAudioEvent(audioClip=under_03d_hero_treadmil_reward_arrive (sfxstatic_assets_areaunderstore.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Shot Effect", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Above
  }

  state Fail {
    PlayParticleEmitter(gameObject=Black Thread States/Normal World/Hero Treadmill/needle behaviour/Rosary Fire Point/Fail Dust, emit=(unset), resetIfPlaying=true)
    on FINISHED → Above
  }

  state Init {
    ActivateGameObject(gameObject=var "Shot Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Below
  }

  var Max Payout: int = 50

  var Value: int = 0
  var Given Rosary: bool = false
  var Shot Effect: gameObject = Black Thread States/Normal World/Hero Treadmill/needle behaviour/Rosary Fire Point/toll_machine_shot
  var Spit Event Tracking: array = var "Spit Event Tracking"
}
