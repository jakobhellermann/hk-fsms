# Animate Cogs
# 1 location(s):
#   Coral_02 / shell_plat shortcut

fsm Animate Cogs {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Cog L")
    FindNamedChild(gameObject=Self, storeResult=var "Cog R")
    on MOVE START → Move
  }

  state Move {
    SendMessage(gameObject=var "Cog L", delivery=SendMessage, options=1, functionCall=StartAnimation())
    SendMessage(gameObject=var "Cog R", delivery=SendMessage, options=1, functionCall=StartAnimation())
    PlayAudioEvent(audioClip=dock_chain_rock_plat_start_move (sfxstatic_assets_areadocks.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on MOVE STOP → End
  }

  state End {
    SendMessage(gameObject=var "Cog L", delivery=SendMessage, options=1, functionCall=StopAtCurrentPoint())
    SendMessage(gameObject=var "Cog R", delivery=SendMessage, options=1, functionCall=StopAtCurrentPoint())
  }

  var Cog L: gameObject = <null>
  var Cog R: gameObject = <null>
}
