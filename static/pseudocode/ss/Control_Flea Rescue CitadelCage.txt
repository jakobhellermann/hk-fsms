# Control
# 2 location(s):
#   Library_01 / Flea Rescue CitadelCage
#   Song_14 / Flea Rescue CitadelCage

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    CheckQuestPdSceneBool(QuestTarget=var "Quest Target", ExpectedValue=true, trueEvent=→"ACTIVATED", falseEvent=(none), storeValue=(unset))
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on ACTIVATED → Broken
    on HIT → Check
    on HEAVY → Check Heavy
  }

  state Check {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Active Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    on FINISHED → Hit
    on CANCEL → Idle
  }

  state Hit {
    PlayRandomAudioClipTable(Table=metal_break_hit (sfxdynamic_assets_areasong.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Flea"), sendEvent="WAKE", delay=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Strike")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Strike", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    MatchScaleSign(Target=var "Strike", MatchTo=var "Hero", EveryFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Hit L
    on R → Hit R
    on BREAK → Break
  }

  state Hit R {
    PlayParticleEmitter(gameObject=var "Pt Hit R", emit=0, resetIfPlaying=false)
    SetPosition(gameObject=var "Art Pivot", vector=(unset), x=0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=-0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimateRotationTo(gameObject=var "Art Pivot", fromValue=0, toValue=-6, worldSpace=false, negativeSpace=false, time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    AnimateRotationTo(gameObject=var "Cage", fromValue=0, toValue=-5, worldSpace=false, negativeSpace=false, time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Return Rotate", floatValue=-6, everyFrame=false)
    SetFloatValue(floatVariable=var "Cage Return Rotate", floatValue=5, everyFrame=false)
    on FINISHED → Topple Back
  }

  state Topple Back {
    AnimateRotationTo(gameObject=var "Art Pivot", fromValue=var "Return Rotate", toValue=0, worldSpace=false, negativeSpace=false, time=0.25, speed=(unset), delay=(unset), easeType=22, reverse=false, finishEvent=(none), realTime=false)
    AnimateRotationTo(gameObject=var "Cage", fromValue=var "Cage Return Rotate", toValue=0, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=23, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")
    on FINISHED → Idle
    on HIT → Hit
  }

  state Hit L {
    PlayParticleEmitter(gameObject=var "Pt Hit L", emit=0, resetIfPlaying=false)
    SetPosition(gameObject=var "Art Pivot", vector=(unset), x=-0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimateRotationTo(gameObject=var "Art Pivot", fromValue=0, toValue=6, worldSpace=false, negativeSpace=false, time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    AnimateRotationTo(gameObject=var "Cage", fromValue=0, toValue=5, worldSpace=false, negativeSpace=false, time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Return Rotate", floatValue=6, everyFrame=false)
    SetFloatValue(floatVariable=var "Cage Return Rotate", floatValue=5, everyFrame=false)
    on FINISHED → Topple Back
  }

  state Break {
    ActivateGameObject(gameObject=var "Flea", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flea Rescue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Strike")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Strike", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    MatchScaleSign(Target=var "Strike", MatchTo=var "Hero", EveryFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Break L
    on R → Break R
  }

  state Break R {
    SendEventByName(eventTarget=GameObject(var "Flea"), sendEvent="RESCUE R", delay=0, everyFrame=false)  // disabled
    SetPosition(gameObject=var "Art Pivot", vector=(unset), x=0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=-0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimateRotationTo(gameObject=var "Art Pivot", fromValue=0, toValue=-90, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=23, reverse=false, finishEvent=→"FINISHED", realTime=false)
  }

  state Break L {
    SendEventByName(eventTarget=GameObject(var "Flea"), sendEvent="RESCUE L", delay=0, everyFrame=false)  // disabled
    SetPosition(gameObject=var "Art Pivot", vector=(unset), x=-0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=0.87, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimateRotationTo(gameObject=var "Art Pivot", fromValue=0, toValue=90, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=23, reverse=false, finishEvent=→"FINISHED", realTime=false)
  }

  state Broken {
    ActivateGameObject(gameObject=var "Flea", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  state Check Heavy {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Active Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    IntAdd(intVariable=var "Hits", add=-10, everyFrame=false)
    on FINISHED → Hit
    on CANCEL → Idle
  }

  var Hero Y: float = 0
  var Return Rotate: float = 0
  var Cage Return Rotate: float = 0
  var Hits: int = 4
  var Art: gameObject = Flea Rescue CitadelCage/Art Pivot/Art
  var Art Pivot: gameObject = Flea Rescue CitadelCage/Art Pivot
  var Self: gameObject = Flea Rescue CitadelCage
  var Strike: gameObject = <null>
  var Cage: gameObject = Flea Rescue CitadelCage/Art Pivot/Art/Cage
  var Flea: gameObject = Flea Rescue CitadelCage/Art Pivot/Art/Flea Distressed
  var Break Effects: gameObject = Flea Rescue CitadelCage/Break Effects
  var Pt Hit L: gameObject = Flea Rescue CitadelCage/Pt Hit L
  var Pt Hit R: gameObject = Flea Rescue CitadelCage/Pt Hit R
  var Flea Rescue: gameObject = Flea Rescue CitadelCage/Art Pivot/Art/Flea Rescue Activation
  var Quest Target: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
