// uses template: bellbeast_children_control
fsm bellbeast_children_control {
  start Check Unlocked

  state Check Unlocked {
    PlayerDataBoolTest(boolName="UnlockedFastTravelTeleport", isTrue=→"UNLOCKED", isFalse=→"LOCKED")
    on UNLOCKED → Set Spawn Amount
    on LOCKED → Locked
  }

  state Set Spawn Amount {
    GetOwner(storeGameObject=var "Self")
    GetChildCount(gameObject=Self, storeResult=var "Spawns")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SendRandomEvent(events=[→"FINISHED", (none)], weights=[0.5, 0.5], delay=0)
    IntAdd(intVariable=var "Spawns", add=-1, everyFrame=false)
    on FINISHED → Clamp Spawns
  }

  state Do Spawn {
    GetRandomChild(gameObject=Self, storeResult=var "Next Node")
    CreateObject(gameObject=Bellbeast Child (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Next Node", position=(unset), rotation=(unset), storeObject=var "")
    SetParent(gameObject=var "Next Node", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    IntAdd(intVariable=var "Spawns", add=-1, everyFrame=false)
    IntCompare(integer1=var "Spawns", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"REPEAT", everyFrame=false)
    on REPEAT → Do Spawn
    on END → End
  }

  state End {
  }

  state Clamp Spawns {
    IntClamp(intVariable=var "Spawns", minValue=2, maxValue=100, everyFrame=false)
    on FINISHED → Do Spawn
  }

  state Locked {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Spawns: int = 0
  var Self: gameObject = <null>
  var Next Node: gameObject = <null>
}
