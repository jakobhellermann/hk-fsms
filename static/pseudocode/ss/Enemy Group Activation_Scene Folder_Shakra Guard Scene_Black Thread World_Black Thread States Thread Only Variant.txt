// uses template: mapper_act3_enemy_activation
fsm Enemy Group Activation {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindNamedChild(gameObject=Self, storeResult=var "Enemy Group A")
    FindNamedChild(gameObject=Self, storeResult=var "Enemy Group B")
    FindNamedChild(gameObject=Self, storeResult=var "Activation  Range")
    GetPlayerDataInt(intName="mapperLocationAct3", storeValue=var "Mapper Location")
    IntCompare(integer1=var "Mapper Location", integer2=var "Anti Location Int", equal=→"NONE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Mapper Location", integer2=2, equal=→"NONE", lessThan=(none), greaterThan=(none), everyFrame=false)
    PlayerDataBoolTest(boolName="mapperIsFightingAct3", isTrue=(none), isFalse=→"NONE")
    GetPlayerDataInt(intName="mapperFightGroup", storeValue=var "Fight Group")
    IntCompare(integer1=var "Fight Group", integer2=1, equal=→"GROUP A", lessThan=→"GROUP A", greaterThan=→"GROUP B", everyFrame=false)
    on GROUP A → Ready Group A
    on GROUP B → Ready Group B
    on NONE → None
  }

  state None {
    ActivateGameObject(gameObject=var "Enemy Group A", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Group B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Ready Group A {
    SetGameObject(variable=var "Enemy Group Crt", gameObject=var "Enemy Group A", everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Group B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait For Entry
  }

  state Ready Group B {
    SetGameObject(variable=var "Enemy Group Crt", gameObject=var "Enemy Group B", everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Group A", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait For Entry
  }

  state Wait For Entry {
    SetPosition2d(gameObject=var "Enemy Group Crt", vector=var "Enemy Cage Offset", x=(unset), y=(unset), space=1, everyFrame=false, lateUpdate=false)
    Trigger2dEvent(gameObject=var "Activation  Range", trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Start Fight
  }

  state Start Fight {
    ActivateGameObject(gameObject=var "Activation  Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition2d(gameObject=var "Enemy Group Crt", vector=(unset), x=0, y=0, space=1, everyFrame=false, lateUpdate=false)
    SetPlayerDataBool(boolName="seenMapperAct3", value=true)
    SetPlayerDataInt(intName="mapperLocationAct3", value=var "Location Int")
  }

  var Location Int: int = 3
  var Anti Location Int: int = 1
  var Enemy Cage Offset: vector2 = (0, 200)

  var Mapper Location: int = 0
  var Fight Group: int = 0
  var Enemy Group A: gameObject = <null>
  var Enemy Group B: gameObject = <null>
  var Activation  Range: gameObject = <null>
  var Enemy Group Crt: gameObject = <null>
}
