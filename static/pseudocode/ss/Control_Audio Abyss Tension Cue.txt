# Control
# 1 location(s):
#   Bonetown / Black Thread States/Black Thread World/Audio Abyss Tension Cue

fsm Control {
  start Init

  state Pause {
    Wait(time=var "Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Tension
  }

  state Start Tension {
    ApplyMusicCue(musicCue=Abyss Tension (audiocuesdynamic_assets_blackthread.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.5)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Trigger2dEvent(gameObject=var "End Range", trigger=1, collideTag="", sendEvent=→"TENSION END", storeCollider=var "")
    on TENSION END → Fade Out
  }

  state Init {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    FindNamedChild(gameObject=Self, storeResult=var "End Range")
    FindNamedChild(gameObject=Self, storeResult=var "Restart Range")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Pause
  }

  state Fade Out {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioStop(gameObject=Self, fadeTime=var "Fade Out Time")  // disabled
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=var "Fade Out Time")
    Trigger2dEvent(gameObject=var "Restart Range", trigger=1, collideTag="", sendEvent=→"RESTART", storeCollider=var "")
    on RESTART → Restart
  }

  state Restart {
    Trigger2dEvent(gameObject=var "End Range", trigger=1, collideTag="", sendEvent=→"TENSION END", storeCollider=var "")
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=3)
    on TENSION END → Fade Out
  }

  var Fade Out Time: float = 8
  var Pause: float = 0

  var Activated: bool = false
  var End Range: gameObject = <null>
  var Restart Range: gameObject = <null>
}
