# Control
# 1 location(s):
#   Song_Tower_01 / Left Wall Hero Saver

fsm Control {
  start Pause

  state Check {
    CheckXPosition(gameObject=var "Hero", compareTo=var "Check Pos", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"SAVE", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on SAVE → Save
  }

  sequence state Pause {
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Check
  }

  state Save {
    SetPosition2d(gameObject=var "Hero", vector=(unset), x=var "Check Pos", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  var Check Pos: float = 33.24
}
