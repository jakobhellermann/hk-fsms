# PassToParent
# 1 location(s):
#   herodynamic_assets_all.bundle / Clockwork Hatchling/Damager

fsm PassToParent {
  start Init
  on HIT LANDED → Pass Hit Landed  // from any state
  on ATTACK BLOCKED → Pass Hit Blocked  // from any state

  state Inert {
  }

  state Pass Hit Landed {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="HIT LANDED", delay=0, everyFrame=false)
    on FINISHED → Inert
  }

  state Pass Hit Blocked {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="ATTACK BLOCKED", delay=0, everyFrame=false)
    on FINISHED → Inert
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Inert
  }

  var Parent: gameObject = <null>
}
