# Control
# 1 location(s):
#   Cradle_Destroyed_Challenge_01 / Centipede Trap Control/Centipede Trap:1

fsm Control {
  start Init
  on TOOK HEAVY DAMAGE → Heavy Damage  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindAlertRange(target=var "Parent", storeResult=var "Spawn Range", childName="Spawn Range")
    on FINISHED → Dormant
  }

  state Dormant {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dormant")
    SetMeshRenderer(gameObject=Self, active=false)
    CheckAlertRange(alertRange=var "Spawn Range", storeResult=(unset), InRangeEvent=→"ALERT", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    on ALERT → Spawn Pause
  }

  state Antic {
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(100, 20), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)  // disabled
    PlayRandomAudioClipTable(Table=enemy_dig_in_antic (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayAudioEventRandom(audioClips=[sand_centipedes_go_back_into_ground_1 (sfxstatic_assets_areacoral.bundle), sand_centipedes_go_back_into_ground_2 (sfxstatic_assets_areacoral.bundle), sand_centipedes_go_back_into_ground_3 (sfxstatic_assets_areacoral.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Burst 1
  }

  state Burst 1 {
    PlayAudioEvent(audioClip=sand_centipedes_eat_hornet_or_enemy (sfxstatic_assets_areacoral.bundle), pitchMin=1.2, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Eat Audio")
    PlayAudioEventRandom(audioClips=[sand_centipedes_burst_1 (sfxstatic_assets_areacoral.bundle), sand_centipedes_burst_2 (sfxstatic_assets_areacoral.bundle), sand_centipedes_burst_3 (sfxstatic_assets_areacoral.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    StopParticleEmitter(gameObject=var "Pt Antic")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetMeshRenderer(gameObject=Self, active=true)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.15)
    PlayParticleEmitter(gameObject=var "Pt Dig", emit=0, resetIfPlaying=false)
    PlayRandomAudioClipTable(Table=enemy_dig_in_emerge (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Burst 2
  }

  state End {
    AudioStopV2(gameObject=var "Eat Audio", fadeTime=0, cancelOnEarlyExit=false)
    StopParticleEmitter(gameObject=var "Pt Dig")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dormant")
    SetMeshRenderer(gameObject=Self, active=false)
    SetTag(gameObject=var "My Node", tag="Teleplane")
    PlayAudioEvent(audioClip=farmer_centipede_burrow (sfxstatic_assets_areagreymoor.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Dormant
  }

  state Spawn Pause {
    WaitRandom(timeMin=var "Wait Min", timeMax=var "Wait Max", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait For Cooldown
  }

  state Get Node {
    FindClosest(gameObject=var "Hero", withTag="Teleplane", ignoreOwner=true, mustBeVisible=false, storeObject=var "My Node", storeDistance=(unset), everyFrame=false)
    SetTag(gameObject=var "My Node", tag="Untagged")
    SetPositionToObject2D(gameObject=Self, targetObject=var "My Node", xOffset=0, yOffset=0, everyFrame=false)
    GetRotation(gameObject=var "My Node", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    RandomlyFlipScale(gameObject=Self)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    FloatCompare(float1=var "Rotation", float2=0, tolerance=1, equal=→"ROOF", lessThan=→"WALL", greaterThan=→"WALL", everyFrame=false)
    on ROOF → Roof
    on WALL → Wall
  }

  state Roof {
    TranslateRandom(gameObject=Self, translateMin=(-0.5, 0, 0), translateMax=(0.5, 0, 0), space=0)
    on FINISHED → Rotate?
  }

  state Wall {
    TranslateRandom(gameObject=Self, translateMin=(0, -0.5, 0), translateMax=(0, 0.5, 0), space=0)
    on FINISHED → Rotate?
  }

  state Heavy Damage {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dormant")
    SetMeshRenderer(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Pt Antic")
    StopParticleEmitter(gameObject=var "Pt Dig")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Heavy Damage Reset
  }

  state Heavy Damage Reset {
    SetTag(gameObject=var "My Node", tag="Teleplane")
    on FINISHED → Dormant
  }

  state Rotate? {
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[1, 1], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=180, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Antic
  }

  state Burst 2 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End
    on MULTI HIT CONNECT → Multihit Anim
  }

  state Multihit Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Damage Hero", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End
  }

  state Wait For Cooldown {
    BoolTest(boolVariable=var "Cooldown", isTrue=(none), isFalse=→"FINISHED", everyFrame=true)
    on FINISHED → Get Node
  }

  var Wait Min: float = 2
  var Wait Max: float = 2.5

  var Rotation: float = 0
  var Cooldown: bool = false
  var In Alert Range: bool = false
  var Pt Antic: gameObject = Centipede Trap Control/Centipede Trap:1/Pt Antic
  var Pt Dig: gameObject = Centipede Trap Control/Centipede Trap:1/Pt Dig
  var Parent: gameObject = <null>
  var My Node: gameObject = <null>
  var Eat Audio: gameObject = <null>
  var Spawn Range: object = <null>
}
