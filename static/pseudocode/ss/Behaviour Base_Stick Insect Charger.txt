# Behaviour Base
# 5 location(s):
#   Shellwood_02 / Stick Insect Charger
#   Shellwood_02 / Stick Insect
#   Shellwood_15 / Stick Insect
#   Shellwood_15 / Stick Insect (1)
#   Shellwood_20 / Stick Insect Charger

// uses template: stick_insect_base
fsm Behaviour Base {
  start Init

  state Init {
    GetHero(storeResult=var "Hero")
    FindChild(gameObject=Self, childName="Emerge Effect", storeResult=var "Burst Effect")
    FindChild(gameObject=Self, childName="Initial Position Markers", storeResult=var "Initial Position Markers")
    FindChild(gameObject=Self, childName="Pt Emerge", storeResult=var "Pt Emerge")
    FindNamedChild(gameObject=Self, storeResult=var "Slam Effect")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Run")
    FindNamedChild(gameObject=Self, storeResult=var "Spawn Pos Alt")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    SetTransformParent(gameObject=var "Initial Position Markers", parent=<null>, worldPositionStays=true)
    FindChild(gameObject=Self, childName="Reposition Markers", storeResult=var "Reposition Markers")
    SetTransformParent(gameObject=var "Reposition Markers", parent=<null>, worldPositionStays=true)
    BoolTest(boolVariable=var "Ground Ambusher", isTrue=→"GROUND", isFalse=(none), everyFrame=false)
    on FINISHED → Initial Position
    on GROUND → Underground
  }

  state Choose Anims {
    SendRandomEvent(events=[→"1", →"2", →"3"], weights=[1, 1, 1], delay=0)
    on 1 → Set Anims 01
    on 2 → Set Anims 02
    on 3 → Set Anims 03
  }

  state Set Anims 01 {
    SetStringValue(stringVariable=var "Hidden Anim", stringValue="Branch 01 Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Hidden Shake Anim", stringValue="Branch 01 Shake", everyFrame=false)
    SetStringValue(stringVariable=var "Shake Loop Anim", stringValue="Branch 03 Shake Loop", everyFrame=false)
    on FINISHED → Hidden
  }

  state Set Anims 02 {
    SetStringValue(stringVariable=var "Hidden Anim", stringValue="Branch 02 Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Hidden Shake Anim", stringValue="Branch 02 Shake", everyFrame=false)
    SetStringValue(stringVariable=var "Shake Loop Anim", stringValue="Branch 01 Shake Loop", everyFrame=false)
    on FINISHED → Hidden
  }

  state Set Anims 03 {
    SetStringValue(stringVariable=var "Hidden Anim", stringValue="Branch 03 Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Hidden Shake Anim", stringValue="Branch 03 Shake", everyFrame=false)
    SetStringValue(stringVariable=var "Shake Loop Anim", stringValue="Branch 03 Shake Loop", everyFrame=false)
    on FINISHED → Hidden
  }

  state Hidden {
    SetLayer(gameObject=Self, layer=Terrain Detector(14))
    SetDamageHeroAmount(target=Self, damageDealt=0)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Hidden Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    WaitRandom(timeMin=4, timeMax=8, finishEvent=→"SHAKE", realTime=false)
    BoolTest(boolVariable=var "Battler", isTrue=→"BATTLER", isFalse=(none), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="AWAKE", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"AWAKE", ActiveOuter=→"AWAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SHAKE → Shake
    on AWAKE → Burst Antic
    on BATTLER → Battle Wait
  }

  state Shake {
    PlayAudioEventRandom(audioClips=[stick_insect_antic (sfxstatic_assets_areashellwood.bundle), stick_insect_charger_turn_around (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.8, pitchMax=0.9, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Hidden Shake Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="AWAKE", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=1, None=(none), ActiveInner=(none), ActiveOuter=→"AWAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=0.33, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst?
    on AWAKE → Burst Antic
  }

  state Burst Out {
    PlayAudioEventRandom(audioClips=[shellwood_twig_wall_hit_1 (sfxstatic_assets_areashellwood.bundle), shellwood_twig_wall_hit_2 (sfxstatic_assets_areashellwood.bundle), shellwood_twig_wall_hit_3 (sfxstatic_assets_areashellwood.bundle)], pitchMin=1.2, pitchMax=1.2, volume=0.75, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Stick_Insect_voice_chirp (sfxdynamic_assets_areashellwood.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SetLayer(gameObject=Self, layer=Enemies(11))
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetVelocity2d(gameObject=Self, vector=(0, 30), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst Out", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Burst Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=true, otherLayerNumber=8, ignoreTriggers=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    on LAND → Land
  }

  state Burst? {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="AWAKE", outOfRangeEvent="", everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"AWAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    on FINISHED → Hidden
    on AWAKE → Burst Antic
  }

  state Fall {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fall", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=true, otherLayerNumber=8, ignoreTriggers=true)
    on LAND → Land
  }

  state Run {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    RendererIsVisible(Target=Self, LostVisibilityTime=2, VisibleEvent=(none), NotVisibleEvent=→"HIDE", EveryFrame=true)
    StartWalker(walkRight=(unset), target=Self, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"PRAY", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"PRAY", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Run")
    on HIDE → Stop Walker
    on PRAY → Pray
    on ATTACK → Attack
    on FALL → Fall
  }

  state Land {
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Run
    on FALL → Fall
  }

  state Stop Walker {
    BoolTest(boolVariable=var "Battler", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetLayer(gameObject=Self, layer=Terrain Detector(14))
    on FINISHED → Reposition
    on CANCEL → Run
  }

  state Pray {
    StopWalker(target=Self, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pray", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayAnimationDelay(gameObject=Self, animLibName="", clipName="Pray", delay=0.15)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"AWAKE", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Stick_Insect_voice_sing (sfxdynamic_assets_areashellwood.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on AWAKE → Idle Pause
    on SING DURATION END → Idle Pause
  }

  state Idle Pause {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pray To Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    WaitRandom(timeMin=0.5, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Run
  }

  state Attack {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"PRAY", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    StopWalker(target=Self, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Attack"), sendEvent="ATTACK", delay=0, everyFrame=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on RETURN → Run
    on FALL → Cancel Attack
    on LEAP START → Leaping
    on PRAY → Pray
  }

  state Cancel Attack {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Attack"), sendEvent="CANCEL ATTACK", delay=0, everyFrame=false)
    on RETURN → Fall
  }

  state Initial Position {
    GetRandomChild(gameObject=var "Initial Position Markers", storeResult=var "Chosen Marker")
    GameObjectCompare(gameObjectVariable=var "Chosen Marker", compareTo=<null>, equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    GetPosition2D(gameObject=var "Chosen Marker", vector=var "Target Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    PositionColliderOnGroundV2(Target=Self, GroundPos=var "Target Pos")
    on FINISHED → Choose Anims
  }

  state Reposition {
    GetRandomChildOffScreen(Target=var "Reposition Markers", ViewportPadding=(0.1, 0.2), StoreObject=var "Chosen Marker")
    GameObjectCompare(gameObjectVariable=var "Chosen Marker", compareTo=<null>, equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    GetPosition2D(gameObject=var "Chosen Marker", vector=var "Target Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    PositionColliderOnGroundV2(Target=Self, GroundPos=var "Target Pos")
    on FINISHED → Choose Anims
  }

  state Burst Antic {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=stick_insect_emerge (sfxstatic_assets_areashellwood.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    AudioPlay(gameObject=Self, volume=1, oneShotClip=stick_insect_charge_antic (sfxstatic_assets_areashellwood.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Shake Loop Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Emerge", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=var "Wake Time", finishEvent=→"AWAKE", realTime=false)
    on AWAKE → Burst Out
  }

  state Leaping {
    on LEAP END → Attack
    on RETURN → Run
  }

  state Battle Wait {
    SetFloatValue(floatVariable=var "Wake Time", floatValue=1, everyFrame=false)
    on BATTLE START → Battle Pause
  }

  state Battle Pause {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst Antic
  }

  state Underground {
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=false)
    on BATTLE START → Emerge Pause
  }

  state Emerge Pause {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → RePos?
  }

  state Emerge Antic {
    PlayParticleEmitterChildren(gameObject=var "Pt Emerge", resetTimeIfPlaying=false, stopOnStateExit=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Emerge Antic")
    SetMeshRenderer(gameObject=Self, active=true)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlay(gameObject=Self, volume=1, oneShotClip=stick_insect_emerge (sfxstatic_assets_areashellwood.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    PlayAudioEventRandom(audioClips=[stick_insect_antic (sfxstatic_assets_areashellwood.bundle), stick_insect_charger_turn_around (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.8, pitchMax=0.9, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Emerge
  }

  state Emerge {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Burst Out
  }

  state RePos? {
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=4, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetPositionToObject2D(gameObject=Self, targetObject=var "Spawn Pos Alt", xOffset=0, yOffset=0, everyFrame=false)
    on FINISHED → Emerge Antic
  }

  var Battler: bool = false
  var Ground Ambusher: bool = false

  var Current Y Velocity: float = 0
  var Turn Duration: float = 0
  var Distance: float = 0
  var Wake Time: float = 0.75
  var Hidden Anim: string = ""
  var Hidden Shake Anim: string = ""
  var Shake Loop Anim: string = ""
  var Target Pos: vector2 = (0, 0)
  var Burst Effect: gameObject = <null>
  var Chosen Marker: gameObject = <null>
  var Hero: gameObject = <null>
  var Initial Position Markers: gameObject = <null>
  var Pt Emerge: gameObject = <null>
  var Reposition Markers: gameObject = <null>
  var Slam Effect: gameObject = <null>
  var Pt Run: gameObject = <null>
  var Spawn Pos Alt: gameObject = <null>
  var Audio Loop Voice: gameObject = <null>
}
