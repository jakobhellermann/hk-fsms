# Horizontal Motion
# 1 location(s):
#   Bone_12 / Black Thread States/Normal World/Pin Gallery States/Here/Pin Challenge/Lady Bug Tiny

fsm Horizontal Motion {
  start Idle
  on HORIZONTAL STOP → Idle  // from any state

  state Idle {
    on HORIZONTAL START → Start
  }

  state Start {
    GetFsmFloat(gameObject=Self, fsmName="Gong Control", variableName="Target X", storeValue=var "X Max", everyFrame=false)
    RandomFloat(min=0.25, max=2, storeResult=var "Multiplier")  // disabled
    FloatMultiply(floatVariable=var "Speed", multiplyBy=var "Multiplier", everyFrame=false)  // disabled
    SetFloatValue(floatVariable=var "Speed R", floatValue=var "Speed", everyFrame=false)
    SetFloatValue(floatVariable=var "Speed L", floatValue=var "Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed L", multiplyBy=-1, everyFrame=false)
    SetFloatValue(floatVariable=var "Accel", floatValue=var "Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Accel", multiplyBy=0.1, everyFrame=false)
    RandomFloat(min=5, max=7, storeResult=var "Distance")
    FloatOperator(float1=var "X Max", float2=var "Distance", operation=Subtract, storeResult=var "X Min", everyFrame=false)
    on FINISHED → Left
  }

  state Right {
    AccelerateToX(gameObject=Self, accelerationFactor=var "Accel", targetSpeed=var "Speed R")
    CheckXPosition(gameObject=Self, compareTo=var "X Max", compareToOffset=0, tolerance=0, equal=→"L", equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"L", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on L → Left
  }

  state Left {
    AccelerateToX(gameObject=Self, accelerationFactor=var "Accel", targetSpeed=var "Speed L")
    CheckXPosition(gameObject=Self, compareTo=var "X Min", compareToOffset=0, tolerance=0, equal=→"R", equalBool=(unset), lessThan=→"R", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on R → Right
  }

  var Accel: float = 0
  var Speed: float = 5
  var Speed L: float = 0
  var Speed R: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var Multiplier: float = 0
  var Distance: float = 0
}
