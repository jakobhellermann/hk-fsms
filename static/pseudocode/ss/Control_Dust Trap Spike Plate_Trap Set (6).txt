# Control
# 1 location(s):
#   Dust_Maze_01 / Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate

fsm Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Step Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spikes", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"STEP", storeCollider=var "")
    on STEP → Step
  }

  state Step {
    PlayAudioEvent(audioClip=trap_spike_plate_button (sfxstatic_assets_areadustmaze.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Step Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatePositionTo(gameObject=var "Sprite", toValue=(0, -0.5, 0), localSpace=true, time=0.1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate Spikes
  }

  state Activate Spikes {
    PlayAudioEvent(audioClip=trap_spike_plate_rise (sfxstatic_assets_areadustmaze.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=shellwood_twig_wall_hit_3 (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Spikes", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Slam", resetTimeIfPlaying=false, stopOnStateExit=false)
    on FINISHED → Damager On
  }

  state Damager On {
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObjectDelay(gameObject=var "Enemy Damager", activate=false, resetOnExit=false, delay=0.1)
  }

  var Sprite: gameObject = Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate/Sprite
  var Step Effects: gameObject = Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate/Step Effects
  var Spikes: gameObject = Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate/Spikes
  var Slam Effect: gameObject = Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate/Slam Effect
  var Damager: gameObject = Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate/Damager
  var Pt Slam: gameObject = Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate/Pt Slam
  var Enemy Damager: gameObject = Mist Maze Controller/Trap Sets/Trap Set (6)/Dust Trap Spike Plate/Enemy Damager
}
