# Music Control
# 1 location(s):
#   Abyss_05 / Cutscene Options/Abyss Dive Cutscene

fsm Music Control {
  start Idle

  state Idle {
    on START MUSIC → Start Music
    on FINAL MUSIC → Silent
  }

  sequence state Start Music {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=LastDivePrologue (audiocuesdynamic_assets_areaabyss.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=4)
    on FINAL MUSIC → Silent
  }

  state Silent {
    CheckTrackTriggerCount(target=var "Music Range Main", count=1, test=Equal, everyFrame=true, successEvent=→"MAIN")
    CheckTrackTriggerCount(target=var "Music Range Sub", count=1, test=Equal, everyFrame=true, successEvent=→"SUB")
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=8)
    on MAIN → Main
    on SUB → Sub
  }

  state Sub {
    ApplyMusicCue(musicCue=LastDivePrologue (audiocuesdynamic_assets_areaabyss.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Action (audioobjects_assets_all.bundle), transitionTime=2)
    CheckTrackTriggerCount(target=var "Music Range Main", count=1, test=Equal, everyFrame=true, successEvent=→"MAIN")
    CheckTrackTriggerCount(target=var "Music Range Silent", count=1, test=Equal, everyFrame=true, successEvent=→"SILENT")
    on MAIN → Main
    on SILENT → Silent
  }

  state Main {
    ApplyMusicCue(musicCue=LastDivePrologue (audiocuesdynamic_assets_areaabyss.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Action (audioobjects_assets_all.bundle), transitionTime=1)
    CheckTrackTriggerCountV2(Target=var "Music Range Main", Count=1, Test=Equal, EveryFrame=true, SetBool=var "In Main Range", SuccessEvent=(none), FailEvent=(none))
    CheckTrackTriggerCountV2(Target=var "Music Range Sub", Count=1, Test=Equal, EveryFrame=true, SetBool=var "In Sub Range", SuccessEvent=(none), FailEvent=(none))
    BoolTestMulti(boolVariables=[var "In Main Range", var "In Sub Range"], boolStates=[false, true], trueEvent=→"SUB", falseEvent=(none), storeResult=(unset), everyFrame=true)
    CheckTrackTriggerCount(target=var "Music Range Silent", count=1, test=Equal, everyFrame=true, successEvent=→"SILENT")
    on SUB → Sub
    on SILENT → Silent
  }

  state Raise Music {
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
    on FINAL MUSIC → Silent
  }

  var In Main Range: bool = false
  var In Sub Range: bool = false
  var Music Range Silent: gameObject = Cutscene Options/Abyss Dive Cutscene/Music Range Silent
  var Music Range Sub: gameObject = Cutscene Options/Abyss Dive Cutscene/Music Range Sub
  var Music Range Main: gameObject = Cutscene Options/Abyss Dive Cutscene/Music Range Main
}
