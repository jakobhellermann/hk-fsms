# FSM
# 1 location(s):
#   Library_10 / Collectable Item Pickup - Melody

fsm FSM {
  start Idle

  state Idle {
    on COLLECT END → Dlg?
    on COLLECT START → Lock Animation Control
  }

  sequence state Updated Quest {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    HeroLockState(heroLockStates=ControlLocked, mode=Remove, setOppositeOnExit=false)
    ShowQuestUpdatedStandalone(Quest=Citadel Ascent Melody Librarian (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    on FINISHED → Idle
  }

  state Dlg? {
    GetPlayerDataBool(boolName="LibrarianAskedForRelic", storeValue=var "Has Met")
    ConvertBoolToString(boolVariable=var "Has Met", stringVariable=var "Dlg", falseString="MELODY_PICKUP_HINT_NOTMET", trueString="MELODY_PICKUP_HINT_MET", everyFrame=false)
    RunDialogue(Sheet="City", Key=var "Dlg", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Updated Quest
  }

  state Lock Animation Control {
    HeroLockState(heroLockStates=ControlLocked, mode=Add, setOppositeOnExit=false)
    on COLLECT END → Dlg?
  }

  var Has Met: bool = false
  var Dlg: string = ""
}
