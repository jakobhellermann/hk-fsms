# Shared Anims
# 1 location(s):
#   Tut_04 / States/Intro Scene/Snail Shamans Set

fsm Shared Anims {
  start Talking Together
  on RITUAL STARTED → Ritual  // from any state

  state Idle {
    on ENTER TALK → Enter Talk
  }

  state Enter Talk {
    ActivateGameObject(gameObject=var "Talk Camlock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    LookAnimNpcOverrideTarget(OverrideTarget=var "Hero", Target=var "Shaman M")
    LookAnimNpcOverrideTarget(OverrideTarget=var "Hero", Target=var "Shaman S")
    LookAnimNpcOverrideTarget(OverrideTarget=var "Hero", Target=var "Shaman S Stand")
    LookAnimNpcOverrideTarget(OverrideTarget=var "Hero", Target=var "Shaman B")
    on FINISHED → Talk Idle
  }

  state Talk Idle {
    on EXIT TALK → Exit Talk
    on S → Small Talking
    on M → Mid Talking
    on B → Big Talking
  }

  state Small Talking {
    ActivateGameObject(gameObject=var "Intro Camlock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="NpcDialogueTitle", methodName="EnableAndShow", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Talk Camlock S", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on LINE_END → Talk Idle
  }

  state Big Talking {
    ActivateGameObject(gameObject=var "Intro Camlock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="NpcDialogueTitle", methodName="EnableAndShow", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Talk Camlock B", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on LINE_END → Talk Idle
  }

  state Mid Talking {
    ActivateGameObject(gameObject=var "Intro Camlock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="NpcDialogueTitle", methodName="EnableAndShow", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Talk Camlock M", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on LINE_END → Talk Idle
  }

  state Exit Talk {
    ActivateGameObject(gameObject=var "Talk Camlock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    LookAnimNpcOverrideTarget(OverrideTarget=<null>, Target=var "Shaman M")
    LookAnimNpcOverrideTarget(OverrideTarget=<null>, Target=var "Shaman S")
    LookAnimNpcOverrideTarget(OverrideTarget=<null>, Target=var "Shaman S Stand")
    LookAnimNpcOverrideTarget(OverrideTarget=<null>, Target=var "Shaman B")
    on FINISHED → Idle
  }

  state Talking Together {
    ActivateGameObject(gameObject=var "Talk Camlock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Talk Camlock M", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Talk Camlock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Talk Camlock S", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Ritual Audio", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on ENTER TALK → Enter Talk
  }

  sequence state Ritual {
    TimeLimitSetV2(TimeDelay=15, StoreValue=var "End Time", CanSet=(unset), EveryFrame=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=var "Shaman B")
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=var "Shaman M")
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=var "Shaman S Stand")
    Wait(time=2, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Ritual Audio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Shaman S Stand", clipName="Stand Small Cast", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FindChild(gameObject=var "Shaman S Stand", childName="Snail Shaman Ritual Burn Effect", storeResult=var "Burn Effect")
    ActivateGameObject(gameObject=var "Burn Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Shaman B", clipName="Big Cast", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FindChild(gameObject=var "Shaman B", childName="Snail Shaman Ritual Burn Effect", storeResult=var "Burn Effect")
    ActivateGameObject(gameObject=var "Burn Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Shaman M", clipName="Mid Cast", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FindChild(gameObject=var "Shaman M", childName="Snail Shaman Ritual Burn Effect", storeResult=var "Burn Effect")
    ActivateGameObject(gameObject=var "Burn Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    TimeLimitCheckV2(StoredValue=var "End Time", AboveEvent=→"FINISHED", BelowEvent=(none), StoreBool=(unset), EveryFrame=true)
    on FINISHED → Ritual Glow
  }

  sequence state Ritual Glow {
    DoSpriteFlashCustom(Target=var "Shaman S Stand", FlashColour=(1, 1, 1, 1), Amount=1, TimeUp=10, StayTime=10, TimeDown=0, StayDownTime=0, Repeating=false, RepeatTimes=0, RepeatingPriority=0, CancelOnExit=false)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    DoSpriteFlashCustom(Target=var "Shaman B", FlashColour=(1, 1, 1, 1), Amount=1, TimeUp=10, StayTime=10, TimeDown=0, StayDownTime=0, Repeating=false, RepeatTimes=0, RepeatingPriority=0, CancelOnExit=false)
    Wait(time=0.1, finishEvent=(none), realTime=false)
    DoSpriteFlashCustom(Target=var "Shaman M", FlashColour=(1, 1, 1, 1), Amount=1, TimeUp=10, StayTime=10, TimeDown=0, StayDownTime=0, Repeating=false, RepeatTimes=0, RepeatingPriority=0, CancelOnExit=false)
  }

  var End Time: float = 0
  var Audio: gameObject = <null>
  var Burn Effect: gameObject = <null>
  var Intro Camlock: gameObject = States/Intro Scene/Snail Shamans Set/Intro Camlock
  var Shaman B: gameObject = States/Intro Scene/Snail Shamans Set/Shaman Big
  var Shaman M: gameObject = States/Intro Scene/Snail Shamans Set/Shaman Mid
  var Shaman S: gameObject = States/Intro Scene/Snail Shamans Set/Shaman Small Sit
  var Shaman S Stand: gameObject = States/Intro Scene/Snail Shamans Set/Shaman Small Stand
  var Talk Camlock: gameObject = States/Intro Scene/Snail Shamans Set/Talk Camlock
  var Talk Camlock B: gameObject = States/Intro Scene/Snail Shamans Set/Talk Camlock B
  var Talk Camlock M: gameObject = States/Intro Scene/Snail Shamans Set/Talk Camlock M
  var Talk Camlock S: gameObject = States/Intro Scene/Snail Shamans Set/Talk Camlock S
  var Ritual Audio: gameObject = States/Intro Scene/Snail Shamans Set/Audio Red Memory Ritual
}
