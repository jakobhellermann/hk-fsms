fsm Bind Response {
  start Idle

  state Idle {
    on DID BIND → Dist. Check
  }

  state Dist. Check {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=6.5, tolerance=0, equal=→"CLOSEBY", lessThan=→"CLOSEBY", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on CLOSEBY → Closeby
  }

  state Closeby {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FRIEND BIND", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Distance: float = 0
}
