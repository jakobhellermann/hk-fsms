fsm Sing Detect {
  start Detect

  state Detect {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.8, MaxReactDelay=1.8, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SING → Send Sing Event
  }

  state Send Sing Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SING", delay=0, everyFrame=false)
    on FINISHED → Detect
  }
}
