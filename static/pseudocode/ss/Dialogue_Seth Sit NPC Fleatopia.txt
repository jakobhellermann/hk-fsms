# Dialogue
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Seth Sit NPC Fleatopia

fsm Dialogue {
  start Idle

  state Idle {
    on INTERACT → Convo Choice
  }

  state Convo Choice {
    IntCompare(integer1=var "Value", integer2=0, equal=→"MEET", lessThan=(none), greaterThan=(none), everyFrame=false)
    GetPlayerDataVariable(VariableName="FleaGamesBestedSeth", StoreValue=var "Beat Seth Scores")
    SavedItemCanGetMore(Item=Memento Seth (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle), trueEvent=(none), falseEvent=(none), storeValue=var "Memento Collected")
    BoolFlip(boolVariable=var "Memento Collected")
    BoolTestMulti(boolVariables=[var "Beat Seth Scores", var "Memento Collected"], boolStates=[true, false], trueEvent=→"MEMENTO", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on MEET → Meet
    on MEMENTO → Memento
    on FINISHED → Repeat
  }

  state End Dialogue {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.3, finishEvent=(none), realTime=false)
    on FINISHED → Award Memento?
  }

  state Meet {
    SetIntValue(intVariable=var "Value", intValue=1, everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key="SETH_FESTIVAL_MEET", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Repeat {
    ConvertBoolToString(boolVariable=var "Memento Collected", stringVariable=var "Convo Key", falseString="SETH_FESTIVAL_REPEAT", trueString="SETH_FESTIVAL_REPEAT_POST_MEMENTO", everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key=var "Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Memento {
    RunDialogue(Sheet="Wanderers", Key="SETH_FESTIVAL_MEMENTO", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetBoolValue(boolVariable=var "Award Memento", boolValue=true, everyFrame=false)
    on CONVO_END → End Dialogue
  }

  state Award Memento? {
    BoolTest(boolVariable=var "Award Memento", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CollectableItemCollect(Amount=1, Item=Memento Seth (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle))
    SetBoolValue(boolVariable=var "Award Memento", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Value: int = 0
  var Award Memento: bool = false
  var Memento Collected: bool = false
  var Beat Seth Scores: bool = false
  var Convo Key: string = ""
}
