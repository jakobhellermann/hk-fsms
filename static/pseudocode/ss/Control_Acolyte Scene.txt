# Control
# 1 location(s):
#   Library_15 / Acolyte Scene

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Post Lights")
    FindNamedChild(gameObject=Self, storeResult=var "Pre Lights")
    on FINISHED → Idle
  }

  state Idle {
    on OPEN → Activations
  }

  state Activations {
    AudioPlaySimple(gameObject=var "Audio Player Cogs", volume=1, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Pre Lights", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Post Lights", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Acolytes Leave
  }

  state Acolytes Leave {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LEAVE", delay=0, everyFrame=false)
    PlayAudioEvent(audioClip=acolytes_collectively_jump_up_stereo (sfxstatic_assets_arealibrary.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
  }

  var Post Lights: gameObject = <null>
  var Pre Lights: gameObject = <null>
  var Audio Player Cogs: gameObject = Acolyte Scene/Audio Player Cogs
}
