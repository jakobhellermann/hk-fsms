# Anim Runner
# 1 location(s):
#   Belltown / Couriers States/Here/Couriers Quest Giver/Courier Tall

// uses template: courier_quest_giver_anim_runner
fsm Anim Runner {
  start Init

  state Inert {
    on PLAY WAKE → Wake
    on PLAY WAVE → Wave
  }

  sequence state Wake {
    SetBoolValue(boolVariable=var "Is Animating", boolValue=true, everyFrame=false)
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=(none), realTime=false)
    EnableBehaviour(gameObject=Self, behaviour="HeroPerformanceSingReaction", component=<null>, enable=false, resetOnExit=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    on FINISHED → Wake 2
  }

  sequence state Wave {
    SetBoolValue(boolVariable=var "Is Animating", boolValue=true, everyFrame=false)
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=(none), realTime=false)
    EnableBehaviour(gameObject=Self, behaviour="HeroPerformanceSingReaction", component=<null>, enable=false, resetOnExit=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=var "Startle Voice Table", pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName=var "Wave Anim", AnimationCompleteEvent=(none))
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=false, Target=Self)
    EnableBehaviour(gameObject=Self, behaviour="HeroPerformanceSingReaction", component=<null>, enable=true, resetOnExit=false)
    SetBoolValue(boolVariable=var "Is Animating", boolValue=false, everyFrame=false)
    on FINISHED → Inert
  }

  state Wake 2 {
    PlayAudioEvent(audioClip=relic_dealer_scavanging_loop (sfxstatic_assets_areabell.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.6, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Antic Player")
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=var "Startle Voice Table", pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Wake Anim", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Wake 3
  }

  state Wake 4 {
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=false, Target=Self)
    EnableBehaviour(gameObject=Self, behaviour="HeroPerformanceSingReaction", component=<null>, enable=true, resetOnExit=false)
    SetBoolValue(boolVariable=var "Is Animating", boolValue=false, everyFrame=false)
    on FINISHED → Inert
  }

  state Wake 3 {
    AudioStopV2(gameObject=var "Antic Player", fadeTime=0, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=garmond_jump (sfxstatic_assets_garmond.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.6, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Antic Player")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wake 4
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    on FINISHED → Inert
  }

  var Wake Anim: string = "Tall Wake"
  var Wave Anim: string = ""
  var Startle Voice Table: object = Courier_tall_alert (sfxdynamic_assets_areabell.bundle)

  var Is Animating: bool = false
  var Courier Short: gameObject = <null>
  var Antic Player: gameObject = <null>
  var Audio Loop Voice: gameObject = <null>
}
