# Dialogue
# 1 location(s):
#   Belltown / Town States/Spinner Defeated/Bagpipers Not Here/Belltown Greeter Act3

fsm Dialogue {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on INTERACT → Convo Check
  }

  state Convo Check {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    IntCompare(integer1=var "Value", integer2=0, equal=→"MEET", lessThan=(none), greaterThan=(none), everyFrame=false)
    PlayerDataBoolTest(boolName="IsAnyCursed", isTrue=→"CURSED", isFalse=(none))
    PlayerDataBoolTest(boolName="hasNeedolinMemoryPowerup", isTrue=(none), isFalse=→"FINISHED")
    IntCompare(integer1=var "Value", integer2=1, equal=→"TALK 1", lessThan=(none), greaterThan=(none), everyFrame=false)
    on MEET → Meet
    on TALK 1 → Talk 1
    on FINISHED → Repeat
    on CURSED → Cursed
  }

  state Meet {
    IntAdd(intVariable=var "Value", add=1, everyFrame=false)
    RunDialogue(Sheet=var "Sheet", Key="GREETER_MEET_ACT3", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state End Dialogue {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Repeat {
    GetPlayerDataBool(boolName="hasNeedolinMemoryPowerup", storeValue=var "Has Deep Melody")
    ConvertBoolToString(boolVariable=var "Has Deep Melody", stringVariable=var "Convo Key", falseString="GREETER_REPEAT_ACT3", trueString="GREETER_REPEAT2_ACT3", everyFrame=false)
    RunDialogue(Sheet=var "Sheet", Key=var "Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Talk 1 {
    IntAdd(intVariable=var "Value", add=1, everyFrame=false)
    RunDialogue(Sheet=var "Sheet", Key="GREETER_TALK1_ACT3", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Cursed {
    PlayerDataBoolTest(boolName="BelltownGreetCursedConvo", isTrue=→"REPEAT", isFalse=→"SPEAK")
    on SPEAK → Cursed Speak
    on REPEAT → Cursed Speak R
  }

  state Cursed Speak {
    RunDialogue(Sheet="Belltown", Key="GREETER_CURSED_ACT_3", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetPlayerDataBool(boolName="BelltownGreetCursedConvo", value=true)
    on CONVO_END → End Dialogue
  }

  state Cursed Speak R {
    RunDialogue(Sheet="Belltown", Key="GREETER_CURSED_REPEAT_ACT_3", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  var Value: int = 0
  var Has Deep Melody: bool = false
  var Sheet: string = "Belltown"
  var Convo Key: string = ""
}
