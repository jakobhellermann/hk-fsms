# Control
# 1 location(s):
#   localpoolprefabs_assets_areaswamp.bundle / Pinstress CrossSlash

fsm Control {
  start Hit1

  state Hit1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="CrossSlash Effect", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    ActivateGameObject(gameObject=var "Damager1", activate=true, recursive=false, resetOnExit=true, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Flash 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flash 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Hit 2
  }

  state Hit 2 {
    ActivateGameObject(gameObject=var "Damager2", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Flash 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flash 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End
  }

  state End {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Damager1: gameObject = Pinstress CrossSlash/Damager1
  var Damager2: gameObject = Pinstress CrossSlash/Damager2
  var Flash 1: gameObject = Pinstress CrossSlash/Sharp Flash
  var Flash 2: gameObject = Pinstress CrossSlash/Sharp Flash (1)
}
