# Conversation
# 1 location(s):
#   Bone_12 / Black Thread States/Normal World/Jail Scene 2/Thief Dummy

fsm Conversation {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wave Region", storeResult=false, sendEvent="WAVE", outOfRangeEvent="", everyFrame=true)
    on WAVE → Wave
    on GRINDLE DOOR HIT → Flinch
  }

  state Wave {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Grindle_call_out (sfxdynamic_assets_areabone.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wave")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wave Region", storeResult=false, sendEvent="", outOfRangeEvent="END", everyFrame=true)
    WaitRandom(timeMin=2.5, timeMax=3, finishEvent=→"REPEAT", realTime=false)
    on END → Wave End Pause
    on INTERACT → Wave End 2
    on GRINDLE DOOR HIT → Wave Flinch
    on REPEAT → Wave
  }

  state Wave End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wave End")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Wave End Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wave Region", storeResult=false, sendEvent="WAVE", outOfRangeEvent="", everyFrame=true)
    on FINISHED → Wave End
    on WAVE → Wave
    on INTERACT → Wave End 2
  }

  state Speak {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wave End")
    BoolTest(boolVariable=var "Activated", isTrue=→"REPEAT", isFalse=→"MEET", everyFrame=false)
    on MEET → Meet
    on REPEAT → Repeat
  }

  state Meet {
    RunDialogue(Sheet="Wanderers", Key="GRINDLE_TRAPPED1", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on CONVO_END → End
  }

  state Repeat {
    RunDialogue(Sheet="Wanderers", Key="GRINDLE_TRAPPED2", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
  }

  state End {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Nearby
  }

  state Wave End 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wave End")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Speak
  }

  state Nearby {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wave Region", storeResult=false, sendEvent="", outOfRangeEvent="END", everyFrame=true)
    on INTERACT → Speak
    on END → Leave Wave
    on GRINDLE DOOR HIT → Flinch 2
  }

  state Leave Wave {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wave")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on INTERACT → Wave End 2
    on FINISHED → Wave
  }

  state Flinch {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Door Flinch")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Wave Flinch {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Door Flinch Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wave
  }

  state Flinch 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Door Flinch")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Nearby
  }

  var Activated: bool = false
  var Audio Loop Voice: gameObject = Black Thread States/Normal World/Jail Scene 2/Thief Dummy/Audio Loop Voice
}
