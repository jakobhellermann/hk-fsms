// uses template: spotlight_control
fsm spotlight_control {
  start Inert
  on FOLLOW HERO → Follow Hero  // from any state
  on FOLLOW → Follow  // from any state
  on SCAN → Scan Dir  // from any state
  on TROBBIO KILLED → Follow  // from any state
  on END → Inert  // from any state

  state Inert {
  }

  state Follow {
    GetAngleToTarget2D(gameObject=Self, target=var "Target", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=true)
    RotateTo(gameObject=Self, targetAngle=var "Angle", speed=55)
  }

  state Follow Hero {
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=true)
    RotateTo(gameObject=Self, targetAngle=var "Angle", speed=55)
  }

  state Scan Dir {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FADE UP", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Scan L", isTrue=→"L", isFalse=→"R", everyFrame=false)
    on L → Scan L
    on R → Scan R
  }

  state Scan L {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Angle", space=0, everyFrame=false)
    EaseFloat(fromValue=var "Start Angle", toValue=var "Scan Min", floatVariable=var "Angle", time=1, speed=(unset), delay=(unset), easeType=2, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → Scan R
  }

  state Scan R {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Angle", space=0, everyFrame=false)
    EaseFloat(fromValue=var "Start Angle", toValue=var "Scan Max", floatVariable=var "Angle", time=1, speed=(unset), delay=(unset), easeType=2, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → Scan L
  }

  var Scan Min: float = 246.612
  var Scan Max: float = 312.431
  var Scan L: bool = true
  var Target: gameObject = Grand Stage Scene/Boss Scene TormentedTrobbio/Tormented Trobbio

  var Angle: float = 0
  var Start Angle: float = 0
}
