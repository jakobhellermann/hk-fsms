# Break Control
# 1 location(s):
#   Abyss_07 / Abyss_Weaver_Hanging_Plat (4)

fsm Break Control {
  start Pause

  state Init {
    ActivateGameObject(gameObject=var "Grab Tendrils", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Will Break", isTrue=(none), isFalse=→"INERT", everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    SetSpriteRendererColor(TintColor=(1, 1, 1, 1), Target=var "Plat")
    on INERT → Inert
    on ACTIVATE → Activated
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inert {
  }

  state Activated {
    ActivateGameObject(gameObject=var "Plat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Break Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Tendrils Up
  }

  state Tendrils Up {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Wait(time=0.21, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Grab Tendrils", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=radiance_tentacles_whip_up (sfxstatic_assets_areaabyss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=abyss_tendrils_emerge (sfxstatic_assets_areaabyss.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Grab Plat
  }

  state Grab Plat {
    PlayAudioEvent(audioClip=bone_boulder_shake_n_break (sfxstatic_assets_areabone.bundle), pitchMin=1.1, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    PlayAudioEvent(audioClip=breakable_wall_hit_2 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    ActivateGameObject(gameObject=var "Plat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Plat Grabbed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitter(gameObject=var "Pt Grab", emit=0, resetIfPlaying=false)
    ObjectJitterOnRender(Target=Self, X=0.05, Y=0.05, Z=(unset), LimitFps=30)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pull In
  }

  state Pull In {
    AudioStop(gameObject=var "Audio Player", fadeTime=0)
    PlayAudioEvent(audioClip=breakable_wall_death (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=big splash (sfxstatic_assets_water.bundle), pitchMin=1.1, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    PlayParticleEmitter(gameObject=var "Pt PullIn", emit=0, resetIfPlaying=false)
    Translate(gameObject=var "Plat Grabbed", vector=(unset), x=(unset), y=var "Pull Speed", z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Grab Tendrils", vector=(unset), x=(unset), y=var "Pull Speed", z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
  }

  var Will Break: bool = false

  var Pull Speed: float = -40
  var Activated: bool = false
  var Break Range: gameObject = Abyss_Weaver_Hanging_Plat (4)/Break Range

  var BG Pole: gameObject = Abyss_Weaver_Hanging_Plat (4)/BG Pole

}
