# Control
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Brew Surprise Flea

fsm Control {
  start Init

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Flea Festival Brew Pot")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Inactive
  }

  state Ready {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Pop Out
  }

  state Pop Out {
    PlayAudioEvent(audioClip=acid_splash (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetParent(gameObject=var "Splash Out Effects", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendMessage(gameObject=var "Flea Festival Brew Pot", delivery=SendMessage, options=0, functionCall=StartTimedJitter())
    ActivateGameObject(gameObject=var "Splash Out Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashingQuickened())
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spin Instant")
    PlayParticleEmitter(gameObject=var "Pt Quickened", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Quickened", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Drip", emit=0, resetIfPlaying=false)
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=10)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, 6.5, 0), time=0.75, speed=(unset), delay=(unset), easeType=19, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shout
  }

  state Shout {
    PlayRandomAudioClipTable(Table=Flea_Howl (sfxdynamic_assets_fleacaravan.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    PlayParticleEmitter(gameObject=var "Pt QuickenedStart", emit=0, resetIfPlaying=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shout End
  }

  state Shout End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RecoverToFly")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  state Shake {
    PlayParticleEmitterInState(gameObject=var "Pt Shake")
    StopParticleEmitter(gameObject=var "Pt Drip")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shake")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetAudioClip(gameObject=Self, audioClip=shellwood_goomba_flyer_grounded_shake (sfxstatic_assets_areashellwood.bundle), autoPlay=true, stopOnExit=false)
    PlayRandomAudioClipTable(Table=Flea_Shake (sfxdynamic_assets_fleacaravan.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Fly Away
  }

  state Fly Away {
    PlayRandomAudioClipTable(Table=flea_fly_away (sfxdynamic_assets_fleacaravan.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    GetPosition(gameObject=var "FlyAway Point", vector=var "FlyAway Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RecoverToFly")
    AnimatePositionTo(gameObject=Self, toValue=var "FlyAway Pos", localSpace=false, time=0.6, speed=(unset), delay=(unset), easeType=24, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetAudioClip(gameObject=Self, audioClip=flee_fly_loop (sfxstatic_assets_pilgrimslowerpilgrimsupper.bundle), autoPlay=true, stopOnExit=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Inactive {
    on ANY GAME COMPLETED → Check
    on TEST → Ready
  }

  state Check {
    RandomInt(min=1, max=100, storeResult=var "Chance", inclusiveMax=true, noRepeat=false)
    IntCompare(integer1=var "Chance", integer2=6, equal=→"CANCEL", lessThan=→"READY", greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → Inactive
    on READY → Ready
  }

  var Chance: int = 0
  var FlyAway Pos: vector3 = (0, 0, 0)
  var Parent: gameObject = <null>
  var Flea Festival Brew Pot: gameObject = <null>
  var Splash Out Effects: gameObject = Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Brew Surprise Flea/Splash Out Effects
  var Pt Quickened: gameObject = Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Brew Surprise Flea/Pt Quickened
  var Pt QuickenedStart: gameObject = Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Brew Surprise Flea/Pt QuickenedStart
  var Pt Shake: gameObject = Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Brew Surprise Flea/Pt Shake
  var Pt Drip: gameObject = Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Brew Surprise Flea/Pt Drip
  var FlyAway Point: gameObject = Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Brew Surprise Flea/FlyAway Point
}
