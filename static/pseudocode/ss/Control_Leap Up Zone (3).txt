# Control
# 3 location(s):
#   Bone_15 / Boss Scene/Leap Up Zone (3)
#   Bone_15 / Boss Scene/Leap Up Zone (2)
#   Bonetown_boss / Boss Scene/Leap Up Zone

fsm Control {
  start Idle

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer=(unset), sendEvent=→"ENTER", storeCollider=var "Skull King")
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="Boss", collideLayer=(unset), sendEvent=→"EXIT", storeCollider=var "Skull King")
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SetFsmBool(gameObject=var "Skull King", fsmName="Control", variableName="Leap Up Zone", setValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  state Exit {
    SetFsmBool(gameObject=var "Skull King", fsmName="Control", variableName="Leap Up Zone", setValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Skull King: gameObject = <null>
}
