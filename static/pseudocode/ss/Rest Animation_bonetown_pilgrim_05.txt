# Rest Animation
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/Pilgrim Crowds/Crowds/Crowd 1/bonetown_pilgrim_05

fsm Rest Animation {
  start Idle

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=LookAnimNPC.PreviousState on Black Thread States/Normal World/Pilgrim Crowds/Crowds/Crowd 1/bonetown_pilgrim_05@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Look Down
    on RIGHT → Return to Left
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="ShouldFaceLeft", parameters=[], storeResult=var "Should Face Left", EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "Should Face Left"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Return to Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="FG Large Look Down Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Default
  }

  state Look Down {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="FG Large Look Down", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Default
  }

  state Default {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="FG Large Idle Frame")
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"BOB", realTime=false)
    on CANCEL → Look Up
    on BOB → Bob
  }

  state Bob {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="FG Large Idle Bob", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Default
  }

  state Look Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="FG Large Look Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  var Max Turn Delay: float = 5
  var Min Turn Delay: float = 3

  var Resting: bool = false
  var Should Face Left: bool = false
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
