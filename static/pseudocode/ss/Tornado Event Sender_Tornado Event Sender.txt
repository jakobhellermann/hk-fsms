// uses template: tornado_event_sender
fsm Tornado Event Sender {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"TRIGGER", storeCollider=var "Collider")
    on TRIGGER → Send
  }

  state Send {
    SendEventByNameUpwards(Target=var "Collider", EventName="TORNADO")
    SendMessage(gameObject=var "Collider", delivery=SendMessage, options=1, functionCall=TornadoEffect())
    on FINISHED → Idle
  }

  var Collider: gameObject = <null>
}
