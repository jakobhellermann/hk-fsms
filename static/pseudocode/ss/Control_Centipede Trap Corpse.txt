# Control
# 1 location(s):
#   enemycorpses_assets_areacradle.bundle / Centipede Trap Corpse

fsm Control {
  start Init

  state Init {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Corpse Spin")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartBounce())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashArmoured())
    Wait(time=0.1, finishEvent=(none), realTime=false)
    RandomlyFlipScale(gameObject=Self)
    RandomFloat(min=0.8, max=1, storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Spin
  }

  state Spin {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=true, lateUpdate=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on BOUNCE → Flail
    on FINISHED → Flail
  }

  state Flail {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Corpse Flail")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Y", space=0, everyFrame=true)
    on BOUNCE → Pop
    on FINISHED → Pop
  }

  state Pop {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Corpse Pop", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopBounce())
    PlayRandomAudioClipTableV2(Table=splat (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Splat", emit=0, resetIfPlaying=false)
    on FINISHED → Recycle
  }

  sequence state Recycle {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=2, finishEvent=(none), realTime=false)
    RecycleSelf()
  }

  var Scale: float = 0
  var Velocity Y: float = 0
  var Pt Splat: gameObject = Centipede Trap Corpse/Pt Splat
}
