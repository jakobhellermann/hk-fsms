fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=6)
    on ENTER → Delay
  }

  state Play Tension {
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=6)
    on TENSION END → Stop Tension
    on LEAVING SCENE → Stop Tension
    on PAUSE TENSION → Idle
  }

  state Stop Tension {
    AudioStopV2(gameObject=Self, fadeTime=1, cancelOnEarlyExit=false)
  }

  state Delay {
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Play Tension
    on LEAVING SCENE → Stop Tension
    on TENSION END → Stop Tension
    on PAUSE TENSION → Idle
  }

  var Delay: float = 0
}
