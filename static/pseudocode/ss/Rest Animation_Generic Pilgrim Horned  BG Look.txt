# Rest Animation
# 1 location(s):
#   Belltown / Black Thread States Thread Only Variant/Normal World/Pilgrim Crowds/Crowds/Crowd 6/Generic Pilgrim Horned  BG Look

fsm Rest Animation {
  start Idle
  on DEACTIVATED → Idle  // from any state

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Black Thread States Thread Only Variant/Normal World/Pilgrim Crowds/Crowds/Crowd 6/Generic Pilgrim Horned  BG Look@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Rest Idle
    on RIGHT → Rest From Right
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[true], storeResult=(unset var), EveryFrame=false)  // disabled
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Horned Rest End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Rest From Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Horned Turn Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Rest Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Horned Rest", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    on FINISHED → Rest End
    on CANCEL → Reset
  }

  var Max Turn Delay: float = 5
  var Min Turn Delay: float = 3

  var Resting: bool = false
  var Should Face Left: bool = false
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
