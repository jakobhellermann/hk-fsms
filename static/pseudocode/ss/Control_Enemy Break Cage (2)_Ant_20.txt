# Control
# 1 location(s):
#   Ant_20 / Enemy Break Cage (2)

fsm Control {
  start Pause
  on INSTA BREAK → Break  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetChild(gameObject=var "Enemy Holder", childName="", withTag="Untagged", storeResult=var "Caged Enemy")
    SetFsmBool(gameObject=var "Caged Enemy", fsmName="Control", variableName="Caged", setValue=true, everyFrame=false)
    SetFsmGameObject(gameObject=var "Caged Enemy", fsmName="Control", variableName="Cage", setValue=var "Self", everyFrame=false)
    FindChild(gameObject=Self, childName="Corpse", storeResult=var "Corpse")
    GetChild(gameObject=var "Corpse", childName="", withTag="Untagged", storeResult=var "Caged Corpse")
    on FINISHED → Init
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=true, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    FloatMultiply(floatVariable=var "Angle", multiplyBy=0.95, everyFrame=false)
    AnimateRotationTo(gameObject=var "Sprite Parent", fromValue=var "Angle", toValue=0, worldSpace=true, negativeSpace=false, time=0.45, speed=(unset), delay=(unset), easeType=27, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on HIT → Hit Dir
    on HIT L → Hit L
    on HIT R → Hit R
  }

  state Hit {
    FlingObject(flungObject=var "Caged Corpse", speedMin=5, speedMax=10, angleMin=70, angleMax=100)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    PlayAudioEventRandom(audioClips=[hunter_cage_hit_1 (sfxstatic_assets_areaant.bundle), hunter_cage_hit_2 (sfxstatic_assets_areaant.bundle)], pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmissionChildren(Target=var "Pt Hit", SetEmission=true, EveryFrame=false, stopOnExit=true)
    ObjectJitterLocal(gameObject=var "Sprite", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=0)
    AnimateRotationTo(gameObject=var "Sprite Parent", fromValue=0, toValue=var "Angle", worldSpace=true, negativeSpace=false, time=0.1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on BREAK → Break
  }

  state Break {
    ActivateGameObject(gameObject=var "Chunks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Caged Enemy", fsmName="Control", variableName="Caged", setValue=false, everyFrame=false)
    PlayAudioEvent(audioClip=hunter_cage_break (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitterChildren(gameObject=var "Pt Break", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Sprite Parent", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="RESCUE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Caged Enemy"), sendEvent="CAGE BREAK", delay=0, everyFrame=false)
  }

  state Init {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SetParent(gameObject=var "Caged Enemy", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Hit Dir {
    CheckTargetDirection(gameObject=Self, target=var "Damager", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Hit L
    on R → Hit R
  }

  state Hit L {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Effect")
    RandomFloat(min=-10, max=10, storeResult=var "Effect Angle")
    SetRotation(gameObject=var "Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Effect Angle", space=0, everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Angle", floatValue=-6, everyFrame=false)
    SetScale(gameObject=var "Effect", vector=(unset), x=-2, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Hit
  }

  state Hit R {
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Effect")
    RandomFloat(min=-10, max=10, storeResult=var "Effect Angle")
    SetRotation(gameObject=var "Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Effect Angle", space=0, everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Angle", floatValue=6, everyFrame=false)
    SetScale(gameObject=var "Effect", vector=(unset), x=2, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Hit
  }

  var Hits: int = 3
  var Break Notify: gameObject = <null>

  var Angle: float = 0
  var Effect Angle: float = 0
  var Activated: bool = false
  var Caged Enemy: gameObject = <null>
  var Damager: gameObject = <null>
  var Effect: gameObject = <null>
  var Enemy Holder: gameObject = Enemy Break Cage (2)/Enemy
  var Pt Break: gameObject = Enemy Break Cage (2)/Pt Break
  var Pt Hit: gameObject = Enemy Break Cage (2)/Pt Hit
  var Self: gameObject = Enemy Break Cage (2)
  var Sprite: gameObject = Enemy Break Cage (2)/Sprite Parent/Sprite
  var Sprite Parent: gameObject = Enemy Break Cage (2)/Sprite Parent
  var Corpse: gameObject = <null>
  var Caged Corpse: gameObject = <null>
  var Chunks: gameObject = Enemy Break Cage (2)/Chunks
}
