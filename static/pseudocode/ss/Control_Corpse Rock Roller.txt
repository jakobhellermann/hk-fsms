# Control
# 1 location(s):
#   enemycorpses_assets_areabone.bundle / Corpse Rock Roller

fsm Control {
  start Wait

  state Wait {
    SetAudioSource(gameObject=Self, active=true)
    Wait(time=0.36, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic 1
  }

  state Antic 1 {
    PlayParticleEmitterInState(gameObject=var "Pt Trail")
    Wait(time=0.85, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterInState(gameObject=var "Pt Trail 2")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashingTarSlug())
    PlayAudioEvent(audioClip=big_enemy_death_pt1 (sfxstatic_assets_enemyheavy.bundle), pitchMin=1.1, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShakeRepeatingV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.15, vibrate=true)
    on FINISHED → Explode
  }

  state Antic 2 {
    PlayParticleEmitterInState(gameObject=var "Pt Trail 2")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Explode
  }

  state Explode {
    SetAudioSource(gameObject=Self, active=false)
    PlayAudioEvent(audioClip=rock_roller_explode (sfxstatic_assets_areabone.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Explode", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Explosion", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    RandomFloat(min=-3, max=-5, storeResult=var "Jump X")
    RandomFloat(min=18, max=20, storeResult=var "Distance")
    SpawnObjectFromGlobalPool(gameObject=Rock Roller Bomb (localpoolprefabs_assets_areabone.bundle), spawnPoint=var "Bomb Spawn Point", position=(-1, 0, -0.001), rotation=(unset), storeObject=var "Bomb")
    SetVelocity2d(gameObject=var "Bomb", vector=(unset), x=var "Jump X", y=var "Distance", everyFrame=false)
    RandomFloat(min=3, max=5, storeResult=var "Jump X")
    RandomFloat(min=18, max=20, storeResult=var "Distance")
    SpawnObjectFromGlobalPool(gameObject=Rock Roller Bomb (localpoolprefabs_assets_areabone.bundle), spawnPoint=var "Bomb Spawn Point", position=(1, 0, -0.001), rotation=(unset), storeObject=var "Bomb")
    SetVelocity2d(gameObject=var "Bomb", vector=(unset), x=var "Jump X", y=var "Distance", everyFrame=false)
    RandomFloat(min=-2, max=2, storeResult=var "Jump X")
    RandomFloat(min=18, max=20, storeResult=var "Distance")
    SpawnObjectFromGlobalPool(gameObject=Rock Roller Bomb (localpoolprefabs_assets_areabone.bundle), spawnPoint=var "Bomb Spawn Point", position=(0, 1.25, -0.001), rotation=(unset), storeObject=var "Bomb")
    SetVelocity2d(gameObject=var "Bomb", vector=(unset), x=var "Jump X", y=var "Distance", everyFrame=false)
    SendEventToRegister(eventName="ROCK ROLLER EXPLODED")
    on FINISHED → Spawn Pickup
  }

  state Inactive {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Spawn Pickup {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    CreateObject(gameObject=Collectable Item Pickup Instant (localpoolprefabs_assets_currency.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Spawned Drop")
    SetCollectablePickupItemV2(Target=var "Spawned Drop", Item=Rock Rollers (dataassets_assets_assets/dataassets/questsystem/quests.bundle), KeepPersistence=true)
    GetProperty(targetProperty=Quest.name on Rock Rollers (dataassets_assets_assets/dataassets/questsystem/quests.bundle), everyFrame=false)
    FormatString(format="{0}_item_{1}", variables=["Rock Roller", var "Item Name"], storeResult=var "Pickup Name", everyFrame=false)
    SetName(gameObject=var "Spawned Drop", name=var "Pickup Name")
    FlingObject(flungObject=var "Spawned Drop", speedMin=15, speedMax=30, angleMin=80, angleMax=100)
    on FINISHED → Inactive
  }

  var Distance: float = 0
  var Jump X: float = 36.53
  var Item Name: string = ""
  var Pickup Name: string = ""
  var Explosion: gameObject = Corpse Rock Roller/Explosion
  var Bomb Spawn Point: gameObject = Corpse Rock Roller/Bomb Spawn Point
  var Pt Trail: gameObject = Corpse Rock Roller/Pt Trail
  var Pt Trail 2: gameObject = Corpse Rock Roller/Pt Trail 2
  var Self: gameObject = <null>
  var Spawned Drop: gameObject = <null>
  var Bomb: gameObject = <null>
}
