# Control
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/Roar Wave Emitter Small
#   Menu_Title / _GameCameras/Roar Wave Emitter Small

fsm Control {
  start Pause
  on ROAR WAVE END → Final Burst  // from any state

  state Emit {
    SetSpriteRenderer(gameObject=var "Wave 2", active=true)
    SetSpriteRenderer(gameObject=var "Lines", active=false)
    SetScale(gameObject=var "Wave 1", vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Wave 2", vector=(unset), x=5.5, y=5.5, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleBy(gameObject=var "Wave 1", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=1, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Wave 2", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=1, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on END → Final Burst
  }

  state Emit Start {
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetSpriteRenderer(gameObject=var "Wave 1", active=true)
    SetSpriteRenderer(gameObject=var "Lines", active=true)
    SetParticleEmission(gameObject=var "Rubble", emission=true, resetOnExit=false)
    SetParticleEmission(gameObject=var "Rubble FG", emission=true, resetOnExit=false)
    iTweenScaleBy(gameObject=var "Wave 1", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Lines", id="", vector=(6, 6, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish Lines
    on END → Final Burst
  }

  state Finish Lines {
    SetScale(gameObject=var "Wave 1", vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    iTweenScaleBy(gameObject=var "Wave 1", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Wave 2", id="", vector=(5.5, 5.5, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleBy(gameObject=var "Lines", id="", vector=(6, 6, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emit
    on END → Final Burst
  }

  state End {
    SetSpriteRenderer(gameObject=var "Wave Final", active=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Particle
  }

  state Finish {
    on FINISHED → Idle
  }

  state End Particle {
    SetParticleEmission(gameObject=var "Rubble", emission=false, resetOnExit=false)
    SetParticleEmission(gameObject=var "Rubble FG", emission=false, resetOnExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmissionRate(gameObject=var "Rubble", emissionRate=25, everyFrame=false)
    SetParticleEmissionRate(gameObject=var "Rubble FG", emissionRate=15, everyFrame=false)
    on FINISHED → Finish
  }

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    SetSpriteRenderer(gameObject=var "Wave 1", active=false)
    SetSpriteRenderer(gameObject=var "Wave 2", active=false)
    SetSpriteRenderer(gameObject=var "Lines", active=false)
    SetScale(gameObject=var "Lines", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Wave 1", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Wave 2", vector=(unset), x=5.5, y=5.5, z=5.5, everyFrame=false, lateUpdate=false)
    on START → Emit Start
  }

  state Final Burst {
    SetSpriteRenderer(gameObject=var "Lines", active=false)
    SetSpriteRenderer(gameObject=var "Wave 1", active=false)
    SetSpriteRenderer(gameObject=var "Wave 2", active=false)
    SetSpriteRenderer(gameObject=var "Wave Final", active=true)
    SetScale(gameObject=var "Wave Final", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    iTweenScaleBy(gameObject=var "Wave Final", id="", vector=(9, 9, 0), time=0.15, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    on FINISHED → End
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Lines: gameObject = _GameCameras/Roar Wave Emitter Small/lines
  var Rubble: gameObject = _GameCameras/Roar Wave Emitter Small/Rubble Fall
  var Rubble FG: gameObject = _GameCameras/Roar Wave Emitter Small/Rubble FG
  var Wave 1: gameObject = _GameCameras/Roar Wave Emitter Small/wave 1
  var Wave 2: gameObject = _GameCameras/Roar Wave Emitter Small/wave 2
  var Wave Final: gameObject = _GameCameras/Roar Wave Emitter Small/wave final
}
