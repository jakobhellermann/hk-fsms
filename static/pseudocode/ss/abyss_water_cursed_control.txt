# abyss_water_cursed_control
# 3 location(s):
#   Abyss_05 / Abyss Water Control
#   Abyss_07 / Abyss Water Control
#   Abyss_08 / Abyss Water Control

// uses template: abyss_water_cursed_control
fsm abyss_water_cursed_control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Surface Water Region")
    FindNamedChild(gameObject=Self, storeResult=var "Cursed Damage Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Damage Dealer")
    CollectableItemGetDataV2(CollectedAmount=(unset), IsHoldingAny=var "Has White Flower", NeededAmount=0, IsCollected=(none), NotCollected=(none), ItemName="", Sprite=var "", Item=White Flower (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle))
    CheckIfCrestEquipped(Crest=Cursed (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=(none), falseEvent=(none), storeValue=var "Is Cursed")
    BoolAllTrue(boolVariables=[var "Has White Flower", var "Is Cursed"], sendEvent=→"CURSED", storeResult=(unset), everyFrame=false)
    on CURSED → Set Cursed
  }

  state Set Cursed {
    ActivateGameObject(gameObject=var "Cursed Damage Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Surface Water Region", active=false, resetOnExit=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Cursed Damage Collider", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Cursed Bind
  }

  state Cursed Bind {
    PlayAudioEvent(audioClip=enemy_splash_in_water (sfxstatic_assets_water.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=hollow_shade_startle (sfxstatic_assets_areaabyss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=Splash In Black (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(0, 0, 0), rotation=(unset), storeObject=var "Splash")
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Splash", position=(0, 1, 0), spawnMin=30, spawnMax=30, speedMin=15, speedMax=25, angleMin=70, angleMax=110, originVariationX=2, originVariationY=0, FSM="", FSMEvent="")
    SetScale(gameObject=var "Splash", vector=(unset), x=2.5, y=2.5, z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FORCE CURSED BIND ABYSS", delay=0, everyFrame=false)
    Wait(time=0.4, finishEvent=(none), realTime=false)
    on FINISHED → Deal Damage
  }

  state Deal Damage {
    DamageHeroDirectly(damager=var "Damage Dealer", damageAmount=1, spikeHazard=true, sinkHazard=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  var Has White Flower: bool = false
  var Is Cursed: bool = false
  var Surface Water Region: gameObject = <null>
  var Cursed Damage Collider: gameObject = <null>
  var Damage Dealer: gameObject = <null>
  var Splash: gameObject = <null>
}
