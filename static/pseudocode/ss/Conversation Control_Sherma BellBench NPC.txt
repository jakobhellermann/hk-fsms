# Conversation Control
# 1 location(s):
#   Bellway_08 / Bell Bench Scene/Sherma BellBench NPC

fsm Conversation Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Asleep
  }

  state Asleep {
    BoolTest(boolVariable=var "Activated", isTrue=→"AWAKE", isFalse=(none), everyFrame=false)  // disabled
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.3, None=(none), ActiveInner=→"WAKE", ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SetRandomAudioClipFromTable(Table=sherma_sleep_loop (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    on INTERACT → Wake Sherma
    on WAKE → S Wake 1
    on AWAKE → S Wake 2
    on BENCH RAISE → S Wake 1
  }

  state Wake Sherma {
    PlayerDataBoolTest(boolName="shermaWokeInSteps", isTrue=→"AWAKE", isFalse=(none))
    SetPlayerDataBool(boolName="metSherma", value=true)
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_WAKE_1", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on CONVO_END → Wake 1
    on AWAKE → Wake 1
  }

  state Wake 1 {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=Self, UseChildren=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=trobbio_recover_or_general_foley_movement (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Dir Check 2
  }

  state Wake 2 {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="SHERMA")
    PlayerDataBoolTest(boolName="shermaWokeInSteps", isTrue=→"REWAKE", isFalse=(none))
    SetPlayerDataBool(boolName="shermaWokeInSteps", value=true)
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_WAKE_2", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
    on REWAKE → Wake 3
  }

  state End {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Listen End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    FaceObjectV5(ObjectA=Self, ObjectB=var "Hero", SpriteFacesRight=true, NewAnimationClip="TurnToIdle", ResetFrame=true, PauseBetweenTurns=0.5, StoreDuration=(unset), turnAudioClipTables=[small_movement_cloak (sfxdynamic_assets_shared.bundle)], EveryFrame=true)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=0, MinReactDelay=1.4, MaxReactDelay=1.8, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT → Dir Check
    on SING → Sing
    on STARTLE → Startle
  }

  state Bench Check {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="SHERMA")
    SetPlayerDataBool(boolName="metSherma", value=true)
    GetFsmBool(gameObject=var "bell_toll_machine", fsmName="Unlock Behaviour", variableName="Activated", storeValue=var "Bench Activated", everyFrame=false)
    BoolTest(boolVariable=var "Bench Activated", isTrue=→"BENCH", isFalse=→"NOBENCH", everyFrame=false)
    on NOBENCH → Nobench Convo
    on BENCH → Bench Convo
  }

  state Bench Talk {
    SetPlayerDataBool(boolName="shermaConvoCoralBench", value=true)
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_BENCH", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Look Back
  }

  state S Wake 1 {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayAudioEvent(audioClip=trobbio_recover_or_general_foley_movement (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=sherma_alert (sfxdynamic_assets_sherma.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → S Wake 2
  }

  state S Wake 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  state Dir Check {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Listen")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Bench Check
    on R → Look Back
  }

  state Look Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookL")
    SetBoolValue(boolVariable=var "Looking Back", boolValue=true, everyFrame=false)
    on FINISHED → Bench Check
  }

  state Dir Check 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Listen")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Wake 2
    on R → Look Back 2
  }

  state Look Back 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookL")
    SetBoolValue(boolVariable=var "Looking Back", boolValue=true, everyFrame=false)
    on FINISHED → Wake 2
  }

  state End Look Back {
    BoolTest(boolVariable=var "Looking Back", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Looking Back", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookR")
    on FINISHED → End
  }

  state Nobench Convo {
    BoolTest(boolVariable=var "Spoken", isTrue=→"REPEAT", isFalse=→"TALK", everyFrame=false)
    on TALK → Nobench Talk
    on REPEAT → Nobench Repeat
  }

  state Nobench Talk {
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    PlayerDataBoolTest(boolName="hasActivatedBellBench", isTrue=(none), isFalse=→"ALT")
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_NOBENCH", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Look Back
    on ALT → Nobench Talk Alt
  }

  state Nobench Repeat {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_NOBENCH_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Look Back
  }

  state Bench Convo {
    PlayerDataBoolTest(boolName="shermaConvoCoralBench", isTrue=(none), isFalse=→"BENCH")
    PlayerDataBoolTest(boolName="shermaConvoCoralJudges", isTrue=(none), isFalse=→"JUDGE")
    on BENCH → Bench Talk
    on JUDGE → Judge Convo
    on FINISHED → Repeat
  }

  state Judge Convo {
    SetPlayerDataBool(boolName="shermaConvoCoralJudges", value=true)
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_JUDGE", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Look Back
  }

  state Repeat {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Look Back
  }

  state Face Bench {
    CheckFacingTarget(facingObject=Self, target=var "bell_toll_machine", spriteFacesRight=true, everyFrame=false, facingEvent=→"FINISHED", notFacingEvent=→"TURN", facingBool=(unset), notFacingBool=(unset))
    on FINISHED → Bench Idle
    on TURN → Turn To Bench
  }

  state Bench Idle {
    on INTERACT → Look Back
  }

  state Turn To Bench {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlipScaleOnExit(gameObject=Self, flipHorizontally=true, flipVertically=false)
    on FINISHED → Start Bench Idle
  }

  state Start Bench Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Bench Idle
  }

  state Wake 3 {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_REWAKE", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
  }

  state Nobench Talk Alt {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CORAL_NOBENCH_ALT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Look Back
  }

  state Sing {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EXCITED", delay=0, everyFrame=false)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=sherma_sing (sfxdynamic_assets_sherma.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=0, MinReactDelay=0.6, MaxReactDelay=0.8, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Sing End
  }

  state Sing End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EXCITED END", delay=0, everyFrame=false)
    on READY → S Wake 2
  }

  state Startle {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle F", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=sherma_alert (sfxdynamic_assets_sherma.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → S Wake 2
  }

  var bell_toll_machine: gameObject = Bell Bench Scene/bell_bench/bell_toll_machine

  var Spoken: bool = false
  var Activated: bool = false
  var Looking Back: bool = false
  var Bench Activated: bool = false
  var Audio Loop Voice: gameObject = Bell Bench Scene/Sherma BellBench NPC/Audio Loop Voice
}
