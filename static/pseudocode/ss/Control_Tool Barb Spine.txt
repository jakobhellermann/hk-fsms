# Control
# 1 location(s):
#   herodynamic_assets_all.bundle / Tool Barb Spine

fsm Control {
  start Init
  on LAVA → Break  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Spark", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    NextFrameEvent(sendEvent=(none))
    FindNamedChild(gameObject=Self, storeResult=var "Pt Poison Thunk")
    on FINISHED → Poison?
  }

  state Fire {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Splinter Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=50, everyFrame=false)
    SetProperty(targetProperty=DamageEnemies.direction on Tool Barb Spine@DamageEnemies, everyFrame=false)
    Wait(time=5, finishEvent=→"END", realTime=false)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"THUNK", storeCollider=var "Thunked Collider", storeForce=(unset))
    on THUNK → Thunk
    on END → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Thunk {
    PlayParticleEmitter(gameObject=var "Pt Poison Thunk", emit=0, resetIfPlaying=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Splinter Thunk", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayAudioEventRandom(audioClips=[spine_floater_projectile_impact_1 (sfxstatic_assets_shared.bundle), spine_floater_projectile_impact_2 (sfxstatic_assets_shared.bundle), spine_floater_projectile_impact_3 (sfxstatic_assets_shared.bundle), spine_floater_projectile_impact_4 (sfxstatic_assets_shared.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Collider2DIsActive(Target=var "Thunked Collider", TrueEvent=(none), FalseEvent=→"SMASH", StoreValue=(unset), EveryFrame=true)
    WaitRandom(timeMin=1.8, timeMax=2.2, finishEvent=→"END", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"SMASH", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on END → Break Antic
    on SMASH → Break
  }

  state Break Antic {
    Collider2DIsActive(Target=var "Thunked Collider", TrueEvent=(none), FalseEvent=→"SMASH", StoreValue=(unset), EveryFrame=true)
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"SMASH", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Break
    on SMASH → Break
  }

  state Break {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Spark", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Poison? {
    Tk2dSpriteSetColor(gameObject=var "Pt Shatter", color=(1, 1, 1, 1), everyframe=false)
    BoolTest(boolVariable=var "Is Poison", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    GetPoisonTintColour(colorVariable=var "Poison Colour")
    Tk2dSpriteSetColor(gameObject=var "Pt Shatter", color=var "Poison Colour", everyframe=false)
    PlayParticleEmitter(gameObject=var "Pt Poison Trail", emit=0, resetIfPlaying=false)
    on FINISHED → Fire
  }

  var Is Poison: bool = false

  var Angle: float = 0
  var Poison Colour: color = (0.9647059, 0.83137256, 0.99215686, 1)
  var Pt Shatter: gameObject = Tool Barb Spine/Pt Shatter
  var Self: gameObject = <null>
  var Spark: gameObject = Tool Barb Spine/Spark
  var Terrain Detector: gameObject = Tool Barb Spine/Terrain Detector
  var Thunk Effect: gameObject = Tool Barb Spine/Thunk Effect
  var Thunked Collider: gameObject = <null>
  var Pt Poison Trail: gameObject = Tool Barb Spine/Pt Poison Trail
  var Pt Poison Thunk: gameObject = <null>
}
