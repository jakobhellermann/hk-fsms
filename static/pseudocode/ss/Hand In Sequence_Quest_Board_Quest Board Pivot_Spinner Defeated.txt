# Hand In Sequence
# 1 location(s):
#   Belltown / Town States/Spinner Defeated/Quest Board Pivot/Quest_Board

fsm Hand In Sequence {
  start Idle

  state Idle {
    on DONATE → Donate Sequence
    on QUESTS → Quests Sequence
    on QUESTS_NOCONSUME → Quests Sequence NoConsume
  }

  state Process {
    CallMethodProper(gameObject=Self, behaviour="QuestBoardInteractable", methodName="ProcessQueuedCompletions", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  sequence state Quests Sequence {
    SendEventByNameV2(eventTarget=GameObject(var "AreaTitle"), sendEvent="NPC CONVO START", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimationWait(Target=var "Hero", ClipName="Collect Stand 1", AnimationCompleteEvent=(none))
    Wait(time=0.2, finishEvent=(none), realTime=false)
    SpawnObjectFromGlobalPoolV2(gameObject=Item Get Effect R (globalpoolprefabs_assets_all.bundle), spawnPoint=Town States/Spinner Defeated/Quest Board Pivot/Quest_Board/Turn In Effect Point, position=(unset), rotation=(unset), amount=1, storeObject=var "", setParent=var "")
    CallMethodProper(gameObject=var "Deposit Box", behaviour="CurveOffsetAnimation", methodName="StartAnimation", parameters=[], storeResult=(unset var), EveryFrame=false)
    PlayAudioEvent(audioClip=shiny_item_pickup (sfxstatic_assets_currency.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Town States/Spinner Defeated/Quest Board Pivot/Quest_Board/Turn In Effect Point, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CallMethodProper(gameObject=Self, behaviour="QuestBoardInteractable", methodName="DepositQuestItems", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWait(Target=var "Hero", ClipName="Collect Stand 3", AnimationCompleteEvent=(none))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Process
  }

  sequence state Donate Sequence {
    SendEventByNameV2(eventTarget=GameObject(var "AreaTitle"), sendEvent="NPC CONVO START", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimationWait(Target=var "Hero", ClipName="Collect Stand 1", AnimationCompleteEvent=(none))
    Wait(time=0.25, finishEvent=(none), realTime=false)
    CallMethodProper(gameObject=Self, behaviour="QuestBoardInteractable", methodName="TakeDonateCurrency", parameters=[], storeResult=(unset var), EveryFrame=false)
    SpawnObjectFromGlobalPoolV2(gameObject=Item Get Effect R (globalpoolprefabs_assets_all.bundle), spawnPoint=Town States/Spinner Defeated/Quest Board Pivot/Quest_Board/Turn In Effect Point, position=(unset), rotation=(unset), amount=1, storeObject=var "", setParent=var "")
    CallMethodProper(gameObject=var "Deposit Box", behaviour="CurveOffsetAnimation", methodName="StartAnimation", parameters=[], storeResult=(unset var), EveryFrame=false)
    PlayAudioEvent(audioClip=shiny_item_pickup (sfxstatic_assets_currency.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Town States/Spinner Defeated/Quest Board Pivot/Quest_Board/Turn In Effect Point, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Wait for Counter
  }

  sequence state Donate Sequence 2 {
    CallMethodProper(gameObject=Self, behaviour="QuestBoardInteractable", methodName="OnDonationSequenceComplete", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  sequence state Donation Sequence Reward {
    Tk2dPlayAnimationWait(Target=var "Hero", ClipName="Collect Stand 3", AnimationCompleteEvent=(none))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    EndQuestV2(ConsumeCurrency=false, ShowPrompt=true, Quest=var "Quest")
    ObjectCompare(objectVariable=var "Reward", compareTo=<null>, equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    GetQuestRewardV2(StoreReward=var "Reward", StoreAmount=var "Reward Amount", Quest=var "Quest")
    SavedItemGetV2(Item=var "Reward", Amount=var "Reward Amount", ShowPopup=true)
    on FINISHED → Donate Sequence 2
  }

  state Wait for Counter {
    RespondToCurrencyCounterEvents(CurrencyType=Money, StateEvent=Stopped, Response=→"FINISHED")
    RespondToCurrencyCounterEvents(CurrencyType=Shard, StateEvent=Stopped, Response=→"FINISHED")
    Wait(time=10, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Donation Sequence Reward
  }

  state Process NoConsume {
    CallMethodProper(gameObject=Self, behaviour="QuestBoardInteractable", methodName="ProcessQueuedCompletions", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  sequence state Quests Sequence NoConsume {
    SendEventByNameV2(eventTarget=GameObject(var "AreaTitle"), sendEvent="NPC CONVO START", delay=0, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="QuestBoardInteractable", methodName="DepositQuestItems", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Process NoConsume
  }

  var Reward Amount: int = 0
  var Deposit Box: gameObject = Town States/Spinner Defeated/Quest Board Pivot/Quest_Board/Deposit Box Animator
  var Quest: object = <null>
  var Reward: object = <null>
}
