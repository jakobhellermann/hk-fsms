# Slide Out
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas
#   Menu_Title / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas

fsm Slide Out {
  start Init
  on DISABLE → Reset  // from any state

  state Init {
    SetBoolValue(boolVariable=var "Is Visible", boolValue=true, everyFrame=false)
    GetPosition(gameObject=Self, vector=var "Start Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    SetVector3Value(vector3Variable=var "Off position", vector3Value=var "Start Position", everyFrame=false)
    Vector3AddXYZ(vector3Variable=var "Off position", addX=0, addY=50, addZ=0, everyFrame=false, perSecond=false)
    on FINISHED → Idle
  }

  state Idle {
    CallMethodProper(gameObject=var "Delivery Icon", behaviour="DeliveryHudIcon", methodName="HudOut", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    on OUT → Go Out
    on OUT INSTANT → Go Out Instant
  }

  state Go Out {
    SetBoolValue(boolVariable=var "Is Visible", boolValue=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Liquid", active=false)
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(0.75, 0.75, 0.75), time=var "Scale Time", delay=0, speed=(unset), easeType=12, loopType=0, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=true)
    on IN → Come In
    on FINISHED → Out
  }

  state Come In {
    SetBoolValue(boolVariable=var "Is Visible", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Counters Parent", behaviour="CurrencyCounterStack", methodName="ReportVisibleChange", parameters=[true], storeResult=(unset var), EveryFrame=false)
    SetFsmFloat(gameObject=_GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Health, fsmName="Low Health FX", variableName="Wait Time", setValue=var "Scale Time", everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Start Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventToRegister(eventName="HUD COMING IN")
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=(1, 1, 1), time=var "Scale Time", delay=0, speed=(unset), easeType=13, loopType=0, startEvent=(none), finishEvent=→"FINISHED", realTime=true, stopOnExit=true, loopDontFinish=true)
    on OUT → Go Out
    on OUT INSTANT → Go Out Instant
    on FINISHED → In
  }

  state Out {
    CallMethodProper(gameObject=var "Counters Parent", behaviour="CurrencyCounterStack", methodName="ReportVisibleChange", parameters=[false], storeResult=(unset var), EveryFrame=false)
    SetPosition(gameObject=Self, vector=var "Off position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Delivery Icon", behaviour="DeliveryHudIcon", methodName="HudOut", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventToRegister(eventName="HUD GONE")
    on IN → In Wait
  }

  state In {
    SetMeshRenderer(gameObject=var "Liquid", active=true)
    CallMethodProper(gameObject=var "Delivery Icon", behaviour="DeliveryHudIcon", methodName="HudIn", parameters=[], storeResult=(unset var), EveryFrame=false)
    on OUT → Go Out
    on OUT INSTANT → Go Out Instant
  }

  state Go Out Instant {
    SetBoolValue(boolVariable=var "Is Visible", boolValue=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Liquid", active=false)
    SetScale(gameObject=Self, vector=(unset), x=0.75, y=0.75, z=0.75, everyFrame=false, lateUpdate=false)
    on FINISHED → Out
  }

  state Reset {
    SetPosition(gameObject=Self, vector=var "Start Position", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Is Visible", boolValue=true, everyFrame=false)
    CallMethodProper(gameObject=var "Counters Parent", behaviour="CurrencyCounterStack", methodName="ReportVisibleChange", parameters=[true], storeResult=(unset var), EveryFrame=false)  // disabled
  }

  state In Wait {
    NextFrameEvent(sendEvent=→"FINISHED")
    on OUT → Go Out
    on OUT INSTANT → Go Out Instant
    on FINISHED → Come In
  }

  var Scale Time: float = 0.1

  var Is Visible: bool = false
  var Off position: vector3 = (0, 0, 0)
  var Start Position: vector3 = (0, 0, 0)
  var Counters Parent: gameObject = _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Counters
  var Liquid: gameObject = <null>
  var Delivery Icon: gameObject = _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Delivery Icon
}
