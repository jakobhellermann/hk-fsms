# Msg Control
# 1 location(s):
#   prompts_assets_all.bundle / UI Msg Get Item Melody

fsm Msg Control {
  start Init

  state Detect {
    ListenForPromptContinue(EventTarget=Self, ContinuePressed=→"PRESS")
    on PRESS → Down
  }

  state Done {
    SendEventToRegister(eventName="GET ITEM MSG END")
    RecycleSelf()
  }

  state Init {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0)
    AnimatorPlay(gameObject=Self, stateName="Down", layer=0, normalizedTime=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Stop", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CONDUCTOR → Set Conductor
    on LIBRARIAN → Set Librarian
    on ARCHITECT → Set Architect
  }

  state Up {
    AnimatorPlayStateWait(target=Self, stateName="Up", finishEvent=→"FINISHED")
    ListenForAnimationEvent(Target=Self, Response=→"FINISHED")
    on FINISHED → Sound
  }

  state Stop Up {
    FadeNestedFadeGroup(Target=var "Stop", ToAlpha=0, FadeTime=0)
    ActivateGameObject(gameObject=var "Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Stop", ToAlpha=1, FadeTime=0.25)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Down {
    PlayAudioEvent(audioClip=button (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatorPlayStateWait(target=Self, stateName="Down", finishEvent=→"FINISHED")
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=2)
    on FINISHED → Done
  }

  state Set Conductor {
    GetLanguageString(sheetName="Quests", convName="SQ_MELODY_CONDUCTOR_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Quests", convName="SQ_MELODY_CONDUCTOR_PROMPT", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Desc", textString=var "Game Text")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=melody_prompts_0001_conductor (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetSpriteRendererSprite(gameObject=var "Icon Glow", sprite=melody_prompts_0001_conductor_glower (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    on FINISHED → Set Common
  }

  state Set Common {
    GetLanguageString(sheetName="Prompts", convName="GET_ITEM_INTRO1", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Up
  }

  state Set Librarian {
    GetLanguageString(sheetName="Quests", convName="SQ_MELODY_LIBRARIAN_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Quests", convName="SQ_MELODY_LIBRARIAN_PROMPT", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Desc", textString=var "Game Text")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=melody_prompts_0002_librarian (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetSpriteRendererSprite(gameObject=var "Icon Glow", sprite=melody_prompts_0002_librarian_glower (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    on FINISHED → Set Common
  }

  state Set Architect {
    GetLanguageString(sheetName="Quests", convName="SQ_MELODY_ARCHITECT_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Quests", convName="SQ_MELODY_ARCHITECT_PROMPT", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Desc", textString=var "Game Text")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=melody_prompts_0000_architect (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetSpriteRendererSprite(gameObject=var "Icon Glow", sprite=melody_prompts_0000_architect_glower (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    on FINISHED → Set Common
  }

  state Stop Playing {
    SendEventToRegister(eventName="GET ITEM MSG COVERED")
    on FINISHED → Wait
  }

  state Wait {
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Up
  }

  state Sound {
    PlayAudioEvent(audioClip=ui_trio_song_get, pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Stop Playing
  }

  var Game Text: string = ""
  var Icon: gameObject = UI Msg Get Item Melody/Contents/Icon
  var Icon Glow: gameObject = UI Msg Get Item Melody/Contents/Icon/Icon Glow
  var Item Name: gameObject = UI Msg Get Item Melody/Contents/Item Name
  var Item Name Prefix: gameObject = UI Msg Get Item Melody/Contents/Item Name Prefix
  var Stop: gameObject = UI Msg Get Item Melody/Contents/Stop
  var Item Desc: gameObject = UI Msg Get Item Melody/Contents/Item Desc
}
