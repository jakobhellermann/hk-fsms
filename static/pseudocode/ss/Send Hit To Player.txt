fsm Send Hit To Player {
  start Init

  state Init {
    on DASH HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObjectFSM(var "Hero", fsm="Sprint"), sendEvent="DASH HIT", delay=0, everyFrame=false)
    on FINISHED → Init
  }
}
