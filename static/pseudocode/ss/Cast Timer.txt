# Cast Timer
# 1 location(s):
#   Clover_19 / Boss Scene/Cloverstag White Boss

fsm Cast Timer {
  start Init

  state Can Cast {
    SetBoolValue(boolVariable=var "Can Cast", boolValue=true, everyFrame=false)
    on CAST → Check Phase
  }

  state Cast P1 {
    SetBoolValue(boolVariable=var "Can Cast", boolValue=false, everyFrame=false)
    Wait(time=3.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Can Cast
  }

  state Init {
    GetHP(target=Self, storeValue=var "P2 HP")
    MultiplyIntByFloat(integer=var "P2 HP", multiplyFloat=0.45, storeResult=var "P2 HP", everyFrame=false, forceRoundUp=false)
    on FINISHED → Can Cast
  }

  state Check Phase {
    SendEvent(eventTarget=Self, sendEvent=→"P2", delay=0, everyFrame=false)  // disabled
    CompareHP(enemy=var "Self", integer2=var "P2 HP", equal=→"P2", lessThan=→"P2", greaterThan=→"P1", everyFrame=false)
    on P1 → Cast P1
    on P2 → Cast P2
  }

  state Cast P2 {
    SetBoolValue(boolVariable=var "Can Cast", boolValue=false, everyFrame=false)
    Wait(time=4.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Can Cast
  }

  var P2 HP: int = 0
  var Can Cast: bool = false
  var Self: gameObject = Boss Scene/Cloverstag White Boss
}
