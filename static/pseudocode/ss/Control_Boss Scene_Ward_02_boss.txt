# Control
# 1 location(s):
#   Ward_02_boss / Boss Scene

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Start Range")
    FindNamedChild(gameObject=Self, storeResult=var "Floor Shuffler")
    FindNamedChild(gameObject=Self, storeResult=var "Husks")
    FindNamedChild(gameObject=Self, storeResult=var "Floor Quaker")
    FindNamedChild(gameObject=Self, storeResult=var "Headpiece Suck Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Conductor Headpiece")
    FindNamedChild(gameObject=Self, storeResult=var "Conductor Boss")
    FindNamedChild(gameObject=Self, storeResult=var "Suck Spears")
    FindNamedChild(gameObject=Self, storeResult=var "Ward Boss Spear Auto")
    FindChild(gameObject=var "Conductor Headpiece", childName="Sprite", storeResult=var "Headpiece Sprite")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Headpiece Antic")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slasher 1")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slammer 1")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slasher 2")
    FindNamedChild(gameObject=var "Husks", storeResult=var "Slammer 2")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop FloorRumble")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Arena Start
  }

  state Arena Start {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    SetDarknessLevel(SetLevel=2, StoreCurrentLevel=(unset))
    GetPlayerDataBool(boolName="wardBossEncountered", storeValue=var "Boss Encountered")
    WaitBool(boolTest=var "Boss Encountered", time=4, finishEvent=→"ENCOUNTERED", realTime=false)
    Wait(time=5, finishEvent=→"NEXT", realTime=false)
    on NEXT → Shuffle R
    on ENCOUNTERED → Encountered Start
  }

  state Boss Title {
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=true, bossTitle="WARD_BOSS")
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    ApplyMusicCue(musicCue=Creepy Boss Ambient (audiocuesdynamic_assets_areabellareashellwoodareaward.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=4)
    on FINISHED → First Slasher
  }

  state First Slasher {
    SendEventByName(eventTarget=GameObject(var "Slasher 1"), sendEvent="SPAWN NEAR HERO", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Floor Shuffler", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetCollider(gameObject=var "Floor Shuffler", active=false, resetOnExit=false)
    FadeAudio(gameObject=var "Floor Shuffler", startVolume=1, endVolume=0, time=0.5)
    on HUSK KILLED → P2 Pause
  }

  state Shuffle R {
    SetCollider(gameObject=var "Floor Shuffler", active=true, resetOnExit=false)
    SetVelocity2d(gameObject=var "Floor Shuffler", vector=(unset), x=6, y=(unset), everyFrame=false)
    Wait(time=2.75, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Boss Encountered", time=2, finishEvent=→"FINISHED", realTime=false)
    ApplyMusicCue(musicCue=Creepy Boss Ambient (audiocuesdynamic_assets_areabellareashellwoodareaward.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=4)
    AudioPlaySimpleV2(gameObject=var "Floor Shuffler", volume=0, startTime=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Floor Shuffler", startVolume=0, endVolume=1, time=0.5)
    on FINISHED → Boss Title
  }

  state P2 Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Boss Encountered", time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → P2 Setup
  }

  state P2 Setup {
    SetPosition(gameObject=var "Floor Shuffler", vector=(unset), x=70.22, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=var "Floor Shuffler", vector=(unset), x=-12, y=(unset), everyFrame=false)
    SetDarknessLevel(SetLevel=1, StoreCurrentLevel=(unset))
    Wait(time=3.75, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=var "Floor Shuffler", active=true, resetOnExit=false)
    WaitBool(boolTest=var "Boss Encountered", time=2.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimpleV2(gameObject=var "Floor Shuffler", volume=0, startTime=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Floor Shuffler", startVolume=0, endVolume=1, time=0.5)
    on FINISHED → P2 Start
  }

  state P2 Start {
    SendEventByName(eventTarget=GameObject(var "Slasher 1"), sendEvent="SPAWN", delay=0.5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Slammer 1"), sendEvent="SPAWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=var "Floor Shuffler", startVolume=1, endVolume=0, time=0.5)
    on FINISHED → P2
  }

  state P2 {
    SetVelocity2d(gameObject=var "Floor Shuffler", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetCollider(gameObject=var "Floor Shuffler", active=false, resetOnExit=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on HUSK KILLED → Remaining Count
    on FINISHED → Remaining Count
  }

  state P3 Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetDarknessLevel(SetLevel=1, StoreCurrentLevel=(unset))
    on FINISHED → P3 Shake
  }

  state Remaining Count {
    GetChildCount(gameObject=var "Husks", storeResult=var "Husks Remaining")
    IntCompare(integer1=var "Husks Remaining", integer2=4, equal=→"NEXT", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → P2
    on NEXT → P3 Pause
  }

  state P3 Shake {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetDarknessLevel(SetLevel=0, StoreCurrentLevel=(unset))
    SetCollider(gameObject=var "Floor Quaker", active=true, resetOnExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimpleV2(gameObject=var "Audio Loop FloorRumble", volume=0, startTime=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Loop FloorRumble", startVolume=0, endVolume=1, time=0.25)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    on FINISHED → Headpiece Antic
  }

  state Headpiece Antic {
    PlayParticleEmitterInState(gameObject=var "Pt Headpiece Antic")
    PlayAudioEvent(audioClip=unravelled_boss_bg_head_shake (sfxstatic_assets_areaward.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Conductor Headpiece", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ObjectJitter(gameObject=var "Headpiece Sprite", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=30)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=var "Conductor Headpiece", vector=(unset), x=(unset), y=-0.5, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Headpiece Pause
  }

  state Headpiece Pause {
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=var "Floor Quaker", active=false, resetOnExit=false)
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    FadeAudio(gameObject=var "Audio Loop FloorRumble", startVolume=1, endVolume=0, time=0.25)
    on FINISHED → Headpiece Suck
  }

  state Headpiece Suck {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=unravelled_boss_bg_head_dissapear (sfxstatic_assets_areaward.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Conductor Headpiece", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Headpiece Suck Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Translate(gameObject=var "Conductor Headpiece", vector=(unset), x=(unset), y=-12, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Boss Phase 1
  }

  state Boss Phase 1 {
    SendEventByName(eventTarget=GameObject(var "Conductor Boss"), sendEvent="WAKE", delay=0, everyFrame=false)
    on TO P2 → Twin Slammers
  }

  state Slammer Battle {
    SetVelocity2d(gameObject=var "Floor Shuffler", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetCollider(gameObject=var "Floor Shuffler", active=false, resetOnExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on HUSK KILLED → Remaining Count 2
    on FINISHED → Remaining Count 2
  }

  state Remaining Count 2 {
    GetChildCount(gameObject=var "Husks", storeResult=var "Husks Remaining")
    IntCompare(integer1=var "Husks Remaining", integer2=4, equal=→"NEXT", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Slammer Battle
    on NEXT → Spear Suck Pause
  }

  state Twin Slammers {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Slammer 1"), sendEvent="SPAWN", delay=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Slammer 2"), sendEvent="SPAWN", delay=2.5, everyFrame=false)
    on FINISHED → Slammer Battle
  }

  state Spear Suck Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → P3 Shake 2
  }

  state P3 Shake 2 {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetCollider(gameObject=var "Floor Quaker", active=true, resetOnExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=var "Audio Loop FloorRumble", startVolume=0, endVolume=1, time=0.25)
    on FINISHED → Suck Spears
  }

  state Suck Spears {
    SendEventByName(eventTarget=GameObject(var "Suck Spears"), sendEvent="SUCK", delay=0, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pos Suck Spears Pause
  }

  state Boss Phase 2 {
    SendEventByName(eventTarget=GameObject(var "Conductor Boss"), sendEvent="CONTINUE", delay=0, everyFrame=false)
    on TO P3 → Auto Spear Pause
  }

  state Slammer, Slasher {
    ActivateGameObject(gameObject=var "Ward Boss Spear Auto", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Slasher 1"), sendEvent="SPAWN L", delay=1, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Slammer 1"), sendEvent="SPAWN M", delay=2, everyFrame=false)
    on FINISHED → Enemy Phase
  }

  state Enemy Phase {
    SetVelocity2d(gameObject=var "Floor Shuffler", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetCollider(gameObject=var "Floor Shuffler", active=false, resetOnExit=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on HUSK KILLED → Remaining Count 3
    on FINISHED → Remaining Count 3
  }

  state Remaining Count 3 {
    GetChildCount(gameObject=var "Husks", storeResult=var "Husks Remaining")
    IntCompare(integer1=var "Husks Remaining", integer2=3, equal=→"NEXT", lessThan=(none), greaterThan=→"NEXT", everyFrame=false)
    on FINISHED → Enemy Phase
    on NEXT → Add Slammer
  }

  state Add Slammer {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Slammer 2"), sendEvent="SPAWN", delay=0, everyFrame=false)
    on FINISHED → Enemy Phase 2
  }

  state Enemy Phase 2 {
    SetVelocity2d(gameObject=var "Floor Shuffler", vector=(unset), x=0, y=(unset), everyFrame=false)
    SetCollider(gameObject=var "Floor Shuffler", active=false, resetOnExit=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on HUSK KILLED → Remaining Count 4
    on FINISHED → Remaining Count 4
  }

  state Remaining Count 4 {
    GetChildCount(gameObject=var "Husks", storeResult=var "Husks Remaining")
    IntCompare(integer1=var "Husks Remaining", integer2=4, equal=→"NEXT", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Enemy Phase 2
    on NEXT → Enemy Phase End Pause
  }

  state Enemy Phase End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFsmBool(gameObject=var "Ward Boss Spear Auto", fsmName="Control", variableName="Auto", setValue=false, everyFrame=false)  // disabled
    on FINISHED → Boss Phase 3
  }

  state Boss Phase 3 {
    SendEventByName(eventTarget=GameObject(var "Conductor Boss"), sendEvent="CONTINUE", delay=0, everyFrame=false)
    on DEATH STEAM → Death Steam
  }

  state Death Steam {
    SetCollider(gameObject=var "Floor Quaker", active=true, resetOnExit=true)
    on DEATH EXPLODE → Death Explode
  }

  state Death Explode {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Silk Heart
  }

  state Silk Heart {
    ActivateGameObject(gameObject=var "Silk Heart", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventToRegister(eventName="SILK HEART APPEAR")
    on HEART COLLECTED → Battle End
  }

  state Battle End {
    SetPlayerDataBool(boolName="wardBossDefeated", value=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open Gate
  }

  state Encountered Start {
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=true, bossTitle="WARD_BOSS")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ApplyMusicCue(musicCue=Creepy Boss Ambient (audiocuesdynamic_assets_areabellareashellwoodareaward.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=2)
    on FINISHED → First Slasher
  }

  state Open Gate {
    SendEventToRegister(eventName="WARD TRAPDOOR OPEN")
  }

  state Auto Spear Pause {
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Slammer, Slasher
  }

  state Pos Suck Spears Pause {
    Wait(time=1.5, finishEvent=(none), realTime=false)
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    SetCollider(gameObject=var "Floor Quaker", active=false, resetOnExit=false)
    FadeAudio(gameObject=var "Audio Loop FloorRumble", startVolume=1, endVolume=0, time=0.25)
    on FINISHED → Boss Phase 2
  }

  state Encountered Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=1)
    on FINISHED → Encountered Start
  }

  var Husks Remaining: int = 0
  var Boss Encountered: bool = false
  var Start Range: gameObject = <null>
  var Floor Shuffler: gameObject = <null>
  var Slasher 1: gameObject = <null>
  var Slammer 1: gameObject = <null>
  var Slasher 2: gameObject = <null>
  var Slammer 2: gameObject = <null>
  var Husks: gameObject = <null>
  var Floor Quaker: gameObject = <null>
  var Headpiece Suck Effects: gameObject = <null>
  var Conductor Headpiece: gameObject = <null>
  var Pt Headpiece Antic: gameObject = <null>
  var Headpiece Sprite: gameObject = <null>
  var Suck Spears: gameObject = <null>
  var Conductor Boss: gameObject = <null>
  var Ward Boss Spear Auto: gameObject = <null>
  var Silk Heart: gameObject = <null>
  var Audio Loop FloorRumble: gameObject = <null>
}
