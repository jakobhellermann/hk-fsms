# Blanker Control
# 4 location(s):
#   End_Credits / Cutscene Blanker
#   End_Game_Completion / Cutscene Blanker
#   Menu_Credits / Cutscene Blanker
#   PermaDeath / Cutscene Blanker

fsm Blanker Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetSpriteRenderer(gameObject=var "Self", active=true)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=(0, 0, 0, 1), everyFrame=false)
    SetFsmGameObject(gameObject=var "MainCamera", fsmName="CameraFade", variableName="Blanker", setValue=var "Self", everyFrame=false)  // disabled
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Out
  }

  state Fade In {
    SetColorValue(colorVariable=var "Colour", color=var "Start Colour", everyFrame=false)
    EaseColor(fromValue=var "Start Colour", toValue=var "End Colour", colorVariable=var "Colour", time=var "Fade In Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)  // disabled
  }

  state Fade Out {
    SetSpriteRenderer(gameObject=Self, active=true)
    SetColorValue(colorVariable=var "Colour", color=var "End Colour", everyFrame=false)
    EaseColor(fromValue=var "End Colour", toValue=var "Start Colour", colorVariable=var "Colour", time=var "Fade Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    on JUST FADE → Fade In
  }

  state Idle {
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  var Fade Time: float = 1
  var Fade In Time: float = 0.5

  var Cancel Reset: bool = false
  var Colour: color = (0, 0, 0, 1)
  var End Colour: color = (0, 0, 0, 1)
  var Start Colour: color = (0, 0, 0, 0)
  var Self: gameObject = <null>
}
