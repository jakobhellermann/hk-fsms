# Stun Cancel
# 1 location(s):
#   localpoolprefabs_assets_areabell.bundle / Spinner_vertical_slash_long_new

fsm Stun Cancel {
  start Pause
  on SPINNER DEFEATED → Recycle  // from any state

  state Pause {
    Wait(time=1.3, finishEvent=→"FINISHED", realTime=false)
    on SPINNER STUN → Recycle
    on FINISHED → Spawn Bell
  }

  state Recycle {
    RecycleSelf()
  }

  state Spawn Bell {
    FlingObjectsFromGlobalPool(gameObject=Spinner AtkBell, spawnPoint=var "Bell Point", position=(unset), spawnMin=1, spawnMax=1, speedMin=30, speedMax=42, angleMin=87, angleMax=93, originVariationX=0, originVariationY=0.25, FSM="Control", FSMEvent="FLING UP")
    on SPINNER STUN → Recycle
  }

  var Bell Point: gameObject = Spinner_vertical_slash_long_new/Bell Point
}
