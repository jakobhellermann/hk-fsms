# Bench Fall Control
# 1 location(s):
#   Wisp_04 / Fall Bench Group

fsm Bench Fall Control {
  start Wait For Persistent

  state Wait For Persistent {
    ActivateGameObject(gameObject=var "Fall Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Real Bench", ToAlpha=0, FadeTime=0)
    ActivateInteractible(Target=var "Real Bench", Activate=false, AllowQueueing=false, UseChildren=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Is Sack Broken?
  }

  state Is Sack Broken? {
    GetPersistentBool(target=Fall Bench Group/Breakable Hang Sack 3, storeValue=var "Is Broken")
    BoolTest(boolVariable=var "Is Broken", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Activate Real Bench
    on FALSE → Wait For Sack Break
  }

  state Activate Real Bench {
    FadeNestedFadeGroup(Target=var "Real Bench", ToAlpha=1, FadeTime=0)
    ActivateInteractible(Target=var "Real Bench", Activate=true, AllowQueueing=false, UseChildren=false)
  }

  state Wait For Sack Break {
    on SACK BREAK → Fake Bench Fall
  }

  state Fake Bench Fall {
    ActivateGameObject(gameObject=var "Fall Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlayStateWait(target=var "Fall Bench", stateName="Fall", finishEvent=→"FINISHED")
    ListenForAnimationEvent(Target=var "Fall Bench", Response=→"FINISHED")
    on FINISHED → Activate Real Bench
  }

  var Is Broken: bool = false
  var Fall Bench: gameObject = Fall Bench Group/Fall Bench
  var Real Bench: gameObject = Fall Bench Group/RestBench
}
