# Protect From Below
# 4 location(s):
#   Dust_05 / Roachkeeper
#   Dust_05 / Garmond Scene/Garmond Not Here/Roachkeeper - Garmond Target
#   Dust_05 / Roachkeeper (2)
#   Dust_06 / Roachkeeper Key Control/Roachkeeper

fsm Protect From Below {
  start Pause

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Immune
  }

  state Immune {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetImmuneToNailAttacks(true))
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"SWAP", belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=-5, reverseIfNegativeScale=false, everyFrame=true)
    on SWAP → Vulnerable
  }

  state Vulnerable {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetImmuneToNailAttacks(false))
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=→"SWAP", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=-5, reverseIfNegativeScale=false, everyFrame=true)
    on SWAP → Immune
  }
}
