# Detect Hit
# 3 location(s):
#   Bone_East_09 / Wind Rock (2)/explode rock
#   Bone_East_09 / Wind Rock (1)/explode rock
#   Bone_East_09 / Wind Rock Double/Wind Rock (3)/explode rock

fsm Detect Hit {
  start Idle

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"HIT", storeCollider=var "Collider")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=<null>, ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=0, storeDirection=0, storeMagnitudeMultiplier=0, firstHitOnly=false)
    on HIT → Check
  }

  state Check {
    GetPosition(gameObject=var "Collider", vector=(unset), x=var "Atk Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Atk Pos", lowerValue=var "Atk Min X", upperValue=var "Atk Max X", boolVariable=(unset), trueEvent=→"VALID", falseEvent=(none), everyFrame=false)
    GetPosition(gameObject=var "Collider", vector=(unset), x=(unset), y=var "Atk Pos", z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Atk Pos", lowerValue=var "Atk Min Y", upperValue=var "Atk Max Y", boolVariable=(unset), trueEvent=→"VALID", falseEvent=(none), everyFrame=false)
    on FINISHED → Idle
    on VALID → Send
  }

  state Send {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="STRIKE", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  var Atk Max X: float = 9999
  var Atk Max Y: float = 9999
  var Atk Min X: float = 22
  var Atk Min Y: float = 0

  var Atk Pos: float = 0
  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
}
