fsm Avoid player entering from room {
  start State 3

  state State 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="door1", equalEvent=→"END", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on END → State 2
  }

  state State 2 {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=500, z=(unset), space=0, everyFrame=false, lateUpdate=false)
  }

  state State 3 {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → State 1
  }

  var Entry Gate: string = ""
}
