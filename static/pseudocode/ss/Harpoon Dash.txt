// uses template: hornet_harpoon_dash
fsm Harpoon Dash {
  start Init
  on HERO DAMAGED → Cancel All  // from any state
  on LEAVING SCENE → Cancel All  // from any state
  on FSM CANCEL → Cancel All  // from any state
  on SPRINT CANCEL → Cancel All  // from any state
  on BOUNCE CANCEL → Cancel All  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Effects", storeResult=var "Effects Folder")
    FindChild(gameObject=var "Effects Folder", childName="Hornet_harpoon_dash", storeResult=var "Dash Effect")
    FindChild(gameObject=var "Effects Folder", childName="Hornet_harpoon_throw_effect", storeResult=var "Throw Effect")
    FindChild(gameObject=var "Effects Folder", childName="Hornet_harpoon_grab_effect", storeResult=var "Grab Effect")

    FindChild(gameObject=Self, childName="Attacks/Harpoon Damager", storeResult=var "Harpoon Damager")
    FindChild(gameObject=Self, childName="Attacks/Harpoon Dash Damager", storeResult=var "Dash Damager")

    FindNamedChild(gameObject=Self, storeResult=var "Centre")
    on FINISHED → Idle
  }

  state Cancel All {

    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=1, functionCall=SetHarpooning(false))

    ActivateGameObject(gameObject=var "Dash Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

    on FINISHED → Unset Nail Charge
    on DO MOVE → Take Control
  }

  state Regain Control {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RegainControl())

  }

    SetFloatValue(floatVariable=var "Return Idle Wait", floatValue=0, everyFrame=false)
    on DO MOVE → Can Do?
    on DO MOVE FIXED → Set Fixed
  }

    on FINISHED → Flip?
  }
