# Cower

// uses template: courier_npc_common
fsm Cower {
  start Init

  state Dormant {
    ActivateGameObject(gameObject=var "Rest Sack", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Start Cowering", isTrue=→"NPC COWER START", isFalse=(none), everyFrame=false)
    on NPC COWER START → Cower
  }

  state Cower {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Cower", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Rest Sack", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetAudioClip(gameObject=var "Audio Loop Voice", audioClip=Courier_scared_loop (sfxstatic_assets_areadustareamoss.bundle), autoPlay=true, stopOnExit=false)
    SetAudioVolume(gameObject=var "Audio Loop Voice", volume=0.5, everyFrame=false)
    on BATTLE END → To Idle
    on COURIER SAVED → To Idle
  }

  state To Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest to Idle", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    CallMethodProper(gameObject=var "Emerge Antic Effect", behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    PlayAudioEvent(audioClip=relic_dealer_scavanging_loop (sfxstatic_assets_areabell.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=var "Startle Voice Table", pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Burst
  }

  state Idle {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=1.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT → Assemble Cell Keys
    on SING → Sing Type
    on STARTLE → Startle
  }

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    FindChild(gameObject=Self, childName="Emerge Antic Effect", storeResult=var "Emerge Antic Effect")
    FindChild(gameObject=Self, childName="Emerge Burst Effect", storeResult=var "Emerge Burst Effect")
    FindChild(gameObject=Self, childName="Rest Sack", storeResult=var "Rest Sack")
    FindChild(gameObject=Self, childName="Fly Away Dust", storeResult=var "Fly Away Dust")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Fly Extra")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    on FINISHED → Dormant
  }

  state Burst {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CallMethodProper(gameObject=var "Emerge Antic Effect", behaviour="PlayParticleEffects", methodName="StopParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Emerge Burst Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rest Sack", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=garmond_jump (sfxstatic_assets_garmond.bundle), pitchMin=1, pitchMax=1, volume=0.7, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioStop(gameObject=var "Audio Player", fadeTime=0)
    on FINISHED → Idle
  }

  state Thank pt1 {
    SendEventToRegister(eventName="TRAILS END")
    RunDialogue(Sheet="Wanderers", Key=var "Cell Thank 1", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Wish Granted
  }

  sequence state Wish Granted {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=Self, UseChildren=false)
    EndQuest(ConsumeCurrency=false, Quest=var "Save Quest")
    on FINISHED → Thank pt2
  }

  state Thank pt2 {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle=var "Name")
    RunDialogue(Sheet="Wanderers", Key=var "Cell Thank 2", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
  }

  state End {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    SendEventToRegister(eventName="TRAILS OFF")
    SendEventToRegister(eventName="TRAIL OBJECTS DEACTIVATE")
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → State 1
  }

  state Fly Away {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocityByScale(gameObject=Self, speed=6, ySpeed=7, everyFrame=false)
    AccelerateVelocity(gameObject=Self, xAccel=(unset), yAccel=1, xMaxSpeed=(unset), yMaxSpeed=40)
    SetTransformParent(gameObject=var "Fly Away Dust", parent=<null>, worldPositionStays=true)
    ActivateGameObject(gameObject=var "Fly Away Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Audio Loop Fly Extra", volume=1, oneShotClip=<null>)
    on FINISHED → Deactivate
  }

  state Assemble Cell Keys {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle=var "Name")
    BuildString(stringParts=[var "Name", "THANK_PT1"], separator="_", addToEnd=false, storeResult=var "Cell Thank 1", everyFrame=false)
    BuildString(stringParts=[var "Name", "THANK_PT2"], separator="_", addToEnd=false, storeResult=var "Cell Thank 2", everyFrame=false)
    on FINISHED → Thank pt1
  }

  state Fly Antic {
    ConvertBoolToFloat(boolVariable=var "Fly Away Right", floatVariable=var "Fly Away Scale", falseValue=-1, trueValue=1, everyFrame=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    SetScale(gameObject=Self, vector=(unset), x=var "Fly Away Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fly Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=relic_dealer_lean_back_to_normal (sfxstatic_assets_areabell.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=var "Flyaway Voice Table", pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Fly Away
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state State 1 {
    BoolTest(boolVariable=var "Fly Away Right", isTrue=(none), isFalse=→"TURN_LEFT", everyFrame=false)
    LookAnimNpcForceFacing(ForceFacing=true, FaceLeft=false, WaitUntilFinished=true, FinishedTurning=(none), Target=Self)
    on TURN_LEFT → Turn Left
    on FINISHED → Fly Antic
  }

  state Turn Left {
    LookAnimNpcForceFacing(ForceFacing=true, FaceLeft=true, WaitUntilFinished=true, FinishedTurning=(none), Target=Self)
    on FINISHED → Fly Antic
  }

  state Sing Short {
    PlayRandomAudioClipTable(Table=garmond_movement (sfxdynamic_assets_garmond.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=false, Target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Courier_short_sing (sfxdynamic_assets_areabell.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    on FINISHED → Idle
  }

  state Startle {
    PlayRandomAudioClipTable(Table=garmond_movement (sfxdynamic_assets_garmond.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=false, Target=Self)

    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=var "Startle Voice Table", pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on CANCEL → Sing End
  }

  state Sing Type {
    StringCompare(stringVariable=var "Name", compareTo="COURIER_SHORT", equalEvent=→"SHORT", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on SHORT → Sing Short
    on FINISHED → Sing Tall
  }

  state Sing Tall {
    PlayRandomAudioClipTable(Table=garmond_movement (sfxdynamic_assets_garmond.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))

    ActivateLookAnimNPC(Activate=false, IsFacingLeft=false, Target=Self)

  }

  var Fly Away Right: bool = true
  var Name: string = "COURIER_SHORT"
  var Save Quest: object = Save Courier Short (dataassets_assets_assets/dataassets/questsystem/quests.bundle)
  var Startle Voice Table: object = Courier_short_alert (sfxdynamic_assets_areabell.bundle)
  var Flyaway Voice Table: object = Courier_short_take_off (sfxdynamic_assets_areamoss.bundle)

  var Fly Away Scale: float = 0
  var Cell Thank 1: string = ""
  var Cell Thank 2: string = ""
  var Audio Loop Fly Extra: gameObject = <null>
  var Emerge Antic Effect: gameObject = <null>
  var Emerge Burst Effect: gameObject = <null>
  var Fly Away Dust: gameObject = <null>
