# Control
# 1 location(s):
#   Peak_07 / Pinstress Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindNamedChild(gameObject=Self, storeResult=var "Pinstress Scene")
    FindNamedChild(gameObject=Self, storeResult=var "No Pinstress Scene")
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=(none), isFalse=→"NO PINSTRESS")
    CheckQuestState(NotTrackedEvent=→"NO PINSTRESS", TrackedEvent=(none), CompletedEvent=(none), Quest=Pinstress Battle (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    GetQuestState(IsAccepted=(unset), IsCompleted=var "Quest Completed", Quest=Pinstress Battle (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    GetPlayerDataBool(boolName="visitedIceCore", storeValue=var "Visited Ice Core")
    BoolAllTrue(boolVariables=[var "Quest Completed", var "Visited Ice Core"], sendEvent=→"NO PINSTRESS", storeResult=(unset), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"PINSTRESS", delay=0, everyFrame=false)
    on PINSTRESS → Pinstress
    on NO PINSTRESS → No Pinstress
  }

  state Pinstress {
    ActivateGameObject(gameObject=var "Pinstress Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "No Pinstress Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state No Pinstress {
    ActivateGameObject(gameObject=var "Pinstress Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "No Pinstress Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Quest Completed: bool = false
  var Visited Ice Core: bool = false
  var Pinstress Scene: gameObject = <null>
  var No Pinstress Scene: gameObject = <null>
}
