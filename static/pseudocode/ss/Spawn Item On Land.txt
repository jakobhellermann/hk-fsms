# Spawn Item On Land
# 1 location(s):
#   Ant_20 / Enemy Break Cage (2)/Corpse/Corpse Pilgrim Fly

fsm Spawn Item On Land {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Collectable Item Pickup")
    FindNamedChild(gameObject=Self, storeResult=var "Pickup Dummy")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    CheckYPosition(gameObject=Self, compareTo=11.3, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAND → Land
  }

  state Land {
    SetPositionToObject2D(gameObject=var "Collectable Item Pickup", targetObject=var "Self", xOffset=0, yOffset=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pickup Dummy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Parent: gameObject = <null>
  var Collectable Item Pickup: gameObject = <null>
  var Self: gameObject = <null>
  var Pickup Dummy: gameObject = <null>
}
