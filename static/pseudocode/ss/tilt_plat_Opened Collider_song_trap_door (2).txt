# tilt_plat
# 1 location(s):
#   Slab_03 / song_trap_door (2)/Opened Collider

// uses template: tink_plat
fsm tilt_plat {
  start Init
  on READY → Idle  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink Up
  }

  state Tink L {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(-0.1, 0, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink R {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(0.1, 0, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink Up {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(0, 0.1, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink Return {
    AnimatePositionTo(gameObject=var "Art", toValue=(0, 0, 0), localSpace=true, time=0.1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  var Art: gameObject = song_trap_door (2)/Door/Tilt Plat/Art

  var Plat Dust: gameObject = <null>
}
