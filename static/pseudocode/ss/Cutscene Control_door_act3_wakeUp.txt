# Cutscene Control
# 1 location(s):
#   Song_Enclave / door_act3_wakeUp

fsm Cutscene Control {
  start Init

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=true)
    on FINISH ENTRY → Take Control
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Prostrate")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    ActivateInteractible(Target=var "Sherma Caretaker", Activate=false, AllowQueueing=false, UseChildren=false)
    TransitionToAudioSnapshot(snapshot=at None (audioobjects_assets_all.bundle), transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Master Silence (audioobjects_assets_all.bundle), transitionTime=0)
    TransitionToAudioSnapshot(snapshot=at None (audioobjects_assets_all.bundle), transitionTime=0.1)
    TransitionToAudioSnapshot(snapshot=Off (audioobjects_assets_all.bundle), transitionTime=0.1)
    ActivateWorldRumble(SetActive=false)
    on FINISHED → Fade Pause
  }

  state Fade Pause {
    Wait(time=1, finishEvent=(none), realTime=false)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    on FINISHED → Sherma Start
  }

  state Fade In {
    TransitionToAudioSnapshot(snapshot=at Layer 1 (audioobjects_assets_all.bundle), transitionTime=8)
    TransitionToAudioSnapshot(snapshot=On (audioobjects_assets_all.bundle), transitionTime=8)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=6)
    CameraBlurPlaneFade(From={Spacing=30, Vibrancy=0, MaskLerp=0}, To={Spacing=0, Vibrancy=0, MaskLerp=0}, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], Duration=5.5)
    Wait(time=8, finishEvent=(none), realTime=false)
    on FINISHED → Get Up
  }

  state Sherma Start {
    SendEventByName(eventTarget=GameObject(var "Sherma Caretaker"), sendEvent="CHIME START", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Master On (audioobjects_assets_all.bundle), transitionTime=0.3)
    TransitionToAudioSnapshot(snapshot=On (audioobjects_assets_all.bundle), transitionTime=15)
    Wait(time=8, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=at Layer 1 (audioobjects_assets_all.bundle), transitionTime=16)
    on FINISHED → Fade In
  }

  sequence state Get Up {
    SendEventByName(eventTarget=GameObject(var "Sherma Caretaker"), sendEvent="CHIME STOP", delay=0, everyFrame=false)
    PlayRandomAudioClipTable(Table=Grunt Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0))
    PlayAudioEvent(audioClip=phantom_short_antic_pre_jump (sfxstatic_assets_areaorgan.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="ProstrateRiseToKneel")
    Wait(time=2, finishEvent=(none), realTime=false)
    PlayAudioEvent(audioClip=trobbio_jump_short (sfxstatic_assets_trobbio.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Talk Pause
  }

  state Talk Pause {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dialogue A
  }

  state Dialogue A {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CARETAKER_MEET_A", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Sherma Caretaker")
    on CONVO_END → Dialogue B
  }

  state Dialogue End {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=var "Sherma Caretaker", UseChildren=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  sequence state End {
    SetDeathRespawnMarker(Target=var "Respawn Marker")
    ActivateWorldRumble(SetActive=true)
    SetPlayerDataBool(boolName="act3_enclaveWakeSceneCompleted", value=true)
    SetPlayerDataBool(boolName="act3MapUpdated", value=true)
    BeginQuestV2(ShowPrompt=true, Quest=Black Thread Pt1 Shamans (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    SaveGame()
    DoGameMapUpdate(silent=true, delay=0, didUpdateEvent=(none))
    Wait(time=1.5, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=Enclave (audiocuesdynamic_assets_areasong.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Sub Area (audioobjects_assets_all.bundle), transitionTime=0.5)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    ActivateInteractible(Target=var "Sherma Caretaker", Activate=true, AllowQueueing=false, UseChildren=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    CallMethodProper(gameObject=var "AreaTitle", behaviour="AreaTitleController", methodName="ForcePlay", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  state Dialogue B {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CARETAKER_MEET_B", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Sherma Caretaker")
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="SHERMA")
    on CONVO_END → Dialogue End
  }

  var Sherma Caretaker: gameObject = Black Thread States/Black Thread World/Enclave Act 3/Sherma Caretaker
  var Respawn Marker: gameObject = Black Thread States/Black Thread World/Enclave Act 3/Town/RestBench

  var Self: gameObject = <null>
}
