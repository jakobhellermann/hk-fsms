# Control
# 1 location(s):
#   localpoolprefabs_assets_areaclover.bundle / Cloverstag White Sickle

fsm Control {
  start Pause
  on MULTI HIT CONNECT → Break  // from any state

  state Init {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Offset X", y=(unset), space=0, everyFrame=false)
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Offset X", float2=var "Hero X", operation=Subtract, storeResult=var "Offset X", everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    WaitRandom(timeMin=0, timeMax=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Summon Up
    on BOSS DEFEATED → Recycle
  }

  state Fly Pause {
    SetMeshRenderer(gameObject=Self, active=true)
    RandomFloat(min=0.8, max=1.2, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=0.4, oneShotClip=<null>)
    PlayRandomAudioClipTable(Table=white_cloverstag_projectile_burst (sfxdynamic_assets_areaclover.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sickle")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    WaitRandom(timeMin=0.4, timeMax=0.8, finishEvent=→"FINISHED", realTime=false)
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)  // disabled
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Extra Delay?
    on BOSS DEFEATED → Break
  }

  state Recycle {
    ActivateGameObject(gameObject=var "Flash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Delay", boolValue=false, everyFrame=false)
    RecycleSelf()
  }

  state Summon Up {
    PlayRandomAudioClipTable(Table=white_cloverstag_projectile_rise (sfxdynamic_assets_areaclover.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    TranslateRandom(gameObject=Self, translateMin=(-2.5, -1.5, 0), translateMax=(2.5, 1.5, 0), space=0)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Summon", emit=0, resetIfPlaying=false)
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Hero X", float2=var "Offset X", operation=Add, storeResult=var "Self X", everyFrame=true)
    SetPosition2D(GameObject=Self, Vector=(unset), X=var "Self X", Y=(unset), Space=0, EveryFrame=true)
    on FINISHED → Fly Pause
    on BOSS DEFEATED → Break
  }

  state Recycle Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    AudioStopV2(gameObject=Self, fadeTime=0.5, cancelOnEarlyExit=false)
    on FINISHED → Recycle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetFloatValue(floatVariable=var "X Force", floatValue=0, everyFrame=false)
    on FINISHED → Init
  }

  state Move {
    SetVelocityByScale(gameObject=Self, speed=8, ySpeed=(unset), everyFrame=false)
    AccelerateToXByScale(gameObject=Self, accelerationFactor=0.375, targetSpeed=-35)
    ChaseObjectVertical(gameObject=Self, target=var "Hero", offset=0.5, speedMax=5, acceleration=0.02)
    TiltBySpeedY(gameObject=Self, tiltFactor=-3, tiltMax=45, rotationSpeed=50)
    PlayParticleEmitterInState(gameObject=var "Pt Idle")
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    CheckYPosition(gameObject=Self, compareTo=8.543973, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"RECYCLE", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → Recycle Pause
    on RECYCLE → Recycle Pause
    on PROJECTILE BREAK → Break
    on BOSS DEFEATED → Break
  }

  state Break {
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Flash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayRandomAudioClipTable(Table=white_cloverstag_projectile_burst (sfxdynamic_assets_areaclover.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Recycle
  }

  state Extra Delay? {
    BoolTest(boolVariable=var "Delay", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move
    on BOSS DEFEATED → Break
  }

  var Wait Time: float = 1
  var Accel: float = 0
  var X Force: float = 0
  var Offset X: float = 0
  var Hero X: float = 0
  var Self X: float = 0
  var Pitch: float = 0
  var Delay: bool = false
  var Pt Summon: gameObject = Cloverstag White Sickle/Pt Summon
  var Pt Idle: gameObject = Cloverstag White Sickle/Pt Idle
  var Flash: gameObject = Cloverstag White Sickle/Flash
}
