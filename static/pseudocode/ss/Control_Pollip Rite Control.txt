# Control
# 1 location(s):
#   Room_Witch / Pollip Rite Control

fsm Control {
  start Init

  state Init {
    on POLLIP RITE START → Init SFX
  }

  state Rite Start {
    SendMessage(gameObject=var "Jitter Parts", delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.2)
    PlayParticleEmitter(gameObject=var "Pt PollipRite", emit=0, resetIfPlaying=false)
    PlayParticleEmitterInState(gameObject=var "Pt PollipRite")
    WaitRandom(timeMin=0.75, timeMax=1, finishEvent=→"SFX", realTime=false)
    on POLLIP RITE END → End
    on SFX → Sfx
  }

  state End {
    SendMessage(gameObject=var "Jitter Parts", delivery=BroadcastMessage, options=1, functionCall=StopJitter())
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitter(gameObject=var "Pt Burst", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=enemy_death_sword (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=greyroot_hornet_grab_pt_6_final_break (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=enemy_death_sword_old (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=0.75, audioPlayerPrefab=var "", spawnPoint=var "Audio Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
  }

  state Init SFX {
    PlayAudioEvent(audioClip=greyroot_hornet_grab_pt_2_grab (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Rite Start
  }

  state Sfx {
    PlayAudioEventRandom(audioClips=[greyroot_hornet_grab_pt_2_grab (sfxstatic_assets_areashellwood.bundle), greyroot_hornet_grab_pt_4_break (sfxstatic_assets_areashellwood.bundle)], pitchMin=1.1, pitchMax=1.3, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[enemy_death_sword (sfxstatic_assets_shared.bundle), flying_stick_insect_corpse_burst (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Audio Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Rite Start
  }

  var Jitter Parts: gameObject = Pollip Rite Control/Jitter Parts
  var Pt PollipRite: gameObject = Pollip Rite Control/Pt PollipRite
  var Pt Burst: gameObject = Pollip Rite Control/Pt Burst
  var Audio Origin: gameObject = Pollip Rite Control/Audio Origin
}
