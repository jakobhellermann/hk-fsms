# Control
# 1 location(s):
#   localpoolprefabs_assets_areapeak.bundle / Crystal Drifter Spikes

fsm Control {
  start Init

  state Init {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetMeshRenderer(gameObject=var "Sprite", active=true)
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spike Damager Small", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spike Damager Main", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Spike Form", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Damager Small
  }

  state Active {
    OnDamagedHeroEvent(Target=var "Spike Damager Main", SendEvent=→"BREAK")
    ActivateGameObject(gameObject=var "Spike Damager Main", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=var "Spawn Terrain Detector", trigger=1, collideTag="", sendEvent=→"BREAK", storeCollider=var "")
    Tk2dWatchAnimationEvents(gameObject=var "Sprite", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on BREAK → Break
    on FINISHED → Wait
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)  // disabled
    PlayAudioEvent(audioClip=crystal_drifter_spikes_dissolve (sfxstatic_assets_areapeak.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Spike Damager Main", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Slash Breaker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Wait {
    OnDamagedHeroEvent(Target=var "Spike Damager Main", SendEvent=→"BREAK")
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.4, finishEvent=→"SPIKES JITTER", realTime=false)
    on SPIKES JITTER → Shatter Antic
    on BREAK → Break
  }

  state Shatter Antic {
    OnDamagedHeroEvent(Target=var "Spike Damager Main", SendEvent=→"BREAK")
    ObjectJitterLocal(gameObject=Self, x=0.05, y=0.05, z=(unset), resetPosOnStateExit=true, limitFps=0)
    Wait(time=0.2, finishEvent=→"BREAK", realTime=false)
    on SPIKES BREAK → Break
    on BREAK → Break
  }

  state Damager Small {
    OnDamagedHeroEvent(Target=var "Spike Damager Small", SendEvent=→"BREAK")
    ActivateGameObject(gameObject=var "Spike Damager Small", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=var "Sprite", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Slash Breaker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Slash Breaker", active=true, resetOnExit=false)
    on FINISHED → Active
    on BREAK → Wait
  }

  var Pt Shatter: gameObject = Crystal Drifter Spikes/Pt Shatter
  var Self: gameObject = <null>
  var Sprite: gameObject = Crystal Drifter Spikes/Sprite
  var Spawn Terrain Detector: gameObject = Crystal Drifter Spikes/Spawn Terrain Detector
  var Spike Damager Small: gameObject = Crystal Drifter Spikes/Spikes Damager Small
  var Spike Damager Main: gameObject = Crystal Drifter Spikes/Spikes Damager Main
  var Slash Breaker: gameObject = Crystal Drifter Spikes/Slash Breaker
}
