# Check Corner
# 4 location(s):
#   Clover_10 / Boss Scene/Dancer Control/Dancer A
#   Clover_10 / Boss Scene/Dancer Control/Dancer B
#   Cog_Dancers_boss / Dancer Control/Dancer B
#   Cog_Dancers_boss / Dancer Control/Dancer A

fsm Check Corner {
  start Idle

  state Idle {
    on CHECK CORNER → Check
  }

  state Check {
    SetBoolValue(boolVariable=var "In Corner", boolValue=false, everyFrame=false)
    GetFsmString(gameObject=Self, fsmName="Control", variableName="Current Pos", storeValue=var "Current Pos", everyFrame=false)
    StringCompare(stringVariable=var "Current Pos", compareTo="Pos1", equalEvent=→"IN CORNER", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Current Pos", compareTo="Pos5", equalEvent=→"IN CORNER", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Current Pos", compareTo="Pos8", equalEvent=→"IN CORNER", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Current Pos", compareTo="Pos12", equalEvent=→"IN CORNER", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Idle
    on IN CORNER → In Corner
  }

  state In Corner {
    SetBoolValue(boolVariable=var "In Corner", boolValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  var In Corner: bool = false
  var Current Pos: string = ""
}
