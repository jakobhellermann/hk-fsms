# Control
# 1 location(s):
#   localpoolprefabs_assets_areashellwood.bundle / Shellwood Gnat

fsm Control {
  start Init
  on FLING OUT → Start Fling Out  // from any state
  on GO UP → Go Up  // from any state

  state Idle {
    DecelerateV2(gameObject=Self, deceleration=0.8, brakeOnExit=true)
    WaitRandom(timeMin=0.7, timeMax=0.8, finishEvent=→"FINISHED", realTime=false)
    DistanceFly(gameObject=Self, target=var "Hero", distance=4.5, speedMax=2.5, acceleration=0.25, targetsHeight=true, height=0.5, minAboveHero=(unset))
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Tiny TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    on FINISHED → Antic
  }

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Fling Out
  }

  state Fling Out {
    SetVelocity2d(gameObject=Self, vector=var "Fling Vector", x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Decelerate
  }

  state Decelerate {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    on FINISHED → Fling Out End
  }

  state Fling Out End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tiny Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CallMethodProper(gameObject=Self, behaviour="ObjectBounce", methodName="StartBounce", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetDamageHero(Target=Self, Enabled=true)
    on FINISHED → Idle
  }

  state Start Fling Out {
    SetTag(gameObject=Self, tag="Untagged")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tiny Roll", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CallMethodProper(gameObject=Self, behaviour="ObjectBounce", methodName="StopBounce", parameters=[], storeResult=(unset var), EveryFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetDamageHero(Target=Self, Enabled=false)
    on FINISHED → Dir
  }

  state Dir {
    SetVector2Value(vector2Variable=var "Fling Vector", vector2Value=(-15, 22), everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"FINISHED", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    SetVector2Value(vector2Variable=var "Fling Vector", vector2Value=(15, 22), everyFrame=false)
    on FINISHED → Fling Out
  }

  state Antic {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=shellwood_gnat_attack (sfxdynamic_assets_areashellwood.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayAudioEvent(audioClip=shellwood_goomba_little_green_mode_notice_hero (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Fly Angle", pause=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tiny Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    on FINISHED → Chase v2
  }

  state Go Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=6, everyFrame=false)
    on FINISHED → Chase v2
  }

  state Chase v2 {
    ChaseObjectGround(gameObject=Self, target=var "Hero", xOffset=0, speedMax=6, acceleration=0.15, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0, snapTo=false, snapSpeedOnly=false, onlyOnStateEntry=false)
    ChaseObjectVertical(gameObject=Self, target=var "Hero", offset=1.3, speedMax=9, acceleration=0.2)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tiny Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Tiny TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=var "In Alert Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    SetFloatValueV2(floatVariable=var "Unalert Timer", floatValue=0, everyFrame=true, activeBool=var "In Alert Range")
    SetBoolValue(boolVariable=var "Not In Alert Range", boolValue=var "In Alert Range", everyFrame=false)
    BoolFlip(boolVariable=var "Not In Alert Range")
    FloatAddV2(floatVariable=var "Unalert Timer", add=1, everyFrame=true, perSecond=true, fixedUpdate=false, activeBool=var "Not In Alert Range")
    FloatCompare(float1=var "Unalert Timer", float2=5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on UNALERT → Unalert
    on SING → Sing
  }

  state Unalert {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Tiny Fly")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="Tiny TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=1.5, speedMax=2, accelerationMin=10, accelerationMax=15, roamingRangeX=3, roamingRangeY=3, manualStartPos=(unset))
    ShoveFromWall(gameObject=Self, shoveForce=10, rayLength=1, checkUp=true, checkDown=true, checkLeft=true, checkRight=true, everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"ALERT", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ALERT → Antic
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Tiny Sing")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.5, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=shellwood_gnat_sing (sfxdynamic_assets_areashellwood.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on END → Antic
    on SING DURATION END → Antic
  }

  var Fly Angle: float = 0
  var Unalert Timer: float = 0
  var In Alert Range: bool = false
  var Not In Alert Range: bool = false
  var Fling Vector: vector2 = (0, 0)
  var Start Pos: vector3 = (0, 0, 0)
  var Audio Loop Voice: gameObject = Shellwood Gnat/Audio Loop Voice
}
