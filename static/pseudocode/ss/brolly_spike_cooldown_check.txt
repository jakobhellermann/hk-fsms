#   globalsettings_assets_all.bundle / Tool_brolly_spike_dj

// uses template: brolly_spike_cooldown_check
fsm brolly_spike_cooldown_check {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    SetTag(gameObject=Self, tag="Brolly Spike Slash")
    on FINISHED → Check
  }

  state Check {
    SetTag(gameObject=Self, tag="Untagged")
    FindGameObject(objectName="", withTag="Brolly Spike Slash", store=var "Slasher")
    GameObjectIsNull(gameObject=var "Slasher", isNull=(none), isNotNull=→"DEACTIVATE", storeResult=(unset), everyFrame=false)
    SetTag(gameObject=Self, tag="Brolly Spike Slash")
    PlayAudioEvent(audioClip=sawtooth_circlet_slash (sfxstatic_assets_shared.bundle), pitchMin=0.925, pitchMax=1.075, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on DEACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Slasher: gameObject = <null>
  var Self: gameObject = <null>
}
