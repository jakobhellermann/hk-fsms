# Repair Sprite
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/RestBench Control/RestBench

fsm Repair Sprite {
  start Init

  state Init {
    EnableBehaviour(gameObject=Self, behaviour="StateTransitionAnimator", component=<null>, enable=false, resetOnExit=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="skullKingBenchMended", isTrue=→"MEND", isFalse=(none))
    on MEND → Set Mended
  }

  state Set Mended {
    SetSpriteRendererSprite(gameObject=Self, sprite=bone_bench_basic_repaired (atlases_assets_assets/sprites/_atlases/hbone.spriteatlas.bundle))
    FindNamedChild(gameObject=Self, storeResult=var "Lit")
    SetSpriteRendererSprite(gameObject=var "Lit", sprite=bone_bench_basic_repaired (atlases_assets_assets/sprites/_atlases/hbone.spriteatlas.bundle))
    EnableBehaviour(gameObject=Self, behaviour="StateTransitionAnimator", component=<null>, enable=true, resetOnExit=false)
    SetPropertyV2(TargetProperty=CustomSceneManager.extraRestZone on _SceneManager@CustomSceneManager, EveryFrame=false, ResetOnExit=false)
    on FINISHED → Idle
  }

  state Idle {
    on AT THIS BENCH → Check Play Sound
  }

  state At Bench {
    SetBoolValue(boolVariable=var "Respawn At Bench", boolValue=false, everyFrame=false)
    on OFF THIS BENCH → Get Up
  }

  state Get Up {
    PlayRandomAudioClipTable(Table=wood_creak (sfxdynamic_assets_areashellwood.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Idle
  }

  state Check Play Sound {
    BoolTest(boolVariable=var "Respawn At Bench", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    PlayRandomAudioClipTable(Table=wood_creak (sfxdynamic_assets_areashellwood.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → At Bench
    on CANCEL → Clear Respawn
  }

  state Clear Respawn {
    SetBoolValue(boolVariable=var "Respawn At Bench", boolValue=false, everyFrame=false)
    on FINISHED → At Bench
  }

  var Respawn At Bench: bool = false
  var Lit: gameObject = <null>
}
