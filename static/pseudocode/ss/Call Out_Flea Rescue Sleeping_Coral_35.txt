# Call Out
# 6 location(s):
#   Coral_35 / Flea Rescue Sleeping
#   Dock_03d / Flea Rescue Sleeping
#   Dock_16 / Flea Rescue Sleeping
#   Greymoor_06 / Flea Rescue Sleeping
#   Library_09 / Flea Rescue Sleeping
#   Song_11 / Flea Rescue Sleeping

fsm Call Out {
  start Init
  on RESCUE → Rescue 1  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Scene")
    on FINISHED → Start Sleep
  }

  state Rescue 1 {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Flea_bark (sfxdynamic_assets_fleacaravan.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayAudioEvent(audioClip=coral_flyer_in_air_dash_1 (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SavedItemGetV2(Item=var "Quest Target", Amount=1, ShowPopup=false)
    PlayParticleEmitter(gameObject=var "Dust", emit=0, resetIfPlaying=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="SmallShake", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RescueToFly")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, 1.8, 0), time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -1, 0), time=0.5, speed=(unset), delay=0.25, easeType=14, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.5, finishEvent=→"NEXT", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on NEXT → Panting
  }

  state Rescue 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.25, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=Flea_Howl_02 (sfxstatic_assets_fleacaravan.bundle))
    on FINISHED → Leave Antic
  }

  state Leave Antic {
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -0.25, 0), time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Flea_bark (sfxdynamic_assets_fleacaravan.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStopV2(gameObject=Self, fadeTime=0.25, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=flee_fly_past_screen (sfxstatic_assets_pilgrimslowerpilgrimsupper.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Leave Dir
  }

  state Leave {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=var "Y Pos", z=var "Z Pos", space=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-10, everyFrame=false)
    FloatOperator(float1=var "X Pos", float2=var "X Scale", operation=Add, storeResult=var "X Target", everyFrame=false)
    FloatOperator(float1=var "Y Pos", float2=20, operation=Add, storeResult=var "Y Target", everyFrame=false)
    FloatOperator(float1=var "Z Pos", float2=-30, operation=Add, storeResult=var "Z Target", everyFrame=false)
    EaseFloat(fromValue=var "X Pos", toValue=var "X Target", floatVariable=var "New X", time=0.4, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "Y Pos", toValue=var "Y Target", floatVariable=var "New Y", time=0.4, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "Z Pos", toValue=var "Z Target", floatVariable=var "New Z", time=0.4, speed=(unset), delay=(unset), easeType=4, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "New X", y=var "New Y", z=var "New Z", space=0, everyFrame=true, lateUpdate=false)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(0, 0, 0, 1), colorVariable=var "Colour", time=0.05, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Scene"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SavedItemGetV2(Item=var "Quest Target", Amount=1, ShowPopup=true)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetMeshRenderer(gameObject=Self, active=true)
  }

  state Sleeping {
    CheckQuestPdSceneBool(QuestTarget=var "Quest Target", ExpectedValue=true, trueEvent=→"ACTIVATED", falseEvent=(none), storeValue=(unset))
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"TOUCH", storeCollider=var "")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=(none), ActiveOuter=→"NOISE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.75, MaxReactDelay=0.75, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    ReceivedNailDamage(Target=Self, collideTag="", sendEvent=→"NAIL", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Nail Attack", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"NOISE", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on NOISE → Range Check
    on TOUCH → Rescue 1
    on ACTIVATED → Activated
    on NAIL → Check LOS
  }

  state Activated {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Range Check {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Rescue Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Sleeping
    on WAKE → Rescue 1
  }

  state Leave Dir {
    BoolTest(boolVariable=var "Swing Left", isTrue=→"SWING L", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Swing Right", isTrue=→"SWING R", isFalse=(none), everyFrame=false)
    on SWING R → Swing R
    on FINISHED → Leave
    on SWING L → Swing L
  }

  state Swing R {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=var "Y Pos", z=var "Z Pos", space=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-10, everyFrame=false)
    FloatOperator(float1=var "X Pos", float2=var "X Scale", operation=Add, storeResult=var "X Target", everyFrame=false)
    FloatOperator(float1=var "Y Pos", float2=20, operation=Add, storeResult=var "Y Target", everyFrame=false)
    FloatOperator(float1=var "Z Pos", float2=-30, operation=Add, storeResult=var "Z Target", everyFrame=false)
    EaseFloat(fromValue=var "X Pos", toValue=var "X Target", floatVariable=var "New X", time=0.4, speed=(unset), delay=(unset), easeType=4, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "Y Pos", toValue=var "Y Target", floatVariable=var "New Y", time=0.4, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "Z Pos", toValue=var "Z Target", floatVariable=var "New Z", time=0.4, speed=(unset), delay=0.1, easeType=4, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "New X", y=var "New Y", z=var "New Z", space=0, everyFrame=true, lateUpdate=false)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(0, 0, 0, 1), colorVariable=var "Colour", time=0.05, speed=(unset), delay=0.1, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Scene"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Sing Hold {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    on SING → Sing
    on FINISHED → Leave Antic
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.75, MaxReactDelay=0.75, None=→"FINISHED", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Leave Antic
  }

  state Swing L {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=var "Y Pos", z=var "Z Pos", space=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-10, everyFrame=false)
    FloatOperator(float1=var "X Pos", float2=var "X Scale", operation=Add, storeResult=var "X Target", everyFrame=false)
    FloatOperator(float1=var "Y Pos", float2=20, operation=Add, storeResult=var "Y Target", everyFrame=false)
    FloatOperator(float1=var "Z Pos", float2=-30, operation=Add, storeResult=var "Z Target", everyFrame=false)
    EaseFloat(fromValue=var "X Pos", toValue=var "X Target", floatVariable=var "New X", time=0.4, speed=(unset), delay=(unset), easeType=4, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "Y Pos", toValue=var "Y Target", floatVariable=var "New Y", time=0.4, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=var "Z Pos", toValue=var "Z Target", floatVariable=var "New Z", time=0.4, speed=(unset), delay=0.1, easeType=4, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "New X", y=var "New Y", z=var "New Z", space=0, everyFrame=true, lateUpdate=false)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(0, 0, 0, 1), colorVariable=var "Colour", time=0.05, speed=(unset), delay=0.1, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=Self, color=var "Colour", everyframe=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Scene"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Start Sleep {
    AudioPlaySimple(gameObject=var "Audio Loop Voice", volume=1, oneShotClip=<null>)
    on FINISHED → Sleeping
  }

  state Panting {
    SetAudioClip(gameObject=var "Audio Loop Voice", audioClip=Flea_pant_loop_02_slower (sfxstatic_assets_fleacaravan.bundle), autoPlay=true, stopOnExit=true)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Rescue 2
  }

  state Check LOS {
    CheckLineOfSight(From=Self, To=var "Nail Attack", blockingLayerMask=256, storeResult=(unset), inSightEvent=→"TOUCH", noSightEvent=(none), everyFrame=false)
    on FINISHED → Sleeping
    on TOUCH → Rescue 1
  }

  var Swing Right: bool = false
  var Swing Left: bool = false

  var New X: float = 0
  var New Y: float = 0
  var New Z: float = 0
  var X Pos: float = 0
  var X Scale: float = 0
  var X Target: float = 0
  var Y Pos: float = 0
  var Y Target: float = 0
  var Z Pos: float = 0
  var Z Target: float = 0
  var Colour: color = (1, 1, 1, 1)
  var Audio Loop Voice: gameObject = Flea Rescue Sleeping/Audio Loop Voice
  var Dust: gameObject = Flea Rescue Sleeping/Dust
  var Nail Attack: gameObject = <null>
  var Scene: gameObject = <null>
  var Self: gameObject = <null>
  var Quest Target: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
