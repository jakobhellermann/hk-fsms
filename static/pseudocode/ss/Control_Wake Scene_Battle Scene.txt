# Control
# 1 location(s):
#   Cog_10_Destroyed / Battle Scene/Wake Scene

fsm Control {
  start Pause
  on BATTLE COMPLETED → Battle Completed  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Abyss_Quake_Instant")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "CamLocks")
    FindNamedChild(gameObject=Self, storeResult=var "Wake Threads")
    FindNamedChild(gameObject=Self, storeResult=var "Pt WakeSlam")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Gates")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "CamLock Pre")
    FindNamedChild(gameObject=var "Gates", storeResult=var "Roof Blocker")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Transition to Silent
    on BATTLE COMPLETED → Battle Completed
  }

  sequence state Threads Up {
    Wait(time=2, finishEvent=(none), realTime=false)
    SendEventByName(eventTarget=GameObject(var "Wake Threads"), sendEvent="IMPACT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Abyss_Quake_Instant", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=3, finishEvent=(none), realTime=false)
    SendEventToRegister(eventName="ABYSS ROAR START")  // disabled
    SendEventByName(eventTarget=GameObject(var "Wake Threads"), sendEvent="FLAIL", delay=0, everyFrame=false)  // disabled
    StartRoarEmitter(spawnPoint=var "Roar Point", delay=0, stunHero=false, roarBurst=false, isSmall=false, noVisualEffect=false, forceThroughBind=false, stopOnExit=true)  // disabled
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)  // disabled
    Wait(time=2, finishEvent=(none), realTime=false)  // disabled
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))  // disabled
    SendEventToRegister(eventName="ABYSS ROAR STOP")  // disabled
    on FINISHED → StartBattle
  }

  state StartBattle {
    CallMethodProper(gameObject=var "Battle Scene", behaviour="BattleScene", methodName="StartBattle", parameters=[], storeResult=(unset var), EveryFrame=false)
    on BATTLE END → End Flail
  }

  state Transition to Silent {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=2)
    SendEventToRegister(eventName="TENSION END")
    ActivateGameObject(gameObject=var "CamLocks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Roof Blocker"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    on FINISHED → Threads Up
  }

  state End Flail {
    SendEventByName(eventTarget=GameObject(var "Wake Threads"), sendEvent="BATTLE END", delay=0, everyFrame=false)
  }

  state Battle Completed {
    ActivateGameObject(gameObject=var "Gates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Pre", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Battle Scene: gameObject = <null>
  var CamLocks: gameObject = <null>
  var Gates: gameObject = <null>
  var Pt WakeSlam: gameObject = <null>
  var Roar Point: gameObject = Battle Scene/Wake Scene/Roar Point
  var Wake Threads: gameObject = <null>
  var Roof Blocker: gameObject = <null>
  var Abyss_Quake_Instant: gameObject = <null>
  var CamLock Pre: gameObject = <null>
}
