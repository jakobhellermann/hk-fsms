# BG Control
# 1 location(s):
#   Shellwood_01b / Battle Scene/Gates/Battle Gate Shellwood (2)

fsm BG Control {
  start Opened
  on BG QUICK OPEN → Quick Open  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close
    on BG QUICK CLOSE → Quick Close
  }

  state Close {
    PlayAudioEvent(audioClip=splinter_queen_spike_appear_2 (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=0.6, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0.15, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Close", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shellwood Gate Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Closed
  }

  state Open {
    PlayAudioEvent(audioClip=splinter_queen_move_left_right_1 (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shellwood Gate Open")
    PlayParticleEmitter(gameObject=var "Pt Open", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on BG CLOSE → Close
    on BG QUICK CLOSE → Quick Close
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shellwood Gate Closed")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Closed
  }

  state Quick Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → 
    on BG QUICK CLOSE → Quick Close
  }

  state Closed {
    on BG OPEN → Open
    on TINK → Tink
  }

  state Tink {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shellwood Gate Hit")
    PlayParticleEmitter(gameObject=var "Pt Hit", emit=0, resetIfPlaying=false)
    on FINISHED → Closed
  }

  var Start Closed: bool = false

  var Self: gameObject = Battle Scene/Gates/Battle Gate Shellwood (2)
  var Slam Effect: gameObject = <null>
  var Pt Close: gameObject = Battle Scene/Gates/Battle Gate Shellwood (2)/Pt Close
  var Pt Open: gameObject = Battle Scene/Gates/Battle Gate Shellwood (2)/Pt Open
  var Pt Hit: gameObject = Battle Scene/Gates/Battle Gate Shellwood (2)/Pt Hit
}
