# Repair webshot
# 1 location(s):
#   Peak_12 / Webshot Scene

fsm Repair webshot {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Fake Hornet", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inert Seat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Has broken SilkShot?
  }

  state Has broken SilkShot? {
    CollectableItemGetData(CollectedAmount=(unset), NeededAmount=1, IsCollected=→"FINISHED", NotCollected=→"CANCEL", ItemName="", Sprite=var "", Item=Broken SilkShot (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle))
    on CANCEL → No Silkshot
    on FINISHED → Has Silkshot
  }

  state No Silkshot {
    ActivateGameObject(gameObject=Webshot Scene/Has Webshot, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateInteractible(Target=var "Interact Controller", Activate=false, AllowQueueing=false, UseChildren=false)
  }

  state Has Silkshot {
    on INTERACT → Inspect Dlg
  }

  state End Dlg 2 {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=var "Interact Controller", UseChildren=false)
    on FINISHED → Has Silkshot
  }

  sequence state Repair Prompt {
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=var "Interact Controller", UseChildren=false)
    DialogueYesNoItemV3(Prompt={Sheet="Inspect", Key="WEB_SHOT_WORKSHOP_PROMPT"}, RequiredItem=Tool Metal (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle), RequiredAmount=1, ShowCounter=false, ConsumeItem=true, WillGetItem=var "", YesEvent=→"FINISHED", NoEvent=→"CANCEL", ReturnHUDAfter=false)
    on CANCEL → End Dlg 2
    on FINISHED → Look Up End
  }

  state Sit Down {
    SetMeshRenderer(gameObject=var "Hero", active=false)
    ActivateGameObject(gameObject=var "Fake Hornet", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inert Seat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 1), duration=1)
    Wait(time=1, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWait(Target=var "Fake Hornet", ClipName="Hornet Desk Sit", AnimationCompleteEvent=(none))
    PlayAudioEvent(audioClip=trobbio_recover_or_general_foley_movement (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Sound
  }

  sequence state Sound {
    PlayAudioEvent(audioClip=workstation_construct_tool (sfxstatic_assets_areaboneareahangareapeakareawisp.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=5, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=Webshot Scene/Has Webshot, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=1)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Stand Up
  }

  state Stand Up {
    Tk2dPlayAnimationWait(Target=var "Fake Hornet", ClipName="Hornet Desk Stand", AnimationCompleteEvent=(none))
    on FINISHED → Anim To Idle
  }

  state Anim To Idle {
    SetMeshRenderer(gameObject=var "Hero", active=true)
    ActivateGameObject(gameObject=var "Fake Hornet", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inert Seat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroAnimationController", methodName="SetPlayRunToIdle", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Get Item
  }

  sequence state End Dlg 3 {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=var "Interact Controller", UseChildren=false)
    on FINISHED → No Silkshot
  }

  state Inspect Dlg {
    RunDialogue(Sheet="Inspect", Key="WEB_SHOT_WORKSHOP_MSG", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=Top, OffsetY=0, Target=var "Interact Controller")
    on CONVO_END → Repair Prompt
  }

  sequence state Look Up End {
    ActivateInteractible(Target=var "Interact Controller", Activate=false, AllowQueueing=false, UseChildren=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Sit Down
  }

  state Get Item {
    CollectableItemTake(Amount=1, ShowCounter=false, Item=Broken SilkShot (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle))
    SavedItemGet(Item=WebShot Weaver (dataassets_assets_assets/dataassets/tools/toolitems.bundle))
    on FINISHED → End Dlg 3
  }

  var Interact Controller: gameObject = Webshot Scene/Inspect Region
  var Fake Hornet: gameObject = Webshot Scene/hornet
  var Inert Seat: gameObject = Webshot Scene/seat
}
