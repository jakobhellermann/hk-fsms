# Control
# 1 location(s):
#   Shellwood_25 / Witch Chapel Door Control

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Door Close")
    FindNamedChild(gameObject=Self, storeResult=var "Door Closed")
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Check
  }

  state Check {
    GetEntryGateName(StoreValue=var "Entry gate Name")
    StringCompare(stringVariable=var "Entry gate Name", compareTo="door1", equalEvent=→"DOOR", notEqualEvent=→"FINISHED", storeResult=(unset), everyFrame=false)
    on DOOR → Do Close
    on FINISHED → Closed
  }

  sequence state Do Close {
    Wait(time=1, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Door Close", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Closed {
    ActivateGameObject(gameObject=var "Door Closed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Entry gate Name: string = ""
  var Door Close: gameObject = <null>
  var Door Closed: gameObject = <null>
}
