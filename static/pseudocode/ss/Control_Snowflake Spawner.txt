# Control
# 1 location(s):
#   Peak_02 / Snowflake Scene (2)/Snowflake Spawner

// uses template: snowflake_spawn
fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Idle")
    on FINISHED → Inactive
  }

  state Pause {
    Wait(time=var "Rate", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Extra Pause
  }

  state Spawn {
    GetRandomChild(gameObject=Self, storeResult=var "Flake")
    SendEventByName(eventTarget=GameObject(var "Flake"), sendEvent="SPAWN", delay=0, everyFrame=false)
    on FINISHED → Pause
  }

  state Extra Pause {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Inactive {
    BoolTest(boolVariable=var "Start Inactive", isTrue=(none), isFalse=→"START SNOWFLAKE", everyFrame=false)
    on START SNOWFLAKE → Start Pause
  }

  state Start Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    PlayParticleEmitterChildren(gameObject=var "Pt Idle", resetTimeIfPlaying=false, stopOnStateExit=false)
    on FINISHED → Spawn
  }

  var Rate: float = 3
  var Start Inactive: bool = false

  var Flake: gameObject = <null>
  var Pt Idle: gameObject = <null>
}
