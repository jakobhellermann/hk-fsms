# Confirm Control
# 1 location(s):
#   shopui_assets_all.bundle / Shop Menu/Item Max Group

fsm Confirm Control {
  start Init

  state Init {
    on FINISHED → Set Text
  }

  state Listen {
    ListenForMenuActionsContinuous(EventTarget=Self, SubmitPressed=→"UI SELECTION MADE", CancelPressed=→"UI SELECTION MADE", NonePressed=(none), IgnoreAttack=false)
    on UI SELECTION MADE → No
  }

  state No {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=var "Fade Time")
    Wait(time=var "Fade Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset
  }

  state Reset {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="RESET SHOP WINDOW", delay=0, everyFrame=false)
    on FINISHED → Inert
  }

  state Set Text {
    on FINISHED → Inert
  }

  state Inert {
    on START INPUT → Listen
  }

  var Fade Time: float = 0.1
}
