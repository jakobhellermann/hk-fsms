# Control

fsm Control {
  start Init
  on POLLYWOG GRAB → Grab Pause  // from any state

    GetParent(gameObject=Self, storeResult=var "Pollywog Control")

  }

    on ENTER → In Range
  }

    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on EXIT → Idle
    on FINISHED → Get Pollywog
  }

  state Get Pollywog {
    GetRandomChild(gameObject=var "Dustroach Pollywog Pool", storeResult=var "Pollywog")
    GameObjectIsNull(gameObject=var "Pollywog", isNull=→"CANCEL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    on CANCEL → In Range
    on FINISHED → Check Spawn Pos
  }

    RandomFloat(min=26, max=33, storeResult=var "Velocity Y")
    SetVector2XY(vector2Variable=var "Velocity", vector2Value=(unset), x=var "Velocity X", y=var "Velocity Y", everyFrame=false)
    SetFsmVector2(gameObject=var "Pollywog", fsmName="Control", variableName="Launch Velocity", setValue=var "Velocity", everyFrame=false)
    SetFsmGameObject(gameObject=var "Pollywog", fsmName="Control", variableName="Home Node", setValue=var "Spawn Y", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Pollywog"), sendEvent="SPAWN", delay=0, everyFrame=false)
    on FINISHED → Get Pollywog L
  }

  state Grab Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → In Range
  }

  state Check Spawn Pos {
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Spawn Pos X", y=(unset), space=0, everyFrame=false)
    CheckXPosition(gameObject=var "Hero", compareTo=var "Spawn Edge L", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"R", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))

    SendRandomEvent(events=[→"L", →"R"], weights=[0.5, 0.5], delay=0)  // disabled
    FloatAddRandom(floatVariable=var "Spawn Pos X", addMin=-10, addMax=12)
    on L → Spawn L
    on R → Spawn R
    on FINISHED → Spawn
  }

  state Spawn L {
    FloatAddRandom(floatVariable=var "Spawn Pos X", addMin=-10, addMax=-6)
    on FINISHED → Spawn
  }

  state Spawn R {
    FloatAddRandom(floatVariable=var "Spawn Pos X", addMin=6, addMax=12)
    on FINISHED → Spawn
  }

  state Spawn Pause {
    Wait(time=var "Spawn Pause", finishEvent=→"FINISHED", realTime=false)

  }
