# Control
# 2 location(s):
#   Coral_10 / Citadel Ambient Music Control
#   Song_19_entrance / Citadel Ambient Music Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="citadelWoken", isTrue=→"AWOKEN", isFalse=→"UNAWOKEN")
    on AWOKEN → Awoken
    on UNAWOKEN → Unawoken
  }

  state Awoken {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=10)
  }

  state Unawoken {
    ApplyMusicCue(musicCue=Ambience Citadel Surrounds (audiocuesdynamic_assets_areacoralarealibraryareasong.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=6)
  }
}
