# Tornado Event Sender
# 1 location(s):
#   localpoolprefabs_assets_trobbio.bundle / Trobbio Tornado

fsm Tornado Event Sender {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"TRIGGER", storeCollider=var "Collider")
    on TRIGGER → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="TORNADO", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Collider: gameObject = <null>
}
