# BG Control
# 3 location(s):
#   Slab_16b / Broodmother Scene Control/Broodmother Scene/Battle Scene Broodmother/Gates/Battle Gate Slab (2)
#   Slab_23 / Battle Scene/Gates/Battle Gate Slab (3)
#   Slab_23 / Battle Scene/Gates/Battle Gate Slab (1)

fsm BG Control {
  start Pause
  on BG QUICK OPEN → Quick Open  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close Pause
    on BG QUICK CLOSE → Close 2
  }

  state Close 1 {
    PlayAudioEventRandomBool(audioClips=[gate_rise_slab_23 (sfxstatic_assets_areaslab.bundle)], doPlay=var "Z Play Audio", pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gate Close", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Close 2
  }

  state Close 2 {
    PlayParticleEmitter(gameObject=var "Pt Slam", emit=0, resetIfPlaying=false)  // disabled
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BG OPEN → Open Pause
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=gate_open (sharedeffects_assets_all.bundle), pitchMin=0.85, pitchMax=1.15, volume=0.85, delay=0, storePlayer=<null>)  // disabled
    PlayAudioEventRandomBool(audioClips=[gate_rise_slab_23 (sfxstatic_assets_areaslab.bundle)], doPlay=var "Z Play Audio", pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Gate Open")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on FINISHED → Opened Effects
  }

  state Close Pause {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    WaitRandom(timeMin=0, timeMax=0.15, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Close 1
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Wall Collider")
    on FINISHED → Opened
  }

  state Opened Effects {
    PlayParticleEmitter(gameObject=var "Pt Open", emit=0, resetIfPlaying=false)  // disabled
    PlayParticleEmitter(gameObject=var "Pt Rocks", emit=0, resetIfPlaying=false)  // disabled
    on FINISHED → Opened
  }

  state Quick Open {
    SetBoolValue(boolVariable=var "Start Closed", boolValue=false, everyFrame=false)
    on FINISHED → Opened
  }

  state Open Pause {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open
  }

  var Start Closed: bool = false
  var Z Play Audio: bool = true

  var Pt Open: gameObject = <null>
  var Pt Rocks: gameObject = <null>
  var Pt Slam: gameObject = <null>
  var Self: gameObject = <null>
  var Wall Collider: gameObject = <null>
}
