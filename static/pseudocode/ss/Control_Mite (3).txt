# Control
# 1 location(s):
#   Greymoor_16 / Mite (3)

fsm Control {
  start Init
  on TRAP → Trap Ready  // from any state
  on ZERO HP → Death  // from any state
  on INSTA KILL → Insta Kill  // from any state
  on FALL → Fall  // from any state

  state Init {
    BoolTest(boolVariable=var "Eating", isTrue=→"EATING", isFalse=(none), everyFrame=false)
    on FINISHED → Run
    on EATING → Wall Stick
  }

  state Eating {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Eating")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="WakeAttack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=(none), ActiveInner=(none), ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    AudioPlayInState(gameObject=var "Audio Loop Munching", volume=1)
    on WAKE → Detach
    on TOOK DAMAGE → Detach
    on ATTACK → Detach 2
  }

  state Run {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    WalkLeftRight(gameObject=Self, walkSpeed=7, spriteFacesLeft=true, groundLayer="Terrain", turnDelay=0.5, wallLookAhead=0, walkAnimName="Run", turnAnimName="Turn", moveWhileTurning=true, startLeft=false, startRight=false, keepDirection=true, flipBeforeTurn=false, turnAudioClipTable=<null>)
    SetAudioClip(gameObject=Self, audioClip=mite_walk_loop (sfxstatic_assets_areagreymoor.bundle), autoPlay=true, stopOnExit=true)
    WaitRandom(timeMin=0.5, timeMax=2, finishEvent=→"IDLE", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    SetRecoilSpeed(target=Self, newRecoilSpeed=15)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=var "Ray Ground Distance", SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on IDLE → Idle
    on ATTACK → Attack Pause
    on SING → Sing
  }

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SelectRandomString(strings=["Idle 1", "Idle 2", "Idle 3"], weights=[1, 1, 1], storeString=var "Anim")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"RUN")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=var "Ray Ground Distance", SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on RUN → Turn?
    on ATTACK → Attack Antic
    on SING → Sing
  }

  state Turn? {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    SendRandomEvent(events=[→"FINISHED", (none)], weights=[0.75, 0.25], delay=0)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=var "Ray Ground Distance", SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → Run
    on ATTACK → Attack Antic
    on SING → Sing
  }

  state Attack Antic {
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[mite_scream (sfxstatic_assets_areagreymoor.bundle)], weights=[1], pitchMin=0.9, pitchMax=1.1, stopPreviousSound=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SelectRandomString(strings=["Attack Antic 1", "Attack Antic 2"], weights=[1, 1], storeString=var "Anim")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=8)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=var "Ray Ground Distance", SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → Wait For Ground
  }

  state Leap {
    PlayAudioEvent(audioClip=mite_jump (sfxstatic_assets_areagreymoor.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Leap")
    RandomFloat(min=12, max=19, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Jump Y", everyFrame=false)
    SetVelocityByScale(gameObject=Self, speed=-15, ySpeed=(unset), everyFrame=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Leap Air
  }

  state Wall? {
    BoolTest(boolVariable=var "On Wall", isTrue=→"ATTACK", isFalse=(none), everyFrame=false)
    on FINISHED → Run
    on ATTACK → Wall Attack Antic
  }

  state Leap Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Land {
    PlayAudioEventRandom(audioClips=[mite_land_1 (sfxstatic_assets_areagreymoor.bundle), mite_land_2 (sfxstatic_assets_areagreymoor.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=var "Ray Ground Distance", SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → Run Away
  }

  state Attack Pause {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=→"FINISHED", realTime=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → Attack Antic
  }

  state Run Away {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetAudioClip(gameObject=Self, audioClip=mite_walk_loop (sfxstatic_assets_areagreymoor.bundle), autoPlay=true, stopOnExit=false)
    WalkLeftRight(gameObject=Self, walkSpeed=7, spriteFacesLeft=true, groundLayer="Terrain", turnDelay=0.5, wallLookAhead=0, walkAnimName="Run", turnAnimName="Turn", moveWhileTurning=true, startLeft=false, startRight=false, keepDirection=true, flipBeforeTurn=false, turnAudioClipTable=<null>)
    WaitRandom(timeMin=0.15, timeMax=0.35, finishEvent=→"FINISHED", realTime=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=var "Ray Ground Distance", SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → Run
  }

  state Wall Attack Antic {
    SelectRandomString(strings=["Attack Antic 1", "Attack Antic 2"], weights=[1, 1], storeString=var "Anim")
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[mite_scream (sfxstatic_assets_areagreymoor.bundle)], weights=[1], pitchMin=0.9, pitchMax=1.1, stopPreviousSound=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wall Leap
  }

  state Wall Leap {
    FaceObjectV2(objectA=Self, objectB=var "Bottom Marker", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    PlayAudioEvent(audioClip=mite_jump (sfxstatic_assets_areagreymoor.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Leap")
    RandomFloat(min=12, max=13, storeResult=var "Jump Y")
    SetVelocityByScale(gameObject=Self, speed=-15, ySpeed=var "Jump Y", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetRecoilSpeed(target=Self, newRecoilSpeed=15)
    on FINISHED → Leap Air
  }

  state Sing {
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=mite_sing (sfxdynamic_assets_areagreymoor.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on CANCEL → Idle
    on SING DURATION END → Idle
  }

  state Trap Ready {
    PlayAudioEvent(audioClip=mite_heavy_antic_call (sfxstatic_assets_areaunderstore.bundle), pitchMin=1.2, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCorpseSpawnPoint(target=Self, spawnPoint=(0, -100, 0))
    FindNamedChild(gameObject=Self, storeResult=var "Corpse Mite(Clone)")
    SetPosition(gameObject=var "Corpse Mite(Clone)", vector=(unset), x=(unset), y=-100, z=(unset), space=0, everyFrame=false, lateUpdate=false)
  }

  state Wall Stick {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=0, tolerance=0.1, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetRecoilSpeed(target=Self, newRecoilSpeed=0)
    SetBoolValue(boolVariable=var "On Wall", boolValue=true, everyFrame=false)
    on FINISHED → Eating
  }

  state Death {
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=8)
  }

  state Detach {
    BoolTest(boolVariable=var "Starts Attached to Art", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Starts Attached to Art", boolValue=false, everyFrame=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Wall?
  }

  state Detach 2 {
    BoolTest(boolVariable=var "Starts Attached to Art", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Starts Attached to Art", boolValue=false, everyFrame=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Attack Pause
  }

  state Insta Kill {
    CheckOutOfCamera(gameObject=Self, margin=1, outsideEvent=→"DISAPPEAR", insideEvent=(none), insideBool=false, outsideBool=false, everyFrame=false)
    SetShardDrop(target=Self, shards=0)
    InstaDeath(target=Self, direction=90)
    on DISAPPEAR → Disappear
  }

  state Fall {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    DecelerateXY(gameObject=Self, decelerationX=0.925, decelerationY=(unset), brakeOnExit=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    on LAND → Land
  }

  state Wait For Ground {
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=var "Ray Ground Distance", SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on LAND → Leap
  }

  state Disappear {
    DestroySelf(detachChildren=false)
  }

  var Eating: bool = true
  var Starts Attached to Art: bool = false

  var Jump Y: float = 0
  var Ray Ground Distance: float = 2.5
  var Rotation: float = 0
  var Scale: float = 0
  var On Wall: bool = false
  var Anim: string = ""
  var Corpse Mite(Clone): gameObject = <null>
  var Bottom Marker: gameObject = Mite (3)/Bottom Marker
  var Audio Loop Munching: gameObject = Mite (3)/Audio Loop Munching
  var Audio Loop Voice: gameObject = Mite (3)/Audio Loop Voice
}
