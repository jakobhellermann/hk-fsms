# Control
# 1 location(s):
#   localpoolprefabs_assets_trobbio.bundle / Trobbio Tornado

fsm Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Tornado Event Sender", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tornado Projectile", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetScale(gameObject=Self, vector=(unset), x=(unset), y=0.7, z=(unset), everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Fly
  }

  state Fly {
    GetScale(gameObject=Self, vector=(unset), xScale=var "Start X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Start X Scale", float2=1.25, operation=Multiply, storeResult=var "End X Scale", everyFrame=false)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=true)
    EaseFloat(fromValue=var "Start X Scale", toValue=var "End X Scale", floatVariable=var "X Scale", time=0.75, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=0.7, toValue=1.7, floatVariable=var "Scale", time=0.75, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=var "Scale", z=(unset), everyFrame=true, lateUpdate=false)
    on FINISHED → End
    on END → End
  }

  state End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tornado Projectile End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopParticleEmittersInChildren(gameObject=Self)
    ActivateGameObject(gameObject=var "Tornado Event Sender", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Pt Pause
  }

  state Recycle {
    RecycleSelf()
  }

  state Pt Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  var Scale: float = 0
  var Start X Scale: float = 0
  var End X Scale: float = 0
  var X Scale: float = 0
  var Tornado Disperse: gameObject = Trobbio Tornado/Tornado Disperse
  var Tornado Event Sender: gameObject = Trobbio Tornado/Tornado Event Sender
}
