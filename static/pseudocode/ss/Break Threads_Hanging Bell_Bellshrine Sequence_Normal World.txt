# Break Threads
# 1 location(s):
#   Bellshrine_02 / Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell

fsm Break Threads {
  start Init
  on THREAD BELL HIT → Hit  // from any state
  on SET BELL RELEASED → Set Bell Released  // from any state

  state Inert {
    SetGameObject(variable=var "Damage Source", gameObject=<null>, everyFrame=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damage Source", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT → Hit
    on HEAVY → Heavy Hit
  }

  state Choose State {
    BoolTest(boolVariable=var "Heavy Hit", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    IntCompare(integer1=var "State", integer2=var "Finish State", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent=var "State", delay=0, everyFrame=false)
    on 1 → Disable Threads
    on 2 → Disable Threads 2
    on 3 → Disable Threads 3
    on FINISHED → Snap Start
  }

  state Disable Threads {
    CallMethodProper(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 1, behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 1, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → Hit Fx
  }

  state Disable Threads 2 {
    CallMethodProper(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 2, behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 2, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → Hit Fx
  }

  state Disable Threads 3 {
    CallMethodProper(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 3, behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 3, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → Hit Fx
  }

  state Release Bell {
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/Threaded Bell, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/Threaded Bell, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/Physics Bell, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Pt Hit 1", storeResult=var "Pt Hit 1")
    FindChild(gameObject=Self, childName="Pt Hit 2", storeResult=var "Pt Hit 2")
    FindChild(gameObject=Self, childName="Pt Hit 3", storeResult=var "Pt Hit 3")
    on FINISHED → Inert
  }

  state Hit {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="FreezeMoment", parameters=[1], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddSilk", parameters=[1, true], storeResult=(unset var), EveryFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Silk_Get_Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Slash Impact")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SetScale(gameObject=var "Slash Impact", vector=(unset), x=1.3, y=1.3, z=(unset), everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Slash Impact", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    IntAdd(intVariable=var "State", add=1, everyFrame=false)
    on FINISHED → Choose State
  }

  state Save State {
    CallMethodProper(gameObject=Black Thread States/Normal World/Bellshrine Sequence, behaviour="StateChangeSequence", methodName="SetState", parameters=[1], storeResult=(unset var), EveryFrame=false)
    FindChild(gameObject=Self, childName="Physics Bell", storeResult=var "Physics Bell")
    ActivateGameObject(gameObject=var "Physics Bell", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=var "Physics Bell", childName="Large Bell", storeResult=var "Physics Bell")
    WakeUp2d(gameObject=var "Physics Bell")
    on FINISHED → Release Bell
  }

  state Snap Large Thread {
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 1, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 2, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Group 3, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEventRandom(audioClips=[silk_prop_slash_1 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_2 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_3 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_4 (sfxstatic_assets_areabellshrine.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Save State
  }

  state Hit Fx {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayAudioEventRandom(audioClips=[silk_prop_slash_1 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_2 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_3 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_4 (sfxstatic_assets_areabellshrine.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[bell_shrine_lower_finish (sfxstatic_assets_areabellshrine.bundle)], pitchMin=1, pitchMax=1.35, volume=0.65, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmission(gameObject=var "Pt Hit 1", emission=true, resetOnExit=true)
    SetParticleEmission(gameObject=var "Pt Hit 2", emission=true, resetOnExit=true)
    SetParticleEmission(gameObject=var "Pt Hit 3", emission=true, resetOnExit=true)
    on FINISHED → Inert
  }

  state Snap Start {
    ScreenFlash(flashColour=(1, 1, 1, 1))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="FreezeMoment", parameters=[1], storeResult=(unset var), EveryFrame=false)
    SetTransformParent(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Silk_thread_snap_effect, parent=<null>, worldPositionStays=true)
    ActivateGameObject(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Silk_thread_snap_effect, activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    GetPosition2D(gameObject=var "Damage Source", vector=var "Hit Source Position", x=(unset), y=(unset), space=0, everyFrame=false)
    GetVector2XY(vector2Variable=var "Hit Source Position", storeX=(unset), storeY=var "Hit Source Position Y", everyFrame=false)
    SetPosition(gameObject=Black Thread States/Normal World/Bellshrine Sequence/Hanging Bell/silk_threads/Silk_thread_snap_effect, vector=(unset), x=(unset), y=var "Hit Source Position Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    PlayAudioEvent(audioClip=silk_prop_kill (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=bell_shrine_arrive (sfxstatic_assets_areabellshrine.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Snap Large Thread
  }

  state Set Bell Released {
    FindChild(gameObject=Self, childName="Physics Bell Return", storeResult=var "Physics Bell")
    ActivateGameObject(gameObject=var "Physics Bell", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Release Bell
  }

  state Heavy Hit {
    SetBoolValue(boolVariable=var "Heavy Hit", boolValue=true, everyFrame=false)
    on FINISHED → Hit
  }

  var Finish State: int = 4

  var Hero Y: float = 0
  var Hit Source Position Y: float = 0
  var State: int = 0
  var Heavy Hit: bool = false
  var Hit Source Position: vector2 = (0, 0)
  var Physics Bell: gameObject = <null>
  var Pt Hit 1: gameObject = <null>
  var Pt Hit 2: gameObject = <null>
  var Pt Hit 3: gameObject = <null>
  var Self: gameObject = <null>
  var Slash Impact: gameObject = <null>
  var Damage Source: gameObject = <null>
}
