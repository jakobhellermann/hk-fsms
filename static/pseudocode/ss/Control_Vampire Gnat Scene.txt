# Control
# 1 location(s):
#   Greymoor_08_boss / Vampire Gnat Scene

fsm Control {
  start Init
  on BOSS DEATH → Killed Gnat  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Fight Start Range")
    FindNamedChild(gameObject=Self, storeResult=var "Tension Range")
    FindNamedChild(gameObject=Self, storeResult=var "Vampire Gnat")
    on FINISHED → Garmond Aid?
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Fight Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter
  }

  state Enter {
    ActivateGameObject(gameObject=var "Fight Start Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmFloat(gameObject=var "Vampire Gnat", fsmName="Control", variableName="Entry Y", setValue=12, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Vampire Gnat"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Tension Range"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="encounteredVampireGnatBoss", value=true)
    on FINISHED → Fighting
  }

  state Fighting {
    SendEventToRegisterDelay(EventName="GARMOND START", delay=8, ExcludeTarget=Self)
    on HORNET DEATH → Hornet Defeated
  }

  state Hornet Defeated {
    PlayerDataIntAdd(intName="vampireGnatDeaths", amount=1)
  }

  state Garmond Aid? {
    ActivateGameObject(gameObject=var "Garmond Fighter", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="vampireGnatRequestedAid", isTrue=(none), isFalse=→"FINISHED")
    PlayerDataBoolTest(boolName="visitedCitadel", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(boolName="garmondInDust05", isTrue=→"FINISHED", isFalse=(none))
    FindNamedChild(gameObject=Self, storeResult=var "Garmond Fighter")
    ActivateGameObject(gameObject=var "Garmond Fighter", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Killed Gnat {
    GetPlayerDataVariable(VariableName="CaravanTroupeLocation", StoreValue=var "Troupe Location")
    EnumCompare(enumVariable=var "Troupe Location", compareTo=Bone, equalEvent=(none), notEqualEvent=(none), storeResult=var "At Bone", everyFrame=false)
    EnumCompare(enumVariable=var "Troupe Location", compareTo=Greymoor, equalEvent=(none), notEqualEvent=(none), storeResult=var "Travelling To Greymoor", everyFrame=false)
    BoolTestMulti(boolVariables=[var "At Bone", var "Travelling To Greymoor"], boolStates=[false, false], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataVariable(VariableName="VampireGnatDefeatedBeforeCaravanArrived", SetValue=true)
    on FINISHED → Inert
  }

  state Inert {
  }

  var At Bone: bool = false
  var Travelling To Greymoor: bool = false
  var Fight Start Range: gameObject = <null>
  var Vampire Gnat: gameObject = <null>
  var Garmond Fighter: gameObject = <null>
  var Tension Range: gameObject = <null>
  var Troupe Location: enum = CaravanTroupeLocations(0)
}
