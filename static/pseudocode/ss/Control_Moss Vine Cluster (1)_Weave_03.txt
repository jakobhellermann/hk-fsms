# Control
# 1 location(s):
#   Weave_03 / Moss Vine Cluster (1)

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on ACTIVATED → Activated
    on FINISHED → Idle
  }

  state Activated {
    SetParent(gameObject=var "Stumps", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Stumps", activate=var "Leave Stumps", recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Nail Attack", sendEvent=→"NAIL HIT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"NAIL HIT", sendEventHeavy=→"DESTROY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetParticleEmission(gameObject=var "Pt Bits", emission=false, resetOnExit=false)
    SetParticleEmission(gameObject=var "Pt Spatter", emission=false, resetOnExit=false)
    on NAIL HIT → Strike
    on DESTROY → Heavy Break
  }

  state Strike {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Hit Cancel Range", storeResult=false, sendEvent="CANCEL", outOfRangeEvent="", everyFrame=false)
    SetParticleEmission(gameObject=var "Pt Bits", emission=true, resetOnExit=false)
    SetParticleEmission(gameObject=var "Pt Spatter", emission=true, resetOnExit=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Effects Origin", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Effects Origin", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Effects Origin", audioClip=enemy_damage_old (sfxstatic_assets_areabellshared.bundle), pitchMin=0.9, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    PlayAudioEvent(audioClip=moss_vine_cut (sfxstatic_assets_areatutorial.bundle), pitchMin=0.9, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Effects Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    BoolTest(boolVariable=var "Immortal", isTrue=→"1", isFalse=(none), everyFrame=false)
    IntSwitch(intVariable=var "Hits", compareTo=[1, 2, 3, 4], sendEvent=[→"1", →"2", →"3", →"4"], everyFrame=false)
    on 1 → Hit 1
    on 2 → Hit 2
    on 3 → Hit 3
    on 4 → PD Bool
    on FINISHED → PD Bool
    on CANCEL → Idle
  }

  state Hit 2 {
    Tk2dPlayAnimation(gameObject=var "Thick", animLibName="", clipName="Thick Snap")
    Tk2dPlayAnimation(gameObject=var "Standard", animLibName="", clipName="Standard Hit")
    Tk2dPlayAnimation(gameObject=var "Standard 2", animLibName="", clipName="Standard Hit")
    Tk2dPlayAnimation(gameObject=var "Twirl", animLibName="", clipName="Twirl Hit")
    ActivateGameObject(gameObject=var "Thick", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recover
  }

  state Recover {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit 3 {
    Tk2dPlayAnimation(gameObject=var "Standard", animLibName="", clipName="Standard Snap")
    Tk2dPlayAnimation(gameObject=var "Standard 2", animLibName="", clipName="Standard Hit")
    Tk2dPlayAnimation(gameObject=var "Twirl", animLibName="", clipName="Twirl Hit")
    ActivateGameObject(gameObject=var "Standard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recover
  }

  state Hit 1 {
    Tk2dPlayAnimation(gameObject=var "Thick", animLibName="", clipName="Thick Hit")
    Tk2dPlayAnimation(gameObject=var "Standard", animLibName="", clipName="Standard Hit")
    Tk2dPlayAnimation(gameObject=var "Standard 2", animLibName="", clipName="Standard Hit")
    Tk2dPlayAnimation(gameObject=var "Twirl", animLibName="", clipName="Twirl Hit")
    on FINISHED → Recover
  }

  state Hit 4 {
    Tk2dPlayAnimation(gameObject=var "Standard 2", animLibName="", clipName="Standard Snap")
    Tk2dPlayAnimation(gameObject=var "Twirl", animLibName="", clipName="Twirl Snap")
    ActivateGameObject(gameObject=var "Standard", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Standard 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Twirl", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thick", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Destroy Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Attached Object", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Attached Object"), sendEvent="VINES CUT", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Hero Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Stumps", activate=var "Leave Stumps", recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Extra Vines", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Mask", ToAlpha=0, FadeTime=0.2)
    PlayAudioEvent(audioClip=moss_vine_death (sfxstatic_assets_areatutorial.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Effects Origin", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Pt
  }

  state End Pt {
    SetParticleEmission(gameObject=var "Pt Bits", emission=false, resetOnExit=false)
    SetParticleEmission(gameObject=var "Pt Spatter", emission=false, resetOnExit=false)
  }

  state PD Bool {
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "PD Bool", value=true)
    on FINISHED → Hit 4
  }

  state Heavy Break {
    SetIntValue(intVariable=var "Hits", intValue=4, everyFrame=false)
    on FINISHED → Strike
  }

  var Leave Stumps: bool = false
  var PD Bool: string = ""
  var Attached Object: gameObject = <null>

  var Hits: int = 0
  var Activated: bool = false
  var Immortal: bool = false
  var Destroy Effects: gameObject = Moss Vine Cluster (1)/Destroy Effects
  var Extra Vines: gameObject = Moss Vine Cluster (1)/Extra Vines
  var Hero Blocker: gameObject = Moss Vine Cluster (1)/Hero Blocker
  var Mask: gameObject = Moss Vine Cluster (1)/Mask
  var Pt Bits: gameObject = Moss Vine Cluster (1)/Pt Bits
  var Pt Spatter: gameObject = Moss Vine Cluster (1)/Pt Spatter
  var Self: gameObject = Moss Vine Cluster (1)
  var Standard: gameObject = Moss Vine Cluster (1)/Standard
  var Standard 2: gameObject = Moss Vine Cluster (1)/Standard 2
  var Stumps: gameObject = Moss Vine Cluster (1)/Stumps
  var Thick: gameObject = Moss Vine Cluster (1)/Thick
  var Twirl: gameObject = Moss Vine Cluster (1)/Twirl
  var Effects Origin: gameObject = Moss Vine Cluster (1)/Effects Origin
}
