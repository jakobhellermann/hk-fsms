# Set Target
# 1 location(s):
#   Bellway_Centipede_Arena / top1

fsm Set Target {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Target
  }

  state Set Target {
    GetPlayerDataVariable(VariableName="FastTravelNPCLocation", StoreValue=var "Fast Travel NPC Location")
    GetFastTravelScene(Location=var "Fast Travel NPC Location", StoreSceneName=var "Fast Travel NPC Scene", InvalidEvent=→"FINISHED")
    CallMethodProper(gameObject=Self, behaviour="TransitionPoint", methodName="SetTargetScene", parameters=[var "Fast Travel NPC Scene"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Inert
  }

  state Inert {
  }

  var Fast Travel NPC Scene: string = ""
  var Fast Travel NPC Location: enum = FastTravelLocations(0)
}
