# Behaviour (special)
# 2 location(s):
#   Under_01b / Understore Toll Bench
#   Under_08 / Understore Toll Bench

fsm Behaviour (special) {
  start Init

  state Inert {
    on INTERACT → Confirm
    on BENCHREST START → Start Sitting
  }

  sequence state Confirm {
    EndDialogue(ReturnControl=false, ReturnHUD=true, Target=Self, UseChildren=false)  // disabled
    GetLocalisedString(Cell={Sheet="Fast Travel", Key="PAY_TOLL_MSG"}, StoreString=var "Prompt Text Override")
    RunFSM(fsmTemplateControl=template=-8990696591063553075 in[Hornet Stays Reaching<-false, Prompt Text Override<-var "Prompt Text Override", Dish<-Understore Toll Bench/Pivot/Regular Appearance/Receptacle Brace/Dish, Cost Reference<-var "Cost Reference"], finishEvent=→"FINISHED", everyFrame=false)
    on FINISHED → Flip to Bench
    on CANCEL → End Interaction
    on CONVO_END_FORCED → End Interaction
  }

  state End Interaction {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Inert
  }

  state Init {
    ActivateInteractible(Target=var "RestBench", Activate=false, AllowQueueing=false, UseChildren=false)
    FadeNestedFadeGroup(Target=var "RestBench", ToAlpha=0, FadeTime=0)
    GetPosition2D(gameObject=var "Bench Sprite", vector=var "Bench Sprite Initial Pos", x=(unset), y=(unset), space=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Dust Parent", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inert
    on BENCHREST START → Start Sitting
  }

  sequence state Flip to Bench {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    PlayAudioEvent(audioClip=understore_toll_bench_activate (sfxstatic_assets_areaunderstorebench.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    VibrationRegionPlayOneShot(target=var "Vibration Region", vibrationDataAsset=understore_toll_bench_activate (vibrationstatic_assets_vibrationdataasset.bundle), requireInside=false, loop=false, isRealTime=false, tag="", stopOnRegionExit=false, stopOnStateExit=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Dust Parent", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlayStateWait(target=Self, stateName="Flip to Bench", finishEvent=(none))
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StopJitter())
    AnimatorPlay(gameObject=Self, stateName="Bench", layer=(unset), normalizedTime=(unset), everyFrame=false)
    on FINISHED → Fade Bench Up
  }

  sequence state Fade Bench Up {
    FadeNestedFadeGroup(Target=var "RestBench", ToAlpha=1, FadeTime=1)
    Wait(time=1, finishEvent=(none), realTime=false)
    ActivateInteractible(Target=var "RestBench", Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Wait For Sit
  }

  state Timer {
    SetPosition2D(GameObject=var "Bench Sprite", Vector=var "Bench Sprite Initial Pos", X=(unset), Y=0, Space=1, EveryFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    AudioStopV2(gameObject=var "Audio Source", fadeTime=var "Sit Audio Fade Time", cancelOnEarlyExit=true)
    on BENCHREST START → On Bench
    on FINISHED → Flip To Toll
  }

  state On Bench {
    SetPosition2D(GameObject=var "Bench Sprite", Vector=var "Bench Sprite Initial Pos", X=(unset), Y=0, Space=1, EveryFrame=false)
    Translate(gameObject=var "Bench Sprite", vector=(unset), x=(unset), y=0.03, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetAudioVolume(gameObject=var "Audio Source", volume=1, everyFrame=false)
    AudioPlayRandom(gameObject=var "Audio Source", audioClips=[generic_metal_creak_1 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_2 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_3 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_4 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_5 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_6 (sfxstatic_assets_areasongareaunderstore.bundle)], weights=[1, 1, 1, 1, 1, 1], pitchMin=0.9, pitchMax=1.1)
    AudioStopV2(gameObject=var "Audio Source", fadeTime=var "Sit Audio Fade Time", cancelOnEarlyExit=true)
    on BENCHREST END → Play Sit Audio
  }

  sequence state Flip To Toll {
    ActivateGameObject(gameObject=var "Dust Parent", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateInteractible(Target=var "RestBench", Activate=false, AllowQueueing=false, UseChildren=false)
    FadeNestedFadeGroup(Target=var "RestBench", ToAlpha=0, FadeTime=0.5)
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    PlayAudioEvent(audioClip=understore_toll_bench_deactivate (sfxstatic_assets_areaunderstorebench.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    VibrationRegionPlayOneShot(target=var "Vibration Region", vibrationDataAsset=understore_toll_bench_activate (vibrationstatic_assets_vibrationdataasset.bundle), requireInside=false, loop=false, isRealTime=false, tag="", stopOnRegionExit=false, stopOnStateExit=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    AnimatorPlayStateWait(target=Self, stateName="Flip to Toll", finishEvent=(none))
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StopJitter())
    AnimatorPlay(gameObject=Self, stateName="Toll", layer=(unset), normalizedTime=(unset), everyFrame=false)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Inert
  }

  state Wait For Sit {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on BENCHREST START → On Bench
    on FINISHED → Flip To Toll
  }

  state Start Sitting {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateInteractible(Target=var "RestBench", Activate=true, AllowQueueing=false, UseChildren=false)
    FadeNestedFadeGroup(Target=var "RestBench", ToAlpha=1, FadeTime=0)
    AnimatorPlay(gameObject=Self, stateName="Bench", layer=(unset), normalizedTime=(unset), everyFrame=false)
    SetPosition2D(GameObject=var "Bench Sprite", Vector=var "Bench Sprite Initial Pos", X=(unset), Y=0, Space=1, EveryFrame=false)
    Translate(gameObject=var "Bench Sprite", vector=(unset), x=(unset), y=0.03, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on BENCHREST END → Timer
  }

  state Inspect {
    RunDialogue(Sheet="Prompts", Key="TOLL_BENCH", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=Top, OffsetY=0, Target=Self)
    on CONVO_END_FORCED → End Interaction
    on CONVO_END → Confirm
  }

  state Play Sit Audio {
    SetAudioVolume(gameObject=var "Audio Source", volume=1, everyFrame=false)
    AudioPlayRandom(gameObject=var "Audio Source", audioClips=[generic_metal_creak_1 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_2 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_3 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_4 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_5 (sfxstatic_assets_areasongareaunderstore.bundle), generic_metal_creak_6 (sfxstatic_assets_areasongareaunderstore.bundle)], weights=[1, 1, 1, 1, 1, 1], pitchMin=0.9, pitchMax=1.1)
    on FINISHED → Timer
  }

  var Sit Audio Fade Time: float = 0.25
  var Cost Reference: object = Understore Toll Bench Default (dataassets_assets_assets/dataassets/costs.bundle)

  var Prompt Text Override: string = ""
  var Bench Sprite Initial Pos: vector2 = (0, 0)
  var RestBench: gameObject = Understore Toll Bench/RestBench
  var Bench Sprite: gameObject = Understore Toll Bench/Pivot/Regular Appearance/Bench
  var Dust Parent: gameObject = Understore Toll Bench/Pivot/Dust
  var Vibration Region: gameObject = Understore Toll Bench/Hero Vibration Region Toll Bench
  var Audio Source: gameObject = Understore Toll Bench
}
