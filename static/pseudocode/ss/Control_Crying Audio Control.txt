# Control
# 1 location(s):
#   Shadow_20 / Collectable Mandrake Scene/Crying Audio Control

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Audio Player Close")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Player Far")
    FindNamedChild(gameObject=Self, storeResult=var "Far Range")
    FindNamedChild(gameObject=Self, storeResult=var "Close Range")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Player Sing")
    on FINISHED → Out of Range
  }

  state Out of Range {
    AudioStop(gameObject=var "Audio Player Close", fadeTime=0)
    AudioStop(gameObject=var "Audio Player Far", fadeTime=0)
    CheckTrackTriggerCount(target=var "Close Range", count=0, test=MoreThan, everyFrame=true, successEvent=→"ENTER CLOSE")
    CheckTrackTriggerCount(target=var "Far Range", count=0, test=MoreThan, everyFrame=true, successEvent=→"ENTER FAR")
    on ENTER CLOSE → Close Range
    on ENTER FAR → Far Range
  }

  state Close Range {
    AudioPlaySimple(gameObject=var "Audio Player Close", volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Player Close", startVolume=0, endVolume=1, time=0.5)
    FadeAudio(gameObject=var "Audio Player Sing", startVolume=1, endVolume=0, time=0.5)
    CheckTrackTriggerCount(target=var "Close Range", count=0, test=LessThanOrEqual, everyFrame=true, successEvent=→"EXIT")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=3, MaxReactDelay=3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on EXIT → Out of Range
    on SING → Sing
  }

  state Far Range {
    AudioPlaySimple(gameObject=var "Audio Player Far", volume=1, oneShotClip=<null>)
    CheckTrackTriggerCount(target=var "Far Range", count=0, test=LessThanOrEqual, everyFrame=true, successEvent=→"EXIT")
    on EXIT → Out of Range
  }

  state Sing {
    FadeAudio(gameObject=var "Audio Player Close", startVolume=1, endVolume=0, time=1)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=→"FINISHED", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    AudioPlaySimple(gameObject=var "Audio Player Sing", volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Player Sing", startVolume=0, endVolume=1, time=1)
    on FINISHED → Close Range
  }

  var Audio Player Close: gameObject = <null>
  var Audio Player Far: gameObject = <null>
  var Far Range: gameObject = <null>
  var Close Range: gameObject = <null>
  var Audio Player Sing: gameObject = <null>
}
