# Control
# 1 location(s):
#   Dust_Maze_Last_Hall / Directional Activation Control

fsm Control {
  start Pause

  state Pause {
    FindNamedChild(gameObject=Self, storeResult=var "Entered From L")
    FindNamedChild(gameObject=Self, storeResult=var "Entered From R")
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Entered From L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Entered From R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Coming From Maze?
  }

  state From Maze Check {
    SetBoolValue(boolVariable=var "Display Silkflies", boolValue=true, everyFrame=false)
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="left1", equalEvent=→"LEFT", notEqualEvent=→"RIGHT", storeResult=(unset), everyFrame=false)
    on LEFT → Maze Is L
    on RIGHT → Maze Is R
  }

  state Maze Is L {
    SetGameObject(variable=var "Folder To Activate", gameObject=var "Entered From L", everyFrame=false)
    on FINISHED → Check Silkflies
  }

  state Maze Is R {
    SetGameObject(variable=var "Folder To Activate", gameObject=var "Entered From R", everyFrame=false)
    on FINISHED → Check Silkflies
  }

  state Check Silkflies {
    ActivateGameObject(gameObject=var "Folder To Activate", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Display Silkflies", isTrue=(none), isFalse=→"NO SILKFLIES", everyFrame=false)
    on NO SILKFLIES → No Silkflies
    on FINISHED → Activate Silkflies
  }

  state No Silkflies {
  }

  state Activate Silkflies {
    FindNamedChild(gameObject=var "Folder To Activate", storeResult=var "Silkflies")
    ActivateGameObject(gameObject=var "Silkflies", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Coming From Maze? {
    PlayerDataVariableTest(VariableName="PreviousMazeScene", ExpectedValue="Dust_09", IsExpectedEvent=→"FROM ORGAN", IsNotExpectedEvent=→"FROM MAZE")
    on FROM ORGAN → From Organ
    on FROM MAZE → From Maze Check
  }

  state From Organ {
    SetBoolValue(boolVariable=var "Display Silkflies", boolValue=false, everyFrame=false)
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="left1", equalEvent=→"LEFT", notEqualEvent=→"RIGHT", storeResult=(unset), everyFrame=false)
    on LEFT → Maze Is R
    on RIGHT → Maze Is L
  }

  var Display Silkflies: bool = false
  var Gate: string = ""
  var Entered From L: gameObject = <null>
  var Entered From R: gameObject = <null>
  var Folder To Activate: gameObject = <null>
  var Silkflies: gameObject = <null>
}
