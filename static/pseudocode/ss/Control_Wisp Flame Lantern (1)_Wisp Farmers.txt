# Control
# 1 location(s):
#   Wisp_02 / Wisp Farmers/Wisp Flame Lantern (1)

fsm Control {
  start Init
  on BREAK → Death Spawn?  // from any state
  on BROKEN → Inert Broken  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    CheckIfToolEquipped(Tool=Wisp Lantern (dataassets_assets_assets/dataassets/tools/toolitems.bundle), RequiredAmountLeft=(unset), trueEvent=(none), falseEvent=(none), storeValue=(unset))
    on FINISHED → Witch Check
    on FRIENDLY → Friendly
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    SetFloatValue(floatVariable=var "Time Since Last Spawn", floatValue=0, everyFrame=false)
    on ENTER → In Range
  }

  state In Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    FloatAdd(floatVariable=var "Time Since Last Spawn", add=1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Time Since Last Spawn", minValue=0, maxValue=100, everyFrame=true)
    on EXIT → Idle
    on FINISHED → Summon Antic
  }

  state Summon {
    Wait(time=1.65, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Wisp Fireball (localpoolprefabs_assets_areawisp.bundle), spawnPoint=var "Pod", position=(0, 0, -0.002), rotation=(unset), storeObject=var "Wisp")
    SetFsmGameObject(gameObject=var "Wisp", fsmName="Control", variableName="Wisp Fireball Master", setValue=var "Wisp Fireball Master", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Wisp"), sendEvent="FLAME IN", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Wisp", fsmName="Control", variableName="Flame In", setValue=true, everyFrame=false)
    RandomFloat(min=60, max=120, storeResult=var "Spawn Angle")
    SetVelocityAsAngle(gameObject=var "Wisp", angle=var "Spawn Angle", speed=4, everyFrame=false)
    AddForce2dAsAngle(gameObject=var "Lamp", forceMode=1, atPosition=(unset), angle=var "Spawn Angle", speed=-20, maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=coral_final_judge_ring_of_flame_appear (sfxstatic_assets_areacoral.bundle), pitchMin=1.1, pitchMax=1.1, volume=0.75, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetFloatValue(floatVariable=var "Time Since Last Spawn", floatValue=0, everyFrame=false)
    FloatAdd(floatVariable=var "Time Since Last Spawn", add=1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Time Since Last Spawn", minValue=0, maxValue=100, everyFrame=true)
    on FINISHED → In Range
    on TOOK DAMAGE → In Range
  }

  state Asleep {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on WAKE → Idle
  }

  state Broken {
    RecordJournalKill(Record=Wisp (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
  }

  state Summon Antic {
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=coral_final_judge_vertical_flame_antic (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FloatAdd(floatVariable=var "Time Since Last Spawn", add=1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Time Since Last Spawn", minValue=0, maxValue=100, everyFrame=true)
    on FINISHED → Summon
  }

  state Friendly {
  }

  state Death Spawn? {
    FloatCompare(float1=var "Time Since Last Spawn", float2=0.25, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Wisp Fireball (localpoolprefabs_assets_areawisp.bundle), spawnPoint=var "Pod", position=(unset), rotation=(unset), storeObject=var "Wisp")
    SetFsmGameObject(gameObject=var "Wisp", fsmName="Control", variableName="Wisp Fireball Master", setValue=var "Wisp Fireball Master", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Wisp"), sendEvent="FLAME IN", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Wisp", fsmName="Control", variableName="Flame In", setValue=true, everyFrame=false)
    RandomFloat(min=60, max=120, storeResult=var "Spawn Angle")
    SetVelocityAsAngle(gameObject=var "Wisp", angle=var "Spawn Angle", speed=4, everyFrame=false)
    AddForce2dAsAngle(gameObject=var "Lamp", forceMode=1, atPosition=(unset), angle=var "Spawn Angle", speed=-20, maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=coral_final_judge_censer_grounded_flare_up_ (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Broken
  }

  state Inert Broken {
    EnableFsmSelf(SetEnabled=false)
  }

  state Witch Check {
    CheckIfCrestEquipped(Crest=Witch (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"WITCH", falseEvent=(none), storeValue=(unset))
    CheckIfCrestEquipped(Crest=Cursed (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"WITCH", falseEvent=(none), storeValue=(unset))
    CheckIfCrestEquipped(Crest=Reaper (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"WITCH", falseEvent=(none), storeValue=(unset))
    CheckIfCrestEquipped(Crest=Warrior (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"WITCH", falseEvent=(none), storeValue=(unset))
    CheckIfCrestEquipped(Crest=Toolmaster (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"WITCH", falseEvent=(none), storeValue=(unset))
    CheckIfCrestEquipped(Crest=Spell (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"WITCH", falseEvent=(none), storeValue=(unset))
    on WITCH → Witch Fix
    on FINISHED → Asleep
  }

  state Witch Fix {
    SetLayer(gameObject=Wisp Farmers/Wisp Flame Lantern (1)/lamp/Active, layer=Interactive Object(19))
    on FINISHED → Asleep
  }

  var Wisp Fireball Master: gameObject = Wisp Farmers/Wisp Fireball Master

  var Spawn Angle: float = 0
  var Angle: float = 0
  var Time Since Last Spawn: float = 100
  var Woken: bool = true
  var Wisp: gameObject = <null>
  var Self: gameObject = <null>
  var Pod: gameObject = Wisp Farmers/Wisp Flame Lantern (1)/lamp
  var Wake Effects: gameObject = Wisp Farmers/Wisp Flame Lantern (1)/Wake Effects
  var Lamp: gameObject = Wisp Farmers/Wisp Flame Lantern (1)/lamp
  var Pt Antic: gameObject = Wisp Farmers/Wisp Flame Lantern (1)/Pt Antic
}
