# Control
# 1 location(s):
#   Dust_01 / Lace Encounter Control/Music Control

fsm Control {
  start Wait
  on LACE PRESENT → Cancel  // from any state
  on LACE LEAVE → Start Music  // from any state

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Music
  }

  state Cancel {
  }

  state Start Music {
    ApplyMusicCue(musicCue=Dustpens (audiocuesdynamic_assets_areadust.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Main Only (audioobjects_assets_all.bundle), transitionTime=4)
  }
}
