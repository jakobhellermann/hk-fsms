# Animation Control
# 1 location(s):
#   Cradle_03 / Boss Scene/Silk_Hair

fsm Animation Control {
  start Inert
  on LOW → Low  // from any state
  on IDLE → Idle  // from any state
  on ROAR → Roar  // from any state
  on STUN → Stun  // from any state
  on DEATH PT1 → Death  // from any state
  on DEATH FRAYED → Death Frayed  // from any state

  state Inert {
  }

  state Low {
    SetAnimatorTrigger(gameObject=var "Animator", trigger="Down")
    SetRotation2DLerp(Target=var "Animator", TargetRotationZ=0, LerpSpeed=var "Rotation Lerp Speed", Space=1)
    RendererFloatRamp(Target=var "Splines", PropertyName="_SpeedY", From=var "Initial Speed", To=0.95, Duration=0.3, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], UseChildren=true)
    SetFloatValue(floatVariable=var "Initial Speed", floatValue=0.95, everyFrame=false)
  }

  state Idle {
    SetAnimatorTrigger(gameObject=var "Animator", trigger="Idle")
    GetScale(gameObject=var "Silk Boss", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=true)
    FloatSignToBool(Value=var "X Scale", StoreIsPositive=var "Facing Right", EveryFrame=true)
    BoolFlipEveryFrame(boolVariable=var "Facing Right", everyFrame=true)
    ConvertBoolToFloat(boolVariable=var "Facing Right", floatVariable=var "Rotation Target", falseValue=10, trueValue=-10, everyFrame=true)
    SetRotation2DLerp(Target=var "Animator", TargetRotationZ=var "Rotation Target", LerpSpeed=var "Rotation Lerp Speed", Space=1)
    RendererFloatRamp(Target=var "Splines", PropertyName="_SpeedY", From=var "Initial Speed", To=0.95, Duration=0.8, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], UseChildren=true)
    SetFloatValue(floatVariable=var "Initial Speed", floatValue=0.95, everyFrame=false)
  }

  state Roar {
    SetAnimatorTrigger(gameObject=var "Animator", trigger="Idle")
    SetRotation2DLerp(Target=var "Animator", TargetRotationZ=0, LerpSpeed=var "Rotation Lerp Speed", Space=1)
    RendererFloatRamp(Target=var "Splines", PropertyName="_SpeedY", From=var "Initial Speed", To=2, Duration=0.3, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], UseChildren=true)
    SetFloatValue(floatVariable=var "Initial Speed", floatValue=2, everyFrame=false)
  }

  state Stun {
    SetAnimatorTrigger(gameObject=var "Animator", trigger="Up")
    GetScale(gameObject=var "Silk Boss", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=true)
    FloatSignToBool(Value=var "X Scale", StoreIsPositive=var "Facing Right", EveryFrame=true)
    BoolFlipEveryFrame(boolVariable=var "Facing Right", everyFrame=true)
    ConvertBoolToFloat(boolVariable=var "Facing Right", floatVariable=var "Rotation Target", falseValue=10, trueValue=-10, everyFrame=true)
    SetRotation2DLerp(Target=var "Animator", TargetRotationZ=var "Rotation Target", LerpSpeed=var "Rotation Lerp Speed", Space=1)
    RendererFloatRamp(Target=var "Splines", PropertyName="_SpeedY", From=var "Initial Speed", To=2, Duration=0.3, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], UseChildren=true)
    SetFloatValue(floatVariable=var "Initial Speed", floatValue=2, everyFrame=false)
  }

  state Death {
    SetAnimatorTrigger(gameObject=var "Animator", trigger="Up")
    SetRotation2DLerp(Target=var "Animator", TargetRotationZ=10, LerpSpeed=var "Rotation Lerp Speed", Space=1)
    RendererFloatRamp(Target=var "Splines", PropertyName="_SpeedY", From=var "Initial Speed", To=2.8, Duration=0.3, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], UseChildren=true)
    SetFloatValue(floatVariable=var "Initial Speed", floatValue=2.8, everyFrame=false)
  }

  state Death Frayed {
    SetAnimatorTrigger(gameObject=var "Animator", trigger="Up")
    SetRotation2DLerp(Target=var "Animator", TargetRotationZ=10, LerpSpeed=var "Rotation Lerp Speed", Space=1)
    ReplaceMaterials(Parent=var "Splines", From=[Silk Hair (materials_assets_areacradle.bundle), Silk Hair Thin (materials_assets_areacradle.bundle)], To=[Silk Hair Frayed (materials_assets_areacradle.bundle), Silk Hair Thin Frayed (materials_assets_areacradle.bundle)])
    RendererFloatRamp(Target=var "Splines", PropertyName="_SpeedY", From=var "Initial Speed", To=4, Duration=0.3, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], UseChildren=true)
    SetFloatValue(floatVariable=var "Initial Speed", floatValue=4, everyFrame=false)
  }

  var Silk Boss: gameObject = Boss Scene/Silk Boss

  var Initial Speed: float = 0.95
  var Rotation Lerp Speed: float = 6
  var Rotation Target: float = 0
  var X Scale: float = 0
  var Facing Right: bool = false
  var Animator: gameObject = Boss Scene/Silk_Hair/Hair Point Follow/Sub Animator
  var Splines: gameObject = Boss Scene/Silk_Hair/Sub Splines
}
