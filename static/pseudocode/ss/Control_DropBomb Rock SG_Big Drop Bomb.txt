# Control
# 1 location(s):
#   Abyss_09 / Big Drop Bomb/DropBomb Rock SG

fsm Control {
  start Pause

  state Pause {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset State
  }

  state Init {
    FindChild(gameObject=Self, childName="Updraft After", storeResult=var "Updraft After")
    FindChild(gameObject=Self, childName="Updraft Before", storeResult=var "Updraft Before")
    FindChild(gameObject=Self, childName="wind_column", storeResult=var "wind_column")
    SetParent(gameObject=var "Updraft After", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Updraft Before", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "wind_column", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    Translate(gameObject=var "Updraft After", vector=(unset), x=-300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FindChild(gameObject=Self, childName="joiner/joiner 1", storeResult=var "joiner 1")
    FindChild(gameObject=Self, childName="Dust Trail", storeResult=var "Dust Trail")
    FindChild(gameObject=Self, childName="joiner/joiner 2", storeResult=var "joiner 2")
    FindChild(gameObject=Self, childName="joiner/joiner 3", storeResult=var "joiner 3")
    on ACTIVATED → Activated
    on FINISHED → Idle
  }

  state Activated {
    Translate(gameObject=var "Updraft Before", vector=(unset), x=-300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayParticleEmitterChildren(gameObject=var "wind_column", resetTimeIfPlaying=false, stopOnStateExit=false)
    ActivateGameObject(gameObject=var "Break Floor", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    BoolAllTrue(boolVariables=[var "Exploded Wall L", var "Exploded Wall R"], sendEvent=→"STRIKE", storeResult=(unset), everyFrame=true)
    on STRIKE → Strike
  }

  state Strike {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    ActivateGameObject(gameObject=var "Clash Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClips=[drop_bomb_rock_initial_hit (sfxstatic_assets_areawilds.bundle)], weights=[1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Crumble", resetTimeIfPlaying=false, stopOnStateExit=false)
    StopParticleEmitter(gameObject=var "Pt Idle")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Explode Instantly", time=0.15, finishEvent=→"FINISHED", realTime=false)
    ObjectJitterLocal(gameObject=var "joiner 1", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=false, limitFps=0)
    ObjectJitterLocal(gameObject=var "joiner 2", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=false, limitFps=0)
    ObjectJitterLocal(gameObject=var "joiner 3", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=false, limitFps=0)
    ActivateGameObject(gameObject=var "glower", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fall Start
  }

  state Explode {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Translate(gameObject=var "Updraft After", vector=(unset), x=300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClips=[drop_bomb_rock_explode_on_chorus (sfxstatic_assets_areawilds.bundle)], weights=[1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Translate(gameObject=var "Updraft Before", vector=(unset), x=-300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    PlayParticleEmitterChildren(gameObject=var "wind_column", resetTimeIfPlaying=false, stopOnStateExit=false)
    SendEventByName(eventTarget=GameObject(var "Break Floor"), sendEvent="EXPLODE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Fall Start {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClips=[drop_bomb_rock_break_free (sfxstatic_assets_areawilds.bundle)], weights=[1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "joiner 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "joiner 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "joiner 3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Golem Bomb", isTrue=→"GOLEM", isFalse=(none), everyFrame=false)
    on GOLEM → Golem Fall
    on FINISHED → Drop Damager
  }

  state Golem Fall {
    SetPlayerDataBool(boolName="destroyedSongGolemRock", value=true)
    AccelerateToY(gameObject=Self, accelerationFactor=2, targetSpeed=-40)
    CheckYPosition(gameObject=Self, compareTo=var "Golem Y", compareToOffset=0, tolerance=0, equal=→"EXPLODE", lessThan=→"EXPLODE", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    PlayParticleEmitterInState(gameObject=var "Dust Trail")
    Rotate(gameObject=var "rock", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=450, space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on EXPLODE → Golem Register
  }

  state Golem Register {
    SendEventToRegister(eventName="GOLEM EXPLOSION")
    on FINISHED → Safer Explosion
  }

  state Fall {
    BoolTest(boolVariable=var "Explode Instantly", isTrue=→"EXPLODE", isFalse=(none), everyFrame=false)
    AccelerateToY(gameObject=Self, accelerationFactor=2, targetSpeed=-40)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"EXPLODE", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer=Enemies(11), sendEvent=→"EXPLODE", storeCollider=var "")
    PlayParticleEmitterInState(gameObject=var "Dust Trail")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Rotate(gameObject=var "rock", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=450, space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetPlayerDataBool(boolName="destroyedSongGolemRock", value=true)
    on EXPLODE → Explosion
  }

  state Explosion {
    SendEventToRegister(eventName="BIG BOMB EXPLOSION")
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Recycle M (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Explode
  }

  state Safer Explosion {
    SpawnObjectFromGlobalPool(gameObject=Safer Explosion (localpoolprefabs_assets_areaabyssareawilds.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Explode
  }

  state Reset State {
    PlayerDataBoolTest(boolName="defeatedSongGolem", isTrue=→"FINISHED", isFalse=(none))
    SetPlayerDataBool(boolName="destroyedSongGolemRock", value=false)
    on FINISHED → Init
  }

  state Drop Damager {
    FindNamedChild(gameObject=Self, storeResult=var "Drop Damager")
    GameObjectIsNull(gameObject=var "Drop Damager", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Drop Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Fall
  }

  var Golem Bomb: bool = false
  var Break Floor: gameObject = Lava Rise Control/Rising Lava

  var Golem Y: float = 15.72
  var Activated: bool = false
  var Exploded Wall R: bool = false
  var Exploded Wall L: bool = false
  var Explode Instantly: bool = false
  var Break Effects: gameObject = Big Drop Bomb/DropBomb Rock SG/Break Effects
  var Clash Effects: gameObject = Big Drop Bomb/DropBomb Rock SG/Clash Effects
  var Dust Trail: gameObject = <null>
  var Pt Crumble: gameObject = Big Drop Bomb/DropBomb Rock SG/Pt Crumble
  var Pt Idle: gameObject = Big Drop Bomb/DropBomb Rock SG/Pt Idle
  var Self: gameObject = <null>
  var Terrain: gameObject = Big Drop Bomb/DropBomb Rock SG/Terrain
  var Terrain Detector: gameObject = Big Drop Bomb/DropBomb Rock SG/Terrain Detector
  var Updraft After: gameObject = <null>
  var Updraft Before: gameObject = <null>
  var joiner 1: gameObject = <null>
  var joiner 2: gameObject = <null>
  var joiner 3: gameObject = <null>
  var rock: gameObject = Big Drop Bomb/DropBomb Rock SG/rock
  var wind_column: gameObject = <null>
  var glower: gameObject = Big Drop Bomb/DropBomb Rock SG/glower
  var Drop Damager: gameObject = <null>
}
