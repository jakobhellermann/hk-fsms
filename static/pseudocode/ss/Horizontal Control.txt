// component disabled: this FSM does not run until something enables it
// uses template: hornet_last_dive_control
fsm Horizontal Control {
  start Init

  state Init {
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=var "Centre X", y=(unset), space=0, everyFrame=false)
    FloatOperator(float1=0, float2=0, operation=Add, storeResult=(unset), everyFrame=false)  // disabled
    on FINISHED → Idle
  }

  state Idle {
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=var "Right Is Pressed", pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=var "Left Is Pressed", pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    AddForce2dConditional(addForceBool=var "Left Is Pressed", gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=-7.5, y=(unset), vector3=(unset), everyFrame=true)
    AddForce2dConditional(addForceBool=var "Right Is Pressed", gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=7.5, y=(unset), vector3=(unset), everyFrame=true)
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=var "Self X", y=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Centre X", float2=var "Self X", operation=Subtract, storeResult=var "Centering Force", everyFrame=true)
    FloatMultiply(floatVariable=var "Centering Force", multiplyBy=12, everyFrame=true)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "Centering Force", y=(unset), vector3=(unset), everyFrame=true)
  }

  var Centre X: float = 0
  var Self X: float = 0
  var Centering Force: float = 0
  var Left Is Pressed: bool = false
  var Right Is Pressed: bool = false
}
