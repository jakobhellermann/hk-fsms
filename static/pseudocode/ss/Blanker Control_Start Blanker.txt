# Blanker Control
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/CameraParent/tk2dCamera/Start Blanker
#   Menu_Title / _GameCameras/CameraParent/tk2dCamera/Start Blanker

fsm Blanker Control {
  start Init
  on START FADE OUT → Idle  // from any state
  on HIDE START BLANKER → Hide  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GameObjectIsNull(gameObject=var "Start Blanker", isNull=(none), isNotNull=→"CANCEL", storeResult=(unset), everyFrame=false)
    SetGameObject(variable=var "Start Blanker", gameObject=var "Self", everyFrame=false)
    SetSpriteRenderer(gameObject=var "Self", active=true)
    SetMaterialColor(gameObject=var "Self", materialIndex=0, material=<null>, namedColor="_Color", color=(1, 1, 1, 0), everyFrame=false)
    SetFsmGameObject(gameObject=var "MainCamera", fsmName="CameraFade", variableName="Blanker", setValue=var "Self", everyFrame=false)  // disabled
    on FINISHED → Idle
    on CANCEL → Invalid
  }

  state Fade In {
    SetColorValue(colorVariable=var "Colour", color=var "Start Colour", everyFrame=false)
    EaseColor(fromValue=var "Start Colour", toValue=var "End Colour", colorVariable=var "Colour", time=2.3, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    SetSpriteRenderer(gameObject=Self, active=true)
    on FADE OUT → Fade Out
    on OUT → Idle
    on START FADE OUT → Fade Out
  }

  state Fade Out {
    SetColorValue(colorVariable=var "Colour", color=var "End Colour", everyFrame=false)
    EaseColor(fromValue=var "End Colour", toValue=var "Start Colour", colorVariable=var "Colour", time=1.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetMaterialColor(gameObject=Self, materialIndex=0, material=<null>, namedColor="_Color", color=var "Colour", everyFrame=true)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FADE IN → Fade In
    on FINISHED → Idle
  }

  state Idle {
    SetSpriteRenderer(gameObject=Self, active=false)
    on START FADE → Fade In
  }

  state Hide {
    SetSpriteRenderer(gameObject=Self, active=false)
  }

  state Invalid {
  }

  var Fade Time: float = 1
  var Cancel Reset: bool = false
  var Colour: color = (0, 0, 0, 1)
  var End Colour: color = (0, 0, 0, 1)
  var Start Colour: color = (0, 0, 0, 0)
  var Self: gameObject = <null>
}
