# End Scene
# 1 location(s):
#   Bone_01 / Battle Scene/End Scene

fsm End Scene {
  start Idle

  state Idle {
    on BATTLE COMPLETED → Deactivate
    on BATTLE END → End Pause
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state End Pause {
    on FINISHED → End Antic
  }

  state End Antic {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BREAK", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Break {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName="Secret Mask", storeResult=var "Secret Mask")
    SendMessage(gameObject=var "Secret Mask", delivery=SendMessage, options=1, functionCall=Uncover())
    FindChild(gameObject=Self, childName="CameraLockArea B", storeResult=var "CameraLockArea B")
    ActivateGameObject(gameObject=var "CameraLockArea B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var CameraLockArea B: gameObject = <null>
  var Secret Mask: gameObject = <null>
}
