# Control
# 1 location(s):
#   Cradle_Destroyed_Challenge_01 / Spike Lazy Flyer (3)

fsm Control {
  start Init
  on ZERO HP → Die  // from any state
  on HAZARD RESPAWNED → Hazard Reset  // from any state

  state Init {
    SetGameObject(variable=var "Next Point", gameObject=var "Patrol Point", everyFrame=false)
    SetParent(gameObject=var "Patrol Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Start Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Fly In Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    RandomFloat(min=0.006, max=0.0069, storeResult=var "Z Pos")
    GetPosition(gameObject=var "Start Point", vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Init Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    on FINISHED → Start Flap Audio
  }

  state Patrol {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    ChaseObjectV3(gameObject=Self, target=var "Next Point", accelerationForce=10, offsetX=0, offsetY=0, speedMax=5, speedMin=0)
    GetDistance(gameObject=Self, target=var "Next Point", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on NEXT → Decel
  }

  state Next Point {
    GameObjectCompare(gameObjectVariable=var "Next Point", compareTo=var "Patrol Point", equalEvent=→"START", notEqualEvent=→"PATROL", storeResult=(unset), everyFrame=false)
    on PATROL → To Patrol
    on START → To Start
  }

  state To Patrol {
    SetGameObject(variable=var "Next Point", gameObject=var "Patrol Point", everyFrame=false)
    on FINISHED → Patrol
  }

  state To Start {
    SetGameObject(variable=var "Next Point", gameObject=var "Start Point", everyFrame=false)
    on FINISHED → Patrol
  }

  state Decel {
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Next Point
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"FINISHED", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Choose Next Point
  }

  state Choose Next Point {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    GetYDistance(gameObject=Self, target=var "Patrol Point", storeResult=var "Patrol Distance", everyFrame=false, allowNegatives=false)
    GetYDistance(gameObject=Self, target=var "Start Point", storeResult=var "Start Distance", everyFrame=false, allowNegatives=false)
    FloatCompare(float1=var "Start Distance", float2=var "Patrol Distance", tolerance=0, equal=→"START", lessThan=→"PATROL", greaterThan=→"START", everyFrame=false)
    on START → To Start
    on PATROL → To Patrol
  }

  state Die {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → Check Distance
  }

  state Check Distance {
    GetYDistance(gameObject=Self, target=var "Hero", storeResult=var "Y Distance", everyFrame=true, allowNegatives=false)
    FloatCompare(float1=var "Y Distance", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"NEXT", everyFrame=true)
    on NEXT → Fly In
  }

  state Fly In {
    PlayAudioEventRandom(audioClips=[coral_flyer_in_air_dash_1 (sfxstatic_assets_areacoral.bundle), coral_flyer_in_air_dash_2 (sfxstatic_assets_areacoral.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayRandomAudioClipTable(Table=enemy_fly_in (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))  // disabled
    SetMeshRenderer(gameObject=Self, active=true)
    SetPositionToObject2D(gameObject=Self, targetObject=var "Fly In Point", xOffset=0, yOffset=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=-10, space=0, everyFrame=false, lateUpdate=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Init Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.5)
    AnimatePositionTo(gameObject=Self, toValue=var "Start Pos", localSpace=false, time=1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    on FINISHED → Activate
  }

  state Activate {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=1, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    AddHP(target=Self, amount=0, healToMax=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    on FINISHED → Start Flap Audio
  }

  state Hazard Reset {
    SetScale(gameObject=Self, vector=(unset), x=var "Init Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetPositionToObject2D(gameObject=Self, targetObject=var "Start Point", xOffset=0, yOffset=0, everyFrame=false)
    on FINISHED → Activate
  }

  state Start Flap Audio {
    RandomFloat(min=0.9, max=1.1, storeResult=var "Pitch")
    RandomiseAudioPosition(gameObject=Self, timeMin=0, timeMax=3.5)
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    on FINISHED → Patrol
  }

  var Distance: float = 0
  var Patrol Distance: float = 0
  var Start Distance: float = 0
  var X Scale: float = 0
  var Z Pos: float = 0
  var Y Distance: float = 0
  var Init Scale: float = 0
  var Pitch: float = 0
  var Start Pos: vector3 = (0, 0, 0)
  var Next Point: gameObject = <null>
  var Patrol Point: gameObject = Spike Lazy Flyer (3)/Patrol Point
  var Start Point: gameObject = Spike Lazy Flyer (3)/Start Point
  var Attack Collider: gameObject = <null>
  var Sprite: gameObject = <null>
  var Attack Range: gameObject = <null>
  var Fly In Point: gameObject = Spike Lazy Flyer (3)/Fly In Point
}
