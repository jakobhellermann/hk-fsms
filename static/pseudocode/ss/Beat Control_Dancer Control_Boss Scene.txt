# Beat Control
# 1 location(s):
#   Clover_10 / Boss Scene/Dancer Control

fsm Beat Control {
  start Init
  on STOP → Init  // from any state

  state Pause {
    Wait(time=var "Beat Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Beat
  }

  state Beat {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BEAT", delay=0, everyFrame=false)
    SendEventToRegisterV2(EventName="BEAT", ExcludeTarget=Self)
    on FINISHED → Pause
  }

  state Init {
    on BEGIN → Beat
  }

  var Beat Time: float = 0.7
}
