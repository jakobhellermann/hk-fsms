# UI Inventory
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Inventory/Inv
#   Menu_Title / _GameCameras/HudCamera/In-game/Inventory/Inv

fsm UI Inventory {
  start Init
  on PANE RESET → Inactive  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Text Completion", storeResult=var "Completion Text")
    FindChild(gameObject=var "Completion Text", childName="Percentage", storeResult=var "Completion Percentage")
    on FINISHED → Completion Rate
  }

  state Inactive {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UI INACTIVE", delay=0, everyFrame=false)
    ActivateInventoryPaneInput(Target=Self, Activate=false)
    on ACTIVATE → Activate
  }

  state Activate {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UI ACTIVE", delay=0, everyFrame=false)
    ActivateInventoryPaneInput(Target=Self, Activate=true)
  }

  state Completion Rate {
    PlayerDataBoolTest(boolName="ConstructedFarsight", isTrue=(none), isFalse=→"FINISHED")
    ActivateGameObject(gameObject=var "Completion Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPlayerDataFloat(gameObject=var "GameManager", floatName="completionPercentage", storeValue=var "Completion percentage Float")
    ConvertFloatToString(floatVariable=var "Completion percentage Float", stringVariable=var "Completion Percentage Str", format="", everyFrame=false)
    BuildString(stringParts=[var "Completion Percentage Str", "%"], separator="", addToEnd=true, storeResult=var "Completion Percentage Str", everyFrame=false)
    SetTextMeshProText(gameObject=var "Completion Percentage", textString=var "Completion Percentage Str")
    FadeGroupUp(target=var "Completion Text")
    on FINISHED → Inactive
  }

  var Completion percentage Float: float = 0
  var Completion Percentage Str: string = ""
  var Completion Percentage: gameObject = <null>
  var Completion Text: gameObject = <null>
  var Self: gameObject = <null>
}
