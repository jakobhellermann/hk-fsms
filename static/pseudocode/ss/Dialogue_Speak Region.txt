# Dialogue
# 1 location(s):
#   Belltown_Shrine / Black Thread States Thread Only Variant/Normal World/Boss Scene/Speak Region

fsm Dialogue {
  start Init
  on CHALLENGE START → Challenge  // from any state

  state Init {
    GetHero(storeResult=var "Hero")
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindChild(gameObject=var "Boss Scene", childName="Intro Spinner", storeResult=var "Spinner")
    FindChild(gameObject=var "Boss Scene", childName="Audio Loop Strum", storeResult=var "Audio Strum")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Challenge Region")
    SetRandomAudioClipFromTable(Table=Widow_mutter_loop (sfxdynamic_assets_areabell.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    RandomiseAudioPosition(gameObject=var "Audio Loop Voice", timeMin=0, timeMax=0)
    on FINISHED → Idle
  }

  state Idle {
    on INTERACT → Turn To Back
  }

  state End {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateGameObject(gameObject=var "Challenge Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Boss Scene"), sendEvent="SPOKEN", delay=0, everyFrame=false)
  }

  state Dlg 1 {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    RunDialogue(Sheet="Belltown", Key="SPINNER_INTRO", OverrideContinue=false, PlayerVoiceTableOverride=Retort Hornet Voice (herodynamic_assets_all.bundle), PreventHeroAnimation=true, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Dlg 1 End
  }

  state Look {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    Tk2dPlayAnimation(gameObject=var "Spinner", animLibName="", clipName="Intro Look")
    Wait(time=1.8, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    FadeAudio(gameObject=var "Spinner", startVolume=1, endVolume=0, time=0.25)
    AudioStopV2(gameObject=var "Audio Strum", fadeTime=0.25, cancelOnEarlyExit=true)
    on FINISHED → Challenge End
  }

  state Challenge {
    SetBoolValue(boolVariable=var "Challenged", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Challenge Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="TENSION STOP")
    on FINISHED → Look
  }

  state Challenge End {
    BoolTest(boolVariable=var "Challenged", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Taunt Back End")
    on FINISHED → End
  }

  state Dlg Look {
    AudioStopV2(gameObject=var "Audio Strum", fadeTime=0, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Spinner", fadeTime=0, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=trobbio_recover_or_general_foley_movement (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=var "Spinner", animLibName="", clipName="Intro Look")
    Wait(time=1.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dlg 2
  }

  state Dlg 2 {
    RunDialogue(Sheet="Belltown", Key="SPINNER_INTRO_2", OverrideContinue=false, PlayerVoiceTableOverride=Retort Hornet Voice (herodynamic_assets_all.bundle), PreventHeroAnimation=true, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Dlg 2 End
  }

  state Turn From Back {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Turn Back Three Quarter EndToIdle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    PlayAudioEvent(audioClip=trobbio_jump_short (sfxstatic_assets_trobbio.bundle), pitchMin=1.15, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Regain Control {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → End
  }

  state Turn To Back {
    PlayAudioEvent(audioClip=trobbio_jump_short (sfxstatic_assets_trobbio.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Turn Back Three Quarter")
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="TENSION STOP")
    SendEventToRegister(eventName="CHALLENGE DISABLE")
    on FINISHED → Dlg 1
  }

  state Dlg 1 End {
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=Self, UseChildren=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dlg Look
  }

  state Dlg 2 End {
    EndDialogue(ReturnControl=false, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Turn From Back
  }

  var Challenged: bool = false
  var Boss Scene: gameObject = <null>
  var Hero: gameObject = <null>
  var Spinner: gameObject = <null>
  var Challenge Region: gameObject = <null>
  var Audio Strum: gameObject = <null>
  var Audio Loop Voice: gameObject = Black Thread States Thread Only Variant/Normal World/Boss Scene/Speak Region/Audio Loop Voice
}
