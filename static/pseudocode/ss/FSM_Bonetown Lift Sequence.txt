# FSM
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/Bonetown Lift Sequence

fsm FSM {
  start Wait for lever

  state Wait for lever {
    SetAnimator(target=Self, active=false)
    on OPEN → Unlock Sequence
    on ACTIVATE → Activate
  }

  state Activate {
    SendEventByNameV2(eventTarget=GameObjectFSM(var "Lift", fsm="Lift Control"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    SetAnimator(target=Self, active=true)
    AnimatorPlay(gameObject=Self, stateName="Unlocked", layer=(unset), normalizedTime=(unset), everyFrame=false)
  }

  state Open {
    SetAnimator(target=Self, active=false)
    SendEventByNameV2(eventTarget=GameObjectFSM(var "Lift", fsm="Lift Control"), sendEvent="OPEN", delay=0, everyFrame=false)
  }

  state Unlock Sequence {
    SetAnimator(target=Self, active=true)
    PlayAudioEvent(audioClip=bonetown_lift_lock_open (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatorPlayStateWait(target=Self, stateName="Unlock", finishEvent=→"FINISHED")
    on FINISHED → Open
  }

  var Cogs: gameObject = Black Thread States/Normal World/Bonetown Lift Sequence/Bone Lift Cogs
  var Lift: gameObject = Black Thread States/Normal World/Bonetown Lift Sequence/Bonetown Lift
}
