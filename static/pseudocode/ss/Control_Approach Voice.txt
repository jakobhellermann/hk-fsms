# Control
# 1 location(s):
#   Bellway_Centipede_Arena / Boss Scene/Approach Voice

fsm Control {
  start Init
  on ARENA START → End  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Voice Point")
    FindNamedChild(gameObject=Self, storeResult=var "Voice Range")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Voice Range", trigger=0, collideTag="", sendEvent=→"SHIFT", storeCollider=var "")
    on SHIFT → Init Pause
  }

  state Init Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(gameObject=var "Voice Range", trigger=2, collideTag="", sendEvent=→"SHIFT", storeCollider=var "")
    on FINISHED → Call
    on SHIFT → Idle
  }

  state Call {
    AudioPlayRandomVoice(gameObject=var "Voice Point", audioClips=[bell_eater_scream_another_1 (sfxstatic_assets_areabellway.bundle), bell_eater_scream_another_2 (sfxstatic_assets_areabellway.bundle), bell_eater_scream_another_3 (sfxstatic_assets_areabellway.bundle)], weights=[1, 1, 1], pitchMin=1, pitchMax=1, stopPreviousSound=false)
    Trigger2dEvent(gameObject=var "Voice Range", trigger=2, collideTag="", sendEvent=→"SHIFT", storeCollider=var "")
    WaitRandom(timeMin=4, timeMax=6, finishEvent=→"FINISHED", realTime=false)
    on SHIFT → Idle 2
    on FINISHED → Call
  }

  state End {
  }

  state Idle 2 {
    Trigger2dEvent(gameObject=var "Voice Range", trigger=0, collideTag="", sendEvent=→"SHIFT", storeCollider=var "")
    on SHIFT → Init Pause 2
  }

  state Init Pause 2 {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(gameObject=var "Voice Range", trigger=2, collideTag="", sendEvent=→"SHIFT", storeCollider=var "")
    on FINISHED → Call
    on SHIFT → Idle 2
  }

  var Voice Point: gameObject = <null>
  var Voice Range: gameObject = <null>
}
