# Control
# 1 location(s):
#   Tut_02 / One Way VineWall

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on ACTIVATED → Activated
    on FINISHED → Idle
  }

  state Activated {
    SetParent(gameObject=var "Stumps", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Stumps", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Nail Attack", sendEvent=→"NAIL HIT", storeCollider=var "")  // disabled
    SetParticleEmission(gameObject=var "Pt Bits", emission=false, resetOnExit=false)
    SetParticleEmission(gameObject=var "Pt Spatter", emission=false, resetOnExit=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"NAIL HIT", sendEventHeavy=→"DESTROY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on NAIL HIT → Strike
    on DESTROY → Destroy Check
  }

  state Strike {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Hero Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    SendMessage(gameObject=var "Wall", delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    SetParticleEmission(gameObject=var "Pt Bits", emission=true, resetOnExit=false)
    SetParticleEmission(gameObject=var "Pt Spatter", emission=true, resetOnExit=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=vine_plat_cut (sfxstatic_assets_areamoss.bundle), pitchMin=0.9, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=breakable_wall_hit_1 (sfxstatic_assets_shared.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=enemy_damage_old (sfxstatic_assets_areabellshared.bundle), pitchMin=0.9, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=grass_cut_2 (sfxstatic_assets_grass.bundle), pitchMin=0.9, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    IntAdd(intVariable=var "Hits", add=1, everyFrame=false)
    IntSwitch(intVariable=var "Hits", compareTo=[1, 2, 3, 4], sendEvent=[→"1", →"2", →"3", →"4"], everyFrame=false)
    on 1 → Hit 1
    on 2 → Hit 1
    on 3 → 
    on 4 → Hit 4
    on FINISHED → Hit 4
    on CANCEL → Idle
  }

  state Recover {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Hit 1 {
    Tk2dPlayAnimation(gameObject=var "Vine 1", animLibName="", clipName="Thick Hit")
    Tk2dPlayAnimation(gameObject=var "Vine 2", animLibName="", clipName="Standard Hit")
    on FINISHED → Recover
  }

  state Hit 4 {
    ActivateGameObject(gameObject=var "Vine 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vine 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Destroy Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Break Wall Effect (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Destroy Effects", position=(unset), rotation=(unset), storeObject=var "")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=breakable_wall_death (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetRemaskerExited(Target=var "Solid", WasDisabled=true, Recursive=true)
    ActivateGameObject(gameObject=var "Solid", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Stumps", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=vine_plat_land (sfxstatic_assets_areamoss.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=breakable_wall_death (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → End Pt
  }

  state End Pt {
    SetParticleEmission(gameObject=var "Pt Bits", emission=false, resetOnExit=false)
    SetParticleEmission(gameObject=var "Pt Spatter", emission=false, resetOnExit=false)
  }

  state Destroy Check {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Hero Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    on FINISHED → Hit 4
    on CANCEL → Idle
  }

  var Hits: int = 0
  var Activated: bool = false
  var Destroy Effects: gameObject = One Way VineWall/Destroy Effects
  var Pt Bits: gameObject = One Way VineWall/Pt Bits
  var Pt Spatter: gameObject = One Way VineWall/Pt Spatter
  var Self: gameObject = One Way VineWall
  var Solid: gameObject = One Way VineWall/Solid
  var Stumps: gameObject = One Way VineWall/Stumps
  var Vine 1: gameObject = One Way VineWall/Vine 1
  var Vine 2: gameObject = One Way VineWall/Vine 2
  var Wall: gameObject = One Way VineWall/Wall
}
