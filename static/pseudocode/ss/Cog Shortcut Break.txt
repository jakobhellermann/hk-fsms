# Cog Shortcut Break
# 1 location(s):
#   Under_05 / cog_05_shortcut

fsm Cog Shortcut Break {
  start Pause Frame

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Broken
  }

  state Broken {
    BoolTest(boolVariable=var "Activated", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Already Broken
    on FALSE → Not Broken
  }

  state Not Broken {
    ActivateGameObject(gameObject=var "Before", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "After", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BREAK → Start Break
  }

  state Already Broken {
    ActivateGameObject(gameObject=var "Before", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "After", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Start Break {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=var "")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendMessageV2(gameObject=var "Blocking Cogs", delivery=BroadcastMessage, options=1, functionCall=StartJitter(), everyFrame=false)
    SendMessageV2(gameObject=var "Blocking Cogs", delivery=BroadcastMessage, options=1, functionCall=StopAtCurrentPoint(), everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Fall 1
  }

  state Fall 1 {
    iTweenMoveBy(gameObject=var "Blocking Cogs", id="", vector=(0, -0.3, 0), time=0.3, delay=0, speed=(unset), easeType=6, loopType=0, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Impact 1
  }

  state Impact 1 {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Large Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    PlayParticleSystemPool(Target=var "Impact Effect Pool")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fall 2
  }

  state Fall 2 {
    iTweenMoveBy(gameObject=var "Blocking Cogs", id="", vector=(0, -0.5, 0), time=0.3, delay=0, speed=(unset), easeType=6, loopType=0, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Impact 2
  }

  state Impact 2 {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Large Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    PlayParticleSystemPool(Target=var "Impact Effect Pool")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Break {
    SendEventToRegister(eventName="WAKE TO IDLE")
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    ActivateGameObject(gameObject=var "Before", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "After", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var After: gameObject = cog_05_shortcut/after
  var Before: gameObject = cog_05_shortcut/before
  var Blocking Cogs: gameObject = cog_05_shortcut/before/blocking cogs
  var Break Effects: gameObject = cog_05_shortcut/break effects
  var Impact Effect Pool: gameObject = cog_05_shortcut/before/blocking cogs/impact effect pool
}
