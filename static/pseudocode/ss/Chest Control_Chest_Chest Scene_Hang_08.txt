# Chest Control
# 2 location(s):
#   Hang_08 / Chest Scene/Chest
#   Song_24 / Black Thread States/Normal World/Enemy Control/Song Fencer/Chest

// uses template: chest_control
fsm Chest Control {
  start Pause Frame

  state Pause Frame {
    FindChild(gameObject=Self, childName="Bouncer", storeResult=var "Bouncer")
    FindChild(gameObject=Self, childName="Item", storeResult=var "Item Holder")
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Opened", storeResult=var "Opened")
    FindChild(gameObject=Self, childName="Idle Glow", storeResult=var "Idle Glow")
    FindChild(gameObject=Self, childName="Open Effects", storeResult=var "Open Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Active While Closed")
    ActivateAllChildren(gameObject=var "Item Holder", activate=false)
    PreBuildTK2DSprites(gameObject=var "Self", useChildren=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activated
    on FINISHED → Create Pool
  }

  state Activated {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    StopParticleEmitter(gameObject=var "Idle Glow")
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Active While Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bouncer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    ActivateGameObject(gameObject=var "Active While Closed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"NAIL HIT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="Nail Attack", sendEvent=→"NAIL HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on NAIL HIT → Range?
    on QUICK OPEN → Quick Open
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=var "Hit Sound", pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Active While Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmitter(gameObject=var "Idle Glow")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Open", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=5)
    on FINISHED → Spawn Items
  }

  state Spawn Items {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=var "Open Sound", pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetLayer(gameObject=Self, layer=Hero Detector(13))
    ActivateGameObject(gameObject=var "Open Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Item Holder", activate=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlingObjectsFromGlobalPoolV3(GameObjects=[Geo Small (globalpoolprefabs_assets_all.bundle)], SpawnPoint=var "Self", Position=(0.25, 0.25, 0), SpawnMin=var "Geo Small", SpawnMax=var "Geo Small", SpeedMin=25, SpeedMax=38, AngleMin=78, AngleMax=102, OriginVariationX=0, OriginVariationY=0, spreadFrames=4)
    FlingObjectsFromGlobalPoolV3(GameObjects=[Geo Med (globalpoolprefabs_assets_all.bundle)], SpawnPoint=var "Self", Position=(0.25, 0.25, 0), SpawnMin=var "Geo Med", SpawnMax=var "Geo Med", SpeedMin=25, SpeedMax=38, AngleMin=78, AngleMax=102, OriginVariationX=0, OriginVariationY=0, spreadFrames=4)
    FlingObjectsFromGlobalPoolV3(GameObjects=[Shell Shard 01 (globalpoolprefabs_assets_all.bundle), Shell Shard 02 (globalpoolprefabs_assets_all.bundle), Shell Shard 03 (globalpoolprefabs_assets_all.bundle)], SpawnPoint=var "Self", Position=(0.25, 0.25, 0), SpawnMin=var "Shards", SpawnMax=var "Shards", SpeedMin=25, SpeedMax=38, AngleMin=78, AngleMax=102, OriginVariationX=0, OriginVariationY=0, spreadFrames=4)
    on FINISHED → Opened
  }

  state Opened {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Opened", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bouncer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Range? {
    FindChild(gameObject=Self, childName="Hero Region", storeResult=var "Hero Region")
    GameObjectIsNull(gameObject=var "Hero Region", isNull=→"NAIL HIT", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Hero Region", storeResult=false, sendEvent="NAIL HIT", outOfRangeEvent="", everyFrame=false)
    on NAIL HIT → Open
    on FINISHED → Idle
  }

  state Quick Open {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Idle Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Activated
  }

  state Create Pool {
    AddPersonalObjectPool(Target=Self, Prefab=Geo Small (globalpoolprefabs_assets_all.bundle), Amount=var "Geo Small", SharePooledInScene=false, FinalCall=false)
    AddPersonalObjectPool(Target=Self, Prefab=Geo Med (globalpoolprefabs_assets_all.bundle), Amount=var "Geo Med", SharePooledInScene=false, FinalCall=false)
    AddPersonalObjectPool(Target=Self, Prefab=Geo Large (globalpoolprefabs_assets_all.bundle), Amount=var "Geo Large", SharePooledInScene=false, FinalCall=false)
    IntOperator(integer1=var "Shards", integer2=3, operation=Divide, storeResult=var "Shard Third", everyFrame=false)
    AddPersonalObjectPool(Target=Self, Prefab=Shell Shard 01 (globalpoolprefabs_assets_all.bundle), Amount=var "Shard Third", SharePooledInScene=false, FinalCall=false)
    AddPersonalObjectPool(Target=Self, Prefab=Shell Shard 02 (globalpoolprefabs_assets_all.bundle), Amount=var "Shard Third", SharePooledInScene=false, FinalCall=false)
    AddPersonalObjectPool(Target=Self, Prefab=Shell Shard 03 (globalpoolprefabs_assets_all.bundle), Amount=var "Shard Third", SharePooledInScene=false, FinalCall=true)
    on FINISHED → Idle
  }

  var Geo Large: int = 2
  var Geo Med: int = 8
  var Geo Small: int = 20
  var Shards: int = 0
  var Hit Sound: object = chest_open (sfxstatic_assets_shared.bundle)
  var Open Sound: object = <null>

  var Shard Third: int = 0
  var Activated: bool = false
  var Bouncer: gameObject = <null>
  var Hero Region: gameObject = <null>
  var Idle Glow: gameObject = <null>
  var Item Holder: gameObject = <null>
  var Open Effects: gameObject = <null>
  var Opened: gameObject = <null>
  var Self: gameObject = <null>
  var Active While Closed: gameObject = <null>
}
