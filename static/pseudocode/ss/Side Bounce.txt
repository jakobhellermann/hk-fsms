# Side Bounce
# 1 location(s):
#   enemycorpses_assets_areaswamp.bundle / Swamp Shaman Projectile

fsm Side Bounce {
  start Init

  state Init {
    on FINISHED → Air
  }

  state Air {
    GetVelocity2dNotZero(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=true)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on WALL R → Wall R
    on WALL L → Wall L
    on TINK UP → Tink Up
    on TINK DOWN → Tink Down
    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
  }

  state Wall R {
    Translate(gameObject=Self, vector=(unset), x=-0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FloatMultiply(floatVariable=var "X Speed", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), everyFrame=false)
    on FINISHED → Air
  }

  state Wall L {
    Translate(gameObject=Self, vector=(unset), x=0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FloatMultiply(floatVariable=var "X Speed", multiplyBy=-1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), everyFrame=false)
    on FINISHED → Air
  }

  state Tink Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=10, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tinked {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TINKED", delay=0, everyFrame=false)
    on FINISHED → Air
  }

  state Tink Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-10, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tink Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-15, y=(unset), everyFrame=false)
    on FINISHED → Tinked
  }

  state Tink Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=15, y=(unset), everyFrame=false)
    on FINISHED → Tinked
  }

  var X Speed: float = 0
}
