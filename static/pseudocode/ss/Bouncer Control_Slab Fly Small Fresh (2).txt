// uses template: bouncer_control
fsm Bouncer Control {
  start Start Pause
  on GO DOWN → Go Down  // from any state
  on GO LEFT → Go Left  // from any state
  on GO RIGHT → Go Right  // from any state
  on GO UP → Go Up  // from any state
  on ZERO HP → Death  // from any state

  state Start Pause {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="MainCamera", store=var "Camera")
    FindNamedChild(gameObject=Self, storeResult=var "Body Damager")
    FindNamedChild(gameObject=Self, storeResult=var "Spawn Pos Alt")
    FindNamedChild(gameObject=Self, storeResult=var "Egg")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    FindNamedChild(gameObject=Self, storeResult=var "Hatch Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Hatch End")
    FindNamedChild(gameObject=Self, storeResult=var "Pt HatchOut")
    FindNamedChild(gameObject=Self, storeResult=var "Drip Spawner")
    FindNamedChild(gameObject=Self, storeResult=var "Up Marker")
    FindNamedChild(gameObject=Self, storeResult=var "Terrain Box - Optional")
    BoolTest(boolVariable=var "z_MotherHatcher", isTrue=→"MOTHER HATCH", isFalse=(none), everyFrame=false)
    GameObjectIsActive(Target=var "Egg", ActiveSpace=1, trueEvent=→"EGG", falseEvent=(none), storeValue=(unset))
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Battle Check
    on EGG → In Egg
    on MOTHER HATCH → Mother Init
  }

  state Left or Right? {
    FloatSwitch(floatVariable=var "Angle", lessThan=[90, 270, 360], sendEvent=[→"RIGHT", →"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Face Left
    on RIGHT → Face Right
  }

  state Face Left {
    SetBoolValue(boolVariable=var "Facing Right", boolValue=false, everyFrame=false)
    on FINISHED → Fly 2
  }

  state Face Right {
    SetBoolValue(boolVariable=var "Facing Right", boolValue=true, everyFrame=false)
    on FINISHED → Fly 2
  }

  state Hit Up {
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Up Right
    on LEFT → Up Left
  }

  state Up Right {
    RandomFloat(min=320, max=350, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Up Left {
    RandomFloat(min=190, max=220, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Down {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    BoolTest(boolVariable=var "Facing Right", isTrue=→"RIGHT", isFalse=→"LEFT", everyFrame=false)
    on RIGHT → Down Right
    on LEFT → Up Left 2
  }

  state Down Right {
    RandomFloat(min=10, max=40, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Up Left 2 {
    RandomFloat(min=140, max=170, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Right {
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Down Right 2
    on DOWN → Right Down
  }

  state Right Down {
    RandomFloat(min=190, max=220, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Down Right 2 {
    RandomFloat(min=140, max=170, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Hit Left {
    FloatSwitch(floatVariable=var "Angle", lessThan=[180, 360], sendEvent=[→"UP", →"DOWN"], everyFrame=false)
    on UP → Left Up
    on DOWN → Left Down
  }

  state Left Down {
    RandomFloat(min=320, max=350, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Left Up {
    RandomFloat(min=10, max=40, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Fly 2 {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    SetVelocityAsAngle(gameObject=var "Self", angle=var "Angle", speed=var "Speed", everyFrame=true)
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=(none), storeCollider=var "", storeForce=(unset))  // disabled
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=25, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=25, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"BONK UP", rightHitEvent=→"BONK RIGHT", bottomHitEvent=→"BONK DOWN", leftHitEvent=→"BONK LEFT", otherLayer=true, otherLayerNumber=24, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=var "Terrain Box - Optional", topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"BONK DOWN", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed Crt", everyFrame=true)
    FloatCompare(float1=var "Speed Crt", float2=0, tolerance=0.1, equal=→"BONK DOWN", lessThan=(none), greaterThan=(none), everyFrame=true)
    on BONK DOWN → Hit Down
    on BONK LEFT → Hit Left
    on BONK RIGHT → Hit Right
    on BONK UP → Hit Up
    on SING → Sing
  }

  state Aim {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    BoolTest(boolVariable=var "Start Up", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle", floatValue=90, everyFrame=false)
    on FINISHED → Left or Right?
  }

  state Go Up {
    RandomFloat(min=20, max=160, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Go Right {
    RandomFloat(min=-70, max=70, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Go Left {
    RandomFloat(min=110, max=250, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state Go Down {
    RandomFloat(min=200, max=340, storeResult=var "Angle")
    on FINISHED → Left or Right?
  }

  state State {
    GetCurrentTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Sleep", equalEvent=→"SLEEP", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Aim
    on SLEEP → Sleep
  }

  state Sleep {
    FindNamedChild(gameObject=Self, storeResult=var "Wake Range")
    ActivateGameObject(gameObject=var "Wake Range", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"WAKE", ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on WAKE → Wake
    on TOOK DAMAGE → Wake
  }

  state Wake {
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=SFS_alert (sfxdynamic_assets_areaslab.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    ActivateGameObject(gameObject=var "Wake Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    RandomFloat(min=15, max=20, storeResult=var "Wake Speed")
    FireAtTarget(gameObject=Self, target=var "Up Marker", speed=var "Wake Speed", position=(0, 0, 0), spread=0, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.88, brakeOnExit=true)
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=7)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    PlayAudioEvent(audioClip=slab_fly_small_fly_notice_hero (sfxstatic_assets_areaslab.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fly Start
  }

  state Fly Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    on FINISHED → Fly Dir
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=→"FINISHED", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=10, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=SFS_sing (sfxdynamic_assets_areaslab.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on FINISHED → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fly Start
  }

  state Cam Check {
    GetDistance(gameObject=var "Self", target=var "Camera", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=44, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → State
  }

  state Fly In Ready {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    ActivateGameObject(gameObject=var "Body Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BATTLE START → Check Pos
    on WAKE → Fly In
  }

  state Check Pos {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=8, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=false)
    on SHIFT → Shift
    on FINISHED → Fly In Pause
  }

  state Shift {
    SetParent(gameObject=var "Spawn Pos Alt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Spawn Pos Alt", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    on FINISHED → Fly In Pause
  }

  state Fly In {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=SFS_flyin (sfxdynamic_assets_areaslab.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayRandomAudioClipTable(Table=enemy_fly_in (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Translate(gameObject=Self, vector=(unset), x=(unset), y=10, z=-15, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetMeshRenderer(gameObject=Self, active=true)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -10, 15), time=0.75, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.55)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    ActivateGameObject(gameObject=var "Body Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    on FINISHED → Aim
  }

  state Battle Check {
    GameObjectIsActive(Target=var "Egg", ActiveSpace=1, trueEvent=→"EGG", falseEvent=(none), storeValue=(unset))
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Fly In", equalEvent=→"FLY IN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Clip", compareTo="Rest", equalEvent=→"REST", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Cam Check
    on FLY IN → Fly In Ready
    on EGG → In Egg
    on REST → Rest
  }

  state In Egg {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Body Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Drip Spawner", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0.4705882, 0.4705882, 0.4705882, 1), everyframe=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    on BATTLE START → Hatch Pause
  }

  state Hatch Pause {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hatch Antic
  }

  state Hatch Antic {
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=30)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=slab_fly_small_fresh_appear_burst (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=slab_fly_mid_spit_antic (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Hatch
  }

  state Hatch {
    ActivateGameObject(gameObject=var "Egg", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=SFS_alert (sfxdynamic_assets_areaslab.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)

    ActivateGameObject(gameObject=var "Hatch Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    RandomlyFlipScale(gameObject=Self)
    SetMeshRenderer(gameObject=Self, active=true)
    PlayAudioEvent(audioClip=slab_fly_small_fresh_appear_burst (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=lifeblood_burst (sfxstatic_assets_lifeblood.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fling Out
  }

  state Fling Out {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hatch")
    GetPosition(gameObject=Self, vector=var "Tween Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    RandomFloat(min=9, max=11, storeResult=var "Tween Y")
    RandomFloat(min=0.006, max=0.0069, storeResult=var "Tween Z")
    SetVector3XYZ(vector3Variable=var "Tween Pos", vector3Value=(unset), x=(unset), y=var "Tween Y", z=var "Tween Z", everyFrame=false)
    AnimatePositionTo(gameObject=Self, toValue=var "Tween Pos", localSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.5)
    PlayParticleEmitterInState(gameObject=var "Pt HatchOut")
    on FINISHED → Hatch End
  }

  state Hatch End {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    ActivateGameObject(gameObject=var "Body Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Hatch End", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Drip Spawner", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    PlayAudioEvent(audioClip=slab_fly_small_fresh_appear_burst (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fly
  }

  state Fly {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    on FINISHED → Activate
  }

  state Fly In Pause {
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fly In
  }

  state Ready for Mother {
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=-100, y=10, z=(unset), space=0, everyFrame=false, lateUpdate=false)

    AddHP(target=Self, amount=(unset), healToMax=true)
    on MOTHER HATCH → Hatch Out
  }

  state Hatch Out {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hatch", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.89, brakeOnExit=true)
    on FINISHED → Hatch Fly
  }

  state Mother Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetSpecialDeath(target=Self, hasSpecialDeath=true)
    on FINISHED → Ready for Mother
  }

  state Death {
    BoolTest(boolVariable=var "z_MotherHatcher", isTrue=→"MOTHER HATCH", isFalse=(none), everyFrame=false)
    on MOTHER HATCH → Simulate Death
  }

  state Simulate Death {
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    on FINISHED → Ready for Mother
  }

  state Hatch Fly {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    ActivateGameObject(gameObject=var "Body Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")

  }

  state Rest {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=true, pauseBetweenTurns=0.5)  // disabled

    RandomFloat(min=140, max=220, storeResult=var "Angle")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"FINISHED", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)

  }

  var Speed: float = 8.5
  var Start Up: bool = false
  var z_MotherHatcher: bool = true

  var Angle: float = 0
  var Distance: float = 0
  var Wake Speed: float = 0
  var Tween Y: float = 0
  var Tween Z: float = 0
  var Speed Crt: float = 0
  var Facing Right: bool = false
  var Sleep: string = ""
  var Clip: string = ""
  var Tween Pos: vector3 = (0, 0, 0)
  var Camera: gameObject = <null>
  var Self: gameObject = <null>
  var Wake Range: gameObject = <null>
  var Spawn Pos Alt: gameObject = <null>
