// uses template: coral_tower_intro_roar
fsm coral_tower_intro_roar {
  start Idle

  state Idle {
    GetGrandParent(gameObject=Self, storeResult=var "Battle Scene")
    GetParent(gameObject=var "Battle Scene", storeResult=var "Battle Scenes Group")
    FindNamedChild(gameObject=var "Battle Scenes Group", storeResult=var "Roar Audio")
    on BATTLE START → Wait
  }

  state Roar {
    ActivateGameObject(gameObject=var "Roar Audio", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Roar Audio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Roar Audio", audioClipTable=CK_roar_short (sfxdynamic_assets_areacoralareamemory.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    StartRoarEmitter(spawnPoint=Self, delay=0, stunHero=true, roarBurst=false, isSmall=false, noVisualEffect=false, forceThroughBind=false, stopOnExit=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=misc_rumble_impact (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=2.5)
    on FINISHED → Start Waves
  }

  state Start Waves {
    ApplyMusicCue(musicCue=Coral Tower Battle (audiocuesdynamic_assets_areacoralareamemory.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.1)
    SendMessage(gameObject=var "Battle Scene", delivery=SendMessage, options=1, functionCall=WaveEnd())
    DestroySelf(detachChildren=false)
  }

  state Wait {
    Wait(time=1.25, finishEvent=(none), realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0.5)
    on FINISHED → Roar
  }

  var Battle Scene: gameObject = <null>
  var Battle Scenes Group: gameObject = <null>
  var Roar Audio: gameObject = <null>
}
