# Control
# 1 location(s):
#   Mosstown_02 / lore_tablet/moss_bone_plaque

fsm Control {
  start Pause

  state Init {
    ActivateGameObject(gameObject=var "Inert Sign", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fall Sign", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fallen Sign Return", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activate
    on ACTIVATED → Activated
  }

  state Activated {
    SetCollider(gameObject=var "Inspect Region", active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Inert Sign", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fallen Sign Return", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Activate {
    SetCollider(gameObject=var "Inspect Region", active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Inert Sign", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fall Sign", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=var "Fall Sign", vector=(10, 20), x=(unset), y=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var Activated: bool = false
  var Inspect Region: gameObject = lore_tablet/moss_bone_plaque/Inspect Region
  var Inert Sign: gameObject = lore_tablet/moss_bone_plaque/Inert Sign
  var Fall Sign: gameObject = lore_tablet/moss_bone_plaque/Fall Sign
  var Fallen Sign Return: gameObject = lore_tablet/moss_bone_plaque/Fallen Sign Return
}
