fsm Control {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "Enemy")
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "Enemy")
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SendEventByName(eventTarget=GameObject(var "Enemy"), sendEvent="SOLID LAND", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Exit {
    SendEventByName(eventTarget=GameObject(var "Enemy"), sendEvent="UNSOLID LAND", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Enemy: gameObject = <null>
}
