# Purr Audio
# 1 location(s):
#   Slab_06 / Bellway_pieces/Bone Beast NPC

fsm Purr Audio {
  start State 1
  on ACTIVE → Choose Audio  // from any state
  on INACTIVE → State 1  // from any state

  state Play Audio {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=var "Audio Clip", WaitForEndOfClip=true, finishedEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Choose Audio {
    SendRandomEventFair(Events=[{SendEvent=→"1", Probability=1}, {SendEvent=→"2", Probability=1}, {SendEvent=→"3", Probability=1}], TrackingArray=var "Audio Fair Tracking Array", MissedMultiplier=2)
    on 1 → Set Audio 1
    on 2 → Set Audio 2
    on 3 → Set Audio 3
  }

  state Set Audio 1 {
    SetAudioClipVariable(audioClipVariable=var "Audio Clip", audioClip=bellbeast_purr_1 (sfxstatic_assets_areabellway.bundle))
    VibrationRegionPlayOneShot(target=var "Vibration Region", vibrationDataAsset=bellbeast_purr_1 (vibrationstatic_assets_vibrationdataasset.bundle), requireInside=false, loop=false, isRealTime=false, tag="", stopOnRegionExit=false, stopOnStateExit=false)
    on FINISHED → Play Audio
  }

  state Set Audio 2 {
    SetAudioClipVariable(audioClipVariable=var "Audio Clip", audioClip=bellbeast_purr_2 (sfxstatic_assets_areabellway.bundle))
    VibrationRegionPlayOneShot(target=var "Vibration Region", vibrationDataAsset=bellbeast_purr_2 (vibrationstatic_assets_vibrationdataasset.bundle), requireInside=false, loop=false, isRealTime=false, tag="", stopOnRegionExit=false, stopOnStateExit=false)
    on FINISHED → Play Audio
  }

  state Set Audio 3 {
    SetAudioClipVariable(audioClipVariable=var "Audio Clip", audioClip=bellbeast_purr_3 (sfxstatic_assets_areabellway.bundle))
    VibrationRegionPlayOneShot(target=var "Vibration Region", vibrationDataAsset=bellbeast_purr_3 (vibrationstatic_assets_vibrationdataasset.bundle), requireInside=false, loop=false, isRealTime=false, tag="", stopOnRegionExit=false, stopOnStateExit=false)
    on FINISHED → Play Audio
  }

  state State 1 {
    AudioStop(gameObject=Self, fadeTime=0)
  }

  state Wait {
    WaitRandom(timeMin=0, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choose Audio
  }

  var Vibration Region: gameObject = Bellway_pieces/Bone Beast NPC/Hero Vibration Region Bone Beast
  var Audio Clip: object = <null>
  var Audio Fair Tracking Array: array = var "Audio Fair Tracking Array"
}
