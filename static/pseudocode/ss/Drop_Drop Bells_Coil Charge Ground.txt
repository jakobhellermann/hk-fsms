# Drop
# 1 location(s):
#   Bellway_Centipede_Arena / Boss Scene/Coil Charge Ground/Drop Bells

fsm Drop {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "DropBell 1")
    FindNamedChild(gameObject=Self, storeResult=var "DropBell 2")
    GetPosition(gameObject=var "DropBell 1", vector=(unset), x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on DROP → Drop
  }

  state Drop {
    SendEventByName(eventTarget=GameObject(var "DropBell 1"), sendEvent="DROP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "DropBell 2"), sendEvent="DROP", delay=0.5, everyFrame=false)
    on FINISHED → Idle
  }

  var DropBell 1: gameObject = <null>
  var DropBell 2: gameObject = <null>
}
