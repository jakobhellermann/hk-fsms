# Heart Container UI
# 1 location(s):
#   heartpieceupgrade_assets_all.bundle / Heart Container UI

fsm Heart Container UI {
  start Init
  on DISABLE → State 1  // from any state

  state Init {
    BoolTest(boolVariable=var "Init", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    FindChild(gameObject=Self, childName="Blanker", storeResult=var "Blanker")
    FindChild(gameObject=Self, childName="Fleur", storeResult=var "Fleur")
    FindChild(gameObject=Self, childName="Piece", storeResult=var "Piece")
    FindChild(gameObject=Self, childName="Mover", storeResult=var "Mover")
    FindChild(gameObject=Self, childName="Mover Trail", storeResult=var "Mover Trail")
    FindChild(gameObject=Self, childName="Get Effect", storeResult=var "Get Effect")
    FindChild(gameObject=Self, childName="Get Glow", storeResult=var "Get Glow")
    ActivateGameObject(gameObject=var "Get Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Init", boolValue=true, everyFrame=false)
    on FINISHED → Wait Frame
  }

  state Darken {
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.5176471), colorVariable=var "Blanker Colour", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Blanker", color=var "Blanker Colour", everyframe=true)
    on FINISHED → Fleur
  }

  state Fleur {
    SetMeshRenderer(gameObject=var "Fleur", active=true)
    Tk2dPlayAnimation(gameObject=var "Fleur", animLibName="", clipName="Fleur Appear")
    GetPlayerDataVariable(VariableName="heartPieces", StoreValue=var "Pieces")
    IntSwitch(intVariable=var "Pieces", compareTo=[0, 1, 2, 3], sendEvent=[→"ONE", →"TWO", →"THREE", →"FOUR"], everyFrame=false)
    on ONE → Get One
    on TWO → Get Two
    on THREE → Get Three
    on FOUR → Get Four
  }

  state Get One {
    SetStringValue(stringVariable=var "Get Anim", stringValue="Get 1", everyFrame=false)
    SetStringValue(stringVariable=var "Piece Land Particles", stringValue="Piece Land 1", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Get 0")
    on FINISHED → Piece Fade
  }

  state Get Anim {
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName=var "Get Anim")
    Tk2dPlayAnimationWithEvents(gameObject=var "Piece", clipName=var "Get Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Check Max 2
  }

  state Fade {
    Tk2dPlayAnimation(gameObject=var "Fleur", animLibName="", clipName="Fleur Disappear")  // disabled
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Piece Colour", time=0.5, speed=(unset), delay=0.5, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=var "Piece Colour", everyframe=true)
    Tk2dSpriteSetColor(gameObject=var "Fleur", color=var "Piece Colour", everyframe=true)
    EaseColor(fromValue=(1, 1, 1, 0.5176471), toValue=(1, 1, 1, 0), colorVariable=var "Blanker Colour", time=0.5, speed=(unset), delay=0.5, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Blanker", color=var "Blanker Colour", everyframe=true)
    Wait(time=1, finishEvent=→"WAIT", realTime=false)
    on WAIT → Check at board
  }

  state Destroy Self {
    PlayAudioEvent(audioClip=hornet_sprint_jump_2nd_spin (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=0.8, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=AffectedByGravity(true))
    SetPlayerDataBool(boolName="isInvincible", value=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    SendEventToRegister(eventName="HEART PIECE COLLECTED")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    AwardQueuedAchievements(delay=1)
    CheckIsInQuestBoardDepositSequence(trueEvent=→"BOARD", falseEvent=(none), storeValue=(unset))
    DoQueuedSaveGame()
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=20, everyFrame=false)  // disabled
    DestroySelf(detachChildren=false)
    on BOARD → Wait For Grounded 2
  }

  state Piece Fade {
    SetMeshRenderer(gameObject=var "Piece", active=true)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Piece Colour", time=1, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=var "Piece Colour", everyframe=true)
    Wait(time=1.8, finishEvent=→"WAIT", realTime=false)
    on WAIT → Get Anim
  }

  state Get Two {
    SetStringValue(stringVariable=var "Get Anim", stringValue="Get 2", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Got 1")
    SetStringValue(stringVariable=var "Piece Land Particles", stringValue="Piece Land 2", everyFrame=false)
    on FINISHED → Piece Fade
  }

  state Get Three {
    SetStringValue(stringVariable=var "Get Anim", stringValue="Get 3", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Got 2")
    SetStringValue(stringVariable=var "Piece Land Particles", stringValue="Piece Land 3", everyFrame=false)
    on FINISHED → Piece Fade
  }

  state Get Four {
    SetStringValue(stringVariable=var "Get Anim", stringValue="Get 4", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Got 3")
    SetStringValue(stringVariable=var "Piece Land Particles", stringValue="Piece Land 4", everyFrame=false)
    on FINISHED → Piece Fade
  }

  state Check Max {
    RecordPlayerStory(EventType=HeartPiece)
    IntCompare(integer1=var "Pieces", integer2=3, equal=→"MAX UP", lessThan=(none), greaterThan=→"MAX UP", everyFrame=false)
    IncrementPlayerDataInt(intName="heartPieces")
    SendEventToRegister(eventName="HEART PIECE SAVE")
    on FINISHED → Update particle achievement progress
    on MAX UP → Fuse
  }

  state Fuse {
    Tk2dPlayAnimation(gameObject=var "Piece", animLibName="", clipName="Fuse")
    Tk2dPlayAnimation(gameObject=var "Fleur", animLibName="", clipName="Fleur Disappear")
    SetMeshRenderer(gameObject=var "Mover", active=true)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 0.42745098), colorVariable=var "Mover Colour", time=0.75, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Mover", color=var "Mover Colour", everyframe=true)
    Wait(time=0.75, finishEvent=→"WAIT", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=silk_heart_get_pt2_1st_flash (sfxstatic_assets_silkshared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on WAIT → Tween Mover
  }

  state Piece Get Particles {
    ActivateGameObject(gameObject=var "Get Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=new_heartpiece_puzzle_bit (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName=var "Piece Land Particles", storeResult=var "Piece Get Object")
    ActivateGameObject(gameObject=var "Piece Get Object", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Max
  }

  state Tween Mover {
    PlayParticleEmitter(gameObject=var "Mover Trail", emit=0, resetIfPlaying=false)
    SetMeshRenderer(gameObject=var "Mover", active=true)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Piece Colour", time=0.5, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Piece", color=var "Piece Colour", everyframe=true)
    EaseColor(fromValue=(1, 1, 1, 0.5254902), toValue=(1, 1, 1, 1), colorVariable=var "Mover Colour", time=0.5, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Tk2dSpriteSetColor(gameObject=var "Mover", color=var "Mover Colour", everyframe=true)
    iTweenMoveTo(gameObject=var "Mover", id="", transformPosition=var "", vectorPosition=(-9.52, 7, 0), time=0.75, delay=0, speed=(unset), space=1, easeType=26, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=true, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleTo(gameObject=var "Mover", id="", transformScale=var "", vectorScale=(1, 1, 1), time=0.75, delay=0, speed=(unset), easeType=26, loopType=0, startEvent=(none), finishEvent=→"WAIT", realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenMoveTo(gameObject=var "Piece", id="", transformPosition=var "", vectorPosition=(-9.52, 7, 0), time=0.75, delay=0, speed=(unset), space=1, easeType=26, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=true, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenScaleTo(gameObject=var "Piece", id="", transformScale=var "", vectorScale=(1, 1, 1), time=0.75, delay=0, speed=(unset), easeType=26, loopType=0, startEvent=(none), finishEvent=→"WAIT", realTime=false, stopOnExit=true, loopDontFinish=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=dream_ghost_appear, pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=silk_boss_move_4 (sfxstatic_assets_areacradle.bundle), pitchMin=1.15, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on WAIT → Get
  }

  state Get {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetParent(gameObject=var "Mover Trail", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    DestroyObject(gameObject=var "Mover", delay=0, detachChildren=false)
    StopParticleEmitter(gameObject=var "Mover Trail")
    PlayParticleEmitter(gameObject=var "Get Effect", emit=0, resetIfPlaying=false)
    DestroyObject(gameObject=var "Piece", delay=0, detachChildren=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=spell_pickup_final_boom (sfxstatic_assets_areacradle.bundle), pitchMin=1.25, pitchMax=1.25, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=bind_for_crest_silk_slam_into_hornet_2d (sfxstatic_assets_crestshrine.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Lifeblood State?
  }

  state Max Up {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="MaxHealth", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataVariable(VariableName="heartPieces", SetValue=0)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddToMaxHealth", parameters=[1], storeResult=(unset var), EveryFrame=false)
    SendEventToRegister(eventName="HEART PIECE SAVE")
    SendEventToRegister(eventName="MAX HP UP")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=silk_heart_get_pt3_2nd_flash (sfxstatic_assets_silkshared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0.5, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=hornet_bind_ready (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0.5, storePlayer=<null>)
    on FINISHED → Fade
  }

  state Full Health? {
    GetPlayerDataInt(intName="health", storeValue=var "HP")
    GetPlayerDataInt(intName="maxHealth", storeValue=var "Max HP")
    IntCompare(integer1=var "HP", integer2=var "Max HP", equal=→"FINISHED", lessThan=→"HEAL", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Max Up
    on HEAL → Heal 1 HP
  }

  state Heal 1 HP {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddHealth", parameters=[1], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Full Health?
  }

  state Hero Anim End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Heart Piece End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Destroy Self
  }

  sequence state Continue Deposit {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    SetPlayerDataVariable(VariableName="disablePause", SetValue=false)
    SendEventToRegister(eventName="CONTINUE QUEST BOARD DEPOSIT")
    DestroySelf(detachChildren=false)
    RemoveHeroInputBlocker(Blocker=Self)
  }

  state Lifeblood State? {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="get_IsInLifebloodState", parameters=[], storeResult=var "Is In Lifeblood State", EveryFrame=false)
    BoolTest(boolVariable=var "Is In Lifeblood State", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="ResetLifebloodState", parameters=[], storeResult=(unset var), EveryFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Blue_Health_Cleanse (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Full Health?
  }

  state Piece Get Particles 2 {
    ActivateGameObject(gameObject=var "Get Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=new_heartpiece_puzzle_bit (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayVibrationV3(vibrationDataAsset=average_impact (vibrationstatic_assets_vibrationdataasset.bundle), motors=None, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    FindChild(gameObject=Self, childName=var "Piece Land Particles", storeResult=var "Piece Get Object")
    ActivateGameObject(gameObject=var "Piece Get Object", activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Max
  }

  state Check Max 2 {
    IntCompare(integer1=var "Pieces", integer2=3, equal=→"MAX UP", lessThan=(none), greaterThan=→"MAX UP", everyFrame=false)
    on FINISHED → Piece Get Particles
    on MAX UP → Piece Get Particles 2
  }

  state Update particle achievement progress {
    UpdateHeartPieceAchievement()
    on FINISHED → Fade
  }

  state Wait For Grounded {
    HeroControllerMethods(method=ResetGravity, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    HeroControllerMethods(method=AffectedByGravity, parameters=[true], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    HeroControllerMethods(method=StartAnimationControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    CheckIsCharacterGrounded(Target=var "Hero", RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    Wait(time=4, finishEvent=→"LAND", realTime=false)
    on LAND → Continue Deposit
  }

  state Init Phase 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=heartpiece_decorative_ui (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetParent(gameObject=var "Mover Trail", parent=var "Mover", resetLocalPosition=false, resetLocalRotation=false)
    AddHeroInputBlocker(Blocker=Self)
    on FINISHED → Darken
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init Phase 2
  }

  state Check at board {
    CheckIsInQuestBoardDepositSequence(trueEvent=→"FINISHED", falseEvent=(none), storeValue=(unset))
    Comment(comment="Skip if at board, we'll play animation manually below.")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SetStartWithDownSpikeEnd())
    on FINISHED → Hero Anim End
  }

  state Wait For Grounded 2 {
    HeroControllerMethods(method=StopAnimationControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Downspike Recovery", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Wait(time=0.1, finishEvent=(none), realTime=false)
    on FINISHED → Wait For Grounded
  }

  state State 1 {
    RemoveHeroInputBlocker(Blocker=Self)
  }

  var HP: int = 0
  var Max HP: int = 0
  var Pieces: int = 1
  var Is In Lifeblood State: bool = false
  var Init: bool = false
  var Get Anim: string = ""
  var Piece Land Particles: string = ""
  var Blanker Colour: color = (1, 1, 1, 0)
  var Mover Colour: color = (1, 1, 1, 0)
  var Piece Colour: color = (1, 1, 1, 0)
  var Blanker: gameObject = <null>
  var Fleur: gameObject = <null>
  var Get Effect: gameObject = <null>
  var Get Glow: gameObject = <null>
  var Hero: gameObject = <null>
  var Mover: gameObject = <null>
  var Mover Trail: gameObject = <null>
  var Piece: gameObject = <null>
  var Piece Get Object: gameObject = <null>
  var Self: gameObject = <null>
}
