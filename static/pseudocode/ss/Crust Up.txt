# Crust Up
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Coral King

fsm Crust Up {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on CRUST UP → Crust Pause
  }

  state Crust Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Crust Up")  // disabled
    PlayParticleEmitterChildren(gameObject=var "Pt Crust Up", resetTimeIfPlaying=false, stopOnStateExit=false)
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=30)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  state Crust Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Crust Up
  }

  state Finish {
    Tk2dSpriteTweenColor(gameObject=Self, color=(0.5294118, 0.5294118, 0.5294118, 0), duration=0.5)
  }

  var Pt Crust Up: gameObject = Boss Scene/Coral King/Pt Crust Up
}
