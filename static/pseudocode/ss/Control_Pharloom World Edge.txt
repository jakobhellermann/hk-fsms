# Control
# 1 location(s):
#   Coral_19 / Pharloom World Edge

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock")
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Dormant
  }

  state Dormant {
    Trigger2dEvent(gameObject=var "Start Range", trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Take Control
  }

  state Take Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=FaceLeft())
    SetPlayerDataBool(boolName="disablePause", value=true)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RunFSM(fsmTemplateControl=template=6159435039333135483 in[Clamp X<-0], finishEvent=→"LAND", everyFrame=false)
    on LAND → Start Pause
  }

  state Meet {
    RunDialogue(Sheet="Coral", Key=var "Meet Convo", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=true, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetPlayerDataBool(boolName=var "Visited Bool", value=true)
    on CONVO_END → End Dialogue
  }

  state End Dialogue {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    EndDialogue(ReturnControl=false, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move Back
  }

  state End {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    on FINISHED → Dormant
  }

  state Start Pause {
    PlayAudioEvent(audioClip=soft_landing_sand (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Act Check
  }

  state Convo Choice {
    PlayerDataBoolTest(boolName=var "Visited Bool", isTrue=→"REPEAT", isFalse=→"MEET")
    on MEET → Meet
    on REPEAT → Repeat
  }

  state Repeat {
    RunDialogue(Sheet="Coral", Key=var "Repeat Convo", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=true, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Act Check {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"ACT 3", isFalse=→"EARLY")
    on EARLY → Early
    on ACT 3 → Act 3
  }

  state Early {
    SetStringValue(stringVariable=var "Visited Bool", stringValue="encounteredPharloomEdge", everyFrame=false)
    SetStringValue(stringVariable=var "Meet Convo", stringValue="WORLD_EDGE", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Convo", stringValue="WORLD_EDGE_REPEAT", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Act 3 {
    SetStringValue(stringVariable=var "Visited Bool", stringValue="encounteredPharloomEdgeAct3", everyFrame=false)
    SetStringValue(stringVariable=var "Meet Convo", stringValue="WORLD_EDGE_ACT3", everyFrame=false)
    SetStringValue(stringVariable=var "Repeat Convo", stringValue="WORLD_EDGE_ACT3_REPEAT", everyFrame=false)
    on FINISHED → Convo Choice
  }

  state Move Back {
    MoveHeroToPosX(Hero=var "Hero", PositionTarget=var "", PositionX=var "Move To X", FinishEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End
  }

  var Move To X: float = 21.1
  var Wake Threads: gameObject = <null>

  var Ground Y: float = 0
  var Visited Bool: string = ""
  var Meet Convo: string = ""
  var Repeat Convo: string = ""
  var CamLock: gameObject = <null>
  var Dust Trail: gameObject = <null>
  var Self: gameObject = <null>
  var Parent: gameObject = <null>
  var Wake Enemies: gameObject = <null>
  var Pt WakeSlam: gameObject = <null>
  var Start Range: gameObject = Pharloom World Edge/Start Range
}
