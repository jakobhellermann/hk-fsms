fsm Jostle Hero On Entry {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Jostle
  }

  state Jostle {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="BONK BACK", delay=0, everyFrame=false)
    on FINISHED → Idle
  }
}
