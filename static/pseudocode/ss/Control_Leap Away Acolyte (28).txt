fsm Control {
  start Idle

  state Idle {
    SetMeshRenderer(gameObject=Self, active=false)
    on LEAVE → Plane
  }

  state Leave Pause {
    WaitRandom(timeMin=0.5, timeMax=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Leave
  }

  state Leave {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Leap")
    SelectRandomVector3(vector3Array=[(-5, 20, 0), (0, 20, 0), (5, 20, 0)], weights=[1, 1, 1], storeVector3=var "Leave Vector")
    AnimatePositionBy(gameObject=Self, shiftBy=var "Leave Vector", time=0.4, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=true)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Plane {
    CheckZPosition(gameObject=Self, compareTo=0, compareToOffset=0, tolerance=0, equal=→"FG", lessThan=→"FG", greaterThan=→"BG", everyFrame=false, space=0)
    on BG → BG
    on FG → FG
  }

  state FG {
    RandomFloat(min=-8, max=-16, storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Leave Pause
  }

  state BG {
    RandomFloat(min=2, max=4, storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Leave Pause
  }

  var Z Pos: float = 0
  var Leave Vector: vector3 = (0, 0, 0)
}
