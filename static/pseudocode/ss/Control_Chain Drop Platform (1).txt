  }

  }

    on BREAK → Slash Break

  }

    ActivateGameObject(gameObject=var "Collider Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled

  }

  }

  state Plat Land {
    PlayAudioEvent(audioClip=bell_chain_plat_land (sfxstatic_assets_areabell.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    PlayParticleEmitter(gameObject=var "Land Dust", emit=0, resetIfPlaying=false)
