fsm Control {
  start Idle

  state Idle {
    on START → Start
  }

  state Start {
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    FadeAudioV2(Target=Self, StartVolume=(unset), EndVolume=0.8, Time=1, SetOnExit=false)
    on STOP → Stop
  }

  state Stop {
    FadeAudioV2(Target=Self, StartVolume=(unset), EndVolume=0, Time=1, SetOnExit=false)  // disabled
    AudioStopV2(gameObject=Self, fadeTime=1, cancelOnEarlyExit=false)
    on START → Start
  }
}
