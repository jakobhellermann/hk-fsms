# Behaviour
# 2 location(s):
#   Memory_Coral_Tower / Enemy Activator Groups/Enemy Activator Mid/Enemy Folder/Coral Goomba L (1)
#   Memory_Coral_Tower / Enemy Activator Groups/Enemy Activator Low/Enemy Folder/Coral Goomba L

// uses template: coral_goomba_behaviour
fsm Behaviour {
  start Init

  state Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Idle Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    WaitRandom(timeMin=0.1, timeMax=0.2, finishEvent=→"WALK", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    on WALK → Walk
    on SING → Sing
  }

  state Walk {
    RandomBool(storeResult=var "Schedule Turn")
    StartCrawler(Target=Self, ScheduleTurn=var "Schedule Turn")
    WaitRandom(timeMin=0.7, timeMax=2, finishEvent=→"IDLE", realTime=false)
    on IDLE → To Idle
  }

  state Init {
    StopCrawler(Target=Self, WaitForTurn=true)
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    on FINISHED → Orientation
  }

  state To Idle {
    StopCrawler(Target=Self, WaitForTurn=true)
    on FINISHED → Idle
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Sing Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on CANCEL → Idle
  }

  state Orientation {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=0, tolerance=0.1, equal=→"HORIZONTAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=180, tolerance=0.1, equal=→"HORIZONTAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=90, tolerance=0.1, equal=→"VERTICAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=270, tolerance=0.1, equal=→"VERTICAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"HORIZONTAL", delay=0, everyFrame=false)
    on HORIZONTAL → Horizontal
    on VERTICAL → Vertical
  }

  state Horizontal {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ConstrainPosition", methodName="StartConstrainY", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ConstrainPosition", methodName="SetYMin", parameters=[var "Y Pos"], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ConstrainPosition", methodName="SetYMax", parameters=[var "Y Pos"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Vertical {
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ConstrainPosition", methodName="StartConstrainX", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ConstrainPosition", methodName="SetXMin", parameters=[var "X Pos"], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ConstrainPosition", methodName="SetXMax", parameters=[var "X Pos"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  var Idle Anim: string = "L Idle"
  var Sing Anim: string = "L Sing"

  var Rotation: float = 0
  var X Pos: float = 0
  var Y Pos: float = 0
  var X Min: float = 0
  var X Max: float = 9999
  var Schedule Turn: bool = false
  var Inner Enemy: gameObject = <null>
  var Self: gameObject = <null>
}
