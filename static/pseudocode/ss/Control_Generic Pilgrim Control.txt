# Control
# 1 location(s):
#   Bone_East_10_Room / Black Thread States/Normal World/Generic Pilgrim Control

fsm Control {
  start Pause

  state Pause {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "Pilgrim 1")
    FindNamedChild(gameObject=Self, storeResult=var "Pilgrim 2")
    FindNamedChild(gameObject=Self, storeResult=var "Pilgrim 3")
    FindNamedChild(gameObject=Self, storeResult=var "Ruckus")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="rhinoRuckus", isTrue=→"RUCKUS", isFalse=(none))
    GetPlayerDataInt(intName="pilgrimRestCrowd", storeValue=var "Crowd Number")
    IntSwitch(intVariable=var "Crowd Number", compareTo=[1, 2, 3, 4, 5, 6], sendEvent=[→"1", →"2", →"3", →"4", →"5", →"6"], everyFrame=false)
    on FINISHED → Set 1
    on 1 → Set 1
    on 2 → Set 2
    on 3 → Set 3
    on 4 → Set 4
    on 5 → Set 5
    on 6 → Set 6
    on RUCKUS → Ruckus
  }

  state Set 1 {
    ActivateGameObject(gameObject=var "Pilgrim 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set 2 {
    ActivateGameObject(gameObject=var "Pilgrim 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set 3 {
    ActivateGameObject(gameObject=var "Pilgrim 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set 4 {
    ActivateGameObject(gameObject=var "Pilgrim 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set 5 {
    ActivateGameObject(gameObject=var "Pilgrim 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set 6 {
    ActivateGameObject(gameObject=var "Pilgrim 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Ruckus {
    ActivateGameObject(gameObject=var "Ruckus", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Crowd Number: int = 0
  var Pilgrim 1: gameObject = <null>
  var Pilgrim 2: gameObject = <null>
  var Pilgrim 3: gameObject = <null>
  var Self: gameObject = <null>
  var Ruckus: gameObject = <null>
}
