    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Orb", active=false)

    on FIRST → First Pause
  }

    SetMeshRenderer(gameObject=var "Orb", active=false)
    EnableFsmSelf(SetEnabled=false)
    on BIND ACTIVE → Bind Full
    on WEAK BIND ACTIVE → Activate Weak
  }

  state Active {
    EnableFsmSelf(SetEnabled=false)
    on BIND INACTIVE → Deactivate
    on WEAK BIND ACTIVE → Full to Weak
  }

  state Deactivate {
    Tk2dPlayAnimationWithEvents(gameObject=var "Orb", clipName="Bind Orb Down", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    on BIND ACTIVE → Bind Full
    on WEAK BIND ACTIVE → Activate Weak
  }

  state Appear Pause {
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Appear
    on WEAK BIND ACTIVE → Appear Skip 2
    on BIND ACTIVE → Appear Skip
  }

  state First Pause {
    EnableFsmSelf(SetEnabled=false)
    on SHOW HP → Appear
  }

  state Appear {
    SetMeshRenderer(gameObject=Self, active=true)

    on FINISHED → Idle
  }

    on BIND ACTIVE → Bind Weak to Full

  state Deactivate Weak {
    Tk2dPlayAnimationWithEvents(gameObject=var "Orb", clipName="Bind Orb Down Weak", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    on BIND ACTIVE → Bind Full
