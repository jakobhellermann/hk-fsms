// uses template: geo_sprite_hud
fsm Geo Sprite {
  start Init
  on DISAPPEAR → Disappear  // from any state

  state Idle {
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Idle Anim")
    SetMeshRenderer(gameObject=Self, active=true)
    on GET → Get
    on TAKE → Take
    on GET SINGLE → Get Single
  }

  state Get {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Get Loop Anim")
    on IDLE → Idle
    on TAKE → Take
  }

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetBoolValue(boolVariable=var "Was Active", boolValue=false, everyFrame=false)
    BoolTest(boolVariable=var "No Disappear", isTrue=→"IDLE", isFalse=(none), everyFrame=false)
    on APPEAR → Appear
    on IDLE → Idle
  }

  state Appear {
    BoolTest(boolVariable=var "Was Active", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Was Active", boolValue=true, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Appear Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Idle
    on GET → Get
  }

  state Take {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Idle Anim")
    ObjectJitterLocal(gameObject=Self, x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=0)
    on IDLE → Idle
    on GET → Get
    on LEVEL LOADED → Idle
  }

  sequence state Disappear {
    NextFrameEvent(sendEvent=(none))
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    BoolTest(boolVariable=var "No Disappear", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Was Active", boolValue=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Disappear Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Init
    on APPEAR → Appear
  }

  state Get Single {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Get Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on IDLE → Idle
    on TAKE → Take
    on FINISHED → Idle
  }

  var No Disappear: bool = true
  var Appear Anim: string = "Coin Appear"
  var Disappear Anim: string = "Coin Disappear"
  var Get Anim: string = "Coin Get"
  var Get Loop Anim: string = "Coin Get Loop"
  var Idle Anim: string = "Coin Idle"

  var Was Active: bool = false
  var Start Pos: vector3 = (0, 0, 0)
}
