fsm Control {
  start State 1
  on END → State 3  // from any state
  on LEAVING SCENE → State 3  // from any state

  state State 1 {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 2
  }

  state State 2 {
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=4)
  }

  state State 3 {
    AudioStopV2(gameObject=Self, fadeTime=1, cancelOnEarlyExit=false)
  }
}
