# Globalise
# 2 location(s):
#   coremanagers_assets__gamemanager.bundle / _GameManager/AudioManager/Atmos
#   Menu_Title / _GameManager/AudioManager/Atmos

fsm Globalise {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetGameObject(variable=var "Atmos", gameObject=var "Self", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Layer 1")
    FindNamedChild(gameObject=Self, storeResult=var "Layer 2")
    FindNamedChild(gameObject=Self, storeResult=var "Layer 3")
    on FINISHED → Idle
  }

  state Idle {
    SetAudioVolume(gameObject=var "Layer 1", volume=var "Volume Normal", everyFrame=false)
    SetAudioVolume(gameObject=var "Layer 2", volume=var "Volume Normal", everyFrame=false)
    SetAudioVolume(gameObject=var "Layer 3", volume=var "Volume Normal", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 1", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 2", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 3", everyFrame=false)
    EnableFsmSelf(SetEnabled=false)
    on ATMOS LOWPASS START → To Lowpass
  }

  state To Lowpass {
    FadeAudio(gameObject=var "Layer 1", startVolume=var "Volume Normal", endVolume=var "Volume Lowpass", time=0.5)
    FadeAudio(gameObject=var "Layer 2", startVolume=var "Volume Normal", endVolume=var "Volume Lowpass", time=0.5)
    FadeAudio(gameObject=var "Layer 3", startVolume=var "Volume Normal", endVolume=var "Volume Lowpass", time=0.5)
    EaseFloat(fromValue=22000, toValue=var "Cutoff Frequency Low", floatVariable=var "Cutoff Frequency Crt", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 1", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 1", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 2", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 2", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 3", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 3", everyFrame=false)
    on FINISHED → Lowpass
    on ATMOS LOWPASS END → To Idle
  }

  state Lowpass {
    SetAudioVolume(gameObject=var "Layer 1", volume=var "Volume Lowpass", everyFrame=false)
    SetAudioVolume(gameObject=var "Layer 2", volume=var "Volume Lowpass", everyFrame=false)
    SetAudioVolume(gameObject=var "Layer 3", volume=var "Volume Lowpass", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 1", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 2", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 3", everyFrame=false)
    EnableFsmSelf(SetEnabled=false)
    on ATMOS LOWPASS END → To Idle
  }

  state To Idle {
    FadeAudio(gameObject=var "Layer 1", startVolume=var "Volume Lowpass", endVolume=var "Volume Normal", time=0.5)
    FadeAudio(gameObject=var "Layer 2", startVolume=var "Volume Lowpass", endVolume=var "Volume Normal", time=0.5)
    FadeAudio(gameObject=var "Layer 3", startVolume=var "Volume Lowpass", endVolume=var "Volume Normal", time=0.5)
    EaseFloat(fromValue=var "Cutoff Frequency Low", toValue=22000, floatVariable=var "Cutoff Frequency Crt", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 1", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 1", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 2", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 2", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Layer 3", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Layer 3", everyFrame=false)
    on FINISHED → Idle
    on ATMOS LOWPASS START → To Lowpass
  }

  var Cutoff Frequency Crt: float = 0
  var Volume Lowpass: float = 0.65
  var Volume Normal: float = 1
  var Cutoff Frequency Low: float = 2500
  var Self: gameObject = <null>
  var Layer 1: gameObject = <null>
  var Layer 2: gameObject = <null>
  var Layer 3: gameObject = <null>
  var Lowpass Layer 1: object = _GameManager/AudioManager/Atmos/Layer 1@AudioLowPassFilter
  var Lowpass Layer 2: object = _GameManager/AudioManager/Atmos/Layer 2@AudioLowPassFilter
  var Lowpass Layer 3: object = _GameManager/AudioManager/Atmos/Layer 3@AudioLowPassFilter
}
