  start Spinning

  state Spinning {
    on CLASH LEFT → Hit L
    on CLASH RIGHT → Hit R
    on CLASH UP → Hit U
    on CLASH DOWN → Hit D
  }

  state Hit L {
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=-10, y=(unset), everyFrame=false)
    on FINISHED → Spinning
  }

  state Hit R {
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=10, y=(unset), everyFrame=false)
    on FINISHED → Spinning
  }

  state Hit U {
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=(unset), y=10, everyFrame=false)
    on FINISHED → Spinning
  }

  state Hit D {
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=(unset), y=-10, everyFrame=false)
    on FINISHED → Spinning
  }

  var Parent: gameObject = Black Thread States/Normal World/Stick Insect Flyer (1)
}
