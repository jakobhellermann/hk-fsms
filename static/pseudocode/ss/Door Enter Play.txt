# Door Enter Play
# 1 location(s):
#   City_Lace_cutscene / Cinematic Player

fsm Door Enter Play {
  start Init

  state Idle {
    on DOOR TOUCHED → Move Scene
  }

  sequence state Gate Cinematic {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    StartCinematic(CinematicPlayer=Self)
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 0), duration=0)
    on CINEMATIC END → Cinematic Ended
  }

  state Cinematic Ended {
    SetPlayerDataVariable(VariableName="visitedCitadel", SetValue=true)
    MenuStyleUnlockActionV2(UnlockKey="ENDING_B", ForceChange=false)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on FINISHED → Next Scene
  }

  state Next Scene {
    SendEventToRegister(eventName="DOOR ENTER COMPLETE")
  }

  state Move Scene {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendEventToRegister(eventName="REST AREA MUSIC STOP")
    MoveGameObjectToScene(gameObject=Self, findRootIfNecessary=false, success=(unset), successEvent=(none), failureEvent=(none), sceneReference=0, sceneAtIndex=0, sceneByName="", sceneByPath="", sceneByGameObject=Self, sceneFound=(unset), sceneFoundEvent=(none), sceneNotFoundEvent=(none))
    MoveGameObjectToScene(gameObject=Act Card, findRootIfNecessary=false, success=(unset), successEvent=(none), failureEvent=(none), sceneReference=0, sceneAtIndex=0, sceneByName="", sceneByPath="", sceneByGameObject=Self, sceneFound=(unset), sceneFoundEvent=(none), sceneNotFoundEvent=(none))
    on DOOR ENTERED → Gate Cinematic
  }

  state Init {
    GetGuidReferenceHolderGameObject(Target=Self, StoreGameObject=var "Door")
    CallMethodProper(gameObject=var "Door", behaviour="TransitionPoint", methodName="SetTargetDoor", parameters=["right_cutsceneEntry"], storeResult=(unset var), EveryFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Target Scene: string = ""
  var Door: gameObject = <null>
}
