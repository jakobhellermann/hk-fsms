# Control
# 1 location(s):
#   Under_07 / Battle Scene/Gates/steam_vent_short (2)

// uses template: steam_vent_gate
fsm Control {
  start Init
  on ANTIC → Antic  // from any state
  on ATTACK → Attack  // from any state
  on IDLE → Re-Idle  // from any state
  on BG OPEN → Open  // from any state
  on BG QUICK OPEN → Open  // from any state
  on BG CLOSE → Close Delay  // from any state

  state Init {
    FindNamedChild(gameObject=var "Pt Attack", storeResult=var "Rubble Fall")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Player SteamLoop")
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=0, tolerance=0.1, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Rubble Fall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Damager")
    FindNamedChild(gameObject=var "Damager", storeResult=var "Air Blow Region")
    SetCollider(gameObject=var "Damager", active=false, resetOnExit=false)
    SetCollider(gameObject=var "Air Blow Region", active=false, resetOnExit=false)
    FindNamedChild(gameObject=Self, storeResult=var "Pt Antic")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Attack")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG CLOSE", isFalse=(none), everyFrame=false)
  }

  state Idle {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP SHAKE", delay=0, everyFrame=false)
    on FINISHED → Antic
  }

  state Antic {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=steam_vent_antic (sfxstatic_assets_areadocks.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Buddy"), sendEvent="ANTIC", delay=0, everyFrame=false)
    on FINISHED → Attack
  }

  state Attack {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=steam_vent_release (sfxstatic_assets_areadocks.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    Wait(time=var "Attack Time", finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=var "Damager", active=true, resetOnExit=true)
    SetCollider(gameObject=var "Air Blow Region", active=true, resetOnExit=true)
    PlayParticleEmitterInState(gameObject=var "Pt Attack")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="START SHAKE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Buddy"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Re-Idle
  }

  state Re-Idle {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Buddy"), sendEvent="IDLE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP SHAKE", delay=0, everyFrame=false)
    on FINISHED → Antic
  }

  state Closed {
    SetCollider(gameObject=var "Damager", active=true, resetOnExit=true)
    SetCollider(gameObject=var "Air Blow Region", active=true, resetOnExit=true)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterInState(gameObject=var "Pt Attack")
    PlayAudioEvent(audioClip=steam_vent_release (sfxstatic_assets_areadocks.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayInState(gameObject=var "Audio Player SteamLoop", volume=(unset))
  }

  state Open {
    PlayAudioEvent(audioClip=steam_vent_release (sfxstatic_assets_areadocks.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
  }

  sequence state Close Delay {
    NextFrameEvent(sendEvent=(none))
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Closed
  }

  var Wait Time: float = 1
  var Attack Time: float = 1
  var Start Closed: bool = true

  var Rotation: float = 0
  var Pt Antic: gameObject = <null>
  var Pt Attack: gameObject = <null>
  var Damager: gameObject = <null>
  var Buddy: gameObject = <null>
  var Rubble Fall: gameObject = <null>
  var Audio Player SteamLoop: gameObject = <null>
  var Air Blow Region: gameObject = <null>
}
