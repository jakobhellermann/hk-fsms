# Control
# 3 location(s):
#   Bone_East_05 / Flea Rescue Barrel
#   Under_21 / Flea Rescue Barrel
#   Under_23 / Flea Rescue Barrel

fsm Control {
  start Init

  sequence state Init {
    FindChild(gameObject=Self, childName="Break Effects", storeResult=var "Break Effects")
    ActivateGameObject(gameObject=var "Flea", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    ActivateGameObject(gameObject=var "Flea", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckQuestPdSceneBool(QuestTarget=var "Quest Tracker", ExpectedValue=true, trueEvent=→"ACTIVATE", falseEvent=(none), storeValue=(unset))
    on FINISHED → To Idle
    on ACTIVATE → Activated
  }

  state Break {
    SetMeshRenderer(gameObject=Self, active=false)
    AudioStop(gameObject=var "Audio Loop Voice", fadeTime=0)
    SendEventToRegister(eventName="FLEA RESCUED")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    PlayAudioEventRandom(audioClips=[barrel_death_1 (sfxstatic_assets_areacogareapeakareawardshared.bundle), barrel_death_2 (sfxstatic_assets_areashellwoodshared.bundle)], pitchMin=1.1, pitchMax=0.9, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Break Effects", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RescueToFly")  // disabled
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)  // disabled
    AnimatePositionBy(gameObject=Self, shiftBy=(0, 2, 0), time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)  // disabled
    AnimatePositionBy(gameObject=Self, shiftBy=(0, 1.5, 0), time=0.5, speed=(unset), delay=0.25, easeType=14, reverse=false, finishEvent=(none), realTime=false)  // disabled
    Wait(time=1.5, finishEvent=→"NEXT", realTime=false)  // disabled
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"R", delay=0, everyFrame=false)
    on L → Emerge R
    on R → Emerge L
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=var "In Alert Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"DAMAGED", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetBoolValueAtTime(BoolVariable=var "Can Shift", BoolValue=true, Time=1, SetOppositeOnStateEntry=true)
    BoolAllTrue(boolVariables=[var "Can Shift", var "In Alert Range"], sendEvent=→"SHIFT", storeResult=(unset), everyFrame=true)
    on DAMAGED → Check Range
    on SHIFT → To Alert
  }

  state Activated {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Emerge R {
    SetFsmBool(gameObject=var "Flea", fsmName="Control", variableName="Emerge Right", setValue=true, everyFrame=false)
    on FINISHED → Activate Flea
  }

  state Activate Flea {
    ActivateGameObject(gameObject=var "Flea", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Emerge L {
    SetFsmBool(gameObject=var "Flea", fsmName="Control", variableName="Emerge Left", setValue=true, everyFrame=false)
    on FINISHED → Activate Flea
  }

  state Check Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Can Rescue Range", storeResult=false, sendEvent="RESCUE", outOfRangeEvent="", everyFrame=false)
    on FINISHED → Alert
    on RESCUE → Break
  }

  state To Idle {
    SetAudioClip(gameObject=var "Audio Loop Voice", audioClip=Flea_scared_loop_01 (sfxstatic_assets_fleacaravan.bundle), autoPlay=true, stopOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barrel")
    PlayAudioEvent(audioClip=grass_goomba_land_n_roll_1 (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    on FINISHED → Idle
  }

  state Alert {
    AudioPlayInState(gameObject=var "Audio Loop Shake", volume=(unset))
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=var "In Alert Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"DAMAGED", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetBoolValueAtTime(BoolVariable=var "Can Shift", BoolValue=true, Time=1, SetOppositeOnStateEntry=true)
    BoolTestMulti(boolVariables=[var "Can Shift", var "In Alert Range"], boolStates=[true, false], trueEvent=→"SHIFT", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on DAMAGED → Check Range
    on SHIFT → To Idle
  }

  state To Alert {
    SetAudioClip(gameObject=var "Audio Loop Voice", audioClip=Flea_scared_loop_excited_01 (sfxstatic_assets_fleacaravan.bundle), autoPlay=true, stopOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barrel Alert")
    on FINISHED → Alert
  }

  var In Alert Range: bool = false
  var Can Shift: bool = false
  var Break Effects: gameObject = <null>
  var Flea: gameObject = Flea Rescue Barrel/Flea Rescue Activation
  var Audio Loop Voice: gameObject = Flea Rescue Barrel/Audio Loop Voice
  var Audio Loop Shake: gameObject = Flea Rescue Barrel/Audio Loop Shake
  var Quest Tracker: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
