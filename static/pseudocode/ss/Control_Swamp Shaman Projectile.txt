# Control
# 1 location(s):
#   enemycorpses_assets_areaswamp.bundle / Swamp Shaman Projectile

fsm Control {
  start Init
  on DAMAGED HERO → Explode  // from any state
  on PROJECTILE BREAK → Explode  // from any state

  state Bounce Antic {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=swamp_bounce_pod_antic_loop (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Looping Audio")
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Glob Fly")
    DecelerateXY(gameObject=Self, decelerationX=0.98, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Slow Down
  }

  state Explode {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    AudioStop(gameObject=var "Looping Audio", fadeTime=0)
    PlayAudioEvent(audioClip=swamp_bounce_pod_explode (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Explode", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    SpawnObjectFromGlobalPool(gameObject=Swamp Bounce Pod Explosion Shaman (localpoolprefabs_assets_areaswamp.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Init {
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    on FINISHED → Air
  }

  state Slow Down {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    DecelerateV2(gameObject=Self, deceleration=0.87, brakeOnExit=false)
    Wait(time=0.45, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Explode
    on TINKED → Explode
  }

  state Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    Wait(time=2, finishEvent=→"END", realTime=false)
    on LAND → Land
    on END → Recycle
    on TINKED → Bounce Antic
  }

  state Land {
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed X", y=(unset), space=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Glob Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=swamp_bounce_pod_initial_hit (sfxstatic_assets_areaswamp.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Bounce Launch
    on TINKED → Bounce Launch
  }

  state Bounce Launch {
    RandomFloat(min=15, max=22, storeResult=var "Bounce Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed X", y=var "Bounce Y", everyFrame=false)
    on FINISHED → Bounce Antic
  }

  var Speed X: float = 0
  var Bounce Y: float = 0
  var Self: gameObject = Swamp Shaman Projectile
  var Looping Audio: gameObject = <null>
  var Antic Effects: gameObject = Swamp Shaman Projectile/Antic Effects
}
