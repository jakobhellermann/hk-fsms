# Control
# 1 location(s):
#   localpoolprefabs_assets_areadust.bundle / Chef Maggot Blob

// uses template: chef_maggot_blob
fsm Control {
  start Launch Pause
  on HERO DAMAGED → Maggot Infect  // from any state

  state Launch Pause {
    AudioStop(gameObject=Self, fadeTime=0)
    RandomFloat(min=var "Launch Y Min", max=var "Launch Y Max", storeResult=var "Velocity Y")
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Rebounded", boolValue=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    FindNamedChild(gameObject=Self, storeResult=var "Pt Burst")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Damage Hero")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Idle")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Launch")
    FindNamedChild(gameObject=Self, storeResult=var "Terrain Detector")
    SetCircleCollider(gameObject=Self, active=true)
    on FINISHED → Launch
  }

  state Launch {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=var "", WaitForEndOfClip=(unset), finishedEvent=(none))
    RandomFloat(min=-0.1031, max=-0.1039, storeResult=var "Pos Z")
    PlayParticleEmitterInState(gameObject=var "Pt Idle")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Pos Z", space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetDamageHero(Target=Self, Enabled=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetVector2XY(vector2Variable=var "Launch Velocity", vector2Value=(unset), x=var "Velocity X", y=var "Velocity Y", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=var "Launch Velocity", x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Glob Idle")
    SetMeshRenderer(gameObject=Self, active=true)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Velocity Y", float2=-0.1, tolerance=0, equal=(none), lessThan=→"FALL", greaterThan=(none), everyFrame=true)
    RayCast2dV2(fromGameObject=Self, fromPosition=(1.2, 0), direction=(-1, 0), space=0, distance=2.4, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "Hitting Wall", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTestMulti(boolVariables=[var "Hitting Wall", var "Rebounded"], boolStates=[true, false], trueEvent=→"WALL", falseEvent=(none), storeResult=(unset), everyFrame=true)
    GetVelocity2dNotZero(gameObject=Self, vector=(unset), x=var "Travel X", y=(unset), space=0, everyFrame=true)
    PlayParticleEmitter(gameObject=var "Pt Launch", emit=0, resetIfPlaying=false)  // disabled
    on FALL → Fall
    on WALL → Wall Bump
    on TINK LEFT → Tink L
    on TINK UP → Tink U
    on TINK DOWN → Tink D
    on TINK RIGHT → Tink R
    on HEAVY → Quick Burst
  }

  state Fall {
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RandomFloat(min=0.003, max=0.0039, storeResult=var "Pos Z")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Pos Z", space=0, everyFrame=false, lateUpdate=false)
    CheckCollisionSide(collidingObject=var "Terrain Detector", topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    RayCast2dV2(fromGameObject=Self, fromPosition=(1.2, 0), direction=(-1, 0), space=0, distance=2.4, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "Hitting Wall", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTestMulti(boolVariables=[var "Hitting Wall", var "Rebounded"], boolStates=[true, false], trueEvent=→"WALL", falseEvent=(none), storeResult=(unset), everyFrame=true)
    GetVelocity2dNotZero(gameObject=Self, vector=(unset), x=var "Travel X", y=(unset), space=0, everyFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Idle")
    on LAND → Land
    on WALL → Wall Bump
    on TINK LEFT → Tink L
    on TINK UP → Tink U
    on TINK DOWN → Tink D
    on TINK RIGHT → Tink R
    on HEAVY → Quick Burst
  }

  state Land {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Glob Burst", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Burst
  }

  state Burst {
    AudioStop(gameObject=Self, fadeTime=0)
    PlayAudioEventRandom(audioClips=[goop_impact_1 (sfxstatic_assets_areaslab.bundle), goop_impact_2 (sfxstatic_assets_areaslab.bundle), goop_impact_3 (sfxstatic_assets_areaslab.bundle), goop_impact_4 (sfxstatic_assets_areaslab.bundle)], pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=great_chef_projectile_explode (sfxstatic_assets_areadust.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHero(Target=Self, Enabled=false)
    PlayParticleEmitter(gameObject=var "Pt Burst", emit=0, resetIfPlaying=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → End Pause
  }

  state End Pause {
    SetCircleCollider(gameObject=Self, active=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Wall Bump {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetBoolValue(boolVariable=var "Rebounded", boolValue=true, everyFrame=false)
    FloatMultiply(floatVariable=var "Travel X", multiplyBy=-0.5, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Travel X", y=(unset), everyFrame=false)
    on FINISHED → Fall
    on HEAVY → Quick Burst
  }

  state Maggot Infect {
    AudioStop(gameObject=Self, fadeTime=0)
    PlayAudioEvent(audioClip=great_chef_projectile_explode (sfxstatic_assets_areadust.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Glob Infect", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetHeroMaggoted(Value=true)
    SetDamageHero(Target=Self, Enabled=false)
    PlayParticleEmitter(gameObject=var "Pt Damage Hero", emit=0, resetIfPlaying=false)
    on FINISHED → End Pause
  }

  state Tink L {
    AddForce2dAsAngle(gameObject=Self, forceMode=1, atPosition=(unset), angle=180, speed=var "Tink Force", maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    on FINISHED → Tink Rebound
  }

  state Tink Rebound {
    PlayAudioEventRandom(audioClips=[goop_impact_1 (sfxstatic_assets_areaslab.bundle), goop_impact_2 (sfxstatic_assets_areaslab.bundle), goop_impact_3 (sfxstatic_assets_areaslab.bundle), goop_impact_4 (sfxstatic_assets_areaslab.bundle)], pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=bounce_shroom (sfxstatic_assets_areamemory.bundle), pitchMin=0.75, pitchMax=0.75, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fall
  }

  state Tink R {
    AddForce2dAsAngle(gameObject=Self, forceMode=1, atPosition=(unset), angle=0, speed=var "Tink Force", maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    on FINISHED → Tink Rebound
  }

  state Tink U {
    AddForce2dAsAngle(gameObject=Self, forceMode=1, atPosition=(unset), angle=90, speed=var "Tink Force", maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    on FINISHED → Tink Rebound
  }

  state Tink D {
    AddForce2dAsAngle(gameObject=Self, forceMode=1, atPosition=(unset), angle=270, speed=var "Tink Force", maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    on FINISHED → Tink Rebound
  }

  state Quick Burst {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Glob Quick Burst")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Burst
  }

  var Launch Y Min: float = 29
  var Launch Y Max: float = 34

  var Velocity Y: float = 0
  var Velocity X: float = 0
  var Travel X: float = 0
  var Pos Z: float = 0
  var Tink Force: float = 10
  var Hitting Wall: bool = false
  var Rebounded: bool = false
  var Launch Velocity: vector2 = (0, 0)
  var Terrain Detector: gameObject = Chef Maggot Blob/Terrain Detector (localpoolprefabs_assets_areadust.bundle)
  var Pt Launch: gameObject = Chef Maggot Blob/Pt Launch (localpoolprefabs_assets_areadust.bundle)
  var Pt Idle: gameObject = Chef Maggot Blob/Pt Idle (localpoolprefabs_assets_areadust.bundle)
  var Pt Damage Hero: gameObject = Chef Maggot Blob/Pt Damage Hero (localpoolprefabs_assets_areadust.bundle)
  var Pt Burst: gameObject = Chef Maggot Blob/Pt Burst (localpoolprefabs_assets_areadust.bundle)
}
