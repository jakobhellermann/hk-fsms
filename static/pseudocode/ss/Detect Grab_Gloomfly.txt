# Detect Grab
# 1 location(s):
#   localpoolprefabs_assets_areaabyss.bundle / Gloomfly

fsm Detect Grab {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventV2(gameObject=var "Grab Collider", trigger=0, collideTag="", collideLayer=Hero Box(20), sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Check
  }

  state Check {
    GetFsmBool(gameObject=var "Hero", fsmName="Roar and Wound States", variableName="Is Grabbed", storeValue=var "Hero Is Grabbed", everyFrame=false)
    BoolTest(boolVariable=var "Hero Is Grabbed", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    CanHeroBeGrabbed(eventTarget=Self, canTakeDmgEvent=→"GRAB", cannotTakeDmgEvent=→"CANCEL")
    on CANCEL → Idle
    on GRAB → Send Grab Event
  }

  state Send Grab Event {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="GRAB HERO", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Hero Is Grabbed: bool = false
  var Grab Collider: gameObject = Gloomfly/Grab Collider
}
