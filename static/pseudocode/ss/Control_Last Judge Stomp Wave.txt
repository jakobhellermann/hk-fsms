# Control
# 1 location(s):
#   localpoolprefabs_assets_areacoral.bundle / Last Judge Stomp Wave

fsm Control {
  start Start Pause

  state Recycle {
    RecycleSelf()
    ActivateGameObject(gameObject=var "Damager 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Damager 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Start Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Damager 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Damager 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Move
  }

  state Move {
    PlayAudioEvent(audioClip=final_judge_horizontal_fire_burst (sfxstatic_assets_areacoral.bundle), pitchMin=0.9, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Speed", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Speed", multiplyBy=-100, everyFrame=false)
    AccelerateToX(gameObject=Self, accelerationFactor=7, targetSpeed=var "X Speed")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObjectDelay(gameObject=var "Damager 1", activate=true, resetOnExit=false, delay=0.1)
    ActivateGameObjectDelay(gameObject=var "Damager 2", activate=true, resetOnExit=false, delay=0.2)
    on FINISHED → Recycle
  }

  var X Speed: float = 0
  var Damager 1: gameObject = Last Judge Stomp Wave/Damager 1
  var Damager 2: gameObject = Last Judge Stomp Wave/Damager 2
}
