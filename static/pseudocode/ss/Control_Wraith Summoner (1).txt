# Control
# 1 location(s):
#   Dust_Maze_02 / Wraith Summoner (1)

// uses template: wraith_summon
fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on WRAITH CALL → Set Amt
  }

  state Set Amt {
    RandomInt(min=2, max=3, storeResult=var "Summons", inclusiveMax=true, noRepeat=false)
    SetFloatValue(floatVariable=var "Distance Multiplier", floatValue=1, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Down", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Left", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Right", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Up", boolValue=false, everyFrame=false)
    on FINISHED → Choose
  }

  state Choose {
    SendRandomEvent(events=[→"DOWN", →"UP", →"LEFT", →"RIGHT"], weights=[1, 1, 1, 1], delay=0)
    on DOWN → Place Down
    on UP → Place Up
    on RIGHT → Place Right
    on LEFT → Place Left
  }

  state Place Down {
    BoolTest(boolVariable=var "Did Down", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Summoned Wraith")
    GameObjectIsNull(gameObject=var "Summoned Wraith", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Add Vector", vector3Value=(0, -27, 0), everyFrame=false)
    Vector3Multiply(vector3Variable=var "Add Vector", multiplyBy=var "Distance Multiplier", everyFrame=false)
    Vector3Add(vector3Variable=var "Summon Pos", addVector=var "Add Vector", everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Summoned Wraith", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=Self, translateMin=(-15, 0, 0), translateMax=(15, 0, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Summoned Wraith"), sendEvent="SUMMON", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Down", boolValue=true, everyFrame=false)
    on CANCEL → Choose
    on FINISHED → Loop
  }

  state Loop {
    FloatAdd(floatVariable=var "Distance Multiplier", add=0.35, everyFrame=false, perSecond=false)
    IntAdd(intVariable=var "Summons", add=-1, everyFrame=false)
    IntCompare(integer1=var "Summons", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"FINISHED", everyFrame=false)
    on END → Idle
    on FINISHED → Choose
  }

  state Place Up {
    BoolTest(boolVariable=var "Did Up", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Summoned Wraith")
    GameObjectIsNull(gameObject=var "Summoned Wraith", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Add Vector", vector3Value=(0, 27, 0), everyFrame=false)
    Vector3Multiply(vector3Variable=var "Add Vector", multiplyBy=var "Distance Multiplier", everyFrame=false)
    Vector3Add(vector3Variable=var "Summon Pos", addVector=var "Add Vector", everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Summoned Wraith", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=Self, translateMin=(-15, 0, 0), translateMax=(15, 0, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Summoned Wraith"), sendEvent="SUMMON", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Up", boolValue=true, everyFrame=false)
    on CANCEL → Choose
    on FINISHED → Loop
  }

  state Place Right {
    BoolTest(boolVariable=var "Did Right", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Summoned Wraith")
    GameObjectIsNull(gameObject=var "Summoned Wraith", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Add Vector", vector3Value=(32, 0, 0), everyFrame=false)
    Vector3Multiply(vector3Variable=var "Add Vector", multiplyBy=var "Distance Multiplier", everyFrame=false)
    Vector3Add(vector3Variable=var "Summon Pos", addVector=var "Add Vector", everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Summoned Wraith", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=Self, translateMin=(0, -10, 0), translateMax=(0, 10, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Summoned Wraith"), sendEvent="SUMMON", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Right", boolValue=true, everyFrame=false)
    on CANCEL → Choose
    on FINISHED → Loop
  }

  state Place Left {
    BoolTest(boolVariable=var "Did Left", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Summoned Wraith")
    GameObjectIsNull(gameObject=var "Summoned Wraith", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector3Value(vector3Variable=var "Add Vector", vector3Value=(-32, 0, 0), everyFrame=false)
    Vector3Multiply(vector3Variable=var "Add Vector", multiplyBy=var "Distance Multiplier", everyFrame=false)
    Vector3Add(vector3Variable=var "Summon Pos", addVector=var "Add Vector", everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Summoned Wraith", vector=var "Summon Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=Self, translateMin=(0, -10, 0), translateMax=(0, 10, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Summoned Wraith"), sendEvent="SUMMON", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Left", boolValue=true, everyFrame=false)
    on CANCEL → Choose
    on FINISHED → Loop
  }

  var Distance Multiplier: float = 0
  var Summons: int = 0
  var Did Up: bool = false
  var Did Down: bool = false
  var Did Left: bool = false
  var Did Right: bool = false
  var Summon Pos: vector3 = (0, 0, 0)
  var Add Vector: vector3 = (0, 0, 0)
  var Summoned Wraith: gameObject = <null>
}
