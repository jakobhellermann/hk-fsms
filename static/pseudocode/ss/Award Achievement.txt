  start Init

  state Init {
    on FINISH ENTRY → Actually Unlocked?
  }

  state Actually Unlocked? {
    PlayerDataVariableTest(VariableName="BelltownHouseState", ExpectedValue=enum(2), IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on FALSE → Inert
    on TRUE → Award
  }

  state Inert {
  }

  state Award {
    AwardAchievementProgress(Key="BELLHOME", CurrentValue=(unset), MaxValue=(unset))
  }
}
