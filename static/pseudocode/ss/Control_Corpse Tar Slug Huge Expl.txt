# Control
# 1 location(s):
#   enemycorpses_assets_areadocks.bundle / Corpse Tar Slug Huge Expl

fsm Control {
  start Pause

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashingTarSlug())
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayInState(gameObject=var "Audio Loop Fireball", volume=1)
    AudioPlayInState(gameObject=var "Audio Loop Goop", volume=1)
    on FINISHED → Explode
  }

  state Explode {
    SpawnObjectFromGlobalPool(gameObject=Goopy Explosion (localpoolprefabs_assets_areadocks.bundle), spawnPoint=var "Explosion Point", position=(unset), rotation=(unset), storeObject=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pop", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelFlash())
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var Pt Antic: gameObject = Corpse Tar Slug Huge Expl/Pt Antic
  var Self: gameObject = Corpse Tar Slug Huge Expl
  var Explosion Point: gameObject = Corpse Tar Slug Huge Expl/Explosion Point
  var Audio Loop Fireball: gameObject = Corpse Tar Slug Huge Expl/Audio Loop Fireball
  var Audio Loop Goop: gameObject = Corpse Tar Slug Huge Expl/Audio Loop Goop
}
