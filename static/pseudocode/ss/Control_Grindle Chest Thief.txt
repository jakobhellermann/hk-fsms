# Control
# 3 location(s):
#   Hang_08 / Chest Scene/Grindle Chest Thief
#   Song_03 / Chest Scene/Grindle Chest Thief
#   Song_17 / Chest Scene/Grindle Chest Thief

fsm Control {
  start Pause

  state Init {
    GetParent(gameObject=Self, storeResult=var "Chest Scene")
    FindNamedChild(gameObject=var "Chest Scene", storeResult=var "Chest")
    GetPersistentBoolFromSaveData(Target=var "Chest", SceneName="", ID="", StoreValue=var "Chest Opened")
    PlayerDataBoolTest(boolName="grindleChestEncountered", isTrue=→"ENCOUNTERED", isFalse=(none))
    BoolTest(boolVariable=var "Chest Opened", isTrue=→"OPENED", isFalse=(none), everyFrame=false)
    on FINISHED → Wait for Approach
    on OPENED → Already Opened
    on ENCOUNTERED → Already Opened
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Wait for Approach {
    Trigger2dEvent(gameObject=var "Approach Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Rummaging
  }

  state Rummaging {
    SetPlayerDataBool(boolName="grindleChestEncountered", value=true)
    SendEventByName(eventTarget=GameObject(var "Chest"), sendEvent="QUICK OPEN", delay=0, everyFrame=false)
    Trigger2dEvent(gameObject=var "Alert Range", trigger=1, collideTag="", sendEvent=→"ALERT", storeCollider=var "")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.2, None=(none), ActiveInner=→"NOISE", ActiveOuter=→"NOISE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    AudioPlaySimple(gameObject=var "Rummage Loop", volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Audio Loop Voice", volume=1, oneShotClip=<null>)
    on NOISE → RummageToLeave
    on ALERT → RummageToLeave
  }

  state Leave Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=dust_roach_newjump_far_2 (sfxstatic_assets_areadust.bundle), pitchMin=1.1, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Leave
  }

  state Already Opened {
    SetPlayerDataInt(intName="grindleChestLocation", value=0)  // disabled
    SetPlayerDataBool(boolName="grindleChestEncountered", value=true)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Leave {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Leap Away")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetVelocityByScale(gameObject=Self, speed=-15, ySpeed=60, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=bone_flyer_dash_burst (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dSpriteTweenColor(gameObject=Self, color=(0, 0, 0, 1), duration=0.15)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state RummageToLeave {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Grindle_alert (sfxdynamic_assets_areasong.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rummage End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=relic_dealer_lean_back_to_normal (sfxstatic_assets_areabell.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=lace_back_or_forward_step_1 (sfxstatic_assets_laceboss.bundle), pitchMin=1.1, pitchMax=1.1, volume=0.75, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Leeave Pause
  }

  state Leeave Pause {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Leave Antic
  }

  var Chest Opened: bool = false
  var Chest Scene: gameObject = <null>
  var Chest: gameObject = Chest Scene/Grindle Chest Thief
  var Approach Range: gameObject = Chest Scene/Grindle Chest Thief/Approach Range
  var Alert Range: gameObject = Chest Scene/Grindle Chest Thief/Alert Range
  var Rummage Loop: gameObject = Chest Scene/Grindle Chest Thief/Rummage Loop
  var Audio Loop Voice: gameObject = Chest Scene/Grindle Chest Thief/Audio Loop Voice
}
