# FSM
# 1 location(s):
#   Demo Start / Demo Start Sequence

fsm FSM {
  start Fade In

  state Fade In {
    SetInputHandlerSkipMode(SkipMode=NOT_SKIPPABLE)
    EaseSpriteColor(target=var "Fader", fromValue=(0, 0, 0, 1), toValue=(0, 0, 0, 0), disableSpriteRendererIfTransparent=false, time=2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enable Skip
  }

  state Wait {
    Wait(time=7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Out
    on CUSTOM CUTSCENE SKIP → Fade Out Instant
  }

  state Next Scene? {
    CheckStaticBool(variableName="DemoIsContinuing", trueEvent=→"CONTINUE GAME", falseEvent=→"NEW GAME", EveryFrame=false)
    on NEW GAME → To New Game
    on CONTINUE GAME → To Continue Game
  }

  state To New Game {
    GetConstantsValue(variableName="STARTING_SCENE", storeValue=var "To Scene")
    BeginSceneTransition(sceneName=var "To Scene", entryGateName="", entryDelay=0, visualization=Custom, preventCameraFadeOut=true)
  }

  state Fade Out {
    EaseSpriteColor(target=var "Fader", fromValue=(0, 0, 0, 0), toValue=(0, 0, 0, 1), disableSpriteRendererIfTransparent=false, time=1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Next Scene?
  }

  state To Continue Game {
    StartCoroutine(gameObject=var "GameManager", behaviour="GameManager", functionCall=RunContinueGame(false), stopOnExit=false)
  }

  state Enable Skip {
    SetInputHandlerSkipMode(SkipMode=SKIP_PROMPT)
    on FINISHED → Wait
  }

  state Fade Out Instant {
    SetSpriteRendererColor(TintColor=(0, 0, 0, 1), Target=var "Fader")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Next Scene?
  }

  var To Scene: string = ""
  var Fader: gameObject = Demo Start Sequence/Blanker
}
