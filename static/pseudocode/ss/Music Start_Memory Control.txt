# Music Start
# 1 location(s):
#   Memory_Red / Memory Control

fsm Music Start {
  start State 1

  state State 1 {
    on AREA TITLE APPEAR → First Time - Start Music
    on DOOR GET UP FINISHED → Reload - Music and Title
  }

  sequence state First Time - Start Music {
    Wait(time=1.5, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=Red Memory (audiocuesdynamic_assets_areamemoryareatutorial.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
  }

  state Reload - Music and Title {
    ApplyMusicCue(musicCue=Red Memory (audiocuesdynamic_assets_areamemoryareatutorial.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
    SendEventToRegister(eventName="AREA TITLE APPEAR")
  }
}
