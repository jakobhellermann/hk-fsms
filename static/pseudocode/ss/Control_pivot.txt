# Control
# 1 location(s):
#   Song_Tower_Destroyed / State Control/song_tower_right_gate/pivot

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Plat Dust")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Rock")
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Rotation", space=1, everyFrame=false)
    FloatOperator(float1=var "Start Rotation", float2=2.5, operation=Add, storeResult=var "Depress Rotation", everyFrame=false)
    FloatOperator(float1=var "Start Rotation", float2=-2, operation=Add, storeResult=var "Raise Rotation", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    on COLLIDE → Check Collision
    on TINK LEFT → Set Raise
    on TINK RIGHT → Set Depress
    on TINK UP → Set Raise
    on TINK DOWN → Set Depress
  }

  state Check Collision {
    GetCollision2dInfo(gameObjectHit=var "", relativeVelocity=(unset), relativeSpeed=(unset), contactPoint=(unset), contactNormal=var "Collision Normal", shapeCount=(unset), physics2dMaterialName="")
    GetVector3XYZ(vector3Variable=var "Collision Normal", storeX=var "Collision X", storeY=var "Collision Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Collision Y", float2=0, tolerance=(unset), equal=→"FINISHED", lessThan=→"DEPRESS", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on DEPRESS → Set Depress
  }

  state Rocks {
    BoolTest(boolVariable=var "Played Rocks", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Rock", emit=0, resetIfPlaying=false)
    SetBoolValue(boolVariable=var "Played Rocks", boolValue=true, everyFrame=false)
    on FINISHED → Wobble
  }

  state Wobble {
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Target Rotation", worldSpace=false, time=0.1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    AudioStop(gameObject=Self, fadeTime=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    PlayAudioEvent(audioClip=platform_dirt_fall_3 (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Return
  }

  state Return {
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Start Rotation", worldSpace=false, time=0.25, speed=(unset), delay=(unset), easeType=22, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on TINK RIGHT → Set Depress
    on TINK LEFT → Set Raise
    on TINK UP → Set Raise
    on TINK DOWN → Set Depress
  }

  state Set Depress {
    SetFloatValue(floatVariable=var "Target Rotation", floatValue=var "Depress Rotation", everyFrame=false)
    on FINISHED → Rocks
  }

  state Set Raise {
    SetFloatValue(floatVariable=var "Target Rotation", floatValue=var "Raise Rotation", everyFrame=false)
    on FINISHED → Rocks
  }

  var Collision X: float = 0
  var Collision Y: float = 0
  var Start Rotation: float = 0
  var Depress Rotation: float = 0
  var Raise Rotation: float = 0
  var Target Rotation: float = 0
  var Played Rocks: bool = false
  var Collision Normal: vector3 = (0, 0, 0)
  var Plat Dust: gameObject = State Control/song_tower_right_gate/pivot/Plat Dust
  var Pt Rock: gameObject = State Control/song_tower_right_gate/pivot/Pt Rock
}
