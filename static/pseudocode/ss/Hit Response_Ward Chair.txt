# Hit Response
# 4 location(s):
#   localpoolprefabs_assets_areaward.bundle / Ward Chair
#   Ward_02 / Ward Chair
#   Ward_02b / Ward Chair
#   Ward_05 / Ward Chair

fsm Hit Response {
  start Init

  state Tink Left {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=-100, y=(unset), vector3=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WHEEL START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="HIT LEFT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Tink Right {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=100, y=(unset), vector3=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WHEEL START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="HIT RIGHT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Rotation", space=0, everyFrame=false)
    FloatCompare(float1=var "Self Rotation", float2=0, tolerance=5, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"CANCEL", delay=0, everyFrame=false)
    on FINISHED → Idle
    on CANCEL → Tipped Over
  }

  state Tink From Hero Dir {
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "Self X", tolerance=0, equal=→"HIT_RIGHT", lessThan=→"HIT_RIGHT", greaterThan=→"HIT_LEFT", everyFrame=false)
    on HIT_LEFT → Tink Left
    on HIT_RIGHT → Tink Right
  }

  state Idle {
    SendMessage(gameObject=Ward Chair/seat pivot/seat, delivery=SendMessage, options=1, functionCall=StartTimedJitter())
    on HIT_LEFT → Tink Left
    on HIT_RIGHT → Tink Right
    on HIT_UP → Tink From Hero Dir
    on HIT_DOWN → Tink From Hero Dir
  }

  state Tink Left 2 {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=-50, y=(unset), vector3=(unset), everyFrame=false)
    on FINISHED → Tipped Over
  }

  state Tink Right 2 {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=50, y=(unset), vector3=(unset), everyFrame=false)
    on FINISHED → Tipped Over
  }

  state Tink From Hero Dir 2 {
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "Self X", tolerance=0, equal=→"HIT_RIGHT", lessThan=→"HIT_RIGHT", greaterThan=→"HIT_LEFT", everyFrame=false)
    on HIT_LEFT → Tink Left 2
    on HIT_RIGHT → Tink Right 2
  }

  state Tipped Over {
    SendMessage(gameObject=Ward Chair/seat pivot/seat, delivery=SendMessage, options=1, functionCall=StartTimedJitter())
    on HIT_LEFT → Tink Left 2
    on HIT_RIGHT → Tink Right 2
    on HIT_UP → Tink From Hero Dir 2
    on HIT_DOWN → Tink From Hero Dir 2
  }

  var Hero X: float = 0
  var Self X: float = 0
  var Self Rotation: float = 0
}
