# FSM
# 1 location(s):
#   Bone_East_LavaChallenge / entrance_door

fsm FSM {
  start Init

  state Wait For Trigger {
    ActivateGameObject(gameObject=var "Door", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckTrackTriggerCountV2(Target=entrance_door/Inside Detector, Count=0, Test=MoreThan, EveryFrame=true, SetBool=(unset), SuccessEvent=→"ENTER", FailEvent=(none))
    on ENTER → Close
  }

  sequence state Complete {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    AnimatorPlay(gameObject=var "Door", stateName="Break Antic", layer=(unset), normalizedTime=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=big_crumbly_door_opened_or_closed (sfxstatic_assets_areaboneareacoralareacrawlareamossareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
  }

  state Close {
    ActivateGameObject(gameObject=var "Door", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=big_crumbly_door_move (sfxstatic_assets_areaboneareacoralareacrawlareamossareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on LAVA CHALLENGE COMPLETE → Complete
  }

  state Init {
    PlayerDataBoolTest(boolName="lavaChallengeEntranceCavedIn", isTrue=→"WAIT FOR TRIGGER", isFalse=→"WAIT FOR EXPLOSION")
    on WAIT FOR TRIGGER → Wait For Trigger
    on WAIT FOR EXPLOSION → Wait For Explosion
  }

  state Wait For Explosion {
    ActivateGameObject(gameObject=var "Door", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckTrackTriggerCountV2(Target=entrance_door/Inside Detector, Count=0, Test=MoreThan, EveryFrame=true, SetBool=(unset), SuccessEvent=→"ENTER", FailEvent=(none))
    on CAVE IN → Close
  }

  var Door: gameObject = entrance_door/cave_in
}
