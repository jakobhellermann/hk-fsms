  state Idle {

  state Splash {
    PlayRandomAudioClipTable(Table=sherma_water_splash (sfxdynamic_assets_sherma.bundle), AudioPlayerPrefab=var "", SpawnPoint=Sherma Citadel Spa NPC/Sherma/Sprite, SpawnPosition=(0, 0, 0))
    WaitRandom(timeMin=0.3, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Splash
  }
}
