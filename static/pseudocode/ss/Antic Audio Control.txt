# Antic Audio Control
# 1 location(s):
#   Shellwood_18 / Boss Scene Parent/Boss Scene/Spikes

fsm Antic Audio Control {
  start Idle

  state Idle {
    on SPIKE ANTIC → Antic
  }

  state Antic {
    AudioPlayInState(gameObject=Self, volume=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on SPIKE ANTIC → Antic
    on FINISHED → Idle
  }
}
