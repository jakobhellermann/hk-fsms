# Check Exit Leading
# 4 location(s):
#   Bellway_08 / Zaza Beckon NPC
#   Coral_33 / Black Thread States/Black Thread World/Garmond Scenes/Garmond Black Threaded Scene/Zaza Beckon NPC
#   Coral_35 / Zaza Beckon NPC
#   Coral_43 / Zaza Beckon NPC

fsm Check Exit Leading {
  start Idle

  state Idle {
    on LEAVING SCENE → Check
  }

  state Check {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"ZAZA LEADING", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"HERO LEADING", delay=0, everyFrame=false)
    on ZAZA LEADING → Zaza Leading
    on HERO LEADING → Hero Leading
  }

  state Zaza Leading {
    SetStaticVariable(variableName="Hero Is Leading Zaza", setValue=false, sceneTransitionsLimit=1)
  }

  state Hero Leading {
    SetStaticVariable(variableName="Hero Is Leading Zaza", setValue=true, sceneTransitionsLimit=1)
  }
}
