# centipede_boss_seg
# 2 location(s):
#   Bellway_Centipede_Arena / Boss Scene/BodyWall L/Segs/Body Seg (2)
#   Bellway_Centipede_Arena / Boss Scene/BodyWall R/Segs/Body Seg (2)

// uses template: centipede_boss_seg
fsm centipede_boss_seg {
  start Init
  on FINAL EXPLODE → Final Explode Pause  // from any state
  on SEGMENTS THRASH → Death  // from any state
  on EXPLODE → Explode  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Slash Collider", storeResult=var "Slash Collider")
    ActivateGameObject(gameObject=var "Slash Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on MOVE → Move Anim
  }

  state Move Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Head Travel")
    BoolTest(boolVariable=var "Head", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body Travel")
    on FINISHED → IntroMove
  }

  state Head End {
  }

  state IntroMove {
    BoolTest(boolVariable=var "Head", isTrue=→"END", isFalse=(none), everyFrame=false)
    on END → Head End
    on IDLE → IntroMove End
  }

  sequence state IntroMove End {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Body Move End", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Idle
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Slash Range", storeResult=false, sendEvent="SLASH", outOfRangeEvent="", everyFrame=true)
    AudioStopV2(gameObject=Self, fadeTime=0.1, cancelOnEarlyExit=false)
    on MOVE → Move
    on TINK → Slash Antic
    on SLASH → Slash Antic
  }

  state Move {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body Travel")
    on IDLE → Move End
  }

  sequence state Move End {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Body Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Idle
  }

  state Slash Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body Slash Antic")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slash
  }

  state Slash {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body Slash")
    ActivateGameObject(gameObject=var "Slash Collider", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Wait(time=0.7, finishEvent=(none), realTime=false)
    PlayAudioEvent(audioClip=grasshopper_fly_attack_slashy (sfxstatic_assets_areaclover.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    on FINISHED → Slash End
  }

  state Slash End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body Slash End")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Death {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Body Slash Antic")
    ActivateGameObject(gameObject=var "Slash Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Final Explode Pause {
    Wait(time=var "Explode Pause", finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPoolOverTimeV2(gameObject=Giant Centipede Death Pop v2 (localpoolprefabs_assets_areabellway.bundle), spawnPoint=var "Self", position=(unset), originVariationX=3.5, originVariationY=3, rotation=(unset), frequency=0.25, scaleMin=1, scaleMax=1)
    on FINISHED → Explode
  }

  state Explode {
    SpawnObjectFromGlobalPool(gameObject=Giant Centipede Death Explode v2 (localpoolprefabs_assets_areabellway.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Explode Pause: float = 1
  var Head: bool = false

  var Slash Collider: gameObject = <null>
  var Self: gameObject = <null>
  var Audio Player: gameObject = <null>
}
