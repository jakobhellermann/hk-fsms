# Control
# 1 location(s):
#   Ant_08 / Battle Scene/Mites/Mitefly (2)

fsm Control {
  start Init
  on INSTA KILL → Insta Kill  // from any state

  state Init {
    GameObjectIsActive(Target=var "Pt Rain", ActiveSpace=1, trueEvent=(none), falseEvent=(none), storeValue=var "Show Rain")
    SetGameObject(variable=var "Next Point", gameObject=var "Patrol Point", everyFrame=false)
    SetParent(gameObject=var "Patrol Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Start Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    BoolTest(boolVariable=var "Battler", isTrue=→"BATTLER", isFalse=(none), everyFrame=false)
    on FINISHED → Patrol
    on BATTLER → Fly In Ready
  }

  state Patrol {
    ChaseObjectGround(gameObject=Self, target=var "Start Point", xOffset=0, speedMax=5, acceleration=1, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0, snapTo=true, snapSpeedOnly=false, onlyOnStateEntry=false)
    ChaseObjectVertical(gameObject=Self, target=var "Next Point", offset=0, speedMax=5, acceleration=0.1)
    GetDistance(gameObject=Self, target=var "Next Point", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=→"NEXT", lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    ActivateGameObjectDelay(gameObject=var "Attack Range", activate=true, resetOnExit=false, delay=0.5)
    on NEXT → Decel
    on SING → Sing
    on ATTACK → Bump Up?
  }

  state Next Point {
    GameObjectCompare(gameObjectVariable=var "Next Point", compareTo=var "Patrol Point", equalEvent=→"START", notEqualEvent=→"PATROL", storeResult=(unset), everyFrame=false)
    on PATROL → To Patrol
    on START → To Start
  }

  state To Patrol {
    SetGameObject(variable=var "Next Point", gameObject=var "Patrol Point", everyFrame=false)
    on FINISHED → Patrol
  }

  state To Start {
    SetGameObject(variable=var "Next Point", gameObject=var "Start Point", everyFrame=false)
    on FINISHED → Patrol
  }

  state Decel {
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    ChaseObjectGround(gameObject=Self, target=var "Start Point", xOffset=0, speedMax=5, acceleration=1, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0, snapTo=true, snapSpeedOnly=false, onlyOnStateEntry=false)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    on FINISHED → Next Point
    on SING → Sing
    on ATTACK → Bump Up?
  }

  state Sing {
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=mite_sing (sfxdynamic_assets_areagreymoor.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    SendEventByName(eventTarget=GameObject(var "Sprite"), sendEvent="STOP", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Sing")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"FINISHED", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Attack End
  }

  state Attack Antic {
    ActivateGameObject(gameObject=var "Pt Rain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sprite"), sendEvent="STOP", delay=0, everyFrame=false)
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[mitefly_antic (sfxstatic_assets_areagreymoor.bundle)], weights=[1], pitchMin=0.9, pitchMax=1.1, stopPreviousSound=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.85, decelerationY=0.85, brakeOnExit=true)
    GetScale(gameObject=var "Sprite", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=12, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Attack Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStop(gameObject=Self, fadeTime=0)
    on FINISHED → Attack
  }

  state Attack {
    PlayAudioEvent(audioClip=mitefly_attack (sfxstatic_assets_areagreymoor.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Attack", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Attack Collider", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    GetScale(gameObject=var "Sprite", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-30, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.95, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Attack Recover
  }

  state Attack Recover {
    Tk2dWatchAnimationEvents(gameObject=var "Sprite", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.88, decelerationY=(unset), brakeOnExit=true)
    ActivateGameObject(gameObject=Self, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Attack End
  }

  state Attack End {
    SendEventByName(eventTarget=GameObject(var "Sprite"), sendEvent="RESTART", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Rain", activate=var "Show Rain", recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Choose Next Point
  }

  state Choose Next Point {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    GetYDistance(gameObject=Self, target=var "Patrol Point", storeResult=var "Patrol Distance", everyFrame=false, allowNegatives=false)
    GetYDistance(gameObject=Self, target=var "Start Point", storeResult=var "Start Distance", everyFrame=false, allowNegatives=false)
    FloatCompare(float1=var "Start Distance", float2=var "Patrol Distance", tolerance=0, equal=→"START", lessThan=→"PATROL", greaterThan=→"START", everyFrame=false)
    on START → To Start
    on PATROL → To Patrol
  }

  state Fly In Ready {
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetCollider(gameObject=var "Sprite", active=false, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=var "Sprite", color=(0, 0, 0, 1), everyframe=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Sprite"), sendEvent="STOP", delay=0, everyFrame=false)
    on BATTLE START → Fly In
    on WAKE → Fly In
  }

  state Fly In {
    Translate(gameObject=var "Sprite", vector=(unset), x=(unset), y=10, z=-8, space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FaceObjectV2(objectA=var "Sprite", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetMeshRenderer(gameObject=var "Sprite", active=true)
    AnimatePositionBy(gameObject=var "Sprite", shiftBy=(0, -10, 8), time=0.75, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=var "Sprite", color=(1, 1, 1, 1), duration=0.75)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=var "Sprite", color=(1, 1, 1, 1), everyframe=false)
    SetCollider(gameObject=var "Sprite", active=true, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Sprite"), sendEvent="RESTART", delay=0, everyFrame=false)
    on FINISHED → Patrol
  }

  state Insta Kill {
    CheckOutOfCamera(gameObject=Self, margin=1, outsideEvent=→"DISAPPEAR", insideEvent=(none), insideBool=false, outsideBool=false, everyFrame=false)
    SetShardDrop(target=Self, shards=0)
    InstaDeath(target=Self, direction=90)
    on DISAPPEAR → Disappear
  }

  state Bump Up? {
    FaceObjectV2(objectA=var "Sprite", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    RayCast2dV2(fromGameObject=var "Ray Pt Floor", fromPosition=(unset), direction=(-1, 0), space=1, distance=3, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"FINISHED", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=20, everyFrame=false)
    on FINISHED → Attack Antic
  }

  state Disappear {
    DestroySelf(detachChildren=false)
  }

  var Battler: bool = false

  var Distance: float = 0
  var X Scale: float = 0
  var Patrol Distance: float = 0
  var Start Distance: float = 0
  var Show Rain: bool = false
  var Next Point: gameObject = <null>
  var Patrol Point: gameObject = Battle Scene/Mites/Mitefly (2)/Patrol Point
  var Start Point: gameObject = Battle Scene/Mites/Mitefly (2)/Start Point
  var Attack Collider: gameObject = Battle Scene/Mites/Mitefly (2)/Sprite/Attack Collider
  var Sprite: gameObject = Battle Scene/Mites/Mitefly (2)/Sprite
  var Attack Range: gameObject = Battle Scene/Mites/Mitefly (2)/Attack Range
  var Alert Range: gameObject = Roachfeeder Short/Alert Range (localpoolprefabs_assets_areadust.bundle)
  var Spawn Pos Alt: gameObject = Roachfeeder Short/Spawn Pos Alt (localpoolprefabs_assets_areadust.bundle)
  var Ray Pt Floor: gameObject = Battle Scene/Mites/Mitefly (2)/Sprite/Ray Pt Floor
  var Pt Rain: gameObject = Battle Scene/Mites/Mitefly (2)/Sprite/Pt Rain
  var Audio Loop Voice: gameObject = Battle Scene/Mites/Mitefly (2)/Audio Loop Voice
}
