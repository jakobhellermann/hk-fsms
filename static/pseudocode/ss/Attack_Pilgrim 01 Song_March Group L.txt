# Attack

fsm Attack {
  start Init

  state Attack Antic {
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="HANDMAIDEN CALL", delay=0, everyFrame=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=pilgrim_leaper_attack (sfxdynamic_assets_pilgrimslower.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Ground Check
  }

  state Attack Launch {
    PlayAudioEvent(audioClip=pilgrim_01_song_jump (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "X Speed", floatValue=var "X Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "X Speed", multiplyBy=-10, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=var "X Speed", y=16, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Air", animationTriggerEvent=(none), animationCompleteEvent=(none))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Attack Air
  }

  state Attack Roll {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Land", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    DecelerateXY(gameObject=Self, decelerationX=0.95, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Attack Land
  }

  state Attack Air {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), everyFrame=true)
    on LAND → Roll Start
  }

  state Idle {
    on ATTACK → Attack Antic
  }

  state Return {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="pilgrim_behaviour"), sendEvent="ATTACK END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Attack Land {
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"AIR", EveryFrame=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.975, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Face Hero
    on AIR → Re-Air
  }

  state Roll Start {
    SetVelocityByScale(gameObject=Self, speed=-15, ySpeed=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=pilgrim_01_song_land_and_roll (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Attack Roll
  }

  state Re-Air {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Re-Air")
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Roll Start
  }

  state Preacher Pray {
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=CancelSetZ())
    AudioStop(gameObject=var "Voice Loop", fadeTime=0)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DORMANT", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on BATTLE START → Enter Idle
  }

  state Enter Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter Antic
  }

  state Enter Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack Antic")
    on FINISHED → Enter Leap
  }

  state Enter Leap {
    PlayRandomAudioClipTable(Table=enemy_jump_in (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetMeshRenderer(gameObject=Self, active=true)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack Air")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=28, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    RandomFloat(min=0.006, max=0.007, storeResult=var "Z Target")
    EaseFloat(fromValue=var "Z Pos", toValue=var "Z Target", floatVariable=var "Z Tween", time=0.4, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Tween", space=0, everyFrame=true, lateUpdate=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.4)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter Fall
  }

  state Enter Fall {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Target", space=0, everyFrame=false, lateUpdate=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Enter Land
  }

  state Enter Land {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    PlayAudioEventRandom(audioClips=[crowman_dagger_ground_land_1 (sfxstatic_assets_areagreymoor.bundle), crowman_dagger_ground_land_2 (sfxstatic_assets_areagreymoor.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Return
  }

  state Stage Sleep {
    AudioStop(gameObject=var "Voice Loop", fadeTime=0)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DORMANT", delay=0, everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on BATTLE START → Wait
    on WAKE → Fake Death Antic
  }

  state Wait {
    WaitRandom(timeMin=0.5, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter Antic
  }

  state Fake Death Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    AudioPlayRandom(gameObject=var "Self", audioClips=[Pilgrim_Leaper_attack_01 (sfxstatic_assets_pilgrimslower.bundle), Pilgrim_Leaper_attack_02 (sfxstatic_assets_pilgrimslower.bundle), Pilgrim_Leaper_attack_03 (sfxstatic_assets_pilgrimslower.bundle)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Attack Launch
  }

  state Init {
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Preacher Pray", equalEvent=→"PREACHER PRAY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Clip", compareTo="Stage Sleep", equalEvent=→"STAGE SLEEP", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Clip", compareTo="Stage Death", equalEvent=→"STAGE SLEEP", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on PREACHER PRAY → Preacher Pray
    on STAGE SLEEP → Stage Sleep
    on FINISHED → Idle
  }

  state Ground Check {
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"AIR", EveryFrame=false)
    on FINISHED → Attack Launch
    on AIR → Fall
  }

  state Fall {
    SetFloatValue(floatVariable=var "X Speed", floatValue=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    on FINISHED → Attack Air
  }

  state Face Hero {
    CheckFacingTarget(facingObject=Self, target=var "Hero", spriteFacesRight=false, everyFrame=false, facingEvent=→"FINISHED", notFacingEvent=(none), facingBool=(unset), notFacingBool=(unset))
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEventsV2(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED", animationInterruptedEvent=(none))
    DecelerateXY(gameObject=Self, decelerationX=0.975, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Return
  }

  var X Scale: float = 0
  var X Speed: float = 0
  var Z Pos: float = 0
  var Z Target: float = 0
  var Z Tween: float = 0
  var Clip: string = ""
  var Self: gameObject = March Group Control/March Group L/Pilgrim 01 Song
  var Voice Loop: gameObject = <null>
  var Audio Loop Voice: gameObject = March Group Control/March Group L/Pilgrim 01 Song/Audio Loop Voice
}
