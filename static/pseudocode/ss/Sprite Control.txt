# Sprite Control
# 1 location(s):
#   Song_10 / Sherma Citadel Spa NPC

fsm Sprite Control {
  start Pause
  on TALK START → Talking  // from any state

  state Init {
    FindNamedChild(gameObject=var "Sherma", storeResult=var "Pt Splash")
    SetRandomAudioClipFromTable(Table=sherma_swim_loop (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    GetPosition(gameObject=var "Swim Point L", vector=(unset), x=var "Min X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Swim Point R", vector=(unset), x=var "Max X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Min X", float2=1.1, operation=Add, storeResult=var "Margin Min X", everyFrame=false)
    FloatOperator(float1=var "Max X", float2=1.1, operation=Subtract, storeResult=var "Margin Max X", everyFrame=false)
    ActivateGameObject(gameObject=var "Swim Point L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Swim Point R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FloatOperator(float1=var "Swim Speed", float2=-1, operation=Multiply, storeResult=var "Swim Speed L", everyFrame=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Swim")
    on FINISHED → Swim R
  }

  state Start Swimming {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Swim")
    GetPosition(gameObject=var "Sherma", vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Self X", float2=var "Margin Min X", tolerance=0, equal=(none), lessThan=→"R", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Self X", float2=var "Margin Max X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"L", everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Turn L
    on R → Turn R
  }

  state Swim L {
    SendRandomEventV2(events=[→"STARTLE", (none)], weights=[0.25, 0.75], trackingInts=[var "Ct Splash", var "Ct Swim"], eventMax=[2, 3])
    SetVelocity2d(gameObject=var "Sherma", vector=(unset), x=var "Swim Speed L", y=(unset), everyFrame=false)
    CheckXPosition(gameObject=var "Sherma", compareTo=var "Min X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"END", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.2, MaxReactDelay=1.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Talk L Active
    on SING → Sing
    on STARTLE → Startle
  }

  state Swim Idle {
    GetPosition(gameObject=var "Sherma", vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    DecelerateXY(gameObject=var "Sherma", decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    WaitRandom(timeMin=2, timeMax=5, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.2, MaxReactDelay=1.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Start Swimming
    on SING → Sing
    on STARTLE → Startle
  }

  state Swim R {
    SendRandomEventV2(events=[→"STARTLE", (none)], weights=[0.25, 0.75], trackingInts=[var "Ct Splash", var "Ct Swim"], eventMax=[2, 3])
    SetVelocity2d(gameObject=var "Sherma", vector=(unset), x=var "Swim Speed", y=(unset), everyFrame=false)
    CheckXPosition(gameObject=var "Sherma", compareTo=var "Max X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"END", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.2, MaxReactDelay=1.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Swim End
    on SING → Sing
    on STARTLE → Startle
  }

  state Turn L {
    SendEventByScale(gameObject=var "Sherma Sprite", eventTarget=Self, xScale=true, positiveEvent=(none), negativeEvent=→"FINISHED", space=0)
    SetScale(gameObject=var "Sherma Sprite", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa TurnToSwim")
    on FINISHED → Swim L
    on STARTLE → Startle
  }

  state Turn R {
    SendEventByScale(gameObject=var "Sherma Sprite", eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)
    SetScale(gameObject=var "Sherma Sprite", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa TurnToSwim")
    on FINISHED → Swim R
    on STARTLE → Startle
  }

  state Noise React 1 {
    SetVelocity2d(gameObject=var "Sherma", vector=(unset), x=0, y=(unset), everyFrame=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Swim End")
    on FINISHED → Noise React 2
  }

  state Noise React 2 {
    FaceObjectV2(objectA=var "Sherma Sprite", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Sing")
    PlayParticleEmitterInState(gameObject=var "Pt Splash")
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Noise React End
  }

  state Noise React End {
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Sing End")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=→"NOISE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Swimming
    on NOISE → Noise React 1
    on SING → Sing React 1
  }

  state Sing React 1 {
    SetVelocity2d(gameObject=var "Sherma", vector=(unset), x=0, y=(unset), everyFrame=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Swim End")
    on FINISHED → Sing React 2
  }

  state Sing React 2 {
    FaceObjectV2(objectA=var "Sherma Sprite", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Sing")
    PlayParticleEmitterInState(gameObject=var "Pt Splash")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=→"FINISHED", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Noise React End
  }

  state Talk L Active {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Swim End
  }

  state Talking {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    DecelerateXY(gameObject=var "Sherma", decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Listen")
    FaceObjectV2(objectA=var "Sherma Sprite", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="Spa TurnToListen", resetFrame=true, everyFrame=false, pauseBetweenTurns=0.5)
    on TALK END → Listen End
  }

  state Swim End {
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Swim End")
    on FINISHED → Swim Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  state Startle {
    SetVelocity2d(gameObject=var "Sherma", vector=(unset), x=0, y=(unset), everyFrame=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    FaceObjectV2(objectA=var "Sherma Sprite", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Sing")
    PlayParticleEmitterInState(gameObject=var "Pt Splash")
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    AudioPlayInState(gameObject=var "Audio Loop Splash", volume=(unset))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPLASH START", delay=0, everyFrame=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=sherma_giggle (sfxdynamic_assets_sherma.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Restart Interactible?
  }

  state Restart Interactible? {
    SetRandomAudioClipFromTable(Table=sherma_swim_loop (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPLASH STOP", delay=0, everyFrame=false)
    CheckXPosition(gameObject=var "Sherma", compareTo=var "Min X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"FINISHED", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Swim End
  }

  state Sing {
    SetVelocity2d(gameObject=var "Sherma", vector=(unset), x=0, y=(unset), everyFrame=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    FaceObjectV2(objectA=var "Sherma Sprite", objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Sing")
    PlayParticleEmitterInState(gameObject=var "Pt Splash")
    AudioPlayInState(gameObject=var "Audio Loop Splash", volume=(unset))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=sherma_sing (sfxdynamic_assets_sherma.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=0, MinReactDelay=0.5, MaxReactDelay=0.6, None=→"FINISHED", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SPLASH START", delay=0, everyFrame=false)
    on FINISHED → Restart Interactible?
  }

  sequence state Listen End {
    Tk2dPlayAnimationWait(Target=var "Sherma Sprite", ClipName="Spa Listen End", AnimationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=var "Sherma Sprite", animLibName="", clipName="Spa Idle")
    SetRandomAudioClipFromTable(Table=sherma_swim_loop (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    on FINISHED → Swim Idle
  }

  var Min X: float = 0
  var Max X: float = 0
  var Margin Min X: float = 0
  var Margin Max X: float = 0
  var Self X: float = 0
  var Swim Speed: float = 1.5
  var Swim Speed L: float = 0
  var Ct Splash: int = 0
  var Ct Swim: int = 0
  var Pt Splash: gameObject = <null>
  var Sherma Sprite: gameObject = Sherma Citadel Spa NPC/Sherma/Sprite
  var Swim Point L: gameObject = Sherma Citadel Spa NPC/Swim Point L
  var Swim Point R: gameObject = Sherma Citadel Spa NPC/Swim Point R
  var Sherma: gameObject = Sherma Citadel Spa NPC/Sherma
  var Audio Loop Swim: gameObject = Sherma Citadel Spa NPC/Sherma/Audio Loop Swim
  var Audio Loop Splash: gameObject = Sherma Citadel Spa NPC/Sherma/Audio Loop Splash
  var Audio Loop Voice: gameObject = Sherma Citadel Spa NPC/Sherma/Audio Loop Voice
}
