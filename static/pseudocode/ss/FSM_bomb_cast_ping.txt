# FSM
# 1 location(s):
#   localpoolprefabs_assets_shared.bundle / bomb_cast_ping

fsm FSM {
  start Init

  state Init {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Activation Delay", everyFrame=false)
    FloatMultiplyV2(floatVariable=var "Activation Delay", multiplyBy=0.025, everyFrame=false, fixedUpdate=false)
    Wait(time=var "Activation Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Do Effects
  }

  state Do Effects {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Activation Delay: float = 0
  var Self: gameObject = bomb_cast_ping
}
