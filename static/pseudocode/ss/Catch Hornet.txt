# Catch Hornet
# 4 location(s):
#   Abyss_Cocoon / Boss Control/Abyss Wave/Abyss Wave L1/abyss_wave/Damager
#   Abyss_Cocoon / Boss Control/Abyss Wave/Abyss Wave R1/abyss_wave/Damager
#   Abyss_Cocoon / Boss Control/Abyss Wave/Abyss Wave R2/abyss_wave/Damager
#   Abyss_Cocoon / Boss Control/Abyss Wave/Abyss Wave L2/abyss_wave/Damager

// uses template: abyss_wave_catch_hero
fsm Catch Hornet {
  start Init

  state Idle {
    on CATCH HERO → Catch
    on MULTI HIT CONNECT → Catch
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Catch Dummy")
    on FINISHED → Idle
  }

  state Catch {
    SetPositionToObject2D(gameObject=var "Catch Dummy", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)  // disabled
    FindClosest(gameObject=var "Hero", withTag="Teleplane", ignoreOwner=true, mustBeVisible=false, storeObject=var "Catch Target", storeDistance=(unset), everyFrame=false)
    GetPosition(gameObject=var "Catch Target", vector=var "Catch Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=true)
    AnimatePositionTo(gameObject=var "Catch Dummy", toValue=var "Catch Pos", localSpace=true, time=0.15, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)  // disabled
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Catch Dummy", xOffset=0, yOffset=0, everyFrame=true)  // disabled
    GetVector3XYZ(vector3Variable=var "Catch Pos", storeX=var "Catch X", storeY=var "Catch Y", storeZ=(unset), everyFrame=true)
    FloatClamp(floatVariable=var "Catch Y", minValue=7, maxValue=9999, everyFrame=true)
    FloatClamp(floatVariable=var "Catch X", minValue=5, maxValue=54, everyFrame=true)
    SetPosition2d(gameObject=var "Hero", vector=(unset), x=var "Catch X", y=var "Catch Y", space=0, everyFrame=true, lateUpdate=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Listen For MultiHit: bool = false

  var Catch X: float = 0
  var Catch Y: float = 0
  var Catch Pos: vector3 = (0, 0, 0)
  var Catch Dummy: gameObject = <null>
  var Catch Target: gameObject = <null>
}
