# Control
# 1 location(s):
#   Crawl_01 / Crypt Worm Dig Effect (3)

fsm Control {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Travel", vector=var "Travel Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetDistance(gameObject=Self, target=var "Travel", storeResult=var "Tween Time", everyFrame=false)
    FloatDivide(floatVariable=var "Tween Time", divideBy=5, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=var "")
    on FINISHED → Range
  }

  state Range {
    FindNamedChild(gameObject=Self, storeResult=var "Alert Range")
    GameObjectIsNull(gameObject=var "Alert Range", isNull=→"ALERT", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    on ALERT → Cam Check
  }

  state Cam Check {
    GetDistance(gameObject=Self, target=var "MainCamera", storeResult=var "Cam Distance", everyFrame=true)
    FloatCompare(float1=var "Cam Distance", float2=42, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Start Pause
  }

  state Start Pause {
    WaitRandom(timeMin=0, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dest Check
  }

  state Dest Check {
    PlayParticleEmitter(gameObject=Self, emit=0, resetIfPlaying=false)
    BoolTest(boolVariable=var "At Home", isTrue=→"TRAVEL", isFalse=→"HOME", everyFrame=false)
    on TRAVEL → To Travel
    on HOME → To Home
  }

  state To Travel {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.5)
    AnimatePositionTo(gameObject=Self, toValue=var "Travel Pos", localSpace=false, time=var "Tween Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    BoolFlip(boolVariable=var "At Home")
    StopParticleEmitter(gameObject=Self)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0, time=0.5)
    WaitRandom(timeMin=3, timeMax=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Cam Check
  }

  state To Home {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.5)
    AnimatePositionTo(gameObject=Self, toValue=var "Start Pos", localSpace=false, time=var "Tween Time", speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  var Cam Distance: float = 0
  var Tween Time: float = 0
  var At Home: bool = true
  var Start Pos: vector3 = (0, 0, 0)
  var Travel Pos: vector3 = (0, 0, 0)
  var Travel: gameObject = Crypt Worm Dig Effect (3)/Travel
  var Alert Range: gameObject = <null>
}
