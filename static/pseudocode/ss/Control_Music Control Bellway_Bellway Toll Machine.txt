fsm Control {
  start Init
  on BELL CENTIPEDE APPEAR → Bell Centipede Appear  // from any state

  state Init {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bellbeast Unrescued?
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Start Region", trigger=1, collideTag="", sendEvent=→"START", storeCollider=var "")
    on START → Start
  }

  state Start {
    SendEventToRegister(eventName="REST AREA MUSIC START")
    Trigger2dEvent(gameObject=var "Stop Region", trigger=1, collideTag="", sendEvent=→"STOP", storeCollider=var "")
    on STOP → Stop
  }

  state Stop {
    Trigger2dEvent(gameObject=var "Start Region", trigger=1, collideTag="", sendEvent=→"START", storeCollider=var "")
    SendEventToRegister(eventName="REST AREA MUSIC STOP")
    on START → Start
  }

  state Wait For Hero Enter {
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Bell Centipede Appear {
    SendEventToRegister(eventName="REST AREA MUSIC STOP")
  }

  state Bellbeast Unrescued {
  }

  state Bellbeast Unrescued? {
    PlayerDataBoolTest(boolName="UnlockedFastTravelTeleport", isTrue=→"FINISHED", isFalse=(none))
    CheckQuestStateV2(NotTrackedEvent=(none), IncompleteEvent=→"BELL CENTIPEDE", CompletedEvent=(none), Quest=Bellbeast Rescue (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    on FINISHED → Wait For Hero Enter
    on BELL CENTIPEDE → Bellbeast Unrescued
  }

  var Defeated Bell Centipede: bool = false
  var Start Region: gameObject = Bellway Toll Machine/Music Control Bellway/Start Region
  var Stop Region: gameObject = Bellway Toll Machine/Music Control Bellway/Stop Region
}
