# Control
# 1 location(s):
#   localpoolprefabs_assets_areahangareasong.bundle / Song Knight Projectile

fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wall Break Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fly
  }

  state Fly {
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    DecelerateXY(gameObject=Self, decelerationX=0.97, decelerationY=(unset), brakeOnExit=false)  // disabled
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"WALL", storeCollider=var "", storeForce=(unset))
    on FINISHED → End
    on WALL → Floor?
  }

  sequence state Recycle {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    RecycleSelf()
  }

  state End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wind Slash ProjectileEnd", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    DecelerateXY(gameObject=Self, decelerationX=0.875, decelerationY=(unset), brakeOnExit=true)
    on FINISHED → Recycle
  }

  state Wall End {
    PlayAudioEvent(audioClip=song_knight_white_projectile_impact (sfxstatic_assets_areahangareasong.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Wall Break Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    on FINISHED → End
  }

  state Trail Dir {
    SetParticleEmitterUVFlip(gameObject=var "Pt Trail", flipU=0, flipV=0)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)
    SetParticleEmitterUVFlip(gameObject=var "Pt Trail", flipU=1, flipV=0)
    on FINISHED → Fly
  }

  state Floor? {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=0, tolerance=10, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=90, space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Wall End
  }

  var Rotation: float = 0
  var Wall Break Effect: gameObject = Song Knight Projectile/Wall Break Effect
  var Pt Trail: gameObject = Song Knight Projectile/Pt Trail
  var Pt Break: gameObject = Song Knight Projectile/Pt Break
}
