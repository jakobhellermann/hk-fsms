# RestArea Music Control
# 2 location(s):
#   coremanagers_assets__gamemanager.bundle / _GameManager/AudioManager/Music/RestArea
#   Menu_Title / _GameManager/AudioManager/Music/RestArea

fsm RestArea Music Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    AudioStop(gameObject=Self, fadeTime=0)
    EnableFsmSelf(SetEnabled=false)
    on REST AREA MUSIC START → Act 3 Check
  }

  sequence state Play {
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=3)
    EnableFsmSelf(SetEnabled=false)
    on REST AREA MUSIC STOP → Stop
    on ENTERING SCENE → Stop
    on REST AREA MUSIC STOP FAST → Stop Fast
  }

  state Stop {
    AudioStopV2(gameObject=Self, fadeTime=7, cancelOnEarlyExit=true)
    Wait(time=7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on REST AREA MUSIC START → Play
  }

  state Stop Fast {
    AudioStopV2(gameObject=Self, fadeTime=1, cancelOnEarlyExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on REST AREA MUSIC START → Play
  }

  state Act 3 Check {
    SetAudioClip(gameObject=Self, audioClip=H6-08 Safe (sfxstatic_assets_shared.bundle), autoPlay=false, stopOnExit=false)
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=(none), isFalse=→"FINISHED")
    SetAudioClip(gameObject=Self, audioClip=Shrine Act 3 (sfxstatic_assets_shared.bundle), autoPlay=false, stopOnExit=false)
    on FINISHED → Play
  }
}
