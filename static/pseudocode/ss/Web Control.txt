# Web Control
# 1 location(s):
#   Belltown / Webbed Hero

fsm Web Control {
  start Init

  state Stuck {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Web Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ListenForJump(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), queueBool=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"ATTACK", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    ListenForLeft(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    ListenForDown(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), queueBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForUp(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForRight(eventTarget=Self, wasPressed=→"WIGGLE", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    Wait(time=10, finishEvent=→"SPINNER", realTime=false)
    on SPINNER → Add Spinner
    on WIGGLE → Wiggle
    on ATTACK → Break Antic
  }

  state Break {
    SendEventToRegister(eventName="LOW HEALTH EFFECT STOP")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Web Cut", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Particles", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Threads", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=enemy_damage_old (sfxstatic_assets_areabellshared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=mimic_spider_transormation_stage_5 (sfxstatic_assets_areabell.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    SetPosition(gameObject=var "Blur Plane", vector=(unset), x=(unset), y=(unset), z=var "Initial Blur Plane Z", space=0, everyFrame=false, lateUpdate=false)
    EndFreeCameraMode()
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Webbed Hero (dataassets_assets_assets/dataassets/camerashake.bundle))
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    on FINISHED → Fall
  }

  state Fall {
    CameraBlurPlaneFadeToV2(Spacing=0, Vibrancy=0, MaskLerp=(unset), Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], Duration=1)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=var "Hero", frame=6)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    CheckIsCharacterGrounded(Target=var "Hero", RayCount=3, GroundDistance=0.05, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    on LAND → Land
  }

  state Land {
    CameraBlurPlaneFadeToV2(Spacing=0, Vibrancy=0, MaskLerp=(unset), Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], Duration=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=hero_land_soft (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Relinquish Control
  }

  state Relinquish Control {
    SetPlayerDataBool(boolName="disableInventory", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateInteractible(Target=var "Rest Bench", Activate=true, AllowQueueing=false, UseChildren=false)
  }

  state Wiggle {
    RemoveNeedolinMsg(Msg=var "Spinner Msg", DidAddTracker=var "Added Spinner Msg")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Web Struggle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ObjectJitter(gameObject=var "Threads", x=0.03, y=(unset), z=(unset), allowMovement=false, limitFps=0)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=grass_cut_1 (sfxstatic_assets_grass.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Stuck
  }

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Particles", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Threads", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    PlayerDataVariableTest(VariableName="spinnerDefeated", ExpectedValue=true, IsExpectedEvent=→"CANCEL", IsNotExpectedEvent=(none))
    SendRandomEvent(events=[→"CANCEL", (none)], weights=[0.6, 0.4], delay=0)
    CheckIsFirstLevelForPlayer(trueEvent=→"TAKE CONTROL", falseEvent=(none), storeValue=(unset))
    on TAKE CONTROL → Set Blur
    on CANCEL → Inert
  }

  state Take Control {
    SetPlayerDataBool(boolName="disableInventory", value=true)
    SendEventToRegister(eventName="CANCEL BENCH")
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Threads", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    GetPosition2D(gameObject=Self, vector=var "Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Hero", Vector=var "Pos", X=0, Y=0, Space=0, EveryFrame=false)
    on FINISHED → Position Camera
  }

  state Wait Enter {
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Wait Frame
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Take Control
  }

  state Break Antic {
    SetDarknessLevel(SetLevel=var "Initial Darkness Level", StoreCurrentLevel=(unset))
    RemoveNeedolinMsg(Msg=var "Spinner Msg", DidAddTracker=var "Added Spinner Msg")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=enemy_damage_old (sfxstatic_assets_areabellshared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    PlayRandomAudioClipTableV2(Table=Attack Heavy Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=true)
    ObjectJitter(gameObject=var "Threads", x=0.05, y=(unset), z=(unset), allowMovement=false, limitFps=0)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Web Cut Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Break
  }

  state Inert {
    on TAKE CONTROL → Set Blur
  }

  state Add Spinner {
    AddNeedolinMsg(Msg=var "Spinner Msg", DidAddTracker=var "Added Spinner Msg")
    on FINISHED → Stuck
  }

  state Set Blur {
    CameraBlurPlaneFade(From={Spacing=0, Vibrancy=0, MaskLerp=0}, To={Spacing=30, Vibrancy=-1, MaskLerp=0}, Curve=curve[(time=0, value=0, inSlope=0, outSlope=1), (time=1, value=1, inSlope=1, outSlope=0)], Duration=0)
    GetPosition(gameObject=var "Blur Plane", vector=(unset), x=(unset), y=(unset), z=var "Initial Blur Plane Z", space=0, everyFrame=false)
    SetPosition(gameObject=var "Blur Plane", vector=(unset), x=(unset), y=(unset), z=5.8, space=0, everyFrame=false, lateUpdate=false)
    SetPlayerDataBool(boolName="disableInventory", value=true)
    on FINISHED → Wait Enter
  }

  state Fade Blur {
    CameraBlurPlaneFadeToV2(Spacing=(unset), Vibrancy=0, MaskLerp=1, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=2, outSlope=2)], Duration=var "Slow Fade Duration")
    Wait(time=var "Slow Fade Duration", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake
  }

  state Wait {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Web Sleep", animationTriggerEvent=(none), animationCompleteEvent=(none))
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Webbed Hero (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    SendEventToRegister(eventName="LOW HEALTH EFFECT START")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Blur
  }

  sequence state Position Camera {
    CameraRepositionToHero()
    NextFixedFrameEvent(SendEvent=(none))
    NextFixedFrameEvent(SendEvent=(none))
    StartFreeCameraMode(StoreObject=var "Camera")
    SetPositionToObject2D(gameObject=var "Camera", targetObject=var "Self", xOffset=(unset), yOffset=(unset), everyFrame=false)
    SetDarknessLevel(SetLevel=2, StoreCurrentLevel=var "Initial Darkness Level")
    on FINISHED → Wait
  }

  sequence state Wake {
    Wait(time=1, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName="Web Wake", AnimationCompleteEvent=(none))
    on FINISHED → Wiggle
  }

  var Blur Plane: gameObject = BlurPlane

  var Slow Fade Duration: float = 8
  var Initial Blur Plane Z: float = 0
  var Initial Darkness Level: int = 0
  var Added Spinner Msg: bool = false
  var Pos: vector2 = (0, 0)
  var Particles: gameObject = Webbed Hero/Particles
  var Self: gameObject = <null>
  var Threads: gameObject = Webbed Hero/Threads
  var Camera: gameObject = <null>
  var Rest Bench: gameObject = RestBench
  var Spinner Msg: object = Hornet Strung Up (dataassets_assets_assets/dataassets/localisedtextcollections.bundle)
}
