# Control
# 1 location(s):
#   Memory_Needolin / Memory Weaver Hang (2)

fsm Control {
  start Pause

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hang Weaver Idle")
    Trigger2dEvent(gameObject=var "Retreat Range", trigger=0, collideTag="", sendEvent=→"RETREAT", storeCollider=var "")
    on RETREAT → Retreat Pause
  }

  state Retreat {
    PlayRandomAudioClipTableV3(Table=Memory_Weaver_leave (sfxdynamic_assets_areacoralareamemory.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hang Weaver Retreat")
    AccelerateToY(gameObject=Self, accelerationFactor=0.75, targetSpeed=18)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEventRandom(audioClips=[widow_insectoid_walk_backwards_on_ground (sfxstatic_assets_areabell.bundle), widow_insectoid_walk_sideways_on_web (sfxstatic_assets_areabell.bundle)], pitchMin=0.85, pitchMax=1.45, volume=0.25, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Disappear
  }

  state Disappear {
    Tk2dSpriteTweenColor(gameObject=Self, color=(0, 0, 0, 0), duration=0.3)
    PlayParticleEmitter(gameObject=var "Pt Disappear", emit=0, resetIfPlaying=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Shuffle Z
  }

  state Shuffle Z {
    RandomFloat(min=-0.001, max=0.001, storeResult=var "Z Translate")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Translate", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Idle
  }

  state Retreat Pause {
    WaitRandom(timeMin=0.3, timeMax=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Retreat
  }

  var Z Translate: float = 0
  var Retreat Range: gameObject = Memory Weaver Hang (2)/Retreat Range
  var Pt Disappear: gameObject = Memory Weaver Hang (2)/Pt Disappear
}
