# Control
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Flea Game - Dodging/Giant Flea DodgeGame

fsm Control {
  start Init
  on RESET FLEA GAMES → Dormant  // from any state

  state Init {
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Idle Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Msg", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on ENTER → Choose Side
  }

  state Choose Side {
    SendRandomEventV4(events=[→"L", →"R"], weights=[0.5, 0.5], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on L → Enter L
    on R → Enter R
  }

  state Enter L {
    SetPosition2d(gameObject=Self, vector=(unset), x=166, y=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Translate X", floatValue=-10, everyFrame=false)
    on FINISHED → Choose Height
  }

  state Enter R {
    SetPosition2d(gameObject=Self, vector=(unset), x=189, y=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Translate X", floatValue=10, everyFrame=false)
    on FINISHED → Choose Height
  }

  state Choose Height {
    SendRandomEventV4(events=[→"LOW", →"HIGH"], weights=[0.5, 0.5], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on LOW → Enter Low
    on HIGH → Enter High
  }

  state Enter Low {
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "Low Pos", space=1, everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Target Y", floatValue=var "High Pos", everyFrame=false)
    on FINISHED → Fly In
  }

  state Enter High {
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "High Pos", space=1, everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Target Y", floatValue=var "Low Pos", everyFrame=false)
    on FINISHED → Fly In
  }

  state Fly In {
    PlayRandomAudioClipTable(Table=enemy_fly_in (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=var "Entry X", y=var "Entry Y", space=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Giant Fly")
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Idle Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=var "Translate X", y=var "Translate Y", z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    AnimateYPositionTo(GameObject=Self, ToValue=var "Entry Y", localSpace=false, time=0.6, speed=(unset), delay=(unset), easeType=4, reverse=false, finishEvent=(none), realTime=false)
    AnimateXPositionTo(GameObject=Self, ToValue=var "Entry X", localSpace=false, time=0.6, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    on FINISHED → Roar
  }

  state Roar {
    PlayRandomAudioClipTable(Table=Giant Flea howl short  (sfxdynamic_assets_areaaqueduct.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Giant Sing", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Antic
  }

  state Antic {
    PlayAudioEvent(audioClip=swamp_muckman_jump_1 (sfxstatic_assets_areaswamp.bundle), pitchMin=1.2, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Giant Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocityByScale(gameObject=Self, speed=20, ySpeed=(unset), everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.85, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Charge
  }

  state Charge {
    ActivateGameObject(gameObject=var "Slam Msg", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=beastfly_horiz_dash_attack (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Giant Charge")
    SetVelocityByScale(gameObject=Self, speed=var "Charge Speed", ySpeed=(unset), everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Idle Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    PlayRandomAudioClipTable(Table=Giant Flea bark  (sfxdynamic_assets_fleacaravan.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Turn
  }

  state Turn {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    PlayAudioEvent(audioClip=beastfly_horiz_dash_attack (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Giant Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.85, decelerationY=(unset), brakeOnExit=false)
    AnimateYPositionTo(GameObject=Self, ToValue=var "Target Y", localSpace=true, time=0.4, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Charge Back
  }

  state Charge Back {
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "Target Y", space=1, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Giant Charge")
    SetVelocityByScale(gameObject=Self, speed=var "Charge Speed", ySpeed=(unset), everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=beastfly_horiz_dash_attack (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayRandomAudioClipTable(Table=Giant Flea bark  (sfxdynamic_assets_fleacaravan.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Score
  }

  state Score {
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    SendEventToRegister(eventName="SCORE")  // disabled
    SpawnObjectFromGlobalPool(gameObject=Flea Game Score Effect (localpoolprefabs_assets_fleacaravan.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")  // disabled
    on FINISHED → Dormant
  }

  var Low Pos: float = 16.25
  var High Pos: float = 19
  var Target Y: float = 0
  var Translate X: float = 0
  var Entry X: float = 0
  var Entry Y: float = 0
  var Translate Y: float = 5
  var Charge Speed: float = -30
  var Idle Collider: gameObject = Caravan_States/Flea Festival/Flea Game - Dodging/Giant Flea DodgeGame/Idle Collider
  var Self: gameObject = Caravan_States/Flea Festival/Flea Game - Dodging/Giant Flea DodgeGame
  var Slam Msg: gameObject = Caravan_States/Flea Festival/Flea Game - Dodging/Giant Flea DodgeGame/Slam Msg
}
