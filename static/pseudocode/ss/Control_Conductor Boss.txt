  on STUN → Stun Hit  // from any state

  }

  state Choice {

    on SHOOT → Set Shoot
    on BARRAGE → Spear Barrage Pause
    on ROAR → Roar Tele Antic
  }

  state Set Charge {
    SetStringValue(stringVariable=var "Next Event", stringValue="CHARGE", everyFrame=false)
    SendRandomEvent(events=[→"TELE L", →"TELE R"], weights=[0.5, 0.5], delay=0)
