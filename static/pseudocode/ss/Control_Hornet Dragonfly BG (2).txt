# Control
# 1 location(s):
#   Clover_16 / Hornet Dragonfly BG (2)

fsm Control {
  start Init
  on RECOIL → Recoil Wait  // from any state
  on DISABLE → Stop  // from any state

  state Init {
    SetParent(gameObject=var "Nodes Parent", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Move Wait
  }

  state Get Child {
    GetChildCount(gameObject=var "Nodes Parent", storeResult=var "Node Count")
    IntAddWrapped(Value=var "Node Index", AddValue=var "Node Index Add", Min=0, Max=var "Node Count", InclusiveMax=false, StoreResult=var "Node Index")
    GetChildNum(gameObject=var "Nodes Parent", childIndex=var "Node Index", store=var "Next Node")
    GameObjectIsActive(Target=var "Next Node", ActiveSpace=1, trueEvent=(none), falseEvent=→"LOOP", storeValue=(unset))
    on FINISHED → Antic
    on LOOP → Get Child
  }

  state Move Pos {
    PlayAudioEventRandom(audioClips=[dragonfly_dash_1 (sfxstatic_assets_areaclover.bundle), dragonfly_dash_2 (sfxstatic_assets_areaclover.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Current Audio")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    RandomFloat(min=1, max=1, storeResult=var "Wait Time")
    FloatAdd(floatVariable=var "Wait Time", add=-1, everyFrame=true, perSecond=true)
    GetPosition2dAs3d(SourceXY=var "Next Node", SourceZ=Self, StoreVector=var "Next Pos")
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Next Pos", time=(unset), delay=0, speed=6, space=0, easeType=10, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Move Wait
  }

  state Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Antic")
    FaceObjectV2(objectA=Self, objectB=var "Next Node", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Large TurnToAntic", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Move Pos
  }

  state Move Wait {
    WaitRandom(timeMin=0.2, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)  // disabled
    FloatAdd(floatVariable=var "Wait Time", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Wait Time", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Reverse?
  }

  state Reverse? {
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.75, 0.25], eventMax=[5, 1], missedMax=[1, 5], activeBool=(unset))
    IntOperator(integer1=var "Node Index Add", integer2=-1, operation=Multiply, storeResult=var "Node Index Add", everyFrame=false)
    on FINISHED → Get Child
  }

  state Recoil Wait {
    on RECOIL END → Reverse?
  }

  state Stop {
    AudioStop(gameObject=var "Current Audio", fadeTime=0)
  }

  var Wait Time: float = 0
  var Node Count: int = 0
  var Node Index: int = 0
  var Node Index Add: int = 1
  var Next Pos: vector3 = (0, 0, 0)
  var Nodes Parent: gameObject = Hornet Dragonfly BG (2)/Nodes
  var Next Node: gameObject = <null>
  var Current Audio: gameObject = <null>
}
