# Control
# 1 location(s):
#   Tut_01 / Tutorial Intro Cutscene/Jump Reminder

fsm Control {
  start Init Pause

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=8.1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"PERFORMED", everyFrame=true)
    on ENTER → Pause
    on PERFORMED → Inactive
  }

  state Send {
    SendEventToRegister(eventName="REMINDER JUMP")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=8.1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"PERFORMED", everyFrame=true)
    on PERFORMED → End
  }

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Y", float2=8.1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"PERFORMED", everyFrame=true)
    on FINISHED → Send
    on PERFORMED → Inactive
  }

  state Inactive {
  }

  state End {
    SendEventToRegister(eventName="DID JUMP")
  }

  state Init Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Hero Y: float = 0
  var Control Reminder: gameObject = <null>
}
