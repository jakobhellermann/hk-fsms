  }

    on WALL R → Wall R
    on WALL L → Wall L
    on SPIKES → Spike Hit
    on SINK → Sink
  }

    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Wounded Bounce to Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=var "Sprite")

  }

  state Hit Right {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    FlingObject(flungObject=Self, speedMin=25, speedMax=30, angleMin=60, angleMax=80)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=var "Sprite")

  }

    SetIsKinematic2d(gameObject=Self, isKinematic=false)

    FloatAdd(floatVariable=var "Time", add=0.05, everyFrame=true, perSecond=true)
    SampleCurve(curve=curve[(time=0, value=0, inSlope=4.822938, outSlope=4.822938), (time=0.20214844, value=0.97494936, inSlope=-1.7944412, outSlope=-1.7944412), (time=0.43076375, value=-0.94812155, inSlope=-0.6529417, outSlope=-0.6529417), (time=0.62706923, value=0.44681293, inSlope=0.6642451, outSlope=0.6642451), (time=0.7761646, value=-0.41457754, inSlope=-1.9626466, outSlope=-1.9626466), (time=1, value=0, inSlope=4.2766485, outSlope=4.2766485)], sampleAt=var "Time", storeValue=var "Force", everyFrame=true)

    FlingObject(flungObject=Self, speedMin=28, speedMax=30, angleMin=260, angleMax=280)
    on HIT_UP → Hit Up
    on FINISHED → Hit Voice
  }

    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Water Surface Y", space=0, everyFrame=false)
    FloatSubtract(floatVariable=var "Water Surface Y", subtract=0.1, everyFrame=false, perSecond=false)
    CallMethodProper(gameObject=Self, behaviour="SurfaceWaterFloater", methodName="Sink", parameters=[], storeResult=(unset var), EveryFrame=false)

  }

  }

    CallMethodProper(gameObject=Self, behaviour="SurfaceWaterFloater", methodName="StopSink", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → In Water
  }

    AnimatorPlay(gameObject=var "Spa Door Anim", stateName="Open", layer=(unset), normalizedTime=(unset), everyFrame=false)

    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Wounded Hit")

    NextFrameEvent(sendEvent=→"FINISHED")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=false)

    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=var "Velocity X", y=4, vector3=(unset), everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=false)

  }

    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Wounded Sing")

    CheckXPosition(gameObject=Self, compareTo=50, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"FINISHED", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
