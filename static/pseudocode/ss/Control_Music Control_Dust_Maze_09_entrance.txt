# Control
# 1 location(s):
#   Dust_Maze_09_entrance / Music Control

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Leaving Maze Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Start Range")
    FindNamedChild(gameObject=Self, storeResult=var "End Range")
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check Entry Type
  }

  state Fade Prev Down {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 2
  }

  state Tension Playing {
    ApplyMusicCue(musicCue=MistMaze (audiocuesdynamic_assets_areadust.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Action Only (audioobjects_assets_all.bundle), transitionTime=3)
    Trigger2dEvent(gameObject=var "Leaving Maze Collider", trigger=0, collideTag="", sendEvent=→"LEAVE MAZE", storeCollider=var "")
    Trigger2dEvent(gameObject=var "End Range", trigger=0, collideTag="", sendEvent=→"END", storeCollider=var "")
    on LEAVE MAZE → Leaving Maze
    on END → Tension Ending
  }

  state Leaving Maze {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=1)
  }

  state State 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    on FINISHED → Wait For Range
  }

  state State 2 {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0.25)
    Wait(time=0.25, finishEvent=(none), realTime=false)
    on FINISHED → State 1
  }

  state Wait For Range {
    Trigger2dEvent(gameObject=var "Start Range", trigger=1, collideTag="", sendEvent=→"START", storeCollider=var "")
    on START → Tension Playing
  }

  state Check Entry Type {
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="right1", equalEvent=→"ENTERING", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Entry Gate", compareTo="left1", equalEvent=→"EXITING", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on ENTERING → Fade Prev Down
    on EXITING → Tension Playing
    on FINISHED → Wake Pause
  }

  state Wake Pause {
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    Wait(time=3, finishEvent=(none), realTime=false)
    on FINISHED → Tension Playing
  }

  state Tension Ending {
    Trigger2dEvent(gameObject=var "Start Range", trigger=1, collideTag="", sendEvent=→"START", storeCollider=var "")
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=8)
    Trigger2dEvent(gameObject=var "Leaving Maze Collider", trigger=0, collideTag="", sendEvent=→"LEAVE MAZE", storeCollider=var "")
    on LEAVE MAZE → Leaving Maze
    on START → Tension Playing
  }

  var Entry Gate: string = ""
  var Leaving Maze Collider: gameObject = <null>
  var Start Range: gameObject = <null>
  var End Range: gameObject = <null>
}
