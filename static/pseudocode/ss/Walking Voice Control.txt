# Walking Voice Control
# 3 location(s):
#   Belltown / Pilgrim Fisher NPC Walker
#   Belltown_07 / Pilgrim Fisher NPC Walker
#   Shellwood_01 / Pilgrim Fisher NPC Walker

fsm Walking Voice Control {
  start Stop

  state Stop {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    on WALK START → Start
  }

  state Start {
    SetRandomAudioClipFromTable(Table=Reed_hum_loop (sfxdynamic_assets_areabellareashellwood.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=1)
    on WALK STOP → Stop
  }

  var Audio Loop Voice: gameObject = Pilgrim Fisher NPC Walker/Audio Loop Voice
}
