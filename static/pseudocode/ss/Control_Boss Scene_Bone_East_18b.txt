# Control
# 1 location(s):
#   Bone_East_18b / Boss Scene

fsm Control {
  start Pause

  state Pause {
    FindNamedChild(gameObject=Self, storeResult=var "TrapBench")
    FindNamedChild(gameObject=Self, storeResult=var "Unencountered Floor")
    FindNamedChild(gameObject=Self, storeResult=var "Re Encounter Remnants")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Bone Hunter Trapper")
    FindNamedChild(gameObject=Self, storeResult=var "Gates")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock B")
    FindNamedChild(gameObject=Self, storeResult=var "Start Range")
    PlayerDataBoolTest(boolName="defeatedAntTrapper", isTrue=→"DEFEATED", isFalse=(none))
    PlayerDataBoolTest(boolName="encounteredAntTrapper", isTrue=→"ENCOUNTERED", isFalse=→"UNENCOUNTERED")
    on DEFEATED → Defeated
    on ENCOUNTERED → Encountered
    on UNENCOUNTERED → Unencountered
  }

  state Defeated {
    ActivateGameObject(gameObject=var "Bone Hunter Trapper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "TrapBench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Unencountered Floor", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Re Encounter Remnants", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Gate Sfx
  }

  state Enter Pause {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    GetPlayerDataBool(boolName="encounteredAntTrapper", storeValue=var "Encountered")
    ConvertBoolToFloat(boolVariable=var "Encountered", floatVariable=var "Enter Pause", falseValue=2, trueValue=0.75, everyFrame=false)
    SetPlayerDataVariable(VariableName="encounteredAntTrapper", SetValue=true)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock B", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=var "Enter Pause", finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Unencountered Floor", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Start Battle
  }

  state Start Battle {
    SendEventByName(eventTarget=GameObject(var "Bone Hunter Trapper"), sendEvent="START", delay=0, everyFrame=false)
    on BATTLE END → End Pause
  }

  state End Battle {
    ActivateGameObject(gameObject=var "CamLock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state End Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Battle
  }

  state Unencountered {
    ActivateGameObject(gameObject=var "TrapBench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Unencountered Floor", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Re Encounter Remnants", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait For Bench Break
  }

  state Encountered {
    ActivateGameObject(gameObject=var "TrapBench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Unencountered Floor", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Re Encounter Remnants", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Wait For Bench Break {
    on BENCH BREAK → Idle
  }

  state Gate Sfx {
    PlayerDataBoolTest(boolName="encounteredAntTrapper", isTrue=(none), isFalse=→"FINISHED")
    PlayAudioEvent(audioClip=mantis_lords_gate_close (sfxstatic_assets_battle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Enter Pause
  }

  var Enter Pause: float = 0
  var Encountered: bool = false
  var Bone Hunter Trapper: gameObject = <null>
  var Gates: gameObject = <null>
  var CamLock B: gameObject = <null>
  var Start Range: gameObject = <null>
  var TrapBench: gameObject = <null>
  var Unencountered Floor: gameObject = <null>
  var Cocoon Barb Trap: gameObject = <null>
  var Re Encounter Remnants: gameObject = <null>
}
