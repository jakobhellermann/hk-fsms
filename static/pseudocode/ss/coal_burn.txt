// uses template: coal_burn
fsm coal_burn {
  start Init
  on CANCEL BURN → Canceled Burn  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Small Burn")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Burn Up")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Coal", collideLayer=Attack(17), sendEvent=→"ENTER", storeCollider=var "Coal Region")
    on ENTER → Burn Pause
  }

  state Burn Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="Coal", collideLayer=Attack(17), sendEvent=→"EXIT", storeCollider=var "Coal Region")
    on FINISHED → Small Burn
    on EXIT → Unburn
  }

  state Small Burn {
    PlayParticleEmitterChildren(gameObject=var "Pt Small Burn", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(0.1544118, 0.12829803, 0.12829803, 1), duration=1.5)
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="Coal", collideLayer=Attack(17), sendEvent=→"EXIT", storeCollider=var "Coal Region")
    SpawnObjectFromGlobalPool(gameObject=Audio Player Actor Loop (audioobjects_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Audio Player")
    SetAudioClip(gameObject=var "Audio Player", audioClip=hot_coal_loop (sfxstatic_assets_areadocks.bundle), autoPlay=false, stopOnExit=false)
    RandomFloat(min=0.85, max=1.15, storeResult=var "Pitch")
    SetAudioPitch(gameObject=var "Audio Player", pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Player", volume=0.1, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Player", startVolume=0, endVolume=0.75, time=1)
    on FINISHED → Burn Up
    on EXIT → Unburn
  }

  state Burn Up {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Burn Up", resetTimeIfPlaying=false, stopOnStateExit=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(0, 0, 0, 0), duration=0.5)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SendMessageV2(gameObject=Self, delivery=SendMessage, options=1, functionCall=BurnUp(), everyFrame=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -0.5, 0), time=0.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=→"FINISHED", realTime=false)
    FadeAudio(gameObject=var "Audio Player", startVolume=1, endVolume=0.5, time=0.5)
    on FINISHED → Burnt
  }

  sequence state Burnt {
    SetMeshRenderer(gameObject=Self, active=false)
    FadeAudio(gameObject=var "Audio Player", startVolume=0.5, endVolume=0, time=0.5)
    AudioStop(gameObject=var "Audio Player", fadeTime=0)
  }

  state Unburn {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Coal", collideLayer=Attack(17), sendEvent=→"ENTER", storeCollider=var "Coal Region")
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.5)
    AudioStop(gameObject=var "Audio Player", fadeTime=0)
    on ENTER → Burn Pause
  }

  state Canceled Burn {
    StopParticleEmittersInChildren(gameObject=var "Pt Burn Up")
    StopParticleEmittersInChildren(gameObject=var "Pt Small Burn")
    AudioStop(gameObject=var "Audio Player", fadeTime=0)
  }

  var Pitch: float = 0
  var Coal Region: gameObject = <null>
  var Pt Small Burn: gameObject = <null>
  var Pt Burn Up: gameObject = <null>
  var Audio Player: gameObject = <null>
  var Self: gameObject = <null>
}
