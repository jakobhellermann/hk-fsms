# Rotater
# 1 location(s):
#   enemycorpses_assets_areashellwood.bundle / Corpse Stick Insect

// uses template: corpse_particle_rotater
fsm Rotater {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    FindNamedChild(gameObject=Self, storeResult=var "particles")
    on FINISHED → Fling
  }

  state Fling {
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Wait(time=0.1, finishEvent=(none), realTime=false)  // disabled
    on FINISHED → State 1
  }

  state State 1 {
    PlayParticleEmitter(gameObject=var "particles", emit=0, resetIfPlaying=false)
  }

  var Angle: float = 0
  var particles: gameObject = <null>
}
