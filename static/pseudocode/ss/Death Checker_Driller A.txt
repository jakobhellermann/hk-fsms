# Death Checker
# 1 location(s):
#   Coral_11 / Black Thread States Thread Only Variant/Normal World/Boss Scene/Driller A

fsm Death Checker {
  start Init

  state Idle {
    on TOOK DAMAGE → Count HP
    on READY TO DIE → Ready To Die
  }

  state Count HP {
    GetHP(target=Self, storeValue=var "Self HP")
    GetHP(target=var "Buddy", storeValue=var "Buddy HP")
    IntAdd(intVariable=var "Buddy HP", add=var "Self HP", everyFrame=false)
    IntOperator(integer1=var "Buddy HP", integer2=var "Self HP", operation=Add, storeResult=var "Total HP", everyFrame=false)
    IntCompare(integer1=var "Buddy HP", integer2=0, equal=→"READY TO DIE", lessThan=→"READY TO DIE", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on READY TO DIE → Ready To Die
  }

  state Ready To Die {
    SendEventByName(eventTarget=GameObject(var "Buddy"), sendEvent="READY TO DIE", delay=0, everyFrame=false)
    on TOOK DAMAGE → Notify Parent
  }

  state Notify Parent {
    GetHeroCStateEvent(VariableName="dead", StoreValue=(unset), TrueEvent=→"CANCEL", FalseEvent=(none), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(var "Boss Scene"), sendEvent="KILL HIT", delay=0, everyFrame=false)
    on CANCEL → Ready To Die
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    on FINISHED → Idle
  }

  var Buddy: gameObject = Black Thread States Thread Only Variant/Normal World/Boss Scene/Driller B

  var Buddy HP: int = 0
  var Self HP: int = 0
  var Total HP: int = 0
  var Boss Scene: gameObject = <null>
}
