    on FINISHED → Talk 2 Start
    on IS_HURT → Is Hurt
  }

    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=Self, UseChildren=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Death
  }

    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)

  }

  }

  var Hero Is Hurt: bool = false
