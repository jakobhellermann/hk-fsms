fsm Control {
  start Pause

  state Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    GetParent(gameObject=var "CameraParent", storeResult=var "GameCameras")
    FindNamedChild(gameObject=var "GameCameras", storeResult=var "Camera Target")
    FindNamedChild(gameObject=Self, storeResult=var "Cam Start Point")
    FindNamedChild(gameObject=Self, storeResult=var "Cam End Point")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock 1")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock 2")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock 3")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock 4")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock 5")
    GetPosition2D(gameObject=var "Cam Start Point", vector=(unset), x=var "Cam Start X", y=var "Cam Start Y", space=0, everyFrame=false)
    GetPosition2D(gameObject=var "Cam End Point", vector=(unset), x=var "Cam End X", y=var "Cam End Y", space=0, everyFrame=false)
    on FINISHED → Init
  }

  state Free {

    SetFloatValue(floatVariable=var "Cam Y", floatValue=8.25, everyFrame=true)
    FloatOperator(float1=var "Cam X", float2=0.2839, operation=Multiply, storeResult=var "Y Add Value", everyFrame=true)
    FloatAdd(floatVariable=var "Cam Y", add=var "Y Add Value", everyFrame=true, perSecond=false)
    CameraFollowYInState(Target=Self, KeepOffset=false)  // disabled
    GetPosition(gameObject=Self, vector=(unset), x=var "Cam X", y=var "Cam Y", z=(unset), space=0, everyFrame=true)  // disabled
    SetCamLock(xLockMin=(unset), xLockMax=(unset), yLockMin=var "Cam Y", yLockMax=var "Cam Y", everyFrame=true)
    Trigger2dEvent(gameObject=var "CamLock 1", trigger=1, collideTag="", sendEvent=→"CAMLOCK 1", storeCollider=var "")
    Trigger2dEvent(gameObject=var "CamLock 2", trigger=0, collideTag="", sendEvent=→"CAMLOCK 2", storeCollider=var "")
    Trigger2dEvent(gameObject=var "CamLock 3", trigger=0, collideTag="", sendEvent=→"CAMLOCK 3", storeCollider=var "")
    Trigger2dEvent(gameObject=var "CamLock 4", trigger=0, collideTag="", sendEvent=→"CAMLOCK 4", storeCollider=var "")
    Trigger2dEvent(gameObject=var "CamLock 5", trigger=1, collideTag="", sendEvent=→"CAMLOCK 5", storeCollider=var "")
    on CAMLOCK 1 → CamLock 1
    on CAMLOCK 2 → CamLock 2
    on CAMLOCK 3 → CamLock 3
    on CAMLOCK 4 → CamLock 4
    on CAMLOCK 5 → CamLock 5
  }

  state Init {

    on FINISHED → Free
  }

  state CamLock 1 {
    SetCamLock(xLockMin=(unset), xLockMax=(unset), yLockMin=10.5, yLockMax=10.5, everyFrame=true)
    Trigger2dEvent(gameObject=var "CamLock 1", trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Free
  }

  state CamLock 2 {
