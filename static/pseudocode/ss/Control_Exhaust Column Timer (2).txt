# Control
# 2 location(s):
#   Library_12b / Exhaust Column Timer (2)
#   Organ_01 / Exhaust Column Timer (1)

fsm Control {
  start Init
  on EXHAUST END → Exhaust End  // from any state

  state Init {
    Wait(time=1, finishEvent=(none), realTime=false)
    FindNamedChild(gameObject=Self, storeResult=var "Shake Region Down")
    FindNamedChild(gameObject=Self, storeResult=var "Shake Region Up")
    FindNamedChild(gameObject=Self, storeResult=var "Shake Region Block")
    SetFsmBool(gameObject=var "Shake Region Down", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    PlayerDataBoolTest(boolName="defeatedPhantom", isTrue=→"EXHAUST END", isFalse=(none))
    on FINISHED → Down Antic
    on EXHAUST END → Exhaust End
  }

  state Down Antic {
    SendEventToRegister(eventName="STEAM DOWN ANTIC")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  state Down {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="STEAM DOWN ON")
    SetFsmBool(gameObject=var "Shake Region Down", fsmName="Shaker", variableName="Do Shake", setValue=true, everyFrame=false)
    on FINISHED → Down End
  }

  state Down End {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="STEAM DOWN OFF")
    SetFsmBool(gameObject=var "Shake Region Down", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    on FINISHED → Up Antic
  }

  state Up Antic {
    SendEventToRegister(eventName="STEAM UP ANTIC")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
  }

  state Up {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="STEAM UP ON")
    SetFsmBool(gameObject=var "Shake Region Up", fsmName="Shaker", variableName="Do Shake", setValue=true, everyFrame=false)
    on FINISHED → Up End
  }

  state Up End {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="STEAM UP OFF")
    SetFsmBool(gameObject=var "Shake Region Up", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    on FINISHED → Down Antic
  }

  state Exhaust End {
    SendEventToRegister(eventName="ALL EXHAUST OFF")
    SendEventToRegister(eventName="BLOCKER OFF")
    SetFsmBool(gameObject=var "Shake Region Down", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Shake Region Up", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Shake Region Block", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
  }

  var Shake Region Down: gameObject = <null>
  var Shake Region Up: gameObject = <null>
  var Shake Region Block: gameObject = <null>
  var Self: gameObject = <null>
}
