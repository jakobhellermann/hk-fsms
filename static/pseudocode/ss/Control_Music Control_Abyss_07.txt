# Control
# 2 location(s):
#   Abyss_07 / Music Control
#   Abyss_12 / Music Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="HasWhiteFlower", isTrue=→"LAST DIVE", isFalse=→"SILENT")
    on LAST DIVE → Last Dive
    on SILENT → Silent
  }

  state Last Dive {
    ApplyMusicCue(musicCue=LastDivePrologue (audiocuesdynamic_assets_areaabyss.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal Soft (audioobjects_assets_all.bundle), transitionTime=4)
  }

  state Silent {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=8)
  }
}
