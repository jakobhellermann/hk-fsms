# Control
# 1 location(s):
#   Greymoor_15b / Memory Orb Challenge/memory_font

fsm Control {
  start Pause
  on COMPLETED → Completed  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Silkfly Cloud")
    on FINISHED → Spawn Silkflies
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → In Range
  }

  state Completed {
    ActivateGameObject(gameObject=var "Uncompleted Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Completed Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
  }

  state In Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.5, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on EXIT → Idle
    on NEEDOLIN → Needolin Start
    on INTERACT → Hero Powerup
  }

  state Needolin Start {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=true, UseChildren=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=2, finishEvent=→"NEXT", realTime=false)
    FadeColorFader(target=var "Summon Glow", fadeType=UP, useChildren=true)
    PlayParticleEmitterInState(gameObject=var "Pt Summon")
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumStart())
    on END → Needolin End
    on NEXT → Wake Cloud
  }

  state Needolin End {
    FadeColorFader(target=var "Summon Glow", fadeType=DOWN, useChildren=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    on FINISHED → Idle
  }

  state Wake Cloud {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Splash Out", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Silkfly Cloud", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ScreenFlash(flashColour=(1, 1, 1, 0.497))
    FadeColorFader(target=var "Summon Glow", fadeType=DOWN, useChildren=true)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    on FINISHED → Challenge Active
  }

  state Challenge Active {
    ActivateGameObject(gameObject=var "Uncompleted Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Completed Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on COMPLETE → Completed
    on TIME OUT → Reset
  }

  state Spawn Silkflies {
    CreateObjectsRandom(SpawnPool=[Silkfly Ambient NP (localpoolprefabs_assets_areagreymoor.bundle)], AmountMin=1, AmountMax=2, SpawnPoint=var "Silkfly Spawn Point", xSpread=0, ySpread=0, Activated=true, StoreArray=[])
    on FINISHED → Idle
  }

  state Reset {
    ActivateGameObject(gameObject=var "Splash Out", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Uncompleted Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Reset Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Completed Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  sequence state Hero Powerup {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimationWait(Target=var "Hero", ClipName="Collect Stand 1", AnimationCompleteEvent=(none))
    Wait(time=0.3, finishEvent=(none), realTime=false)
    HeroChargeEffectStart(TintColor=(1, 1, 1, 1))
    HeroChargeEffectRespond(ChargeBurstEvent=→"CHARGE BURST", ChargeEndEvent=(none))
    on CHARGE BURST → Heal
  }

  state End Dialogue {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → In Range
  }

  state Heal {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="MaxHealthKeepBlue", parameters=[], storeResult=(unset var), EveryFrame=false)
    FindChild(gameObject=var "Hero", childName="Bind Effects/Pt Heal", storeResult=var "Pt Heal")
    PlayParticleEmitter(gameObject=var "Pt Heal", emit=0, resetIfPlaying=false)
    HeroChargeEffectRespond(ChargeBurstEvent=(none), ChargeEndEvent=→"CHARGE END")
    on CHARGE END → End Dialogue
  }

  var Completed Effects: gameObject = Memory Orb Challenge/memory_font/Completed
  var Parent: gameObject = <null>
  var Pt Summon: gameObject = Memory Orb Challenge/memory_font/Pt Summon
  var Reset Glow: gameObject = Memory Orb Challenge/memory_font/Reset Glow
  var Silkfly Cloud: gameObject = <null>
  var Silkfly Spawn Point: gameObject = Memory Orb Challenge/memory_font/Silkfly Spawn Point
  var Splash Out: gameObject = Memory Orb Challenge/memory_font/Splash Out
  var Summon Glow: gameObject = Memory Orb Challenge/memory_font/Summon Glow
  var Uncompleted Effects: gameObject = Memory Orb Challenge/memory_font/Uncompleted
  var Pt Heal: gameObject = <null>
  var Strings: gameObject = Memory Orb Challenge/memory_font/Strings
}
