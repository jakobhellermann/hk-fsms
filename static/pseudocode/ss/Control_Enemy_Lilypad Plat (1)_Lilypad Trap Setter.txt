# Control
# 1 location(s):
#   Clover_04b / Lilypad Trap Setter/Lilypad Plat (1)/Enemy

fsm Control {
  start Snap 1
  on ZERO HP → Kill Parent  // from any state
  on DISABLE → OnDIsable  // from any state

  state Snap 1 {
    FindNamedChild(gameObject=Self, storeResult=var "Collider 1")
    FindNamedChild(gameObject=Self, storeResult=var "Collider 2")
    FindNamedChild(gameObject=Self, storeResult=var "Enemy Chomper")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Snap")
    ActivateGameObject(gameObject=var "Collider 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collider 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Trap Attack", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    CheckTrackTriggerCountV2(Target=var "Collider 1", Count=0, Test=MoreThan, EveryFrame=true, SetBool=(unset), SuccessEvent=→"HERO", FailEvent=(none))
    on HERO → Catch Hero 1
    on FINISHED → Snap 2?
  }

  state End Std {
    ActivateGameObjectDelay(gameObject=var "Enemy Chomper", activate=false, resetOnExit=false, delay=0.25)
  }

  state Kill Parent {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="DIE", delay=0, everyFrame=false)
  }

  state Snap 2? {
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Collider 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collider 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Chomper", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Snap", emit=0, resetIfPlaying=false)
    BoolTest(boolVariable=var "Caught Hero", isTrue=→"HERO", isFalse=(none), everyFrame=false)
    on HERO → Catch Hero 2?
    on FINISHED → End Std
  }

  state Catch Hero 2? {
    BoolTest(boolVariable=var "Caught Hero", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Caught Hero", boolValue=true, everyFrame=false)
    RunFSM(fsmTemplateControl=template=-5200671031615995940 in[Grab Effect Point<-var "Grab Effect Point"], finishEvent=(none), everyFrame=false)
    on FINISHED → Wait
  }

  state End Hero {
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Grab Point", xOffset=0, yOffset=0, everyFrame=false)
    RunFSM(fsmTemplateControl=template=-7631507063665673418, finishEvent=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Caught Hero", boolValue=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Trap Attack", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=3)
    ActivateGameObject(gameObject=var "Enemy Chomper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state OnDIsable {
    BoolTest(boolVariable=var "Caught Hero", isTrue=→"HERO", isFalse=(none), everyFrame=false)
    on HERO → End Hero
  }

  state Wait {
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Grab Point", xOffset=0, yOffset=0, everyFrame=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Hero
  }

  state Catch Hero 1 {
    SetBoolValue(boolVariable=var "Caught Hero", boolValue=true, everyFrame=false)
    RunFSM(fsmTemplateControl=template=-5200671031615995940 in[Grab Effect Point<-var "Grab Effect Point"], finishEvent=(none), everyFrame=false)
    on FINISHED → Hero
    on CANCEL → No Hero
  }

  state No Hero {
    SetBoolValue(boolVariable=var "Caught Hero", boolValue=false, everyFrame=false)
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=false)
    on FINISHED → Snap 2?
  }

  state Hero {
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Grab Point", xOffset=0, yOffset=0, everyFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Trap Attack Hero", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Snap 2?
  }

  var Caught Hero: bool = false
  var Collider 1: gameObject = <null>
  var Collider 2: gameObject = <null>
  var Pt Snap: gameObject = <null>
  var Enemy Chomper: gameObject = <null>
  var Parent: gameObject = <null>
  var Grab Effect Point: gameObject = Lilypad Trap Setter/Lilypad Plat (1)/Enemy/Grab Effect Point
  var Grab Point: gameObject = Lilypad Trap Setter/Lilypad Plat (1)/Enemy/Grab Point
}
