# Save Courier
# 1 location(s):
#   Aspid_01 / Courier Quest Scene/Courier Scene

fsm Save Courier {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Enemies")
    FindNamedChild(gameObject=Self, storeResult=var "Courier Short")
    on FINISHED → Idle
  }

  state Idle {
    Wait(time=3, finishEvent=→"KILLED", realTime=false)
    on KILLED → Count Enemies
  }

  sequence state Count Enemies {
    Wait(time=0.1, finishEvent=(none), realTime=false)
    GetChildCount(gameObject=var "Enemies", storeResult=var "Enemy Count")
    IntCompare(integer1=var "Enemy Count", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on END → Send Event
  }

  sequence state Send Event {
    Wait(time=1, finishEvent=(none), realTime=false)
    SendEventByName(eventTarget=GameObject(var "Courier Short"), sendEvent="COURIER SAVED", delay=0, everyFrame=false)
  }

  var Enemy Count: int = 0
  var Enemies: gameObject = <null>
  var Courier Short: gameObject = <null>
}
