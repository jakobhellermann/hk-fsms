# Control
# 4 location(s):
#   Arborium_05 / Bloom Puncher
#   Shellwood_01b / Bloom Puncher
#   Shellwood_10 / Bloom Puncher
#   Shellwood_11 / Bloom Puncher

fsm Control {
  start Init

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    SetParent(gameObject=var "Head", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetParent(gameObject=var "H1", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetParent(gameObject=var "H2", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetParent(gameObject=var "H3", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetParent(gameObject=var "H4", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetParent(gameObject=var "H5", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    SetParent(gameObject=var "H6", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)  // disabled
    on FINISHED → Idle
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Lift Range", storeResult=false, sendEvent="LOOK", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=→"LOOK", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ATTACK → Attack Antic
    on TOOK DAMAGE → Attack Antic
    on ZERO HP → Spawn 1
    on SING → Sing
    on LOOK → Look
  }

  state Attack Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Punch Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitterChildren(gameObject=var "Pt Open", resetTimeIfPlaying=false, stopOnStateExit=false)
    PlayAudioEvent(audioClip=bloom_puncher_open (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Extend Player")
    on FINISHED → Punch 1
    on ZERO HP → Spawn 1
  }

  state Punch 1 {
    PlayAudioEvent(audioClip=bloom_puncher_extend (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Extend Player")
    PlayParticleEmitterChildren(gameObject=var "Pt Attack", resetTimeIfPlaying=false, stopOnStateExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Punching F")
    Tk2dPlayAnimation(gameObject=var "BG", animLibName="", clipName="Punching B")
    ActivateGameObject(gameObject=var "BG", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Head", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Head", clipName="Punch Head", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=var "Head", frame=0)
    ActivateGameObject(gameObject=var "H1", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Punch 2
    on ZERO HP → Spawn 1
  }

  state Punch 2 {
    Tk2dWatchAnimationEvents(gameObject=var "Head", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "H2", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Punch 3
    on ZERO HP → Spawn 2
  }

  state Punch 3 {
    Tk2dWatchAnimationEvents(gameObject=var "Head", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "H3", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Punch 4
    on ZERO HP → Spawn 3
  }

  state Punch 4 {
    Tk2dWatchAnimationEvents(gameObject=var "Head", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "H4", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Punch 5
    on ZERO HP → Spawn 3
  }

  state Punch 5 {
    PlayAudioEvent(audioClip=bloom_puncher_withdraw (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Extend Player")
    Tk2dWatchAnimationEvents(gameObject=var "Head", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "H5", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Punch 6
    on ZERO HP → Spawn 3
  }

  state Punch 6 {
    Tk2dWatchAnimationEvents(gameObject=var "Head", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "H6", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Punch 7
    on ZERO HP → Spawn 2
  }

  state Punch 7 {
    Tk2dWatchAnimationEvents(gameObject=var "Head", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Punch End
    on ZERO HP → Spawn 1
  }

  state Punch End {
    ActivateGameObject(gameObject=var "BG", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Head", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Punch End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Punch Recover
    on ZERO HP → Spawn 1
  }

  state Spawn Head {
    AudioStop(gameObject=var "Extend Player", fadeTime=0)
    CreateObject(gameObject=Bloom Puncher Head (enemycorpses_assets_areashellwood.bundle), spawnPoint=<null>, position=var "Spawn Point", rotation=(unset), storeObject=var "Death Head")
    FlingObject(flungObject=var "Death Head", speedMin=20, speedMax=25, angleMin=75, angleMax=105)
    ActivateGameObject(gameObject=var "Head", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "H1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "H2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "H3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "H4", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "H5", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "H6", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Spawn 1 {
    GetPosition(gameObject=Self, vector=var "Spawn Point", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Spawn Head
  }

  state Spawn 2 {
    GetPosition(gameObject=var "Spawn 2", vector=var "Spawn Point", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Spawn Head
  }

  state Spawn 3 {
    GetPosition(gameObject=var "Spawn 3", vector=var "Spawn Point", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Spawn Head
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=flower_drifter_singing (sfxdynamic_assets_areashellwood.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Punch End
    on SING DURATION END → Punch Recover
  }

  state Look {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitterChildren(gameObject=var "Pt Open", resetTimeIfPlaying=false, stopOnStateExit=false)
    PlayAudioEvent(audioClip=bloom_puncher_open (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Extend Player")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    on FINISHED → Look End
    on ATTACK → Look AttackAntic
    on ZERO HP → Spawn 1
  }

  state Look End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Punch End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
    on ZERO HP → Spawn 1
  }

  state Punch Recover {
    WaitRandom(timeMin=0.25, timeMax=0.4, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.1, None=(none), ActiveInner=→"SING", ActiveOuter=→"SING", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Idle
    on ZERO HP → Spawn 1
    on SING → Sing
  }

  state Look AttackAntic {
    PlayAudioEvent(audioClip=bloom_puncher_open (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Extend Player")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look Punch Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on ZERO HP → Spawn 1
    on FINISHED → Punch 1
  }

  var Spawn Point: vector3 = (0, 0, 0)
  var BG: gameObject = Bloom Puncher/BG
  var Death Head: gameObject = <null>
  var Extend Player: gameObject = <null>
  var H1: gameObject = Bloom Puncher/H1
  var H2: gameObject = Bloom Puncher/H2
  var H3: gameObject = Bloom Puncher/H3
  var H4: gameObject = Bloom Puncher/H4
  var H5: gameObject = Bloom Puncher/H5
  var H6: gameObject = Bloom Puncher/H6
  var Head: gameObject = Bloom Puncher/Head
  var Pt Attack: gameObject = Bloom Puncher/Pt Attack
  var Pt Open: gameObject = Bloom Puncher/Pt Open
  var Self: gameObject = <null>
  var Spawn 2: gameObject = Bloom Puncher/Spawn 2
  var Spawn 3: gameObject = Bloom Puncher/Spawn 3
  var Audio Loop Voice: gameObject = Bloom Puncher/Audio Loop Voice
}
