# Damage Slowdown
# 1 location(s):
#   herodynamic_assets_all.bundle / Tool Wheel/EnemyDamager

fsm Damage Slowdown {
  start State 1

  state State 1 {
    on HIT LANDED → State 2
  }

  state State 2 {
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=0, y=(unset), everyFrame=false)  // disabled
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Velocity X", multiplyBy=0.5, everyFrame=false)
    SetVelocity2d(gameObject=var "Parent", vector=(unset), x=var "Velocity X", y=(unset), everyFrame=false)
    on FINISHED → State 1
  }

  var Velocity X: float = 0
  var Parent: gameObject = Tool Wheel
}
