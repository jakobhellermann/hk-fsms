# Camera Control
# 1 location(s):
#   Under_01 / Understore Lift Scene/Understore_lift

fsm Camera Control {
  start Init
  on CAM END → Cam End  // from any state

  state Init {
    on CAM START → Get
  }

  state Get {
    FindGameObject(objectName="", withTag="CameraTarget", store=var "CamTarget")
    CallMethodProper(gameObject=var "CamTarget", behaviour="CameraTarget", methodName="StartFreeMode", parameters=[false], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Follow Down
  }

  state Bot Lock {
    AnimatePositionTo(gameObject=var "CamTarget", toValue=(81.24, 11.26, 0), localSpace=false, time=(unset), speed=5, delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    on FINAL PHASE → Rock Lock
  }

  state Rock Lock {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Target Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Target Y", float2=12.53, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatClamp(floatVariable=var "Target Y", minValue=11.26, maxValue=23.25, everyFrame=true)
    SetPosition(gameObject=var "CamTarget", vector=(unset), x=(unset), y=var "Target Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → State 1
  }

  state State 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Rock Lock
  }

  state Cam End {
    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=1, functionCall=EndFreeMode())  // disabled
  }

  state Follow Down {
    GetPosition(gameObject=Self, vector=(unset), x=var "Target X", y=var "Target Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Target Y", add=-0.5, everyFrame=true, perSecond=false)
    SetPosition(gameObject=var "CamTarget", vector=(unset), x=var "Target X", y=var "Target Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on CAM LAND → State 2
  }

  state State 2 {
    GetPosition(gameObject=Self, vector=(unset), x=var "Target X", y=var "Target Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Target Y", add=0.5, everyFrame=true, perSecond=false)
    SetPosition(gameObject=var "CamTarget", vector=(unset), x=var "Target X", y=var "Target Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on CAM FOLLOW → Follow Down
    on CAM FALL → Cam Fall
  }

  state Cam Fall {
    GetPosition(gameObject=Self, vector=(unset), x=var "Target X", y=var "Target Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Target Y", add=-2, everyFrame=true, perSecond=false)
    SetPosition(gameObject=var "CamTarget", vector=(unset), x=var "Target X", y=var "Target Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
  }

  var Target Y: float = 0
  var Target X: float = 0
  var CamTarget: gameObject = <null>
}
