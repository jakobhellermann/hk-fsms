# Control
# 2 location(s):
#   Abyss_02 / Gloomfly Cocoon S (3)
#   Abyss_02b / Gloomfly Cocoon S (3)

fsm Control {
  start Pause
  on BREAK → Break  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    CreateObject(gameObject=Gloomfly (localpoolprefabs_assets_areaabyss.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Gloomfly")
    ActivateGameObject(gameObject=var "Gloomfly", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=var "Egg", storeResult=var "Active")
    on FINISHED → Idle
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    on ALERT → Hatch Antic
  }

  state Hatch Antic {
    ActivateGameObject(gameObject=var "Radial Burst Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEventRandom(audioClips=[coral_goomba_large_vox_medium_1 (sfxstatic_assets_areacoral.bundle), coral_goomba_large_vox_medium_2 (sfxstatic_assets_areacoral.bundle), coral_goomba_large_vox_medium_3 (sfxstatic_assets_areacoral.bundle), coral_goomba_large_vox_medium_4 (sfxstatic_assets_areacoral.bundle)], pitchMin=1.7, pitchMax=1.7, volume=0.7, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    ObjectJitterLocal(gameObject=var "Active", x=0.05, y=0.05, z=(unset), resetPosOnStateExit=true, limitFps=30)
    RandomFloatEither(value1=0, value2=180, storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=var "Egg", angle=var "Angle", speed=5, everyFrame=false)
    PlayAudioEvent(audioClip=broodmother_spit_antic (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=broodmother_give_birth_antic (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    on FINISHED → Hatch
  }

  state Hatch {
    SendMessage(gameObject=var "Active", delivery=SendMessage, options=1, functionCall=BreakSelf())
  }

  state Break {
    ActivateGameObject(gameObject=var "Gloomfly", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObject(flungObject=var "Gloomfly", speedMin=15, speedMax=15, angleMin=260, angleMax=280)
    PlayAudioEvent(audioClip=broodmother_give_birth (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=lifeblood_burst (sfxstatic_assets_lifeblood.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
  }

  var Angle: float = 0
  var Self: gameObject = Gloomfly Cocoon S (3)
  var Gloomfly: gameObject = <null>
  var Radial Burst Effect: gameObject = Gloomfly Cocoon S (3)/Radial Burst Effect
  var Egg: gameObject = Gloomfly Cocoon S (3)/Egg
  var Active: gameObject = <null>
  var Pt Antic: gameObject = Gloomfly Cocoon S (3)/Egg/Active/Sprite/Pt Antic
}
