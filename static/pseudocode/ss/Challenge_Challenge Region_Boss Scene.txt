# Challenge
# 1 location(s):
#   Memory_Ant_Queen / Boss Scene/Challenge Region

fsm Challenge {
  start Init
  on SPECIAL CHALLENGE → Take Control  // from any state

  state Init {
    on FINISHED → Idle
  }

  state In Region {
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=true, everyFrame=false)
    Trigger2dEvent(gameObject=var "Taunt Region", trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on INTERACT → Hornet Voice
    on EXIT → Exit
    on SPECIAL CHALLENGE → Take Control
  }

  state Facing {
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=false, everyFrame=false)
    BoolTest(boolVariable=var "Face Receiver", isTrue=→"RECEIVER", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Face Left", isTrue=→"L", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Face Right", isTrue=→"R", isFalse=(none), everyFrame=false)
    on L → Face L
    on R → Face R
    on FINISHED → Challenge 1
    on RECEIVER → Straight Back?
  }

  state Face L {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Challenge 1
  }

  state Challenge 1 {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Receiver"), sendEvent="CHALLENGE START", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Taunt Back", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Challenge 2
  }

  state Face R {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    on FINISHED → Challenge 1
  }

  state Challenge 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=sword_5 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Challenge Complete
  }

  state Challenge Complete {
    SendEventByName(eventTarget=GameObject(var "Receiver"), sendEvent="CHALLENGE", delay=0, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Taunt Region", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In Region
  }

  state Exit {
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Take Control {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=false, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=var "Disable Pause")
    on FINISHED → Hornet Voice
  }

  state Face Receiver {
    CheckTargetDirection(gameObject=var "Hero", target=var "Receiver", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Face L
    on R → Face R
  }

  state Straight Back? {
    FloatCompare(float1=var "Straight Back Distance", float2=0, tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    GetXDistance(gameObject=var "Straight Back Object", target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=var "Straight Back Distance", tolerance=0, equal=→"STRAIGHT BACK", lessThan=→"STRAIGHT BACK", greaterThan=(none), everyFrame=false)
    on FINISHED → Face Receiver
    on STRAIGHT BACK → Face Straight Back
  }

  state Face Straight Back {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Receiver"), sendEvent="CHALLENGE START", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Taunt Straight Back", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Challenge 2
  }

  state Hornet Voice {
    PlayRandomAudioClipTable(Table=var "Taunt Voice Table", AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0))
    on FINISHED → Facing
  }

  var Straight Back Distance: float = 0
  var Face Left: bool = false
  var Face Receiver: bool = true
  var Face Right: bool = false
  var Receiver: gameObject = Boss Scene/Hunter Queen Boss
  var Straight Back Object: gameObject = <null>

  var Distance: float = 0
  var Disable Pause: bool = false
  var Taunt Region: gameObject = Boss Scene/Challenge Region/Taunt Region
  var Taunt Voice Table: object = Taunt Seriously Hornet Voice (herodynamic_assets_all.bundle)
}
