# Hand Control
# 2 location(s):
#   Cradle_03 / Boss Scene/Silk Boss/Hand R
#   Cradle_03 / Boss Scene/Silk Boss/Hand L

// uses template: silk_boss_hand_control
fsm Hand Control {
  start Init
  on SWIPE → Set Swipe Std  // from any state
  on SWIPE QUICK → Set Swipe Quick  // from any state
  on STOMP → Set Stomp Std  // from any state
  on STOMP QUICK → Set Stomp Quick  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Silk Boss")
    FindNamedChild(gameObject=Self, storeResult=var "Finger Blade L")
    FindNamedChild(gameObject=Self, storeResult=var "Finger Blade M")
    FindNamedChild(gameObject=Self, storeResult=var "Finger Blade R")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Offset", y=var "Y Offset", z=(unset), space=1, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    FloatOperator(float1=var "Y Offset", float2=5, operation=Add, storeResult=var "Y Offset High", everyFrame=false)
    on FINISHED → Deparent
  }

  state Idle {
    SetBoolValue(boolVariable=var "Ready", boolValue=true, everyFrame=false)
    MatchScaleSign(Target=Self, MatchTo=var "Silk Boss", EveryFrame=true)
    Wait(time=3, finishEvent=→"CLAW", realTime=false)  // disabled
    SetPositionToObject2D(gameObject=Self, targetObject=var "Silk Boss", xOffset=var "X Offset", yOffset=var "Y Offset", everyFrame=true)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), space=0, everyFrame=true)
    FloatClamp(floatVariable=var "Self X", minValue=23, maxValue=57, everyFrame=true)
    SetPosition2D(GameObject=Self, Vector=(unset), X=var "Self X", Y=(unset), Space=0, EveryFrame=true)
    on STOMP → Set Stomp Std
    on SWIPE → Set Swipe Std
    on CLAW → Claw Dir
    on CLAW FAR L → Claw Far L
    on CLAW FAR R → Claw Far R
    on STOMP FAR L → Stomp Aim L
    on STOMP FAR R → Stomp Aim R
    on PINCER HIGH L → Swipe High L
    on PINCER HIGH R → Swipe High R
    on PREPARE → Prepare
    on STOMP DASH L → Stomp Dash L
    on STOMP DASH R → Stomp Dash R
  }

  state Stomp {
    SetBoolValue(boolVariable=var "Ready", boolValue=false, everyFrame=false)
    FloatClamp(floatVariable=var "Hero X", minValue=22, maxValue=55, everyFrame=false)
    SetFloatValue(floatVariable=var "Attack X", floatValue=var "Hero X", everyFrame=false)
    FloatAddRandom(floatVariable=var "Attack X", addMin=-5, addMax=-4)
    SetFsmFloat(gameObject=var "Finger Blade L", fsmName="Control", variableName="Attack X", setValue=var "Attack X", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Finger Blade L"), sendEvent="STOMP", delay=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Attack X", floatValue=var "Hero X", everyFrame=false)
    FloatAddRandom(floatVariable=var "Attack X", addMin=4, addMax=5)
    SetFsmFloat(gameObject=var "Finger Blade R", fsmName="Control", variableName="Attack X", setValue=var "Attack X", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Finger Blade R"), sendEvent="STOMP", delay=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Attack X", floatValue=var "Hero X", everyFrame=false)
    FloatAddRandom(floatVariable=var "Attack X", addMin=-1, addMax=1)
    SetFsmFloat(gameObject=var "Finger Blade M", fsmName="Control", variableName="Attack X", setValue=var "Attack X", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Finger Blade M"), sendEvent="STOMP", delay=0, everyFrame=false)
    on FINISHED → Attack Ready Frame
  }

  state Swipe Dir {
    SetBoolValue(boolVariable=var "Ready", boolValue=false, everyFrame=false)
    CheckXPosition(gameObject=var "Hero", compareTo=47, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"R", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    CheckXPosition(gameObject=var "Hero", compareTo=33, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"L", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Swipe L
    on R → Swipe R
  }

  state Swipe R {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Attack X", floatValue=var "Hero X", everyFrame=false)
    FloatAddRandom(floatVariable=var "Attack X", addMin=-11, addMax=-12)
    RandomFloat(min=133, max=134, storeResult=var "Attack Y")
    SetFsmFloat(gameObject=var "Finger Blade L", fsmName="Control", variableName="Attack X", setValue=var "Attack X", everyFrame=false)
    SetFsmFloat(gameObject=var "Finger Blade L", fsmName="Control", variableName="Attack Y", setValue=var "Attack Y", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Finger Blade L"), sendEvent=var "Swipe R Event", delay=0, everyFrame=false)

    FloatAddRandom(floatVariable=var "Attack X", addMin=-11, addMax=-12.5)
    RandomFloat(min=141.5, max=142.5, storeResult=var "Attack Y")
    SetFsmFloat(gameObject=var "Finger Blade M", fsmName="Control", variableName="Attack X", setValue=var "Attack X", everyFrame=false)
    SetFsmFloat(gameObject=var "Finger Blade M", fsmName="Control", variableName="Attack Y", setValue=var "Attack Y", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Finger Blade M"), sendEvent=var "Swipe R Event", delay=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Attack X", floatValue=var "Hero X", everyFrame=false)
    FloatAddRandom(floatVariable=var "Attack X", addMin=-11, addMax=-12.5)
    RandomFloatEither(value1=136, value2=139.5, storeResult=var "Attack Y")
    FloatAddRandom(floatVariable=var "Attack Y", addMin=-0.5, addMax=0.5)
    SetFsmFloat(gameObject=var "Finger Blade R", fsmName="Control", variableName="Attack X", setValue=var "Attack X", everyFrame=false)
    SetFsmFloat(gameObject=var "Finger Blade R", fsmName="Control", variableName="Attack Y", setValue=var "Attack Y", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Finger Blade R"), sendEvent=var "Swipe R Event", delay=0, everyFrame=false)
    on FINISHED → Attack Ready Frame
  }

  }

    GetFsmBool(gameObject=var "Finger Blade R", fsmName="Control", variableName="Ready", storeValue=var "Ready R", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Ready L", var "Ready M", var "Ready R"], sendEvent=→"READY", storeResult=(unset), everyFrame=true)
    SetPositionToObject2D(gameObject=Self, targetObject=var "Silk Boss", xOffset=var "X Offset", yOffset=var "Y Offset", everyFrame=true)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), space=0, everyFrame=true)
    FloatClamp(floatVariable=var "Self X", minValue=23, maxValue=57, everyFrame=true)
    SetPosition2D(GameObject=Self, Vector=(unset), X=var "Self X", Y=(unset), Space=0, EveryFrame=true)
    on READY → Idle
  }

  state Attack Ready Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Attacking
  }
