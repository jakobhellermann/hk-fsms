# Trap Bench
# 1 location(s):
#   Ant_17 / Trap Scene/RestBench

fsm Trap Bench {
  start Init
  on HERO DAMAGED → Get Up 2  // from any state

  state Init {
    GetHero(storeResult=var "Hero")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Original Y", z=(unset), space=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Step Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Step Effects", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Idle {
    PlayerDataBoolTest(boolName="antBenchTrapDefused", isTrue=→"OPEN", isFalse=(none))
    on EXTRA BENCH SIT → Sit
    on OPEN → Activate
  }

  state Sit {
    SendEventToRegister(eventName="HIDE INTERACT REMINDER")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AffectedByGravity", parameters=[false], storeResult=(unset var), EveryFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=bench_rest (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Sit")
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Sit Vector", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Sit Vector", vector3Value=(unset), x=var "Self X", y=(unset), z=(unset), everyFrame=false)
    Vector3Add(vector3Variable=var "Sit Vector", addVector=var "Adjust Vector", everyFrame=false, perSecond=false)
    iTweenMoveTo(gameObject=var "Hero", id="", transformPosition=var "", vectorPosition=var "Sit Vector", time=0.2, delay=0, speed=(unset), space=0, easeType=13, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    on FINISHED → Click
  }

  state Sitting {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Sit Idle")
    ListenForJump(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), queueBool=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    ListenForUp(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForDown(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), queueBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForLeft(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    ListenForRight(eventTarget=Self, wasPressed=→"GET UP", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    Wait(time=1, finishEvent=→"GET UP", realTime=false)
    on GET UP → Get Up
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Get Off", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AffectedByGravity", parameters=[true], storeResult=(unset var), EveryFrame=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, 0.2, 0), time=0.1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  state Finish {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControlToIdle", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Original Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Set Interactible
  }

  state Click {
    PlayAudioEvent(audioClip=bone_trap_button (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Step Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Step Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -0.2, 0), time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    AnimatePositionBy(gameObject=var "Hero", shiftBy=(0, -0.2, 0), time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Sickle"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    SendEventToRegister(eventName="TRAP SPRUNG")
    on FINISHED → Sitting
  }

  state Activate {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="BENCH ACTIVE", delay=0, everyFrame=false)
  }

  state Set Interactible {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  state Get Up 2 {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AffectedByGravity", parameters=[true], storeResult=(unset var), EveryFrame=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, 0.2, 0), time=0.1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  var Sickle: gameObject = Trap Scene/Sickle Holder/Hunter Sickle Trap

  var Original Y: float = 0
  var Self X: float = 0
  var Defused: bool = false
  var Adjust Vector: vector3 = (0, 0.2, 0)
  var Sit Vector: vector3 = (0, 0, 0)
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
  var Step Effects: gameObject = Trap Scene/RestBench/Step Effects
}
