# Control
# 1 location(s):
#   Room_Caravan_Interior_Travel / Hornet Spider Travel

fsm Control {
  start Resting Init

  state Resting Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sit Idle")
    on FINISHED → Resting
  }

  state Idle Anim {
    SelectRandomString(strings=["Sit Idle", "SitLook 1", "SitLook 2", "SitLook 3", "SitLook 4"], weights=[0.5, 0.125, 0.125, 0.125, 0.125], storeString=var "Sit Anim")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Sit Anim")
    on FINISHED → Resting
  }

  state Resting {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"ANIM END")
    GetPlayerDataVariable(VariableName="hasNeedolin", StoreValue=var "Has Needolin")
    ListenForSilkSpecial(eventTarget=Self, wasPressed=→"NEEDOLIN", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=var "Has Needolin", stateEntryOnly=false)
    on ANIM END → Idle Anim
    on NEEDOLIN → Needolin Start
  }

  state Needolin Start {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="HasNeedolin", parameters=[], storeResult=var "Can Play Needolin", EveryFrame=false)
    BoolTest(boolVariable=var "Can Play Needolin", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="NeedolinSit Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Set Playing
    on CANCEL → Frame
  }

  state Needolin Play {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NeedolinSit Play")
    ListenForSilkSpecial(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"END", isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Self X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Self X Scale", float2=1, tolerance=0, equalBool=var "Facing Left", lessThanBool=var "Facing Right", greaterThanBool=(unset), everyFrame=false)
    ListenForSilkSpecial(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=→"BUTTON UP", isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    on END → Needolin End
    on LEVEL LOADED → Needolin End
  }

  state Needolin End {
    ControlNeedolin(target=var "Hero", isPlaying=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="NeedolinSit End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle Anim
  }

  state Set Playing {
    ControlNeedolin(target=var "Hero", isPlaying=true)
    on FINISHED → Needolin Play
  }

  state Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Resting
  }

  var Facing Right: bool = true

  var Self X Scale: float = 0
  var Facing Left: bool = false
  var Has Needolin: bool = false
  var Can Play Needolin: bool = false
  var Sit Anim: string = ""
}
