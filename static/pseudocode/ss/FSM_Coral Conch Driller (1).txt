// uses template: flyer_go_up_response
fsm FSM {
  start Init
  on FLYER RESPONSE STOP → Stop  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on GO UP → Force Up
    on GO DOWN → Force Down
    on GO LEFT → Force Left
    on GO RIGHT → Force Right
  }

  state Force Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Rise Speed", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Force Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Rise Speed", everyFrame=false)  // disabled
    SetVelocityAsAngle(gameObject=Self, angle=270, speed=var "Rise Speed", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Force Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Rise Speed", everyFrame=false)  // disabled
    SetVelocityAsAngle(gameObject=Self, angle=180, speed=var "Rise Speed", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Force Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Rise Speed", everyFrame=false)  // disabled
    SetVelocityAsAngle(gameObject=Self, angle=0, speed=var "Rise Speed", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Stop {
    on FLYER RESPONSE START → Idle
  }

  var Rise Speed: float = 10
}
