// component disabled: this FSM does not run until something enables it
fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Jail Gate Door")
    on FINISHED → Idle
  }

  state Idle {
    on BATTLE COMPLETED → Open Door
    on BATTLE END → Open Door
  }

  state Open Door {
    AnimatorPlay(gameObject=var "Jail Gate Door", stateName="Open", layer=(unset), normalizedTime=(unset), everyFrame=false)
  }

  var Battle Scene: gameObject = <null>
  var Jail Gate Door: gameObject = <null>
}
