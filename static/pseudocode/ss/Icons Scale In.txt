# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Tool Icons

fsm Icons Scale In {
  start Init
  on HUD APPEAR RESET → Set Scale  // from any state

  state Init {
    GetScale(gameObject=var "Tool Icon U", vector=var "Tool Icon U Scale", xScale=(unset), yScale=(unset), zScale=(unset), space=1, everyFrame=false)
    GetScale(gameObject=var "Tool Icon N", vector=var "Tool Icon N Scale", xScale=(unset), yScale=(unset), zScale=(unset), space=1, everyFrame=false)
    GetScale(gameObject=var "Tool Icon D", vector=var "Tool Icon D Scale", xScale=(unset), yScale=(unset), zScale=(unset), space=1, everyFrame=false)
    on FINISHED → Set Scale
  }

  state First? {
    PlayerDataBoolTest(boolName="bindCutscenePlayed", isTrue=→"WAIT FOR HP", isFalse=→"WAIT FOR INTRO")
    on WAIT FOR HP → Wait for HP
    on WAIT FOR INTRO → Wait for Intro
  }

  state Wait for HP {
    EnableFsmSelf(SetEnabled=false)
    on LAST HP ADDED → Wait
  }

  state Wait for Intro {
    EnableFsmSelf(SetEnabled=false)
    on SPOOL APPEAR → Wait
  }

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Scale In U
  }

  state Scale In U {
    GameObjectIsActive(Target=var "Tool Icon U", ActiveSpace=1, trueEvent=(none), falseEvent=→"FINISHED", storeValue=(unset))
    iTweenScaleTo(gameObject=var "Tool Icon U", id="", transformScale=var "", vectorScale=var "Tool Icon U Scale", time=0.2, delay=0, speed=(unset), easeType=1, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Scale In N
  }

  state Scale In N {
    GameObjectIsActive(Target=var "Tool Icon N", ActiveSpace=1, trueEvent=(none), falseEvent=→"FINISHED", storeValue=(unset))
    iTweenScaleTo(gameObject=var "Tool Icon N", id="", transformScale=var "", vectorScale=var "Tool Icon N Scale", time=0.2, delay=0, speed=(unset), easeType=1, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Scale In D
  }

  state Scale In D {
    GameObjectIsActive(Target=var "Tool Icon D", ActiveSpace=1, trueEvent=(none), falseEvent=→"FINISHED", storeValue=(unset))
    iTweenScaleTo(gameObject=var "Tool Icon D", id="", transformScale=var "", vectorScale=var "Tool Icon D Scale", time=0.2, delay=0, speed=(unset), easeType=1, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=false, loopDontFinish=false)
    on FINISHED → Inert
  }

  state Inert {
    SetScale(gameObject=var "Tool Icon U", vector=var "Tool Icon U Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Tool Icon N", vector=var "Tool Icon N Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Tool Icon D", vector=var "Tool Icon D Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SendEventToRegister(eventName="TOOLS APPEAR")
    EnableFsmSelf(SetEnabled=false)
  }

  state Set Scale {
    SetScale(gameObject=var "Tool Icon U", vector=(unset), x=0, y=0, z=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Tool Icon N", vector=(unset), x=0, y=0, z=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Tool Icon D", vector=(unset), x=0, y=0, z=0, everyFrame=false, lateUpdate=false)
    on FINISHED → First?
  }

  var Tool Icon D Scale: vector3 = (0, 0, 0)
  var Tool Icon N Scale: vector3 = (0, 0, 0)
  var Tool Icon U Scale: vector3 = (0, 0, 0)
  var Tool Icon D: gameObject = _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Tool Icons/Tool Icon D
  var Tool Icon N: gameObject = _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Tool Icons/Tool Icon N
  var Tool Icon U: gameObject = _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Tool Icons/Tool Icon U
}
