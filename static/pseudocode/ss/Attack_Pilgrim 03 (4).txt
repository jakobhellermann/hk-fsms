# Attack
# 1 location(s):
#   Bone_11 / Pilgrim 03 (4)

fsm Attack {
  start Init

  state Idle {
    on ATTACK → Attack Antic
  }

  state Attack Antic {
    StopWalker(target=Self, everyFrame=false)
    SetIntValue(intVariable=var "Leaps", intValue=2, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=pilgrim_03_pounce (sfxstatic_assets_pilgrimslower.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Pilgrim_Groveller_Attack (sfxdynamic_assets_pilgrimslower.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.15)
    on FINISHED → Wait For Ground
  }

  state Attack {
    IntAdd(intVariable=var "Leaps", add=-1, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetVelocityByScale(gameObject=Self, speed=-13, ySpeed=(unset), everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Jump Height", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetGravity2dScale(gameObject=Self, gravityScale=0.75)
    SetRecoilBlocked(Target=Self, IsUpBlocked=false, IsDownBlocked=false, IsLeftBlocked=false, IsRightBlocked=false)
    on FINISHED → Wait For Land
  }

  state Stop {
    PlayAudioEvent(audioClip=pilgrim_staff_wielder_land (sfxstatic_assets_pilgrimslower.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=0, y=(unset), everyFrame=false)  // disabled
    TimeLimitSet(timeDelay=0.35, storeValue=var "Attack Time")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Breaker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Return
    on FALL → Fall
  }

  state Return {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="pilgrim_behaviour"), sendEvent="ATTACK END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Wait For Land {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    ActivateGameObjectDelay(gameObject=var "Breaker", activate=true, resetOnExit=false, delay=0.25)
    on LAND → Stop
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Breaker")
    on FINISHED → Idle
  }

  state Attack Height {
    SetFloatValue(floatVariable=var "Jump Height", floatValue=10, everyFrame=false)
    RayCast2dV2(fromGameObject=var "Step Ray", fromPosition=(unset), direction=(-1, 0), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"FINISHED", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    SetFloatValue(floatVariable=var "Jump Height", floatValue=13, everyFrame=false)
    on FINISHED → Attack
  }

  state Wait For Ground {
    BoolTest(boolVariable=var "Dont Wait For Ground", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"FINISHED", NotGroundedEvent=(none), EveryFrame=true)
    on FINISHED → Attack Height
  }

  state Fall {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    on FINISHED → Wait For Land
  }

  var Attack Time: float = 0
  var X Scale: float = 0
  var X Speed: float = 0
  var X Distance: float = 0
  var Jump Height: float = 0
  var Leaps: int = 2
  var Dont Wait For Ground: bool = false
  var Self: gameObject = Pilgrim 03 (4)
  var Breaker: gameObject = <null>
  var Step Ray: gameObject = Pilgrim 03 (4)/Step Ray
  var Audio Loop Voice: gameObject = Pilgrim 03 (4)/Audio Loop Voice
}
