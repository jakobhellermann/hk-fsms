# Control
# 1 location(s):
#   Coral_Judge_Arena / Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio

fsm Control {
  start Init
  on AMBIENT FLEA AUDIO STOP → Stop All  // from any state

  state Init {
    ActivateGameObject(gameObject=var "Fly Loop 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fly Loop 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Festival", isTrue=→"MULTIPLE", isFalse=(none), everyFrame=false)
    GetPlayerDataVariable(VariableName="SavedFleasCount", StoreValue=var "Little Fleas Saved")
    IntSwitch(intVariable=var "Little Fleas Saved", compareTo=[0, 1, 2], sendEvent=[→"NONE", →"ONE", →"TWO"], everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"MULTIPLE", delay=0, everyFrame=false)
    on NONE → None Saved
    on ONE → One Saved
    on TWO → Two Saved
    on MULTIPLE → Multiple Saved
  }

  state None Saved {
    on FINISHED → Idle
  }

  state One Saved {
    ActivateGameObject(gameObject=var "Fly Loop 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Multiple Saved {
    ActivateGameObject(gameObject=var "Fly Loop 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fly Loop 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Two Saved {
    ActivateGameObject(gameObject=var "Fly Loop 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fly Loop 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on LITTLE FLEA SING START → Increment Singers
    on LITTLE FLEA SING END → Decrement Singers
    on LITTLE FLEA SLEEP START → Increment Sleepers
    on LITTLE FLEA SLEEP END → Decerement Sleepers
  }

  state Increment Singers {
    IntAdd(intVariable=var "Flea Singers Amount", add=1, everyFrame=false)
    on FINISHED → Check Singer Amount
  }

  state Check Singer Amount {
    IntClamp(intVariable=var "Flea Singers Amount", minValue=0, maxValue=100, everyFrame=false)
    IntSwitch(intVariable=var "Flea Singers Amount", compareTo=[0, 1, 2], sendEvent=[→"NONE", →"ONE", →"TWO"], everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"MULTIPLE", delay=0, everyFrame=false)
    on NONE → No Singers
    on ONE → One Singer
    on TWO → Two Singers
    on MULTIPLE → Multiple Singers
  }

  state Decrement Singers {
    IntAdd(intVariable=var "Flea Singers Amount", add=-1, everyFrame=false)
    on FINISHED → Check Singer Amount
  }

  state No Singers {
    AudioStopV2(gameObject=var "Sing Loop 1", fadeTime=0.1, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Sing Loop 2", fadeTime=0.1, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Sing Loop 3", fadeTime=0.1, cancelOnEarlyExit=false)
    on FINISHED → Idle
  }

  state One Singer {
    AudioPlaySimple(gameObject=var "Sing Loop 1", volume=1, oneShotClip=<null>)
    AudioStopV2(gameObject=var "Sing Loop 2", fadeTime=0.1, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Sing Loop 3", fadeTime=0.1, cancelOnEarlyExit=false)
    on FINISHED → Idle
  }

  state Two Singers {
    AudioPlaySimple(gameObject=var "Sing Loop 1", volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Sing Loop 2", volume=1, oneShotClip=<null>)
    AudioStopV2(gameObject=var "Sing Loop 3", fadeTime=0.1, cancelOnEarlyExit=false)
    on FINISHED → Idle
  }

  state Multiple Singers {
    AudioPlaySimple(gameObject=var "Sing Loop 1", volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Sing Loop 2", volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Sing Loop 3", volume=1, oneShotClip=<null>)
    on FINISHED → Idle
  }

  state Increment Sleepers {
    IntAdd(intVariable=var "Flea Sleepers Amount", add=1, everyFrame=false)
    on FINISHED → Check Sleeper Amount
  }

  state Decerement Sleepers {
    IntAdd(intVariable=var "Flea Sleepers Amount", add=-1, everyFrame=false)
    on FINISHED → Check Sleeper Amount
  }

  state Check Sleeper Amount {
    IntClamp(intVariable=var "Flea Sleepers Amount", minValue=0, maxValue=100, everyFrame=false)
    IntSwitch(intVariable=var "Flea Sleepers Amount", compareTo=[0, 1], sendEvent=[→"NONE", →"ONE"], everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"MULTIPLE", delay=0, everyFrame=false)
    on NONE → No Sleepers
    on ONE → One Sleeper
    on MULTIPLE → Multiple Sleepers
  }

  state No Sleepers {
    AudioStopV2(gameObject=var "Sleep Loop 1", fadeTime=0.1, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Sleep Loop 2", fadeTime=0.1, cancelOnEarlyExit=false)
    on FINISHED → Idle
  }

  state One Sleeper {
    AudioPlaySimple(gameObject=var "Sleep Loop 1", volume=0.4, oneShotClip=<null>)
    AudioStopV2(gameObject=var "Sleep Loop 2", fadeTime=0.1, cancelOnEarlyExit=false)
    on FINISHED → Idle
  }

  state Multiple Sleepers {
    AudioPlaySimple(gameObject=var "Sleep Loop 1", volume=0.3, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Sleep Loop 2", volume=0.2, oneShotClip=<null>)
    on FINISHED → Idle
  }

  state Stop All {
    AudioStopV2(gameObject=var "Fly Loop 1", fadeTime=1, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Fly Loop 2", fadeTime=1, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Sleep Loop 1", fadeTime=1, cancelOnEarlyExit=false)
    AudioStopV2(gameObject=var "Sleep Loop 2", fadeTime=1, cancelOnEarlyExit=false)
    on AMBIENT FLEA AUDIO START → Restart All
  }

  state Restart All {
    AudioPlaySimple(gameObject=var "Fly Loop 1", volume=0.2, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Fly Loop 2", volume=0.2, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Sleep Loop 1", volume=1, oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Sleep Loop 2", volume=1, oneShotClip=<null>)
  }

  var Festival: bool = false

  var Little Fleas Saved: int = 0
  var Flea Singers Amount: int = 0
  var Flea Sleepers Amount: int = 0
  var Fly Loop 1: gameObject = Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio/Fly Loop 1
  var Fly Loop 2: gameObject = Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio/Fly Loop 2
  var Sing Loop 1: gameObject = Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio/Sing Loop 1
  var Sing Loop 2: gameObject = Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio/Sing Loop 2
  var Sing Loop 3: gameObject = Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio/Sing Loop 3
  var Sleep Loop 1: gameObject = Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio/Sleep Loop 1
  var Sleep Loop 2: gameObject = Caravan_Set/Caravan/Active/Caravan Ambient Fly Audio/Sleep Loop 2
}
