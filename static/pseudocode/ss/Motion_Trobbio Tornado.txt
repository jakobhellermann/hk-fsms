# Motion
# 1 location(s):
#   localpoolprefabs_assets_trobbio.bundle / Trobbio Tornado

fsm Motion {
  start Init

  state Motion {
    SetVelocityByScale(gameObject=Self, speed=30, ySpeed=(unset), everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.96, decelerationY=(unset), brakeOnExit=false)
    AccelerateToY(gameObject=Self, accelerationFactor=0.4, targetSpeed=30)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
    on MULTI HIT CONNECT → Multihit
  }

  state End {
  }

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Tornado Disperse", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Motion
  }

  state Multihit {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.0029, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    GetPosition2D(gameObject=var "Hero", vector=var "Hero Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    Vector2AddXY(vector2Variable=var "Hero Pos", addX=0, addY=-1.75, everyFrame=false, perSecond=false)
    AnimateRigidBody2DPositionTo(GameObject=Self, ToValue=var "Hero Pos", time=0.1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)  // disabled
    Wait(time=0.37, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Multihit End
  }

  state Multihit End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="END", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Tornado Disperse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Hero Pos: vector2 = (0, 0)
  var Tornado Disperse: gameObject = Trobbio Tornado/Tornado Disperse
}
