# Control
# 1 location(s):
#   localpoolprefabs_assets_areadust.bundle / Grove Pilgrim Fly

fsm Control {
  start Init
  on SING → Sing  // from any state

  state Init {
    FindAlertRange(target=Self, storeResult=var "Alert Range", childName="Alert Range")
    SetParent(gameObject=var "Patrol Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Start Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    RandomInt(min=1, max=2, storeResult=var "Spawns", inclusiveMax=true, noRepeat=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Start Anim")
    StringCompare(stringVariable=var "Start Anim", compareTo="Fly In", equalEvent=→"FLY IN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Hatchling
    on FLY IN → Fly In Ready
  }

  state To Patrol {
    SetBoolValue(boolVariable=var "Returning", boolValue=false, everyFrame=false)
    ChaseObjectSpread(gameObject=Self, target=var "Patrol Point", speedMax=3, acceleration=0.2, targetSpreadX=0, targetSpreadY=1, spreadResetTimeMin=1, spreadResetTimeMax=2, offsetX=0, offsetY=0)
    GetXDistance(gameObject=Self, target=var "Patrol Point", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    on SHIFT → To Start
    on TOOK DAMAGE → 
    on ALERT → Startle
  }

  state To Start {
    TiltBySpeedY(gameObject=Self, tiltFactor=-1, tiltMax=15, rotationSpeed=50)
    SetBoolValue(boolVariable=var "Returning", boolValue=true, everyFrame=false)
    ChaseObjectSpread(gameObject=Self, target=var "Start Point", speedMax=3, acceleration=0.2, targetSpreadX=0, targetSpreadY=1, spreadResetTimeMin=1, spreadResetTimeMax=2, offsetX=0, offsetY=0)
    GetXDistance(gameObject=Self, target=var "Start Point", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    on SHIFT → To Patrol
    on TOOK DAMAGE → 
    on ALERT → Startle
  }

  state Unalert Patrol {
    IdleBuzzV2(gameObject=Self, waitMin=0.5, waitMax=1, speedMax=2.5, accelerationMax=15, roamingRangeX=4, roamingRangeY=4, manualStartPos=(unset))
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ALERT → Startle
    on TOOK DAMAGE → Startle
    on GO UP → Go Up
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=true)
    on CANCEL → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → 
  }

  state Start Check {
    GetCurrentTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Spawn Sing", equalEvent=→"SPAWN SING", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → To Patrol
    on SPAWN SING → Spawn Sing
  }

  state Spawn Sing {
    DecelerateXY(gameObject=Self, decelerationX=(unset), decelerationY=0.9, brakeOnExit=true)
    SetVelocity2d(gameObject=Self, vector=(0, 10), x=(unset), y=(unset), everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sing End
  }

  state Startle {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=5, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Chase
  }

  state Chase {
    UnalertTimer(alertRange=var "Unalert Range", timerVariable=var "Unalert Timer", unalertTime=6, eventTarget=Self, unalertEvent=→"UNALERT", resetOnStateEntry=false)
    DistanceFly(gameObject=Self, target=var "Hero", distance=5, speedMax=4.5, acceleration=0.2, targetsHeight=false, height=(unset), minAboveHero=3)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ATTACK → Attack Antic
    on UNALERT → Unalert Patrol
    on GO UP → Go Up Chase
  }

  state Attack Antic {
    DistanceFly(gameObject=Self, target=var "Hero", distance=5, speedMax=2, acceleration=0.2, targetsHeight=false, height=(unset), minAboveHero=3)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    on FINISHED → Shake Start
  }

  state Shake {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shake")
    DistanceFly(gameObject=Self, target=var "Hero", distance=5, speedMax=1, acceleration=0.2, targetsHeight=false, height=(unset), minAboveHero=3)
    WaitRandom(timeMin=0.75, timeMax=0.9, finishEvent=→"FINISHED", realTime=false)
    FlingObjectsFromGlobalPoolTime(gameObject=SilkAcid BurstCloud (localpoolprefabs_assets_areacloverareamoss.bundle), spawnPoint=var "Self", position=(0, 0, -0.0025), frequency=0.04, spawnMin=2, spawnMax=2, speedMin=15, speedMax=30, angleMin=0, angleMax=360, originVariationX=0.5, originVariationY=0.5, startDelay=0)
    on FINISHED → Recover
  }

  state Recover {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    DistanceFly(gameObject=Self, target=var "Hero", distance=7, speedMax=4.5, acceleration=0.2, targetsHeight=false, height=(unset), minAboveHero=2)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    FireAtTarget(gameObject=Self, target=var "Hero", speed=-5, position=(0, 0.5, 0), spread=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Shake")
    on FINISHED → Chase
  }

  state Shake Start {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shake")
    DistanceFly(gameObject=Self, target=var "Hero", distance=5, speedMax=1.5, acceleration=0.2, targetsHeight=false, height=(unset), minAboveHero=3)
    PlayParticleEmitter(gameObject=var "Pt Shake", emit=0, resetIfPlaying=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=1.5, everyFrame=false)
    on FINISHED → Shake
  }

  state Hatchling {
    IntCompare(integer1=var "Spawns", integer2=0, equal=→"END", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CreateObject(gameObject=Aspid Hatchling (localpoolprefabs_assets_areaclover.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Hatchling")
    SetFsmGameObject(gameObject=var "Hatchling", fsmName="Control", variableName="Follow Target", setValue=var "Self", everyFrame=false)
    IntAdd(intVariable=var "Spawns", add=-1, everyFrame=false)
    on FINISHED → Hatchling
    on END → Start Check
  }

  state Go Up {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=1, everyFrame=false)
    on FINISHED → Unalert Patrol
  }

  state Go Up Chase {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=1, everyFrame=false)
    on FINISHED → Chase
  }

  state Fly In Ready {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    StopParticleEmitter(gameObject=var "Pt Idle")
    on BATTLE START → Check Pos
    on WAKE → Fly In
  }

  state Check Pos {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=8, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=false)
    on SHIFT → Shift
    on FINISHED → Fly In
  }

  state Shift {
    SetParent(gameObject=var "Spawn Pos Alt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Spawn Pos Alt", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    on FINISHED → Fly In
  }

  state Fly In {
    PlayParticleEmitter(gameObject=var "Pt Idle", emit=0, resetIfPlaying=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=10, z=-15, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetMeshRenderer(gameObject=Self, active=true)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -10, 15), time=0.75, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.55)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    on FINISHED → Chase
  }

  var Distance: float = 0
  var Unalert Timer: float = 0
  var Spawns: int = 0
  var Returning: bool = false
  var Clip: string = ""
  var Start Anim: string = ""
  var Patrol Point: gameObject = Grove Pilgrim Fly/Patrol Point
  var Start Point: gameObject = Grove Pilgrim Fly/Start Point
  var Self: gameObject = Grove Pilgrim Fly
  var Pt Shake: gameObject = Grove Pilgrim Fly/Pt Shake
  var Hatchling: gameObject = <null>
  var Spawn Pos Alt: gameObject = Grove Pilgrim Fly/Spawn Pos Alt
  var Pt Idle: gameObject = Grove Pilgrim Fly/Pt Idle
  var Alert Range: object = Grove Pilgrim Fly/Alert Range@AlertRange
  var Unalert Range: object = Grove Pilgrim Fly/Unalert Range@AlertRange
}
