# Interaction
# 1 location(s):
#   Bellway_04_boss / Bone Beast NPC

fsm Interaction {
  start Init
  on DISABLE → Stop Rumbles  // from any state
  on TEST ARRIVE → Wait Finished Entering  // from any state
  on DISABLE → Disable  // from any state

  state Init {
    AddHeroInputBlocker(Blocker=Self)
    ActivateGameObject(gameObject=var "Antic Bell Shaker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Travel Interaction", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Temp Warmth Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Active From Arrive", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Antic Particles", storeResult=var "Antic Particle Parent")
    FindChild(gameObject=Self, childName="Burst Particles", storeResult=var "Burst Particle Parent")
    FindChild(gameObject=Self, childName="Sleep Hit Response", storeResult=var "Sleep Hit Response")
    ActivateGameObject(gameObject=var "Sleep Hit Response", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Awake Hit Response", storeResult=var "Awake Hit Response")

    ActivateGameObject(gameObject=var "Burst Out Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Burst Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait For Boss Load
  }

  state Idle {
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=var "Start Facing Left", Target=Self)
    ActivateInteractible(Target=var "Travel Interaction", Activate=true, AllowQueueing=false, UseChildren=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.5, MaxReactDelay=3, None=(none), ActiveInner=(none), ActiveOuter=→"SING", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    WaitRandom(timeMin=5, timeMax=15, finishEvent=→"SHAKE", realTime=false)
    CheckTrackTriggerCountV2(Target=Bone Beast NPC/Look Detector Exit, Count=0, Test=LessThanOrEqual, EveryFrame=true, SetBool=(unset), SuccessEvent=→"EXIT", FailEvent=(none))
    on INTERACT → Open map
    on SING → Start Sing
    on SHAKE → Wait for Turn
    on HIT → Hit
    on EXIT → Outside Range
  }

  state End Travel Interaction {
    GetDirection2D(From=Self, To=var "Hero", StoreVector=(unset), StoreX=var "Direction To Hero", StoreY=(unset), EveryFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Direction To Hero", float2=var "X Scale", operation=Multiply, storeResult=var "Direction To Hero", everyFrame=false)
    FloatSignToBool(Value=var "Direction To Hero", StoreIsPositive=var "Start Facing Left", EveryFrame=false)
    BoolFlip(boolVariable=var "Start Facing Left")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=var "Travel Interaction", UseChildren=true)
    on FINISHED → Idle
  }

  state Is Already Present? {
    GetPlayerDataVariable(VariableName="FastTravelNPCLocation", StoreValue=var "Current Location")
    EnumCompare(enumVariable=var "Current Location", compareTo=var "This Location", equalEvent=→"TRUE", notEqualEvent=→"FALSE", storeResult=(unset), everyFrame=false)
    on FALSE → Wait For Call
    on TRUE → Start State
  }

  state Activate {

    ActivateGameObject(gameObject=var "Travel Interaction", activate=true, recursive=false, resetOnExit=false, everyFrame=false)

    SendEventByNameV2(eventTarget=GameObjectFSM(Self, fsm="Purr Audio"), sendEvent="ACTIVE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

    SendEventToRegister(eventName="BELL RUMBLE START")
    DoCameraShake(VisibleRenderer=var "", Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Bone Beast NPC Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)

  }

    SendEventToRegister(eventName="BELL SHAKE")
    SendEventToRegister(eventName="BEAST APPEAR")

    on FINISHED → Burst Out End
  }

    ActivateInteractible(Target=var "Travel Interaction", Activate=false, AllowQueueing=true, UseChildren=true)
    AudioPlay(gameObject=var "Sing Player", volume=1, oneShotClip=<null>, WaitForEndOfClip=false, finishedEvent=(none))
    ConvertBoolToString(boolVariable=var "Was Facing Left", stringVariable=var "Anim", falseString="Sing Right", trueString="Sing", everyFrame=false)

    SendEventToRegister(eventName="BELL RUMBLE START")
    DoCameraShake(VisibleRenderer=var "", Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)

  }

  }

    PlayerDataVariableTest(VariableName="IsTeleporting", ExpectedValue=true, IsExpectedEvent=→"WAKE", IsNotExpectedEvent=(none))
    PlayerDataVariableTest(VariableName="travelling", ExpectedValue=true, IsExpectedEvent=→"TRAVEL", IsNotExpectedEvent=(none))

    on SLEEP → Start Asleep
    on TRAVEL → Wait Finished Entering
  }

    SetInteractableLabel(Target=var "Travel Interaction", Label=Call, UseChildren=true)
    on INTERACT → Hero Call
    on WAKE → Wake
  }

  }

    SendEventToRegister(eventName="END FOLLOWERS INSTANT")
    ClearHeroEffectsLite()
    SetPlayerDataString(gameObject=var "GameManager", stringName="nextScene", value=var "To Scene")

    on INVALID → Invalid
    on FINISHED → Fade
  }
