# Control
# 1 location(s):
#   Song_10 / Steam Column Timer

fsm Control {
  start Init

  state Init {
    Wait(time=1, finishEvent=(none), realTime=false)
    FindNamedChild(gameObject=Self, storeResult=var "Shake Region v2")
    SetFsmBool(gameObject=var "Shake Region v2", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    on FINISHED → Hero In Range?
  }

  state Steam On {
    Wait(time=1.5, finishEvent=(none), realTime=false)
    SendEventToRegister(eventName="STEAM ON")
    SetFsmBool(gameObject=var "Shake Region v2", fsmName="Shaker", variableName="Do Shake", setValue=true, everyFrame=false)
    on FINISHED → Steam Off
  }

  state Steam Off {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    SendEventToRegister(eventName="STEAM OFF")
    SetFsmBool(gameObject=var "Shake Region v2", fsmName="Shaker", variableName="Do Shake", setValue=false, everyFrame=false)
    on FINISHED → Hero In Range?
  }

  state Hero In Range? {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Active Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Start Pause
  }

  state Start Pause {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Steam On
  }

  var Shake Region v2: gameObject = <null>
}
