  start State 1

  state State 1 {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"BREAK", storeCollider=var "")
    on BREAK → State 2
  }

  state State 2 {
    GetParent(gameObject=Self, storeResult=var "PArent")
    SendEventByName(eventTarget=GameObject(var "PArent"), sendEvent="BREAK", delay=0, everyFrame=false)
    SendEvent(eventTarget=GameObject(var "PArent"), sendEvent=(none), delay=0, everyFrame=false)  // disabled
    on FINISHED → State 1
  }

  var PArent: gameObject = <null>
}
