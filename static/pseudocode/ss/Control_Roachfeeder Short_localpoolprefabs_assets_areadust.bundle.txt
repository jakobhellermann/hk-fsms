# Control
# 1 location(s):
#   localpoolprefabs_assets_areadust.bundle / Roachfeeder Short

fsm Control {
  start Init
  on ZERO HP → Zero HP  // from any state
  on TO PATROL → Return To Patrol  // from any state

  state Init {
    SetParent(gameObject=var "Patrol Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Start Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetGameObject(variable=var "Next Point", gameObject=var "Patrol Point", everyFrame=false)
    GameObjectIsNull(gameObject=var "Enemy Target", isNull=(none), isNotNull=→"ENEMY TARGET", storeResult=(unset), everyFrame=false)
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Start Anim")
    StringCompare(stringVariable=var "Start Anim", compareTo="Flap In", equalEvent=→"FLY IN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Start Anim", compareTo="Find Spawn Point", equalEvent=→"SPAWN POINT", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → To Next Point
    on FLY IN → Fly In Ready
    on SPAWN POINT → Fly In Ready 2
    on ENEMY TARGET → Enemy Target Buzz
  }

  state Next Point {
    RandomBool(storeResult=var "Will Turn")
    GameObjectCompare(gameObjectVariable=var "Next Point", compareTo=var "Patrol Point", equalEvent=→"START", notEqualEvent=→"PATROL", storeResult=(unset), everyFrame=false)
    on PATROL → To Patrol
    on START → To Start
  }

  state To Patrol {
    SetGameObject(variable=var "Next Point", gameObject=var "Patrol Point", everyFrame=false)
    on FINISHED → To Next Point
  }

  state To Start {
    SetGameObject(variable=var "Next Point", gameObject=var "Start Point", everyFrame=false)
    on FINISHED → To Next Point
  }

  state Startle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Startle")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToStartle", resetFrame=true, everyFrame=false, pauseBetweenTurns=0.5)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=8)
    ActivateGameObject(gameObject=var "Attack Range", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ST_yell (sfxdynamic_assets_areaswamp.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Fly
  }

  state To Next Point {
    FireAtTarget(gameObject=Self, target=var "Next Point", speed=8, position=(0, 0, 0), spread=0, everyFrame=false)
    GetDistance(gameObject=Self, target=var "Next Point", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=→"ARRIVE", lessThan=→"ARRIVE", greaterThan=(none), everyFrame=true)
    Wait(time=1, finishEvent=→"ARRIVE", realTime=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=(unset), sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ARRIVE → Arrive
    on ALERT → Startle
    on TOOK DAMAGE → Startle
    on SING → Sing
  }

  state Arrive {
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    WaitRandom(timeMin=0.75, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=(unset), sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Move Choice
    on ALERT → Startle
    on TOOK DAMAGE → Startle
    on SING → Sing
  }

  state Move Choice {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    SendRandomEventV4(events=[→"MOVE", →"FLIT"], weights=[1, 1], eventMax=[2, 5], missedMax=[9, 9], activeBool=(unset))
    on MOVE → Next Point
    on FLIT → Flit
    on SING → Sing
  }

  state Flit {
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    RandomFloat(min=8, max=12, storeResult=var "Speed")
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    FireAtTarget(gameObject=Self, target=var "Next Point", speed=var "Speed", position=(0, 0, 0), spread=2, everyFrame=false)
    WaitRandom(timeMin=0.5, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=(unset), sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Move Choice
    on ALERT → Startle
    on TOOK DAMAGE → Startle
    on SING → Sing
  }

  state Fly {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    DistanceFly(gameObject=Self, target=var "Hero", distance=6, speedMax=6, acceleration=0.35, targetsHeight=false, height=(unset), minAboveHero=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Chase")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToChase", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"ATTACK", realTime=false)
    on TOOK DAMAGE → Get In Range
    on ATTACK → Get In Range
    on SING → Sing
  }

  state Get In Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=(unset), sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    DistanceFly(gameObject=Self, target=var "Hero", distance=5, speedMax=6, acceleration=0.35, targetsHeight=false, height=(unset), minAboveHero=1)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToChase", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    CheckAlertRange(alertRange=Roachfeeder Short/Unalert Range@AlertRange, storeResult=(unset), InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=→"UNALERT", OutOfRangeDelay=3, everyFrame=true)
    on ATTACK → Throw Antic
    on SING → Sing
    on UNALERT → Idle Fly
  }

  state Throw Antic {
    PlayAudioEvent(audioClip=roach_feeder_short_antic_and_throw (sfxstatic_assets_areagreymoor.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Throw Player")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ST_yell_short (sfxdynamic_assets_areaswamp.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Throw Antic", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    DecelerateV2(gameObject=Self, deceleration=0.875, brakeOnExit=false)  // disabled
    DistanceFly(gameObject=Self, target=var "Hero", distance=8, speedMax=8, acceleration=0.5, targetsHeight=false, height=(unset), minAboveHero=1)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="TurnToChase", resetFrame=false, everyFrame=true, pauseBetweenTurns=0.5)
    on FINISHED → Aim Lock
  }

  state Throw {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Throw", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ST_yell (sfxdynamic_assets_areaswamp.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SpawnObjectFromGlobalPool(gameObject=Caltrop Ball, spawnPoint=var "Self", position=(0, 0, -0.0016), rotation=(unset), storeObject=var "Ball")
    SetVelocityAsAngle(gameObject=var "Ball", angle=var "Angle", speed=20, everyFrame=false)
    AddForce2d(gameObject=var "Ball", forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=5, vector3=(unset), everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)  // disabled
    on FINISHED → Throw Recover
  }

  state Throw Recover {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    RandomFloat(min=0.5, max=1, storeResult=var "Speed")
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fly
    on TOOK DAMAGE → Fly
  }

  state Aim Lock {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)  // disabled
    DecelerateV2(gameObject=Self, deceleration=0.875, brakeOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=2, storeAngle=var "Angle", pause=0, everyFrame=false)
    ClampAngleByScale(gameObject=Self, space=0, angleVariable=var "Angle", positiveMinValue=180, positiveMaxValue=260, negativeMinValue=280, negativeMaxValue=360, everyFrame=false)
    on FINISHED → Throw
  }

  state Sing {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    EnemySingControl(enemyGameObject=Self, audioPlayer=Roachfeeder Short/Audio Loop Voice, singAudioTable=roachfeeder_short_sing (sfxdynamic_assets_areadust.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fly
  }

  state Zero HP {
    AudioStop(gameObject=var "Throw Player", fadeTime=0)
  }

  state Fly In Ready {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on BATTLE START → Check Pos
    on WAKE → Fly In
  }

  state Check Pos {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=8, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=false)
    on SHIFT → Shift
    on FINISHED → Fly In
  }

  state Shift {
    SetParent(gameObject=var "Spawn Pos Alt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Spawn Pos Alt", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    on FINISHED → Fly In
  }

  state Fly In {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=10, z=-15, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ST_yell (sfxdynamic_assets_areaswamp.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetMeshRenderer(gameObject=Self, active=true)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -10, 15), time=0.75, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.75)
    on FINISHED → Activate
  }

  state Activate {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    ActivateGameObject(gameObject=var "Attack Range", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fly
  }

  state Idle Fly {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=(unset), sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    IdleBuzzV3(gameObject=Self, waitMin=1, waitMax=3, speedMax=5, accelerationMin=3, accelerationMax=5, roamingRangeX=2, roamingRangeY=2, manualStartPos=(unset))
    ActivateGameObject(gameObject=var "Attack Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Alert Range", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on ALERT → Startle
    on TOOK DAMAGE → Startle
  }

  state Fly In Ready 2 {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    SetBoolValue(boolVariable=var "Stay Alert", boolValue=true, everyFrame=false)
    on BATTLE START → Find Pos
    on WAKE → Find Pos
  }

  state Find Pos {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Hero", xOffset=0, yOffset=0, zOffset=0, overrideZ=var "Z Pos", everyFrame=false)
    FindClosest(gameObject=Self, withTag="Teleplane", ignoreOwner=true, mustBeVisible=false, storeObject=var "Spawn Point", storeDistance=(unset), everyFrame=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Spawn Point", xOffset=0, yOffset=0, zOffset=0, overrideZ=var "Z Pos", everyFrame=false)
    on FINISHED → Fly In
  }

  state Return To Patrol {
    SetGameObject(variable=var "Next Point", gameObject=var "Patrol Point", everyFrame=false)
    DistanceFly(gameObject=Self, target=var "Next Point", distance=0, speedMax=7, acceleration=0.3, targetsHeight=false, height=(unset), minAboveHero=(unset))
    GetDistanceV2(gameObject=Self, target=var "Next Point", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=3, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    on FINISHED → Next Point
  }

  state Enemy Target Buzz {
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=5, everyFrame=false)
    DistanceFly(gameObject=Self, target=var "Enemy Target", distance=4, speedMax=7, acceleration=0.3, targetsHeight=true, height=2, minAboveHero=(unset))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Chase")
    FaceObjectV2(objectA=Self, objectB=var "Enemy Target", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToChase", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"ALERT", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    WaitRandom(timeMin=1.5, timeMax=2.5, finishEvent=→"ATTACK", realTime=false)
    on TOOK DAMAGE → Startle
    on ALERT → Startle
    on ATTACK → NPC Shout
  }

  state NPC Shout {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Startle")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ST_yell (sfxdynamic_assets_areaswamp.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Enemy Target Buzz
    on TOOK DAMAGE → Startle
  }

  var Stay Alert: bool = false
  var Enemy Target: gameObject = <null>

  var Angle: float = 0
  var Distance: float = 0
  var Speed: float = 0
  var Z Pos: float = 0
  var Will Turn: bool = false
  var Start Anim: string = ""
  var Alert Range: gameObject = Roachfeeder Short/Alert Range
  var Attack Range: gameObject = Roachfeeder Short/Attack Range
  var Ball: gameObject = <null>
  var Next Point: gameObject = <null>
  var Patrol Point: gameObject = Roachfeeder Short/Patrol Point
  var Self: gameObject = Roachfeeder Short
  var Start Point: gameObject = Roachfeeder Short/Start Point
  var Throw Player: gameObject = <null>
  var Spawn Pos Alt: gameObject = Roachfeeder Short/Spawn Pos Alt
  var Spawn Point: gameObject = <null>
  var Audio Loop Voice: gameObject = Roachfeeder Short/Audio Loop Voice
}
