// uses template: steam_slash_effect
fsm steam_slash_effect {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"DAMAGED", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on DAMAGED → Dmg Effect
  }

  state Dmg Effect {
    BoolTest(boolVariable=var "Junk Chute", isTrue=→"JUNK CHUTE", isFalse=(none), everyFrame=false)
    GetColliderRange(gameObject=Self, minX=var "Min X", maxX=var "Max X", minY=var "Min Y", maxY=var "Max Y", everyFrame=false)
    GetPosition(gameObject=var "Damager", vector=(unset), x=var "Effect X", y=var "Effect Y", z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Effect X", minValue=var "Min X", maxValue=var "Max X", everyFrame=false)
    FloatClamp(floatVariable=var "Effect Y", minValue=var "Min Y", maxValue=var "Max Y", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Steam Cut Effect (localpoolprefabs_assets_areaorganareaunderstore.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Effect")
    SetPosition(gameObject=var "Effect", vector=(unset), x=var "Effect X", y=var "Effect Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
    on JUNK CHUTE → Junk Chute
  }

  state Junk Chute {
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=var "Min X", y=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Max X", floatValue=var "Min X", everyFrame=false)
    FloatAdd(floatVariable=var "Min X", add=-2, everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Max X", add=2, everyFrame=false, perSecond=false)
    GetPosition(gameObject=var "Damager", vector=(unset), x=var "Effect X", y=var "Effect Y", z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Effect X", minValue=var "Min X", maxValue=var "Max X", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Steam Cut Effect (localpoolprefabs_assets_areaorganareaunderstore.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Effect")
    SetPosition(gameObject=var "Effect", vector=(unset), x=var "Effect X", y=var "Effect Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  var Junk Chute: bool = true

  var Effect X: float = 0
  var Effect Y: float = 0
  var Max X: float = 0
  var Max Y: float = 0
  var Min X: float = 0
  var Min Y: float = 0
  var Damager: gameObject = <null>
  var Effect: gameObject = <null>
}
