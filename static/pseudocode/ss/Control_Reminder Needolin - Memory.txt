# Control
# 1 location(s):
#   Memory_Needolin / Reminder Needolin - Memory

fsm Control {
  start Pause
  on GATE OPEN → Gate Open  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Show Reminder Range", trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    BoolTest(boolVariable=var "Weaver Gate Open", isTrue=→"GATE OPEN", isFalse=(none), everyFrame=true)
    on ENTER → Send
  }

  state Activated {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
  }

  state Send {
    SendEventToRegister(eventName="REMINDER NEEDOLIN")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "Weaver Gate Open", isTrue=→"GATE OPEN", isFalse=(none), everyFrame=true)
    on FINISHED → Sent
  }

  state Sent {
    Trigger2dEvent(gameObject=var "Hide Reminder Range", trigger=1, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    BoolTest(boolVariable=var "Weaver Gate Open", isTrue=→"GATE OPEN", isFalse=(none), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"SING", IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on EXIT → Hide Reminder
    on SING → Wait to Hide
  }

  state Hide Reminder {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="REMINDER NEEDOLIN END")
    BoolTest(boolVariable=var "Weaver Gate Open", isTrue=→"GATE OPEN", isFalse=(none), everyFrame=true)
    on FINISHED → Idle
  }

  state Gate Open {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventToRegister(eventName="REMINDER NEEDOLIN END")
    BoolTest(boolVariable=var "Weaver Gate Open", isTrue=(none), isFalse=→"GATE CLOSED", everyFrame=true)
    on GATE CLOSED → Hide Reminder
  }

  state Init {
    GetEntryGateName(StoreValue=var "Gate Name")
    StringCompare(stringVariable=var "Gate Name", compareTo="right1", equalEvent=→"ACTIVATE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activated
  }

  state Hide While Playing {
    SendEventToRegister(eventName="REMINDER NEEDOLIN END")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on CANCEL → Send
  }

  state Wait to Hide {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Hide While Playing
    on CANCEL → Sent
  }

  var Y Distance: float = 0

  var Activated: bool = false
  var Weaver Gate Open: bool = false
  var Gate Name: string = ""
  var Show Reminder Range: gameObject = Reminder Needolin - Memory/Show Reminder Range
  var Hide Reminder Range: gameObject = Reminder Needolin - Memory/Hide Reminder Range
}
