# Control
# 1 location(s):
#   Dock_09 / Boss Scene/Dock Guard Slasher/Projectile

fsm Control {
  start Init

  state Init {
    on FINISHED → Dormant
  }

  state Fly {
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=var "Pull Force", y=(unset), vector3=(unset), everyFrame=true)
    PlayParticleEmitterChildren(gameObject=var "Pt Projectile Trail", resetTimeIfPlaying=false, stopOnStateExit=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on CATCH → Dormant
  }

  state Launch {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetScale(gameObject=Self, vector=(unset), xScale=var "Throw Speed", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Throw Speed", multiplyBy=-36, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Throw Speed", y=(unset), everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Pull Force", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Pull Force", multiplyBy=50, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    on FINISHED → Fly
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on THROW → Launch
  }

  var Throw Speed: float = 0
  var Pull Force: float = 0
  var Pt Projectile Trail: gameObject = Boss Scene/Dock Guard Slasher/Projectile/Pt Projectile Trail
}
