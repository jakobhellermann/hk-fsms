# Control
# 1 location(s):
#   Dock_09 / Boss Scene

fsm Control {
  start Pause
  on LAVA START → Lava Start  // from any state

  state Pause {
    FindNamedChild(gameObject=Self, storeResult=var "Dock Guard Thrower")
    ActivateGameObject(gameObject=var "Dock Guard Thrower", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava_Waterfall L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava_Waterfall R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Dock Guard Slasher")
    FindNamedChild(gameObject=Self, storeResult=var "Minions")
    FindNamedChild(gameObject=Self, storeResult=var "Minions 2")
    FindNamedChild(gameObject=Self, storeResult=var "Battle Start Range")
    FindNamedChild(gameObject=Self, storeResult=var "Gates")
    FindNamedChild(gameObject=Self, storeResult=var "CamLocks")
    FindNamedChild(gameObject=Self, storeResult=var "CamLocks Pre")
    FindNamedChild(gameObject=Self, storeResult=var "Lava_Waterfall L")
    FindNamedChild(gameObject=Self, storeResult=var "Lava_Waterfall R")
    FindNamedChild(gameObject=Self, storeResult=var "Lava_Waterfall M")
    FindNamedChild(gameObject=Self, storeResult=var "Silk Heart")
    FindNamedChild(gameObject=Self, storeResult=var "Intro Minions")
    FindNamedChild(gameObject=Self, storeResult=var "Slide Floors")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop LavaFall")
    FindNamedChild(gameObject=Self, storeResult=var "CamLocks LavaFall")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Tension")  // disabled
    ActivateGameObject(gameObject=var "CamLocks LavaFall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dock Guard Thrower", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava_Waterfall L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava_Waterfall R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="defeatedDockForemen", isTrue=→"DEFEATED", isFalse=(none))
    PlayerDataBoolTest(boolName="encounteredDockForemen", isTrue=→"ENCOUNTERED", isFalse=(none))
    on FINISHED → Battle Ready
    on DEFEATED → Defeated
    on ENCOUNTERED → Battle Ready
  }

  state Battle Ready {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Battle Start Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    ActivateGameObject(gameObject=var "CamLocks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckHeroPerformanceRegionV2(Target=var "Dock Guard Slasher", Radius=7, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=(none), ActiveOuter=→"MINION DAMAGED", IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ENTER → Arena Start
    on MINION DAMAGED → Wait until Hero In Arena
  }

  state Arena Start {
    SetPlayerDataBool(boolName="encounteredDockForemen", value=true)
    ActivateGameObject(gameObject=var "Battle Start Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Intro Minions"), sendEvent="LEAVE", delay=0, everyFrame=false)
    SendEventToRegister(eventName="TENSION END")
    SendEventByName(eventTarget=GameObject(var "Audio Loop Tension"), sendEvent="END", delay=0, everyFrame=false)  // disabled
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "CamLocks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks LavaFall", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Slasher Start
  }

  state Slasher Start {
    SendEventByName(eventTarget=GameObject(var "Dock Guard Slasher"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Slasher Dead", isTrue=→"SLASHER KILLED", isFalse=(none), everyFrame=true)
    on CALL ENEMIES → Send Minions 1
    on SLASHER KILLED → Send Thrower Pause
    on CALL THROWER → Send Thrower Pause
  }

  state Send Minions 1 {
    SendEventByName(eventTarget=GameObject(var "Minions"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Slasher Dead", isTrue=→"SLASHER KILLED", isFalse=(none), everyFrame=true)
    FloatAdd(floatVariable=var "Minions 1 Time", add=1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Minions 1 Time", minValue=0, maxValue=100, everyFrame=true)
    on CALL THROWER → Send Thrower Pause
    on SLASHER KILLED → Send Thrower Pause
    on CALL ENEMIES → Send Minions 2
  }

  state Send Thrower {
    ActivateGameObject(gameObject=var "Dock Guard Thrower", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Slasher Dead", isTrue=→"SLASHER KILLED", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Thrower Dead", isTrue=→"THROWER KILLED", isFalse=(none), everyFrame=true)
    on SLASHER KILLED → Upgrade Thrower
    on THROWER KILLED → Upgrade Slasher
  }

  state Upgrade Thrower {
    SetBoolValue(boolVariable=var "Slasher Dead", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Music End", variableName="Slasher Defeated", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dock Guard Thrower", fsmName="Control", variableName="Lava Mode", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dock Guard Thrower", fsmName="Control", variableName="Short Death", setValue=false, everyFrame=false)
    SetFsmFloat(gameObject=var "Dock Guard Thrower", fsmName="Control", variableName="Idle Time", setValue=0.25, everyFrame=false)
    CompareHP(enemy=var "Dock Guard Thrower", integer2=350, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "Dock Guard Thrower", behaviour="HealthManager", methodName="AddHP", parameters=[200, 350], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Final Phase
  }

  state Lava Pause {
    Wait(time=0, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Final Phase
  }

  state Upgrade Slasher {
    SetBoolValue(boolVariable=var "Thrower Dead", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Music End", variableName="Thrower Defeated", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dock Guard Slasher", fsmName="Control", variableName="Lava Mode", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dock Guard Slasher", fsmName="Control", variableName="Short Death", setValue=false, everyFrame=false)
    SetFsmFloat(gameObject=var "Dock Guard Slasher", fsmName="Control", variableName="Idle Time", setValue=0.1, everyFrame=false)
    CompareHP(enemy=var "Dock Guard Slasher", integer2=350, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "Dock Guard Slasher", behaviour="HealthManager", methodName="AddHP", parameters=[200, 350], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Final Phase
  }

  state Final Phase {
    ActivateGameObject(gameObject=var "Lava_Waterfall L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava_Waterfall R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolAllTrue(boolVariables=[var "Thrower Dead", var "Slasher Dead"], sendEvent=→"END", storeResult=(unset), everyFrame=true)
    SendEventByName(eventTarget=GameObject(var "Lava_Waterfall M"), sendEvent="LAVA START", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Loop LavaFall", volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Loop LavaFall", startVolume=0, endVolume=0.75, time=1)
    ActivateGameObject(gameObject=var "CamLocks LavaFall", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on END → End Pause
  }

  state Defeated {
    ActivateGameObject(gameObject=var "CamLocks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks LavaFall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks Pre", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dock Guard Slasher", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Minions", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Intro Minions", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state End Pause {
    SetPlayerDataBool(boolName="defeatedDockForemen", value=true)
    RecordJournalKill(Record=Dock Guard Thrower (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    SendEventToRegister(eventName="FANFARE BOSS DEFEAT")
    on FINISHED → Lava End
  }

  sequence state Lava End {
    SendEventByName(eventTarget=GameObject(var "Lava_Waterfall L"), sendEvent="LAVA END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lava_Waterfall R"), sendEvent="LAVA END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lava_Waterfall M"), sendEvent="LAVA END", delay=0, everyFrame=false)
    on FINISHED → Slide floors
  }

  state Finish {
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks LavaFall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Spawn Silkheart {
    ActivateGameObject(gameObject=var "Silk Heart", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lava_Waterfall M"), sendEvent="LAVA END", delay=0, everyFrame=false)
    on HEART COLLECTED → Lava End
  }

  state Wait until Hero In Arena {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Aggro Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Arena Start
  }

  state Send Thrower Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Thrower
  }

  sequence state Slide floors {
    FadeAudio(gameObject=var "Audio Loop LavaFall", startVolume=1, endVolume=0, time=2)
    SendEventByName(eventTarget=GameObject(var "Slide Floors"), sendEvent="CLOSE", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    on FINISHED → Finish
  }

  state Lava Start {
    ActivateGameObject(gameObject=var "Lava_Waterfall L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava_Waterfall R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Loop LavaFall", volume=0, oneShotClip=<null>)
    BoolTest(boolVariable=var "Slasher Dead", isTrue=→"SLASHER KILLED", isFalse=(none), everyFrame=true)
    on CALL ENEMIES → Send Minions 1
    on SLASHER KILLED → Send Thrower Pause
    on CALL THROWER → Send Thrower Pause
  }

  state Send Minions 2 {
    SendEventByName(eventTarget=GameObject(var "Minions 2"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Slasher Dead", isTrue=→"SLASHER KILLED", isFalse=(none), everyFrame=true)
    FloatAdd(floatVariable=var "Minions 2 Time", add=1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Minions 2 Time", minValue=0, maxValue=100, everyFrame=true)
    on CALL THROWER → Send Thrower Pause
    on SLASHER KILLED → Send Thrower Pause
  }

  var Minions 1 Time: float = 0
  var Minions 2 Time: float = 0
  var Slasher Dead: bool = false
  var Thrower Dead: bool = false
  var Dock Guard Slasher: gameObject = <null>
  var Dock Guard Thrower: gameObject = <null>
  var Minions: gameObject = <null>
  var Battle Start Range: gameObject = <null>
  var Gates: gameObject = <null>
  var CamLocks: gameObject = <null>
  var Lava_Waterfall L: gameObject = <null>
  var Lava_Waterfall R: gameObject = <null>
  var Lava_Waterfall M: gameObject = <null>
  var Silk Heart: gameObject = <null>
  var Intro Minions: gameObject = <null>
  var CamLocks Pre: gameObject = <null>
  var Slide Floors: gameObject = <null>
  var Audio Loop LavaFall: gameObject = <null>
  var CamLocks LavaFall: gameObject = <null>
  var Audio Loop Tension: gameObject = <null>
  var Minions 2: gameObject = <null>
}
