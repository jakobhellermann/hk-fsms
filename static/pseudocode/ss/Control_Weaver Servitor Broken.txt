# Control
# 6 location(s):
#   Abyss_08 / Weaver Servitor Broken
#   Bone_East_Weavehome / Weaver Servitor Broken
#   Peak_02 / Weaver Servitor Broken
#   Peak_08b / Weaver Servitor Broken
#   Peak_10 / Weaver Servitor Broken
#   Shadow_Weavehome / Weaver Servitor Broken

fsm Control {
  start Wait Frame
  on ZERO HP → Death  // from any state

  state Init {
    FindNamedChild(gameObject=var "Head Swiveller", storeResult=var "Head")
    FindNamedChild(gameObject=var "Head", storeResult=var "Beam Light")
    SetParent(gameObject=var "Head Swiveller", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetScale(gameObject=var "Head Swiveller", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=2.5, MaxReactDelay=3.5, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on NEEDOLIN → Check Dir
  }

  state Check Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Aim L
    on R → Aim R
  }

  state Aim L {
    SetFloatValue(floatVariable=var "Angle", floatValue=20, everyFrame=false)
    on FINISHED → Look
  }

  state Look {
    AnimateRotationTo(gameObject=var "Head", fromValue=(unset), toValue=var "Angle", worldSpace=false, negativeSpace=false, time=1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    ActivateGameObject(gameObject=var "Beam Light", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Head", fsmName="Control", variableName="Active", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Head Swiveller"), sendEvent="LOOK START", delay=0, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=0.6, oneShotClip=weaver_servitor_notice_hero (sfxstatic_assets_areaweaver.bundle))
    AudioPlaySimple(gameObject=var "Whir Loop", volume=1, oneShotClip=<null>)
    on END → End
  }

  state Aim R {
    SetFloatValue(floatVariable=var "Angle", floatValue=-20, everyFrame=false)
    on FINISHED → Look
  }

  state End {
    AnimateRotationTo(gameObject=var "Head", fromValue=var "Angle", toValue=0, worldSpace=false, negativeSpace=false, time=1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Head Swiveller"), sendEvent="LOOK STOP", delay=0, everyFrame=false)
    on FINISHED → Head End
  }

  state Death {
    ActivateGameObject(gameObject=var "Head", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Head End {
    AudioStop(gameObject=var "Whir Loop", fadeTime=0)
    SetFsmBool(gameObject=var "Head", fsmName="Control", variableName="Active", setValue=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Beam Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=weaver_servitor_go_to_sleep (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Idle
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  var Angle: float = 0
  var Head Swiveller: gameObject = Weaver Servitor Broken/Head Swiveller
  var Head: gameObject = <null>
  var Beam Light: gameObject = <null>
  var Whir Loop: gameObject = Weaver Servitor Broken/Whir Loop
}
