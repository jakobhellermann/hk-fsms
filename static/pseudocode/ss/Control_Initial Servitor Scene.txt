# Control
# 1 location(s):
#   Weave_02 / Group:1/Initial Servitor Scene

fsm Control {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WAKE", delay=0, everyFrame=false)
  }
}
