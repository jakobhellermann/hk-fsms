# Control
# 1 location(s):
#   Crawl_07 / Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm

fsm Control {
  start Init
  on ZERO HP → Death 1  // from any state
  on MOVE STOP → Stop  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Nest Parent")
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Small Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Body Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Chomp Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on SPAWN → Position
  }

  state Reparent {
    SetParent(gameObject=Self, parent=var "Nest Parent", resetLocalPosition=false, resetLocalRotation=false)
    StopParticleEmitter(gameObject=var "Dig Dust")
    AddHP(target=Self, amount=(unset), healToMax=true)
    SetBoolValue(boolVariable=var "Silk Dash Death", boolValue=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=var "Ground Bouncer", active=false, resetOnExit=false)
    on FINISHED → Dormant
  }

  state Position {
    SetParent(gameObject=Self, parent=var "Nest", resetLocalPosition=false, resetLocalRotation=false)
    GetFsmFloat(gameObject=var "Nest", fsmName="Nest Control", variableName="Max X", storeValue=var "Max X", everyFrame=false)
    GetFsmFloat(gameObject=var "Nest", fsmName="Nest Control", variableName="Min X", storeValue=var "Min X", everyFrame=false)
    GetFsmFloat(gameObject=var "Nest", fsmName="Nest Control", variableName="Ground Y", storeValue=var "Ground Y", everyFrame=false)
    GetPosition2D(gameObject=var "Nest", vector=(unset), x=(unset), y=var "Nest Y", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Ground Y", add=var "Nest Y", everyFrame=false, perSecond=false)
    GetFsmFloat(gameObject=var "Nest", fsmName="Nest Control", variableName="Top Y", storeValue=var "Top Y", everyFrame=false)
    RandomFloat(min=var "Min X", max=var "Max X", storeResult=var "Spawn X")
    SetPosition(gameObject=Self, vector=(unset), x=var "Spawn X", y=var "Top Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Antic
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Antic Dust")
    PlayParticleEmitterInState(gameObject=var "Antic Rocks")
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=var "Solid", Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)  // disabled
    SetPosition(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=7, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Translate(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=-10, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SetCollider(gameObject=var "Ground Bouncer", active=true, resetOnExit=false)
    PlayAudioEventRandomBool(audioClips=[bone_worm_antic_1 (sfxstatic_assets_areabone.bundle), bone_worm_antic_2 (sfxstatic_assets_areabone.bundle), bone_worm_antic_3 (sfxstatic_assets_areabone.bundle)], doPlay=var "Play Sounds", pitchMin=0.85, pitchMax=1.15, volume=0.7, audioPlayerPrefab=Audio Player Actor ShortRange@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    on FINISHED → Drop Start
  }

  state Drop Start {
    SetMeshRenderer(gameObject=Self, active=true)
    PlayAudioEventRandomBool(audioClips=[bone_worm_fall_1 (sfxstatic_assets_areabone.bundle), bone_worm_fall_2 (sfxstatic_assets_areabone.bundle), bone_worm_fall_3 (sfxstatic_assets_areabone.bundle)], doPlay=var "Play Sounds", pitchMin=0.85, pitchMax=1.15, volume=0.7, audioPlayerPrefab=Audio Player Actor VeryShortRange@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Crawl")
    DoCameraShake(VisibleRenderer=var "Solid", Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)  // disabled
    ActivateGameObject(gameObject=var "Body Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Chomp Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Drop
  }

  state Dig 1 {
    ActivateGameObject(gameObject=var "Chomp Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEventRandomBool(audioClips=[bone_worm_ground_impact_1 (sfxstatic_assets_areabone.bundle), bone_worm_ground_impact_2 (sfxstatic_assets_areabone.bundle), bone_worm_ground_impact_3 (sfxstatic_assets_areabone.bundle)], doPlay=var "Play Sounds", pitchMin=0.85, pitchMax=1.15, volume=0.7, audioPlayerPrefab=Audio Player Actor VeryShortRange@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dig", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dig Dust", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Bits", emit=0, resetIfPlaying=false)
    SetPosition(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    Translate(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=-8, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SetCollider(gameObject=var "Ground Bouncer", active=true, resetOnExit=false)
    on FINISHED → Dig 2
  }

  state Dig 2 {
    ActivateGameObject(gameObject=var "Small Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Body Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Translate(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=-8, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Dig 3
  }

  state Dig 3 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Small Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Translate(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=-8, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Dig 4
  }

  state Death 1 {
    SetCollider(gameObject=var "Ground Bouncer", active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Small Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStop(gameObject=var "Current Audio Player", fadeTime=0)
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    StopParticleEmittersInChildren(gameObject=Self)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Body Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Chomp Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Death Dir
  }

  state Death Air {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=1, everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=var "Ground Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    on LAND → Splat
  }

  sequence state Splat {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Heal
  }

  state Stop {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
  }

  state Death Dir {
    BoolTest(boolVariable=var "Silk Dash Death", isTrue=→"SILK DASH", isFalse=(none), everyFrame=false)
    SetVelocityByScale(gameObject=Self, speed=-10, ySpeed=12, everyFrame=false)
    on FINISHED → Death Air
    on SILK DASH → Silk Dashed
  }

  state Silk Dashed {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetVelocityByScale(gameObject=Self, speed=-6, ySpeed=18, everyFrame=false)
    on FINISHED → Death Air
  }

  state Multichomp {
    PlayAudioEvent(audioClip=hornet_bone_worm_multi_damager (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.003, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-0.25, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Chomp")
    SetCollider(gameObject=var "Ground Bouncer", active=false, resetOnExit=false)
    on FINISHED → Multichomp End
  }

  state Drop {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-19, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=1, everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=var "Ground Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Attack")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Crawl")
    on LAND → Dig 1
    on MULTI HIT CONNECT → Multichomp
  }

  state Multichomp End {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Drop
  }

  state Dig 4 {
    StopParticleEmitter(gameObject=var "Dig Dust")
    SetMeshRenderer(gameObject=Self, active=false)
    Translate(gameObject=var "Ground Bouncer", vector=(unset), x=(unset), y=-8, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reparent
  }

  state Heal {
    AddHP(target=Self, amount=9999, healToMax=false)
    on FINISHED → Reparent
  }

  var Ground Y: float = 0
  var Max X: float = 0
  var Min X: float = 0
  var Spawn X: float = 0
  var Top Y: float = 0
  var Y Pos: float = 0
  var Z Pos: float = 0
  var Nest Y: float = 0
  var Play Sounds: bool = false
  var Silk Dash Death: bool = false
  var Antic Dust: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Antic Dust
  var Antic Rocks: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Antic Rocks
  var Chomp Collider: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Chomp Collider
  var Current Audio Player: gameObject = <null>
  var Dig Dust: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Dig Dust
  var Nest: gameObject = <null>
  var Pt Attack: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Pt Attack
  var Pt Bits: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Pt Bits
  var Slam Effect: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Slam Effect
  var Small Hit: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Small Hit
  var Solid: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/black_solid
  var Body Damager: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Body Damager
  var Nest Parent: gameObject = <null>
  var Ground Bouncer: gameObject = Area_States/Uninfected/Bone Worm Nests/Worm Pool/Bone Worm/Ground Bouncer
}
