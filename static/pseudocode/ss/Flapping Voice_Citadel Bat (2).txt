# Flapping Voice
# 2 location(s):
#   Organ_01 / Citadel Bat (2)
#   Under_07c / Citadel Bat (2)

fsm Flapping Voice {
  start Idle
  on STOP FLAP → Idle  // from any state

  state Idle {
    on START FLAP → Pause
  }

  state Pause {
    WaitRandom(timeMin=1, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Voice
  }

  state Voice {
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[citadel_bat_flying_utterance_1 (sfxstatic_assets_areasongareaunderstore.bundle), citadel_bat_flying_utterance_2 (sfxstatic_assets_areasongareaunderstore.bundle), citadel_bat_flying_utterance_3 (sfxstatic_assets_areasongareaunderstore.bundle), citadel_bat_flying_utterance_4 (sfxstatic_assets_areasongareaunderstore.bundle), citadel_bat_flying_utterance_5 (sfxstatic_assets_areasongareaunderstore.bundle)], weights=[1, 1, 1, 1, 1], pitchMin=0.9, pitchMax=1.1, stopPreviousSound=false)
    on FINISHED → Pause
  }

  var Audio Loop Voice: gameObject = Citadel Bat (2)/Audio Loop Voice
}
