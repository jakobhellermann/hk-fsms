fsm Detect Water Entry {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer=Soft Terrain(25), sendEvent=→"SPLASH", storeCollider=var "")
    on SPLASH → Splash In
  }

  state Splash In {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="SPLASH IN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Pt Splash: gameObject = Vampire Gnat/Pt Splash (localpoolprefabs_assets_areaaqueduct.bundle)
  var Splash Effect: gameObject = Vampire Gnat/Splash Effect (localpoolprefabs_assets_areaaqueduct.bundle)
}
