# Conversation
# 1 location(s):
#   Coral_42 / Thief NPC Shop

fsm Conversation {
  start Pause
  on CLOSE SHOP WINDOW → End  // from any state

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=1.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    FaceObjectV4(ObjectA=Self, ObjectB=var "Hero", SpriteFacesRight=false, NewAnimationClip="TurnToIdle2", ResetFrame=true, PauseBetweenTurns=0.5, StoreDuration=(unset), turnAudioClipTable=small_movement_cloak (sfxdynamic_assets_shared.bundle), EveryFrame=true)
    on INTERACT → Speak
    on SING → Sing
    on STARTLE → Startle
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.7, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Grindle_sing (sfxdynamic_assets_areasong.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on END → Sing Finish
  }

  state Sing Finish {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state End {
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Idle
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetProperty(targetProperty=PlayMakerNPC.InteractLabel on Thief NPC Shop@PlayMakerNPC, everyFrame=false)
    on FINISHED → Resting
  }

  state Speak {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToIdle2", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    PlayerDataBoolTrueAndFalse(trueBool="blackThreadWorld", falseBool="metGrindleAct3", isTrue=→"MEET ACT 3", isFalse=(none))
    BoolTest(boolVariable=var "Activated", isTrue=(none), isFalse=→"MEET", everyFrame=false)
    BoolTest(boolVariable=var "Spoken", isTrue=(none), isFalse=→"GENERIC", everyFrame=false)
    on MEET → Intro
    on GENERIC → Repeat
    on FINISHED → Play Voice
    on MEET ACT 3 → Act 3 Intro
  }

  state Pause {
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Intro {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key="GRINDLE_SHOP_INTRO", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Hide Box
  }

  state Shop Up {
    GetShopObject(StoreObject=var "Shop Object", Target=Self)
    SendEventByName(eventTarget=GameObject(var "Shop Object"), sendEvent="SHOP UP", delay=0, everyFrame=false)
    SetProperty(targetProperty=PlayMakerNPC.InteractLabel on Thief NPC Shop@PlayMakerNPC, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on SHOP CLOSED → End
  }

  state Hide Box {
    EndDialogue(ReturnControl=false, ReturnHUD=true, Target=Self, UseChildren=false)
    SetBoolValue(boolVariable=var "Spoken", boolValue=true, everyFrame=false)
    on FINISHED → Shop Up
  }

  state Resting {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Enclave Rest")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Close Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"WAKE", ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on WAKE → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Enclave Wake", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Grindle_shop_react (sfxdynamic_assets_areacoral.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Come Forward
  }

  state Wake Land {
    PlayAudioEvent(audioClip=hornet_ground_land (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Start Interactive
  }

  state Come Forward {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.006, space=0, everyFrame=false, lateUpdate=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Wake Land
  }

  state Start Interactive {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  state Repeat {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    GetPlayerDataBool(boolName="blackThreadWorld", storeValue=var "Is Act 3")
    ConvertBoolToString(boolVariable=var "Is Act 3", stringVariable=var "Convo Key", falseString="GRINDLE_SHOP_REPEAT", trueString="GRINDLE_SHOP_REPEAT_ACT3", everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key=var "Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Hide Box
  }

  state Act 3 Intro {
    SetPlayerDataBool(boolName="metGrindleAct3", value=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key="GRINDLE_SHOP_INTRO_ACT3", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Hide Box
  }

  state Startle {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Grindle_alert (sfxdynamic_assets_areasong.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayAudioEvent(audioClip=grass_hopper_jump (sfxstatic_assets_areacoralareasong.bundle), pitchMin=1.15, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Idle
  }

  state Play Voice {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Grindle_talk (sfxdynamic_assets_areasong.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Shop Up
  }

  var Activated: bool = false
  var Spoken: bool = false
  var Is Act 3: bool = false
  var Convo Key: string = ""
  var CamLock: gameObject = Thief NPC Shop/Shop Camlock
  var Shop Object: gameObject = <null>
  var Audio Loop Voice: gameObject = Thief NPC Shop/Audio Loop Voice
}
