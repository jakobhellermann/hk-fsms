# arena impact
# 3 location(s):
#   Cog_Dancers_boss / Dancer Control/Death Chunks 1/Chunks/Chunk (3)
#   Cog_Dancers_boss / Dancer Control/Death Chunks 1/Chunks/Chunk (2)
#   Cog_Dancers_boss / Dancer Control/Death Chunks 1/Chunks/Chunk

// uses template: corpse_chunk_arena_impact
fsm arena impact {
  start Idle

  state Idle {
    on ARENA IMPACT → Impact
  }

  state Impact {
    RandomFloat(min=2, max=5, storeResult=var "Speed")
    RandomFloat(min=70, max=110, storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Idle
  }

  var Speed: float = 0
  var Angle: float = 0
}
