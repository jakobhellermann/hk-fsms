# Custom Corpse
# 1 location(s):
#   enemycorpses_assets_areacoralareamemory.bundle / Corpse Coral Goomba M

fsm Custom Corpse {
  start Wait for land

  state Wait for land {
    on LAND → Spawn
  }

  state Spawn {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetTransformParent(gameObject=var "Break Particles", parent=<null>, worldPositionStays=true)
    ActivateGameObject(gameObject=var "Break Particles", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPreInstantiatedGameObject(target=Self, storeGameObject=var "Inner")  // disabled
    GetPosition2D(gameObject=Self, vector=var "Self Pos 2D", x=(unset), y=(unset), space=0, everyFrame=false)  // disabled
    SetPosition2D(GameObject=var "Inner", Vector=var "Self Pos 2D", X=0, Y=0, Space=0, EveryFrame=false)  // disabled
    MatchScaleSign(Target=var "Inner", MatchTo=var "Self", EveryFrame=false)  // disabled
    SendEventByNameV2(eventTarget=GameObject(var "Inner"), sendEvent="FLING OUT", delay=0, everyFrame=false)  // disabled
  }

  var Self Pos 2D: vector2 = (0, 0)
  var Break Particles: gameObject = <null>
  var Inner: gameObject = <null>
  var Self: gameObject = <null>
}
