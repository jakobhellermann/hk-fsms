# Tink Bounce
# 1 location(s):
#   localpoolprefabs_assets_areacoral.bundle / Coral Crust Tree Pearl

fsm Tink Bounce {
  start Init
  on BREAK → Broken  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetRigidbody2DConstraints(gameObject=Self, freezePositionX=(unset), freezePositionY=(unset), freezeRotation=false)
    SetIntValue(intVariable=var "Bounces", intValue=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on TINK LEFT → Left
    on TINK RIGHT → Right
    on TINK UP → Up
    on TINK DOWN → Down
    on BOUNCED → Bounced
  }

  state Left {
    RandomFloat(min=130, max=140, storeResult=var "Angle")
    RandomFloat(min=17, max=22, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Effect
  }

  state Right {
    RandomFloat(min=40, max=50, storeResult=var "Angle")
    RandomFloat(min=17, max=22, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Effect
  }

  state Up {
    RandomFloat(min=85, max=95, storeResult=var "Angle")
    RandomFloat(min=23, max=27, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Effect
  }

  state Down {
    RandomFloat(min=250, max=290, storeResult=var "Angle")
    RandomFloat(min=17, max=22, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Effect
  }

  state Broken {
    SetRigidbody2DConstraints(gameObject=Self, freezePositionX=(unset), freezePositionY=(unset), freezeRotation=true)
  }

  state Bounced {
    IntAddV2(intVariable=var "Bounces", add=1, everyFrame=false)
    IntCompare(integer1=var "Bounces", integer2=3, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="BreakWhenNotMoving", methodName="Break", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Effect {
    PlayAudioEventRandom(audioClips=[choral_or_other_break_1 (sfxstatic_assets_areasong.bundle), choral_or_other_break_2 (sfxstatic_assets_areasong.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Idle
  }

  var Angle: float = 0
  var Speed: float = 0
  var Bounces: int = 0
  var Self: gameObject = <null>
}
