fsm FSM {
  start Inert

  state Inert {
    on ACTIVE → Wait
  }

  state Wait {
    Wait(time=5, finishEvent=(none), realTime=false)
    on FINISHED → Exit?
  }

  state Exit? {
    CheckTrackTriggerCount(target=Self, count=0, test=LessThanOrEqual, everyFrame=true, successEvent=→"EXIT")
    on EXIT → Close
  }

  state Close {
    CallMethodProper(gameObject=var "Gate L", behaviour="Gate", methodName="Close", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Inert
  }

  var Gate L: gameObject = Vent Pass Through/Variants/vent_pass_through_door_blocker
}
