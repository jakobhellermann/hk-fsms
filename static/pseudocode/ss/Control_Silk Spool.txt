fsm Control {
  start Pause
  on DISABLE → State 1  // from any state

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Get Cloud", storeResult=var "Get Cloud")
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite 1")
    FindChild(gameObject=Self, childName="Get Anim", storeResult=var "Get Anim")
    FindChild(gameObject=Self, childName="Idle Particles", storeResult=var "Idle Particles")
    FindNamedChild(gameObject=Self, storeResult=var "Idle Pulse")
    on ACTIVATED → Activated
    on FINISHED → PD Bool
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=(unset), sendEvent=→"GET", storeCollider=var "")
    on GET → Pickup Disabled?
  }

  state Get {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Warmth Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Idle Pulse", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStop(gameObject=Self, fadeTime=0)
    SetPlayerDataBool(boolName="isInvincible", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=heartpiece_collect (heartpieceupgrade_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Heart Piece")
    SetGravity2dScale(gameObject=var "Hero", gravityScale=0)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="GET", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Idle Particles")
    ActivateGameObject(gameObject=var "Sprite 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Get Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=spool_shard_new_1_touch (sfxstatic_assets_currency.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → UI
  }

  state Activated {
    DestroySelf(detachChildren=false)
  }

  state UI {
    CreateObject(gameObject=Silk Spool UI (silkspoolupgrade_assets_all.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "UI")
    on SILK SPOOL SAVE → Save
  }

  state Pause {
    ActivateGameObject(gameObject=var "Warmth Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  sequence state Pickup Disabled? {
    NextFrameEvent(sendEvent=(none))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="GetCState", parameters=["recoiling"], storeResult=var "IsRecoiling", EveryFrame=false)
    BoolTest(boolVariable=var "IsRecoiling", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Idle
    on FALSE → Set PD
    on FINISHED → Set PD
  }

  state Fling Up? {
    BoolTest(boolVariable=var "Fling Up", isTrue=→"FLING", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on FLING → Fling 
  }

  state Fling  {
    GetPosition(gameObject=Self, vector=(unset), x=var "Fling X", y=var "Fling Y", z=var "Fling Z", space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Fling X", minValue=var "X Min", maxValue=var "X Max", everyFrame=false)
    FloatAdd(floatVariable=var "Fling Y", add=2, everyFrame=false, perSecond=false)
    SetVector3XYZ(vector3Variable=var "Fling Pos", vector3Value=(unset), x=var "Fling X", y=var "Fling Y", z=var "Fling Z", everyFrame=false)
    AnimatePositionTo(gameObject=Self, toValue=var "Fling Pos", localSpace=false, time=1, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Fling Up Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Idle
  }

  state PD Bool {
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName=var "PD Bool", isTrue=→"ACTIVATED", isFalse=(none))
    on FINISHED → Fling Up?
    on ACTIVATED → Activated
  }

  state Set PD {
    AddHeroInputBlocker(Blocker=Self)
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "PD Bool", value=true)
    on FINISHED → Shift Up?
  }

  state End {
    ActivateGameObject(gameObject=var "Warmth Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RemoveHeroInputBlocker(Blocker=Self)
  }

  state Shift Up? {
    RayCast2dV2(fromGameObject=var "Hero", fromPosition=(unset), direction=(0, -1), space=1, distance=2.5, minDepth=(unset), maxDepth=(unset), hitEvent=→"SHIFT", noHitEvent=→"FINISHED", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on SHIFT → Get And Shift Up
    on FINISHED → Get
  }

  state Get And Shift Up {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Warmth Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Idle Pulse", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStop(gameObject=Self, fadeTime=0)
    SetPlayerDataBool(boolName="isInvincible", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=heartpiece_collect (heartpieceupgrade_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Heart Piece")
    SetGravity2dScale(gameObject=var "Hero", gravityScale=0)
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="GET", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Idle Particles")
    ActivateGameObject(gameObject=var "Sprite 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Get Anim", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="AverageShake", delay=0, everyFrame=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=spool_shard_new_1_touch (sfxstatic_assets_currency.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatePositionBy(gameObject=var "Hero", shiftBy=(0, 0.75, 0), time=0.2, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → UI
  }

  state State 1 {
    RemoveHeroInputBlocker(Blocker=Self)
  }

  state Save {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on SILK SPOOL UI END → End
  }

  var X Min: float = 0
  var X Max: float = 9999
  var Fling Up: bool = false
  var PD Bool: string = ""

  var Fling X: float = 0
  var Fling Y: float = 0
  var Fling Z: float = 0
  var Activated: bool = false
  var IsRecoiling: bool = false
  var Was On Ground: bool = false
  var Fling Pos: vector3 = (0, 0, 0)
  var Get Anim: gameObject = <null>
  var Get Cloud: gameObject = <null>
  var Idle Particles: gameObject = <null>
  var Self: gameObject = <null>
  var Sprite 1: gameObject = <null>
  var UI: gameObject = <null>
  var Fling Up Effects: gameObject = Silk Spool/Fling Up Effects
  var Idle Pulse: gameObject = <null>
  var Warmth Region: gameObject = Silk Spool/Warmth Region
}
