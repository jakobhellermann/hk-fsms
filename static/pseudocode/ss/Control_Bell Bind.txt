# Control
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/Tool Effects/Bell Bind

// uses template: bell_bind_tool
fsm Control {
  start Wait
  on LEAVING SCENE → Ended  // from any state
  on HAZARD RESPAWN RESET → Ended  // from any state

  state Wait {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Appear
    on HERO DAMAGED → Disable
    on DISAPPEAR → Disable
  }

  state Appear {
    SetBoolValue(boolVariable=var "Is Shielding", boolValue=true, everyFrame=false)
    SetStaticVariable(variableName="IsBindBellActive", setValue=true, sceneTransitionsLimit=(unset))
    PlayAudioEvent(audioClip=hornet_warding_bell_bind (sfxstatic_assets_shared.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    SpawnObjectFromGlobalPool(gameObject=bind_bell_appear (localpoolprefabs_assets_shared.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Spawned Bell")
    GetPosition2D(gameObject=var "Hero", vector=var "Hero 2D Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Spawned Bell", Vector=var "Hero 2D Pos", X=0, Y=0, Space=0, EveryFrame=false)
    on FINISHED → Idle
    on HIT → Hit
    on DISAPPEAR → Disable
  }

  state Idle {
    on DISAPPEAR → Disappear 2
    on HIT → Hit
  }

  state Hit {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Burst
  }

  state Disappear {
    SetAnimatorTrigger(gameObject=var "Spawned Bell", trigger="Disappear")
    ListenForAnimationEvent(Target=var "Spawned Bell", Response=→"FINISHED")
    GameObjectIsActiveEveryFrame(Target=var "Spawned Bell", ActiveSpace=1, TrueEvent=(none), FalseEvent=→"FINISHED", StoreValue=(unset), EveryFrame=true)
    on FINISHED → Ended
  }

  state Disable {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Burst {
    PlayAudioEvent(audioClip=hornet_warding_bell_bind_break (sfxstatic_assets_shared.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=bind_bell_burst (localpoolprefabs_assets_shared.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Spawned Burst")
    GetPosition2D(gameObject=var "Hero", vector=var "Hero 2D Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Spawned Burst", Vector=var "Hero 2D Pos", X=0, Y=0, Space=0, EveryFrame=false)
    on FINISHED → Disappear
  }

  state Ended {
    SetBoolValue(boolVariable=var "Is Shielding", boolValue=false, everyFrame=false)
    SetStaticVariable(variableName="IsBindBellActive", setValue=false, sceneTransitionsLimit=(unset))
    on FINISHED → Disable
  }

  state Disappear 2 {
    SetAnimatorTrigger(gameObject=var "Spawned Bell", trigger="Disappear")
    ListenForAnimationEvent(Target=var "Spawned Bell", Response=→"FINISHED")
    GameObjectIsActiveEveryFrame(Target=var "Spawned Bell", ActiveSpace=1, TrueEvent=(none), FalseEvent=→"FINISHED", StoreValue=(unset), EveryFrame=true)
    on FINISHED → Ended
    on HIT → Hit 2
  }

  state Wait For Disappear {
    ListenForAnimationEvent(Target=var "Spawned Bell", Response=→"FINISHED")
    GameObjectIsActiveEveryFrame(Target=var "Spawned Bell", ActiveSpace=1, TrueEvent=(none), FalseEvent=→"FINISHED", StoreValue=(unset), EveryFrame=true)
    on FINISHED → Ended
  }

  state Hit 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Burst 2
  }

  state Burst 2 {
    PlayAudioEvent(audioClip=hornet_warding_bell_bind_break (sfxstatic_assets_shared.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=bind_bell_burst (localpoolprefabs_assets_shared.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Spawned Burst")
    GetPosition2D(gameObject=var "Hero", vector=var "Hero 2D Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Spawned Burst", Vector=var "Hero 2D Pos", X=0, Y=0, Space=0, EveryFrame=false)
    on FINISHED → Wait For Disappear
  }

  var Is Shielding: bool = false
  var Hero 2D Pos: vector2 = (0, 0)
  var Spawned Bell: gameObject = <null>
  var Spawned Burst: gameObject = <null>
  var Audio Player: gameObject = <null>
}
