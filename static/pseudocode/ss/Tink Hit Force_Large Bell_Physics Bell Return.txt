fsm Tink Hit Force {
  start Inert

  state Inert {
    on TINK LEFT → Add Force Left
    on TINK RIGHT → Add Force Right
    on TINK UP → Random
    on TINK DOWN → Random
  }

  state Add Force Left {
    AddTorque2d(gameObject=Self, forceMode=1, torque=-0.5, everyFrame=false)
    on FINISHED → Audio
  }

  state Add Force Right {
    AddTorque2d(gameObject=Self, forceMode=1, torque=0.5, everyFrame=false)
    on FINISHED → Audio
  }

  state Random {
    SendRandomEvent(events=[→"TINK LEFT", →"TINK RIGHT"], weights=[1, 1], delay=0)
    on TINK LEFT → Add Force Left
    on TINK RIGHT → Add Force Right
  }

  state Audio {
    PlayAudioEvent(audioClip=bellshrine_sword_hit (sfxstatic_assets_areabellshrine.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=<null>)
    on FINISHED → Inert
  }
}
