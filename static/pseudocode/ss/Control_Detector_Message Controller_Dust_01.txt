# Control
# 5 location(s):
#   Dust_01 / Message Controller/Detector
#   Dust_02 / Message Controller/Detector
#   Dust_11 / Message Controller (2)/Detector

#   Dust_Barb / Message Controller/Detector

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Above Set")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Below Set")
    on FINISHED → Above
  }

  state Above {
    ActivateGameObject(gameObject=var "Above Set", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Below Set", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Below
  }

  state Below {
    ActivateGameObject(gameObject=var "Above Set", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Below Set", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Above
  }

  var Parent: gameObject = <null>
  var Above Set: gameObject = <null>
  var Below Set: gameObject = <null>
}
