# Control
# 1 location(s):
#   Dock_03 / Dock Bomber Ambush

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Dock Bomber Amb1")
    FindNamedChild(gameObject=Self, storeResult=var "Dock Flyer Amb")
    FindNamedChild(gameObject=Self, storeResult=var "Explode Roof")
    FindNamedChild(gameObject=Self, storeResult=var "Explode Effects")
    ActivateGameObject(gameObject=var "Dock Bomber Amb1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dock Flyer Amb", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Explode
  }

  state Explode {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Recycle M (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Explode Roof", position=(unset), rotation=(unset), storeObject=var "")
    ActivateGameObject(gameObject=var "Explode Roof", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Explode Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="docksBomberAmbush", value=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Explode Effects", audioClip=explode_wall_explode (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Bomber Entry
  }

  state Bomber Entry {
    SetFsmBool(gameObject=var "Dock Bomber Amb1", fsmName="Behaviour", variableName="Bomb Ambush", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dock Flyer Amb", fsmName="Behaviour", variableName="Bomb Ambush", setValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Dock Bomber Amb1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dock Flyer Amb", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait {
    ActivateGameObject(gameObject=var "Dock Flyer Amb", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dock Bomber Amb1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Dock Bomber Amb1: gameObject = <null>
  var Explode Roof: gameObject = <null>
  var Explode Effects: gameObject = <null>
  var Dock Bomber Amb2: gameObject = <null>
  var Dock Flyer Amb: gameObject = <null>
}
