# Record Active Tubes
# 1 location(s):
#   Weave_10 / Crest Upgrade Shrine/crest_shrine_break_tube parent

fsm Record Active Tubes {
  start Idle

  state Idle {
    on TUBES ACTIVE → Increment
    on TUBES DESTROYED → Decrement
  }

  state Increment {
    IntAdd(intVariable=var "Tubes Active", add=1, everyFrame=false)
    SendEventToRegister(eventName="TUBES CHANGE")
    on FINISHED → Idle
  }

  state Decrement {
    IntAdd(intVariable=var "Tubes Active", add=-1, everyFrame=false)
    SendEventToRegister(eventName="TUBES CHANGE")
    on FINISHED → Idle
  }

  var Tubes Active: int = 0
}
