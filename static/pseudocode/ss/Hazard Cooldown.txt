fsm Hazard Cooldown {
  start State 1

  state State 1 {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Cooldown", setValue=false, everyFrame=false)
    on KNIGHT SPIKE HIT → State 2
  }

  state State 2 {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Cooldown", setValue=true, everyFrame=false)
    Wait(time=3, finishEvent=(none), realTime=false)
    on FINISHED → State 1
  }
}
