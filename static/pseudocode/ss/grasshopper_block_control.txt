// component disabled: this FSM does not run until something enables it
// uses template: grasshopper_block_control
fsm grasshopper_block_control {
  start Idle

  state Idle {
    on TOOK DAMAGE → Set Block
    on BLOCKED HIT → Unset Block
  }

  state Set Block {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Do Block", setValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  state Unset Block {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Do Block", setValue=false, everyFrame=false)
    on FINISHED → Idle
  }
}
