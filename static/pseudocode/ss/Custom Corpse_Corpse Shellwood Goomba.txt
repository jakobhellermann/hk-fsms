# Custom Corpse
# 1 location(s):
#   enemycorpses_assets_areashellwood.bundle / Corpse Shellwood Goomba

fsm Custom Corpse {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Spawn Effects
  }

  state Wait for land {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=true, otherLayerNumber=8, ignoreTriggers=true)
    Wait(time=0.5, finishEvent=→"LAND", realTime=false)
    on LAND → Scene Spawn?
  }

  state Spawn {
    GetPreInstantiatedGameObject(target=Self, storeGameObject=var "Gnat")
    GetPosition2D(gameObject=Self, vector=var "Self Pos 2D", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Gnat", Vector=var "Self Pos 2D", X=0, Y=0, Space=0, EveryFrame=false)
    MatchScaleSign(Target=var "Gnat", MatchTo=var "Self", EveryFrame=false)
    SendEventByNameV2(eventTarget=GameObject(var "Gnat"), sendEvent="FLING OUT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Spawn Effects {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    FindChild(gameObject=Self, childName="Break Particles", storeResult=var "Break Particles")
    ActivateGameObject(gameObject=var "Break Particles", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait for land
  }

  state Scene Spawn? {
    PlayAudioEvent(audioClip=enemy_bone_crack_1 (sfxstatic_assets_areabone.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetTransformParent(gameObject=var "Break Particles", parent=<null>, worldPositionStays=true)
    ActivateGameObject(gameObject=var "Break Particles", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FindClosest(gameObject=Self, withTag="Boss", ignoreOwner=false, mustBeVisible=false, storeObject=var "Scene Spawn", storeDistance=(unset), everyFrame=false)
    GameObjectIsNull(gameObject=var "Scene Spawn", isNull=→"FINISHED", isNotNull=→"SCENE SPAWN", storeResult=(unset), everyFrame=false)
    on FINISHED → Spawn
    on SCENE SPAWN → Scene Spawn
  }

  state Scene Spawn {
    PlayAudioEvent(audioClip=shellwood_goomba_flyer_shell_break (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1.2, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetPosition2D(gameObject=Self, vector=var "Self Pos 2D", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Scene Spawn", Vector=var "Self Pos 2D", X=0, Y=0, Space=0, EveryFrame=false)
    MatchScaleSign(Target=var "Scene Spawn", MatchTo=var "Self", EveryFrame=false)
    SendEventByNameV2(eventTarget=GameObject(var "Scene Spawn"), sendEvent="FLING OUT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Self Pos 2D: vector2 = (0, 0)
  var Break Particles: gameObject = <null>
  var Gnat: gameObject = <null>
  var Self: gameObject = <null>
  var Scene Spawn: gameObject = <null>
}
