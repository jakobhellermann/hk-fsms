# FSM
# 4 location(s):
#   Song_02 / Black Thread States/Normal World/Citadel Woken Scene/March Group 2/Song Pilgrim 01 (1)
#   Song_02 / Black Thread States/Normal World/Citadel Woken Scene/March Group/Song Pilgrim 01
#   Song_27 / Song Knight Control/Song Knight Absent/March Groups/March Group L/Song Pilgrim 01
#   Song_27 / Song Knight Control/Song Knight Absent/March Groups/March Group R/Song Pilgrim 01 (1)

// uses template: call_handmaiden
fsm FSM {
  start Pause
  on HANDMAIDEN CALLED → Inactive  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Call Check
  }

  state Wait {
    AddEventRegister(target=Self, eventName="HANDMAIDEN CALLED")
    CreateObject(gameObject=Song Handmaiden SpriteCache (localpoolprefabs_assets_areahangareasong.bundle), spawnPoint=<null>, position=(-20, 0, 0), rotation=(unset), storeObject=var "")  // disabled
    CreateObject(gameObject=Song Handmaiden (localpoolprefabs_assets_areahangareasong.bundle), spawnPoint=<null>, position=(-100, 0, 0), rotation=(unset), storeObject=var "Handmaiden")
    on HANDMAIDEN CALL → Call Pause
  }

  state Call Check {
    PlayerDataBoolTest(boolName="marionettesBurned", isTrue=→"INACTIVE", isFalse=(none))
    GetPlayerDataBool(boolName="marionettesMet", storeValue=var "Marionettes Met")
    BoolTestMulti(boolVariables=[var "Marionettes Met", var "Force If Marionettes Not Met"], boolStates=[false, true], trueEvent=→"ACTIVE", falseEvent=(none), storeResult=(unset), everyFrame=false)
    RandomInt(min=1, max=100, storeResult=var "Chance", inclusiveMax=true, noRepeat=false)
    IntCompare(integer1=var "Chance", integer2=var "Call Percent", equal=(none), lessThan=(none), greaterThan=→"INACTIVE", everyFrame=false)
    StringCompare(stringVariable=var "PD Bool Required", compareTo="", equalEvent=→"ACTIVE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName=var "PD Bool Required", isTrue=(none), isFalse=→"INACTIVE")
    SendEvent(eventTarget=Self, sendEvent=→"ACTIVE", delay=0, everyFrame=false)
    on ACTIVE → Wait
    on INACTIVE → Inactive
  }

  state Inactive {
  }

  state Call Pause {
    WaitRandom(timeMin=3, timeMax=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Call
  }

  state Call {
    CreateObject(gameObject=Song Handmaiden (localpoolprefabs_assets_areahangareasong.bundle), spawnPoint=<null>, position=(-20, 0, 0), rotation=(unset), storeObject=var "Handmaiden")  // disabled
    SetFsmBool(gameObject=var "Handmaiden", fsmName="Control", variableName="CalledByEnemy", setValue=true, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Handmaiden"), sendEvent="CALL", delay=0, everyFrame=false)
    SendEventToRegisterV2(EventName="HANDMAIDEN CALLED", ExcludeTarget=Self)
  }

  var Call Percent: int = 25
  var Force If Marionettes Not Met: bool = false
  var PD Bool Required: string = "hasHarpoonDash"

  var Chance: int = 25
  var Marionettes Met: bool = false
  var Handmaiden: gameObject = <null>
}
