# FSM
# 1 location(s):
#   Arborium_01 / one way moss vine/Moss Vine Cluster

fsm FSM {
  start State 1

  state State 1 {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → State 2
  }

  state State 2 {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=→"HERO BELOW", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on HERO BELOW → Deactivate
  }

  state Deactivate {
    EnableFSM(gameObject=Self, fsmName="Control", enable=false, resetOnExit=false)
  }
}
