  state State 1 {
    on GARMOND BATTLE → State 2
  }

  state State 2 {
    SetDamageEnemyAmount(target=Self, damageDealt=0)
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"ENTER", storeCollider=var "Enemy")

    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventByNameUpwards(Target=var "Enemy", EventName="SPIKE ENTER")

  }

  var Enemy: gameObject = <null>
