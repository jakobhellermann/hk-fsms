fsm BG Control {
  start Wait

  state Wait {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)

    FindNamedChild(gameObject=var "Parent", storeResult=var "Top Mask")
    ActivateGameObject(gameObject=var "Impact Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Battle Complete?
  }

  state Battle Complete? {
    GetPersistentBool(target=var "Battle Scene", storeValue=var "Battle Complete")
    BoolTest(boolVariable=var "Battle Complete", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Opened
    on FALSE → Wait for open
  }

  state Open {
    AnimatorPlay(gameObject=Battle Scene/Gates/drop bridge/greymoor_flip_bridge_BG (1)/controls/Gears, stateName="Open", layer=0, normalizedTime=0, everyFrame=false)

    on FINISHED → Impact
  }

  state Wait for open {
    GetPosition(gameObject=var "Closed Point", vector=var "Closed Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=Self, vector=var "Closed Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on BG OPEN → Open
  }

  state Opened {
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Top Mask", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Impact {
    AudioStop(gameObject=Self, fadeTime=0)
    PlayAudioEvent(audioClip=shellwood_lift_arrive (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    ActivateGameObject(gameObject=var "Impact Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Opened
  }

  var Battle Scene: gameObject = Battle Scene

  var Battle Complete: bool = false
  var Start Pos: vector3 = (0, 0, 0)
  var Closed Pos: vector3 = (0, 0, 0)
  var Closed Point: gameObject = Battle Scene/Gates/drop bridge top point

  var Parent: gameObject = <null>
}
