# BG Control
# 1 location(s):
#   Ant_19 / Boss Control/Boss Scene/Gates/Battle Gate Ant (1)

fsm BG Control {
  start Pause
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state
  on ACTIVATE → Destroy  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close Pause
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor Distant V2 (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=hunter_trap_cage_come_up (sfxstatic_assets_areaant.bundle), pitchMin=0.75, pitchMax=1.25, volume=0.8, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ant Gate Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=var "Wall Collider", active=true, resetOnExit=false)
    on FINISHED → Close 2
  }

  state Close 2 {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Slam", emit=0, resetIfPlaying=false)
    on BG OPEN → Open
    on OPEN → Open
  }

  state Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ant Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=var "Wall Collider", active=false, resetOnExit=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor Distant V2 (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=bone_trapdoor_move_open (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1.4, volume=0.85, delay=0, storePlayer=<null>)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Opened Effects
  }

  state Quick Close {
    SetBoolValue(boolVariable=var "Start Closed", boolValue=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Ant Gate Closed")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=var "Wall Collider", active=true, resetOnExit=false)
    on BG OPEN → Open
    on OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Wall Collider", active=false, resetOnExit=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close Pause {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Close 1
  }

  state Pause {
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Opened
  }

  state Opened Effects {
    PlayParticleEmitter(gameObject=var "Pt Open", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Rocks", emit=0, resetIfPlaying=false)
    on FINISHED → Opened
  }

  var Start Closed: bool = true

  var Dust: gameObject = <null>
  var Pt Open: gameObject = Boss Control/Boss Scene/Gates/Battle Gate Ant (1)/Pt Open
  var Pt Rocks: gameObject = Boss Control/Boss Scene/Gates/Battle Gate Ant (1)/Pt Rocks
  var Pt Slam: gameObject = Boss Control/Boss Scene/Gates/Battle Gate Ant (1)/Pt Slam
  var Raise Dust: gameObject = Boss Control/Boss Scene/Gates/Battle Gate Ant (1)/Plat Dust
  var Self: gameObject = <null>
  var Slam Effect: gameObject = Boss Control/Boss Scene/Gates/Battle Gate Ant (1)/Close Effect
  var Wall Collider: gameObject = Boss Control/Boss Scene/Gates/Battle Gate Ant (1)/Wall Collider
}
