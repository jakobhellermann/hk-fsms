# Control
# 1 location(s):
#   Ant_08 / Battle Scene/End Scene

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Collectable Item Pickup")
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Ant Regions")
    on BATTLE END → Activate Item
  }

  state Activate Item {
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Ant Regions", delivery=SendMessage, options=1, functionCall=StopEmitting())
  }

  var Collectable Item Pickup: gameObject = <null>
  var Ant Regions: gameObject = <null>
  var Battle Scene: gameObject = <null>
}
