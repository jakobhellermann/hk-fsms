fsm Control {
  start Rotation
  on STOP → Stop  // from any state
  on ANTIC → Antic  // from any state

  state Init {
    SendRandomEvent(events=[→"DOWN", →"UP"], weights=[1, 1], delay=0)
    on DOWN → Down
    on UP → Up
  }

  state Down {
    RandomFloat(min=0.75, max=1, storeResult=var "Time")
    AnimatePositionTo(gameObject=Self, toValue=(0, -0.02, 0), localSpace=true, time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
  }

  state Up {
    RandomFloat(min=0.75, max=1, storeResult=var "Time")
    AnimatePositionTo(gameObject=Self, toValue=(0, 0.02, 0), localSpace=true, time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  state Stop {
    AnimatePositionTo(gameObject=Self, toValue=(0, 0, 0), localSpace=true, time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on BOB → Init
  }

  state Antic {
    AnimatePositionTo(gameObject=Self, toValue=(0, 0.3, 0), localSpace=true, time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on BOB → Init
  }

  state Rotation {
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)  // disabled
    on FINISHED → Init
  }

  var Time: float = 0
}
