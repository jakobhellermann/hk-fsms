# Loop Beat Anim
# 1 location(s):
#   Cradle_03 / Boss Scene/silk_cocoon_core

fsm Loop Beat Anim {
  start Init
  on NEXT_SCENE → Stop 2  // from any state
  on STOP_SILK_HEARTBEAT → Stop 2  // from any state

  state Init {
    on FINISHED → Start
  }

  state Start {
    AnimatorPlayStateWait(target=Self, stateName="Silk_Cocoon_Intro_Beat", finishEvent=→"LOOP")
    on LOOP → Loop
    on STOP → Stop
  }

  state Loop {
    SendEventByName(eventTarget=GameObject(var "Audio Heartbeat"), sendEvent="PLAY", delay=0, everyFrame=false)
    AnimatorPlay(gameObject=Self, stateName="Silk_Cocoon_Intro_Beat", layer=0, normalizedTime=0, everyFrame=false)
    AnimatorPlayStateWait(target=Self, stateName="Silk_Cocoon_Intro_Beat", finishEvent=→"LOOP")
    on LOOP → Loop
    on STOP → Stop
  }

  state Stop {
  }

  state Stop 2 {
    EnableBehaviour(gameObject=Self, behaviour="Animator", component=<null>, enable=false, resetOnExit=false)
  }

  var Audio Heartbeat: gameObject = Boss Scene/silk_cocoon_core/Audio Heartbeat
}
