# BG Control
# 1 location(s):
#   Room_CrowCourt / Battle Scene/Gates/greymoor_flip_bridge_BG

fsm BG Control {
  start Wait

  state Wait {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Inverted?
  }

  state Battle Complete? {
    GetPersistentBool(target=var "Battle Scene", storeValue=var "Battle Complete")
    BoolTest(boolVariable=var "Battle Complete", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Open
    on FALSE → Wait for open
  }

  state Open {
    CallMethodProper(gameObject=Self, behaviour="Trapdoor", methodName="OpenDoorCustom", parameters=[-1], storeResult=(unset var), EveryFrame=false)
  }

  state Wait for open {
    on BG OPEN → Open
  }

  state Inverted? {
    BoolTest(boolVariable=var "Is Inverted", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Battle Complete?
    on TRUE → Battle Complete? 2
  }

  state Battle Complete? 2 {
    GetPersistentBool(target=var "Battle Scene", storeValue=var "Battle Complete")
    BoolTest(boolVariable=var "Battle Complete", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on TRUE → Close
    on FALSE → Wait for close
  }

  state Close {
    CallMethodProper(gameObject=Self, behaviour="Trapdoor", methodName="CloseDoorCustom", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  state Wait for close {
    CallMethodProper(gameObject=Self, behaviour="Trapdoor", methodName="OpenDoorCustom", parameters=[-1], storeResult=(unset var), EveryFrame=false)
    on BG OPEN → Close
  }

  var Is Inverted: bool = false
  var Battle Scene: gameObject = Battle Scene

  var Battle Complete: bool = false
}
