# Fake Death
# 1 location(s):
#   Slab_10b / Boss Scene/First Weaver

fsm Fake Death {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on TOOK DAMAGE → Check
  }

  state Check {
    CompareHP(enemy=var "Self", integer2=1, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CompareHP(enemy=var "Self", integer2=var "P2 HP", equal=(none), lessThan=→"DEATH", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on DEATH → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FAKE DEATH", delay=0, everyFrame=false)
  }

  var P2 HP: int = 0
  var Self: gameObject = <null>
}
