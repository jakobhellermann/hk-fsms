# Break Threads
# 1 location(s):
#   Hang_14 / Handmaiden Scene/break_thread_core

fsm Break Threads {
  start Init
  on THREAD BELL HIT → Hit  // from any state

  state Inert {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.75, MaxReactDelay=0.75, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on HIT → Hit
    on SING → Needolin Twitch
  }

  state Choose State {
    SendEventByName(eventTarget=Self, sendEvent=var "State", delay=0, everyFrame=false)
    on 1 → Hit Fx
    on 2 → Hit Fx
    on 3 → Hit Fx
    on FINISHED → Snap Start
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Hit 1", storeResult=var "Pt Hit 1")
    FindChild(gameObject=Self, childName="Pt Hit 2", storeResult=var "Pt Hit 2")
    FindChild(gameObject=Self, childName="Pt Hit 3", storeResult=var "Pt Hit 3")
    ActivateGameObject(gameObject=var "Deactivate On Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Activate On Break", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Broken?
  }

  state Hit {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="CanDamage Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="FreezeMoment", parameters=[1], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddSilk", parameters=[1, true], storeResult=(unset var), EveryFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Silk_Get_Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Slash Impact")
    SetScale(gameObject=var "Slash Impact", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Slash Impact", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    MatchScaleSign(Target=var "Slash Impact", MatchTo=var "Hero", EveryFrame=false)
    IntAdd(intVariable=var "State", add=1, everyFrame=false)
    SendMessage(gameObject=Handmaiden Scene/break_thread_core/silk_threads, delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    on FINISHED → Choose State
    on CANCEL → Inert
  }

  state Snap Large Thread {
    PlayAudioEventRandom(audioClips=[silk_prop_slash_1 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_2 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_3 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_4 (sfxstatic_assets_areabellshrine.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=Handmaiden Scene/break_thread_core/silk_threads, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Event Target"), sendEvent="BREAK", delay=0, everyFrame=false)
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataVariable(VariableName=var "PD Bool", SetValue=true)
    on FINISHED → Broken
  }

  state Hit Fx {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayAudioEventRandom(audioClips=[silk_prop_slash_1 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_2 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_3 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_4 (sfxstatic_assets_areabellshrine.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=wraith_death (sfxstatic_assets_areadustmaze.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmission(gameObject=var "Pt Hit 1", emission=true, resetOnExit=true)  // disabled
    SetParticleEmission(gameObject=var "Pt Hit 2", emission=true, resetOnExit=true)  // disabled
    SetParticleEmission(gameObject=var "Pt Hit 3", emission=true, resetOnExit=true)  // disabled
    PlayParticleEmitter(gameObject=var "Pt Hit 1", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Hit 2", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Hit 3", emit=0, resetIfPlaying=false)
    on FINISHED → Inert
  }

  state Snap Start {
    ScreenFlash(flashColour=(1, 1, 1, 1))
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="FreezeMoment", parameters=[1], storeResult=(unset var), EveryFrame=false)
    SetTransformParent(gameObject=Handmaiden Scene/break_thread_core/silk_threads/Silk_thread_snap_effect, parent=<null>, worldPositionStays=true)
    ActivateGameObject(gameObject=Handmaiden Scene/break_thread_core/silk_threads/Silk_thread_snap_effect, activate=true, recursive=true, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=silk_prop_kill (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=wraith_death (sfxstatic_assets_areadustmaze.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CallMethodProper(gameObject=Handmaiden Scene/break_thread_core/Break Particles, behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    Comment(comment="Popup to be shown after sequence completed", Record=Song Handmaiden (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle), ShowPopup=false)
    CompleteJournalRecord(Record=Song Handmaiden (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle), ShowPopup=false)
    on FINISHED → Snap Large Thread
  }

  state Broken? {
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FALSE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataVariableTest(VariableName=var "PD Bool", ExpectedValue=true, IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on TRUE → Broken
    on FALSE → Inert
  }

  state Broken {
    SendEventByName(eventTarget=GameObject(var "Event Target"), sendEvent="BROKEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Deactivate On Break", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Activate On Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Needolin Twitch {
    SendMessage(gameObject=Handmaiden Scene/break_thread_core/silk_threads, delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    PlayAudioEventRandom(audioClips=[hanging_by_thread_01 (sfxstatic_assets_silkshared.bundle), hanging_by_thread_02 (sfxstatic_assets_silkshared.bundle), hanging_by_thread_03 (sfxstatic_assets_silkshared.bundle)], pitchMin=1, pitchMax=1.3, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Inert
  }

  var PD Bool: string = "marionettesBurned"
  var Deactivate On Break: gameObject = Handmaiden Scene/web_mist_set
  var Activate On Break: gameObject = <null>
  var Event Target: gameObject = Handmaiden Scene

  var Hero Y: float = 0
  var State: int = 0
  var Pt Hit 1: gameObject = <null>
  var Pt Hit 2: gameObject = <null>
  var Pt Hit 3: gameObject = <null>
  var Self: gameObject = <null>
  var Slash Impact: gameObject = <null>
}
