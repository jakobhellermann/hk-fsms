# Control
# 1 location(s):
#   localpoolprefabs_assets_areaclover.bundle / Aspid Hatchling

fsm Control {
  start Pause
  on BREAK → Break  // from any state
  on ZERO HP → Break  // from any state
  on CLEAR HATCHLING → Clear  // from any state

  state Init {
    RandomFloat(min=15, max=17, storeResult=var "Lifetime")
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Start Anim")
    StringCompare(stringVariable=var "Start Anim", compareTo="Fly In", equalEvent=→"FLY IN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Start Alert", isTrue=→"CHASE", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Spawn Up", isTrue=→"SPAWN UP", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Spawn Down", isTrue=→"SPAWN DOWN", isFalse=(none), everyFrame=false)
    GameObjectIsNull(gameObject=var "Follow Target", isNull=→"FINISHED", isNotNull=→"FOLLOW", storeResult=(unset), everyFrame=false)
    on FINISHED → Idle
    on CHASE → Chase
    on FOLLOW → Follow
    on SPAWN UP → Spawn Up
    on SPAWN DOWN → Spawn Down
    on FLY IN → Fly In Ready
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    IdleBuzzV2(gameObject=Self, waitMin=0.5, waitMax=1, speedMax=1, accelerationMax=10, roamingRangeX=1, roamingRangeY=1, manualStartPos=(unset))
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    SetScale(gameObject=var "Alert Range", vector=(unset), x=2, y=2, z=(unset), everyFrame=false, lateUpdate=false)
    on ALERT → Startle
  }

  state Break {
    FlingObjectsFromGlobalPool(gameObject=SilkAcid BurstCloud (localpoolprefabs_assets_areacloverareamoss.bundle), spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=3, speedMin=2, speedMax=10, angleMin=0, angleMax=360, originVariationX=0.5, originVariationY=0.5, FSM="", FSMEvent="")
    InstaDeath(target=Self, direction=0)
  }

  state Chase {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Chase")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToChase", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=8.5, acceleration=0.35, targetSpread=1, spreadResetTimeMin=1, spreadResetTimeMax=2, offsetX=0, offsetY=0.5)
    on GO UP → Go Up Chase
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Follow {
    RandomFloat(min=-3, max=3, storeResult=var "Offset X")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    GameObjectIsNull(gameObject=var "Follow Target", isNull=→"IDLE", isNotNull=(none), storeResult=(unset), everyFrame=true)
    RandomFloat(min=2, max=0.75, storeResult=var "Offset Y")
    ChaseObject(gameObject=Self, target=var "Follow Target", speedMax=3, acceleration=0.15, targetSpread=0.5, spreadResetTimeMin=1, spreadResetTimeMax=2, offsetX=var "Offset X", offsetY=var "Offset Y")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    SetScale(gameObject=var "Alert Range", vector=(unset), x=1.5, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    WaitRandom(timeMin=3, timeMax=5, finishEvent=→"SHIFT", realTime=false)
    on ALERT → Startle
    on SHIFT → Follow
    on IDLE → Idle
  }

  state Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=8, everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=0.9, brakeOnExit=false)
    on FINISHED → Chase
  }

  state Spawn Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=20, everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=0.9, brakeOnExit=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    on FINISHED → Range Check
  }

  state Range Check {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="", storeResult=false, sendEvent="ALERT", outOfRangeEvent="IDLE", everyFrame=false)
    SetScale(gameObject=var "Alert Range", vector=(unset), x=2.5, y=2.5, z=(unset), everyFrame=false, lateUpdate=false)
    on ALERT → Chase
    on IDLE → Idle
  }

  state Spawn Down {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    RandomFloat(min=-12, max=-25, storeResult=var "Y Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Speed", everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=0.9, brakeOnExit=false)
    on FINISHED → Ran Fire
  }

  state Ran Fire {
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    RandomFloat(min=2, max=8, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Range Check
  }

  state Go Up Chase {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=1, everyFrame=false)
    on FINISHED → Chase
  }

  state Clear {
    DestroySelf(detachChildren=false)
  }

  state Fly In Ready {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    StopParticleEmitter(gameObject=var "Pt Idle")
    on BATTLE START → Check Pos
    on WAKE → Fly In
  }

  state Check Pos {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=8, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=false)
    on SHIFT → Shift
    on FINISHED → Fly In
  }

  state Shift {
    SetParent(gameObject=var "Spawn Pos Alt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Spawn Pos Alt", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    on FINISHED → Fly In
  }

  state Fly In {
    PlayParticleEmitter(gameObject=var "Pt Idle", emit=0, resetIfPlaying=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=10, z=-15, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetMeshRenderer(gameObject=Self, active=true)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -10, 15), time=0.75, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.55)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    on FINISHED → Chase
  }

  var Offset X: float = 0
  var Offset Y: float = 0
  var Y Speed: float = 0
  var Speed: float = 0
  var Angle: float = 0
  var Lifetime: float = 0
  var Distance: float = 0
  var Spawn Up: bool = false
  var Start Alert: bool = false
  var Spawn Down: bool = false
  var Start Anim: string = ""
  var Self: gameObject = Aspid Hatchling
  var Follow Target: gameObject = <null>
  var Alert Range: gameObject = Aspid Hatchling/Alert Range
  var Pt Idle: gameObject = Aspid Hatchling/Pt Idle
  var Spawn Pos Alt: gameObject = Aspid Hatchling/Spawn Pos Alt
}
