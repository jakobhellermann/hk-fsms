# Control
# 1 location(s):
#   globalpoolprefabs_assets_all.bundle / Burst Line R

fsm Control {
  start Burst

  state Burst {
    SetFloatValue(floatVariable=var "Scale", floatValue=1, everyFrame=false)
    EaseFloat(fromValue=1, toValue=14, floatVariable=var "Scale", time=0.15, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=true, lateUpdate=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    RecycleSelf()
  }

  var Scale: float = 0
}
