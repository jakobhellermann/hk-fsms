# Control
# 1 location(s):
#   Cog_07 / Black Thread States/Normal World/Repairable Scene/Repairing Scene/Song Automaton Ball Inert

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="Boss Corpse", collideLayer=(unset), sendEvent=→"COG SMASH", storeCollider=var "")
    on TINK DOWN → Random Dir
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Random Dir
    on TINK DOWN R → Tink R
    on TINK DOWN L → Tink L
    on COG SMASH → Cog Destroy
  }

  state Tink L {
    SetAngularVelocity2d(gameObject=Self, angularVelocity=100, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-10, y=(unset), everyFrame=false)
    on FINISHED → First Hit
  }

  state Random Dir {
    SendRandomEvent(events=[→"TINK LEFT", →"TINK RIGHT"], weights=[1, 1], delay=0)
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
  }

  state Tink R {
    SetAngularVelocity2d(gameObject=Self, angularVelocity=-100, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=10, y=(unset), everyFrame=false)
    on FINISHED → First Hit
  }

  state First Hit {
    BoolTest(boolVariable=var "Did First Hit", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Did First Hit", boolValue=true, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SendEventToRegister(eventName="ALERT")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FIRST HIT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Cog Destroy {
    SetPlayerDataBool(boolName="cog7_automatonDestroyed", value=true)
    PlayParticleEmitter(gameObject=var "Cog Grind Particle", emit=0, resetIfPlaying=false)
    SetFloatValue(floatVariable=var "Death Time", floatValue=1, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.4)
    on FINISHED → Audio
  }

  state Audio {
    PlayAudioEventRandom(audioClips=[automaton_generic_death_1 (sfxstatic_assets_areacog.bundle), automaton_generic_death_2 (sfxstatic_assets_areacog.bundle), automaton_generic_death_3 (sfxstatic_assets_areacog.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FloatAdd(floatVariable=var "Death Time", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Death Time", float2=0, tolerance=0, equal=→"DESTROY", lessThan=→"DESTROY", greaterThan=(none), everyFrame=true)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Audio
    on DESTROY → Destroy
  }

  state Destroy {
    SetParent(gameObject=var "Cog Grind Particle", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    StopParticleEmitter(gameObject=var "Cog Grind Particle")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Death Time: float = 0
  var Did First Hit: bool = false
  var Cog Grind Particle: gameObject = Black Thread States/Normal World/Repairable Scene/Repairing Scene/Song Automaton Ball Inert/Cog Grind Particle
}
