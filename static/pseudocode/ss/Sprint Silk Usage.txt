# Sprint Silk Usage
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet

// uses template: hornet_sprint_silk_usage
fsm Sprint Silk Usage {
  start Check Pause
  on TOOL EQUIPS CHANGED → Check Pause  // from any state

  state Idle {
    Wait(time=4, finishEvent=→"RESET", realTime=false)
    on START SILK USAGE → Start Usage
    on RESET → Reset Timer
    on START SILK USAGE BURST → 
  }

  state Usage {
    GetSilk(StoreAmount=var "Current Silk")
    IntCompare(integer1=var "Current Silk", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    AddUsingSilk(Amount=1, UsingType=Normal, DidAddTracker=var "Added Using Silk")
    PerSecond(floatValue=1, storeResult=var "Timer Add", everyFrame=true)
    FloatOperator(float1=var "Use Silk Timer", float2=var "Timer Add", operation=Add, storeResult=var "Use Silk Timer", everyFrame=true)
    FloatCompare(float1=var "Use Silk Timer", float2=6, tolerance=0, equal=→"USE SILK", lessThan=(none), greaterThan=→"USE SILK", everyFrame=true)
    on STOP SILK USAGE → StopUsage
    on USE SILK → Use Silk
    on CANCEL → Pause Usage
  }

  state StopUsage {
    SetFsmBool(gameObject=Self, fsmName="Sprint", variableName="Sprintmaster Active", setValue=false, everyFrame=false)
    RemoveUsingSilk(Amount=1, UsingType=Normal, DidAddTracker=var "Added Using Silk")
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetSilkRegenBlocked", parameters=[false], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Use Silk {
    RemoveUsingSilk(Amount=1, UsingType=Normal, DidAddTracker=var "Added Using Silk")
    SetFloatValue(floatVariable=var "Use Silk Timer", floatValue=0, everyFrame=false)
    CheckIsSilkDrainDisabled(trueEvent=→"FINISHED", falseEvent=(none), storeValue=(unset))
    TakeSilkV2(Target=var "Hero", Amount=1, TakeSource=ActiveUse)
    on FINISHED → Usage
  }

  state Reset Timer {
    SetFloatValue(floatVariable=var "Use Silk Timer", floatValue=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Start Usage {
    GetSilk(StoreAmount=var "Current Silk")
    IntCompare(integer1=var "Current Silk", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Sprint", variableName="Sprintmaster Active", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetSilkRegenBlocked", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on CANCEL → Pause Usage
    on FINISHED → Usage
  }

  state Burst {
    FloatOperator(float1=var "Use Silk Timer", float2=1, operation=Add, storeResult=var "Use Silk Timer", everyFrame=false)
    on FINISHED → Start Usage
  }

  state Check Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check Sprintmaster
  }

  state Check Sprintmaster {
    CheckIfToolEquipped(Tool=Sprintmaster (dataassets_assets_assets/dataassets/tools/toolitems.bundle), RequiredAmountLeft=(unset), trueEvent=→"EQUIPPED", falseEvent=→"UNEQUIPPED", storeValue=(unset))
    on EQUIPPED → Idle
    on UNEQUIPPED → Unequipped
  }

  state Unequipped {
    SetFsmBool(gameObject=Self, fsmName="Sprint", variableName="Sprintmaster Active", setValue=false, everyFrame=false)
  }

  state Pause Usage {
    SetFsmBool(gameObject=Self, fsmName="Sprint", variableName="Sprintmaster Active", setValue=false, everyFrame=false)
    RemoveUsingSilk(Amount=1, UsingType=Normal, DidAddTracker=var "Added Using Silk")
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetSilkRegenBlocked", parameters=[false], storeResult=(unset var), EveryFrame=false)
    GetSilkV2(StoreAmount=var "Current Silk", EveryFrame=true)
    IntCompare(integer1=var "Current Silk", integer2=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=true)
    on CANCEL → Start Usage
    on STOP SILK USAGE → Idle
  }

  var Use Silk Timer: float = 0
  var Timer Add: float = 0
  var Current Silk: int = 0
  var Added Using Silk: bool = false
}
