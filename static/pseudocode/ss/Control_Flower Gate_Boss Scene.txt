# Control
# 1 location(s):
#   Shellwood_22 / Boss Scene/Flower Gate

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "vine_wall")
    FindNamedChild(gameObject=Self, storeResult=var "Destroy Effects")
    on FINISHED → Idle
  }

  state Idle {
    on SETH DEFEATED → Break Antic
  }

  state Break Antic {
    ObjectJitter(gameObject=Self, x=0.1, y=(unset), z=(unset), allowMovement=false, limitFps=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=shellwood_twig_wall_hit_2 (sfxstatic_assets_areashellwood.bundle))
    on FINISHED → Break
  }

  state Break {
    ActivateGameObject(gameObject=var "Destroy Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "vine_wall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=shellwood_twig_wall_break (sfxstatic_assets_areashellwood.bundle))
  }

  var vine_wall: gameObject = <null>
  var Destroy Effects: gameObject = <null>
}
