# FSM
# 4 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival Outro/Audio Flea Voice (2)
#   Aqueduct_05_festival / Caravan_States/Flea Festival Outro/Audio Flea Voice (3)
#   Aqueduct_05_festival / Caravan_States/Flea Festival Outro/Audio Flea Voice
#   Aqueduct_05_festival / Caravan_States/Flea Festival Outro/Audio Flea Voice (1)

fsm FSM {
  start State 1
  on FINAL PINATA HIT → Sing  // from any state

  state State 1 {
    AddEventRegister(target=Self, eventName="PINATA HIT")
    AddEventRegister(target=Self, eventName="FINAL PINATA HIT")
    on FINISHED → State 2
  }

  state State 2 {
    on PINATA HIT → Howl
    on FINAL PINATA HIT → Sing
  }

  sequence state Howl {
    WaitRandom(timeMin=0.3, timeMax=0.7, finishEvent=(none), realTime=false)
    PlayRandomAudioClipTable(Table=Flea_Howl (sfxdynamic_assets_fleacaravan.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → State 2
  }

  state Sing {
    WaitRandom(timeMin=0.3, timeMax=0.7, finishEvent=(none), realTime=false)
    GetRandomAudioClipFromTable(Table=Flea_sing (sfxdynamic_assets_fleacaravan.bundle), ForcePlay=true, StoreClip=var "Clip")
    SetAudioClip(gameObject=Self, audioClip=var "Clip", autoPlay=true, stopOnExit=false)
    RandomiseAudioPosition(gameObject=Self, timeMin=0, timeMax=5)
  }

  var Clip: object = <null>
}
