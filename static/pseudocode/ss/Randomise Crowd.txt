# Randomise Crowd
# 1 location(s):
#   Coral_42 / Rosary Thief Crowd

fsm Randomise Crowd {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    RandomInt(min=2, max=4, storeResult=var "Crowd Amount", inclusiveMax=true, noRepeat=false)
    on FINISHED → Activate Thief
  }

  state Activate Thief {
    GetRandomChild(gameObject=Self, storeResult=var "Thief")
    ActivateGameObject(gameObject=var "Thief", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Thief", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    IntAdd(intVariable=var "Crowd Amount", add=-1, everyFrame=false)
    IntCompare(integer1=var "Crowd Amount", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"REPEAT", everyFrame=false)
    on REPEAT → Activate Thief
    on END → End
  }

  state End {
  }

  var Crowd Amount: int = 0
  var Self: gameObject = <null>
  var Thief: gameObject = <null>
}
