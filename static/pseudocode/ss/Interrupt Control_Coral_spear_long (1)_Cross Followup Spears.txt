# Interrupt Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Cross Followup Spears/Coral_spear_long (1)

fsm Interrupt Control {
  start Antic

  state Antic {
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    on BOSS DEFEATED → Deactivate
    on FINISHED → Animating
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Animating {
    on BOSS DEFEATED → Break
  }

  state Break {
    AnimatorStopPlayback(gameObject=Self)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    FindNamedChild(gameObject=Self, storeResult=var "Cancel Break Effects")
    ActivateGameObject(gameObject=var "Cancel Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Damager: gameObject = <null>
  var spear_set: gameObject = <null>
  var Self: gameObject = Boss Scene/Cross Followup Spears/Coral_spear_long (1)
  var Cancel Break Effects: gameObject = <null>
}
