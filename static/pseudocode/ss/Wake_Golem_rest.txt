# Wake
# 1 location(s):
#   Bone_East_08_boss_golem_rest / Boss Scene/Golem_rest

fsm Wake {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="Rumble Loop", storeResult=var "Rumble Loop")
    FindChild(gameObject=Self, childName="Golem", storeResult=var "Golem")
    FindChild(gameObject=Self, childName="Break Burst", storeResult=var "Break Burst")
    FindChild(gameObject=Self, childName="Break Effects", storeResult=var "Break Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Thrown Pin")
    FindChild(gameObject=var "Golem", childName="Head", storeResult=var "Head")
    FindChild(gameObject=var "Head", childName="Eye Glow", storeResult=var "Eyes")
    FindChild(gameObject=var "Golem", childName="Bod", storeResult=var "Bod")
    FindChild(gameObject=var "Golem", childName="Torso", storeResult=var "Torso")
    FindChild(gameObject=Self, childName="Pieces", storeResult=var "Pieces")
    FindChild(gameObject=var "Pieces", childName="Piece BL", storeResult=var "Piece BL")
    FindChild(gameObject=var "Pieces", childName="Piece BR", storeResult=var "Piece BR")
    FindChild(gameObject=var "Pieces", childName="Piece TL", storeResult=var "Piece TL")
    FindChild(gameObject=var "Pieces", childName="Piece TR", storeResult=var "Piece TR")
    FindChild(gameObject=Self, childName="Glows", storeResult=var "Glows")
    FindChild(gameObject=Self, childName="Pt Antic", storeResult=var "Pt Antic")
    on FINISHED → Idle
  }

  state Idle {
    on WAKE → Head Wake
  }

  state Head Wake {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    EaseFloat(fromValue=0, toValue=9, floatVariable=var "Rotation", time=0.5, speed=(unset), delay=(unset), easeType=28, reverse=false, finishEvent=(none), realTime=false)
    SetRotation(gameObject=var "Head", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true, lateUpdate=false)
    ActivateGameObject(gameObject=var "Eyes", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Golem", audioClip=chorus_eyes_open (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Golem", audioClip=chorus_under_rocks_shake_initiate (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Rumble
  }

  state Rumble {
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=true, Delay=0)
    ObjectJitter(gameObject=var "Bod", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Head", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Torso", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Piece BL", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Piece BR", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Piece TL", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "Piece TR", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    AudioPlayInState(gameObject=var "Rumble Loop", volume=1)
    on FINISHED → Break Apart
  }

  state Break Apart {
    ActivateGameObject(gameObject=var "Active Art", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="HugeShake", delay=0, everyFrame=false)
    StartRoarEmitter(spawnPoint=var "Head", delay=0, stunHero=true, roarBurst=true, isSmall=false, noVisualEffect=false, forceThroughBind=false, stopOnExit=false)  // disabled
    ActivateGameObject(gameObject=var "Golem", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Glows", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="ACTIVATE GOLEM")
    PlayParticleEmitterChildren(gameObject=var "Break Effects", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetAnimatorTrigger(gameObject=var "Pieces", trigger="break")
    ScreenFlash(flashColour=(0.875, 0.22398578, 0.032169096, 0.141))
    ActivateGameObject(gameObject=var "Break Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ScaleTo(gameObject=var "Break Burst", target=(12, 12, 12), duration=0.25, delay=0, curve=Linear)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Golem", audioClip=chorus_under_rocks_break_free (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Thrown Pin", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=var "Break Burst", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Rotation: float = 0
  var Bod: gameObject = <null>
  var Break Burst: gameObject = <null>
  var Break Effects: gameObject = <null>
  var Eyes: gameObject = <null>
  var Glows: gameObject = <null>
  var Golem: gameObject = <null>
  var Head: gameObject = <null>
  var Piece BL: gameObject = <null>
  var Piece BR: gameObject = <null>
  var Piece TL: gameObject = <null>
  var Piece TR: gameObject = <null>
  var Pieces: gameObject = <null>
  var Pt Antic: gameObject = <null>
  var Torso: gameObject = <null>
  var Rumble Loop: gameObject = <null>
  var Thrown Pin: gameObject = <null>
  var Active Art: gameObject = Boss Scene/Golem_rest/Active Art
}
