// uses template: interactive_bell
fsm interactive_bell {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=Self, childName="glow_rune", storeResult=var "Glow Rune")
    FindNamedChild(gameObject=Self, storeResult=var "Strike Point")
    FindChild(gameObject=var "Parent", childName="glow_rune", storeResult=var "Glow Rune Parent")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Nail Attack", sendEvent=→"NAIL HIT", storeCollider=var "Hitter")
    on NAIL HIT → Get Dir
  }

  state Get Dir {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=var "Strike Clip", pitchMin=0.85, pitchMax=1.15, volume=0.6, delay=0, storePlayer=<null>)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Strike Point", position=(0, 0, 0), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    GetDamagerDirection(target=var "Hitter", storeDirection=var "Atk Dir")
    SetFloatValue(floatVariable=var "Hit Force", floatValue=12.5, everyFrame=false)
    FloatCompare(float1=var "Atk Dir", float2=0, tolerance=0, equal=→"R", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Atk Dir", float2=180, tolerance=0, equal=→"L", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Hit Force", floatValue=9, everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Hit L
    on R → Apply Force
  }

  state Hit L {
    FloatMultiply(floatVariable=var "Hit Force", multiplyBy=-1, everyFrame=false)
    on FINISHED → Apply Force
  }

  state Hit Recovery {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Apply Force {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=var "Hit Force", y=(unset), vector3=(unset), everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="STRUCK", delay=0, everyFrame=false)
    on FINISHED → Bench Bell FX
  }

  state Bench Bell FX {
    BoolTest(boolVariable=var "Bench Bell", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    AnimatorPlay(gameObject=var "Glow Rune", stateName="Hit", layer=(unset), normalizedTime=0, everyFrame=false)  // disabled
    AnimatorPlay(gameObject=var "Glow Rune Parent", stateName="Hit", layer=(unset), normalizedTime=0, everyFrame=false)  // disabled
    on FINISHED → Hit Recovery
  }

  var Bench Bell: bool = false
  var Strike Clip: object = bell_bench_bell_small_right (sfxstatic_assets_bench.bundle)

  var Atk Dir: float = 0
  var Hit Force: float = 0
  var Glow Rune: gameObject = <null>
  var Glow Rune Parent: gameObject = <null>
  var Hitter: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Strike Point: gameObject = <null>
}
