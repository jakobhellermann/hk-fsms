# Rest Animation
# 1 location(s):
#   localpoolprefabs_assets_areasong.bundle / Enclave Pilgrim Small

fsm Rest Animation {
  start Idle

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on RESTING → Get Previous State
    on SING → Sing
    on STARTLE → Startle
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Enclave Pilgrim Small@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → To Idle
    on RIGHT → To Idle
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "Should Face Left"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"WAKE", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=(none), ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on WAKE → Wake Up
  }

  state To Idle {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="To Rest", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Sing {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.7, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Pilgrim_npc_female_mid_sing (sfxdynamic_assets_areamoss.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on CANCEL → Sing End
  }

  sequence state Sing End {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWait(Target=Self, ClipName="To Rest", AnimationCompleteEvent=(none))
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  state Wake Up {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Awake End
  }

  state Awake End {
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=true, Target=Self)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    SetBoolValue(boolVariable=var "Resting", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Startle {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Wait(time=0.5, finishEvent=→"CANCEL", realTime=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Pilgrim_npc_female_mid_alert (sfxdynamic_assets_areamoss.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on CANCEL → Sing End
  }

  var Resting: bool = false
  var Should Face Left: bool = false
  var Audio Loop Voice: gameObject = Enclave Pilgrim Small/Audio Loop Voice
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
