  start State 1

  state State 1 {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → State 2
  }

  state State 2 {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControlNotVelocity", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataBool(boolName="crashingIntoGreymoor", value=true)
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 1), duration=0.1)
    TransitionToAudioSnapshot(snapshot=Off (audioobjects_assets_all.bundle), transitionTime=0.3)
    TransitionToAudioSnapshot(snapshot=at None (audioobjects_assets_all.bundle), transitionTime=0.3)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=true)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 18), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → State 3
  }

  state State 3 {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=EnterWithoutInput(true))
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    BeginSceneTransition(sceneName="Greymoor_01", entryGateName="bot1", entryDelay=0, visualization=Default, preventCameraFadeOut=false)
  }
}
