# Catch Control
# 1 location(s):
#   localpoolprefabs_assets_laceboss.bundle / lost_lace_cross_slash_catcher

fsm Catch Control {
  start Init
  on LACE DEATH START → Recycle  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Hornet Catch Dummy")  // disabled
    FindNamedChild(gameObject=Self, storeResult=var "Pt Grab")  // disabled
    GetParent(gameObject=Self, storeResult=var "Boss Scene")  // disabled
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Lost Lace Boss")  // disabled
    Wait(time=0.2, finishEvent=(none), realTime=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Grab Start {
    SetBoolValue(boolVariable=var "Grabbing Hero", boolValue=true, everyFrame=false)
    RunFSM(fsmTemplateControl=template=-5200671031615995940 in[Bypass Checks<-false, Dont Deal Damage<-var "Dont Deal Damage", No Hero Invuln<-false, Grab Effect Point<-var "Hero"], finishEvent=(none), everyFrame=false)
    on CANCEL → Grab Cancel
    on FINISHED → Grabbed
  }

  state Grabbed {
    SetPositionToObject2D(gameObject=var "Hornet Catch Dummy", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    SetPositionToObject2D(gameObject=var "Pt Grab", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    MatchScaleSign(Target=var "Hornet Catch Dummy", MatchTo=var "Hero", EveryFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Hornet Catch Dummy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    PlayRandomAudioClipTable(Table=Wound Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0))
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=grab_damage (sfxstatic_assets_areamemory.bundle), pitchMin=1, pitchMax=1.1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Grab Effect Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    PlayAudioEvent(audioClip=hero_damage (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Grab", emit=0, resetIfPlaying=false)  // disabled
    SetMeshRenderer(gameObject=var "Hero", active=true)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Wound Double Strike")
    on FINISHED → State 1
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"GRAB HERO", storeCollider=var "")
    CheckXPosition(gameObject=Self, compareTo=6, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"CANCEL", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    CheckXPosition(gameObject=Self, compareTo=52, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"CANCEL", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on GRAB HERO → Grab Start
    on CANCEL → Recycle
  }

  state Grab Cancel {
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  state State 1 {
    SendEventToRegister(eventName="CAUGHT HORNET")
    SetFsmBool(gameObject=var "Lost Lace Boss", fsmName="Control", variableName="Caught Hornet", setValue=true, everyFrame=false)  // disabled
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Recycle {
    RecycleSelf()
  }

  var Grabbing Hero: bool = false
  var Grab Effect Point: gameObject = <null>
  var Hornet Catch Dummy: gameObject = <null>
  var Boss Scene: gameObject = <null>
  var Lost Lace Boss: gameObject = <null>
  var Pt Grab: gameObject = <null>
}
