# Arena Setup
# 1 location(s):
#   Memory_Coral_Tower / Battle Scenes/Battle Scene Chamber 4/Wave 6 - side plats and fish

fsm Arena Setup {
  start State 1

  state State 1 {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    on WAVE QUEUED → State 2
  }

  sequence state State 2 {
    Wait(time=var "Delay", finishEvent=(none), realTime=false)
    SetHazardRespawn(hazardRespawnMarker=var "Respawn Marker")
    on WAVE START → Pause
  }

  state Pause {
    GetConstantsValue(variableName="CORAL_SPEAR_SPAWN_TIME", storeValue=var "Spawn Time")
    FloatAdd(floatVariable=var "Spawn Time", add=-0.65, everyFrame=false, perSecond=false)
    Wait(time=var "Spawn Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Break {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Platform Set A")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Platform Break Effect A")
    FindChild(gameObject=var "Platform Set A", childName="Activator", storeResult=var "Activator A")  // disabled
    SendMessage(gameObject=var "Activator A", delivery=SendMessage, options=1, functionCall=ForceDeactivate())  // disabled
    ActivateGameObject(gameObject=var "Platform Set A", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Platform Break Effect A", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Delay: float = 0
  var Respawn Marker: gameObject = Battle Scenes/Battle Scene Chamber 4/Hazard Respawn Marker L

  var Spawn Time: float = 0
  var Battle Scene: gameObject = <null>
  var Platform Set HazA: gameObject = <null>
  var Activator A: gameObject = <null>
  var Platform Set A: gameObject = <null>
  var Platform Break Effect A: gameObject = <null>
}
