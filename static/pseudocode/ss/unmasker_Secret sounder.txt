# unmasker
# 1 location(s):
#   Tut_01 / Secret sounder

// uses template: unmasker
fsm unmasker {
  start Pause

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on UNCOVER → Fade
    on STAY → Idle Stay
    on FINISHED → Idle Stay
    on ACTIVATE → Activate
  }

  state Fade {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.5, delay=0, easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Play Sound", isTrue=→"SOUND", isFalse=(none), everyFrame=false)
    on SOUND → Sound
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Idle
  }

  state Idle Stay {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="Player", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Fade
  }

  state Sound {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=secret_discovered_temp (sfxstatic_assets_shared.bundle))  // disabled
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClip=secret_discovered_temp (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)  // disabled
    SendEventToRegister(eventName="SECRET TONE")
  }

  state Activate {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.1, delay=0, easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
  }

  var Play Sound: bool = true

  var Activated: bool = false
}
