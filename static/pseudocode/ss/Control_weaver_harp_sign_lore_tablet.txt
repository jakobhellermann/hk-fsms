# Control
# 1 location(s):
#   Mosstown_02 / lore_tablet/weaver_harp_sign

fsm Control {
  start Pause
  on MAP TAKEN → Map Taken  // from any state
  on MAP TAKE → Map Take  // from any state

  state Pause {
    ActivateGameObject(gameObject=var "Inspect Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)  // disabled
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0)
    FadeNestedFadeGroup(Target=var "Glow Text", ToAlpha=0, FadeTime=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Inactive
  }

  state Inactive {
    AudioStop(gameObject=Self, fadeTime=0)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Needolin Range
  }

  state Needolin Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on EXIT → Inactive
    on NEEDOLIN → Needolin Start
  }

  state Needolin Start {
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=var "")
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.5)
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=1, FadeTime=2)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    GetAttunementTime(storeVariable=var "Needolin Time", baseTime=4, extraTime=0)
    GetAttunementTime(storeVariable=var "Up Delay", baseTime=1, extraTime=0)
    GetAttunementTime(storeVariable=var "Up Time", baseTime=3, extraTime=0)
    CallMethodProper(gameObject=var "Summon Glow", behaviour="ColorFader", methodName="SetUpTime", parameters=[var "Up Time"], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Summon Glow", behaviour="ColorFader", methodName="SetUpDelay", parameters=[var "Up Delay"], storeResult=(unset var), EveryFrame=false)
    Wait(time=var "Needolin Time", finishEvent=→"NEXT", realTime=false)
    FadeColorFader(target=var "Summon Glow", fadeType=UP, useChildren=true)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumStart())
    on END → Needolin End
    on NEXT → Activate Burst
  }

  state Needolin End {
    FadeColorFader(target=var "Summon Glow", fadeType=DOWN, useChildren=true)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0, time=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0.5)
    on FINISHED → Inactive
  }

  state Activate Burst {
    PlayAudioEvent(audioClip=weaver_harp_sign_burst (sfxstatic_assets_areaweaver.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Flashes", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ScreenFlash(flashColour=(1, 1, 1, 0.516))
    FadeColorFader(target=var "Summon Glow", fadeType=DOWN, useChildren=true)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    SendEventByName(eventTarget=GameObject(var "Notify"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0.2)
    on FINISHED → Text Up
  }

  state Text Up {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Glow Text", ToAlpha=1, FadeTime=0.5)
    ActivateGameObject(gameObject=var "Glow Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "text_bg", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Wisp", emit=0, resetIfPlaying=false)
    on FINISHED → Inspectable
  }

  state Readable {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.3, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on NEEDOLIN → Idle Strum
    on EXIT → Away Pause
  }

  state Deactivate {
    FadeNestedFadeGroup(Target=var "Glow Text", ToAlpha=0, FadeTime=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    StopParticleEmitter(gameObject=var "Pt Wisp")
    ActivateGameObject(gameObject=var "Inspect Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flashes", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inactive
  }

  state Activated {
    SendEventByName(eventTarget=GameObject(var "Notify"), sendEvent="ACTIVATED", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Wisp", emit=0, resetIfPlaying=false)
    on FINISHED → Deactivate
  }

  state Away Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on FINISHED → Deactivate
    on ENTER → Readable
  }

  state Fade Text In {
    FadeColorFader(target=var "Glow Text", fadeType=UP, useChildren=true)
    ActivateGameObject(gameObject=var "text_bg", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)  // disabled
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=QuickStrum())  // disabled
    on FINISHED → Readable
  }

  state Inspectable {
    ActivateGameObject(gameObject=var "Inspect Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Readable
  }

  state Map Taken {
    ActivateGameObject(gameObject=var "Glow Text", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Wisp", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Summon Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Strings", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Map Take {
    FadeColorFader(target=var "Glow Text", fadeType=DOWN, useChildren=true)
    StopParticleEmitter(gameObject=var "Pt Wisp")
    FadeColorFader(target=var "Summon Glow", fadeType=DOWN, useChildren=true)
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=0, FadeTime=2)
  }

  state Idle Strum {
    ActivateInteractible(Target=var "Inspect Region", Activate=false, AllowQueueing=false, UseChildren=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=1, FadeTime=2)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.3, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumStart())
    on CANCEL → Not In Range
    on END → Idle Strum End
  }

  state Not In Range {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Readable
  }

  state Idle Strum End {
    ActivateInteractible(Target=var "Inspect Region", Activate=true, AllowQueueing=false, UseChildren=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0.5)
    on FINISHED → Readable
  }

  var Inspect Region: gameObject = lore_tablet/weaver_harp_sign/Inspect Region
  var Notify: gameObject = lore_tablet/moss_bone_plaque

  var Needolin Time: float = 0
  var Up Delay: float = 0
  var Up Time: float = 0
  var Strings Group Idle Alpha: float = 0.25
  var Activated: bool = false
  var Flashes: gameObject = lore_tablet/weaver_harp_sign/flashes
  var Glow Text: gameObject = lore_tablet/weaver_harp_sign/glow_text
  var Pt Wisp: gameObject = lore_tablet/weaver_harp_sign/Pt Wisp
  var Strings: gameObject = lore_tablet/weaver_harp_sign/Strings
  var Summon Glow: gameObject = lore_tablet/weaver_harp_sign/Summon Glow
  var text_bg: gameObject = lore_tablet/weaver_harp_sign/text_bg
}
