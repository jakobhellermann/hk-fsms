# Control
# 2 location(s):
#   Peak_02 / Weaver Servitor Large/Legs Container/Legs/Head Swiveller/Head
#   Peak_04d / Weaver Servitor Large/Legs Container/Legs/Head Swiveller/Head

fsm Control {
  start Init
  on LOCK TARGET → Lock Target  // from any state
  on LOOK STOP → Get Reset Angle  // from any state

  state Init {
    SetParent(gameObject=var "Aim Lock", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Aim Lock Random", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Rest
  }

  state Rest {
    on LOOK START → Target Hero
    on LOOK RANDOM → Target Random
  }

  state Look L {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Angle", add=90, everyFrame=true, perSecond=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Target", offsetX=0, offsetY=0, storeAngle=var "Target Angle", pause=0, everyFrame=true)
    GetDirectionBetweenAngles(startAngle=var "Self Angle", targetAngle=var "Target Angle", storeDirection=var "Clockwise", storeDistance=var "Angle Distance", everyFrame=true)
    BoolTest(boolVariable=var "Clockwise", isTrue=(none), isFalse=→"R", everyFrame=true)
    FloatAdd(floatVariable=var "Rotate Speed", add=1200, everyFrame=true, perSecond=true)
    FloatOperator(float1=var "Angle Distance", float2=-4, operation=Multiply, storeResult=var "Angle Clamp Min", everyFrame=true)
    FloatOperator(float1=var "Angle Distance", float2=4, operation=Multiply, storeResult=var "Angle Clamp Max", everyFrame=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=var "Angle Clamp Min", maxValue=var "Angle Clamp Max", everyFrame=true)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on LOOK STOP → Get Reset Angle
    on R → Look R
    on LOOK PAUSE → Look Pause
  }

  state Reset {
    AnimateRotationTo(gameObject=Self, fromValue=var "Self Angle", toValue=0, worldSpace=false, negativeSpace=false, time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rest
  }

  state Start Dir {
    SetFloatValue(floatVariable=var "Rotate Speed", floatValue=0, everyFrame=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Self Angle", add=90, everyFrame=false, perSecond=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Target", offsetX=0, offsetY=0, storeAngle=var "Target Angle", pause=0, everyFrame=false)
    FloatOperator(float1=360, float2=var "Self Angle", operation=Subtract, storeResult=var "Delta", everyFrame=false)  // disabled
    GetDirectionBetweenAngles(startAngle=var "Self Angle", targetAngle=var "Target Angle", storeDirection=var "Clockwise", storeDistance=0, everyFrame=false)
    BoolTest(boolVariable=var "Clockwise", isTrue=→"L", isFalse=→"R", everyFrame=false)
    on L → Look L
    on R → Look R
  }

  state Look R {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Self Angle", add=90, everyFrame=true, perSecond=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Target", offsetX=0, offsetY=0, storeAngle=var "Target Angle", pause=0, everyFrame=true)
    GetDirectionBetweenAngles(startAngle=var "Self Angle", targetAngle=var "Target Angle", storeDirection=var "Clockwise", storeDistance=var "Angle Distance", everyFrame=true)
    BoolTest(boolVariable=var "Clockwise", isTrue=→"L", isFalse=(none), everyFrame=true)
    FloatAdd(floatVariable=var "Rotate Speed", add=-1200, everyFrame=true, perSecond=true)
    FloatOperator(float1=var "Angle Distance", float2=-4, operation=Multiply, storeResult=var "Angle Clamp Min", everyFrame=true)
    FloatOperator(float1=var "Angle Distance", float2=4, operation=Multiply, storeResult=var "Angle Clamp Max", everyFrame=true)
    FloatClamp(floatVariable=var "Rotate Speed", minValue=var "Angle Clamp Min", maxValue=var "Angle Clamp Max", everyFrame=true)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on LOOK STOP → Get Reset Angle
    on L → Look L
    on LOOK PAUSE → Look Pause
  }

  state Get Reset Angle {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Self Angle", space=1, everyFrame=false)
    FloatCompare(float1=var "Self Angle", float2=180, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatAdd(floatVariable=var "Self Angle", add=-360, everyFrame=false, perSecond=false)
    on FINISHED → Reset
  }

  state Look Pause {
    on LOOK START → Target Hero
    on LOOK RANDOM → Target Random
  }

  state Target Hero {
    SetGameObject(variable=var "Target", gameObject=var "Hero", everyFrame=false)
    on FINISHED → Start Dir
  }

  state Lock Target {
    SetPositionToObject2D(gameObject=var "Aim Lock", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    SetGameObject(variable=var "Target", gameObject=var "Aim Lock", everyFrame=false)
    on FINISHED → Start Dir
  }

  state Target Random {
    RandomFloat(min=3, max=10, storeResult=var "Random Pos X")
    RandomlyFlipFloat(storeResult=var "Random Pos X")
    SetPositionToObject2D(gameObject=var "Aim Lock Random", targetObject=var "Self", xOffset=var "Random Pos X", yOffset=6, everyFrame=false)
    SetGameObject(variable=var "Target", gameObject=var "Aim Lock Random", everyFrame=false)
    on FINISHED → Start Dir
  }

  var Angle: float = 0
  var Angle Clamp Max: float = 0
  var Angle Clamp Min: float = 0
  var Angle Distance: float = 0
  var Delta: float = 0
  var Rotate Speed: float = 0
  var Self Angle: float = 0
  var Target Angle: float = 0
  var Random Pos X: float = 0
  var Clockwise: bool = false
  var Aim Lock: gameObject = <null>
  var Target: gameObject = <null>
  var Aim Lock Random: gameObject = Weaver Servitor Large/Aim Lock Random
  var Self: gameObject = Weaver Servitor Large
}
