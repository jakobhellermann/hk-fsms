# FSM
# 1 location(s):
#   Coral_44 / Secret Mask New Sharp Ultra

fsm FSM {
  start Idle

  state Idle {
    on BREAK → Fade Out
    on BROKEN → Deactivate
  }

  state Fade Out {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0.2)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on BROKEN → Deactivate
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
