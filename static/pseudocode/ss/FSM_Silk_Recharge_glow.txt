# FSM
# 1 location(s):
#   localpoolprefabs_assets_bench.bundle / Silk_Recharge_glow

fsm FSM {
  start Appear

  state Appear {
    AnimatorPlay(gameObject=Self, stateName="Appear", layer=0, normalizedTime=0, everyFrame=false)
    on STOP → Disappear
    on MAX → Maxed
  }

  state Disappear {
    AnimatorPlayStateWait(target=Self, stateName="Disappear", finishEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  sequence state Maxed {
    AnimatorPlayStateWait(target=Self, stateName="Disappear Maxed", finishEvent=→"FINISHED")
    on FINISHED → Recycle
  }
}
