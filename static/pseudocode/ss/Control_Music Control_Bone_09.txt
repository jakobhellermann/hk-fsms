# Control
# 1 location(s):
#   Bone_09 / Music Control

fsm Control {
  start Pause

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check
  }

  state Check {
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="right1", equalEvent=→"FROM DOCKS", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Gate", compareTo="right2", equalEvent=→"FROM DOCKS", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FROM DOCKS → From Docks
    on FINISHED → Normal
  }

  state Normal {
    ApplyMusicCue(musicCue=Vaults (audiocuesdynamic_assets_areabonearealibrary.bundle), delayTime=0, transitionTime=0)  // disabled
    TransitionToAudioSnapshot(snapshot=Boneforest Lava (audioobjects_assets_all.bundle), transitionTime=4)
  }

  sequence state From Docks {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    Wait(time=3, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=Boneforest (audiocuesdynamic_assets_areabone.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Boneforest Lava (audioobjects_assets_all.bundle), transitionTime=4)
  }

  var Gate: string = ""
}
