# Control
# 1 location(s):
#   Shellwood_14 / Ambush Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Jitter Parent")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Antic")
    FindNamedChild(gameObject=Self, storeResult=var "Enemies Up")
    FindNamedChild(gameObject=Self, storeResult=var "Enemies Down")
    PlayerDataBoolTest(boolName="shellwood14_ambushed", isTrue=→"AMBUSHED", isFalse=(none))
    on FINISHED → Idle
    on AMBUSHED → Re Ambush
  }

  state Idle {
    on FLEA RESCUED → Ambush Pause
    on HEART PIECE COLLECTED → Ambush Pause
  }

  state Ambush Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetPlayerDataBool(boolName="shellwood14_ambushed", value=true)
    on FINISHED → Ambush Antic
  }

  state Ambush Antic {
    SendMessage(gameObject=var "Jitter Parent", delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=vine_ball_antic_for_emerge (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=vine_spikes_wall_disappear (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Enemies Enter
  }

  state Enemies Enter {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=vine_ball_emerge (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=vine_spikes_wall_appear (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=flower_side_dash (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Enemies Up", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemies Down", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemies Up"), sendEvent="ENTER UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemies Down"), sendEvent="ENTER DOWN", delay=0, everyFrame=false)
  }

  state Re Ambush {
    ActivateGameObject(gameObject=var "Enemies Up", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemies Down", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Jitter Parent: gameObject = <null>
  var Pt Antic: gameObject = <null>
  var Enemies Up: gameObject = <null>
  var Enemies Down: gameObject = <null>
}
