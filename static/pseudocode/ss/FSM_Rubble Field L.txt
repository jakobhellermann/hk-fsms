# FSM
# 3 location(s):
#   Cradle_03 / Boss Scene/Rubble Fields/Rubble Field L
#   Cradle_03 / Boss Scene/Rubble Fields/Rubble Field M
#   Cradle_03 / Boss Scene/Rubble Fields/Rubble Field R

// uses template: silk_boss_rubble_field
fsm FSM {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Antic")
    FindNamedChild(gameObject=Self, storeResult=var "Silk_yank")
    FindNamedChild(gameObject=Self, storeResult=var "Silk_yank (1)")
    FindNamedChild(gameObject=Self, storeResult=var "Silk_yank (2)")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop RockRumble")
    FindNamedChild(gameObject=Self, storeResult=var "Boulder Field")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Boulders")
    FindNamedChild(gameObject=Self, storeResult=var "Field Damager")
    ActivateGameObject(gameObject=var "Silk_yank", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Silk_yank (1)", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Silk_yank (2)", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    ActivateGameObject(gameObject=var "Field Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"ATTACK", realTime=false)  // disabled
    on ATTACK → Set Attack
    on ATTACK LONG → Set Attack Long
  }

  state Silk Yank {
    PlayAudioEvent(audioClip=silk_boss_tendrils_fly_up_and_grab_rock (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, -10, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Silk_yank", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    RandomFloat(min=0.1, max=0.2, storeResult=var "Delay")
    ActivateGameObjectDelay(gameObject=var "Silk_yank (1)", activate=true, resetOnExit=false, delay=var "Delay")
    RandomFloat(min=0.1, max=0.2, storeResult=var "Delay")
    ActivateGameObjectDelay(gameObject=var "Silk_yank (2)", activate=true, resetOnExit=false, delay=var "Delay")
    on FINISHED → Antic
    on ATTACK CLEAR → End
  }

  state Antic {
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=var "Audio Loop RockRumble", volume=1, oneShotClip=<null>)
    on FINISHED → Rock Spawn
    on ATTACK CLEAR → End
  }

  state Rock Spawn {
    SpawnObjectFromGlobalPoolOverTimeV2(gameObject=Silk Boulder Mid (localpoolprefabs_assets_areacradle.bundle), spawnPoint=var "Self", position=(0, 2, 0), originVariationX=6.5, originVariationY=0, rotation=(unset), frequency=0.25, scaleMin=1, scaleMax=1)
    SpawnObjectFromGlobalPoolOverTimeV2(gameObject=Silk Boulder Large (localpoolprefabs_assets_areacradle.bundle), spawnPoint=var "Self", position=(0, 4.5, 0), originVariationX=4, originVariationY=0, rotation=(unset), frequency=0.9, scaleMin=1, scaleMax=1)
    Wait(time=var "Attack Time", finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Boulder Field", ToAlpha=1, FadeTime=0.5)
    PlayParticleEmitterChildren(gameObject=var "Pt Boulders", resetTimeIfPlaying=false, stopOnStateExit=true)
    SetPosition2d(gameObject=var "Field Damager", vector=(unset), x=(unset), y=30, space=1, everyFrame=false, lateUpdate=false)
    ActivateGameObjectDelay(gameObject=var "Field Damager", activate=true, resetOnExit=false, delay=0.1)
    AnimatePositionTo(gameObject=var "Field Damager", toValue=(0, 0, 0), localSpace=true, time=0.3, speed=(unset), delay=0.1, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    on ATTACK CLEAR → End
    on FINISHED → End
  }

  state End {
    StopParticleEmittersInChildren(gameObject=var "Pt Antic")
    AudioStopV2(gameObject=var "Audio Loop RockRumble", fadeTime=0, cancelOnEarlyExit=false)
    FadeNestedFadeGroup(Target=var "Boulder Field", ToAlpha=0, FadeTime=0.5)
    ActivateGameObjectDelay(gameObject=var "Field Damager", activate=false, resetOnExit=false, delay=0.4)
    on FINISHED → Idle
  }

  state Set Attack {
    SetFloatValue(floatVariable=var "Attack Time", floatValue=2.5, everyFrame=false)
    on FINISHED → Silk Yank
  }

  state Set Attack Long {
    SetFloatValue(floatVariable=var "Attack Time", floatValue=5, everyFrame=false)
    on FINISHED → Silk Yank
  }

  var Attack Time: float = 0
  var Delay: float = 0
  var Pt Antic: gameObject = <null>
  var Silk_yank: gameObject = <null>
  var Self: gameObject = <null>
  var Silk_yank (1): gameObject = <null>
  var Silk_yank (2): gameObject = <null>
  var Audio Loop RockRumble: gameObject = <null>
  var Boulder Field: gameObject = <null>
  var Pt Boulders: gameObject = <null>
  var Field Damager: gameObject = <null>
}
