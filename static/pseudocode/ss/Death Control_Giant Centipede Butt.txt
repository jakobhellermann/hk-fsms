# Death Control
# 1 location(s):
#   Bellway_Centipede_Arena / Boss Scene/Giant Centipede Butt

// uses template: centipede_death_control
fsm Death Control {
  start Init
  on FINAL EXPLODE → Pause  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Death Point")
    FindNamedChild(gameObject=Self, storeResult=var "Corpse")
    FindNamedChild(gameObject=Self, storeResult=var "Chunks")
    on FINISHED → Alive
  }

  state Can Die {
    on TOOK DAMAGE → Killable?
  }

  state Alive {
    on CAN DIE → Can Die
  }

  state Killable? {
    BoolTest(boolVariable=var "Killable", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Death Hit V2
    on CANCEL → Can Die
  }

  state Death Hit {
    SendEventToRegisterV2(EventName="CENTIPEDE DYING", ExcludeTarget=Self)
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0)
    SpawnObjectFromGlobalPool(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Death Point", position=(unset), rotation=(unset), storeObject=var "")
    SetFsmBool(gameObject=var "Partner", fsmName="Control", variableName="Centipede Dying", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    on FINISHED → Dying
  }

  state Dying {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Death Anim")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DEATH", delay=0, everyFrame=false)
    on FINAL EXPLODE → Explode
  }

  state Explode {
    SpawnObjectFromGlobalPool(gameObject=Giant Centipede Death Explode v2 (localpoolprefabs_assets_areabellway.bundle), spawnPoint=var "Death Point", position=(unset), rotation=(unset), storeObject=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    EnableFSM(gameObject=Self, fsmName="Control", enable=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Away")
    ActivateGameObject(gameObject=var "Chunks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    Wait(time=var "Explode Pause", finishEvent=(none), realTime=false)
    on FINISHED → Explode
  }

  state Death Hit V2 {
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    SpawnObjectFromGlobalPool(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Death Point", position=(unset), rotation=(unset), storeObject=var "")  // disabled
    SendEventToRegisterV2(EventName="CENTIPEDE DYING", ExcludeTarget=Self)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Leave For Death", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Partner", fsmName="Control", variableName="Leave For Death", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
  }

  var Explode Pause: float = 3
  var Death Anim: string = "Head Spit"
  var Partner: gameObject = <null>

  var Killable: bool = false
  var Death Point: gameObject = <null>
  var Corpse: gameObject = <null>
  var Chunks: gameObject = <null>
}
