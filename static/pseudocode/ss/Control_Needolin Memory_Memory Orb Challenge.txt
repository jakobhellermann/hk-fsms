# Control
# 2 location(s):
#   Dock_11 / Memory Orb Challenge/Needolin Memory
#   Greymoor_15b / Memory Orb Challenge/Needolin Memory

fsm Control {
  start Init

  state Init {
    FadeNestedFadeGroup(Target=var "Silhouettes", ToAlpha=0, FadeTime=0)
    ActivateGameObject(gameObject=var "Finish Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → In Range
  }

  state In Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on EXIT → Idle
    on NEEDOLIN → Needolin Start
  }

  state Needolin Start {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=3, finishEvent=→"NEXT", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Play", emit=0, resetIfPlaying=true)
    FadeNestedFadeGroup(Target=var "Silhouettes", ToAlpha=0.5, FadeTime=3)
    StopParticleEmittersInChildren(gameObject=var "Activated Effects")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="NEEDOLIN PLAYING", delay=0, everyFrame=false)
    on END → Needolin End
    on NEXT → Complete
  }

  state Needolin End {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Silhouettes", ToAlpha=0, FadeTime=0.75)
    StopParticleEmitter(gameObject=var "Pt Play")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="NEEDOLIN STOPPED", delay=0, everyFrame=false)
    on FINISHED → In Range
  }

  state Complete {
    FadeNestedFadeGroup(Target=var "Silhouettes", ToAlpha=1, FadeTime=0.25)
    ActivateGameObject(gameObject=var "Finish Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SendEventByName(eventTarget=FSMComponent, sendEvent="DISPLAY", delay=0, everyFrame=false)
    on END → Final Fade
  }

  state Final Fade {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Silhouettes", ToAlpha=0, FadeTime=0.75)
    StopParticleEmitter(gameObject=var "Pt Play")
    SetFsmBool(gameObject=Self, fsmName="Needolin Msg", variableName="Display End", setValue=true, everyFrame=false)
    on FINISHED → UI Msg
  }

  state UI Msg {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SavedItemGet(Item=Attunement Increase (dataassets_assets_assets/dataassets/collectables/fakecollectables.bundle))
  }

  var Activated: bool = false
  var Activated Effects: gameObject = Memory Orb Challenge/Needolin Memory/Activated Effects
  var Finish Effects: gameObject = Memory Orb Challenge/Needolin Memory/Finish Effects
  var Pt Play: gameObject = Memory Orb Challenge/Needolin Memory/Pt Play
  var Silhouettes: gameObject = Memory Orb Challenge/Needolin Memory/Silhouettes
  var Parent: gameObject = <null>
}
