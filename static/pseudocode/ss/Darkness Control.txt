# Darkness Control
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/Vignette

fsm Darkness Control {
  start Pause
  on DAMAGE → Damage  // from any state
  on DEATH → Death  // from any state
  on HERO RESPAWNED → Recover  // from any state
  on RESET → Pause  // from any state
  on SCENE RESET → Hazard Death?  // from any state
  on VIGNETTE OPEN SLOW → Open Slow  // from any state

  state Init {
    FindGameObject(objectName="", withTag="SceneManager", store=var "Scene Manager")
    on FINISHED → Dark Lev Check
  }

  state Idle {
    SetPlayerDataInt(intName="previousDarkness", value=var "Darkness Level")
  }

  state Damage {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Damage Scale", time=0.2, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    AnimateScaleTo(Target=Self, ToLocalScale=var "Damage Scale", time=0.2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Damage Pause
  }

  state Damage Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recover
  }

  state Recover {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Idle Scale", time=1, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    AnimateScaleTo(Target=Self, ToLocalScale=var "Idle Scale", time=1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Death {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Damage Scale", time=2, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    AnimateScaleTo(Target=Self, ToLocalScale=var "Damage Scale", time=2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
  }

  state Pause {
    SetSpriteRenderer(gameObject=Self, active=true)
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Lantern Glow", storeResult=var "Lantern Glow")
    FindChild(gameObject=Self, childName="Darkness Border", storeResult=var "Darkness Border")
    ActivateGameObject(gameObject=var "Darkness Border", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FindChild(gameObject=Self, childName="Darkness Plates", storeResult=var "Darkness Plates")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Dark Lev Check {
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    IntSwitch(intVariable=var "Darkness Level", compareTo=[0, -1], sendEvent=[→"NORMAL", →"DARK -1"], everyFrame=false)
    PlayerDataBoolTest(boolName="hasLantern", isTrue=→"LANTERN", isFalse=(none))  // disabled
    IntSwitch(intVariable=var "Darkness Level", compareTo=[1, 2], sendEvent=[→"DARK 1", →"DARK 2"], everyFrame=false)
    on DARK 1 → Dark 1
    on DARK 2 → Dark 2
    on NORMAL → Normal
    on LANTERN → Lantern
    on DARK -1 → Dark -1
  }

  state Normal {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=5.5, y=5.5, z=5.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=4, y=4, z=4, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Set {
    SetScale(gameObject=Self, vector=var "Idle Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Lantern Glow", vector=var "Idle Scale", x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Dark 1 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=2.7, y=2.7, z=2.7, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=2.4, y=2.4, z=2.4, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Dark 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=1.8, y=1.8, z=1.8, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=1.6, y=1.6, z=1.6, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Plates", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → Set
  }

  state Lantern {
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=3, y=3, z=3, everyFrame=false)
    ActivateGameObject(gameObject=var "Lantern Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=2.2, y=2.2, z=2.2, everyFrame=false)
    on FINISHED → Previously Dark?
  }

  state Previously Dark? {
    SetScale(gameObject=var "Lantern Glow", vector=(unset), x=1, y=1, z=1, everyFrame=false, lateUpdate=false)
    GetPlayerDataInt(intName="previousDarkness", storeValue=var "Previous Dark Level")
    IntCompare(integer1=var "Previous Dark Level", integer2=0, equal=(none), lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    IntSwitch(intVariable=var "Darkness Level", compareTo=[1, 2], sendEvent=[→"DARK 1", →"DARK 2"], everyFrame=false)
    on FINISHED → Set
    on DARK 1 → Dark 1 Start
    on DARK 2 → Dark 2 Start
  }

  state Dark 1 Start {
    SetScale(gameObject=Self, vector=var "Idle Scale", x=2.7, y=2.7, z=2.7, everyFrame=false, lateUpdate=false)
    on FINISHED → Lantern Shrink
  }

  state Lantern Up {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Idle Scale", time=4, delay=0.5, speed=(unset), easeType=13, loopType=0, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    AnimateScaleTo(Target=Self, ToLocalScale=var "Idle Scale", time=4, speed=(unset), delay=0.5, easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    iTweenScaleTo(gameObject=var "Lantern Glow", id="", transformScale=var "", vectorScale=(3, 3, 3), time=4, delay=0.5, speed=(unset), easeType=13, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    AnimateScaleTo(Target=var "Lantern Glow", ToLocalScale=(3, 3, 3), time=4, speed=(unset), delay=0.5, easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Darkness Plates"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Dark 2 Start {
    SetScale(gameObject=Self, vector=var "Idle Scale", x=1.8, y=1.8, z=1.8, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Lantern Shrink
  }

  state Scene Reset {
    FindGameObject(objectName="", withTag="SceneManager", store=var "Scene Manager")
    IntSwitch(intVariable=var "Darkness Level", compareTo=[0, -1], sendEvent=[→"NORMAL", →"DARK -1"], everyFrame=false)
    PlayerDataBoolTest(boolName="hasLantern", isTrue=→"LANTERN", isFalse=(none))  // disabled
    IntSwitch(intVariable=var "Darkness Level", compareTo=[1, 2], sendEvent=[→"DARK 1", →"DARK 2"], everyFrame=false)
    on NORMAL → Normal 2
    on DARK 1 → Dark 1 2
    on DARK 2 → Dark 2 2
    on LANTERN → Lantern 2
    on DARK -1 → Dark -1 2
  }

  state Normal 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=5.5, y=5.5, z=5.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=4, y=4, z=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Dark 1 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=2.7, y=2.7, z=2.7, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=2.4, y=2.4, z=2.4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Dark 2 2 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=1.8, y=1.8, z=1.8, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=1.6, y=1.6, z=1.6, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Lantern 2 {
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=3, y=3, z=3, everyFrame=false)
    ActivateGameObject(gameObject=var "Lantern Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=2.2, y=2.2, z=2.2, everyFrame=false)
    on FINISHED → Lantern Up
  }

  state Lantern Shrink {
    SetScale(gameObject=var "Lantern Glow", vector=(unset), x=0.1, y=0.1, z=0.1, everyFrame=false, lateUpdate=false)
    on FINISHED → Lantern Up
  }

  state Dark -1 {
    ActivateGameObject(gameObject=var "Lantern Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=10.5, y=10.5, z=10.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=5, y=5, z=5, everyFrame=false)
    ActivateGameObject(gameObject=var "Darkness Plates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Dark -1 2 {
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=10.5, y=10.5, z=10.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=5, y=5, z=5, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lantern Glow"), sendEvent="SHRINK", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Hazard Death? {
    GetLayer(gameObject=var "Parent", storeResult=var "Hero Layer", everyFrame=false)
    IntCompare(integer1=var "Hero Layer", integer2=9, equal=→"FINISHED", lessThan=→"CANCEL", greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → In Hazard Death
    on FINISHED → Scene Reset
  }

  state In Hazard Death {
  }

  state Open Slow {
    SetVector3XYZ(vector3Variable=var "Idle Scale", vector3Value=(unset), x=5.5, y=5.5, z=5.5, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Damage Scale", vector3Value=(unset), x=4, y=4, z=4, everyFrame=false)
    AnimateScaleTo(Target=Self, ToLocalScale=var "Idle Scale", time=3, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Darkness Level: int = 0
  var Previous Dark Level: int = 0
  var Hero Layer: int = 0
  var Darkness String: string = ""
  var Damage Scale: vector3 = (4, 4, 4)
  var Idle Scale: vector3 = (5, 5, 5)
  var Darkness Border: gameObject = <null>
  var Darkness Plates: gameObject = <null>
  var Lantern Glow: gameObject = <null>
  var Parent: gameObject = <null>
  var Scene Manager: gameObject = <null>
}
