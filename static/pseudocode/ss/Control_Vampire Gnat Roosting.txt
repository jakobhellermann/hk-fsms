# Control
# 1 location(s):
#   Greymoor_05 / Scene Control/Vampire Gnat Roosting

fsm Control {
  start Pause

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Perch")
    PlayerDataBoolTest(boolName="encounteredVampireGnat_05", isTrue=→"ACTIVATED", isFalse=(none))
    PlayerDataVariableTest(VariableName="VampireGnatBossInAltLoc", ExpectedValue=true, IsExpectedEvent=→"ACTIVATED", IsNotExpectedEvent=(none))
    PlayerDataBoolTest(boolName="encounteredVampireGnatBoss", isTrue=→"ACTIVATED", isFalse=(none))
    on FINISHED → Idle
    on ACTIVATED → Activated
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    SendEventToRegisterDelay(EventName="MOORWING ROOSTING", delay=0.5, ExcludeTarget=Self)
    on ENTER → Leave Pause
  }

  state Leave Antic {
    SetPlayerDataBool(boolName="encounteredVampireGnat_05", value=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayAudioEvent(audioClip=moorwing_flap_1 (sfxstatic_assets_vampiregnat.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -18), SpawnedPlayerRef=var "")
    on FINISHED → Zoom
  }

  state Zoom {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly Away")
    AnimatePositionTo(gameObject=Self, toValue=(0, 71, 18.38), localSpace=false, time=1.5, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=0, toValue=-10, floatVariable=var "Angle", time=0.5, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=(none), realTime=false)
    SetRotation(gameObject=var "Perch", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=true, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt LaunchDust", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt LaunchRag", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt JumpTrail", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=moorwing_dive_attack (sfxstatic_assets_vampiregnat.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -18), SpawnedPlayerRef=var "")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    PlayAudioEvent(audioClip=arrow_sign_first_hit (sfxstatic_assets_signs.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -18), SpawnedPlayerRef=var "")
    Wait(time=3.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetMeshRenderer(gameObject=Self, active=false)
    SendEventToRegister(eventName="MOORWING LEFT")
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activated {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Leave Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Leave Antic
  }

  var Angle: float = 0
  var Parent: gameObject = <null>
  var Perch: gameObject = <null>
  var Pt LaunchDust: gameObject = Scene Control/Vampire Gnat Roosting/Pt LaunchDust
  var Pt LaunchRag: gameObject = Scene Control/Vampire Gnat Roosting/Pt LaunchRag
  var Pt JumpTrail: gameObject = Scene Control/Vampire Gnat Roosting/Pt JumpTrail
}
