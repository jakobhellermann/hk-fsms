# unmasker
# 1 location(s):
#   Bonegrave / Col_Glow_Remasker

// uses template: remasker_inverse
fsm unmasker {
  start Pause
  on HERO LEAVE → Hero Leave  // from any state

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="Player", sendEvent=→"UNCOVER", storeCollider=var "")
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=0.01, delay=0, easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on UNCOVER → Fade Out
  }

  state Fade Out {
    iTweenFadeTo(gameObject=Self, id="", alpha=1, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="Player", sendEvent=→"COVER", storeCollider=var "")
    on COVER → Fade In
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Fade In {
    iTweenFadeTo(gameObject=Self, id="", alpha=0, includeChildren=true, namedValueColor="_Color", time=var "Fade Time", delay=0, easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="Player", sendEvent=→"UNCOVER", storeCollider=var "")
    on UNCOVER → Fade Out
  }

  state Hero Leave {
  }

  var Fade Time: float = 1.2

  var Activated: bool = false
}
