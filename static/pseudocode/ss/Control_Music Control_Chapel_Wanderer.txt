# Control
# 1 location(s):
#   Chapel_Wanderer / Music Control

fsm Control {
  start Pause
  on CHAPEL MUSIC STOP → Inactive  // from any state

  state Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Up
  }

  state Fade Up {
    GetIsCrestUnlocked(Crest=Wanderer (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"INACTIVE", falseEvent=(none), storeValue=(unset))
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)  // disabled
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=2)  // disabled
    ApplyMusicCue(musicCue=Chapel (audiocuesdynamic_assets_candle.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Main Only (audioobjects_assets_all.bundle), transitionTime=2)
    on CREST STAB START → Fade Down
    on INACTIVE → Inactive
    on LEAVING SCENE → Leave Scene
  }

  state Fade Down {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=2)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=2)
  }

  state Inactive {
  }

  state Leave Scene {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=1)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=1)
  }
}
