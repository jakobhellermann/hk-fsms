# Noise React
# 1 location(s):
#   Slab_16 / Event Control/Visiting Cloakless Scene/Kit Guard Needle

fsm Noise React {
  start Init

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Kit Guard Needle")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=(none), ActiveOuter=→"NOISE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    WaitRandom(timeMin=1, timeMax=3, finishEvent=→"TALK", realTime=false)
    on NOISE → Startle
    on TALK → Talk
  }

  state Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Kit Guard Needle Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=10)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=SF_hop_large (sfxdynamic_assets_areagreymoorareaslabareaswampareawilds.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Idle
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    on FINISHED → Idle
  }

  state Talk {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=SF_angry_talk_large (sfxdynamic_assets_areaslab.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=(none), ActiveOuter=→"NOISE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Idle
    on NOISE → Startle
  }

  var Audio Loop Voice: gameObject = <null>
}
