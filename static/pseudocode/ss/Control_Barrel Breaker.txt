# Control
# 1 location(s):
#   Clover_10_web / Return Ambush/Ambush/Barrel Breaker

fsm Control {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Antic
  }

  state Antic {
    ObjectJitter(gameObject=var "Barrel", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Break
  }

  state Break {
    SendMessage(gameObject=var "Barrel", delivery=SendMessage, options=1, functionCall=BreakSelf())
  }

  var Barrel: gameObject = Return Ambush/Ambush/Grove_barrel return ambush (21)
}
