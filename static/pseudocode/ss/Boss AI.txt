# Boss AI
# 1 location(s):
#   Coral_11 / Black Thread States Thread Only Variant/Normal World/Boss Scene

fsm Boss AI {
  start Init
  on STOP AI → Stop  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Driller A")
    FindNamedChild(gameObject=Self, storeResult=var "Driller B")
    on FINISHED → Dormant
  }

  state Dormant {
    on START → Start Pause
  }

  state Start Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Double V
  }

  state Choice {
    SendEvent(eventTarget=Self, sendEvent=→"STOMP SHOOT", delay=0, everyFrame=false)  // disabled
    GetFsmFloat(gameObject=Self, fsmName="needolin_capture_timer", variableName="Needolin Timer", storeValue=var "Needolin Timer", everyFrame=false)
    FloatCompare(float1=var "Needolin Timer", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SING", everyFrame=false)
    SendRandomEventV4(events=[→"H THEN V", →"V THEN H", →"DOUBLE H", →"DOUBLE V", →"SHOOT", →"STOMP", →"STOMP SHOOT"], weights=[0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5], eventMax=[1, 1, 1, 1, 1, 1, 1], missedMax=[7, 7, 7, 7, 6, 6, 6], activeBool=(unset))
    on H THEN V → H Then V
    on V THEN H → V Then H
    on DOUBLE H → Double H
    on DOUBLE V → Double V
    on SHOOT → Shoot
    on STOMP → Stomp
    on STOMP SHOOT → Stomp Shoot
    on SING → Sing
  }

  state H Then V {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="ATTACK H", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="ATTACK V", delay=0.5, everyFrame=false)
    Wait(time=1.8, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Driller A", audioClips=[driller_boss_misc_scream_1 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_3 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_4 (sfxstatic_assets_areacoral.bundle), <null>], weights=[1, 1, 1, 0], pitchMin=0.8, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    on FINISHED → Idle Pause
  }

  state V Then H {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="ATTACK V", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="ATTACK H", delay=0.5, everyFrame=false)
    Wait(time=1.8, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Driller A", audioClips=[driller_boss_misc_scream_1 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_3 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_4 (sfxstatic_assets_areacoral.bundle), <null>], weights=[1, 1, 1, 0], pitchMin=0.8, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    on FINISHED → Idle Pause
  }

  state Double H {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="ATTACK HT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="ATTACK HB", delay=0.1, everyFrame=false)
    Wait(time=1.7, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Driller A", audioClips=[driller_boss_misc_scream_1 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_3 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_4 (sfxstatic_assets_areacoral.bundle), <null>], weights=[1, 1, 1, 0], pitchMin=0.8, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    on FINISHED → Idle Pause
  }

  state Double V {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="ATTACK VL", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="ATTACK VR", delay=0.1, everyFrame=false)
    Wait(time=1.7, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShot(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Driller A", audioClips=[driller_boss_misc_scream_1 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_3 (sfxstatic_assets_areacoral.bundle), driller_boss_misc_scream_4 (sfxstatic_assets_areacoral.bundle), <null>], weights=[1, 1, 1, 0], pitchMin=0.8, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)
    on FINISHED → Idle Pause
  }

  state Shoot {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="SHOOT", delay=0, everyFrame=false)
    Wait(time=4.2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="ATTACK V", delay=2.5, everyFrame=false)
    on FINISHED → Idle Pause
  }

  state Stomp {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="STOMP", delay=0, everyFrame=false)
    Wait(time=4.2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="ATTACK H", delay=2.5, everyFrame=false)
    on FINISHED → Idle Pause
  }

  state Stomp Shoot {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="STOMP", delay=0, everyFrame=false)
    Wait(time=6.2, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="SHOOT", delay=2.5, everyFrame=false)
    on FINISHED → Idle Pause
  }

  state Stop {
  }

  state Idle Pause {
    GetHP(target=var "Driller A", storeValue=var "HP remaining")
    IntTestToBool(int1=var "HP remaining", int2=180, equalBool=(unset), lessThanBool=var "P2", greaterThanBool=(unset), everyFrame=false)
    ConvertBoolToFloat(boolVariable=var "P2", floatVariable=var "Idle Time", falseValue=0.75, trueValue=0.1, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait For Ready
  }

  state Wait For Ready {
    GetFsmBool(gameObject=var "Driller A", fsmName="Control", variableName="Ready", storeValue=var "Driller A Ready", everyFrame=true)
    GetFsmBool(gameObject=var "Driller B", fsmName="Control", variableName="Ready", storeValue=var "Driller B Ready", everyFrame=true)
    BoolAllTrue(boolVariables=[var "Driller A Ready", var "Driller B Ready"], sendEvent=→"FINISHED", storeResult=(unset), everyFrame=true)
    on FINISHED → Choice
  }

  state Sing {
    SendEventByName(eventTarget=GameObject(var "Driller A"), sendEvent="SING L", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Driller B"), sendEvent="SING R", delay=0, everyFrame=false)
    GetFsmFloat(gameObject=Self, fsmName="needolin_capture_timer", variableName="Needolin Timer", storeValue=var "Needolin Timer", everyFrame=true)
    FloatCompare(float1=var "Needolin Timer", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    WaitRandom(timeMin=4, timeMax=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Sing Pause
  }

  state End Sing {
    SetFsmBool(gameObject=var "Driller A", fsmName="Control", variableName="End Sing", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Driller B", fsmName="Control", variableName="End Sing", setValue=true, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Capture Reset
  }

  state End Sing Pause {
    Wait(time=0.3, finishEvent=(none), realTime=false)
    on FINISHED → End Sing
  }

  state Capture Reset {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="NEEDOLIN CAPTURE COOLDOWN", delay=0, everyFrame=false)
    on FINISHED → Wait For Ready
  }

  var Idle Time: float = 0
  var Needolin Timer: float = 0
  var HP remaining: int = 0
  var P2: bool = false
  var Driller A Ready: bool = false
  var Driller B Ready: bool = false
  var Driller A: gameObject = <null>
  var Driller B: gameObject = <null>
}
