# Bell State
# 1 location(s):
#   Shadow_18 / Bellshrine Sequence Swamp

fsm Bell State {
  start Pause

  state Idle {
    ActivateInteractible(Target=var "RestBench", Activate=false, AllowQueueing=false, UseChildren=false)
    FadeNestedFadeGroup(Target=var "RestBench", ToAlpha=0, FadeTime=0)
    on BATTLE COMPLETED → Already Completed
    on BATTLE END → Completed Sequence
  }

  state Pause {
    ActivateGameObject(gameObject=var "Lever Plat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bench Pieces", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Already Completed {
    CallMethodProper(gameObject=var "Battle Scene", behaviour="BattleScene", methodName="BattleCompleted", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Lever Plat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=var "Lever Plat", stateName="Appear", layer=0, normalizedTime=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Bench Pieces", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=var "Bench Pieces", stateName="Appear", layer=0, normalizedTime=1, everyFrame=false)
    FadeNestedFadeGroup(Target=var "RestBench", ToAlpha=1, FadeTime=0)
    ActivateInteractible(Target=var "RestBench", Activate=true, AllowQueueing=false, UseChildren=false)
  }

  state Completed Sequence {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Plat Appear
  }

  sequence state Plat Appear {
    PlayAudioEvent(audioClip=bone_cogs_move (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlaySimple(gameObject=var "Audio Loop Bubbling", volume=1, oneShotClip=<null>)
    PlayParticleEmitter(gameObject=var "Plat Antic Pt", emit=0, resetIfPlaying=true)
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    AudioStop(gameObject=var "Audio Loop Bubbling", fadeTime=0)
    PlayAudioEvent(audioClip=big splash (sfxstatic_assets_water.bundle), pitchMin=1.2, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    StopParticleEmitter(gameObject=var "Plat Antic Pt")
    ActivateGameObject(gameObject=var "Lever Plat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlayStateWait(target=var "Lever Plat", stateName="Appear", finishEvent=(none))
    Wait(time=3, finishEvent=(none), realTime=false)
    on FINISHED → Bench Appear
  }

  sequence state Bench Appear {
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    PlayAudioEvent(audioClip=bone_cogs_move (sfxstatic_assets_areabone.bundle), pitchMin=1.25, pitchMax=1.25, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player")
    Wait(time=1.5, finishEvent=(none), realTime=false)
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    ActivateGameObject(gameObject=var "Bench Pieces", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStop(gameObject=var "Audio Player", fadeTime=0)
    PlayAudioEvent(audioClip=bonetown_lift_arrive (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=0.25, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.35, finishEvent=(none), realTime=false)
    PlayAudioEvent(audioClip=bell_bench_open (sfxstatic_assets_bench.bundle), pitchMin=1, pitchMax=1, volume=1.25, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=(none), realTime=false)
    FadeNestedFadeGroup(Target=var "RestBench", ToAlpha=1, FadeTime=2)
    ActivateInteractible(Target=var "RestBench", Activate=true, AllowQueueing=false, UseChildren=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → BG Open
  }

  sequence state BG Open {
    SaveGame()
    Wait(time=1, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "CamLock BellSequence", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Battle Scene", behaviour="BattleScene", methodName="SendEventToChildren", parameters=["BG OPEN"], storeResult=(unset var), EveryFrame=false)
  }

  var Battle Scene: gameObject = Battle Scene
  var RestBench: gameObject = RestBench

  var Lever Plat: gameObject = Bellshrine Sequence Swamp/swamp_bell_bench_plat
  var Bench Pieces: gameObject = Bellshrine Sequence Swamp/bench_pieces
  var Plat Antic Pt: gameObject = Bellshrine Sequence Swamp/Pt Bubble
  var Audio Loop Bubbling: gameObject = Bellshrine Sequence Swamp/Audio Loop Bubbling
  var Audio Player: gameObject = <null>
  var CamLock BellSequence: gameObject = Bellshrine Sequence Swamp/CamLock BellSequence
}
