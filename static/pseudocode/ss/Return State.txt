# Return State
# 1 location(s):
#   Bone_05_boss / Boss Scene

fsm Return State {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="encounteredBellBeast", isTrue=→"RETURN", isFalse=(none))
    on RETURN → Set Return State
  }

  state Set Return State {
    FindChild(gameObject=Self, childName="Thick Silk Vines Beast", storeResult=var "Thick Silk Vines Beast")
    FindChild(gameObject=Self, childName="Return Battle", storeResult=var "Return Battle")
    ActivateGameObject(gameObject=var "Return Battle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thick Silk Vines Beast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Return Battle: gameObject = <null>
  var Thick Silk Vines Beast: gameObject = <null>
}
