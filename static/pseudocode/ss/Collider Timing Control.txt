# Collider Timing Control
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/Attacks/Scythe/DashUpper Slash/Extra Damager

fsm Collider Timing Control {
  start Collider Timing

  sequence state Collider Timing {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    Wait(time=0.1, finishEvent=(none), realTime=false)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
  }
}
