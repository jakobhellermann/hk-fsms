# Control
# 1 location(s):
#   enemycorpses_assets_areadocks.bundle / Corpse Tar Slug Expl

fsm Control {
  start Pause

  state Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashingTarSlug())
    PlayAudioEvent(audioClip=tar_slug_death_antic (sfxstatic_assets_areadocks.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.85, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Explode
  }

  state Explode {
    ActivateGameObject(gameObject=var "Explosion", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pop", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelFlash())
    PlayAudioEvent(audioClip=swamp_bounce_pod_explode (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=enemy_death_sword (sfxstatic_assets_shared.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=small_fireball_projectile_death (sfxstatic_assets_areabone.bundle), pitchMin=0.95, pitchMax=1.05, volume=0.5, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetMeshRenderer(gameObject=Self, active=false)
  }

  var Pt Antic: gameObject = Corpse Tar Slug Expl/Pt Antic
  var Explosion: gameObject = Corpse Tar Slug Expl/Explosion
}
