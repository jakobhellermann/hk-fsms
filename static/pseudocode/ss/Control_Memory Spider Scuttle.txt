# Control
# 1 location(s):
#   Memory_First_Sinner / Memory Spider Scuttle

fsm Control {
  start Pause

  state Idle {
    Trigger2dEvent(gameObject=var "Retreat Range", trigger=0, collideTag="", sendEvent=→"RETREAT", storeCollider=var "")
    on RETREAT → Retreat
    on PROSTRATE RISE STAND → Retreat Pause
  }

  state Retreat {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    PlayAudioEventRandom(audioClips=[blade_spider_hang_misc_1 (sfxstatic_assets_areacoral.bundle), blade_spider_hang_misc_2 (sfxstatic_assets_areacoral.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    PlayRandomAudioClipTable(Table=Spider_alert (sfxdynamic_assets_areacoral.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="TurnToWalk")
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Retreat Angle", speed=12, everyFrame=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    RayCast2dV2(fromGameObject=var "Retreat Ray", fromPosition=(unset), direction=(0, -1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"FINISHED", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Disappear
  }

  state Disappear {
    Tk2dSpriteTweenColor(gameObject=Self, color=(0, 0, 0, 0), duration=0.3)
    PlayParticleEmitter(gameObject=var "Pt Disappear", emit=0, resetIfPlaying=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    AudioStopV2(gameObject=Self, fadeTime=0.2, cancelOnEarlyExit=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Shuffle Z
  }

  state Shuffle Z {
    RandomFloat(min=-0.001, max=0.001, storeResult=var "Z Translate")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Translate", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Direction Marker", offsetX=0, offsetY=0, storeAngle=var "Retreat Angle", pause=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Retreat Pause {
    WaitRandom(timeMin=0.4, timeMax=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Retreat
  }

  var Z Translate: float = 0
  var Retreat Angle: float = 0
  var Retreat Range: gameObject = Memory Spider Scuttle/Retreat Range
  var Pt Disappear: gameObject = Memory Spider Scuttle/Pt Disappear
  var Direction Marker: gameObject = Memory Spider Scuttle/Direction Marker
  var Retreat Ray: gameObject = Memory Spider Scuttle/Retreat Ray
}
