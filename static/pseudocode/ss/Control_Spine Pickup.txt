# Control
# 1 location(s):
#   localpoolprefabs_assets_areawilds.bundle / Spine Pickup

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Spawn Replacement
  }

  state Recycle {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RecycleSelf()
  }

  state Spawn Replacement {
    SpawnObjectFromGlobalPool(gameObject=Collectable Item Pickup Instant (localpoolprefabs_assets_currency.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Spawned Pickup")
    SetCollectablePickupItem(Target=var "Spawned Pickup", Item=Common Spine (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle))
    SetPosition2D(GameObject=Self, Vector=(unset), X=0, Y=0, Space=1, EveryFrame=false)
    GetVelocity2d(gameObject=var "Self", vector=var "Velocity", x=(unset), y=(unset), space=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Spawned Pickup", vector=var "Velocity", x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Recycle
  }

  var Velocity: vector2 = (0, 0)
  var Self: gameObject = <null>
  var Spawned Pickup: gameObject = <null>
}
