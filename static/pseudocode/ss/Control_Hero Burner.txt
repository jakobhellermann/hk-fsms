# Control
# 1 location(s):
#   Mosstown_02c / moss_hut_set/NPC Scene/moss pot_large/Hero Burner

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Pt Burn")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Burn")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Burn Pause
  }

  state Burn Pause {
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXIT", storeCollider=var "")
    PlayAudioEvent(audioClip=acid_hiss (sfxstatic_assets_shared.bundle), pitchMin=0.85, pitchMax=1.15, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Burn Time
    on EXIT → Idle
  }

  state Burn Effect {
    CallMethodProper(gameObject=var "Hero", behaviour="SpriteFlash", methodName="FlashingAcid", parameters=[], storeResult=(unset var), EveryFrame=false)
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXIT", storeCollider=var "")
    Wait(time=var "Burn Time", finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Burn", resetTimeIfPlaying=false, stopOnStateExit=true)
    AudioPlaySimpleV2(gameObject=var "Audio Loop Burn", volume=(unset), startTime=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Loop Burn", startVolume=0.25, endVolume=1, time=1)
    on EXIT → End Effect
    on FINISHED → Damage
  }

  state End Effect {
    CallMethodProper(gameObject=var "Hero", behaviour="SpriteFlash", methodName="CancelFlash", parameters=[], storeResult=(unset var), EveryFrame=false)
    FadeAudio(gameObject=var "Audio Loop Burn", startVolume=(unset), endVolume=0, time=0.25)
    on FINISHED → Idle
  }

  state Damage {
    PlayAudioEvent(audioClip=acid_splash (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioStopV2(gameObject=var "Audio Loop Burn", fadeTime=0, cancelOnEarlyExit=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXIT", storeCollider=var "")
    on FINISHED → Damage End
    on EXIT → Damage End
  }

  state Damage End {
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on FINISHED → End Effect
  }

  state Set Burn Time {
    CheckIfToolEquipped(Tool=Mosscreep Tool 1 (dataassets_assets_assets/dataassets/tools/toolitems.bundle), RequiredAmountLeft=(unset), trueEvent=→"TOOL 1", falseEvent=(none), storeValue=(unset))
    CheckIfToolEquipped(Tool=Mosscreep Tool 2 (dataassets_assets_assets/dataassets/tools/toolitems.bundle), RequiredAmountLeft=(unset), trueEvent=→"TOOL 2", falseEvent=(none), storeValue=(unset))
    SetFloatValue(floatVariable=var "Burn Time", floatValue=1, everyFrame=false)
    on FINISHED → Burn Effect
    on TOOL 1 → Tool 1
    on TOOL 2 → Tool 2
  }

  state Tool 1 {
    SetFloatValue(floatVariable=var "Burn Time", floatValue=1.4, everyFrame=false)
    on FINISHED → Burn Effect
  }

  state Tool 2 {
    SetFloatValue(floatVariable=var "Burn Time", floatValue=1.7, everyFrame=false)
    on FINISHED → Burn Effect
  }

  var Burn Time: float = 0
  var Parent: gameObject = <null>
  var Pt Burn: gameObject = <null>
  var Audio Loop Burn: gameObject = <null>
}
