# Detect
# 1 location(s):
#   Bone_05_boss / Boss Scene/Thick Silk Vines Beast/Special Detector

fsm Detect {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Special Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")
    on HIT → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="DESTROY", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  var Parent: gameObject = <null>
}
