# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Roar Spikes

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Spike Holder 1")
    FindNamedChild(gameObject=Self, storeResult=var "Spike Holder 2")
    FindNamedChild(gameObject=Self, storeResult=var "Spike Holder 3")
    FindNamedChild(gameObject=Self, storeResult=var "Spike Holder 4")
    on FINISHED → Idle
  }

  state Idle {
    on SPIKE UP → Pause
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choose Ptn
  }

  state Do Spikes {
    GetRandomChild(gameObject=Self, storeResult=var "Spike Holder")  // disabled
    SendEventByName(eventTarget=GameObject(var "Spike Holder"), sendEvent="SPIKE UP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Choose Ptn {
    SendRandomEventV4(events=[→"1", →"2", →"3", →"4"], weights=[1, 1, 1, 0], eventMax=[1, 1, 1, 1], missedMax=[4, 4, 4, 4], activeBool=(unset))
    on 1 → Ptn 1
    on 2 → Ptn 2
    on 3 → Ptn 3
    on 4 → Ptn 4
  }

  state Ptn 1 {
    SetGameObject(variable=var "Spike Holder", gameObject=var "Spike Holder 1", everyFrame=false)
    on FINISHED → Do Spikes
  }

  state Ptn 2 {
    SetGameObject(variable=var "Spike Holder", gameObject=var "Spike Holder 1", everyFrame=false)
    on FINISHED → Do Spikes
  }

  state Ptn 3 {
    SetGameObject(variable=var "Spike Holder", gameObject=var "Spike Holder 3", everyFrame=false)
    on FINISHED → Do Spikes
  }

  state Ptn 4 {
    SetGameObject(variable=var "Spike Holder", gameObject=var "Spike Holder 4", everyFrame=false)
    on FINISHED → Do Spikes
  }

  var Spike Holder: gameObject = <null>
  var Spike Holder 1: gameObject = <null>
  var Spike Holder 2: gameObject = <null>
  var Spike Holder 3: gameObject = <null>
  var Spike Holder 4: gameObject = <null>
}
