# Camera Control
# 1 location(s):
#   Bone_East_08 / Boss Scene

fsm Camera Control {
  start Init
  on CAM END → Cam End  // from any state
  on FINAL PHASE → Set Final Phase  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "")
    on CAM START → Get
  }

  state Get {
    FindGameObject(objectName="", withTag="CameraTarget", store=var "CamTarget")
    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=1, functionCall=StartFreeMode(false))
    on FINISHED → Bot Lock
  }

  state Bot Lock {
    AnimatePositionTo(gameObject=var "CamTarget", toValue=(81.24, 11.26, 0), localSpace=false, time=(unset), speed=5, delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Custom Cam Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    on FINAL PHASE → Set Final Phase
    on EXIT → Range Out
  }

  state Rock Lock {
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Target Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Target Y", float2=12.53, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)  // disabled
    FloatClamp(floatVariable=var "Target Y", minValue=11.26, maxValue=23.25, everyFrame=true)
    SetPosition(gameObject=var "CamTarget", vector=(unset), x=81.24, y=var "Target Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Custom Cam Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    on FINISHED → State 1
    on EXIT → Range Out
  }

  state State 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Rock Lock
  }

  state Cam End {
    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=1, functionCall=EndFreeMode())
  }

  state Set Final Phase {
    SetBoolValue(boolVariable=var "Final Phase", boolValue=true, everyFrame=false)
    on FINISHED → Rock Lock
  }

  state Range Out {
    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=1, functionCall=EndFreeMode())
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Custom Cam Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Restart
  }

  state Restart {
    SendMessage(gameObject=var "CamTarget", delivery=SendMessage, options=1, functionCall=StartFreeMode(false))
    BoolTest(boolVariable=var "Final Phase", isTrue=→"FINAL PHASE", isFalse=→"FINISHED", everyFrame=false)
    on FINAL PHASE → Rock Lock
    on FINISHED → Bot Lock
  }

  var Target Y: float = 0
  var Final Phase: bool = false
  var CamTarget: gameObject = <null>
}
