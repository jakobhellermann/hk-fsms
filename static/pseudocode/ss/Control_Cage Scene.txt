# Control
# 1 location(s):
#   Dust_01 / Cage Scene

fsm Control {
  start Idle

  state Idle {
    FindChild(gameObject=Self, childName="Dustroach", storeResult=var "Dustroach")
    FindChild(gameObject=Self, childName="Sleeper", storeResult=var "Sleeper")
    on BREAK → Break
    on BROKEN → Broken
  }

  state Broken {
    ActivateGameObject(gameObject=var "Dustroach", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sleeper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Break {
    ActivateGameObject(gameObject=var "Dustroach", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sleeper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wake
  }

  state Wake {
    SendEventByName(eventTarget=GameObject(var "Dustroach"), sendEvent="WAKE", delay=0, everyFrame=false)
  }

  var Dustroach: gameObject = <null>
  var Sleeper: gameObject = <null>
}
