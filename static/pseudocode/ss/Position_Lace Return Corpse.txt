# Position
# 1 location(s):
#   Song_Tower_01 / Boss Scene/Lace Return Corpse

fsm Position {
  start Pause

  state Pause {
    FindNamedChild(gameObject=Self, storeResult=var "Lace Corpse")
    FindNamedChild(gameObject=var "Lace Corpse", storeResult=var "Added Effects")
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Added Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Set
  }

  state Set {
    GetPlayerDataFloat(gameObject=var "GameManager", floatName="laceCorpseScaleX", storeValue=var "Scale X")
    GetPlayerDataFloat(gameObject=var "GameManager", floatName="laceCorpsePosX", storeValue=var "Pos X")
    SetScale(gameObject=var "Lace Corpse", vector=(unset), x=var "Scale X", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetPosition2d(gameObject=var "Lace Corpse", vector=(unset), x=var "Pos X", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition2d(gameObject=var "Added Effects", vector=(unset), x=var "Pos X", y=(unset), space=0, everyFrame=false, lateUpdate=false)  // disabled
    on FINISHED → Added Effects?
  }

  state Added Effects? {
    PlayerDataBoolTest(boolName="laceCorpseAddedEffects", isTrue=→"ADDED EFFECTS", isFalse=(none))
    on ADDED EFFECTS → Added Effects
  }

  state Added Effects {
    ActivateGameObject(gameObject=var "Added Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Pos X: float = 0
  var Scale X: float = 0
  var Added Effects: gameObject = <null>
  var Lace Corpse: gameObject = <null>
}
