# Detect Grab
# 2 location(s):
#   localpoolprefabs_assets_areadustmaze.bundle / Wraith/Collider Charge
#   Dust_Maze_01 / Wraith/Collider Charge

fsm Detect Grab {
  start Idle

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"TOUCH", storeCollider=var "")
    on TOUCH → Grab Check
  }

  state Notify Parent {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="GRAB", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Grab Check {
    RunFSM(fsmTemplateControl=template=-5200671031615995940 in[Bypass Checks<-false, Dont Deal Damage<-var "Dont Deal Damage", No Hero Invuln<-false, Grab Effect Point<-var "Hero"], finishEvent=(none), everyFrame=false)
    on FINISHED → Notify Parent
    on CANCEL → Idle
  }

  var Parent: gameObject = Wraith
  var Grab Effect Point: gameObject = <null>
}
