# Control
# 3 location(s):
#   Bone_East_03 / explode_wall_norck
#   Bone_East_20 / explode_wall_norck
#   Bone_East_24 / explode_wall_norck

fsm Control {
  start Pause
  on EXPLODE → Explode Audio  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindChild(gameObject=Self, childName="Updraft After", storeResult=var "Updraft After")
    FindChild(gameObject=Self, childName="Updraft Before", storeResult=var "Updraft Before")
    SetParent(gameObject=var "Updraft After", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Updraft Before", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    Translate(gameObject=var "Updraft After", vector=(unset), x=-300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on ACTIVATED → Activated
    on FINISHED → Idle
  }

  state Activated {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetFsmBool(gameObject=var "Parent", fsmName="Updraft Control", variableName="Wall Exploded", setValue=true, everyFrame=false)
    Translate(gameObject=var "Updraft Before", vector=(unset), x=-300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    on STRIKE → Strike
    on STRIKE S → Rumble S
  }

  state Strike {
    PlayAudioEvent(audioClip=explode_wall_initial_hit (sfxstatic_assets_areawilds.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=<null>)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    ActivateGameObject(gameObject=var "Clash Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmittersInChildren(gameObject=var "Pt Idle")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Strike", emit=0, resetIfPlaying=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rumble
  }

  state Rumble {
    PlayAudioEvent(audioClip=explode_wall_charge_up (sfxstatic_assets_areawilds.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=<null>)
    PlayParticleEmitterChildren(gameObject=var "Pt Rumble", resetTimeIfPlaying=false, stopOnStateExit=true)
    ActivateGameObject(gameObject=var "Glow Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Glow Rumble", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Sequence", isTrue=→"SEQUENCE", isFalse=(none), everyFrame=false)
    ObjectJitter(gameObject=var "Explode Rock", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall1", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall2", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall3", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flash
    on SEQUENCE → Get Refs
  }

  state Flash {
    ActivateGameObject(gameObject=var "Clash Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set PD
  }

  state Explode {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetFsmBool(gameObject=var "Parent", fsmName="Updraft Control", variableName="Wall Exploded", setValue=true, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Recycle M Silent (localpoolprefabs_assets_areawilds.bundle), spawnPoint=var "Explosion Point", position=(unset), rotation=(unset), storeObject=var "")
    Translate(gameObject=var "Updraft Before", vector=(unset), x=-300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Translate(gameObject=var "Updraft After", vector=(unset), x=300, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    ActivateGameObject(gameObject=var "Explode Rock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Glow Rumble", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "wall1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "wall2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "wall3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Pt Explode", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Death Pieces", activate=true)
    FlingObjects(containerObject=var "Death Pieces", adjustPosition=(0, 0, 0), randomisePosition=false, speedMin=25, speedMax=40, angleMin=0, angleMax=360, space=0)
    FadeGroupDown(target=var "masks", fast=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Seq Rumble {
    ActivateGameObject(gameObject=var "Ceiling Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fake Ceiling", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    ObjectJitter(gameObject=var "Explode Rock", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall1", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall2", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall3", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    SendEventToRegister(eventName="STRIKE S")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flash
  }

  state Get Refs {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Fake Ceiling", storeResult=var "Fake Ceiling")
    FindChild(gameObject=var "Parent", childName="Ceiling Break", storeResult=var "Ceiling Break")
    on FINISHED → Seq Rumble
  }

  state Rumble S {
    PlayParticleEmitterChildren(gameObject=var "Pt Rumble", resetTimeIfPlaying=false, stopOnStateExit=true)
    ActivateGameObject(gameObject=var "Glow Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Glow Rumble", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ObjectJitter(gameObject=var "Explode Rock", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall1", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall2", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    ObjectJitter(gameObject=var "wall3", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sequence Pause
  }

  state Flash S {
    ActivateGameObject(gameObject=var "Clash Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set PD
  }

  state Sequence Pause {
    Wait(time=var "Sequence Pause", finishEvent=(none), realTime=false)
    on FINISHED → Flash
  }

  state Set PD {
    StringCompare(stringVariable=var "Set PlayerData Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "Set PlayerData Bool", value=true)
    on FINISHED → Explode Audio
  }

  state Explode Audio {
    BoolTest(boolVariable=var "No Explosion Audio", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    PlayAudioEvent(audioClip=explode_wall_explode (sfxstatic_assets_areawilds.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Explode
  }

  var Sequence Pause: float = 0
  var Sequence: bool = false
  var Set PlayerData Bool: string = ""

  var Activated: bool = false
  var No Explosion Audio: bool = false
  var CamLock: gameObject = explode_wall_norck/CamLock
  var Ceiling Break: gameObject = <null>
  var Clash Effect: gameObject = explode_wall_norck/Clash Effect
  var Death Pieces: gameObject = explode_wall_norck/death_pieces
  var Explode Rock: gameObject = explode_wall_norck/explode rock
  var Explosion Point: gameObject = explode_wall_norck/Explosion Point
  var Fake Ceiling: gameObject = <null>
  var Glow Idle: gameObject = explode_wall_norck/Glow Idle
  var Glow Rumble: gameObject = explode_wall_norck/Glow Rumble
  var Parent: gameObject = <null>
  var Pt Explode: gameObject = explode_wall_norck/Pt Explode
  var Pt Idle: gameObject = explode_wall_norck/Pt Idle
  var Pt Rumble: gameObject = explode_wall_norck/Pt Rumble
  var Pt Strike: gameObject = explode_wall_norck/Pt Strike
  var Updraft After: gameObject = <null>
  var Updraft Before: gameObject = <null>
  var masks: gameObject = explode_wall_norck/masks
  var wall1: gameObject = explode_wall_norck/wall1
  var wall2: gameObject = explode_wall_norck/wall2
  var wall3: gameObject = explode_wall_norck/wall3
}
