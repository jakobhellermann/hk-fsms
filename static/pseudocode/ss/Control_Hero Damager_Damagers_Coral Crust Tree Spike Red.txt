fsm Control {
  start Idle

  state Idle {
    SetDamageHero(Target=Self, Enabled=true)
    on HOOK → Hooked
  }

  state Hooked {
    SetDamageHero(Target=Self, Enabled=false)
    on HOOK CANCEL → Idle
    on HOOK END → Wait For Hero Pos
  }

  state Wait For Hero Pos {
    CheckTrackTriggerCount(target=var "Hero Tracker", count=0, test=Equal, everyFrame=true, successEvent=→"END")  // disabled
    CheckTrackTriggerCountV2(Target=var "Hero Tracker", Count=0, Test=Equal, EveryFrame=true, SetBool=var "Hero Out Of Tracker", SuccessEvent=(none), FailEvent=(none))
    SetBoolValueAtTime(BoolVariable=var "Waited", BoolValue=true, Time=0.1, SetOppositeOnStateEntry=true)
    BoolAllTrue(boolVariables=[var "Hero Out Of Tracker", var "Waited"], sendEvent=→"END", storeResult=(unset), everyFrame=true)
    on END → Pause
    on HOOK → Hooked
    on HOOK CANCEL → Idle
  }

  state Pause {
    Wait(time=0.1, finishEvent=(none), realTime=false)
    on FINISHED → Idle
    on HOOK → Hooked
    on HOOK CANCEL → Idle
  }

  var Hero Out Of Tracker: bool = false
  var Waited: bool = false
  var Hero Tracker: gameObject = <null>
}
