// uses template: node_control
fsm node_control {
  start Idle

  state Idle {
    on REQUEST NODE → Assign Node
  }

  state Assign Node {
    GetEventSender(sentByGameObject=var "Requester")
    GetRandomChild(gameObject=Self, storeResult=var "Node To Assign")
    SetFsmGameObject(gameObject=var "Requester", fsmName="Control", variableName="My Node", setValue=var "Node To Assign", everyFrame=false)
    SetParent(gameObject=var "Node To Assign", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(var "Requester"), sendEvent="NODE ASSIGNED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Requester: gameObject = <null>
  var Node To Assign: gameObject = <null>
}
