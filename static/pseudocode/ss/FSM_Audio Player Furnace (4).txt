// uses template: lowpass_audio_by_area
fsm FSM {
  start Add Lowpass Component

  state Init {
    AddEventRegister(target=Self, eventName="LOWPASS AREA ENTER")
    AddEventRegister(target=Self, eventName="LOWPASS AREA EXIT")
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Component", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Component", everyFrame=false)
    on FINISHED → Use Vent Region?
  }

  state Normal {
    SetAudioVolume(gameObject=Self, volume=var "Volume Normal", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Component", everyFrame=false)
    on LOWPASS AREA ENTER → To Lowpass
    on VENT REGION ENTER → To Lowpass
  }

  state To Lowpass {
    FadeAudio(gameObject=Self, startVolume=var "Volume Normal", endVolume=var "Volume Lowpass", time=0.5)
    GetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Component", everyFrame=false)
    EaseFloat(fromValue=var "Cutoff Frequency Start", toValue=var "Cutoff Frequency", floatVariable=var "Cutoff Frequency Crt", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Component", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Component", everyFrame=false)
    on FINISHED → Lowpass
    on LOWPASS AREA EXIT → To Normal
    on VENT REGION EXIT → To Normal
  }

  state Lowpass {
    SetAudioVolume(gameObject=Self, volume=var "Volume Lowpass", everyFrame=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Component", everyFrame=false)
    on LOWPASS AREA EXIT → To Normal
    on VENT REGION EXIT → To Normal
  }

  state To Normal {
    GetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Component", everyFrame=false)
    FadeAudio(gameObject=Self, startVolume=var "Volume Lowpass", endVolume=var "Volume Normal", time=0.5)
    EaseFloat(fromValue=var "Cutoff Frequency Start", toValue=22000, floatVariable=var "Cutoff Frequency Crt", time=0.5, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetProperty(targetProperty=AudioLowPassFilter.cutoffFrequency on var "Lowpass Component", everyFrame=true)
    SetProperty(targetProperty=AudioLowPassFilter.enabled on var "Lowpass Component", everyFrame=false)
    on FINISHED → Normal
    on LOWPASS AREA ENTER → To Lowpass
    on VENT REGION ENTER → To Lowpass
  }

  state Use Vent Region? {
    BoolTest(boolVariable=var "Use Vent Events", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    AddEventRegister(target=Self, eventName="VENT REGION ENTER")
    AddEventRegister(target=Self, eventName="VENT REGION EXIT")
    on FINISHED → Start Lowpassed?
  }

  state Start Lowpassed? {
    BoolTest(boolVariable=var "z Start Lowpassed", isTrue=→"LOWPASS AREA ENTER", isFalse=(none), everyFrame=false)
    on FINISHED → To Normal
    on LOWPASS AREA ENTER → To Lowpass
  }

  state Add Lowpass Component {
    ObjectIsNull(Target=var "Lowpass Component", trueEvent=(none), falseEvent=→"FINISHED", storeValue=(unset))
    AddComponent(gameObject=Self, component="UnityEngine.AudioLowPassFilter", storeComponent=var "Lowpass Component", removeOnExit=false)
    on FINISHED → Init
  }

  var Cutoff Frequency: float = 500000
  var Volume Normal: float = 1
  var Volume Lowpass: float = 0
  var Use Vent Events: bool = false
  var z Start Lowpassed: bool = false
  var Lowpass Component: object = <null>

  var Cutoff Frequency Crt: float = 0
  var Cutoff Frequency Start: float = 0
}
