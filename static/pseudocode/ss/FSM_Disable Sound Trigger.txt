# FSM
# 1 location(s):
#   Under_08 / Disable Sound Trigger

fsm FSM {
  start Wait

  state Wait {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Check
  }

  state Check {
    BoolTest(boolVariable=var "Activated", isTrue=→"END", isFalse=(none), everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on END → Deactivate
    on ENTER → Send Event
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Send Event {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetPersistentBool(Target=var "Secret Sound Region", SetValue=true)
    ActivateGameObject(gameObject=var "Secret Sound Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Deactivate
  }

  var Activated: bool = false
  var Secret Sound Region: object = Secret Sound Region
}
