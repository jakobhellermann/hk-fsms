# Control
# 1 location(s):
#   Arborium_02 / Rhino Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Rhino")
    FindNamedChild(gameObject=var "Rhino", storeResult=var "Behind Marker")
    on FINISHED → Check
  }

  state Check {
    CheckTargetDirection(gameObject=var "Rhino", target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"SHIFT", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on SHIFT → Shift
  }

  state Shift {
    SetPosition(gameObject=var "Rhino", vector=(unset), x=103.83, y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Rhino", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Behind Marker", vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
  }

  var Behind Marker: gameObject = <null>
  var Rhino: gameObject = <null>
}
