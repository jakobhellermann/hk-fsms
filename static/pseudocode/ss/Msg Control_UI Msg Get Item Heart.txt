# Msg Control
# 1 location(s):
#   localpoolprefabs_assets_areamemory.bundle / UI Msg Get Item Heart

fsm Msg Control {
  start Init

  state Detect {
    ListenForPromptContinue(EventTarget=Self, ContinuePressed=→"PRESS")
    on PRESS → Down
  }

  sequence state Done {
    SendEventToRegister(eventName="GET ITEM MSG END")
    CallMethodProper(gameObject=Self, behaviour="UIMsgProxy", methodName="SetIsInMsg", parameters=[false], storeResult=(unset var), EveryFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    ShowQuestUpdatedStandalone(Quest=var "Quest")
    RecycleSelf()
  }

  state Init {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0)
    AnimatorPlay(gameObject=Self, stateName="Down", layer=0, normalizedTime=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Stop", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CORAL → Set Coral
    on FLOWER → Set Flower
    on HUNTER → Set Hunter
    on CLOVER → Set Clover
  }

  state Up {
    CallMethodProper(gameObject=Self, behaviour="UIMsgProxy", methodName="SetIsInMsg", parameters=[true], storeResult=(unset var), EveryFrame=false)
    AnimatorPlayStateWait(target=Self, stateName="Up", finishEvent=→"FINISHED")
    ListenForAnimationEvent(Target=Self, Response=→"FINISHED")
    on FINISHED → Sound
  }

  state Stop Up {
    FadeNestedFadeGroup(Target=var "Stop", ToAlpha=0, FadeTime=0)
    ActivateGameObject(gameObject=var "Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Stop", ToAlpha=1, FadeTime=0.25)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Down {
    PlayAudioEvent(audioClip=button (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatorPlayStateWait(target=Self, stateName="Down", finishEvent=→"FINISHED")
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=2)
    on FINISHED → Done
  }

  state Set Coral {
    GetLanguageString(sheetName="UI", convName="INV_NAME_HEART_CORAL", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_DESC_HEART_CORAL", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Desc", textString=var "Game Text")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=heart_prompt_coral (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetSpriteRendererSprite(gameObject=var "Icon Glow", sprite=heart_prompt_coral_glow (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetObjectValue(objectVariable=var "Quest", objectValue=Black Thread Pt5 Heart Coral (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle), everyFrame=false)
    on FINISHED → Set Common
  }

  state Set Common {
    GetLanguageString(sheetName="Prompts", convName="GET_ITEM_INTRO_GET", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Up
  }

  state Set Flower {
    GetLanguageString(sheetName="UI", convName="INV_NAME_HEART_BLOOM", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_DESC_HEART_FLOWER", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Desc", textString=var "Game Text")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=heart_prompt_flower (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetSpriteRendererSprite(gameObject=var "Icon Glow", sprite=heart_prompt_flower_glow (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetObjectValue(objectVariable=var "Quest", objectValue=Black Thread Pt5 Heart Flower (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle), everyFrame=false)
    on FINISHED → Set Common
  }

  state Set Hunter {
    GetLanguageString(sheetName="UI", convName="INV_NAME_HEART_HUNTER", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_DESC_HEART_HUNTER", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Desc", textString=var "Game Text")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=heart_prompt_hunter (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetSpriteRendererSprite(gameObject=var "Icon Glow", sprite=heart_prompt_hunter_glow (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetObjectValue(objectVariable=var "Quest", objectValue=Black Thread Pt5 Heart Ant (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle), everyFrame=false)
    on FINISHED → Set Common
  }

  state Stop Playing {
    SendEventToRegister(eventName="GET ITEM MSG COVERED")
    on FINISHED → Wait
  }

  state Wait {
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Up
  }

  state Sound {
    PlayAudioEvent(audioClip=ui_ability_get_from_weavers (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Stop Playing
  }

  state Set Clover {
    GetLanguageString(sheetName="UI", convName="INV_NAME_HEART_CLOVER", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_DESC_HEART_CLOVER", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Desc", textString=var "Game Text")
    SetSpriteRendererSprite(gameObject=var "Icon", sprite=heart_prompt_clover (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetSpriteRendererSprite(gameObject=var "Icon Glow", sprite=heart_prompt_clover_glow (atlases_assets_assets/sprites/_atlases/prompts.spriteatlas.bundle))
    SetObjectValue(objectVariable=var "Quest", objectValue=Black Thread Pt5 Heart (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle), everyFrame=false)
    on FINISHED → Set Common
  }

  var Game Text: string = ""
  var Icon: gameObject = UI Msg Get Item Heart/Contents/Icon
  var Icon Glow: gameObject = UI Msg Get Item Heart/Contents/Icon/Icon Glow
  var Item Name: gameObject = UI Msg Get Item Heart/Contents/Item Name
  var Item Name Prefix: gameObject = UI Msg Get Item Heart/Contents/Item Name Prefix
  var Stop: gameObject = UI Msg Get Item Heart/Contents/Stop
  var Item Desc: gameObject = UI Msg Get Item Heart/Contents/Item Desc
  var Quest: object = <null>
}
