# Combo Hit Control
# 1 location(s):
#   Room_Forge / _NPCs/Forge Worker

fsm Combo Hit Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  sequence state Combo {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    SendEventToRegister(eventName="WORKER HIT COMBO")
    SetFloatValue(floatVariable=var "Combo", floatValue=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    FloatAdd(floatVariable=var "Combo", add=-0.25, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Combo", minValue=0, maxValue=10, everyFrame=true)
    on WORKER HIT → Hit
  }

  state Hit {
    FloatAddRandom(floatVariable=var "Combo", addMin=0.75, addMax=1)
    FloatCompare(float1=var "Combo", float2=4, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"COMBO", everyFrame=false)
    on FINISHED → Idle
    on COMBO → Combo
  }

  var Combo: float = 0
}
