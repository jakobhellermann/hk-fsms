# Control
# 2 location(s):
#   Bone_05_boss / Boss Scene/Threaded Bell Gate
#   Bone_05_boss / Boss Scene/Threaded Bell Gate (1)

fsm Control {
  start Init
  on BG OPEN → Open  // from any state

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Opened
  }

  state Opened {
    on BG CLOSE → Close
  }

  state Close {
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=bell_beast_thread_break (sfxstatic_assets_areabell.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.75, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=bell_chain_plat_land (sfxstatic_assets_areabell.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Closed
  }

  state Closed {
    on TINK → Tink
  }

  state Tink {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Closed
  }

  state Open {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
  }

  var Self: gameObject = <null>
}
