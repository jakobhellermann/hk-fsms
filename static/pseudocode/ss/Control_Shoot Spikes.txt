# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Shoot Spikes

fsm Control {
  start Init

  state Idle {
    on ATTACK → Check Dir
    on BOSS DEATH TEST → Test
  }

  state Attack R {
    GetRandomChild(gameObject=var "Patterns R", storeResult=var "Next Pattern")  // disabled
    ConvertBoolToString(boolVariable=var "Do Pattern 1", stringVariable=var "Pattern Name", falseString="Pattern 2", trueString="Pattern 1", everyFrame=false)
    BoolFlip(boolVariable=var "Do Pattern 1")
    FindChild(gameObject=var "Patterns R", childName=var "Pattern Name", storeResult=var "Next Pattern")
    SendEventByName(eventTarget=GameObject(var "Next Pattern"), sendEvent="SHOOT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Patterns L")
    FindNamedChild(gameObject=Self, storeResult=var "Patterns R")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Coral King")
    on FINISHED → Idle
  }

  state Check Dir {
    SendEventByScale(gameObject=var "Coral King", eventTarget=Self, xScale=true, positiveEvent=→"R", negativeEvent=→"L", space=0)
    on R → Attack R
    on L → Attack L
  }

  state Attack L {
    GetRandomChild(gameObject=var "Patterns L", storeResult=var "Next Pattern")  // disabled
    ConvertBoolToString(boolVariable=var "Do Pattern 1", stringVariable=var "Pattern Name", falseString="Pattern 2", trueString="Pattern 1", everyFrame=false)
    BoolFlip(boolVariable=var "Do Pattern 1")
    FindChild(gameObject=var "Patterns R", childName=var "Pattern Name", storeResult=var "Next Pattern")
    SendEventByName(eventTarget=GameObject(var "Next Pattern"), sendEvent="SHOOT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Test {
    SendEventToRegister(eventName="BOSS DEFEATED")
    on FINISHED → Idle
  }

  var Do Pattern 1: bool = false
  var Pattern Name: string = ""
  var Coral King: gameObject = <null>
  var Next Pattern: gameObject = <null>
  var Parent: gameObject = <null>
  var Patterns L: gameObject = <null>
  var Patterns R: gameObject = <null>
}
