# Detect
# 2 location(s):
#   herodynamic_assets_all.bundle / Tool Bomb/Interactor
#   herodynamic_assets_all.bundle / Tool Dust Bomb New/Interactor

// uses template: tool_bomb_interact
fsm Detect {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Explosion", collideLayer=(unset), sendEvent=→"EXPLODE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Attack(17), sendEvent=→"CHECK ATTACK", storeCollider=var "Attack Object")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Enemy Attack(22), sendEvent=→"CHECK ATTACK", storeCollider=var "Attack Object")
    on EXPLODE → Explode
    on TINK DOWN L → Tink Down L
    on TINK DOWN R → Tink Down R
    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
    on TINK UP → Tink Up
    on CHECK ATTACK → Check Attack
    on TORNADO → Tornado
  }

  state Explode {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="INSTA EXPLODE", delay=0, everyFrame=false)
  }

  state Tink Down L {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK DOWN L", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Tink Down R {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK DOWN R", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Tink Left {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK LEFT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Tink Right {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK RIGHT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Tink Up {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK UP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Check Attack {
    HasComponent(gameObject=var "Attack Object", component="DamageHero", removeOnExit=false, trueEvent=(none), falseEvent=→"CANCEL", store=(unset), everyFrame=false)
    CallMethodProper(gameObject=var "Attack Object", behaviour="DamageHero", methodName="IsDamagerSpikes", parameters=[], storeResult=var "Is Damager Spikes", EveryFrame=false)
    BoolTest(boolVariable=var "Is Damager Spikes", isTrue=→"EXPLODE", isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Idle
    on EXPLODE → Explode
  }

  state Tornado {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TORNADO", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Swallow {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="SWALLOW", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Is Damager Spikes: bool = false
  var Parent: gameObject = <null>
  var Attack Object: gameObject = <null>
}
