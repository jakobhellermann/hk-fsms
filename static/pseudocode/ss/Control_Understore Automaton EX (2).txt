# Control
# 1 location(s):
#   Under_21 / Understore Automaton EX (2)

fsm Control {
  start Init
  on EXPLODE → Explode  // from any state
  on LAVA DEATH → Explode  // from any state
  on HARPOON DAMAGE → Die to harpoon  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Battle Enter EX", equalEvent=→"BATTLE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Clip", compareTo="Ambush EX", equalEvent=→"AMBUSH", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    WaitRandom(timeMin=0, timeMax=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on BATTLE → Dormant
    on AMBUSH → Ambush Ready
  }

  state Idle {
    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=1.5, speedMax=2, accelerationMin=5, accelerationMax=10, roamingRangeX=0.25, roamingRangeY=0.25, manualStartPos=var "Start Pos")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle EX")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXPLODE", storeCollider=var "")
    AudioPlayInState(gameObject=Self, volume=1)
    RandomFloat(min=0.8, max=1.2, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlayInState(gameObject=var "Loop Sizzle", volume=1)
    RandomFloat(min=0.8, max=1.2, storeResult=var "Pitch")
    SetAudioPitch(gameObject=var "Loop Sizzle", pitch=var "Pitch", everyFrame=false)
    SetAudioClip(gameObject=var "Loop Sizzle", audioClip=understore_automaton_ex_idle_sizzle (sfxstatic_assets_areaunderstore.bundle), autoPlay=false, stopOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1.2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ALERT → Alert
    on TOOK DAMAGE → Alert
    on STARTLE → Alert
    on SING → Sing
  }

  state Alert {
    PlayAudioEvent(audioClip=song_automaton_fly_antic (sfxstatic_assets_areacog.bundle), pitchMin=0.7, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=automaton_creature_chirp (sfxdynamic_assets_areacog.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    ActivateGameObject(gameObject=var "Glow Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXPLODE", storeCollider=var "")
    ActivateGameObject(gameObject=var "Glow Chase", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Alert Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Alert EX", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopParticleEmitter(gameObject=var "Pt Idle")
    PlayParticleEmitter(gameObject=var "Pt Chase", emit=0, resetIfPlaying=false)
    CallMethodProper(gameObject=Self, behaviour="SpriteFlash", methodName="FlashingBomb", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    SetAudioClip(gameObject=var "Loop Sizzle", audioClip=understore_automaton_ex_chase_sizzle (sfxstatic_assets_areaunderstore.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=var "Loop Sizzle", volume=1, oneShotClip=<null>)
    on FINISHED → Chase
  }

  state Chase {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Chase EX")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXPLODE", storeCollider=var "")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="TurnToChase EX", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    ChaseObjectV2(gameObject=Self, target=var "Hero", speedMax=8, accelerationForce=15, offsetX=0, offsetY=1.5)
    FloatAdd(floatVariable=var "Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=0, tolerance=0, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    on NEXT → Explode Antic
  }

  state Explode Antic {
    ObjectJitter(gameObject=Self, x=0.15, y=0.15, z=(unset), allowMovement=true, limitFps=30)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.98, brakeOnExit=false)
    CallMethodProper(gameObject=Self, behaviour="SpriteFlash", methodName="FlashingBombFast", parameters=[], storeResult=(unset var), EveryFrame=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXPLODE", storeCollider=var "")
    SetHP(target=Self, hp=1)
    on FINISHED → Explode
  }

  state Explode {
    DoNotSpawnCorpse(target=Self)
    SpawnObjectFromGlobalPool(gameObject=Understore Automaton EX Death Chunks (localpoolprefabs_assets_areaunderstore.bundle), spawnPoint=var "Self", position=(0, 0, 0), rotation=(0, 0, 0), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Gas Explosion Recycle M (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    InstaDeath(target=Self, direction=0)
  }

  state Dormant {
    ActivateGameObject(gameObject=var "Damageable Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Glow Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Idle")
    on BATTLE START → Delay
  }

  state Enter {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle EX")
    ActivateGameObject(gameObject=var "Damageable Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayRandomAudioClipTable(Table=automaton_hit (sfxdynamic_assets_areacog.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayRandomAudioClipTable(Table=enemy_dig_in_emerge (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SetCircleCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Glow Idle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXPLODE", storeCollider=var "")
    RandomFloat(min=-25, max=-35, storeResult=var "Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Speed", everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Alert
  }

  state Delay {
    Wait(time=var "Enter Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter Antic
  }

  state Fling Corpse {
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    on FINISHED → Dead
  }

  state Activate {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Alert EX", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Damageable Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=true)
    CallMethodProper(gameObject=Self, behaviour="HealthManager", methodName="RefillHP", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Fly In {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle EX")
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=Self, translateMin=(-0.5, -0.5, 0), translateMax=(0.5, 0.5, 0), space=1)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=40, z=25, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    CallMethodProper(gameObject=Self, behaviour="ResetDynamicHierarchy", methodName="DoReset", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Damageable Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -40, -25), time=1, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  state Dead {
    SendEvent(eventTarget=Self, sendEvent=→"CANCEL", delay=0, everyFrame=false)
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Battle Enter EX", equalEvent=→"CANCEL", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    CallMethodProper(gameObject=Self, behaviour="SpriteFlash", methodName="CancelRepeatingFlash", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Distance
    on CANCEL → Disable
  }

  state Disable {
    DestroySelf(detachChildren=false)  // disabled
    SetPosition2d(gameObject=Self, vector=(unset), x=-100, y=-50, space=0, everyFrame=false, lateUpdate=false)
    InstaDeath(target=Self, direction=0)
  }

  state Distance {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=15, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Fly In
  }

  state Enter Antic {
    PlayRandomAudioClipTable(Table=enemy_emerge_antic (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayParticleEmitterInState(gameObject=var "Pt EnterAntic")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Enter
  }

  state Ambush Ready {
    ActivateGameObject(gameObject=var "Damageable Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Glow Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Idle")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="AMBUSH", outOfRangeEvent="", everyFrame=true)
    on AMBUSH → Ambush Fly In
  }

  state Activate 2 {
    ActivateGameObject(gameObject=var "Damageable Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=true)
    CallMethodProper(gameObject=Self, behaviour="HealthManager", methodName="RefillHP", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Alert
  }

  state Ambush Fly In {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle EX")
    SetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=15, z=5, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Damageable Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -15, -5), time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=enemy_fly_in (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Activate 2
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing EX")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=automaton_creature_sing (sfxdynamic_assets_areacog.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    PlayAudioEvent(audioClip=song_automaton_fly_antic (sfxstatic_assets_areacog.bundle), pitchMin=0.7, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Glow Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Glow Chase", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Alert Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Idle")
    PlayParticleEmitter(gameObject=var "Pt Chase", emit=0, resetIfPlaying=false)
    SetAudioClip(gameObject=var "Loop Sizzle", audioClip=understore_automaton_ex_chase_sizzle (sfxstatic_assets_areaunderstore.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=var "Loop Sizzle", volume=1, oneShotClip=<null>)
    FloatAdd(floatVariable=var "Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=0, tolerance=0, equal=(none), lessThan=→"NEXT", greaterThan=(none), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on CANCEL → Chase
    on NEXT → Explode Antic
  }

  state Die to harpoon {
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatTestToBool(float1=var "Hero X Scale", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Hero Facing R", greaterThanBool=(unset), everyFrame=false)
    ConvertBoolToFloat(boolVariable=var "Hero Facing R", floatVariable=var "Angle", falseValue=180, trueValue=0, everyFrame=false)
    InstaDeath(target=Self, direction=var "Angle")
  }

  var Enter Delay: float = 0

  var Speed: float = 0
  var Pitch: float = 0
  var Distance: float = 0
  var Timer: float = 1.5
  var Hero X Scale: float = 0
  var Angle: float = 0
  var Hero Facing R: bool = false
  var Clip: string = ""
  var Start Pos: vector3 = (0, 0, 0)
  var Damageable Collider: gameObject = Understore Automaton EX (2)/Damageable Collider
  var Pt Idle: gameObject = Understore Automaton EX (2)/Pt Idle
  var Glow Idle: gameObject = Understore Automaton EX (2)/Glow Idle
  var Glow Chase: gameObject = Understore Automaton EX (2)/Glow Chase
  var Pt Chase: gameObject = Understore Automaton EX (2)/Pt Chase
  var Alert Effects: gameObject = Understore Automaton EX (2)/Alert Effects
  var Self: gameObject = Understore Automaton EX (2)
  var Pt EnterAntic: gameObject = Understore Automaton EX (2)/Pt EnterAntic
  var Loop Sizzle: gameObject = Understore Automaton EX (2)/Loop Sizzle
  var Audio Loop Voice: gameObject = Understore Automaton EX (2)/Audio Loop Voice
}
