# Particle Control
# 2 location(s):
#   localpoolprefabs_assets_areadustmaze.bundle / Wraith
#   Dust_Maze_01 / Wraith

fsm Particle Control {
  start Range Out

  state Range Out {
    CheckOutOfCamera(gameObject=Self, margin=10, outsideEvent=(none), insideEvent=→"IN", insideBool=false, outsideBool=false, everyFrame=true)
    on IN → Range In
  }

  state Range In {
    PlayParticleEmitterInState(gameObject=var "Pt Idle")
    CheckOutOfCamera(gameObject=Self, margin=10, outsideEvent=→"OUT", insideEvent=(none), insideBool=false, outsideBool=false, everyFrame=true)
    on OUT → Range Out
  }

  var Pt Idle: gameObject = Wraith/Pt Idle
}
