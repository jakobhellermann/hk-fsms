# Duck Reaction
# 1 location(s):
#   Bone_12 / Black Thread States/Normal World/Pin Gallery States/Here/Lady Bug Large

fsm Duck Reaction {
  start Inert

  state Inert {
    SetBoolValue(boolVariable=var "Is Ducked", boolValue=false, everyFrame=false)
    TriggerEnterEventSubscribe(trigger=var "Attack Detector", triggerEnteredEvent=→"DUCK", triggerExitedEvent=(none), triggerStayedEvent=(none))
    on DUCK → Duck
  }

  state Duck {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=loddie_startle (sfxdynamic_assets_areabone.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    EnableBehaviour(gameObject=Self, behaviour="HeroPerformanceSingReaction", component=<null>, enable=false, resetOnExit=false)
    SetBoolValue(boolVariable=var "Is Ducked", boolValue=true, everyFrame=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Duck", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Ducking
  }

  state Ducking {
    TriggerEnterEventSubscribe(trigger=var "Attack Detector", triggerEnteredEvent=→"DUCK", triggerExitedEvent=(none), triggerStayedEvent=(none))
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on DUCK → Ducking
    on FINISHED → Recover
    on DUCK CANCEL → Recover
  }

  state Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Duck Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  sequence state Reset {
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    NextFrameEvent(sendEvent=(none))
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    EnableBehaviour(gameObject=Self, behaviour="HeroPerformanceSingReaction", component=<null>, enable=true, resetOnExit=false)
    on FINISHED → Inert
  }

  var Is Ducked: bool = false
  var Audio Loop Voice: gameObject = Black Thread States/Normal World/Pin Gallery States/Here/Lady Bug Large/Audio Loop Voice
  var Attack Detector: object = Black Thread States/Normal World/Pin Gallery States/Here/Lady Bug Large/Attack Detector@TriggerEnterEvent
}
