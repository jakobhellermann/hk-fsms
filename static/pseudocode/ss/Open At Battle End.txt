# Open At Battle End
# 1 location(s):
#   Ward_02 / Boss Scene Parent/Pipe_Vent_Hatch

fsm Open At Battle End {
  start State 3

  state State 1 {
    PlayerDataBoolTest(boolName="wardBossHatchOpened", isTrue=→"WARD TRAPDOOR OPEN", isFalse=(none))
    GetPlayerDataBool(boolName="wardBossDefeated", storeValue=var "Boss Defeated")
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatTestToBool(float1=var "Distance", float2=6, tolerance=0, equalBool=(unset), lessThanBool=var "Hero Nearby", greaterThanBool=(unset), everyFrame=true)
    BoolAllTrue(boolVariables=[var "Boss Defeated", var "Hero Nearby"], sendEvent=→"WARD TRAPDOOR OPEN", storeResult=(unset), everyFrame=true)
    on WARD TRAPDOOR OPEN → Set Opened
    on HEART COLLECTED → State 2
  }

  state State 2 {
    SetPlayerDataBool(boolName="wardBossHatchOpened", value=true)
    CallMethodProper(gameObject=Self, behaviour="Trapdoor", methodName="OpenDoorCustom", parameters=[0], storeResult=(unset var), EveryFrame=false)
  }

  state State 3 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State 1
  }

  state Set Opened {
    CallMethodProper(gameObject=Self, behaviour="Trapdoor", methodName="OpenDoorCustomSilent", parameters=[0], storeResult=(unset var), EveryFrame=false)
  }

  var Distance: float = 0
  var Boss Defeated: bool = false
  var Hero Nearby: bool = false
}
