// uses template: hornet_multi_wounder
fsm FSM {
  start Init

  state Idle {
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"HIT", storeCollider=var "", storeForce=(unset))  // disabled
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"COLLIDE", storeCollider=var "Collider")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"COLLIDE", storeCollider=var "")
    on COLLIDE → Hit
    on PARRIED → Parried Recover
    on TINK DOWN → Tink D
  }

  state Hit {
    PlayerDataBoolTest(boolName="isInvincible", isTrue=→"CANCEL", isFalse=(none))
    BoolTest(boolVariable=var "Hazard Hit", isTrue=→"HAZARD", isFalse=(none), everyFrame=false)
    GetHeroCState(VariableName="parrying", StoreValue=var "Hero Parrying", EveryFrame=false)
    BoolTestMulti(boolVariables=[var "Hero Parrying", var "Ignore Parrying"], boolStates=[true, false], trueEvent=→"CROSS STITCH", falseEvent=(none), storeResult=(unset), everyFrame=false)
    HeroControllerMethods(method=IsParryInvulnerable, parameters=[], everyFrame=false, storeValue=(unset var), isTrue=→"PARRIED", isFalse=(none))
    CanHeroTakeDamage(eventTarget=Self, canTakeDmgEvent=(none), cannotTakeDmgEvent=→"CANCEL")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="WillDoBellBindHit", parameters=[], storeResult=var "Did Bind Bell Hit", EveryFrame=false)
    BoolTest(boolVariable=var "Did Bind Bell Hit", isTrue=→"BELL BIND", isFalse=(none), everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Roar and Wound States", variableName="Damaged By Void Multihitter", setValue=var "Black Threaded", everyFrame=false)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Roar and Wound States", variableName="Multi Wounder", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MULTI HIT CONNECT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="MULTI HIT CONNECT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Grandparent"), sendEvent="MULTI HIT CONNECT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Great Grandparent"), sendEvent="MULTI HIT CONNECT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent=var "Hero Event", delay=0, everyFrame=false)
    PlayAudioEvent(audioClip=hero_damage (herosfxstatic_assets_all.bundle), pitchMin=1.2, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=standard_health_hit (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    Wait(time=0.1, finishEvent=→"END", realTime=false)
    on END → Idle
    on BELL BIND → Bell Bind Hit
    on HAZARD → Hazard Hit
    on CROSS STITCH → Cross Stitch Parry
    on CANCEL → Idle
    on PARRIED → Parried Recover
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetGrandparent(gameObject=Self, storeResult=var "Grandparent")
    GetParent(gameObject=var "Grandparent", storeResult=var "Great Grandparent")
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Check Black Threaded
  }

  state Parried Recover {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on PARRIED → Parried Recover
  }

  state Bell Bind Hit {
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=false)
    SendEventByNameUpwards(Target=Self, EventName="CANCEL MULTI HIT")
    on FINISHED → Idle
  }

  sequence state Hazard Hit {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent=var "Hero Event Hazard", delay=0, everyFrame=false)
    Wait(time=0.3, finishEvent=(none), realTime=false)
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=true, sinkHazard=false)
    Wait(time=1.35, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  state Cross Stitch Parry {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="PARRIED", delay=0, everyFrame=false)
    on FINISHED → Parried Recover
  }

  state Check Steam Hazard {
    SetStringValue(stringVariable=var "Hero Event Hazard", stringValue="MULTI WOUND HAZARD", everyFrame=false)
    BoolTest(boolVariable=var "z2 Steam Hazard", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Hazard Hit", boolValue=true, everyFrame=false)
    SetStringValue(stringVariable=var "Hero Event Hazard", stringValue="MULTI WOUND STEAM", everyFrame=false)
    on FINISHED → Set Tink Event
  }

  state Check Dmg Hero Event {
    SetStringValue(stringVariable=var "Hero Event", stringValue="MULTI WOUND 3", everyFrame=false)
    EnumCompare(enumVariable=var "Multihit Type", compareTo=LagHit, equalEvent=→"LAG HIT", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    EnumCompare(enumVariable=var "Multihit Type", compareTo=Zap, equalEvent=→"ZAP HIT", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    EnumCompare(enumVariable=var "Multihit Type", compareTo=DoubleStrike, equalEvent=→"DOUBLE STRIKE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    EnumCompare(enumVariable=var "Multihit Type", compareTo=Pollen, equalEvent=→"POLLEN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    EnumCompare(enumVariable=var "Multihit Type", compareTo=NoEffect, equalEvent=→"NO EFFECT", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    EnumCompare(enumVariable=var "Multihit Type", compareTo=Weak, equalEvent=→"WEAK", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Idle
    on LAG HIT → Set Lag Hit
    on ZAP HIT → Set Zap Hit
    on DOUBLE STRIKE → Set Double Strike
    on POLLEN → Set Pollen Hit
    on NO EFFECT → Set No Effect
    on WEAK → Set Weak
  }

  state Set Lag Hit {
    SetStringValue(stringVariable=var "Hero Event", stringValue="MULTI LAG HIT", everyFrame=false)
    on FINISHED → Idle
  }

  state Set Zap Hit {
    SetStringValue(stringVariable=var "Hero Event", stringValue="MULTI ZAP HIT", everyFrame=false)
    on FINISHED → Idle
  }

  state Set Double Strike {
    SetStringValue(stringVariable=var "Hero Event", stringValue="MULTI DOUBLE STRIKE", everyFrame=false)
    on FINISHED → Idle
  }

  state Set Tink Event {
    HasComponent(gameObject=Self, component="TinkEffect", removeOnExit=false, trueEvent=(none), falseEvent=→"FINISHED", store=(unset), everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="TinkEffect", methodName="SetFsmEvent", parameters=["TINKED"], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="TinkEffect", methodName="SetSendDirectionalFSMEvents", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Check Dmg Hero Event
  }

  state Check Black Threaded {
    CheckIsBlackThreaded(Target=Self, TrueEvent=(none), FalseEvent=(none), StoreValue=var "Black Threaded", EveryFrame=false)
    BoolTest(boolVariable=var "z3 Force Black Threaded", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Black Threaded", boolValue=true, everyFrame=false)
    on FINISHED → Check Steam Hazard
  }

  state Set Pollen Hit {
    SetStringValue(stringVariable=var "Hero Event", stringValue="MULTI POLLEN HIT", everyFrame=false)
    on FINISHED → Idle
  }

  state Tink D {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Set No Effect {
    SetStringValue(stringVariable=var "Hero Event", stringValue="MULTI HIT NO EFFECT", everyFrame=false)
    on FINISHED → Idle
  }

  state Set Weak {
    SetStringValue(stringVariable=var "Hero Event", stringValue="MULTI WOUND WEAK", everyFrame=false)
    SetStringValue(stringVariable=var "Hero Event Hazard", stringValue="MULTI WOUND HAZARD WEAK", everyFrame=false)
    on FINISHED → Idle
  }

  var Hazard Hit: bool = false
  var Ignore Parrying: bool = false
  var z2 Steam Hazard: bool = false
  var z3 Force Black Threaded: bool = false
  var Multihit Type: enum = MultihitTypes(0)

  var Black Threaded: bool = false
  var Did Bind Bell Hit: bool = false
  var Hero Parrying: bool = false
  var Hero Event: string = ""
  var Hero Event Hazard: string = ""
  var Self: gameObject = <null>
  var Parent: gameObject = <null>
  var Grandparent: gameObject = <null>
  var Collider: gameObject = <null>
  var Great Grandparent: gameObject = <null>
}
