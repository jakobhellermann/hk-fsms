// uses template: coal_region
fsm Damage Hero {
  start Init

  state Init {
    CreateObject(gameObject=Lava Bell Coal_Region_Shield (localpoolprefabs_assets_lava.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Lava Bell Shield")
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    SetParent(gameObject=var "Lava Bell Shield", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Lava Bell Shield", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="Pt Smoke", storeResult=var "Pt Smoke")
    FindChild(gameObject=Self, childName="Pt Sparks", storeResult=var "Pt Sparks")
    on FINISHED → Idle
  }

  state Idle {
    RemoveStatusVignetteStatus(Status=InCoalRegion, StoredSet=var "Did Set Vignetta")
    SendEventToRegister(eventName="LAVA BELL SIZZLE STOP")
    CallMethodProper(gameObject=var "Lava Bell Shield", behaviour="LavaBellCoalRegionShield", methodName="End", parameters=[], storeResult=(unset var), EveryFrame=false)
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.4)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on ENTER → Lava Bell?
    on FINISHED → Idle Inactive
  }

  state Sizzle Short {
    SetPositionToObject(gameObject=var "Pt Smoke", targetObject=var "Hero", xOffset=0, yOffset=-1.8, zOffset=-0.002, overrideZ=(unset), everyFrame=false)
    SetPositionToObject(gameObject=var "Pt Sparks", targetObject=var "Hero", xOffset=0, yOffset=-1.8, zOffset=-0.003, overrideZ=(unset), everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=0.2)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXIT", storeCollider=var "")
    SendMessageDelay(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FlashCoal(), delay=0.2)
    SendMessageDelay(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FlashCoal(), delay=0.4)
    SendMessageDelay(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FlashCoal(), delay=0.55)  // disabled
    SetPositionToObject(gameObject=var "Pt Smoke", targetObject=var "Hero", xOffset=0, yOffset=-1.8, zOffset=-0.002, overrideZ=(unset), everyFrame=true)
    SetPositionToObject(gameObject=var "Pt Sparks", targetObject=var "Hero", xOffset=0, yOffset=-1.8, zOffset=-0.003, overrideZ=(unset), everyFrame=true)
    GetSpeed2d(gameObject=var "Hero", storeResult=var "Hero Speed", everyFrame=true)
    FloatCompare(float1=var "Hero Speed", float2=14, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SPRINT", everyFrame=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="IsLavaBellActive", parameters=[], storeResult=var "Is Bell Active", EveryFrame=true)
    BoolTest(boolVariable=var "Is Bell Active", isTrue=→"TRUE", isFalse=(none), everyFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Smoke")
    PlayParticleEmitterInState(gameObject=var "Pt Sparks")
    on EXIT → Idle
    on FINISHED → Damage
    on SPRINT → Sprinting
    on TRUE → No Sizzle
  }

  state Damage {
    CallMethodProper(gameObject=var "Lava Bell Shield", behaviour="LavaBellCoalRegionShield", methodName="End", parameters=[], storeResult=(unset var), EveryFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetDamageHeroAmount(target=Self, damageDealt=1)  // disabled
    DamageHeroDirectlyV2(damager=Self, damageAmount=1, hazardType=ENEMY, damageDirection=FromHeroFacingDirection, invertDirection=false, overrideDirection=top)
    AddTempStatusVignette(Status=None)
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=false)
    SpawnObjectFromGlobalPool(gameObject=Coal Damage Effect (localpoolprefabs_assets_lavashared.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Frame 2
  }

  state Frame 2 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Still Inside
  }

  state Lava Bell? {
    GetHeroCStateEvent(VariableName="isInCutsceneMovement", StoreValue=(unset), TrueEvent=→"CANCEL", FalseEvent=(none), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="IsLavaBellActive", parameters=[], storeResult=var "Is Bell Active", EveryFrame=false)
    BoolTest(boolVariable=var "Is Bell Active", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on CANCEL → Idle Inactive
    on TRUE → No Sizzle
    on FALSE → Sizzle Start
  }

  state No Sizzle {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="IsLavaBellActive", parameters=[], storeResult=var "Is Bell Active", EveryFrame=true)
    BoolTest(boolVariable=var "Is Bell Active", isTrue=(none), isFalse=→"FALSE", everyFrame=true)
    GetSpeed2d(gameObject=var "Hero", storeResult=var "Hero Speed", everyFrame=true)
    FloatCompare(float1=var "Hero Speed", float2=14, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SPRINT", everyFrame=true)
    SendEventToRegister(eventName="LAVA BELL SIZZLE START")
    ActivateGameObject(gameObject=var "Lava Bell Shield", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Lava Bell Shield", behaviour="LavaBellCoalRegionShield", methodName="Begin", parameters=[], storeResult=(unset var), EveryFrame=false)
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=0.2)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXIT", storeCollider=var "")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on EXIT → Idle
    on FALSE → Lava Bell End
    on FINISHED → Break Bell
    on SPRINT → Lava Bell End
  }

  state Sprinting {
    GetSpeed2d(gameObject=var "Hero", storeResult=var "Hero Speed", everyFrame=true)
    FloatCompare(float1=var "Hero Speed", float2=13, tolerance=0, equal=(none), lessThan=→"SPRINT END", greaterThan=(none), everyFrame=true)
    SetPositionToObject(gameObject=var "Pt Smoke", targetObject=var "Hero", xOffset=0, yOffset=-1.8, zOffset=-0.002, overrideZ=(unset), everyFrame=true)
    SetPositionToObject(gameObject=var "Pt Sparks", targetObject=var "Hero", xOffset=0, yOffset=-1.8, zOffset=-0.003, overrideZ=(unset), everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Idle
    on SPRINT END → Lava Bell?
    on TRUE → No Sizzle
  }

  state Lava Bell End {
    SendEventToRegister(eventName="LAVA BELL SIZZLE STOP")
    CallMethodProper(gameObject=var "Lava Bell Shield", behaviour="LavaBellCoalRegionShield", methodName="End", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Sizzle Start
  }

  state Break Bell {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="UseLavaBell", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Lava Bell End
  }

  state Check Still Inside {
    Trigger2dEventWait(gameObject=Self, trigger=1, collideTag="", collideLayer=Hero Box(20), sendEvent=→"ENTER", storeCollider=var "", waitFrames=1, finishEvent=(none))
    on FINISHED → Idle
    on ENTER → No Sizzle
  }

  state Idle Inactive {
    Comment(comment="Only disable if has component that will wake it up. (in case template is used in other places not accounted for)")
    HasComponent(gameObject=Self, component="TriggerEnterEvent", removeOnExit=false, trueEvent=(none), falseEvent=(none), store=var "Will Disable", everyFrame=false)
    BoolFlip(boolVariable=var "Will Disable")
    EnableFsmSelf(SetEnabled=var "Will Disable")
    on ENTER → Lava Bell?
  }

  state Sizzle Start {
    AddStatusVignetteStatus(Status=InCoalRegion, StoreSet=var "Did Set Vignetta", InState=false)
    on FINISHED → Sizzle Short
  }

  var Hero Speed: float = 0
  var Is Bell Active: bool = false
  var Will Disable: bool = false
  var Did Set Vignetta: bool = false
  var Pt Smoke: gameObject = <null>
  var Pt Sparks: gameObject = <null>
  var Lava Bell Shield: gameObject = <null>
  var Self: gameObject = <null>
}
