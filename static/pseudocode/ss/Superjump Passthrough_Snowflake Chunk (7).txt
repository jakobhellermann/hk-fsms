# Superjump Passthrough
# 1 location(s):
#   Peak_05c / Snowflake Chunk (7)

fsm Superjump Passthrough {
  start Idle

  state Idle {
    SetLayer(gameObject=var "Collider Core", layer=Terrain(8))
    SetLayer(gameObject=var "Collider Outer", layer=Terrain(8))
    EnableFsmSelf(SetEnabled=false)
    on SUPER JUMP THROW NEEDLE → Check Y
  }

  state Super Jumping {
    SetLayer(gameObject=var "Collider Core", layer=Terrain Detector(14))
    SetLayer(gameObject=var "Collider Outer", layer=Terrain Detector(14))
    EnableFsmSelf(SetEnabled=false)
    on SUPER JUMP ENDED → Idle
  }

  state Check Y {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"CANCEL", belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on FINISHED → Super Jumping
    on CANCEL → Idle
  }

  var Collider Core: gameObject = Snowflake Chunk (7)/Active/Collider Core
  var Collider Outer: gameObject = Snowflake Chunk (7)/Active/Collider Outer
}
