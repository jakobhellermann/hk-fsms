# Death
# 1 location(s):
#   localpoolprefabs_assets_areashellwood.bundle / Corpse Flower Queen

fsm Death {
  start Wait Frame

  state Stagger {
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[flower_medium_3 (sfxstatic_assets_areashellwood.bundle)], weights=[1], pitchMin=1, pitchMax=1, stopPreviousSound=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventToRegister(eventName="SPIKES DOWN")
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0.1)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashBossFinalHit())
    on FINISHED → Heart Death
  }

  state Steam {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Boss Death Antic (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, 0), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=boss_death_pt_2 (sfxstatic_assets_boss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Blow
  }

  state Blow {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    SpawnObjectFromGlobalPool(gameObject=Boss Death Explosion (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, 0), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Blow", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=boss_death_pt_3 (sfxstatic_assets_boss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Fall
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Stagger
  }

  state Fall {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Fall")
    SetVelocityByScale(gameObject=Self, speed=20, ySpeed=20, everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.95, decelerationY=(unset), brakeOnExit=true)
    CheckYPosition(gameObject=Self, compareTo=86.6, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"BREAK", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on BREAK → Break
  }

  state Break {
    SendEventToRegister(eventName="FINAL BREAK")
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAND → Land
  }

  state Land {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=false_knight_land (sfxstatic_assets_enemyheavy.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Land")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "X Pos", minValue=14, maxValue=23, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=(unset), X=var "X Pos", Y=var "Ground Y", Space=0, EveryFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Activate NPC
  }

  state Activate NPC {
    ActivateGameObject(gameObject=var "NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on INTERACT → 
  }

  state Heart Death {
    SetPlayerDataBool(boolName="defeatedFlowerQueen", value=true)
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[flower_scream_long (sfxstatic_assets_areashellwood.bundle)], weights=[1], pitchMin=1, pitchMax=1, stopPreviousSound=true)
    RunFSM(fsmTemplateControl=template=6292096786850507184 in[Boss Type<-"Flower Queen"], finishEvent=(none), everyFrame=false)
  }

  var Ground Y: float = 6.48
  var X Pos: float = 0
  var Pt Blow: gameObject = Corpse Flower Queen/Pt Blow
  var Self: gameObject = Corpse Flower Queen
  var NPC: gameObject = Corpse Flower Queen/NPC
  var Audio Loop Voice: gameObject = Corpse Flower Queen/Audio Loop Voice
}
