# Control
# 1 location(s):
#   Ward_03 / Song Creeper Wake

fsm Control {
  start State 1

  state State 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State 2
  }

  state State 2 {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on ACTIVATED → State 3
    on CREEPER WAKE → State 4
  }

  state State 3 {
    SendEventByName(eventTarget=GameObject(var "Creeper"), sendEvent="SOFT WAKE", delay=0, everyFrame=false)
    Translate(gameObject=var "Creeper", vector=(unset), x=15, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
  }

  state State 4 {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  var Creeper: gameObject = Song Creeper

  var Activated: bool = false
}
