# Appear
# 1 location(s):
#   globalpoolprefabs_assets_all.bundle / Dazed Effect

fsm Appear {
  start Init
  on STUN EFFECT END → End  // from any state

  state Init {
    ActivateGameObject(gameObject=var "Dazed Particle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dazed End Basic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)  // disabled
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Current Y", space=0, everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    EnableBehaviour(gameObject=Self, behaviour="", component=Dazed Effect@tk2dSpriteAnimator, enable=true, resetOnExit=false)  // disabled
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect Y
  }

  state Detect Y {
    SetFloatValue(floatVariable=var "Prev Y", floatValue=var "Current Y", everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Current Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Current Y", float2=0, tolerance=0.01, equal=→"LANDED", lessThan=(none), greaterThan=(none), everyFrame=true)
    GetGravity2dScale(gameObject=var "Parent", storeGravity=var "Parent Gravity", everyFrame=false)
    FloatCompare(float1=var "Parent Gravity", float2=0, tolerance=0.1, equal=→"LANDED", lessThan=→"LANDED", greaterThan=(none), everyFrame=false)
    on LANDED → Play
  }

  state Play {
    EnableBehaviour(gameObject=Self, behaviour="", component=Dazed Effect@tk2dSpriteAnimator, enable=true, resetOnExit=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Dazed")  // disabled
    Tk2dPlayFrame(gameObject=Self, frame=0)  // disabled
    SetMeshRenderer(gameObject=Self, active=true)  // disabled
    ActivateGameObject(gameObject=var "Dazed Particle", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
  }

  sequence state End {
    ActivateGameObject(gameObject=var "Dazed End Basic", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    RecycleSelf()
  }

  var Current Y: float = 0
  var Prev Y: float = 0
  var Parent Gravity: float = 0
  var Parent: gameObject = <null>
  var Dazed Particle: object = Dazed Effect/Dazed Particle
  var Dazed End Basic: object = Dazed Effect/Dazed End Basic
}
