# Spin
# 1 location(s):
#   Bone_01 / Great Bone Gate Scene/Great Bone Gate/Wheel 1

fsm Spin {
  start Idle

  state Idle {
    on SPIN → Spinning
  }

  state Spinning {
    SetFloatValue(floatVariable=var "Spin Crt", floatValue=0, everyFrame=false)
    EaseFloat(fromValue=0, toValue=var "Spin", floatVariable=var "Spin Crt", time=1, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin Crt", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on STOP → Stop
  }

  state Stop {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin Crt", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    EaseFloat(fromValue=var "Spin Crt", toValue=0, floatVariable=var "Spin Crt", time=0.5, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Spin: float = -200

  var Spin Crt: float = 0
}
