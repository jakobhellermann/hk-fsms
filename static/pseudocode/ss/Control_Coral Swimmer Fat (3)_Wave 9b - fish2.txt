# Control
# 1 location(s):
#   Memory_Coral_Tower / Battle Scenes/Battle Scene Chamber 3/Wave 9b - fish2/Coral Swimmer Fat (3)

fsm Control {
  start Init
  on ZERO HP → Death Check  // from any state

  state Init {
    SetParent(gameObject=var "Patrol Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Start Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetPosition2D(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    BoolTest(boolVariable=var "Spear Spawner", isTrue=→"SPEAR SPAWNER", isFalse=(none), everyFrame=false)
    on FINISHED → Set Respawner
    on SPEAR SPAWNER → Spear Spawn Pause
  }

  state To Patrol {
    TiltBySpeedY(gameObject=Self, tiltFactor=-3, tiltMax=15, rotationSpeed=50)
    SetBoolValue(boolVariable=var "Returning", boolValue=false, everyFrame=false)
    ChaseObjectSpread(gameObject=Self, target=var "Patrol Point", speedMax=3, acceleration=0.15, targetSpreadX=0, targetSpreadY=2.5, spreadResetTimeMin=1, spreadResetTimeMax=2, offsetX=0, offsetY=0)
    GetXDistance(gameObject=Self, target=var "Patrol Point", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Fat TurnToSwim", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    ActivateGameObjectDelay(gameObject=var "Attack Range Obj", activate=true, resetOnExit=false, delay=0.5)
    on SHIFT → To Start
    on ATTACK → Attack Antic
  }

  state To Start {
    TiltBySpeedY(gameObject=Self, tiltFactor=-3, tiltMax=15, rotationSpeed=50)
    SetBoolValue(boolVariable=var "Returning", boolValue=true, everyFrame=false)
    ChaseObjectSpread(gameObject=Self, target=var "Start Point", speedMax=3, acceleration=0.15, targetSpreadX=0, targetSpreadY=2.5, spreadResetTimeMin=1, spreadResetTimeMax=2, offsetX=0, offsetY=0)
    GetXDistance(gameObject=Self, target=var "Start Point", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Fat TurnToSwim", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    ActivateGameObjectDelay(gameObject=var "Attack Range Obj", activate=true, resetOnExit=false, delay=0.5)
    on SHIFT → To Patrol
    on ATTACK → Attack Antic
  }

  state Set Respawner {
    BoolTest(boolVariable=var "Respawner", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetSpecialDeath(target=Self, hasSpecialDeath=true)
    on FINISHED → To Patrol
  }

  state Death Check {
    BoolTest(boolVariable=var "Respawner", isTrue=→"RESPAWNER", isFalse=(none), everyFrame=false)
    on RESPAWNER → Die
  }

  state Die {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    GetDistance(gameObject=Self, target=var "MainCamera", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=45, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"RESPAWN", everyFrame=true)
    on RESPAWN → Respawn
  }

  state Respawn {
    SetMeshRendererChildren(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    CallMethodProper(gameObject=Self, behaviour="HealthManager", methodName="RefillHP", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPosition2D(GameObject=Self, Vector=var "Start Pos", X=(unset), Y=(unset), Space=0, EveryFrame=false)
    on FINISHED → To Patrol
  }

  state Spear Spawn Pause {
    GetPosition2D(gameObject=Self, vector=var "Spawn Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=(unset), X=-100, Y=100, Space=0, EveryFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    GetConstantsValue(variableName="CORAL_SPEAR_SPAWN_TIME", storeValue=var "Spawn Time")
    Wait(time=var "Spawn Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst Out
  }

  state Burst Out {
    SetPosition2D(GameObject=Self, Vector=var "Spawn Pos", X=(unset), Y=(unset), Space=0, EveryFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=2, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    on FINISHED → To Patrol
  }

  state Attack Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fat Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    PlayAudioEvent(audioClip=tar_slug_antic (sfxstatic_assets_areadocks.bundle), pitchMin=1.2, pitchMax=1.4, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    on FINISHED → Y Charge
  }

  state Attack {
    PlayAudioEvent(audioClip=coral_poke_swimmer_attack (sfxstatic_assets_areacoral.bundle), pitchMin=1.2, pitchMax=1.4, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fat Attack", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocityByScale(gameObject=Self, speed=-30, ySpeed=(unset), everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.925, brakeOnExit=false)
    ActivateGameObject(gameObject=var "Attack Range Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Check Return
  }

  state Check Return {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fat Swim")
    CheckFacingTarget(facingObject=Self, target=var "Patrol Point", spriteFacesRight=false, everyFrame=false, facingEvent=→"PATROL", notFacingEvent=(none), facingBool=(unset), notFacingBool=(unset))
    on PATROL → To Patrol
    on FINISHED → To Start
  }

  state Y Charge {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"CHARGE UP", belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=2, reverseIfNegativeScale=false, everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=→"CHARGE DOWN", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=-2, reverseIfNegativeScale=false, everyFrame=false)
    on CHARGE UP → Charge Up
    on CHARGE DOWN → Charge Down
    on FINISHED → Attack
  }

  state Charge Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=10, everyFrame=false)
    on FINISHED → Attack
  }

  state Charge Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-10, everyFrame=false)
    on FINISHED → Attack
  }

  var Respawner: bool = false
  var Spear Spawner: bool = true

  var Distance: float = 0
  var Spawn Time: float = 0
  var Returning: bool = false
  var Start Pos: vector2 = (0, 0)
  var Spawn Pos: vector2 = (0, 0)
  var Patrol Point: gameObject = Battle Scenes/Battle Scene Chamber 3/Wave 9b - fish2/Coral Swimmer Fat (3)/Patrol Point
  var Start Point: gameObject = Battle Scenes/Battle Scene Chamber 3/Wave 9b - fish2/Coral Swimmer Fat (3)/Start Point
  var Attack Range Obj: gameObject = Battle Scenes/Battle Scene Chamber 3/Wave 9b - fish2/Coral Swimmer Fat (3)/Attack Range
}
