# Control

fsm Control {
  start Init
  on ZERO HP → Die  // from any state
  on BOSS DEFEATED → Boss Defeated  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Attack Circle", storeResult=var "Attack Circle")
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Enemy Container")
    on FINISHED → Dormant
  }

  state Attack Antic {
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Attack
  }

  state Attack {
    PlayAudioEvent(audioClip=bone_circler_vicious_fly_attack (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Attack")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)

    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Attack Scale", floatValue=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Attack Circle", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    EaseFloat(fromValue=1, toValue=1.4, floatVariable=var "Attack Scale", time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)

    Tk2dPlayAnimationWithEvents(gameObject=var "Attack Circle", clipName="Attack Circle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FireAtTarget(gameObject=Self, target=var "Hero", speed=5.5, position=(0, 0.5, 0), spread=0.5, everyFrame=false)
    on FINISHED → Attack End
  }

  state Attack End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Chase
  }

  state Chase {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Aggro")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToAggro", resetFrame=false, everyFrame=true)
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=4, acceleration=0.1, targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0, offsetX=0, offsetY=0)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SING → Sing
    on ATTACK → Attack Antic
  }

  state Chase Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Aggro")
    on FINISHED → Chase
  }

  state Sing {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)

    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Chase Start
  }

  state Dormant {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)  // disabled
    on SUMMON → Swoop In
  }

  state Swoop In {
    SetBoolValue(boolVariable=var "Active", boolValue=true, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)

    EaseFloat(fromValue=var "Z Pos", toValue=var "Z Target", floatVariable=var "Z Tween", time=0.5, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)

    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=25, everyFrame=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Chase
  }

  state Die {
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)

    SetVelocityByScale(gameObject=Self, speed=15, ySpeed=15, everyFrame=false)

  }

  state Reset {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetParent(gameObject=Self, parent=var "Enemy Container", resetLocalPosition=false, resetLocalRotation=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=HealToMax())
    on FINISHED → Dormant
  }

  state Boss Defeated {
    BoolTest(boolVariable=var "Active", isTrue=→"DIE", isFalse=(none), everyFrame=false)
    on DIE → Die
  }

  var Attack Scale: float = 0
  var Chase Distance: float = 12
  var Distance: float = 0
  var Z Pos: float = 0
  var Z Target: float = 0
  var Z Tween: float = 0
  var Alert: bool = false
  var Range Out: bool = false
  var Active: bool = false
  var Attack Circle: gameObject = <null>
  var Home: gameObject = <null>
  var Self: gameObject = <null>
  var Terrain Box: gameObject = Boss Control/Boss Scene/Summon Enemies/Bone Circler Vicious Smn/Terrain Box
  var Enemy Container: gameObject = <null>
  var Audio Loop Sing: gameObject = Boss Control/Boss Scene/Summon Enemies/Bone Circler Vicious Smn/Audio Loop Sing
}
