# Behaviour
# 2 location(s):
#   Hang_14 / laundry_basket (1)
#   Hang_16 / laundry_basket (1)

fsm Behaviour {
  start Init

  state Idle {
    GetPosition(gameObject=var "Tilt Collider", vector=(unset), x=var "Tilt Collider X", y=(unset), z=(unset), space=0, everyFrame=true)
    SetFsmFloat(gameObject=var "Tilt Collider", fsmName="tilt_plat", variableName="Self X", setValue=var "Tilt Collider X", everyFrame=true)
    Collision2dEvent(gameObject=var "Tilt Collider", collision=0, collideTag="Player", sendEvent=→"LAND", storeCollider=var "", storeForce=(unset))
    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
    on TINK UP → Tink From Hero Dir
    on TINK DOWN → Tink From Hero Dir
    on LAND → Decelerate
    on SPRINT BONKED → Tink From Hero Dir
  }

  state Tink Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-5, y=(unset), everyFrame=false)
    on FINISHED → Tinked
  }

  state Tink Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=5, y=(unset), everyFrame=false)
    on FINISHED → Tinked
  }

  state Init {
    on FINISHED → Idle
  }

  state Tink From Hero Dir {
    SendEventToRegister(eventName="LAUNDRY BONK")
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "Self X", tolerance=0, equal=→"TINK RIGHT", lessThan=→"TINK RIGHT", greaterThan=→"TINK LEFT", everyFrame=false)
    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
  }

  state Decelerate {
    Collision2dEvent(gameObject=var "Tilt Collider", collision=2, collideTag="Player", sendEvent=→"LEAVE", storeCollider=var "", storeForce=(unset))
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=(unset), brakeOnExit=false)
    on LEAVE → Idle
  }

  state Tinked {
    CallMethodProper(gameObject=laundry_basket (1)/Hit Effect Pool, behaviour="ParticleSystemPool", methodName="PlayParticles", parameters=[], storeResult=(unset var), EveryFrame=false)
    PlayAudioEventRandom(audioClips=[laundry_basket_hit_01 (sfxstatic_assets_areahang.bundle), laundry_basket_hit_02 (sfxstatic_assets_areahang.bundle), laundry_basket_hit_03 (sfxstatic_assets_areahang.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "Wheel Audio"), sendEvent="WHEEL START", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Hero X: float = 0
  var Self X: float = 0
  var Tilt Collider X: float = 0
  var Tilt Collider: gameObject = laundry_basket (1)/Top/top collider
  var Wheel Audio: gameObject = laundry_basket (1)/Wheel Audio
}
