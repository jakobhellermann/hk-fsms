# Tink Hit Force
# 1 location(s):
#   Dust_Chef / Battle Parent/Kitchen Pipe Gong/kitchen_string_offset/kitchen_string/kitchen_gong

fsm Tink Hit Force {
  start Inert

  state Inert {
    on TINK LEFT → Add Force Left
    on TINK RIGHT → Add Force Right
    on TINK UP → Random
    on TINK DOWN → Random
  }

  state Add Force Left {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=-30, y=10, vector3=(unset), everyFrame=false)
    on FINISHED → Hit
  }

  state Add Force Right {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=30, y=10, vector3=(unset), everyFrame=false)
    on FINISHED → Hit
  }

  state Random {
    SendRandomEvent(events=[→"TINK LEFT", →"TINK RIGHT"], weights=[1, 1], delay=0)
    on TINK LEFT → Add Force Left
    on TINK RIGHT → Add Force Right
  }

  state Hit {
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    PlayAudioEvent(audioClip=bellshrine_sword_hit (sfxstatic_assets_areabellshrine.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventToRegister(eventName="GONG HIT")
    on FINISHED → Inert
  }
}
