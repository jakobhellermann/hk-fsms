# Wobble
# 1 location(s):
#   Belltown_08 / Boss Scene/Wisp Pyre Effigy

fsm Wobble {
  start Idle

  state Idle {
    on WOBBLE L → Wobble L
    on WOBBLE R → Wobble R
  }

  state Wobble L {
    AnimateRotationTo(gameObject=Self, fromValue=0, toValue=2, worldSpace=false, negativeSpace=false, time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Start Rotation", floatValue=3, everyFrame=false)
    on FINISHED → Wobble Back
  }

  state Wobble Back {
    AnimateRotationTo(gameObject=Self, fromValue=var "Start Rotation", toValue=var "Idle Rotation", worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=22, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on WOBBLE L → Wobble L
    on WOBBLE R → Wobble R
  }

  state Wobble R {
    AnimateRotationTo(gameObject=Self, fromValue=0, toValue=-2, worldSpace=false, negativeSpace=false, time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Start Rotation", floatValue=-3, everyFrame=false)
    on FINISHED → Wobble Back
  }

  var Hit Rotation: float = 0

  var Idle Rotation: float = 0
  var Start Rotation: float = 0
}
