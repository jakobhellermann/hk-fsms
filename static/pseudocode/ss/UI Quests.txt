fsm UI Quests {
  start Init
  on PANE RESET → Inactive  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Inactive
  }

  state Inactive {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UI INACTIVE", delay=0, everyFrame=false)
    ActivateInventoryPaneInput(Target=Self, Activate=false)
    on ACTIVATE → Activate
  }

  state Activate {
    SendEventByName(eventTarget=GameObject(var "Self"), sendEvent="UI ACTIVE", delay=0, everyFrame=false)
    ActivateInventoryPaneInput(Target=Self, Activate=true)
  }

  var Completion percentage Float: float = 0
  var Heart Pieces Int: int = 0
  var Nail Upgrades: int = 0
  var Only Pane: bool = false

  var H Pieces Amount String: string = ""
  var Item Pos: vector3 = (0, 0, 0)
  var Completion Percentage: gameObject = <null>

  var Cursor: gameObject = <null>
  var Dream Nail: gameObject = <null>
  var Equipment: gameObject = <null>
  var Geo: gameObject = <null>

  var Inv_Items: gameObject = <null>
  var Nail: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
}
