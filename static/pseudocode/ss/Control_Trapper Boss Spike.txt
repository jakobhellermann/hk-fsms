    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Retract
    on TINKED HEAVY → Retract
  }

  state Antic {
    SetPosition2D(GameObject=Self, Vector=(unset), X=var "Trap X", Y=(unset), Space=0, EveryFrame=false)

    on R → R

  }

    on FINISHED → Antic Sfx
  }

  state R {
    FloatOperator(float1=var "Init X", float2=3, operation=Add, storeResult=var "Trap X", everyFrame=false)
    on FINISHED → Antic Sfx
  }

  state Trap Pause {
