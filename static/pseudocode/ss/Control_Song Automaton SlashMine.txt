# Control
# 1 location(s):
#   localpoolprefabs_assets_areacog.bundle / Song Automaton SlashMine

fsm Control {
  start Init
  on HEAVY → Heavy Break  // from any state

  state Init {
    SetScale(gameObject=Self, vector=(unset), x=1.1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomlyFlipScale(gameObject=Self)
    RandomFloat(min=1.15, max=1.75, storeResult=var "Detonate Time")
    SetVelocityAsAngle(gameObject=Self, angle=45, speed=10, everyFrame=false)  // disabled
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    ActivateGameObject(gameObject=var "Tink Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hero Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "automaton_mine_antic_effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Air
  }

  state Air {
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"WALL L", rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"WALL L", rightHitEvent=→"WALL R", bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    GetVelocity2dNotZero(gameObject=Self, vector=(unset), x=var "X Velocity", y=var "Y Velocity", space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Detonate Time", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Detonate Time", float2=0, tolerance=0, equal=(none), lessThan=→"ANTIC", greaterThan=(none), everyFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)  // disabled
    on LAND → Land
    on WALL L → Wall L
    on WALL R → Wall R
    on TINKED → Antic
    on ANTIC → Antic
    on MULTI HIT CONNECT → Quick Detonate
  }

  state Land {
    FloatMultiplyV2(floatVariable=var "Y Velocity", multiplyBy=-0.8, everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Velocity", everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Bounce Effects
  }

  state Bounce Effects {
    PlayRandomAudioClipTable(Table=metal_bounce (sfxdynamic_assets_areagreymoor.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Air
  }

  state Wall L {
    FloatMultiplyV2(floatVariable=var "X Velocity", multiplyBy=-1, everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=-0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Bounce Effects
  }

  state Wall R {
    FloatMultiplyV2(floatVariable=var "X Velocity", multiplyBy=-1, everyFrame=false, fixedUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Bounce Effects
  }

  state Antic {
    RandomFloat(min=0.003, max=0.0039, storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Mine Antic")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    PlayAudioEvent(audioClip=song_automaton_fly_antic (sfxstatic_assets_areacog.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "automaton_mine_antic_effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ShoveFromWall(gameObject=Self, shoveForce=var "Shove Force", rayLength=2, checkUp=true, checkDown=true, checkLeft=true, checkRight=true, everyFrame=true)
    on FINISHED → Detonate
    on MULTI HIT CONNECT → Quick Detonate
  }

  state Detonate {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    RandomlyFlipScale(gameObject=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mine Detonate", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    PlayAudioEvent(audioClip=bone_circler_vicious_fly_attack (sfxstatic_assets_areabone.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    ShoveFromWall(gameObject=Self, shoveForce=var "Shove Force", rayLength=2, checkUp=true, checkDown=true, checkLeft=true, checkRight=true, everyFrame=true)
    on FINISHED → Slash
    on MULTI HIT CONNECT → Quick Detonate
  }

  state Slash {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Slash Collider", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Tink Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    ShoveFromWall(gameObject=Self, shoveForce=var "Shove Force", rayLength=2, checkUp=true, checkDown=true, checkLeft=true, checkRight=true, everyFrame=true)
    ActivateGameObject(gameObject=var "Hero Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Slash End
    on MULTI HIT CONNECT → Multihit
  }

  state Slash End {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Multihit {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Mine Detonate", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    ActivateGameObject(gameObject=var "Hero Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Quick Detonate {
    PlayAudioEvent(audioClip=bone_circler_vicious_fly_attack (sfxstatic_assets_areabone.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Tink Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RandomFloat(min=0.003, max=0.0039, storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "automaton_mine_antic_effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Multihit
  }

  state Heavy Break {
    PlayRandomAudioClipTable(Table=metal_break_destroy (sfxdynamic_assets_areagreymoor.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SetMeshRenderer(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Hero Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slash Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Tink Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "automaton_mine_antic_effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Detonate Time: float = 0
  var X Velocity: float = 0
  var Y Velocity: float = 0
  var Z Pos: float = 0
  var Shove Force: float = 30
  var Slash Collider: gameObject = Song Automaton SlashMine/Slash Collider
  var automaton_mine_antic_effect: gameObject = Song Automaton SlashMine/automaton_mine_antic_effect
  var Tink Detector: gameObject = Song Automaton SlashMine/Tink Detector
  var Break Effects: gameObject = Song Automaton SlashMine/Break Effects
  var Hero Detector: gameObject = Song Automaton SlashMine/Hero Detector
}
