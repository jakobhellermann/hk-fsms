# Control
# 1 location(s):
#   localpoolprefabs_assets_areapeak.bundle / Crystal Drifter Giant Spike

fsm Control {
  start Init

  state Grow {
    PlayAudioEvent(audioClip=crystal_drifter_giant_attack (sfxstatic_assets_areapeak.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Spike Grow", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetMeshRenderer(gameObject=var "Sprite", active=true)
    on FINISHED → Antic
    on FIRE → Fire
  }

  state Antic {
    ObjectJitterLocal(gameObject=Self, x=0.03, y=0.03, z=(unset), resetPosOnStateExit=true, limitFps=0)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slash Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Slash Detector", active=true, resetOnExit=false)
    Trigger2dEvent(gameObject=var "Spawn Terrain Detector", trigger=1, collideTag="", sendEvent=→"BREAK", storeCollider=var "")
    Wait(time=0.75, finishEvent=→"BREAK", realTime=false)
    on FIRE → Break
    on BREAK → Break
  }

  state Fire {
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slash Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Spike Fly")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=25, everyFrame=false)
    Wait(time=5, finishEvent=→"END", realTime=false)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"THUNK", storeCollider=var "", storeForce=(unset))
    on THUNK → Thunk
    on END → Break
    on BREAK → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    PlayAudioEvent(audioClip=crystal_drifter_spikes_dissolve (sfxstatic_assets_areapeak.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    ActivateGameObject(gameObject=var "Spark", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Slash Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Thunk {
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Break
  }

  state Init {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    ActivateGameObject(gameObject=var "Spark", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Grow
  }

  var Angle: float = 0
  var Pt Shatter: gameObject = Crystal Drifter Giant Spike/Pt Shatter
  var Self: gameObject = <null>
  var Spark: gameObject = Crystal Drifter Giant Spike/Spark
  var Terrain Detector: gameObject = Crystal Drifter Giant Spike/Terrain Detector
  var Thunk Effect: gameObject = Crystal Drifter Giant Spike/Thunk Effect
  var Sprite: gameObject = Crystal Drifter Giant Spike/Sprite
  var Spawn Terrain Detector: gameObject = Crystal Drifter Giant Spike/Spawn Terrain Detector
  var Slash Detector: gameObject = Crystal Drifter Giant Spike/Slash Breaker
}
