# Control
# 4 location(s):
#   Clover_01b / Grass Goomba
#   Clover_02c / Grass Goomba
#   Clover_03 / Grass Goomba
#   Clover_16 / Grass Goomba

fsm Control {
  start Init

  state Init {
    RandomlyFlipScale(gameObject=Self)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State
  }

  state Walk {
    AudioPlayInState(gameObject=Self, volume=1)
    WalkLeftRight(gameObject=Self, walkSpeed=3, spriteFacesLeft=true, groundLayer="Terrain", turnDelay=0.35, wallLookAhead=0, walkAnimName="Run", turnAnimName="Turn", moveWhileTurning=true, startLeft=false, startRight=false, keepDirection=true, flipBeforeTurn=true, turnAudioClipTable=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FLOAT", EveryFrame=true)
    WaitRandom(timeMin=0.5, timeMax=1.5, finishEvent=→"END", realTime=false)  // disabled
    ActivateGameObjectDelay(gameObject=var "Attack Range", activate=true, resetOnExit=false, delay=0.25)
    on ATTACK → Antic
    on SING → Sing
    on TOOK DAMAGE → Check Alert Range
    on FLOAT → Float
  }

  state Sing {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToSing", resetFrame=true, everyFrame=false, pauseBetweenTurns=0.5)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Walk
  }

  state Check Alert Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=false)
    on FINISHED → Walk
    on ATTACK → Antic
  }

  state Antic {
    ActivateGameObject(gameObject=var "Attack Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=grass_goomba_jump_antic (sfxstatic_assets_areawilds.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEventRandom(audioClips=[grass_goomba_attack_1 (sfxstatic_assets_areawilds.bundle), grass_goomba_attack_2 (sfxstatic_assets_areawilds.bundle), grass_goomba_attack_3 (sfxstatic_assets_areawilds.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fly Up
  }

  state Fly Up {
    PlayAudioEvent(audioClip=grass_goomba_jump (sfxstatic_assets_areawilds.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=30, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    AudioPlayInState(gameObject=var "Audio Loop Flutter", volume=1)
    on FINISHED → Charge
  }

  state Charge {
    PlayAudioEventRandom(audioClips=[grass_goomba_dash_attack_1 (sfxstatic_assets_areawilds.bundle), grass_goomba_dash_attack_2 (sfxstatic_assets_areawilds.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Charge")
    SetVelocityByScale(gameObject=Self, speed=-24, ySpeed=-16, everyFrame=true)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL", bottomHitEvent=(none), leftHitEvent=→"WALL", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    ActivateGameObject(gameObject=var "Dash Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"END", realTime=false)
    on LAND → Roll
    on END → Float
    on WALL → Wall Bonk
  }

  state Roll {
    PlayAudioEventRandom(audioClips=[grass_goomba_land_n_roll_1 (sfxstatic_assets_areawilds.bundle), grass_goomba_land_n_roll_2 (sfxstatic_assets_areawilds.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Velocity", multiplyBy=0.65, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roll")
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=10, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Speed", float2=5, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Roll End
  }

  state Roll End {
    DecelerateXY(gameObject=Self, decelerationX=0.95, decelerationY=(unset), brakeOnExit=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roll End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Walk
  }

  state Float {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.35)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    AudioPlayInState(gameObject=var "Audio Loop Flutter", volume=1)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state State {
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Hidden", equalEvent=→"HIDDEN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Clip", compareTo="Quick Wake", equalEvent=→"HIDDEN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Walk
    on HIDDEN → Set Hidden
  }

  state Hidden {
    WaitRandom(timeMin=4, timeMax=7, finishEvent=→"RUSTLE", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on RUSTLE → Rustle
    on WAKE → Emerge Antic
    on TOOK DAMAGE → Emerge
  }

  state Rustle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Rustle")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    PlayAudioEvent(audioClip=moss_knight_rustle (sfxstatic_assets_areashellwoodareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Hidden
  }

  state Set Hidden {
    SetDamageHero(Target=Self, Enabled=false)
    SetRecoilSpeed(target=Self, newRecoilSpeed=0)
    on FINISHED → Hidden
  }

  state Rustle Sing {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rustle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Still Singing?
  }

  state Still Singing? {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    on SING → Rustle Sing
    on FINISHED → Hidden
  }

  state Emerge Antic {
    StringCompare(stringVariable=var "Clip", compareTo="Quick Wake", equalEvent=→"QUICK WAKE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayAudioEventRandom(audioClips=[grass_goomba_alert_1 (sfxstatic_assets_areawilds.bundle), grass_goomba_alert_2 (sfxstatic_assets_areawilds.bundle), grass_goomba_alert_3 (sfxstatic_assets_areawilds.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=moss_knight_rustle (sfxstatic_assets_areashellwoodareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Emerge
    on QUICK WAKE → Quick Wake
  }

  state Emerge {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHero(Target=Self, Enabled=true)
    SetRecoilSpeed(target=Self, newRecoilSpeed=15)
    PlayAudioEvent(audioClip=moss_knight_rustle (sfxstatic_assets_areashellwoodareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Walk
  }

  state Idle {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FLOAT", EveryFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    WaitRandom(timeMin=0.5, timeMax=0.75, finishEvent=→"WALK", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateGameObjectDelay(gameObject=var "Attack Range", activate=true, resetOnExit=false, delay=0.25)
    on FLOAT → Float
    on ATTACK → Antic
    on TOOK DAMAGE → Check Alert Range
    on SING → Sing
    on WALK → Walk
  }

  state Wall Bonk {
    PlayAudioEvent(audioClip=hornet_hit_reject_dull (herosfxstatic_assets_all.bundle), pitchMin=1.15, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wall Bonk")
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SetVelocityByScale(gameObject=Self, speed=5, ySpeed=5, everyFrame=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    on LAND → Roll
  }

  state Quick Wake {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHero(Target=Self, Enabled=true)
    SetRecoilSpeed(target=Self, newRecoilSpeed=15)
    on FINISHED → Walk
  }

  var Speed: float = 0
  var X Velocity: float = 0
  var Clip: string = ""
  var Dash Effect: gameObject = Grass Goomba/Dash Effect
  var Audio Loop Flutter: gameObject = Grass Goomba/Audio Loop Flutter
  var Attack Range: gameObject = Grass Goomba/Attack Range
}
