# Control
# 1 location(s):
#   localpoolprefabs_assets_areaswamp.bundle / Swamp Muckman Dart

fsm Control {
  start Init
  on PROJECTILE BREAK → Break  // from any state

  state Init {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetDamageHero(Target=Self, Enabled=true)
    ActivateGameObject(gameObject=var "Splasher", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fly
  }

  state Fly {
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"HIT TERRAIN", storeCollider=var "", storeForce=(unset))
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    FaceAngle(gameObject=Self, angleOffset=0, angleOffsetIfMirrored=180, otherGameObject=<null>, everyFrame=true)
    Wait(time=3, finishEvent=→"BREAK", realTime=false)
    Trigger2dEventV2(gameObject=Self, trigger=0, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"WATER", storeCollider=var "")
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"BREAK", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT TERRAIN → Stick
    on PARRIED → Check
    on BREAK → Break
    on WATER → Splash In
    on DAMAGED HERO → Break
  }

  state Stick {
    PlayAudioEventRandom(audioClips=[swamp_stake_shooter_land_1 (sfxstatic_assets_areaswamp.bundle), swamp_stake_shooter_land_2 (sfxstatic_assets_areaswamp.bundle), swamp_stake_shooter_land_3 (sfxstatic_assets_areaswamp.bundle), swamp_stake_shooter_land_4 (sfxstatic_assets_areaswamp.bundle)], pitchMin=1, pitchMax=1.25, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Dart Hit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=0.7)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"BREAK", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=true, ignoreHunterWeapon=false, ignoreTraps=true, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetDamageHero(Target=Self, Enabled=false)
    on BREAK → Break
    on FINISHED → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    AudioStop(gameObject=Self, fadeTime=0)
    ActivateGameObject(gameObject=var "Splasher", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayAudioEvent(audioClip=swamp_muckman_projectile_dart_explode (sfxstatic_assets_areaswamp.bundle), pitchMin=0.85, pitchMax=1.15, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[wooden_stake_break_1 (sfxstatic_assets_areashellwood.bundle), wooden_stake_break_2 (sfxstatic_assets_areashellwood.bundle), wooden_stake_break_3 (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.9, pitchMax=1.25, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=true, recursive=false, resetOnExit=true, everyFrame=false)  // disabled
    SetDamageHero(Target=Self, Enabled=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Swamp Muckman_Dart_Explosion, spawnPoint=var "Explosion Point", position=(unset), rotation=(unset), storeObject=var "")
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    on FINISHED → Recycle
  }

  state Deflect {
    FloatAdd(floatVariable=var "Angle", add=var "Angle Mod", everyFrame=false, perSecond=false)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=2, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    SetDamageHero(Target=Self, Enabled=false)
    on FINISHED → Fly
  }

  state Check {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    FloatCompare(float1=var "Angle", float2=0, tolerance=1, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Angle", float2=180, tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on L → Deflect L
    on R → Deflect R
    on FINISHED → Deflect Other
  }

  state Deflect L {
    RandomFloat(min=-10, max=-13, storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Deflect R {
    RandomFloat(min=10, max=13, storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Deflect Other {
    RandomFloat(min=5, max=10, storeResult=var "Angle Mod")
    RandomlyFlipFloat(storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Splash In {
    AccelerateToY(gameObject=Self, accelerationFactor=2, targetSpeed=-20)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  var Angle: float = 0
  var Angle Mod: float = 0
  var Speed: float = 0
  var Break Effect: gameObject = Swamp Muckman Dart/Break Effects
  var Pt Break: gameObject = Swamp Muckman Dart/Pt Break
  var Thunk Effect: gameObject = Swamp Muckman Dart/Thunk Effect
  var Explosion Point: gameObject = Swamp Muckman Dart/Explosion Point
  var Splasher: gameObject = Swamp Muckman Dart/Splasher
}
