# Strike
# 1 location(s):
#   globalpoolprefabs_assets_all.bundle / Strike Nail R

fsm Strike {
  start Vibrate

  state Effect {
    RandomFloat(min=1.5, max=1.6, storeResult=var "Randomiser")
    SetScale(gameObject=Self, vector=(unset), x=var "Randomiser", y=var "Randomiser", z=(unset), everyFrame=false, lateUpdate=false)
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sword")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"WAIT")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on WAIT → End
  }

  state End {
    RecycleSelf()
  }

  state Vibrate {
    CheckSceneName(sceneName="Menu_Title", equalEvent=→"FINISHED", notEqualEvent=(none))
    VibrationPlayerPlay(target=Self)
    on FINISHED → Effect
  }

  var Randomiser: float = 0
}
