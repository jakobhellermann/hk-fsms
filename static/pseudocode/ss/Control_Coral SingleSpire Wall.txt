# Control
# 1 location(s):
#   localpoolprefabs_assets_areacoralareamemory.bundle / Coral SingleSpire Wall

fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Spire Extra L", active=false)
    SetMeshRenderer(gameObject=var "Spire Extra R", active=false)
    on FINISHED → Antic
  }

  state Antic {
    PlayAudioEvent(audioClip=rock_roller_wall_disturbed (sfxstatic_assets_areabone.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitterInState(gameObject=var "Pt AnticDust")
    PlayParticleEmitterInState(gameObject=var "Pt AnticRock")
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Pt CreepDust"), sendEvent="END", delay=0, everyFrame=false)
    on FINISHED → Up
  }

  state Up {
    PlayAudioEvent(audioClip=coral_knight_small_ceiling_coral_appear (sfxstatic_assets_areacoralareamemory.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="SingleSpire Up", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayAnimationWithEvents(gameObject=var "Spire Extra L", clipName="SingleSpire Up", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayAnimationWithEvents(gameObject=var "Spire Extra R", clipName="SingleSpire Up", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=var "Spire Extra L", active=true)
    SetMeshRenderer(gameObject=var "Spire Extra R", active=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Active
    on CORAL BREAK → Smash
  }

  state Active {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    Wait(time=1.25, finishEvent=→"NEXT", realTime=false)
    on NEXT → Break
    on CORAL BREAK → Smash
  }

  state Break {
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="SingleSpire Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimationWithEvents(gameObject=var "Spire Extra L", clipName="SingleSpire Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimationWithEvents(gameObject=var "Spire Extra R", clipName="SingleSpire Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayAudioEvent(audioClip=coral_wall_break (sfxstatic_assets_areacoral.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Particle End
  }

  state Ground Check {
    SetPositionToObject(gameObject=Self, targetObject=var "Hero", xOffset=0, yOffset=0, zOffset=-0.0005, overrideZ=(unset), everyFrame=false)
    RayCast2dV2(fromGameObject=var "Edge Check 1", fromPosition=(unset), direction=(0, -1), space=1, distance=10, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "Edge 1 Found", storeHitObject=var "", storeHitPoint=var "Hit Point 1", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "Edge 1 Found", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    RayCast2dV2(fromGameObject=var "Edge Check 2", fromPosition=(unset), direction=(0, -1), space=1, distance=10, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "Edge 2 Found", storeHitObject=var "", storeHitPoint=var "Hit Point 2", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "Edge 2 Found", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    GetVector2XY(vector2Variable=var "Hit Point 1", storeX=(unset), storeY=var "Hit Y 1", everyFrame=false)
    GetVector2XY(vector2Variable=var "Hit Point 2", storeX=(unset), storeY=var "Hit Y 2", everyFrame=false)
    FloatCompare(float1=var "Hit Y 1", float2=var "Hit Y 2", tolerance=0.1, equal=→"FINISHED", lessThan=→"CANCEL", greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Appear Pause
    on CANCEL → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Appear Pause {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Hit Y 1", z=(unset), space=0, everyFrame=false, lateUpdate=false)  // disabled
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)  // disabled
    ActivateGameObject(gameObject=var "Pt CreepDust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    EaseFloat(fromValue=7, toValue=0, floatVariable=var "Dust X", time=0.3, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)  // disabled
    SetPosition(gameObject=var "Pt CreepDust", vector=(unset), x=var "Dust X", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)  // disabled
    on FINISHED → Antic
  }

  state Particle End {
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Spire Extra L", active=false)
    SetMeshRenderer(gameObject=var "Spire Extra R", active=false)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Smash {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Break
  }

  var Dust X: float = 0
  var Hit Y 1: float = 0
  var Hit Y 2: float = 0
  var Edge 1 Found: bool = false
  var Edge 2 Found: bool = false
  var Hit Point 1: vector2 = (0, 0)
  var Hit Point 2: vector2 = (0, 0)
  var Edge Check 1: gameObject = Coral SingleSpire Wall/Edge Check 1
  var Edge Check 2: gameObject = Coral SingleSpire Wall/Edge Check 2
  var Pt Break: gameObject = Coral SingleSpire Wall/Pt Break
  var Pt CreepDust: gameObject = Coral SingleSpire Wall/Pt CreepDust
  var Spire Extra R: gameObject = Coral SingleSpire Wall/Spire Extra R
  var Spire Extra L: gameObject = Coral SingleSpire Wall/Spire Extra L
  var Pt AnticDust: object = Coral SingleSpire Wall/Pt AnticDust
  var Pt AnticRock: object = Coral SingleSpire Wall/Pt AnticRock
  var Slam Effect: object = <null>
}
