# Damage Effects
# 1 location(s):
#   Cradle_Destroyed_Challenge_01 / Centipede Trap Control/Centipede Trap:0

fsm Damage Effects {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on TOOK DAMAGE → Spawn Single
    on TOOK HEAVY DAMAGE → Heavy Damage
  }

  state Spawn Single {
    PlayAudioEventRandom(audioClips=[bone_hunter_tiny_death_1 (sfxstatic_assets_areacradle.bundle), bone_hunter_tiny_death_2 (sfxstatic_assets_areacradle.bundle), bone_hunter_tiny_death_3 (sfxstatic_assets_areacradle.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    RandomFloat(min=-15, max=-20, storeResult=var "Speed")
    RecordJournalKill(Record=Centipede Trap (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
    FireAtTarget(gameObject=var "Corpse", target=var "Hero", speed=var "Speed", position=(0, 0, 0), spread=5, everyFrame=false)
    GetVelocity2d(gameObject=var "Corpse", vector=(unset), x=(unset), y=var "Speed Y", space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Speed Y", minValue=15, maxValue=30, everyFrame=false)
    SetVelocity2d(gameObject=var "Corpse", vector=(unset), x=(unset), y=var "Speed Y", everyFrame=false)
    on FINISHED → Cool Down
  }

  state Cool Down {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Heavy Damage {
    FlingObjectsFromGlobalPool(gameObject=Centipede Trap Corpse (enemycorpses_assets_areacradle.bundle), spawnPoint=var "Corpse Spawn Point", position=(unset), spawnMin=4, spawnMax=4, speedMin=15, speedMax=25, angleMin=40, angleMax=140, originVariationX=1, originVariationY=0.5, FSM="", FSMEvent="")
    PlayAudioEventRandom(audioClips=[bone_hunter_tiny_death_1 (sfxstatic_assets_areacradle.bundle), bone_hunter_tiny_death_2 (sfxstatic_assets_areacradle.bundle), bone_hunter_tiny_death_3 (sfxstatic_assets_areacradle.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    RecordJournalKill(Record=Centipede Trap (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
    RecordJournalKill(Record=Centipede Trap (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
    RecordJournalKill(Record=Centipede Trap (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
    RecordJournalKill(Record=Centipede Trap (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    on FINISHED → Cool Down
  }

  var Speed: float = 0
  var Speed Y: float = 0
  var Damager: gameObject = <null>
  var Corpse Spawn Point: gameObject = Centipede Trap Control/Centipede Trap:0/Corpse Spawn Point
  var Corpse: gameObject = <null>
}
