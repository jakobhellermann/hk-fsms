# Bind

// uses template: hornet_bind
fsm Bind {
  start Pause
  on FSM CANCEL → Cancel All  // from any state
  on LEAVING SCENE → Leave Scene Bind?  // from any state
  on FORCE CURSED BIND → Cursed Bind Start  // from any state
  on HERO DAMAGED → Witch Binding?  // from any state
  on BIND END → End Bind  // from any state
  on FORCE CURSED BIND ABYSS → Abyss Cursed Bind Start  // from any state
  on HERO GRAB BREAK → Witch Binding?  // from any state

  state Idle {

    SetBoolValue(boolVariable=var "Witch Binding", boolValue=false, everyFrame=false)

    ListenForCast(eventTarget=Self, wasPressed=→"BUTTON DOWN", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    SetBoolValue(boolVariable=var "Can Bind Break", boolValue=false, everyFrame=false)
    on BUTTON DOWN → Can Bind?
  }

  state Take Control {
    CallMethodProper(gameObject=Self, behaviour="HeroController", methodName="SetCState", parameters=["focusing", true], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RelinquishControlNotVelocity())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="BindCharge Ground")
    BoolTest(boolVariable=var "Is Warrior Equipped", isTrue=→"WARRIOR", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Is Witch Equipped", isTrue=→"WITCH", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Is Sprinting", isTrue=→"SPRINTING", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Is Shuttlecock", isTrue=→"SHUTTLECOCK", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "On Ground", isTrue=→"ON GROUND", isFalse=→"OFF GROUND", everyFrame=false)
    on ON GROUND → Start Bind Ground
    on SPRINTING → Sprint Adjust
    on SHUTTLECOCK → Shuttlecock
    on OFF GROUND → Off Ground
    on WITCH → Witch
    on WARRIOR → Warrior Kickup
  }

    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartAnimationControlToIdle())
    on FINISHED → Cooldown
  }
