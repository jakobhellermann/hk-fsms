# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene

fsm Control {
  start Pause

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Coral King")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Boss")
    FindNamedChild(gameObject=Self, storeResult=var "Entry Spears")
    FindNamedChild(gameObject=Self, storeResult=var "coral_king_final_throne")
    FindNamedChild(gameObject=var "coral_king_final_throne", storeResult=var "throne_full")
    on FINISHED → Ready
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Ready {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter
  }

  state Enter {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0.5)
    SendEventToRegister(eventName="BOSS SCENE START")
    ActivateGameObject(gameObject=var "CamLock Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    PlayerDataBoolTest(boolName="encounteredCoralKing", isTrue=→"ENCOUNTERED", isFalse=(none))
    on FINISHED → Pre Intro Roar
    on ENCOUNTERED → Encountered Pause
  }

  state Start Fight {
    SetPlayerDataBool(boolName="encounteredCoralKing", value=true)
    SendEventByName(eventTarget=GameObject(var "Coral King"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on END KING SCENE → Exit Memory
  }

  state Spears Up {
    ActivateAllChildren(gameObject=var "Entry Spears", activate=true)
    Wait(time=2.7, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    on FINISHED → Start Fight
  }

  sequence state Exit Memory {
    StartPreloadingScene(SceneName="Coral_Tower_01", LoadMode=LoadSceneMode(1))
    ScreenFader(startColour=(1, 1, 1, 0), endColour=(1, 1, 1, 1), duration=2)
    Wait(time=4, finishEvent=(none), realTime=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    BeginSceneTransition(sceneName="Coral_Tower_01", entryGateName="door_wakeOnGround", entryDelay=0, visualization=Default, preventCameraFadeOut=true)
  }

  state Throne Rumble {
    AnimatorPlay(gameObject=var "coral_king_final_throne", stateName="Coral_King_Boss_Throne_Intro", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=4, finishEvent=(none), realTime=false)
    SendMessage(gameObject=var "coral_king_final_throne", delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClip=CK_roar_short_02 (sfxstatic_assets_areacoralareamemory.bundle), pitchMin=1, pitchMax=1, volume=1, delay=1.5, storePlayer=<null>)
    on FINISHED → Start Fight
  }

  state Encountered Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pre Intro Roar
  }

  state Start Refight {
    SendEventByName(eventTarget=GameObject(var "Coral King"), sendEvent="BATTLE START REFIGHT", delay=0, everyFrame=false)
    on END KING SCENE → Exit Memory
  }

  state Pre Intro Roar {
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    AnimatorPlay(gameObject=var "coral_king_final_throne", stateName="Coral_King_Boss_Throne_Pre_Intro_Roar", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=4, finishEvent=(none), realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClip=CK_throne_yell_01 (sfxstatic_assets_areacoralareamemory.bundle), pitchMin=1, pitchMax=1, volume=1, delay=1.25, storePlayer=<null>)
    on FINISHED → Throne Rumble
  }

  state State 1 {
    on FINISHED → Pre Intro Roar
  }

  var CamLock Boss: gameObject = <null>
  var Coral King: gameObject = <null>
  var Entry Spears: gameObject = <null>
  var coral_king_final_throne: gameObject = <null>
  var throne_full: gameObject = <null>
  var Song Region Boss: gameObject = <null>
}
