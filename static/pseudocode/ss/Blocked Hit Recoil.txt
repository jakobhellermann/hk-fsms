  state Idle {
    on BLOCKED HIT L → Recoil Left

    on BLOCKED HIT U → Recoil Up
    on BLOCKED DOWN → Recoil Down

  state Recoil Down {
    CallMethodProper(gameObject=Self, behaviour="Recoil", methodName="RecoilByDirection", parameters=[3, 1], storeResult=(unset var), EveryFrame=false)
