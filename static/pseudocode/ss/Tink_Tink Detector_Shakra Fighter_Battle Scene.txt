#   Hang_04_boss / Battle Scene/Shakra Fighter/Tink Detector

fsm Tink {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"TINK", storeCollider=var "")
    on TINK → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="TINK", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Parent: gameObject = Battle Scene/Shakra Fighter
}
