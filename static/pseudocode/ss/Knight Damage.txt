# Knight Damage
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/Effects/Damage Effect

fsm Knight Damage {
  start Init

  state Gen {
    PlayerDataBoolTest(boolName="damagedPurple", isTrue=→"PURPLE", isFalse=(none))
    PlayerDataBoolTest(boolName="damagedBlue", isTrue=→"BLUE", isFalse=(none))
    SpawnObjectFromGlobalPool(gameObject=standard_health_hit (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0.15, 0), rotation=(unset), storeObject=var "Hit Crack")  // disabled
    BoolTest(boolVariable=var "Hero Leak", isTrue=→"LEAK", isFalse=(none), everyFrame=false)
    on LEAK → Check Focus Prompt
    on BLUE → Blue Effects
    on PURPLE → Purple Effects
    on FINISHED → Check Focus Prompt
  }

  state Leak {
    BoolTest(boolVariable=var "Hazard", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    StopParticleEmitter(gameObject=var "Leak")  // disabled
    PlayParticleEmitter(gameObject=var "Leak", emit=0, resetIfPlaying=false)  // disabled
    on FINISHED → Idle
  }

  state Check Focus Prompt {
    GetPlayerDataInt(intName="health", storeValue=var "HP")
    PlayerDataBoolTest(boolName="promptFocus", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(boolName="seenFocusTablet", isTrue=(none), isFalse=→"FINISHED")
    GetPlayerDataInt(intName="maxHealth", storeValue=var "Max HP")
    IntCompare(integer1=var "HP", integer2=var "Max HP", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    GetPlayerDataInt(intName="MPCharge", storeValue=var "MP")
    IntCompare(integer1=var "MP", integer2=33, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetPlayerDataBool(boolName="promptFocus", value=true)
    on FINISHED → Last HP?
  }

  state Blue Effects {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=flashHealBlue())
    SpawnObjectFromGlobalPool(gameObject=Blue_Health_Blast (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Idle
  }

  state Muffle? {
    BoolTest(boolVariable=var "Muffle", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="DAMAGE LOWPASS", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="DAMAGE LOWPASS")
    on FINISHED → Regular Damage
  }

  state Last HP? {
    IntCompare(integer1=var "HP", integer2=1, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    ActivateGameObject(gameObject=var "Low Health Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Leak
  }

  state Idle {
    SetBoolValue(boolVariable=var "Hazard", boolValue=false, everyFrame=false)
    on DAMAGE → Muffle?
    on HAZARD DAMAGE → Set Hazard
    on FROST DAMAGE → Frost Damage
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindGameObject(objectName="", withTag="CameraParent", store=var "Camera")
    ActivateGameObject(gameObject=var "Frost Damage Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Set Hazard {
    SetBoolValue(boolVariable=var "Hazard", boolValue=true, everyFrame=false)
    on FINISHED → Regular Damage
  }

  state Regular Damage {
    SendEventByName(eventTarget=GameObject(var "Vignette"), sendEvent="DAMAGE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HeroLight"), sendEvent="DAMAGE", delay=0, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Hero Damage Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Gen
  }

  state Frost Damage {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Frost Damage Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Frost Damage Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Gen
  }

  state Purple Effects {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=flashHealPoison())
    SpawnObjectFromGlobalPool(gameObject=Blue_Health_Blast Poison Variant (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Idle
  }

  var HeroLight: gameObject = Hero_Hornet/HeroLight
  var Leak: gameObject = Hero_Hornet/Effects/Damage Effect/Leak
  var Vignette: gameObject = Hero_Hornet/Vignette

  var HP: int = 0
  var MP: int = 0
  var Max HP: int = 0
  var Hazard: bool = false
  var Hero Leak: bool = true
  var Muffle: bool = true
  var Camera: gameObject = <null>
  var Effect: gameObject = <null>
  var Hit Crack: gameObject = <null>
  var Low Health Effect: gameObject = Hero_Hornet/Effects/Damage Effect/low health hit effect
  var Self: gameObject = <null>
  var Frost Damage Effect: gameObject = Hero_Hornet/Effects/Damage Effect/Frost Dmg Pt
}
