  state Init {

  }

  }

  state Bump {

    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Floor Bump", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    on FINISHED → Idle
  }

  state Open {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Floor Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    on FINISHED → Idle
  }

  var Boss Scene: gameObject = <null>
  var Trobbio: gameObject = <null>

}
