# Control

fsm Control {
  start Init
  on COMPLETED → Completed  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Silkfly Cloud")
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"CUT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Idle", resetTimeIfPlaying=false, stopOnStateExit=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=var "Needolin Glow", ToAlpha=0, FadeTime=0.75)
    on CUT → Cut
    on NEEDOLIN → Needolin Glow
  }

  state Cut {
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect Small (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "active", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Cut", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Silkfly Cloud"), sendEvent="START", delay=0, everyFrame=false)
    PlayAudioEventRandom(audioClips=[grass_cut_1 (sfxstatic_assets_grass.bundle), grass_cut_2 (sfxstatic_assets_grass.bundle), grass_cut_3 (sfxstatic_assets_grass.bundle), grass_cut_4 (sfxstatic_assets_grass.bundle), grass_cut_5 (sfxstatic_assets_grass.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on RESET → Reset
  }

  state Reset {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    PlayAudioEvent(audioClip=memory_orb_main_appear (sfxstatic_assets_areamemory.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[grass_cut_1 (sfxstatic_assets_grass.bundle), grass_cut_2 (sfxstatic_assets_grass.bundle), grass_cut_3 (sfxstatic_assets_grass.bundle), grass_cut_4 (sfxstatic_assets_grass.bundle), grass_cut_5 (sfxstatic_assets_grass.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "active", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Reset Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Reset Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Break", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Cut", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Needolin Glow {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.5, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=var "Needolin Glow", ToAlpha=1, FadeTime=0.5)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"CUT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on END → Idle
    on CUT → Cut
  }

  state Completed {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var active: gameObject = Memory Orb Group New/White Cloverbush Orb/active
  var Pt Break: gameObject = Memory Orb Group New/White Cloverbush Orb/Pt Break
  var Pt Cut: gameObject = Memory Orb Group New/White Cloverbush Orb/Pt Cut
  var Self: gameObject = Memory Orb Group New/White Cloverbush Orb
  var Reset Effects: gameObject = Memory Orb Group New/White Cloverbush Orb/Reset Effects
  var Silkfly Cloud: gameObject = <null>
  var Parent: gameObject = <null>
  var Pt Idle: gameObject = Memory Orb Group New/White Cloverbush Orb/Pt Idle
  var Needolin Glow: gameObject = Memory Orb Group New/White Cloverbush Orb/Needolin Glow
}
