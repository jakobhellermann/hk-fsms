fsm Eva Heal {
  start Inert

  state Inert {
    SendEventByName(eventTarget=GameObject(var "Spawned Effect"), sendEvent="STOP", delay=0, everyFrame=false)
    on AT THIS BENCH INIT → Wake Wait
    on AT THIS BENCH → Start
  }

  state At Bench {
    GetFsmBool(gameObject=Self, fsmName="Bench Control", variableName="At This Bench", storeValue=var "At This Bench", everyFrame=true)
    BoolTest(boolVariable=var "At This Bench", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    PlayerDataBoolTest(boolName="atBench", isTrue=(none), isFalse=→"CANCEL")
    GetSilk(StoreAmount=var "Current Silk")
    GetPlayerDataVariable(VariableName="CurrentSilkMax", StoreValue=var "Max Silk")
    IntCompare(integer1=var "Current Silk", integer2=var "Max Silk", equal=→"MAX", lessThan=(none), greaterThan=→"MAX", everyFrame=false)
    Wait(time=0.26, finishEvent=→"FINISHED", realTime=false)
    on CANCEL → Inert
    on FINISHED → Give Silk
    on OFF THIS BENCH → Inert
    on MAX → Silk Maxed
  }

  state Give Silk {
    AddSilk(target=var "Hero", amount=1, playHeroEffect=false)
    PlayAudioEvent(audioClip=focus_ready (herosfxstatic_assets_all.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → At Bench
    on OFF THIS BENCH → Inert
  }

  sequence state Start {
    PlayerDataVariableTest(VariableName="HasBoundCrestUpgrader", ExpectedValue=true, IsExpectedEvent=(none), IsNotExpectedEvent=→"CANCEL")
    GetSilk(StoreAmount=var "Current Silk")
    GetPlayerDataVariable(VariableName="CurrentSilkMax", StoreValue=var "Max Silk")
    IntCompare(integer1=var "Current Silk", integer2=var "Max Silk", equal=→"CANCEL", lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    GetFsmBool(gameObject=Self, fsmName="Bench Control", variableName="At This Bench", storeValue=var "At This Bench", everyFrame=false)
    BoolTest(boolVariable=var "At This Bench", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    PlayerDataBoolTest(boolName="atBench", isTrue=(none), isFalse=→"CANCEL")
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    GetFsmVector3(gameObject=Self, fsmName="Bench Control", variableName="Adjust Vector", storeValue=var "Adjust Vector", everyFrame=false)  // disabled
    SpawnObjectFromGlobalPool(gameObject=Silk_Recharge_glow (localpoolprefabs_assets_bench.bundle), spawnPoint=var "Self", position=var "Adjust Vector", rotation=(unset), storeObject=var "Spawned Effect")  // disabled
    SpawnObjectFromGlobalPool(gameObject=Silk_Recharge_glow (localpoolprefabs_assets_bench.bundle), spawnPoint=var "Hero", position=(0, -0.3676853, 0), rotation=(unset), storeObject=var "Spawned Effect")
    Wait(time=0.16666667, finishEvent=(none), realTime=false)
    on CANCEL → Inert
    on FINISHED → At Bench
  }

  state Wake Wait {
    on WAKE END → Wait
  }

  state Wait {
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Start
  }

  state Silk Maxed {
    SendEventByName(eventTarget=GameObject(var "Spawned Effect"), sendEvent="MAX", delay=0, everyFrame=false)
    on FINISHED → Inert
  }

  var Current Silk: int = 0
  var Max Silk: int = 0
  var At This Bench: bool = false
  var Adjust Vector: vector3 = (0, 0, 0)
  var Hero Position: vector3 = (0, 0, 0)
  var Self: gameObject = <null>
  var Spawned Effect: gameObject = <null>
}
