# Control
# 1 location(s):
#   Slab_16 / Event Control/Battle Cloakless Start Scene/Dancer Fly Death

fsm Control {
  start Init

  state Init {
    on FINISHED → Fling
  }

  state Fling {
    SetVelocityAsAngle(gameObject=Self, angle=70, speed=22, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Air
  }

  state Air {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Land {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fly Death Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=big_enemy_corpse_ground_land (sfxstatic_assets_enemyheavy.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fade
  }

  state Bounce Away {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly Death Air")
    SetVelocity2d(gameObject=Self, vector=(5, 16), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Fade {
    Tk2dSpriteTweenColor(gameObject=Self, color=(0.7490196, 0.7490196, 0.7490196, 1), duration=0.25)
  }

  var Pokes: int = 0
  var Going R: bool = false
}
