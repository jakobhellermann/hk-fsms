# Control
# 2 location(s):
#   Peak_01 / One Way Wall Crystal
#   Peak_04c / One Way Wall Crystal

fsm Control {
  start Pause

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=Self, storeResult=var "Masks")
    FindNamedChild(gameObject=Self, storeResult=var "Active After Break")
    ActivateGameObject(gameObject=var "Active After Break", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Broken
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"DAMAGE", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on DAMAGE → Check
  }

  state Check {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Can Break Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    PlayAudioEventRandom(audioClips=[crystal_break_1 (sfxstatic_assets_areapeak.bundle), crystal_break_2 (sfxstatic_assets_areapeak.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    on CANCEL → Idle
    on FINISHED → Hit
    on BREAK → Set PD Bool
  }

  state Break {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    FindNamedChild(gameObject=var "Active", storeResult=var "CamLock")
    SetCollider(gameObject=var "CamLock", active=false, resetOnExit=false)
    SetParent(gameObject=var "CamLock", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Active", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="START SNOWFLAKE", delay=0, everyFrame=false)
    PlayAudioEventRandom(audioClips=[breakable_wall_death (sfxstatic_assets_shared.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[breakable_wall_hit_1 (sfxstatic_assets_shared.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Chunks", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Break Wall Effect (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Break Wall Effect Point", position=(unset), rotation=(unset), storeObject=var "")
    ActivateGameObject(gameObject=var "Active After Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Active After Break", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendMessage(gameObject=var "Masks", delivery=BroadcastMessage, options=1, functionCall=Uncover())
  }

  state Hit {
    SendMessage(gameObject=var "Break Chunks", delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    SendMessage(gameObject=var "Active", delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    PlayAudioEventRandom(audioClips=[breakable_wall_hit_1 (sfxstatic_assets_shared.bundle), breakable_wall_hit_2 (sfxstatic_assets_shared.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetRandomChild(gameObject=var "Break Chunks", storeResult=var "Chunk")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    FindChild(gameObject=var "Chunk", childName="Break Effects", storeResult=var "Chunk Effects")
    SetParent(gameObject=var "Chunk", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Chunk Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetSpriteRenderer(gameObject=var "Chunk", active=false)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Chunk", position=(unset), rotation=(unset), storeObject=var "")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Broken {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="START SNOWFLAKE", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Active After Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Active After Break", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set PD Bool {
    StringCompare(stringVariable=var "Set PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetPlayerDataBool(boolName=var "Set PD Bool", value=true)
    on FINISHED → Break
  }

  var Hits: int = 3
  var Set PD Bool: string = ""

  var Activated: bool = false
  var Active: gameObject = One Way Wall Crystal/Active
  var Break Chunks: gameObject = One Way Wall Crystal/Break Chunks
  var Break Effects: gameObject = One Way Wall Crystal/Break Effects
  var Break Wall Effect Point: gameObject = One Way Wall Crystal/Break Wall Effect Point
  var Chunk: gameObject = <null>
  var Chunk Effects: gameObject = <null>
  var Masks: gameObject = <null>
  var Parent: gameObject = <null>
  var CamLock: gameObject = <null>
  var Active After Break: gameObject = <null>
}
