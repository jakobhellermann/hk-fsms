# Control
# 1 location(s):

  on TORNADO → Break  // from any state

  state Init {

    ActivateGameObject(gameObject=var "Spark", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Tendril Target", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFloatValue(floatVariable=var "Fire Timer", floatValue=1, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    on FINISHED → Poison?
  }

  state Recycle {
    RecycleSelf()
  }

  state Ground {

  }

    ActivateGameObject(gameObject=var "Throw Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Ball Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetParent(gameObject=var "Surface Water", storeResult=var "Surface Water")
    GetPosition2D(gameObject=Self, vector=var "Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    CallMethodProper(gameObject=var "Surface Water", behaviour="MaggotRegion", methodName="ReportExplosion", parameters=[var "Pos"], storeResult=(unset var), EveryFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Water Zap L", y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Water Zap R", y=(unset), z=(unset), space=0, everyFrame=false)
    RayCast2dV2(fromGameObject=Self, fromPosition=(0, 1), direction=(0, -1), space=1, distance=4, minDepth=(unset), maxDepth=(unset), hitEvent=→"WATER", noHitEvent=→"END", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Hit Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Soft Terrain(25)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Recycle
  }

  state Set Zap Start {
    GetVector2XY(vector2Variable=var "Hit Point", storeX=(unset), storeY=var "Water Zap Y", everyFrame=false)
    SetBoolValue(boolVariable=var "Zapping L", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Zapping R", boolValue=true, everyFrame=false)
    SetIntValue(intVariable=var "Zaps Limit", intValue=8, everyFrame=false)
    on FINISHED → Zap R
  }

  state Zap R {
    BoolTest(boolVariable=var "Zapping R", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    FloatAdd(floatVariable=var "Water Zap R", add=var "Water Zap Distance", everyFrame=false, perSecond=false)
    SetVector2XY(vector2Variable=var "Zap Origin", vector2Value=(unset), x=var "Water Zap R", y=var "Water Zap Y", everyFrame=false)
    Vector2AddXY(vector2Variable=var "Zap Origin", addX=0, addY=1, everyFrame=false, perSecond=false)
    RayCast2dV2(fromGameObject=<null>, fromPosition=var "Zap Origin", direction=(0, -1), space=1, distance=4, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"END", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Hit Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Soft Terrain(25)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)

    on FINISHED → Zap L
  }

  state Zap R End {
    SetBoolValue(boolVariable=var "Zapping R", boolValue=false, everyFrame=false)
    on FINISHED → Zap L
  }

  state Zap L {
    BoolTest(boolVariable=var "Zapping L", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    FloatSubtract(floatVariable=var "Water Zap L", subtract=var "Water Zap Distance", everyFrame=false, perSecond=false)
    SetVector2XY(vector2Variable=var "Zap Origin", vector2Value=(unset), x=var "Water Zap L", y=var "Water Zap Y", everyFrame=false)

    on FINISHED → Zap Pause
  }

  state Zap L End {
    SetBoolValue(boolVariable=var "Zapping L", boolValue=false, everyFrame=false)
    on FINISHED → Zap Pause
  }

  state Zap Pause {
    IntAdd(intVariable=var "Zaps Limit", add=-1, everyFrame=false)
    IntCompare(integer1=var "Zaps Limit", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    BoolNoneTrue(boolVariables=[var "Zapping L", var "Zapping R"], sendEvent=→"END", storeResult=(unset), everyFrame=false)
    Wait(time=0.025, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Zap R
    on END → Recycle
  }

  state Air {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer=(unset), sendEvent=→"WATER", storeCollider=var "Surface Water")

    ActivateGameObjectDelay(gameObject=var "Throw Damager", activate=true, resetOnExit=false, delay=0.05)
    on WALL L → Wall L
    on WALL R → Wall R
    on GROUND → Ground
    on DAMAGED ENEMY → Damaged Enemy
    on WATER → Water Explode
    on BREAK → Break
  }

  state Break {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Tendril Target", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    ActivateGameObject(gameObject=var "Spark", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)

    ActivateGameObject(gameObject=var "Ball Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

  }
