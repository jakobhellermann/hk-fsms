# Multihitter
# 3 location(s):
#   Bone_East_12 / Boss Scene/Lace Boss1/Cross Slash/hero damager
#   Song_Tower_01 / Boss Scene/Lace Boss2 New/Cross Slash/hero damager
#   Song_Tower_01 / Boss Scene/Lace Boss2 New/Cross Slash Repeat - NOT USED/hero damager

// uses template: lace_cross_slash_damager
fsm Multihitter {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Lace")
    on FINISHED → Idle
  }

  state Idle {
    on TRIGGER STAY 2D → Start Hit?
  }

  state Hit {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="TakeQuickDamage", parameters=[1, true], storeResult=(unset var), EveryFrame=false)
    SendEventToRegister(eventName="HERO DAMAGED")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="WOUND START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lace"), sendEvent="MULTIHIT", delay=0, everyFrame=false)
    on MULTIHIT END → Idle
  }

  state Start Hit? {
    GetHeroCState(VariableName="parrying", StoreValue=var "Parrying", EveryFrame=false)
    BoolTest(boolVariable=var "Parrying", isTrue=→"CROSS STITCH", isFalse=(none), everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="WillDoBellBindHit", parameters=[], storeResult=var "Bind Bell Hit", EveryFrame=false)
    BoolTest(boolVariable=var "Bind Bell Hit", isTrue=→"BIND BELL", isFalse=(none), everyFrame=false)
    CanHeroTakeDamage(eventTarget=Self, canTakeDmgEvent=→"HIT", cannotTakeDmgEvent=→"CANCEL")
    on CANCEL → 
    on HIT → Hit
    on BIND BELL → Bind Bell Damage
    on CROSS STITCH → Cross Stitch
  }

  state Bind Bell Damage {
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=false)
    on FINISHED → Idle
  }

  state Cross Stitch {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="PARRIED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Bind Bell Hit: bool = false
  var Parrying: bool = false
  var Lace: gameObject = <null>
}
