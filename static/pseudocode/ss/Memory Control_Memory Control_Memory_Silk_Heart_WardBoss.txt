# Memory Control
# 1 location(s):
#   Memory_Silk_Heart_WardBoss / Memory Control

fsm Memory Control {
  start Init

  state Begin {
    RunFSM(fsmTemplateControl=template=2940330214353294867, finishEvent=(none), everyFrame=false)
    on NEXT SCENE → To White
  }

  sequence state End Scene {
    SetPlayerDataVariable(VariableName="disablePause", SetValue=false)
    RunFSM(fsmTemplateControl=template=4498518172459336583 in[Set PD Bool<-var "Set PD Bool", Set Respawn Scene<-var "Set Respawn Scene"], finishEvent=(none), everyFrame=false)
    RunFSM(fsmTemplateControl=template=6518013566008060279 in[To Scene<-var "To Scene", To Scene Door<-var "To Scene Door"], finishEvent=(none), everyFrame=false)
  }

  state Init {
    RunFSM(fsmTemplateControl=template=6942462098519755694, finishEvent=(none), everyFrame=false)
    on DOOR GET UP FINISHED → Begin
  }

  state To White {
    SetPlayerDataVariable(VariableName="disablePause", SetValue=true)
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(1, 1, 1, 1), duration=0.3)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=at None (audioobjects_assets_all.bundle), transitionTime=0.5)
    on FINISHED → Memory Msg
  }

  sequence state Memory Msg {
    AddMemoryMsg(Source=Self, Text={Sheet="Lore", Key="SILK_HEART_MEMORY_WARDBOSS_FINAL"})
    Wait(time=6, finishEvent=(none), realTime=false)
    RemoveMemoryMsg(Source=Self, DisappearDelay=0)
    Wait(time=2, finishEvent=(none), realTime=false)
    on FINISHED → To Black
  }

  state To Black {
    ScreenFader(startColour=(unset), endColour=(0, 0, 0, 1), duration=2)
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Scene
  }

  var UI Msg: gameObject = <null>
}
