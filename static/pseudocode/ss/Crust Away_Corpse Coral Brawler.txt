// uses template: corpse_coral_crust_away
fsm Crust Away {
  start Wait

  state Wait {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Crust Up")
    Wait(time=var "Crust Pause", finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetNoTint(true))
    on FINISHED → Crust Up
  }

  state Crust Up {
    PlayAudioEvent(audioClip=grey_warrior_stone_turn_into (sfxstatic_assets_areacoral.bundle), pitchMin=0.85, pitchMax=1.15, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Crust Up", resetTimeIfPlaying=false, stopOnStateExit=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(0.5294118, 0.5294118, 0.5294118, 0), duration=1)
    ObjectJitterOnRender(Target=Self, X=0.07, Y=0.07, Z=(unset), LimitFps=30)
    Wait(time=7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
  }

  state Deactivate {
    DestroySelf(detachChildren=false)
  }

  var Crust Pause: float = 3.5

  var Pt Crust Up: gameObject = <null>
}
