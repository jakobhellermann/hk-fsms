# Detect Lava Damage
# 1 location(s):
#   Bone_East_12 / Boss Scene/Lace Boss1

fsm Detect Lava Damage {
  start Idle

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=→"LAVA", sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)  // disabled
    CheckYPosition(gameObject=Self, compareTo=var "Lava Pos Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"LAVA", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAVA → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LAVA DAMAGE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Lava Pos Y: float = 1
}
