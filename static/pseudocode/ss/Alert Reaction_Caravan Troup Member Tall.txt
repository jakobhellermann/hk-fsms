# Alert Reaction
# 1 location(s):
#   Room_Caravan_Interior / Caravan Troup Member Tall

fsm Alert Reaction {
  start Pause

  state Idle {
    CheckTrackTriggerCount(target=var "Track Trigger Enter", count=0, test=Equal, everyFrame=true, successEvent=→"ALERT")
    PlayerDataBoolTest(boolName="MetCaravanTroupeLeader", isTrue=(none), isFalse=→"CANCEL")
    on ALERT → Alert Anim
    on CANCEL → Not Met
  }

  state Alert Anim {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="CT Greet", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reactivate
  }

  state Pending Idle {
  }

  state Not Alert {
    CheckTrackTriggerCount(target=var "Track Trigger Exit", count=0, test=CheckTrackTriggerCount+IntTest(3), everyFrame=true, successEvent=→"CANCEL")
    on CANCEL → Pending Idle
  }

  state Reactivate {
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=false, Target=Self)
    on FINISHED → Pending Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Not Met {
  }

  var Track Trigger Enter: gameObject = Caravan Troup Member Tall/Enter Detector
  var Track Trigger Exit: gameObject = Caravan Troup Member Tall/Exit Detector
}
