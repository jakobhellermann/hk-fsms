# Control
# 1 location(s):
#   Memory_Coral_Tower / Battle Scenes/Battle Scene Chamber 3/Wave 11/Coral Brawler/Stomp Spire R

fsm Control {
  start Pause

  state Init {
    FindChild(gameObject=Self, childName="Hit 1", storeResult=var "Hit 1")
    FindChild(gameObject=Self, childName="Hit 2", storeResult=var "Hit 2")
    FindChild(gameObject=Self, childName="Hit 3", storeResult=var "Hit 3")
    FindChild(gameObject=Self, childName="Pt Break", storeResult=var "Pt Break")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Offset", y=var "Y Offset", z=var "Z Offset", space=1, everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Dormant
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Pt AnticDust")
    PlayParticleEmitterInState(gameObject=var "Pt AnticRock")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Parent", xOffset=var "X Offset", yOffset=var "Y Offset", zOffset=var "Z Offset", overrideZ=(unset), everyFrame=false)
    on FINISHED → Up 1
  }

  state Break {
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="TripleSpire Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Hit 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hit 3", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    on SPIKE UP → Antic
  }

  state Up 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="TripleSpire Up", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetMeshRenderer(gameObject=Self, active=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Up 2
  }

  state Up 2 {
    ActivateGameObject(gameObject=var "Hit 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Up 3
  }

  state Up 3 {
    ActivateGameObject(gameObject=var "Hit 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Up 4
  }

  state Up 4 {
    ActivateGameObject(gameObject=var "Hit 3", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"NEXT", realTime=false)
    on NEXT → Break
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  var X Offset: float = 0
  var Y Offset: float = 0
  var Z Offset: float = 0
  var Start Pos: vector3 = (0, 0, 0)
  var Hit 1: gameObject = <null>
  var Hit 2: gameObject = <null>
  var Hit 3: gameObject = <null>
  var Parent: gameObject = <null>
  var Pt Break: gameObject = <null>
  var Pt AnticDust: object = Battle Scenes/Battle Scene Chamber 3/Wave 11/Coral Brawler/Stomp Spire R/Pt AnticDust
  var Pt AnticRock: object = Battle Scenes/Battle Scene Chamber 3/Wave 11/Coral Brawler/Stomp Spire R/Pt AnticRock
}
