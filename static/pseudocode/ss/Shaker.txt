#   Organ_01 / Exhaust Column Timer (1)/Shake Region Up
#   Organ_01 / Exhaust Column Timer (1)/Shake Region Down
#   Shadow_20 / Exhaust Column Timer/Shake Region Down
#   Song_10 / Steam Column Timer/Shake Region v2

fsm Shaker {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Wait
  }

  state Wait {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    Wait(time=var "Interval", finishEvent=→"SHAKE", realTime=false)
    on EXIT → Idle
    on SHAKE → Shake
  }

  state Shake {
    BoolTest(boolVariable=var "Do Shake", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=var "Shake Event", cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Wait
    on EXIT → Idle
  }

  var Interval: float = 0.15
  var Do Shake: bool = false
  var Shake Event: object = Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle)
}
