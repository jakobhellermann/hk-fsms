# Multicircle
# 1 location(s):
#   Song_Tower_01 / Boss Scene/Lace Boss2 New

fsm Multicircle {
  start Init
  on MULTICIRCLE STOP → Idle  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on MULTICIRCLE START → First Slash
  }

  state First Slash {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Next Y Pos", z=(unset), space=0, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=lace_wild_slash_blade (localpoolprefabs_assets_areasong.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Slash")  // disabled
    SetScale(gameObject=var "Slash", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    on FINISHED → Do Slashes
  }

  state Check Dir {
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"R", negativeEvent=→"L", space=0)
    on R → Set R
    on L → Set L
  }

  state Set R {
    SpawnObjectFromGlobalPool(gameObject=lace_wild_slash_blade (localpoolprefabs_assets_areasong.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Slash")
    SetPosition(gameObject=var "Slash", vector=(unset), x=var "Next Y Pos", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Slash", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    FloatAdd(floatVariable=var "Next Y Pos", add=var "Slash Distance", everyFrame=false, perSecond=false)
    CheckXPosition(gameObject=Self, compareTo=var "Next Y Pos", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"FINISHED", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → Set R
  }

  state Set L {
    SpawnObjectFromGlobalPool(gameObject=lace_wild_slash_blade (localpoolprefabs_assets_areasong.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Slash")
    SetPosition(gameObject=var "Slash", vector=(unset), x=var "Next Y Pos", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Slash", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    FloatSubtract(floatVariable=var "Next Y Pos", subtract=var "Slash Distance", everyFrame=false, perSecond=false)
    CheckXPosition(gameObject=Self, compareTo=var "Next Y Pos", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"FINISHED", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → Set L
  }

  state Do Slashes {
    SpawnObjectFromGlobalPool(gameObject=lace_circle_slash (localpoolprefabs_assets_laceboss.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Slash")
    SetPosition(gameObject=var "Slash", vector=(unset), x=(unset), y=var "Next Y Pos", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetScale(gameObject=var "Slash", vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    FloatAdd(floatVariable=var "Next Y Pos", add=var "Slash Distance", everyFrame=false, perSecond=false)
    CheckYPosition(gameObject=Self, compareTo=var "Next Y Pos", compareToOffset=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → Do Slashes
  }

  var X Scale: float = 0
  var Next Y Pos: float = 0
  var Slash Distance: float = 3
  var Self: gameObject = Boss Scene/Lace Boss2 New
  var Slash: gameObject = <null>
}
