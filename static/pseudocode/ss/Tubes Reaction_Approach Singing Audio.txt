# Tubes Reaction
# 1 location(s):
#   Weave_10 / Crest Upgrade Shrine/Approach Singing Audio

fsm Tubes Reaction {
  start Idle

  state Idle {
    AddEventRegister(target=Self, eventName="TUBES CHANGE")
    on TUBES CHANGE → Check
    on RESING → Restart Singing
  }

  state Check {
    GetFsmInt(gameObject=var "Tubes Parent", fsmName="Record Active Tubes", variableName="Tubes Active", storeValue=var "Tubes Active", everyFrame=false)
    IntSwitch(intVariable=var "Tubes Active", compareTo=[0, 1, 2, 3], sendEvent=[→"NONE", →"1", →"2", →"3"], everyFrame=false)
    on NONE → None
    on 1 → 1
    on 2 → 2
    on 3 → 3
  }

  state None {
    SetAudioVolume(gameObject=Self, volume=0.75, everyFrame=false)
    on FINISHED → Idle
  }

  state 1 {
    SetAudioVolume(gameObject=Self, volume=0.65, everyFrame=false)
    on FINISHED → Idle
  }

  state 2 {
    SetAudioVolume(gameObject=Self, volume=0.55, everyFrame=false)
    on FINISHED → Idle
  }

  state 3 {
    SetAudioVolume(gameObject=Self, volume=0.4, everyFrame=false)
    on FINISHED → Idle
  }

  state Restart Singing {
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=0.75, time=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Tubes Parent: gameObject = Crest Upgrade Shrine/crest_shrine_break_tube parent

  var Tubes Active: int = 0
}
