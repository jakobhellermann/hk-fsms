# State Control
# 3 location(s):
#   Bonetown / Black Thread States/Normal World/RestBench Control/bonetown_globe
#   Song_Enclave / Black Thread States/Normal World/Enclave States/States/Level 1/bonetown_globe
#   Song_Enclave / Black Thread States/Black Thread World/Enclave Act 3/Town/bonetown_globe

fsm State Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=Self, storeResult=var "base")
    FindNamedChild(gameObject=var "Parent", storeResult=var "bonetown_globe_repaired")
    FindNamedChild(gameObject=var "Parent", storeResult=var "restbench_debris")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="skullKingBenchMended", isTrue=→"REPAIRED", isFalse=(none))
    PlayerDataBoolTest(boolName="skullKingInvaded", isTrue=→"BROKEN", isFalse=(none))
    PlayerDataBoolTest(boolName="skullKingKilled", isTrue=→"BROKEN", isFalse=(none))
    SendEvent(eventTarget=Self, sendEvent=→"UNBROKEN", delay=0, everyFrame=false)
    on UNBROKEN → Unbroken
    on BROKEN → Broken
    on REPAIRED → Repaired
  }

  state Unbroken {
  }

  state Repaired {
    ActivateGameObject(gameObject=var "bonetown_globe_repaired", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Broken {
    ActivateGameObject(gameObject=var "base", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "restbench_debris", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetParent(gameObject=var "base", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    PlayerDataBoolTest(boolName="skullKingKilled", isTrue=(none), isFalse=→"BENCH DEBRIS")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BENCH DEBRIS → Bench Debris
  }

  state Bench Debris {
    ActivateGameObject(gameObject=var "restbench_debris", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Parent: gameObject = <null>
  var base: gameObject = <null>
  var bonetown_globe_repaired: gameObject = <null>
  var restbench_debris: gameObject = <null>
}
