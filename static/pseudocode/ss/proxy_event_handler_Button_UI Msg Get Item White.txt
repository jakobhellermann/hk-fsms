# proxy_event_handler
# 1 location(s):
#   prompts_assets_all.bundle / UI Msg Get Item White/Button

fsm proxy_event_handler {
  start Init

  state Init {
    on DOWN → Send Down
    on UP → Send Up
    on DOWN INSTANT → 
    on UP INSTANT → 
    on STOP → 
    on PULSE DOWN → 
  }

  state Send Down {
    SendEventByName(eventTarget=GameObject(UI Msg Get Item White/Button/Label), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Init
  }

  state Send Up {
    SendEventByName(eventTarget=GameObject(UI Msg Get Item White/Button/Label), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Init
  }
}
