# Play Control
# 1 location(s):
#   Mosstown_01 / Bone Trapdoor Slide/Rumble Particles/Rumble Dust

fsm Play Control {
  start Idle

  state Idle {
    on MANTIS FLOOR RUMBLE → Start
  }

  state Start {
    PlayParticleEmitter(gameObject=Self, emit=0, resetIfPlaying=false)
    on MANTIS FLOOR FINISH → End
  }

  state End {
    StopParticleEmitter(gameObject=Self)
  }
}
