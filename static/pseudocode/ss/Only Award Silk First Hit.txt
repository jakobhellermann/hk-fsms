# Only Award Silk First Hit
# 2 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/Attacks/Charge Slash Witch/damager 01
#   heroloading_assets_all.bundle / Hero_Hornet/Attacks/Charge Slash Witch/damager 02

// uses template: witch_charge_slash
fsm Only Award Silk First Hit {
  start Idle

  state Idle {
    on DEALT ACTUAL DAMAGE → Dealt Damage
  }

  state Dealt Damage {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "damager 01")
    FindNamedChild(gameObject=var "Parent", storeResult=var "damager 02")
    CallMethodProper(gameObject=var "damager 01", behaviour="DamageEnemies", methodName="SetSilkGenerationNone", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "damager 02", behaviour="DamageEnemies", methodName="SetSilkGenerationNone", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=var "damager 01", delivery=SendMessage, options=0, functionCall=SetSilkGenerationNone())  // disabled
    SendMessage(gameObject=var "damager 02", delivery=SendMessage, options=0, functionCall=SetSilkGenerationNone())  // disabled
  }

  var Parent: gameObject = <null>
  var damager 01: gameObject = <null>
  var damager 02: gameObject = <null>
}
