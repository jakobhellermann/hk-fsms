# Control
# 1 location(s):
#   Library_01 / Acolyte Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "First Cluster")
    FindNamedChild(gameObject=Self, storeResult=var "Return")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="scholarAcolytesReleased", isTrue=(none), isFalse=→"END")
    PlayerDataBoolTest(boolName="seenScholarAcolytes", isTrue=→"SEEN", isFalse=→"FIRST")
    on SEEN → Seen
    on FIRST → First
    on END → Unreleased
  }

  state First {
    SetPlayerDataBool(boolName="seenScholarAcolytes", value=true)
    ActivateAllChildren(gameObject=var "First Cluster", activate=true)
  }

  state Seen {
    RandomInt(min=4, max=5, storeResult=var "Spawns", inclusiveMax=true, noRepeat=false)
    on FINISHED → Activate
  }

  state Activate {
    GetRandomChild(gameObject=var "Return", storeResult=var "Acolyte")
    ActivateGameObject(gameObject=var "Acolyte", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Acolyte", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    IntAdd(intVariable=var "Spawns", add=-1, everyFrame=false)
    IntCompare(integer1=var "Spawns", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"LOOP", everyFrame=false)
    on END → End
    on LOOP → Activate
  }

  state End {
  }

  state Unreleased {
  }

  var Spawns: int = 0
  var First Cluster: gameObject = <null>
  var Return: gameObject = <null>
  var Acolyte: gameObject = <null>
}
