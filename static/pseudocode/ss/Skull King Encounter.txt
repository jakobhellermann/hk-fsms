# Skull King Encounter
# 1 location(s):
#   Bonetown_boss / Skull King Encounter - deprecated?

fsm Skull King Encounter {
  start Init

  state Init {
    FindGameObject(objectName="", withTag="Boss", store=var "Skull King")
    GetHero(storeResult=var "Hero")
    on FINISHED → State?
  }

  state State? {
    SendEventToRegister(eventName="BOSS ENCOUNTER SETUP")
    PlayerDataBoolTest(boolName="EncounteredBonetownBoss", isTrue=→"RESUME ENCOUNTER", isFalse=→"START ENCOUNTER")
    on START ENCOUNTER → Wait For Encounter
    on RESUME ENCOUNTER → Get Closest Pos
  }

  state Wait For Encounter {
    ActivateGameObject(gameObject=var "Fake Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Skull King", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Broken Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BENCH SAT → Kill Pilby?
  }

  state Start Encounter {
    ActivateGameObject(gameObject=var "Skull King", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByNameV2(eventTarget=GameObject(var "Skull King"), sendEvent="START", delay=0, everyFrame=false)
    SendEventToRegister(eventName="NPC COWER START")
    on BOSS DEAD → End Encounter
  }

  state End Encounter {
    SendEventToRegister(eventName="NPC COWER END")
  }

  state Wait {
    SetPlayerDataBool(boolName="EncounteredBonetownBoss", value=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Encounter
  }

  state Resume Encounter {
    ActivateGameObject(gameObject=var "Fake Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Broken Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Skull King", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByNameV2(eventTarget=GameObject(var "Skull King"), sendEvent="RESUME", delay=0, everyFrame=false)
    SendEventToRegister(eventName="NPC COWER START")
    on BOSS DEAD → End Encounter
  }

  state Get Closest Pos {
    GetClosestGameObject(Target=var "Hero", GameObjects=[var "Right Boss Start", var "Left Boss Start"], MatchEvents=[→"RIGHT", →"LEFT"], StoreGameObject=var "", EveryFrame=false)
    on RIGHT → Spawn Left
    on LEFT → Spawn Right
  }

  state Spawn Right {
    GetPosition2D(gameObject=var "Right Boss Start", vector=var "Start Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Skull King", Vector=var "Start Pos", X=0, Y=0, Space=0, EveryFrame=false)
    on FINISHED → Resume Encounter
  }

  state Spawn Left {
    GetPosition2D(gameObject=var "Left Boss Start", vector=var "Start Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Skull King", Vector=var "Start Pos", X=0, Y=0, Space=0, EveryFrame=false)
    FlipScale(gameObject=var "Skull King", flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Resume Encounter
  }

  state Kill Pilby? {
    PlayerDataBoolTest(boolName="visitedCitadel", isTrue=→"FINISHED", isFalse=(none))
    SetPlayerDataBool(boolName="pilbyKilled", value=true)
    on FINISHED → Wait
  }

  var Skull King: gameObject = Boss Scene/Skull King

  var Start Pos: vector2 = (0, 0)
  var Broken Bench: gameObject = Skull King Encounter - deprecated?/Broken Bench
  var Fake Bench: gameObject = Skull King Encounter - deprecated?/Fake Bench
  var Hero: gameObject = <null>
  var Left Boss Start: gameObject = Skull King Encounter - deprecated?/Left Boss Start
  var Right Boss Start: gameObject = Skull King Encounter - deprecated?/Right Boss Start
}
