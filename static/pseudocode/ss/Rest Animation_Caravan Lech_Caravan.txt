# Rest Animation
# 1 location(s):
#   Greymoor_08_caravan / Caravan/Caravan Lech

fsm Rest Animation {
  start Wait Hero In Position

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Caravan/Caravan Lech@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Antic 2
    on RIGHT → Antic 2
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="ShouldFaceLeft", parameters=[], storeResult=var "Should Face Left", EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "Should Face Left"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest Idle {
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest Left", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on CANCEL → Greet?
  }

  state Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Antic 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Wait Hero In Position {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Reset
  }

  state Greet? {
    BoolTest(boolVariable=var "Did Greet", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Did Greet", boolValue=true, everyFrame=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=kratt_notice (sfxdynamic_assets_fleacaravan.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Antic
  }

  var Max Turn Delay: float = 5
  var Min Turn Delay: float = 3

  var Resting: bool = false
  var Should Face Left: bool = false
  var Did Greet: bool = false
  var Audio Loop Voice: gameObject = Caravan/Caravan Lech/Audio Loop Voice
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
