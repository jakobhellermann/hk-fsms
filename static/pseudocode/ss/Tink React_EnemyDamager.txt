# Tink React
# 1 location(s):
#   herodynamic_assets_all.bundle / Tool Wheel/EnemyDamager

fsm Tink React {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on TINK DOWN → Tink D
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink U
  }

  state Move Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Tink D {
    AddForce2d(gameObject=var "Parent", forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=14, vector3=(unset), everyFrame=false)
    on FINISHED → Move Pause
  }

  state Tink L {
    AddForce2d(gameObject=var "Parent", forceMode=1, atPosition=(unset), vector=(unset), x=-10, y=20, vector3=(unset), everyFrame=false)
    on FINISHED → Move Pause
  }

  state Tink R {
    AddForce2d(gameObject=var "Parent", forceMode=1, atPosition=(unset), vector=(unset), x=10, y=20, vector3=(unset), everyFrame=false)
    on FINISHED → Move Pause
  }

  state Tink U {
    AddForce2d(gameObject=var "Parent", forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=28, vector3=(unset), everyFrame=false)
    on FINISHED → Move Pause
  }

  var Parent: gameObject = Tool Wheel
}
