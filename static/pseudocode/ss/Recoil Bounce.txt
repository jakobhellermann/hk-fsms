fsm Recoil Bounce {
  start Idle

  state Idle {
    on HIT RIGHT → Hit Right
    on HIT LEFT → Hit Left
    on HIT UP → Hit Up
    on HIT DOWN → Hit Down
  }

  state Hit Right {
    ClampVelocity2D(gameObject=Self, xMin=5, xMax=(unset), yMin=(unset), yMax=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Hit Left {
