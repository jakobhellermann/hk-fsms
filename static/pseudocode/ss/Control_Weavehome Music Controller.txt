# Control
# 3 location(s):
#   Bone_East_Weavehome / Weavehome Music Controller
#   Shadow_Weavehome / Weavehome Music Controller
#   Weave_04 / Weavehome Music Controller

fsm Control {
  start Wait

  state Idle {
    on WEAVER GATE CLOSED → Play Weaver Music
  }

  sequence state Play Weaver Music {
    Wait(time=1.5, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=Weaverlands (audiocuesdynamic_assets_areaswampareaweaverareawilds.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
    on WEAVER GATE OPEN → Fade To Silent
  }

  state Init {
    on FINISHED → Idle
  }

  state Wait {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check Gate
  }

  state Check Gate {
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo=var "Exit Gate", equalEvent=→"FADE MUSIC", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FADE MUSIC → Fade To Silent
    on FINISHED → Play Weaver Music
  }

  state Fade To Silent {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    on WEAVER GATE CLOSED → Play Weaver Music
  }

  var Exit Gate: string = "left1"

  var Entry Gate: string = ""
}
