# Control
# 1 location(s):
#   Bone_East_18 / Trap Cage/Hunter Trap Cage

fsm Control {
  start Pause

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Start Sprung", isTrue=→"SPRUNG", isFalse=(none), everyFrame=false)
    SetMeshRenderer(gameObject=var "Cage FG", active=false)
    SetMeshRenderer(gameObject=var "Cage BG", active=false)
    on ACTIVATED → Activated
    on ACTIVATE → Range Check
    on SPRUNG → Miss
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=→"BREAK", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=true, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT → Hit
    on BREAK → Heavy Break
  }

  state Hit {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=15, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Slash Impact R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    PlayAudioEventRandom(audioClips=[hunter_cage_hit_1 (sfxstatic_assets_areaant.bundle), hunter_cage_hit_2 (sfxstatic_assets_areaant.bundle)], pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=var "Cage FG", animLibName="", clipName="Cage Hit")
    Tk2dPlayFrame(gameObject=var "Cage FG", frame=0)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetParticleEmissionChildren(Target=var "Pt Hit", SetEmission=true, EveryFrame=false, stopOnExit=true)
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=30)
    on FINISHED → Idle
    on BREAK → Break
  }

  state Break {
    ActivateGameObject(gameObject=var "Chunks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=hunter_cage_break (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventByName(eventTarget=GameObject(var ""), sendEvent="RESCUE", delay=0.25, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Trap Terrain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Cage BG", animLibName="", clipName="Cage BG Broken")
    Tk2dPlayAnimation(gameObject=var "Cage FG", animLibName="", clipName="Cage Broken")
    PlayParticleEmitterChildren(gameObject=var "Pt Break", resetTimeIfPlaying=false, stopOnStateExit=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activated {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Trap Terrain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Cage BG", animLibName="", clipName="Cage BG Broken")
    Tk2dPlayAnimation(gameObject=var "Cage FG", animLibName="", clipName="Cage Broken")
  }

  state Range Check {
    PlayAudioEvent(audioClip=hunter_trap_cage_come_up (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Capture Range", storeResult=false, sendEvent="IN", outOfRangeEvent="OUT", everyFrame=false)
    on IN → Capture
    on OUT → Miss
  }

  state Capture {
    SendEventToRegister(eventName="HORNET CAGED")
    SendEventToRegister(eventName="CAGE SPRUNG")
    SendEventToRegister(eventName=var "Extra Register Event")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR BURST ENTER", delay=0, everyFrame=false)
    SetPositionToObject(gameObject=var "Hero", targetObject=var "Capture Point", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    SetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=(unset), z=0.006, space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt TrapDust", emit=0, resetIfPlaying=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Trap Terrain", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Cage FG", active=true)
    SetMeshRenderer(gameObject=var "Cage BG", active=true)
    Tk2dPlayAnimation(gameObject=var "Cage BG", animLibName="", clipName="Cage BG Up")
    Tk2dPlayAnimation(gameObject=var "Cage FG", animLibName="", clipName="Cage Up")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=Wound Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Idle
  }

  state Miss {
    SendEventToRegister(eventName="CAGE SPRUNG")
    PlayParticleEmitter(gameObject=var "Pt TrapDust", emit=0, resetIfPlaying=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Terrain", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Cage FG", active=true)
    SetMeshRenderer(gameObject=var "Cage BG", active=true)
    Tk2dPlayAnimation(gameObject=var "Cage BG", animLibName="", clipName="Cage BG Up")
    Tk2dPlayAnimation(gameObject=var "Cage FG", animLibName="", clipName="Cage Up")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Heavy Break {
    SetIntValue(intVariable=var "Hits", intValue=0, everyFrame=false)
    on FINISHED → Hit
  }

  var Start Sprung: bool = false
  var Extra Register Event: string = ""

  var Distance: float = 0
  var Hits: int = 4
  var Activated: bool = false
  var Cage BG: gameObject = Trap Cage/Hunter Trap Cage/Cage BG
  var Cage FG: gameObject = Trap Cage/Hunter Trap Cage/Cage FG
  var Capture Point: gameObject = Trap Cage/Hunter Trap Cage/Capture Point
  var Pt Break: gameObject = Trap Cage/Hunter Trap Cage/Pt Break
  var Pt Hit: gameObject = Trap Cage/Hunter Trap Cage/Pt Hit
  var Pt TrapDust: gameObject = Trap Cage/Hunter Trap Cage/Pt TrapDust
  var Self: gameObject = Trap Cage/Hunter Trap Cage
  var Slam Effect: gameObject = Trap Cage/Hunter Trap Cage/Slam Effect
  var Terrain: gameObject = Trap Cage/Hunter Trap Cage/Terrain
  var Trap Terrain: gameObject = Trap Cage/Hunter Trap Cage/Trap Terrain
  var Chunks: gameObject = Trap Cage/Hunter Trap Cage/Chunks
}
