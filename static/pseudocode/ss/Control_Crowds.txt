# Control
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/Pilgrim Crowds/Crowds

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetOwner(storeGameObject=var "Self")
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataBool(boolName="skullKingWillInvade", storeValue=var "skullKingWillInvade")
    GetPlayerDataBool(boolName="skullKingInvaded", storeValue=var "skullKingInvaded")
    BoolAnyTrue(boolVariables=[var "skullKingWillInvade", var "skullKingInvaded"], sendEvent=(none), storeResult=var "Skull King Present", everyFrame=false)
    GetPlayerDataBool(boolName="skullKingKilled", storeValue=var "skullKingKilled")
    BoolTestMulti(boolVariables=[var "Skull King Present", var "skullKingKilled"], boolStates=[true, false], trueEvent=→"CROWD 1", falseEvent=(none), storeResult=(unset), everyFrame=false)
    GetPlayerDataInt(intName="bonetownCrowd", storeValue=var "Crowd Value")
    IntSwitch(intVariable=var "Crowd Value", compareTo=[1, 2, 3, 4, 5, 6, 7], sendEvent=[→"CROWD 1", →"CROWD 2", →"CROWD 3", →"CROWD 4", →"CROWD 5", →"CROWD 6", →"CROWD 7"], everyFrame=false)
    on CROWD 1 → Crowd 1
    on CROWD 2 → Crowd 2
    on CROWD 3 → Crowd 3
    on CROWD 4 → Crowd 4
    on CROWD 5 → Crowd 5
    on CROWD 6 → Crowd 6
    on CROWD 7 → Crowd 7
  }

  state Crowd 1 {
    FindChild(gameObject=Self, childName="Crowd 1", storeResult=var "Crowd")
    on FINISHED → Activate
  }

  state Activate {
    ActivateGameObject(gameObject=var "Crowd", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Crowd 2 {
    FindChild(gameObject=Self, childName="Crowd 2", storeResult=var "Crowd")
    on FINISHED → Activate
  }

  state Crowd 3 {
    FindChild(gameObject=Self, childName="Crowd 3", storeResult=var "Crowd")
    on FINISHED → Activate
  }

  state Crowd 4 {
    FindChild(gameObject=Self, childName="Crowd 4", storeResult=var "Crowd")
    on FINISHED → Activate
  }

  state Crowd 5 {
    FindChild(gameObject=Self, childName="Crowd 5", storeResult=var "Crowd")
    on FINISHED → Activate
  }

  state Crowd 6 {
    FindChild(gameObject=Self, childName="Crowd 6", storeResult=var "Crowd")
    on FINISHED → Activate
  }

  state Crowd 7 {
    FindChild(gameObject=Self, childName="Crowd 7", storeResult=var "Crowd")
    on FINISHED → Activate
  }

  var Crowd Value: int = 0
  var skullKingWillInvade: bool = false
  var skullKingInvaded: bool = false
  var skullKingKilled: bool = false
  var Skull King Present: bool = false
  var Crowd: gameObject = <null>
  var Self: gameObject = <null>
}
