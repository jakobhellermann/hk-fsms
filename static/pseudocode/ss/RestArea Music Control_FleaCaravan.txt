# RestArea Music Control
# 2 location(s):
#   coremanagers_assets__gamemanager.bundle / _GameManager/AudioManager/Music/FleaCaravan
#   Menu_Title / _GameManager/AudioManager/Music/FleaCaravan

fsm RestArea Music Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    EnableFsmSelf(SetEnabled=false)
    on FLEA MUSIC START → Check Should Play
    on FLEA MUSIC RESET → Reset
  }

  sequence state Play {
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=3)
    EnableFsmSelf(SetEnabled=false)
    on FLEA MUSIC STOP → Stop Wait
    on FLEA MUSIC STOP FAST → Stop Fast
    on BEAST BABY WARP → Stop
  }

  state Stop {
    AudioStopV2(gameObject=Self, fadeTime=7, cancelOnEarlyExit=true)
    Wait(time=7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on FLEA MUSIC START → Check Should Play
    on FLEA MUSIC RESET → Reset
  }

  state Stop Fast {
    AudioStopV2(gameObject=Self, fadeTime=1, cancelOnEarlyExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on FLEA MUSIC START → Check Should Play
    on FLEA MUSIC RESET → Reset
  }

  state Reset {
    AudioStop(gameObject=Self, fadeTime=0)
    on FINISHED → Play
  }

  state Stop Wait {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Stop
    on FLEA MUSIC START → Check Should Play
    on FLEA MUSIC RESET → Reset
  }

  state Check Should Play {
    CaravanLocationSwitch(Locations=[{Location=CaravanTroupeLocations(1), MatchEvent=→"TRUE"}, {Location=CaravanTroupeLocations(2), MatchEvent=→"TRUE"}, {Location=CaravanTroupeLocations(3), MatchEvent=→"FLEATOPIA"}])
    on FLEA MUSIC STOP → Idle
    on FINISHED → Idle
    on TRUE → Play
    on FLEA MUSIC STOP FAST → Idle
    on FLEATOPIA → Set Fleatopia
  }

  state Set Fleatopia {
    PlayerDataBoolTest(boolName="FleaGamesStarted", isTrue=→"FESTIVAL", isFalse=(none))
    SetAudioClip(gameObject=Self, audioClip=Fleatopia Paired Back B (audiocuesstatic_assets_shared.bundle), autoPlay=false, stopOnExit=false)
    on FINISHED → Play
    on FESTIVAL → Set Festival
  }

  state Set Festival {
    SetAudioClip(gameObject=Self, audioClip=Flea Festival (audiocuesstatic_assets_shared.bundle), autoPlay=false, stopOnExit=false)
    on FINISHED → Play
  }
}
