# Land
# 2 location(s):
#   Under_05 / Barrel Plat Lift (2)/Fall_platform_barrel wide/Hero Land
#   Under_27 / Floor Break Set/Fall_platform_barrel/Hero Land

fsm Land {
  start Init

  state Init {
    GetPosition(gameObject=var "Vine", vector=var "Vine Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=var "Plat Sprite", vector=var "Plat Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"LAND", storeCollider=var "")
    SetPosition(gameObject=var "Vine", vector=var "Vine Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Plat Sprite", vector=var "Plat Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Plat Sprite", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    on LAND → Land
  }

  state Land {
    SendEventByName(eventTarget=GameObject(var "Vine"), sendEvent="Bob", delay=0, everyFrame=false)  // disabled
    SetParticleEmissionRate(gameObject=var "Land Grass", emissionRate=35, everyFrame=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    iTweenMoveBy(gameObject=var "Vine", id="", vector=(0, -0.1, 0), time=0.1, delay=0, speed=(unset), easeType=13, loopType=0, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenMoveBy(gameObject=var "Vine", id="", vector=(0, 0.1, 0), time=0.3, delay=0.1, speed=(unset), easeType=14, loopType=0, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenMoveBy(gameObject=var "Plat Sprite", id="", vector=(0, -0.1, 0), time=0.1, delay=0, speed=(unset), easeType=13, loopType=0, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    iTweenMoveBy(gameObject=var "Plat Sprite", id="", vector=(0, 0.1, 0), time=0.3, delay=0.1, speed=(unset), easeType=14, loopType=0, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Land End
  }

  state Land End {
    SetParticleEmissionRate(gameObject=var "Land Grass", emissionRate=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Plat Pos: vector3 = (0, 0, 0)
  var Vine Pos: vector3 = (0, 0, 0)
  var Land Grass: gameObject = <null>
  var Plat Sprite: gameObject = <null>
  var Vine: gameObject = <null>
}
