# Control
# 1 location(s):
#   Mosstown_02 / Black Thread States Thread Only Variant/Normal World/Thick Silk Vines

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on DESTROY → Destroy
    on ACTIVATE → Activate
  }

  state Destroy {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="AverageShake", delay=0, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    ActivateGameObject(gameObject=var "Wall", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Tutorial Msg", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Thick Vines"), sendEvent="DESTROY", delay=0, everyFrame=false)
  }

  state Activate {
    ActivateGameObject(gameObject=var "Tutorial Msg", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Tutorial Msg: gameObject = Black Thread States Thread Only Variant/Normal World/Reminder Silk Skill (1)

  var Activated: bool = false
  var Thick Vines: gameObject = Black Thread States Thread Only Variant/Normal World/Thick Silk Vines/Thick Vines
  var Wall: gameObject = Black Thread States Thread Only Variant/Normal World/Thick Silk Vines/Wall
}
