# Control
# 1 location(s):
#   Ant_02 / Tension Music Control

fsm Control {
  start Init

  state Init {
    on FINISHED → Check
  }

  state Check {
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="left1", equalEvent=(none), notEqualEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName="ant02GuardDefeated", isTrue=→"CANCEL", isFalse=(none))
    PlayerDataBoolTest(boolName="visitedHuntersTrail", isTrue=→"CANCEL", isFalse=(none))
    PlayerDataBoolTest(boolName="visitedGreymoor", isTrue=→"CANCEL", isFalse=(none))
    PlayerDataBoolTest(boolName="visitedShellwood", isTrue=→"CANCEL", isFalse=(none))
    on CANCEL → Cancel
    on FINISHED → Range Check
  }

  state Cancel {
  }

  state Tension {
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=5)
    on LEAVING SCENE → End
    on ANT GUARD WOKE → End
  }

  state End {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.5)
  }

  state Range Check {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Tension
  }

  var Entry Gate: string = ""
}
