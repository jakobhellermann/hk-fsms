// uses template: map_set_room_compass
fsm Set Compass Point {
  start Idle

  state Idle {
    on SET COMPASS POINT → Inert
  }

  state Set {
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", space=0, everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="GetSceneNameString", parameters=[], storeResult=var "Scene Name", EveryFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="GetCurrentMapZone", parameters=[], storeResult=var "Map Zone", EveryFrame=false)
    CallMethodProper(gameObject=var "Game Map", behaviour="GameMap", methodName="SetDoorValues", parameters=[var "Hero X", var "Hero Y", var "Scene Name", var "Map Zone"], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataFloat(gameObject=var "GameManager", floatName="gMap_doorX", value=var "Hero X")
    SetPlayerDataFloat(gameObject=var "GameManager", floatName="gMap_doorY", value=var "Hero Y")
    SetPlayerDataString(gameObject=var "GameManager", stringName="gMap_doorScene", value=var "Scene Name")
    SetPlayerDataString(gameObject=var "GameManager", stringName="gMap_doorMapZone", value=var "Map Zone")
    CallMethodProper(gameObject=var "Game Map", behaviour="GameMap", methodName="SetCompassPoint", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Inert {
  }

  var Hero X: float = 0
  var Hero Y: float = 0
  var Map Zone: string = ""
  var Scene Name: string = ""
}
