# DAMAGE EFFECTS
# 1 location(s):
#   Shellwood_01 / Wasp Scene/Hive Scene/Shellwood Hive/Control

fsm DAMAGE EFFECTS {
  start Init 2

  state Idle {
    on TOOK DAMAGE → Alert BG
  }

  state Damage {
    ObjectJitterLocal(gameObject=var "Art", x=0.1, y=0.05, z=(unset), resetPosOnStateExit=true, limitFps=0)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Hit", emit=0, resetIfPlaying=false)
    SendEventByName(eventTarget=GameObject(var "Wasps"), sendEvent="ALERT", delay=0, everyFrame=false)
    SetSpriteRendererColor(TintColor=(0, 0, 0, 1), Target=var "")
    PlayAudioEvent(audioClip=shellwood_hive_break (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on TOOK DAMAGE → Damage
    on FINISHED → Idle
  }

  state Alert BG {
    BoolTest(boolVariable=var "Alerted", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "BG Wasps"), sendEvent="ANGRY", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Alerted", boolValue=true, everyFrame=false)
    on FINISHED → Damage
  }

  state Init 2 {
    FindChild(gameObject=Self, childName="Pt Hit", storeResult=var "Pt Hit")
    FindChild(gameObject=Self, childName="Wasps", storeResult=var "Wasps")
    FindNamedChild(gameObject=var "Parent", storeResult=var "BG Spawn 1")
    FindNamedChild(gameObject=var "Parent", storeResult=var "BG Spawn 2")
    FindNamedChild(gameObject=Self, storeResult=var "Pt ShootAntic")
    FindNamedChild(gameObject=Self, storeResult=var "Emerge Effect")
    on FINISHED → Idle
  }

  var BG Wasps: gameObject = <null>

  var Spawn Pause: float = 0
  var Sideline Amount: int = 3
  var Spawn HP1: int = 0
  var Spawn HP2: int = 0
  var Spawn HP3: int = 0
  var Alerted: bool = false
  var Spawned 1: bool = false
  var Spawned 2: bool = false
  var Spawned 3: bool = false
  var Art: gameObject = Wasp Scene/Hive Scene/Shellwood Hive/Art
  var Pt Hit: gameObject = <null>
  var Self: gameObject = Wasp Scene/Hive Scene/Shellwood Hive/Control
  var Wasps: gameObject = <null>
  var BG Spawn 1: gameObject = <null>
  var BG Spawn 2: gameObject = <null>
  var Emerge Effect: gameObject = <null>
  var Parent: gameObject = Wasp Scene/Hive Scene/Shellwood Hive
  var Pt ShootAntic: gameObject = <null>
}
