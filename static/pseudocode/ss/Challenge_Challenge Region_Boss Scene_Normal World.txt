# Challenge
# 1 location(s):
#   Belltown_Shrine / Black Thread States Thread Only Variant/Normal World/Boss Scene/Challenge Region

fsm Challenge {
  start Init
  on CHALLENGE DISABLE → Inert  // from any state

  state Init {
    on FINISHED → Idle
  }

  state In Region {
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=true, everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Exit
    on SPECIAL CHALLENGE → Take Control
  }

  state Facing {
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=false, everyFrame=false)
    BoolTest(boolVariable=var "Face Left", isTrue=→"L", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Face Right", isTrue=→"R", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Face Self", isTrue=→"SELF", isFalse=(none), everyFrame=false)
    on L → Face L
    on R → Face R
    on FINISHED → Challenge 1
    on SELF → Face Self
  }

  state Face L {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Challenge 1
  }

  state Challenge 1 {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayRandomAudioClipTable(Table=Taunt Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SendEventByName(eventTarget=GameObject(var "Receiver"), sendEvent="CHALLENGE START", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Taunt Back", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=hornet_taunt (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=hornet_taunt (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Challenge 2
  }

  state Face R {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    on FINISHED → Challenge 1
  }

  state Challenge 2 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=sword_5 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Challenge Complete
  }

  state Challenge Complete {
    SendEventByName(eventTarget=GameObject(var "Receiver"), sendEvent="CHALLENGE", delay=0, everyFrame=false)
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In Region
  }

  state Exit {
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Take Control {
    SetFsmBool(gameObject=var "Hero", fsmName="Silk Specials", variableName="In Challenge Region", setValue=false, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RelinquishControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    on FINISHED → Facing
  }

  state Face Self {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Face L
    on R → Face R
  }

  state Inert {
  }

  var Face Right: bool = false
  var Face Left: bool = false
  var Receiver: gameObject = Black Thread States Thread Only Variant/Normal World/Boss Scene/Speak Region

  var Face Self: bool = false
}
