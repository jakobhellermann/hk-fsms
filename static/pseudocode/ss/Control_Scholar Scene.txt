# Control
# 1 location(s):
#   Library_14 / Scholar Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindNamedChild(gameObject=Self, storeResult=var "Scholar Ambush")
    FindNamedChild(gameObject=Self, storeResult=var "Scholar Revisit")
    PlayerDataBoolTest(boolName="library_14_ambush", isTrue=→"REVISIT", isFalse=→"AMBUSH")
    on REVISIT → Revisit
    on AMBUSH → Ambush
  }

  state Revisit {
    ActivateGameObject(gameObject=var "Scholar Ambush", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Scholar Revisit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Ambush {
    ActivateGameObject(gameObject=var "Scholar Ambush", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Scholar Revisit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Scholar Ambush: gameObject = <null>
  var Scholar Revisit: gameObject = <null>
}
