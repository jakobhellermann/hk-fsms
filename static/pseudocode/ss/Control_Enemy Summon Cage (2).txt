fsm Control {
  start Init

  state Init {
    GetGrandParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Summon Enemies")
    on FINISHED → Dormant
  }

  state Dormant {
    on SUMMON → Pause
  }

  state Fall {
    ActivateGameObject(gameObject=var "Cage", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEventRandom(audioClips=[hunter_cage_hit_1 (sfxstatic_assets_areaant.bundle), hunter_cage_hit_2 (sfxstatic_assets_areaant.bundle)], pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Pos", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Y Pos", float2=var "Land Y", tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-32, everyFrame=false)
    on LAND → Summon Enemy
    on BOSS DEFEATED → Break
  }

  state Break {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Cage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Break", resetTimeIfPlaying=false, stopOnStateExit=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=hunter_cage_break (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Dormant
  }

  state Pause {
    WaitRandom(timeMin=0.25, timeMax=0.4, finishEvent=→"FINISHED", realTime=false)
    RandomFloat(min=var "Min X", max=var "Max X", storeResult=var "X Pos")
    SetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=53.17, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Antic
    on BOSS DEFEATED → Dormant
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fall
    on BOSS DEFEATED → Dormant
  }

  state Summon Enemy {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Land Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetFsmGameObject(gameObject=var "Summon Enemies", fsmName="Control", variableName="Summon Point", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Summon Enemies"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on FINISHED → Break
  }

  var Min X: float = 54.93
  var Max X: float = 62.93

  var X Pos: float = 0
  var Y Pos: float = 0
  var Land Y: float = 36.03
  var Cage: gameObject = Boss Control/Boss Scene/Summon Cages/Enemy Summon Cage (2)/Cage

}
