# FSM
# 1 location(s):
#   Crawl_03 / Area_States/Uninfected/Bone Worm Nests/Bone Worm Nest (12)

// uses template: bone_worm_nest
fsm FSM {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Nest Parent")
    FindNamedChild(gameObject=var "Nest Parent", storeResult=var "Worm Pool")
    FindNamedChild(gameObject=Self, storeResult=var "Spawn Range")
    WaitRandom(timeMin=0, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    AddEventRegister(target=Self, eventName="HORNET DEATH")
    on FINISHED → Dormant
  }

  state Idle {
    Wait(time=var "Spawn Pause", finishEvent=→"SPAWN", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Spawn Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    on NEEDOLIN → Singing
    on SPAWN → Extra Pause
    on EXIT → Exit Pause
    on HORNET DEATH → Keep Spawning
  }

  state Singing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Idle
  }

  state Spawn {
    GetRandomChild(gameObject=var "Worm Pool", storeResult=var "Spawn")
    SetFsmGameObject(gameObject=var "Spawn", fsmName="Control", variableName="Nest", setValue=var "Self", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Spawn"), sendEvent="SPAWN", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Spawn", fsmName="Control", variableName="Play Sounds", setValue=var "z Do Sounds", everyFrame=false)
    on FINISHED → Idle
  }

  state Extra Pause {
    WaitRandom(timeMin=0, timeMax=0.15, finishEvent=(none), realTime=false)
    on FINISHED → Spawn
  }

  state Dormant {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Spawn Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Extra Pause
  }

  state Exit Pause {
    on FINISHED → Dormant
    on HORNET DEATH → Keep Spawning
  }

  state Keep Spawning {
    FindNamedChild(gameObject=Self, storeResult=var "Alert Range")
    ActivateGameObject(gameObject=var "Alert Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Ground Y: float = -6.62
  var Max X: float = 1
  var Min X: float = -2
  var Spawn Pause: float = 0.5
  var Top Y: float = 0
  var z Do Sounds: bool = false

  var Cam X Min: float = 0
  var Cam Y Min: float = 0
  var Cam X Max: float = 0
  var Cam Y Max: float = 0
  var Spawn: gameObject = <null>
  var Nest Parent: gameObject = <null>
  var Worm Pool: gameObject = <null>
  var Spawn Range: gameObject = <null>
  var Self: gameObject = <null>
  var Alert Range: gameObject = <null>
}
