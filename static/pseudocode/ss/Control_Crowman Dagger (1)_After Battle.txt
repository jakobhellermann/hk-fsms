# Control
# 1 location(s):
#   Room_CrowCourt / Battle Scene/After Battle/Crowman Dagger (1)

fsm Control {
  start Init
  on ZERO HP → Death  // from any state
  on GO UP → Flap Up  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Throw Point", storeResult=var "Throw Point")
    FindAlertRange(target=Self, storeResult=var "Throw Range", childName="Throw Range")
    BoolTest(boolVariable=var "Spawner", isTrue=→"SPAWNER", isFalse=(none), everyFrame=false)
    on FINISHED → Random Scale
    on SPAWNER → Spawn Ready
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=5)
    SendEventToRegister(eventName="CROWMAN WAKE")  // disabled
    ActivateGameObject(gameObject=var "Pt Rain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=crawman_med_yelp (sfxdynamic_assets_areagreymoor.bundle), pitchOffset=var "Voice Pitch Offset", stopPreviousSound=true, forcePlay=false)
    on FINISHED → Ground Idle
  }

  state Ground Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    CompareUnalertTime(alertRange=var "Throw Range", compareTo=8, lessThanOrEqualEvent=(none), greatherThanEvent=(none), greatherThanBool=var "Unalert Exceeded", everyFrame=false)
    BoolTestMulti(boolVariables=[var "Battler", var "Unalert Exceeded"], boolStates=[false, true], trueEvent=→"UNALERT", falseEvent=(none), storeResult=(unset), everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Slash Range", storeResult=false, sendEvent="CLOSE", outOfRangeEvent="", everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToIdle", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    WaitRandom(timeMin=0.4, timeMax=0.75, finishEvent=→"FINISHED", realTime=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Velocity", space=0, everyFrame=true)
    FloatCompare(float1=var "Y Velocity", float2=-0.1, tolerance=0, equal=(none), lessThan=→"DROP", greaterThan=(none), everyFrame=true)
    on FINISHED → Range Check
    on TOOK DAMAGE → Range Check
    on DROP → Drop
    on CLOSE → Close
    on SING → Sing Ground
    on UNALERT → Unalert Pause
  }

  state Drop {
    SetGravity2dScale(gameObject=Self, gravityScale=var "Gravity")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hop")
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)

    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=→"FLY", belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=3, reverseIfNegativeScale=false, everyFrame=true)
    on LAND → Land
    on FLY → Set Flap
  }

  state Land {
    PlayAudioEventRandom(audioClips=[crowman_dagger_ground_land_1 (sfxstatic_assets_areagreymoor.bundle), crowman_dagger_ground_land_2 (sfxstatic_assets_areagreymoor.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)

    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Throw Range", storeResult=false, sendEvent="MID", outOfRangeEvent="FAR", everyFrame=false)
    on CLOSE → Close
    on MID → Mid
    on FAR → Far
    on SING → Sing Ground
  }

  state Close {
    SendRandomEventV4(events=[→"HOP AWAY", →"FLY"], weights=[1, 0.5], eventMax=[2, 1], missedMax=[1, 2], activeBool=(unset))
    on HOP AWAY → Hop Away
    on FLY → Room Overhead?
  }

  state Hop Away {
    SetFloatValue(floatVariable=var "Hop Speed", floatValue=10, everyFrame=false)
    RandomInt(min=2, max=3, storeResult=var "Hops", inclusiveMax=true, noRepeat=false)
    on FINISHED → Hop Antic
  }

  state Hop Antic {

  }

  state Hop Air {

    on FLY → Set Flap
  }

  state Hop Bounce {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hop Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hop Launch
  }

  state Hop Finish {
    IntAdd(intVariable=var "Hops", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hops", integer2=0, equal=→"LAND", lessThan=→"LAND", greaterThan=→"BOUNCE", everyFrame=false)
    on LAND → Hop Land
    on BOUNCE → Hop Bounce
  }

  state Hop To {

  }

    on HOP AWAY → Hop Away
    on FLY → Room Overhead?
    on THROW → G Throw Antic
  }

  state Far {

    on FLY → Room Overhead?
  }

    GetAngleBetweenPointsV2(originX=var "Self X", originY=var "Self Y", targetX=var "X Target", targetY=var "Y Target", storeAngle=var "Angle", everyFrame=false)
    RandomFloat(min=8, max=10, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    RandomFloat(min=11, max=13, storeResult=var "Speed")

    IntCompare(integer1=var "Flaps", integer2=0, equal=(none), lessThan=(none), greaterThan=→"REPEAT", everyFrame=false)
    on REPEAT → Flap
    on CLOSE → 
    on FINISHED → Flap Move
    on SING → Sing Air
  }
