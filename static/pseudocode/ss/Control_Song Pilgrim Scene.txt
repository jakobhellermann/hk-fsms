# Control
# 1 location(s):
#   Song_15 / Song Pilgrim Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State
  }

  state State {
    FindNamedChild(gameObject=Self, storeResult=var "Woken")
    FindNamedChild(gameObject=Self, storeResult=var "Unwoken")
    BoolTest(boolVariable=var "Activated", isTrue=→"WOKEN", isFalse=→"UNWOKEN", everyFrame=false)
    on WOKEN → Woken
    on UNWOKEN → Unwoken
  }

  state Woken {
    ActivateGameObject(gameObject=var "Unwoken", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Woken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Unwoken {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    ActivateGameObject(gameObject=var "Unwoken", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Woken", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on ENTER → Wake
  }

  state Wake {
    SendEventByName(eventTarget=GameObject(var "Unwoken"), sendEvent="WAKE", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  var Activated: bool = false
  var Woken: gameObject = <null>
  var Unwoken: gameObject = <null>
}
