# Start Control
# 1 location(s):
#   Under_18 / Battle Scene/Battle Start Sequence

fsm Start Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Idle {
    on OPEN → Start Pause
  }

  state Start Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Gate Close
  }

  state Gate Close {
    CallMethodProper(gameObject=Self, behaviour="BattleScene", methodName="StartBattle", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    ActivateGameObject(gameObject=var "CamLocks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Light Open
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=Self, storeResult=var "City Spot Lamp Closed")
    FindNamedChild(gameObject=Self, storeResult=var "City Spot Lamp Open")
    FindNamedChild(gameObject=Self, storeResult=var "Darkness Region")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Gates")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "CamLocks")
    on FINISHED → Idle
  }

  state Light Open {
    PlayAudioEvent(audioClip=city_spot_lamp_open (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "City Spot Lamp Closed", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCollider(gameObject=var "Darkness Region", active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "City Spot Lamp Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "City Spot Lamp Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0.1)
    on FINISHED → Begin Batle
  }

  state Begin Batle {
    CallMethodProper(gameObject=var "Battle Scene", behaviour="BattleScene", methodName="StartBattle", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  var Battle Scene: gameObject = <null>
  var City Spot Lamp Closed: gameObject = <null>
  var City Spot Lamp Open: gameObject = <null>
  var Darkness Region: gameObject = <null>
  var Gates: gameObject = <null>
  var CamLocks: gameObject = <null>
}
