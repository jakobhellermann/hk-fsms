# Custom Dialogue - Act3 Rescue
# 1 location(s):
#   Greymoor_24 / Black Thread States/Black Thread World/Caravan Lech

fsm Custom Dialogue - Act3 Rescue {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=1.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT → Convo Check
    on STARTLE → Startle
    on SING → Sing
  }

  state Convo Check {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="FLEA_LECH")
    PlayerDataBoolTest(boolName="CaravanLechSaved", isTrue=→"REPEAT", isFalse=→"MEET")
    on MEET → Meet
    on REPEAT → Repeat
  }

  state Meet {
    RunDialogue(Sheet="Caravan", Key="CARAVAN_LECH_SAVED_ACT3", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetPlayerDataBool(boolName="CaravanLechSaved", value=true)
    SetPlayerDataBool(boolName="CaravanLechAct3Convo", value=true)
    on CONVO_END → End
  }

  state Repeat {
    RunDialogue(Sheet="Caravan", Key="CARAVAN_LECH_SAVED_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
  }

  state End {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Sing {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.7, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=kratt_sing (sfxdynamic_assets_fleacaravan.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
  }

  sequence state Sing End {
    Tk2dPlayAnimationWait(Target=Self, ClipName="Sing End", AnimationCompleteEvent=(none))
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    on FINISHED → Idle
  }

  state Startle {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    Wait(time=0.5, finishEvent=→"CANCEL", realTime=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=kratt_alert (sfxdynamic_assets_fleacaravan.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on CANCEL → Sing End
  }

  var Audio Loop Voice: gameObject = Black Thread States/Black Thread World/Caravan Lech/Audio Loop Voice
}
