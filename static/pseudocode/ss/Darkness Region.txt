# Darkness Region
# 1 location(s):
#   Library_15 / Darkness Region

fsm Darkness Region {
  start Pause
  on DISABLE → Exit 2  // from any state

  state Init {
    AddTrackTrigger(target=Self)
    on FINISHED → Out
  }

  state Out {
    GetTrackTriggerCount(Target=Self, StoreCount=var "Trigger Count", EveryFrame=true)
    IntCompareToBool(integer1=var "Trigger Count", integer2=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Trigger Change", everyFrame=true)
    BoolTestMulti(boolVariables=[var "Trigger Change", var "Change Blocked"], boolStates=[true, false], trueEvent=→"ENTER", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on ENTER → Enter
  }

  state Enter {
    GetTrackTriggerCount(Target=Self, StoreCount=var "Trigger Count", EveryFrame=true)
    IntCompareToBool(integer1=var "Trigger Count", integer2=0, equalBool=var "Trigger Change", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Trigger Change", var "Change Blocked"], boolStates=[true, false], trueEvent=→"EXIT", falseEvent=(none), storeResult=(unset), everyFrame=true)
    SetBoolValue(boolVariable=var "Is Inside", boolValue=true, everyFrame=false)
    SetDarknessLevel(SetLevel=var "Darkness", StoreCurrentLevel=var "Darkness Before")
    on EXIT → Exit
  }

  state Exit {
    GetTrackTriggerCount(Target=Self, StoreCount=var "Trigger Count", EveryFrame=true)
    IntCompareToBool(integer1=var "Trigger Count", integer2=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Trigger Change", everyFrame=true)
    BoolTestMulti(boolVariables=[var "Trigger Change", var "Change Blocked"], boolStates=[true, false], trueEvent=→"ENTER", falseEvent=(none), storeResult=(unset), everyFrame=true)
    SetBoolValue(boolVariable=var "Is Inside", boolValue=false, everyFrame=false)
    SetDarknessLevel(SetLevel=var "Darkness Before", StoreCurrentLevel=(unset))
    on ENTER → Enter
  }

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → Init
  }

  state Exit 2 {
    BoolTest(boolVariable=var "Is Inside", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Is Inside", boolValue=false, everyFrame=false)
    SetDarknessLevel(SetLevel=var "Darkness Before", StoreCurrentLevel=(unset))
    on FINISHED → Inert
  }

  state Inert {
  }

  var Darkness: int = 1

  var Darkness Before: int = 0
  var Trigger Count: int = 0
  var Is Inside: bool = false
  var Trigger Change: bool = false
  var Change Blocked: bool = false
}
