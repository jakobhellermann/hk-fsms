# Battle Range Check
# 1 location(s):
#   Peak_07 / Pinstress Control/Pinstress Scene/Pinstress Boss

fsm Battle Range Check {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  state Init {
    FindAlertRange(target=Self, storeResult=var "Battle Range", childName="Battle Range")
    SetParent(gameObject=var "Battle Range Obj", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Range In {
    SetFloatValue(floatVariable=var "Range Out Timer", floatValue=0, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Battle Range Out", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="In Range Out Grace", setValue=false, everyFrame=false)
    CheckAlertRange(alertRange=var "Battle Range", storeResult=(unset), InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=→"RANGE OUT", OutOfRangeDelay=0, everyFrame=true)
    on RANGE OUT → Range Out Grace
  }

  state Idle {
    on BATTLE START → Range In
  }

  state Range Out Grace {
    FloatAdd(floatVariable=var "Range Out Timer", add=1, everyFrame=true, perSecond=true)
    CheckAlertRange(alertRange=var "Battle Range", storeResult=(unset), InRangeEvent=→"RANGE IN", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    FloatCompare(float1=var "Range Out Timer", float2=3, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"TIMEOUT", everyFrame=true)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="In Range Out Grace", setValue=true, everyFrame=false)
    on RANGE IN → Range In
    on TIMEOUT → Timed Out
  }

  state Timed Out {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Battle Range Out", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="In Range Out Grace", setValue=false, everyFrame=false)
    CheckAlertRange(alertRange=var "Battle Range", storeResult=(unset), InRangeEvent=→"RANGE IN", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    on RANGE IN → Range In
  }

  var Range Out Timer: float = 0
  var Battle Range Obj: gameObject = Pinstress Control/Pinstress Scene/Pinstress Boss/Battle Range
  var Battle Range: object = Pinstress Control/Pinstress Scene/Pinstress Boss/Battle Range@AlertRange
}
