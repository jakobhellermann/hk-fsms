# Chime Sfx
# 1 location(s):
#   Song_Enclave / Black Thread States/Black Thread World/Enclave Act 3/Sherma Caretaker

fsm Chime Sfx {
  start Idle
  on CHIME END → Idle  // from any state
  on CHIME START → Frame Check  // from any state

  state Idle {
  }

  state Frame Check {
    Tk2dSpriteGetId(gameObject=Self, spriteID=var "Sprite ID", everyframe=true)
    IntCompare(integer1=var "Sprite ID", integer2=13, equal=→"CHIME", lessThan=(none), greaterThan=(none), everyFrame=true)
    on CHIME → Play Chime
  }

  state Play Chime {
    PlayAudioEventRandom(audioClips=[sherma_bell_ring_single_1 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_2 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_3 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_4 (sfxstatic_assets_sherma.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Frame Check
  }

  var Sprite ID: int = 0
}
