# FSM
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/Skull King Audio Control

fsm FSM {
  start Wait Event

  state Wait Event {
    PlayerDataBoolTest(boolName="skullKingWillInvade", isTrue=(none), isFalse=→"CANCEL")
    on CANCEL → Deactivate
    on NPC COWER START → Transition
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Transition {
    ActivateGameObject(gameObject=var "Bonetown Crowd invasion audio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeAudio(gameObject=var "Bonetown Crowd calm audio", startVolume=1, endVolume=0, time=0.3)
    FadeAudio(gameObject=var "Bonetown Crowd invasion audio loop", startVolume=0, endVolume=1, time=0.3)
    Wait(time=0.3, finishEvent=(none), realTime=false)
    on SKULL KING KILLED → Wait
    on FINISHED → Deactivate
  }

  state Skull King Killed {
    FadeAudio(gameObject=var "Bonetown Crowd calm audio", startVolume=0, endVolume=1, time=0.3)
    FadeAudio(gameObject=var "Bonetown Crowd invasion audio loop", startVolume=1, endVolume=0, time=0.3)
    Wait(time=0.3, finishEvent=(none), realTime=false)
    on FINISHED → Deactivate
  }

  state Wait {
    Wait(time=0.15, finishEvent=(none), realTime=false)
    on FINISHED → Skull King Killed
  }

  var Bonetown Crowd calm audio: object = Black Thread States/Normal World/Bonetown Crowd calm audio
  var Bonetown Crowd invasion audio: object = Black Thread States/Normal World/Bonetown Crowd invasion audio
  var Bonetown Crowd invasion audio loop: object = Black Thread States/Normal World/Bonetown Crowd invasion audio/Bonetown Crowd invasion audio loop
  var Bonetown Crowd invasion audio start: object = Black Thread States/Normal World/Bonetown Crowd invasion audio/Bonetown Crowd invasion audio start
}
