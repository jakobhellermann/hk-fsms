# Camera Control
# 1 location(s):
#   Abyss_09 / Lava Rise Control/Rising Lava/Cam Controller

fsm Camera Control {
  start Pause

  state Init {
    FindGameObject(objectName="", withTag="CameraTarget", store=var "CamTarget")
    FindNamedChild(gameObject=Self, storeResult=var "Cam Point")  // disabled
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Idle {
    on CAM START → Cam Follow
  }

  state Cam Follow {
    CallMethodProper(gameObject=var "CamTarget", behaviour="CameraTarget", methodName="StartFreeMode", parameters=[false], storeResult=(unset var), EveryFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Cam Point Y", z=(unset), space=0, everyFrame=true)
    SetFloatToHighest(floatVariable=var "Target Y", value1=var "Cam Point Y", value2=var "Hero Y", everyFrame=true)
    SetPosition(gameObject=var "CamTarget", vector=(unset), x=(unset), y=var "Target Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on CAM END → End Free Mode
  }

  state End Free Mode {
    CallMethodProper(gameObject=var "CamTarget", behaviour="CameraTarget", methodName="EndFreeMode", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  var Target X: float = 0
  var Target Y: float = 0
  var Hero Y: float = 0
  var Cam Point Y: float = 0
  var CamTarget: gameObject = <null>
  var Cam Point: gameObject = <null>
}
