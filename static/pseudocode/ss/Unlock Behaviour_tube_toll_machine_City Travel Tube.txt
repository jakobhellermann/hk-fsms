fsm Unlock Behaviour {
  start Init

  state Inert {
    StringCompare(stringVariable=var "Unlocked PD Bool", compareTo="", equalEvent=→"ACTIVATED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName=var "Unlocked PD Bool", isTrue=→"ACTIVATED", isFalse=(none))
    on INTERACT → Prompt
    on ACTIVATED → Disabled
  }

  state End Interaction {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Inert
  }

  state Init {
    SetAnimator(target=Self, active=false)
    GetFsmString(gameObject=var "Unlock", fsmName="Tube Travel", variableName="Unlocked PD Bool", storeValue=var "Unlocked PD Bool", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Inert
  }

  state Retract Animation {
    PlayAudioEvent(audioClip=bell_bench_pay_dish_retract_start_on_rise (sfxstatic_assets_bench.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetPlayerDataBool(boolName=var "Unlocked PD Bool", value=true)
    SendMessageV2(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl(), everyFrame=false)
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SetAnimator(target=Self, active=true)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="CheckTubeAchievements", parameters=[], storeResult=(unset var), EveryFrame=false)
    AnimatorPlayStateWait(target=Self, stateName="Retract", finishEvent=→"FINISHED")
    on FINISHED → After Retract Pause
  }

  state After Retract Pause {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop
  }

  state Disabled {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Stop {
    SendEventByName(eventTarget=GameObject(var "Unlock"), sendEvent="UNLOCK", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Prompt {
    RunFSM(fsmTemplateControl=template=-8990696591063553075 in[Hornet Stays Reaching<-false, Prompt Text Override<-"", Dish<-var "Dish", Cost Reference<-var "Cost Reference"], finishEvent=→"FINISHED", everyFrame=false)
    on CANCEL → End Interaction
    on FINISHED → Retract Animation
  }

  var Cost Reference: object = City Tube Toll Default (dataassets_assets_assets/dataassets/costs.bundle)

  var Activated: bool = false
  var Unlocked PD Bool: string = ""
  var Dish: gameObject = City Travel Tube/tube_toll_machine/toll_machine/toll_machine_dish
  var Unlock: gameObject = City Travel Tube
}
