# Dialogue
# 1 location(s):
#   Bone_18 / Quest Scene/Final Encounter/Battle Scene/Pilgrim Stomper Fight NPC

fsm Dialogue {
  start Pause

  state Init {
    CheckQuestState(NotTrackedEvent=(none), TrackedEvent=→"ACTIVATED", CompletedEvent=(none), Quest=<null>)  // disabled
    on FINISHED → Refight Ready
  }

  state Idle {
    on INTERACT → Pre Prompt
  }

  state Met? {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    BeginQuestV2(ShowPrompt=false, Quest=<null>)
    on FINISHED → Final Dialogue
  }

  state Final Dialogue {
    GetPlayerDataBool(boolName="blackThreadWorld", storeValue=var "Is Act 3")
    ConvertBoolToString(boolVariable=var "Is Act 3", stringVariable=var "Convo Key", falseString="PILGRIM_STOMPER_QUEST_FINAL", trueString="PILGRIM_STOMPER_QUEST_FINAL_ACT3", everyFrame=false)
    RunDialogue(Sheet="Bonebottom", Key=var "Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Fight Pause
  }

  state Fight Pause {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Battle
  }

  state Start Battle {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    SendMessage(gameObject=var "Battle Scene", delivery=SendMessage, options=1, functionCall=StartBattle())
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Refight Ready {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="START", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"CHECK ARENA", ActiveOuter=→"CHECK ARENA", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"CHECK ARENA", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=<null>, ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=0, storeMagnitudeMultiplier=0, firstHitOnly=false)
    on START → Start Battle
    on CHECK ARENA → Check Arena
  }

  state Give Prompt {
    QuestCompleteYesNo(Quest=<null>, ConsumeCurrency=true, YesEvent=→"TRUE", NoEvent=→"FALSE", ReturnHUDAfter=false)
    on TRUE → Met?
    on FALSE → Decline
  }

  state End {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    on FINISHED → Idle
  }

  state Pre Prompt {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=true, npcTitle="LEARNEDPILGRIM")
    RunDialogue(Sheet="Bonebottom", Key="PILGRIM_STOMPER_QUEST_READY", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Give Prompt
  }

  state Decline {
    RunDialogue(Sheet="Bonebottom", Key="PILGRIM_STOMPER_QUEST_FINAL_DECLINE", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
  }

  state Check Arena {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Arena Range", storeResult=false, sendEvent="START", outOfRangeEvent="FINISHED", everyFrame=false)
    on FINISHED → Refight Ready
    on START → Start Battle
  }

  var Is Act 3: bool = false
  var Convo Key: string = ""
  var Battle Scene: gameObject = <null>
  var Attack Detector: gameObject = Quest Scene/Final Encounter/Battle Scene/Pilgrim Stomper Fight NPC/Attack Detector
}
