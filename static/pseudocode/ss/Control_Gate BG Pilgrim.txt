# Control
# 1 location(s):
#   Coral_Judge_Arena / Boss Scene/Pilgrims/Gate BG Pilgrim

fsm Control {
  start Init
  on KILL PILGRIMS → Die  // from any state
  on SING → Sing Sequence  // from any state
  on SCARED → Scared  // from any state
  on SING MANUAL → Sing  // from any state

  state Die {
    FindNamedChild(gameObject=Self, storeResult=var "Death")
    ActivateGameObject(gameObject=var "Death", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayRandomAudioClipTableV3(Table=choir_envoy_death (sfxdynamic_assets_areacoral.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    SetParent(gameObject=var "Death", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Init {
    PreBuildTK2DSprites(gameObject=Boss Scene/Pilgrims/Gate BG Pilgrim/Death, useChildren=true)
    on FINISHED → Rest
  }

  state Rest {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Round Rest")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=2, None=(none), ActiveInner=→"SING MANUAL", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ALERT → Rest End
    on STARTLE → Startle
  }

  state Rest End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Round Rest End Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))  // disabled
    on FINISHED → Facing Right
    on STARTLE → Startle
  }

  state Facing Right {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Round Idle Right")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"TURN", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=2, None=(none), ActiveInner=→"SING MANUAL", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on TURN → Turn Left
    on STARTLE → Startle
  }

  sequence state Turn Left {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"CANCEL", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))  // disabled
    Tk2dPlayAnimationWait(Target=Self, ClipName="Round Turn Left", AnimationCompleteEvent=(none))
    on FINISHED → Facing Left
    on CANCEL → Facing Right
    on STARTLE → Startle
  }

  state Facing Left {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Round Idle Left")
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"TURN", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="", outOfRangeEvent="UNALERT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=2, None=(none), ActiveInner=→"SING MANUAL", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on TURN → Turn Right
    on UNALERT → Return To Rest
    on STARTLE → Startle
  }

  sequence state Turn Right {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"CANCEL", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))  // disabled
    Tk2dPlayAnimationWait(Target=Self, ClipName="Round Turn Right", AnimationCompleteEvent=(none))
    on FINISHED → Facing Right
    on CANCEL → Facing Left
    on STARTLE → Startle
  }

  sequence state Return To Rest {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="CANCEL", outOfRangeEvent="", everyFrame=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))  // disabled
    Tk2dPlayAnimationWait(Target=Self, ClipName="Round Turn Right", AnimationCompleteEvent=(none))
    on FINISHED → Rest
    on CANCEL → Facing Left
    on STARTLE → Startle
  }

  state Sing Sequence {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Round Sing")
    SetRandomAudioClipFromTable(Table=pilgrim_solemn_gate_sing (sfxdynamic_assets_areacoral.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    on SING END → Sing Sequence End
  }

  state Sing Sequence End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Round Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0.2, cancelOnEarlyExit=false)
    on FINISHED → Facing Right
  }

  state Scared {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Round Sing")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=pilgrim_solemn_alert (sfxdynamic_assets_areacoral.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Round Sing")
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=pilgrim_solemn_gate_sing (sfxdynamic_assets_areacoral.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.7, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on CANCEL → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Round Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest
  }

  state Startle {
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=pilgrim_solemn_alert (sfxdynamic_assets_areacoral.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Round Sing")
    Wait(time=0.4, finishEvent=→"CANCEL", realTime=false)
    on CANCEL → Sing End
  }

  var Death: gameObject = <null>
  var Audio Loop Voice: gameObject = Boss Scene/Pilgrims/Gate BG Pilgrim/Audio Loop Voice
}
