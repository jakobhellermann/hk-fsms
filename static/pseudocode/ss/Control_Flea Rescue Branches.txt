# Control
# 4 location(s):
#   Bone_06 / Flea Rescue Branches
#   Dust_09 / Flea Rescue Branches
#   Shadow_10 / Flea Rescue Branches
#   Shellwood_03 / Flea Rescue Branches

fsm Control {
  start Wait

  state Init {
    CheckQuestPdSceneBool(QuestTarget=var "Quest Target", ExpectedValue=true, trueEvent=→"ACTIVATE", falseEvent=(none), storeValue=(unset))
    on ACTIVATE → Activate
    on FINISHED → Idle
  }

  state Activate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    AudioPlaySimple(gameObject=var "Audio Loop Voice", volume=1, oneShotClip=<null>)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"CUT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"CUT", sendEventHeavy=→"BREAK", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on CUT → Hit
    on BREAK → Cut
  }

  state Cut {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=shellwood_twig_wall_hit_3 (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "Notfify"), sendEvent="FLEA RESCUED", delay=0, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Silk", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flea", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flea Rescue Activation", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Fx", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Flea"), sendEvent="RESCUE", delay=0, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
  }

  state Wait {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Hit {
    CheckTrackTriggerCount(target=var "Hittable Range", count=1, test=LessThan, everyFrame=false, successEvent=→"CANCEL")
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    PlayAudioEventRandom(audioClips=[shellwood_twig_wall_hit_1 (sfxstatic_assets_areashellwood.bundle), shellwood_twig_wall_hit_2 (sfxstatic_assets_areashellwood.bundle), shellwood_twig_wall_hit_3 (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    SetScale(gameObject=var "Slash", vector=(unset), x=(unset), y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Slash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Hit Fx", resetTimeIfPlaying=false, stopOnStateExit=false)
    SendMessage(gameObject=var "Silk", delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Flea", position=(unset), rotation=(unset), storeObject=var "")
    on BREAK → Cut
    on FINISHED → Idle
    on CANCEL → Cancel Frame
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Notfify: gameObject = <null>

  var X Scale: float = 0
  var Hits: int = 3
  var Break Fx: gameObject = Flea Rescue Branches/Break Fx
  var Flea: gameObject = Flea Rescue Branches/Flea Distressed
  var Silk: gameObject = Flea Rescue Branches/Silk
  var Slash: gameObject = Flea Rescue Branches/Slash
  var Hit Fx: gameObject = Flea Rescue Branches/Hit Fx
  var Hittable Range: gameObject = Flea Rescue Branches/Hittable Range
  var Flea Rescue Activation: gameObject = Flea Rescue Branches/Flea Rescue Activation
  var Audio Loop Voice: gameObject = <null>
  var Quest Target: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
