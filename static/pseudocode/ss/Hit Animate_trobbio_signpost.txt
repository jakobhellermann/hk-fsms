# Hit Animate
# 1 location(s):
#   Library_13 / Grand Stage Scene/First_Stage/Post_defeat_set/trobbio_signpost

fsm Hit Animate {
  start Idle

  state Idle {
    on HIT_UP → Choose Dir
    on HIT_DOWN → Choose Dir
    on HIT_LEFT → Hit L
    on HIT_RIGHT → Hit R
    on TINK UP → Choose Dir
    on TINK DOWN → Choose Dir
    on TINK LEFT → Hit L
    on TINK RIGHT → Hit R
  }

  state Hit L {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignToBool(Value=var "X Scale", StoreIsPositive=var "Scale Positive", EveryFrame=false)
    ConvertBoolToString(boolVariable=var "Scale Positive", stringVariable=var "Anim", falseString="Hit Right", trueString="Hit Left", everyFrame=false)
    AnimatorPlay(gameObject=Self, stateName=var "Anim", layer=0, normalizedTime=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-2, y=(unset), everyFrame=false)
    on FINISHED → Hit Fx
  }

  state Choose Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HIT_LEFT", leftEvent=→"HIT_RIGHT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=true, everyFrame=false)
    on HIT_LEFT → Hit L
    on HIT_RIGHT → Hit R
  }

  state Hit R {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignToBool(Value=var "X Scale", StoreIsPositive=var "Scale Positive", EveryFrame=false)
    ConvertBoolToString(boolVariable=var "Scale Positive", stringVariable=var "Anim", falseString="Hit Left", trueString="Hit Right", everyFrame=false)
    AnimatorPlay(gameObject=Self, stateName=var "Anim", layer=0, normalizedTime=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=2, y=(unset), everyFrame=false)
    on FINISHED → Hit Fx
  }

  state Hit Fx {
    PlayAudioEvent(audioClip=hornet_hit_reject_dull (herosfxstatic_assets_all.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateInteractible(Target=var "Inspect Region", Activate=false, AllowQueueing=false, UseChildren=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "inspect plink", activate=false, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Ac Interactible
  }

  state Ac Interactible {
    ActivateInteractible(Target=var "Inspect Region", Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  var X Scale: float = 0
  var Scale Positive: bool = false
  var Anim: string = ""
  var Inspect Region: gameObject = Grand Stage Scene/First_Stage/Post_defeat_set/trobbio_signpost/Inspect Region
  var inspect plink: gameObject = Grand Stage Scene/First_Stage/Post_defeat_set/trobbio_signpost/inspect plink
}
