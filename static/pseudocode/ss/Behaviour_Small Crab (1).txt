# Behaviour
# 1 location(s):
#   Crawl_04 / Little Crabs/Crabs/Small Crab (1)

fsm Behaviour {
  start Init

  state Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"WALK", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on WALK → Walk
    on SING → Sing
  }

  state Walk {
    RandomBool(storeResult=var "Schedule Turn")
    StartCrawler(Target=Self, ScheduleTurn=var "Schedule Turn")
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"IDLE", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Crawl")
    on IDLE → To Idle
    on SING → To Sing
  }

  state Init {
    StopCrawler(Target=Self, WaitForTurn=true)
    on FINISHED → Idle
  }

  state To Idle {
    StopCrawler(Target=Self, WaitForTurn=true)
    on FINISHED → Idle
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=<null>, noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state To Sing {
    StopCrawler(Target=Self, WaitForTurn=true)
    on FINISHED → Sing
  }

  var Schedule Turn: bool = false
  var Pt Crawl: gameObject = Little Crabs/Crabs/Small Crab (1)/Pt Crawl
  var Audio Loop Voice: gameObject = Little Crabs/Crabs/Small Crab (1)/Audio Loop Voice
}
