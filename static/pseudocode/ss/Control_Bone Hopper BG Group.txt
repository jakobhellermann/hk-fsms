# Control
# 1 location(s):
#   Bone_East_24 / Bone Hopper BG Group

fsm Control {
  start Start Pause

  state Start Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Scene Start {
    GetRandomChild(gameObject=Self, storeResult=var "Next Hopper")
    SetPosition2d(gameObject=var "Next Hopper", vector=(unset), x=110, y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Hopper"), sendEvent="MANUAL SPAWN", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Hopper")
    SetPosition2d(gameObject=var "Next Hopper", vector=(unset), x=130, y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Hopper"), sendEvent="MANUAL SPAWN", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Hopper")
    SetPosition2d(gameObject=var "Next Hopper", vector=(unset), x=150, y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Hopper"), sendEvent="MANUAL SPAWN", delay=0, everyFrame=false)
    on FINISHED → Wait
  }

  state Wait {
    WaitRandom(timeMin=2, timeMax=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spawn
  }

  state Spawn {
    GetRandomChild(gameObject=Self, storeResult=var "Next Hopper")
    SendEventByName(eventTarget=GameObject(var "Next Hopper"), sendEvent="SPAWN", delay=0, everyFrame=false)
    on FINISHED → Wait
  }

  var Next Hopper: gameObject = <null>
}
