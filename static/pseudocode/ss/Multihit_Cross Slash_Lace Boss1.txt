# Multihit
# 1 location(s):
#   Bone_East_12 / Boss Scene/Lace Boss1/Cross Slash

fsm Multihit {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=Self, childName="Hornet Hit", storeResult=var "Hornet Hit")
    ActivateGameObject(gameObject=var "Hornet Hit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Attacking {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor 2D (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClip=lace_multi_slash_hornet_damage (sfxstatic_assets_laceboss.bundle), pitchMin=1, pitchMax=1, volume=0, delay=0, storePlayer=var "Audio Player")
    on MULTIHIT → Register Hit
  }

  state Register Hit {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="WillDoBellBindHit", parameters=[], storeResult=var "Bind Bell Hit", EveryFrame=false)
    BoolTest(boolVariable=var "Bind Bell Hit", isTrue=→"BIND BELL", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Hornet Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ScreenFlash(flashColour=(1, 1, 1, 0.528))
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=true)
    SetPositionToObject(gameObject=var "Hero", targetObject=var "Self", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SetFsmBool(gameObject=var "Parent", fsmName="Control", variableName="CrossSlashing Hero", setValue=true, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    FadeAudio(gameObject=var "Audio Player", startVolume=0, endVolume=1, time=0.1)
    on BIND BELL → Bind Bell Damage
  }

  state Idle {
    on ATTACK START → Attacking
    on MULTIHIT → Register Hit
  }

  state Bind Bell Damage {
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=false)
    on FINISHED → Idle
  }

  var Bind Bell Hit: bool = false
  var Audio Player: gameObject = <null>
  var Hornet Hit: gameObject = <null>
  var Parent: gameObject = <null>
  var Self: gameObject = <null>
  var Lace: gameObject = <null>
}
