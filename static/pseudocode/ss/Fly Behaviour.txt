# Fly Behaviour
# 1 location(s):
#   localpoolprefabs_assets_lifeblood.bundle / Health Flyer

fsm Fly Behaviour {
  start Idle

  state Idle {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopBounce())
    on FLY → Flip Out
  }

  state Flip Out {
    RandomFloatV2(min=120, max=60, storeResult=var "Fling Angle", everyFrame=false)
    RandomFloatV2(min=5, max=8, storeResult=var "Fling Speed", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Fling Angle", speed=var "Fling Speed", everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    GetVelocity2d(gameObject=Self, vector=var "Velocity", x=(unset), y=(unset), space=0, everyFrame=true)
    GetVector2Length(vector2=var "Velocity", storeLength=var "Velocity Magnitude", everyFrame=true)
    FloatCompare(float1=var "Velocity Magnitude", float2=1, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Fly Away
  }

  state Fly Away {
    DistanceFlySmooth(gameObject=Self, target=var "Hero", distance=20, speedMax=9, accelerationForce=1, targetRadius=2, deceleration=0.9, offset=(0, 0, 0))  // disabled
    DistanceFly(gameObject=Self, target=var "Hero", distance=30, speedMax=6, acceleration=0.05, targetsHeight=false, height=0, minAboveHero=(unset))
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartBounce())
    ShoveFromWall(gameObject=Self, shoveForce=10, rayLength=2, checkUp=true, checkDown=true, checkLeft=true, checkRight=true, everyFrame=true)
  }

  var Fling Angle: float = 0
  var Fling Speed: float = 0
  var Velocity Magnitude: float = 0
  var Velocity: vector2 = (0, 0)
}
