# Pop on boss death and set defeated
# 1 location(s):
#   Clover_19 / Boss Scene/Memory Orb Group (1)

fsm Pop on boss death and set defeated {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Orb Spawn")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Gates")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Gate Disappear Effects")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "CamLock B")
    PlayerDataBoolTest(boolName="defeatedWhiteCloverstag", isTrue=→"ALREADY DEFEATED", isFalse=(none))
    on WHITE CLOVERSTAG POP → Get Corpse
    on ALREADY DEFEATED → Activate 2
  }

  state Activate {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=Appear())
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetPlayerDataBool(boolName="defeatedWhiteCloverstag", value=true)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)  // disabled
    on FINISHED → End Arena
  }

  state End Arena {
    ActivateGameObject(gameObject=var "CamLock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gates", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate Disappear Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Get Corpse {
    FindGameObject(objectName="", withTag="Boss Corpse", store=var "Corpse")
    SetPositionToObject2D(gameObject=var "Orb Spawn", targetObject=var "Corpse", xOffset=0, yOffset=0, everyFrame=false)
    on FINISHED → Activate
  }

  state Activate 2 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=Appear())
    on FINISHED → 
    on LAST ORB COLLECTED → State 1
  }

  state State 1 {
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
  }

  var Activated: bool = false
  var Boss Scene: gameObject = <null>
  var CamLock B: gameObject = <null>
  var Corpse: gameObject = <null>
  var Gates: gameObject = <null>
  var Orb Spawn: gameObject = <null>
  var Gate Disappear Effects: gameObject = <null>
}
