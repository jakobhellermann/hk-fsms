# Death
# 1 location(s):
#   enemycorpses_assets_areashellwood.bundle / Corpse Seth

fsm Death {
  start Wait Frame

  state Stagger {
    GetOwner(storeGameObject=var "Self")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetPlayerDataBool(boolName="defeatedSeth", value=true)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    CreateObject(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=0.009, space=0, everyFrame=false, lateUpdate=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=SB_death_hit (sfxdynamic_assets_seth.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Steam
  }

  state Steam {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Boss Death Antic Unthreaded (localpoolprefabs_assets_boss.bundle), spawnPoint=var "Self", position=(0, 0, 0), rotation=(unset), storeObject=var "")
    PlayParticleEmitterInState(gameObject=var "Pt Steam")
    PlayAudioEvent(audioClip=boss_death_pt_2 (sfxstatic_assets_boss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=SB_pre_death (sfxdynamic_assets_seth.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Blow
  }

  state Blow {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=SB_post_death (sfxdynamic_assets_seth.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SpawnObjectFromGlobalPool(gameObject=Boss Death Explosion Unthreaded (localpoolprefabs_assets_boss.bundle), spawnPoint=var "Self", position=(0, 0, 0), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=boss_death_pt_3 (sfxstatic_assets_boss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    SendEventToRegisterDelay(EventName="SETH DEFEATED", delay=3, ExcludeTarget=Self)  // disabled
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetVelocityByScale(gameObject=Self, speed=8, ySpeed=20, everyFrame=false)
    SetParent(gameObject=var "Chunks", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Chunks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="SETH EXPLODE")
    on FINISHED → Fling Type
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Stagger
  }

  state Splash In {
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "Water Y", space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Splash Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Splash Effect", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendEventToRegisterDelay(EventName="SETH DEFEATED", delay=3, ExcludeTarget=Self)
  }

  state Fling Type {
    GetPosition2d(gameObject=var "Splash Check", vector_2d=(unset), x=var "Splash Check X", y=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Splash Check X", lowerValue=88, upperValue=146, boolVariable=(unset), trueEvent=→"FG", falseEvent=→"BG", everyFrame=false)
    on FG → Fling FG
    on BG → Fling Air
  }

  state Fling Air {
    CheckYPosition(gameObject=Self, compareTo=var "Water Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAND → Splash In
  }

  state Fling FG {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=-0.75, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Fling Air
  }

  var Water Y: float = 2.7
  var Splash Check X: float = 0
  var Chunks: gameObject = Corpse Seth/Chunks
  var Pt Steam: gameObject = <null>
  var Self: gameObject = <null>
  var Splash Effect: gameObject = Corpse Seth/Splash Effect
  var Splash Check: gameObject = Corpse Seth/Splash Check
  var Audio Loop Voice: gameObject = Corpse Seth/Audio Loop Voice
}
