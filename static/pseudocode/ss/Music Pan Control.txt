# Music Pan Control
# 2 location(s):
#   coremanagers_assets__gamemanager.bundle / _GameManager/AudioManager/Music
#   Menu_Title / _GameManager/AudioManager/Music

fsm Music Pan Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    EnableFsmSelf(SetEnabled=false)
    on LEAVING SCENE → Reset Pan
  }

  state Reset Pan {
    PanAudio(gameObject=var "Sub", targetPan=0, overTime=0.5)
    on FINISHED → Idle
  }

  var Sub: gameObject = _GameManager/AudioManager/Music/Sub
}
