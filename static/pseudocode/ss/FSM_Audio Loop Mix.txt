# FSM
# 1 location(s):
#   Mosstown_02c / moss_hut_set/NPC Scene/Audio Loop Mix

fsm FSM {
  start Init

  state Init {
    on FINISHED → Mixing
  }

  state Mixing {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0.5, time=0.25)  // disabled
    SetAudioPitch(gameObject=Self, pitch=1, everyFrame=false)  // disabled
    SetAudioClip(gameObject=Self, audioClip=moss_druid_idle (sfxstatic_assets_areamoss.bundle), autoPlay=true, stopOnExit=false)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    on MIX STOP → Stop
  }

  state Stop {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.15)  // disabled
    SetAudioPitch(gameObject=Self, pitch=0.75, everyFrame=false)  // disabled
    SetAudioClip(gameObject=Self, audioClip=swamp_water_loop_when_close (sfxstatic_assets_areaswamp.bundle), autoPlay=true, stopOnExit=false)
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    on MIX START → Mixing
    on MIX FAST → Mix Fast
  }

  state Mix Fast {
    SetAudioPitch(gameObject=Self, pitch=1.5, everyFrame=false)  // disabled
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0.75, time=0.15)  // disabled
    SetAudioVolume(gameObject=Self, volume=0.5, everyFrame=false)
    SetAudioClip(gameObject=Self, audioClip=moss_druid_mixing_rapidly (sfxstatic_assets_areamoss.bundle), autoPlay=true, stopOnExit=false)
    on MIX STOP → Stop
  }
}
