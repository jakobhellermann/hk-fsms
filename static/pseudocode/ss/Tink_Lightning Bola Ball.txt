# Tink
# 1 location(s):
#   herodynamic_assets_all.bundle / Lightning Bola Ball

fsm Tink {
  start State 1

  state State 1 {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=Attack(17), sendEvent=→"NAIL", storeCollider=var "Nail Object")
    on NAIL → Nail Hit
  }

  state Nail Hit {
    GetDamagerDirection(target=var "Nail Object", storeDirection=var "Direction")
    FloatAddRandom(floatVariable=var "Direction", addMin=-5, addMax=5)
    SetVelocityAsAngle(gameObject=Self, angle=var "Direction", speed=30, everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 1
  }

  var Direction: float = 0
  var Nail Object: gameObject = <null>
  var Pt Bounce: gameObject = Lightning Bola Ball/Pt Bounce
}
