# Control
# 4 location(s):
#   Ant_Queen / Memory Group/here/Deep Memory Tension
#   Clover_01 / Memory Scene Control/Memory Scene/Deep Memory Tension
#   Coral_Tower_01 / Memory Group/before/Deep Memory Tension
#   Shellwood_11b / memory_font/Uncompleted/Memory Group/Deep Memory Tension

fsm Control {
  start Init
  on ENTERING MEMORY → Entering Memory  // from any state

  state Init {
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Delay
  }

  state Play Tension Low Start {
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=0, endVolume=0.6, time=4)
    on TENSION END → Stop Tension
    on LEAVING SCENE → Stop Tension
    on ENTER DEEP MEMORY ZONE → Tension High
  }

  state Stop Tension {
    AudioStopV2(gameObject=Self, fadeTime=1, cancelOnEarlyExit=false)
  }

  state Delay {
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Play Tension Low Start
    on LEAVING SCENE → Stop Tension
    on TENSION END → Stop Tension
  }

  state Tension High {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=2)
    on TENSION END → Stop Tension
    on LEAVING SCENE → Stop Tension
    on EXIT DEEP MEMORY ZONE → Tension Low
  }

  state Tension Low {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0.6, time=4)
    on TENSION END → Stop Tension
    on LEAVING SCENE → Stop Tension
    on ENTER DEEP MEMORY ZONE → Tension High
  }

  state Entering Memory {
    AudioStopV2(gameObject=Self, fadeTime=3, cancelOnEarlyExit=false)
  }

  var Delay: float = 0
}
