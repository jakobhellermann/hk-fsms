# Sequence
# 1 location(s):
#   Belltown_Room_Spare / Farsight Cinematic Player

fsm Sequence {
  start Init
  on CONSTRUCT FARSIGHT → Cinematic  // from any state

  state Init {
    EnableFsmSelf(SetEnabled=false)
  }

  sequence state End {
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=1)
    Wait(time=1, finishEvent=(none), realTime=false)
    HeroControllerMethods(method=RegainControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    SendEventByNameV2(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    EnableFsmSelf(SetEnabled=false)
  }

  sequence state Cinematic {
    SetPlayerDataBool(boolName="disablePause", value=true)
    HeroControllerMethods(method=RelinquishControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    HeroControllerMethods(method=StopAnimationControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    TransitionToAudioSnapshot(snapshot=Cinematic (audioobjects_assets_all.bundle), transitionTime=1)
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 1), duration=1)
    Wait(time=1, finishEvent=(none), realTime=false)
    SendEventToRegister(eventName="ACTIVATE DISPLAYS")
    ScreenFader(startColour=(unset), endColour=(0, 0, 0, 0), duration=0)
    StartCinematic(CinematicPlayer=Self)
    on CINEMATIC END → UI Msg
  }

  sequence state UI Msg {
    HeroControllerMethods(method=StartAnimationControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    ScreenFader(startColour=(1, 1, 1, 0), endColour=(1, 1, 1, 1), duration=0)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    ScreenFader(startColour=(1, 1, 1, 1), endColour=(0, 0, 0, 1), duration=1)
    Wait(time=1, finishEvent=(none), realTime=false)
    CreateUIMsgGetItem(gameObject=UI Msg Get Item (prompts_assets_all.bundle), storeObject=var "UI Msg", sprite=hornet_world_sense (textures_assets_areabell.bundle))
    SetFsmString(gameObject=var "UI Msg", fsmName="Msg Control", variableName="Item", setValue="Farsight", everyFrame=false)
    on GET ITEM MSG END → End
  }

  var Item Name: string = ""
  var UI Msg: gameObject = <null>
}
