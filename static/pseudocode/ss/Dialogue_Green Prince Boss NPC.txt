# Dialogue
# 1 location(s):
#   Clover_10 / Boss Scene/Green Prince Boss NPC

fsm Dialogue {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "CamLock B")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Song Region Boss")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Song Region Full")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Dancer Control")
    FindNamedChild(gameObject=var "Dancer Control", storeResult=var "Pos5")
    on FINISHED → Wait for Enter
  }

  sequence state Wait for Enter {
    TriggerEnterEventSubscribe(trigger=Boss Scene/Green Prince Boss NPC/Enter Detector@TriggerEnterEvent, triggerEnteredEvent=→"ENTERED", triggerExitedEvent=(none), triggerStayedEvent=(none))
    on ENTERED → Encountered?
  }

  state Take Hero Control {
    SendEventToRegister(eventName="TENSION END")
    RunFSM(fsmTemplateControl=template=6159435039333135483 in[Clamp X<-91.5], finishEvent=→"FINISHED", everyFrame=false)
    on FINISHED → Turn Hero
  }

  state Convo 1 {
    RunDialogue(Sheet="Wanderers", Key="GREEN_PRINCE_VERDANIA_BOSS_1", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Turn To Hornet
  }

  sequence state Turn To Hornet {
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=Self, UseChildren=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    PlayRandomAudioClipTable(Table=small_movement_shell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NPC Look")
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Convo 2
  }

  state Convo 2 {
    RunDialogue(Sheet="Wanderers", Key="GREEN_PRINCE_VERDANIA_BOSS_2", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Dialogue End
  }

  sequence state Dialogue End {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Tele Out
  }

  state Encountered? {
    ActivateGameObject(gameObject=var "CamLock B", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="encounteredCloverDancers", isTrue=→"ENCOUNTERED", isFalse=→"UNENCOUNTERED")
    on UNENCOUNTERED → Take Hero Control
    on ENCOUNTERED → Encountered Start
  }

  state Battle Start {
    SendEventToRegister(eventName="BATTLE START")
    SetPlayerDataBool(boolName="encounteredCloverDancers", value=true)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Song Region Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Song Region Full", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Tele Out {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=clover_dancer_dash_03 (sfxstatic_assets_areacloverareacog.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Tele In
  }

  state Tele In {
    SetPositionToObject2D(gameObject=Self, targetObject=var "Pos5", xOffset=0, yOffset=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele In Q", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Battle Start
  }

  state Encountered Start {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventToRegister(eventName="TENSION END")
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Tele Out
  }

  state Turn Hero {
    HeroTurnToFace(Target=Self)
    on FINISHED → Convo 1
  }

  var Activated: bool = false
  var CamLock B: gameObject = <null>
  var Dancer Control: gameObject = <null>
  var Parent: gameObject = <null>
  var Pos5: gameObject = <null>
  var Song Region Boss: gameObject = <null>
  var Song Region Full: gameObject = <null>
}
