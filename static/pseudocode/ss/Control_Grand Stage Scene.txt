# Control
# 1 location(s):
#   Library_13 / Grand Stage Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Boss Scene Trobbio")
    FindNamedChild(gameObject=Self, storeResult=var "Boss Scene TormentedTrobbio")
    FindNamedChild(gameObject=Self, storeResult=var "First_Stage")
    FindNamedChild(gameObject=Self, storeResult=var "Second_Stage")
    FindNamedChild(gameObject=Self, storeResult=var "Third_Stage_refight")
    ActivateGameObject(gameObject=var "Boss Scene Trobbio", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Boss Scene TormentedTrobbio", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "First_Stage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Second_Stage", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Third_Stage_refight", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Act Check
  }

  state Act 2 {
    ActivateGameObject(gameObject=var "First_Stage", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="defeatedTrobbio", isTrue=→"COMPLETE", isFalse=→"TROBBIO")
    on TROBBIO → Trobbio Ready
    on COMPLETE → Boss Defeated
  }

  state Trobbio Ready {
    ActivateGameObject(gameObject=var "Boss Scene Trobbio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Boss Defeated {
  }

  state Act Check {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"ACT 3", isFalse=→"ACT 2")
    on ACT 2 → Act 2
    on ACT 3 → Act 3
  }

  state Act 3 {
    CheckQuestState(NotTrackedEvent=→"PREP", TrackedEvent=→"TROBBIO", CompletedEvent=→"COMPLETE", Quest=Tormented Trobbio (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    on TROBBIO → Trobbio Ready 2
    on COMPLETE → Boss Defeated Act3
    on PREP → Preparing Stage
  }

  state Trobbio Ready 2 {
    ActivateGameObject(gameObject=var "Boss Scene TormentedTrobbio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Third_Stage_refight", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on TORMENTED BATTLE END → Spawn Tormented Item
  }

  state Boss Defeated Act3 {
    ActivateGameObject(gameObject=var "Third_Stage_refight", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Preparing Stage {
    ActivateGameObject(gameObject=var "Second_Stage", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Spawn Tormented Item {
    SetPlayerDataBool(boolName="defeatedTormentedTrobbio", value=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="SPAWN ITEM")
    on FINISHED → End Battle
  }

  state End Battle {
    FindNamedChild(gameObject=var "Boss Scene TormentedTrobbio", storeResult=var "CamLock Boss")
    FindNamedChild(gameObject=var "Boss Scene TormentedTrobbio", storeResult=var "Gates")
    ActivateGameObject(gameObject=var "CamLock Boss", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  var Boss Scene TormentedTrobbio: gameObject = <null>
  var Boss Scene Trobbio: gameObject = <null>
  var First_Stage: gameObject = <null>
  var Second_Stage: gameObject = <null>
  var Third_Stage_refight: gameObject = <null>
  var CamLock Boss: gameObject = <null>
  var Gates: gameObject = <null>
}
