# Protect From Below
# 2 location(s):
#   Bone_15 / Boss Scene/Skull King
#   Bonetown_boss / Boss Scene/Skull King

fsm Protect From Below {
  start Can Attack

  state Can Attack {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetImmuneToNailAttacks(false))
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Can Nail Attack Range", storeResult=false, sendEvent="", outOfRangeEvent="SHIFT", everyFrame=true)
    on SHIFT → Can't Attack
  }

  state Can't Attack {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetImmuneToNailAttacks(true))
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Can Nail Attack Range", storeResult=false, sendEvent="SHIFT", outOfRangeEvent="", everyFrame=true)
    on SHIFT → Can Attack
  }
}
