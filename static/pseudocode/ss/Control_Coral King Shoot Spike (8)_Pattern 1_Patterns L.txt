# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Shoot Spikes/Patterns L/Pattern 1/Coral King Shoot Spike (8)

fsm Control {
  start Init

  state Init {
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Base", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shot", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on SHOOT → Delay
  }

  state Delay {
    FloatOperator(float1=var "Sequence", float2=0.05, operation=Multiply, storeResult=var "Delay Crt", everyFrame=false)
    Wait(time=var "Delay Crt", finishEvent=→"FINISHED", realTime=false)
    BoolTest(boolVariable=var "Boss Defeated", isTrue=→"BOSS DEFEATED", isFalse=(none), everyFrame=true)  // disabled
    on FINISHED → Antic 1 
    on BOSS DEFEATED → Dormant
  }

  state Antic 1  {
    PlayAudioEvent(audioClip=coral_knight_small_ceiling_coral_appear (sfxstatic_assets_areacoralareamemory.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shot Antic", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetMeshRenderer(gameObject=Self, active=true)
    SetPosition(gameObject=var "Shot", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    on FINISHED → Antic 2
  }

  state Antic 2 {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Delay 2
  }

  state Delay 2 {
    FloatOperator(float1=var "Sequence", float2=0.125, operation=Multiply, storeResult=var "Delay Crt", everyFrame=false)
    Wait(time=var "Delay Crt", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fire
  }

  state Fire {
    SetMeshRenderer(gameObject=Self, active=false)
    PlayAudioEvent(audioClip=coral_knight_small_ceiling_coral_shoot (sfxstatic_assets_areacoralareamemory.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Base", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Boss Defeated", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Shot", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=var "Shot", vector=(unset), x=(unset), y=-50, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End
  }

  state End {
    on FINISHED → Dormant
  }

  var Sequence: float = 1

  var Delay Crt: float = 0
  var Boss Defeated: bool = false
  var Base: gameObject = Boss Scene/Shoot Spikes/Patterns L/Pattern 1/Coral King Shoot Spike (8)/Base
  var Pt Antic: gameObject = Boss Scene/Shoot Spikes/Patterns L/Pattern 1/Coral King Shoot Spike (8)/Pt Antic
  var Shot: gameObject = Boss Scene/Shoot Spikes/Patterns L/Pattern 1/Coral King Shoot Spike (8)/Shot
  var Slam Effect: gameObject = Boss Scene/Shoot Spikes/Patterns L/Pattern 1/Coral King Shoot Spike (8)/Slam Effect
}
