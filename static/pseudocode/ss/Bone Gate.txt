# Bone Gate
# 1 location(s):
#   Bone_04 / Bone Gate

fsm Bone Gate {
  start Pause Frame

  state Idle {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=true)
    on OPEN → Open Audio
    on ACTIVATED → Deactivate
  }

  state Open Audio {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=switch_gate_gate (sfxstatic_assets_areaantareaboneareamemoryareasongareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    ObjectJitter(gameObject=Self, x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=30)
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    on FINISHED → Open Pause
  }

  state Deactivate {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on ACTIVATED → Deactivate
  }

  state Open Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Open")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Lift", emit=0, resetIfPlaying=false)
    on FINISHED → Deactivate
  }

  state Open Pause {
    Wait(time=0.55, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Open Anim
  }

  var Activated: bool = false
  var Self: gameObject = <null>
  var Pt Antic: gameObject = Bone Gate/Pt Antic
  var Pt Lift: gameObject = Bone Gate/Pt Lift
}
