# Change Range if plat down
# 1 location(s):
#   Bone_01 / Rock Roller Scene/Rock Roller

fsm Change Range if plat down {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Set {
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Range Max", setValue=99.37, everyFrame=false)
  }

  state Check {
    PlayerDataBoolTest(boolName="bone01shortcutPlat", isTrue=→"SET", isFalse=(none))
    on SET → Set
  }
}
