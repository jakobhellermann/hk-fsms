# Detect Hit
# 1 location(s):
#   Bone_East_08 / Boss Scene/Big Drop Bomb/explode_wall (1)/explode rock

fsm Detect Hit {
  start Idle

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Collider", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT → Check
  }

  state Check {
    CheckTrackTriggerCount(target=Boss Scene/Big Drop Bomb/explode_wall (1)/explode rock/Hero Detector, count=0, test=LessThanOrEqual, everyFrame=false, successEvent=→"FINISHED")
    GetPosition(gameObject=var "Collider", vector=(unset), x=var "Atk Pos", y=(unset), z=(unset), space=0, everyFrame=false)  // disabled
    FloatInRange(floatVariable=var "Atk Pos", lowerValue=var "Atk Min X", upperValue=var "Atk Max X", boolVariable=(unset), trueEvent=(none), falseEvent=→"FINISHED", everyFrame=false)  // disabled
    GetPosition(gameObject=var "Collider", vector=(unset), x=(unset), y=var "Atk Pos", z=(unset), space=0, everyFrame=false)  // disabled
    FloatInRange(floatVariable=var "Atk Pos", lowerValue=var "Atk Min Y", upperValue=var "Atk Max Y", boolVariable=(unset), trueEvent=→"VALID", falseEvent=→"FINISHED", everyFrame=false)  // disabled
    SendEvent(eventTarget=Self, sendEvent=→"VALID", delay=0, everyFrame=false)
    on FINISHED → Idle
    on VALID → Send
  }

  state Send {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="STRIKE", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  var Atk Max X: float = 9999
  var Atk Max Y: float = 9999
  var Atk Min X: float = 22
  var Atk Min Y: float = 0

  var Atk Pos: float = 0
  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
}
