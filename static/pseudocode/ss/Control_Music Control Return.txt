# Control
# 1 location(s):
#   Tut_01 / Music Control Return

fsm Control {
  start Pause

  state Pause {
    on FINISHED → Check State
  }

  state Check State {
    PlayerDataBoolTest(boolName="entered_Tut01b", isTrue=(none), isFalse=→"INACTIVE")
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="top1", equalEvent=→"TOP", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on INACTIVE → Inactive
    on TOP → Fade Out Bonetown
    on FINISHED → Moss Music Up
  }

  state Inactive {
  }

  state Moss Music Up {
    ApplyMusicCue(musicCue=MossCave (audiocuesdynamic_assets_areatutorial.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Sub Area (audioobjects_assets_all.bundle), transitionTime=6)
  }

  state Fade Out Bonetown {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=3)
    Wait(time=3, finishEvent=(none), realTime=false)
    on FINISHED → Moss Music Up
  }

  var Entry Gate: string = ""
}
