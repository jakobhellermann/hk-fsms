fsm Control {
  start Init
  on TINK L → Tink L  // from any state
  on TINK R → Tink R  // from any state

  state Init {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Orig Angle", space=0, everyFrame=false)  // disabled
    BoolTest(boolVariable=var "Flip", isTrue=→"FLIP", isFalse=(none), everyFrame=false)
    FloatAdd(floatVariable=var "Angle L", add=var "Orig Angle", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Angle R", add=var "Orig Angle", everyFrame=false, perSecond=false)
    on FLIP → Flip
  }

  state Tink R {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Angle", space=0, everyFrame=false)  // disabled
    SetFloatValue(floatVariable=var "Start Angle", floatValue=var "Angle R", everyFrame=false)
    AnimateRotationTo(gameObject=Self, fromValue=var "Orig Angle", toValue=var "Angle R", worldSpace=false, negativeSpace=false, time=0.1, speed=(unset), delay=(unset), easeType=1, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=metal_creak (sfxdynamic_assets_areasong.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Return
  }

  state Tink L {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Angle", space=0, everyFrame=false)  // disabled
    SetFloatValue(floatVariable=var "Start Angle", floatValue=var "Angle L", everyFrame=false)
    AnimateRotationTo(gameObject=Self, fromValue=var "Orig Angle", toValue=var "Angle L", worldSpace=false, negativeSpace=false, time=0.1, speed=(unset), delay=(unset), easeType=1, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=metal_creak (sfxdynamic_assets_areasong.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Return
  }

  state Return {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Angle", space=0, everyFrame=false)  // disabled
    AnimateRotationTo(gameObject=Self, fromValue=var "Start Angle", toValue=var "Orig Angle", worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=22, reverse=false, finishEvent=→"FINISHED", realTime=false)
  }

  state Flip {
    SetFloatValue(floatVariable=var "Temp", floatValue=var "Angle L", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle L", floatValue=var "Angle R", everyFrame=false)
    SetFloatValue(floatVariable=var "Angle R", floatValue=var "Temp", everyFrame=false)
    FloatAdd(floatVariable=var "Angle L", add=var "Orig Angle", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Angle R", add=var "Orig Angle", everyFrame=false, perSecond=false)
  }

  var Orig Angle: float = 90
  var Flip: bool = false

  var Start Angle: float = 0
  var Angle R: float = -3.5
  var Angle L: float = 3.5
  var Temp: float = 0
}
