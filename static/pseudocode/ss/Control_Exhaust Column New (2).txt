# Control
# 1 location(s):
#   Organ_01 / Exhaust Column New (2)

fsm Control {
  start Init
  on ALL EXHAUST OFF → State 1  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Audio Player")
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Damager", active=false, resetOnExit=false)
    SetCollider(gameObject=var "Air Blow Region", active=false, resetOnExit=false)
    BoolTest(boolVariable=var "z Blocker", isTrue=→"BLOCKER", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Is Up", isTrue=→"UP", isFalse=→"DOWN", everyFrame=false)
    on UP → Up Start
    on DOWN → Down Start
    on BLOCKER → Blocker
  }

  state Down On {
    PlayParticleEmitterChildren(gameObject=var "Pt Attack", resetTimeIfPlaying=false, stopOnStateExit=true)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Damager", active=true, resetOnExit=false)
    SetCollider(gameObject=var "Air Blow Region", active=true, resetOnExit=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Audio Player", audioClip=organ_note (sfxstatic_assets_areaorgan.bundle), pitchMin=0.75, pitchMax=1.25, volume=1, delay=0, storePlayer=var "Organ Note Player")
    SetAudioClip(gameObject=var "Audio Player", audioClip=organ_steam_attack 1 (sfxstatic_assets_areaorgan.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=var "Audio Player", volume=0.75, oneShotClip=<null>)
    on STEAM DOWN OFF → Down Off
  }

  state Down Start {
    on STEAM DOWN ANTIC → Down Antic
  }

  state Down Antic {
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    SetAudioClip(gameObject=var "Audio Player", audioClip=organ_steam_antic 1 (sfxstatic_assets_areaorgan.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=var "Audio Player", volume=1, oneShotClip=<null>)
    on STEAM DOWN ON → Down On
  }

  state Down Off {
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Damager", active=false, resetOnExit=false)
    SetCollider(gameObject=var "Air Blow Region", active=false, resetOnExit=false)
    PlayParticleEmitterChildren(gameObject=var "Pt After", resetTimeIfPlaying=false, stopOnStateExit=true)
    AudioStopV2(gameObject=var "Organ Note Player", fadeTime=1, cancelOnEarlyExit=false)
    on STEAM DOWN ANTIC → Down Antic
  }

  state Up On {
    PlayParticleEmitterChildren(gameObject=var "Pt Attack", resetTimeIfPlaying=false, stopOnStateExit=true)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Damager", active=true, resetOnExit=false)
    SetCollider(gameObject=var "Air Blow Region", active=true, resetOnExit=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Audio Player", audioClip=organ_note (sfxstatic_assets_areaorgan.bundle), pitchMin=0.75, pitchMax=1.25, volume=1, delay=0, storePlayer=var "Organ Note Player")
    SetAudioClip(gameObject=var "Audio Player", audioClip=organ_steam_attack 1 (sfxstatic_assets_areaorgan.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=var "Audio Player", volume=0.75, oneShotClip=<null>)
    on STEAM UP OFF → Up Off
  }

  state Up Start {
    on STEAM UP ANTIC → Up Antic
  }

  state Up Antic {
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    SetAudioClip(gameObject=var "Audio Player", audioClip=organ_steam_antic 1 (sfxstatic_assets_areaorgan.bundle), autoPlay=false, stopOnExit=false)
    AudioPlaySimple(gameObject=var "Audio Player", volume=1, oneShotClip=<null>)
    on STEAM UP ON → Up On
  }

  state Up Off {
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Damager", active=false, resetOnExit=false)
    SetCollider(gameObject=var "Air Blow Region", active=false, resetOnExit=false)
    PlayParticleEmitterChildren(gameObject=var "Pt After", resetTimeIfPlaying=false, stopOnStateExit=true)
    AudioStopV2(gameObject=var "Organ Note Player", fadeTime=1, cancelOnEarlyExit=false)
    on STEAM UP ANTIC → Up Antic
  }

  state Blocker {
    PlayParticleEmitterChildren(gameObject=var "Pt Attack", resetTimeIfPlaying=false, stopOnStateExit=true)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Damager", active=true, resetOnExit=false)
    on BLOCKER OFF → Blocker Off
  }

  state Blocker Off {
    PlayParticleEmitterChildren(gameObject=var "Pt After", resetTimeIfPlaying=false, stopOnStateExit=true)
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetCollider(gameObject=var "Damager", active=false, resetOnExit=false)
  }

  state State 1 {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Is Up: bool = false
  var z Blocker: bool = false

  var Pt Attack: gameObject = Exhaust Column New (2)/Pt Attack
  var Damager: gameObject = Exhaust Column New (2)/Damager
  var Pt After: gameObject = Exhaust Column New (2)/Pt After
  var Pt Antic: gameObject = Exhaust Column New (2)/Pt Antic
  var Audio Player: gameObject = <null>
  var Organ Note Player: gameObject = <null>
  var Air Blow Region: gameObject = Exhaust Column New (2)/Damager/Air Blow Region
}
