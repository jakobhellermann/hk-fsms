# Detect Grab
# 1 location(s):
#   Shadow_18 / Battle Scene/Wave 6 - Boss/Swamp Shaman/Suck Capture Collider

// uses template: detect_grab
fsm Detect Grab {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventV2(gameObject=Self, trigger=0, collideTag="", collideLayer=Hero Box(20), sendEvent=→"ENTER", storeCollider=var "")
    Trigger2dEventV2(gameObject=Self, trigger=0, collideTag="", collideLayer=Interactive Object(19), sendEvent=→"INTERACTIVE", storeCollider=var "Colliding Object")
    on ENTER → Check
    on INTERACTIVE → Check Interactive
  }

  state Check {
    CanHeroBeGrabbedV2(eventTarget=Self, canTakeDmgEvent=→"GRAB", cannotTakeDmgEvent=→"CANCEL", ignoreParryState=var "ignoreParryState", ignoreBellBind=false, triggerBellBindEffect=false)
    on CANCEL → Idle
    on GRAB → Send Grab Event
  }

  state Send Grab Event {
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="GRAB HERO", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Check Interactive {
    BoolTest(boolVariable=var "Check For Explosives", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    HasComponent(gameObject=var "Colliding Object", component="CustomTag", removeOnExit=false, trueEvent=(none), falseEvent=→"FINISHED", store=(unset), everyFrame=false)
    CallMethodProper(gameObject=var "Colliding Object", behaviour="CustomTag", methodName="GetCustomTagAsString", parameters=[], storeResult=var "Custom Tag", EveryFrame=false)
    StringSwitch(stringVariable=var "Custom Tag", compareTo=["Bomb", "DustBomb", "LightningBolaBall", "LightningBola", "SkinnyMosquito", "CogworkHatchling"], sendEvent=[→"BOMB", →"DUST BOMB", →"LIGHTNING BOLA BALL", →"LIGHTNING BOLA", →"MOSQUITO", →"HATCHLING"], everyFrame=false)
    on FINISHED → Idle
    on BOMB → Send Bomb
    on DUST BOMB → Send Dust Bomb
    on LIGHTNING BOLA BALL → Send Lightning
    on LIGHTNING BOLA → Send Lightning
  }

  state Send Bomb {
    SendEventByNameUpwards(Target=var "Colliding Object", EventName="SWALLOW")
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="GRAB BOMB", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Send Dust Bomb {
    SendEventByNameUpwards(Target=var "Colliding Object", EventName="SWALLOW")
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="GRAB DUST BOMB", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Send Lightning {
    SendEventByNameUpwards(Target=var "Colliding Object", EventName="SWALLOW")
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="GRAB LIGHTNING BOLA", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Recycle Object {
    SendEventByNameUpwards(Target=var "Colliding Object", EventName="SWALLOW")
    on FINISHED → Idle
  }

  var Check For Explosives: bool = true
  var ignoreParryState: bool = true
  var Event Receiver: gameObject = Battle Scene/Wave 6 - Boss/Swamp Shaman

  var Custom Tag: string = ""
  var Grab Collider: gameObject = <null>
  var Colliding Object: gameObject = <null>
}
