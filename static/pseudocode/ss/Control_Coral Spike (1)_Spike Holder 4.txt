# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Roar Spikes/Spike Holder 4/Coral Spike (1)

fsm Control {
  start Init

  state Init {
    BuildString(stringParts=[var "Anim", " Up"], separator="", addToEnd=true, storeResult=var "Anim Up", everyFrame=false)
    BuildString(stringParts=[var "Anim", " Break"], separator="", addToEnd=true, storeResult=var "Anim Break", everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on SPIKE UP → Antic
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Pt AnticDust")
    PlayParticleEmitterInState(gameObject=var "Pt AnticRock")
    WaitRandom(timeMin=1, timeMax=1.2, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="SPEAR ANTIC START")
    on FINISHED → Up
  }

  state Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Up", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    PlayRandomAudioClipTable(Table=Coral Spear Enter (sfxdynamic_assets_areacoralareamemory.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SendEventToRegister(eventName="SPEAR ANTIC END")
    on FINISHED → Active
  }

  state Active {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    Wait(time=0.5, finishEvent=→"NEXT", realTime=false)
    on NEXT → Break
  }

  state Break {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Break", resetTimeIfPlaying=false, stopOnStateExit=false)
    PlayRandomAudioClipTable(Table=Coral Spear Destroy (sfxdynamic_assets_areacoralareamemory.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Dormant
  }

  var Anim: string = "Spear1"

  var Anim Break: string = ""
  var Anim Up: string = ""
  var Pt Break: gameObject = Boss Scene/Roar Spikes/Spike Holder 4/Coral Spike (1)/Pt Break
  var Dust Barrier: object = Boss Scene/Roar Spikes/Spike Holder 4/Coral Spike (1)/Dust Barrier
  var Pt AnticDust: object = Boss Scene/Roar Spikes/Spike Holder 4/Coral Spike (1)/Pt AnticDust
  var Pt AnticRock: object = Boss Scene/Roar Spikes/Spike Holder 4/Coral Spike (1)/Pt AnticRock
  var Slam Effect: object = Boss Scene/Roar Spikes/Spike Holder 4/Coral Spike (1)/Slam Effect
}
