# Control
# 1 location(s):
#   Dust_12 / Dustroach Caged (4)

fsm Control {
  start Init
  on ZERO HP → Death  // from any state

  state Init {
    RandomInt(min=1, max=57, storeResult=var "Start Frame", inclusiveMax=false, noRepeat=false)
    Tk2dPlayFrame(gameObject=Self, frame=var "Start Frame")
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SetParent(gameObject=var "Pos L", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Pos R", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    RandomlyFlipScale(gameObject=Self)
    on FINISHED → Sleeping
  }

  state Sleeping {
    AudioPlaySimple(gameObject=var "Audio Loop Sleep", volume=1, oneShotClip=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"WAKE", ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    FloatCompare(float1=var "Z Pos", float2=3, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"BG", everyFrame=false)
    on WAKE → Wake Pause
    on RAISE ALARM → Wake Pause
    on SHAKE → Wake Pause
    on BG → Sleeping BG
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Wake", emit=0, resetIfPlaying=false)
    PlayRandomAudioClipTable(Table=Dust Roach Bark (sfxdynamic_assets_areadust.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    AudioStop(gameObject=Self, fadeTime=0)
    AudioStop(gameObject=var "Audio Loop Sleep", fadeTime=0)
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    WaitRandom(timeMin=0.4, timeMax=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bark
  }

  state Bark {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bark")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToBark", resetFrame=true, everyFrame=true, pauseBetweenTurns=2)
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=10)
    SendEventToRegisterDelay(EventName="RAISE ALARM", delay=0.5, ExcludeTarget=Self)  // disabled
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=Dust Roach Bark (sfxdynamic_assets_areadust.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Barking
    on SING → Sing
  }

  state Wake Pause {
    WaitRandom(timeMin=0.1, timeMax=0.3, finishEvent=(none), realTime=false)
    on FINISHED → Wake
  }

  state Sleeping BG {
    SetBoolValue(boolVariable=var "BG", boolValue=true, everyFrame=false)
    on RAISE ALARM → Wake Pause
  }

  state Barking {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"SHAKE", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bark")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToBark", resetFrame=true, everyFrame=true, pauseBetweenTurns=2)
    PlayRandomAudioClipTable(Table=Dust Roach Roar (sfxdynamic_assets_areadust.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=var "Noise Radius")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SHAKE → Move Choice
    on SING → Sing
  }

  state Shake Antic {
    RandomlyFlipScale(gameObject=Self)  // disabled
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Screen Shake?
  }

  state Slam {
    AnimatePositionTo(gameObject=Self, toValue=var "Start Pos", localSpace=false, time=0.2, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Barking
  }

  state Slam Translate {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "Tinker"), sendEvent="RATTLE", delay=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "X Scale", multiplyBy=-1, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Slam
  }

  state Screen Shake? {
    BoolTest(boolVariable=var "BG", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Slam Translate
  }

  state Move Choice {
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Move To L
    on R → Move To R
  }

  state Move To L {
    AccelerateToX(gameObject=Self, accelerationFactor=1, targetSpeed=-10)
    CheckTargetDirection(gameObject=Self, target=var "Pos L", aboveEvent=(none), belowEvent=(none), rightEvent=→"FINISHED", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToRun", everyFrame=true, pauseBetweenTurns=true, pauseTime=0)
    on FINISHED → Stick To L
  }

  state Stick To L {
    SetPositionToObject(gameObject=Self, targetObject=var "Pos L", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    on FINISHED → Shake Antic
  }

  state Move To R {
    AccelerateToX(gameObject=Self, accelerationFactor=1, targetSpeed=10)
    CheckTargetDirection(gameObject=Self, target=var "Pos R", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"FINISHED", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToRun", everyFrame=true, pauseBetweenTurns=true, pauseTime=0)
    on FINISHED → Stick To R
  }

  state Stick To R {
    SetPositionToObject(gameObject=Self, targetObject=var "Pos R", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    on FINISHED → Shake Antic
  }

  state Death {
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Caged")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bark")
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=dust_roach_sing_loop (sfxdynamic_assets_areadust.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Move Choice
    on SING DURATION END → Move Choice
  }

  var Tinker: gameObject = cage_mid_wide_sway (1)/Plat Collider

  var Z Pos: float = 0
  var X Scale: float = 0
  var Noise Radius: float = 10
  var Start Frame: int = 0
  var BG: bool = false
  var Start Pos: vector3 = (0, 0, 0)
  var Pt Wake: gameObject = Dustroach Caged (4)/Pt Wake
  var Pos L: gameObject = Dustroach Caged (4)/Pos L
  var Pos R: gameObject = Dustroach Caged (4)/Pos R
  var Audio Loop Sleep: gameObject = Dustroach Caged (4)/Audio Loop Sleep
  var Audio Loop Voice: gameObject = Dustroach Caged (4)/Audio Loop Voice
}
