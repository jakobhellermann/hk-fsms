# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Death Spikes 1/Coral Death Spike (1)

fsm Control {
  start Init

  state Init {
    BuildString(stringParts=[var "Anim", " Up"], separator="", addToEnd=true, storeResult=var "Anim Up", everyFrame=false)
    BuildString(stringParts=[var "Anim", " Break"], separator="", addToEnd=true, storeResult=var "Anim Break", everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on SPIKE UP → Appear Pause
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Pt AnticDust")
    PlayParticleEmitterInState(gameObject=var "Pt AnticRock")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
  }

  state Up {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim Up", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Active
  }

  state Active {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
  }

  state Appear Pause {
    WaitRandom(timeMin=0, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  var Anim: string = "Spear3"

  var Anim Break: string = ""
  var Anim Up: string = ""
  var Pt Break: gameObject = <null>
  var Dust Barrier: object = <null>
  var Pt AnticDust: object = Boss Scene/Death Spikes 1/Coral Death Spike (1)/Pt AnticDust
  var Pt AnticRock: object = Boss Scene/Death Spikes 1/Coral Death Spike (1)/Pt AnticRock
  var Slam Effect: object = Boss Scene/Death Spikes 1/Coral Death Spike (1)/Slam Effect
}
