# Prompt Control
# 1 location(s):
#   Cog_09 / puzzle cylinders

fsm Prompt Control {
  start Init
  on SHOW PROMPT → Get Melody  // from any state
  on ENTER NEEDOLIN → Needolin Play Prompt  // from any state

  state Idle {
    ActivateGameObject(gameObject=var "Needolin Override", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  sequence state Get Melody {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SetDeathRespawnMarker(Target=puzzle cylinders/Death Respawn Marker)
    RunFSM(fsmTemplateControl=template=708463260513842076 in[Set Just Gave Item<-var "Set Just Gave Item", UI Msg Event<-var "UI Msg Event", Give Item<-var "Give Item"], finishEvent=(none), everyFrame=false)
    SendEventByNameV2(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    SetPlayerDataBool(boolName="disableInventory", value=false)
    on FINISHED → Idle
  }

  sequence state Needolin Play Prompt {
    ActivateGameObject(gameObject=var "Needolin Override", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RunFSM(fsmTemplateControl=template=-6849325147229470167 in[Skip Begin<-var "Skip Begin", Skip End<-var "Skip End", Timeout Override<-false], finishEvent=(none), everyFrame=false)
    SendEventToRegister(eventName="NEEDOLIN TIME END")
    on FINISHED → Idle
  }

  state Init {
    ActivateGameObject(gameObject=var "Needolin Override", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Needolin Override: gameObject = puzzle cylinders/Choir Needolin Override
}
