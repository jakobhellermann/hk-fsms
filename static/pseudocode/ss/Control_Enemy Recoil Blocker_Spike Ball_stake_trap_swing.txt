fsm Control {
  start Solid

  state Solid {
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "Collider", storeForce=(unset))
    SetCollider2dIsTrigger(gameObject=Self, isTrigger=false, setAllColliders=false)
    on COLLIDE → Recoil Check
  }

  state Recoil Check {
    GetIsRecoiling(target=var "Collider", storeIsRecoiling=var "Is Recoiling", everyframe=false)
    BoolTest(boolVariable=var "Is Recoiling", isTrue=→"RECOIL", isFalse=(none), everyFrame=false)
    on FINISHED → Solid
    on RECOIL → Trigger Start
  }

  state Trigger {
    SetCollider2dIsTrigger(gameObject=Self, isTrigger=true, setAllColliders=false)  // disabled
    CheckTrackTriggerCount(target=Self, count=0, test=MoreThan, everyFrame=true, successEvent=→"SOLID")
    on SOLID → Solid
  }

  state Trigger Start {
    SetCollider2dIsTrigger(gameObject=Self, isTrigger=true, setAllColliders=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Trigger
  }

  var Enemies Inside: int = 0
  var Is Recoiling: bool = false
  var Collider: gameObject = <null>
}
