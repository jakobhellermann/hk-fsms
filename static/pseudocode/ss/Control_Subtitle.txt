# Control
# 1 location(s):
#   Menu_Title / LogoTitle/Subtitle

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Fade In
  }

  state Fade In {
    DestroyAllChildren(gameObject=var "Self", disable=false)  // disabled
    GetLanguageString(sheetName="General", convName="GAME_DISPLAY_TITLE", storeValue=var "Text")
    SetTextMeshProText(gameObject=Self, textString=var "Text")
    SetColorValue(colorVariable=var "Colour", color=(1, 1, 1, 0), everyFrame=false)
    EaseColor(fromValue=(1, 1, 1, 0), toValue=(1, 1, 1, 1), colorVariable=var "Colour", time=0.25, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetTextMeshProColor(gameObject=Self, color=var "Colour", everyFrame=true)
    CallMethodProper(gameObject=Self, behaviour="ChangeFontByLanguage", methodName="SetFont", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FADE OUT → Fade Out
  }

  state Fade Out {
    SetColorValue(colorVariable=var "Colour", color=(1, 1, 1, 1), everyFrame=false)
    EaseColor(fromValue=(1, 1, 1, 1), toValue=(1, 1, 1, 0), colorVariable=var "Colour", time=0.25, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetTextMeshProColor(gameObject=Self, color=var "Colour", everyFrame=true)
    on FADE IN → Fade In
  }

  var Text: string = ""
  var Colour: color = (0, 0, 0, 1)
  var Self: gameObject = <null>
}
