# Control
# 1 location(s):
#   Bone_East_08 / Boss Scene

fsm Control {
  start Pause
  on BATTLE END → End Pause  // from any state

  sequence state Pause {
    WaitForBossLoad(sendEvent=(none))
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=true)
    NextFrameEvent(sendEvent=(none))
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  sequence state Init {
    FindChild(gameObject=Self, childName="CamLock Boss", storeResult=var "CamLock Boss")
    FindChild(gameObject=Self, childName="CamLock Normal", storeResult=var "CamLock Normal")
    FindChild(gameObject=Self, childName="Pre Activation Floor", storeResult=var "Pre Activation Floor")
    FindChild(gameObject=Self, childName="Wall L", storeResult=var "Wall L")
    FindChild(gameObject=Self, childName="Wall R", storeResult=var "Wall R")
    FindChild(gameObject=Self, childName="Golem_rest", storeResult=var "Golem_rest")
    FindChild(gameObject=Self, childName="Start Range", storeResult=var "Start Range")
    FindChild(gameObject=Self, childName="Remeet Lava", storeResult=var "Remeet Lava")
    FindChild(gameObject=Self, childName="RemeetAntic Lava", storeResult=var "RemeetAntic Lava")
    FindChild(gameObject=Self, childName="Battle End Floor", storeResult=var "Battle End Floor")
    FindNamedChild(gameObject=Self, storeResult=var "Big Drop Bomb")
    FindNamedChild(gameObject=Self, storeResult=var "Refight Rumble Loop")
    FindNamedChild(gameObject=Self, storeResult=var "Big Drop Bomb Return")
    ActivateGameObject(gameObject=var "Big Drop Bomb Return", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Lava Plats")
    ActivateGameObject(gameObject=var "Lava Plats", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="defeatedSongGolem", isTrue=→"DEFEATED", isFalse=(none))
    GetGuidReferenceHolderGameObject(Target=var "Golem_rest", StoreGameObject=var "Golem_rest")
    PlayerDataBoolTrueAndFalse(trueBool="hasBrolly", falseBool="encounteredSongGolem", isTrue=→"MEET READY", isFalse=(none))
    PlayerDataBoolTrueAndFalse(trueBool="encounteredSongGolem", falseBool="defeatedSongGolem", isTrue=→"REMEET READY", isFalse=(none))
    on MEET READY → Meet Ready
    on REMEET READY → Remeet Ready
    on DEFEATED → Beastfly?
  }

  state Meet Ready {
    Trigger2dEvent(gameObject=var "Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=5)
    SendEventByName(eventTarget=GameObject(var "Battle End Floor"), sendEvent="AWAY", delay=0, everyFrame=false)
    on ENTER → Wake
  }

  state Wake {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CAM START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Pre Activation Floor"), sendEvent="AWAY", delay=0, everyFrame=false)
    SetPlayerDataBool(boolName="encounteredSongGolem", value=true)
    ActivateGameObject(gameObject=var "CamLock Normal", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Start Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Golem_rest"), sendEvent="WAKE", delay=0, everyFrame=false)
    SendEventToRegister(eventName="GOLEM FIGHT START")
  }

  state Remeet Ready {
    Trigger2dEvent(gameObject=var "Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    ActivateGameObject(gameObject=var "Golem_rest", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=5)
    SendEventByName(eventTarget=GameObject(var "Battle End Floor"), sendEvent="AWAY", delay=0, everyFrame=false)
    on ENTER → Remeet 1
  }

  state Remeet 1 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CAM START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Pre Activation Floor"), sendEvent="AWAY", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Start Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Normal", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=chorus_under_rocks_shake_initiate (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlaySimple(gameObject=var "Refight Rumble Loop", volume=1, oneShotClip=<null>)
    SendEventToRegister(eventName="GOLEM FIGHT START")
    on FINISHED → Remeet 2
  }

  state Remeet 2 {
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=true, Delay=0)
    PlayParticleEmitterInState(gameObject=var "RemeetAntic Lava")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Remeet 3
  }

  state Remeet 3 {
    PlayAudioEvent(audioClip=chorus_under_rocks_break_free (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioStop(gameObject=var "Refight Rumble Loop", fadeTime=0)
    SendEventToRegister(eventName="ACTIVATE GOLEM")
    PlayParticleEmitterChildren(gameObject=var "Remeet Lava", resetTimeIfPlaying=false, stopOnStateExit=false)
  }

  state End Pause {
    AwardQueuedAchievements(delay=0)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=GameObject(var "Battle End Floor"), sendEvent="RETURN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="CAM END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Pre Activation Floor"), sendEvent="RETURN", delay=0, everyFrame=false)
    SaveGame()
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)  // disabled
  }

  state Defeated {
    ActivateGameObject(gameObject=var "Big Drop Bomb", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="destroyedSongGolemRock", isTrue=(none), isFalse=→"RETURN BOMB")
    on RETURN BOMB → Activate Return Bomb Rock
  }

  state Activate Return Bomb Rock {
    ActivateGameObject(gameObject=var "Big Drop Bomb Return", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Beastfly? {
    CheckQuestStateV2(NotTrackedEvent=→"FINISHED", IncompleteEvent=(none), CompletedEvent=→"FINISHED", Quest=Beastfly Hunt (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    CanEndQuest(CannotEndEvent=(none), CanEndEvent=→"FINISHED", StoreValue=(unset), Quest=Beastfly Hunt (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    ActivateGameObject(gameObject=var "Lava Plats", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Defeated
  }

  var Battle End Floor: gameObject = <null>
  var CamLock Boss: gameObject = <null>
  var CamLock Normal: gameObject = <null>
  var Golem_rest: gameObject = <null>
  var Pre Activation Floor: gameObject = <null>
  var Remeet Lava: gameObject = <null>
  var RemeetAntic Lava: gameObject = <null>
  var Silk Heart: gameObject = <null>
  var Start Range: gameObject = <null>
  var Wall L: gameObject = <null>
  var Wall R: gameObject = <null>
  var Big Drop Bomb: gameObject = <null>
  var Refight Rumble Loop: gameObject = <null>
  var Big Drop Bomb Return: gameObject = <null>
  var Lava Plats: gameObject = <null>
}
