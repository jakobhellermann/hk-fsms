  start Play
  on END → State 2  // from any state

  state Play {
    PlayParticleEmitterInState(gameObject=Self)
    RayCast2dV2(fromGameObject=Self, fromPosition=(0, 0.1), direction=(0, -1), space=0, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "Grounded", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "Grounded", isTrue=(none), isFalse=→"FINISHED", everyFrame=true)
    on FINISHED → No Ground
  }

  state State 2 {
    StopParticleEmitter(gameObject=Self)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 3
  }

  state State 3 {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state No Ground {
    RayCast2dV2(fromGameObject=Self, fromPosition=(0, 0.1), direction=(0, -1), space=0, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=→"FINISHED", noHitEvent=(none), storeDidHit=var "Grounded", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Play
  }

  var Grounded: bool = false
}
