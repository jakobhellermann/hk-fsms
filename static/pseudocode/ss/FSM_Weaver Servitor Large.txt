# FSM
# 1 location(s):
#   Peak_02 / Weaver Servitor Large

// uses template: weaver_drop_death
fsm FSM {
  start Idle

  state Idle {
    CheckYPosition(gameObject=Self, compareTo=var "Death Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"READY", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on READY → Check Collision
  }

  state Check Collision {
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=→"DIE", storeCollider=var "", storeForce=(unset))
    on DIE → Die
  }

  state Die {
    InstaDeath(target=Self, direction=0)
  }

  var Death Y: float = 164.75
}
