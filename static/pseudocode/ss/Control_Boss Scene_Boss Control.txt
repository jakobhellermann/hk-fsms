# Control
# 1 location(s):
#   Ant_19 / Boss Control/Boss Scene

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Bone Flyer Giant")
    FindNamedChild(gameObject=Self, storeResult=var "Top Blocker")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Boss")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Intro")
    FindNamedChild(gameObject=Self, storeResult=var "Gates")
    FindNamedChild(gameObject=Self, storeResult=var "Collectable Item Pickup")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Start Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Early Start Range", storeResult=false, sendEvent="EARLY ENTER", outOfRangeEvent="", everyFrame=true)
    PlayerDataBoolTest(boolName="defeatedBoneFlyerGiant", isTrue=→"BOSS DEFEATED", isFalse=(none))
    SetInvincible(target=var "Bone Flyer Giant", Invincible=true, InvincibleFromDirection=0, resetOnStateExit=false)
    on ENTER → Start
    on BOSS DEFEATED → Defeated
    on EARLY ENTER → Early Start Range
  }

  state Start {
    ActivateGameObject(gameObject=var "Top Blocker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Bone Flyer Giant", fsmName="Control", variableName="Hero Entered", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SetInvincible(target=var "Bone Flyer Giant", Invincible=false, InvincibleFromDirection=0, resetOnStateExit=false)
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Hero X", minValue=0, maxValue=65.85, everyFrame=false)
    SetPosition2D(GameObject=var "Hero", Vector=(unset), X=var "Hero X", Y=(unset), Space=0, EveryFrame=false)
    on BOSS DEFEATED → Defeated Pause
  }

  state Defeated Pause {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Boss", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Top Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="CHAPEL MUSIC START")
  }

  state Defeated {
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG QUICK OPEN", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Boss", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Top Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bone Flyer Giant", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Early Start Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Start Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Early Start Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    SetInvincible(target=var "Bone Flyer Giant", Invincible=false, InvincibleFromDirection=0, resetOnStateExit=false)
    on ENTER → Start
    on EXIT → Idle
    on BOSS DAMAGED → Start
  }

  var Hero X: float = 0
  var Bone Flyer Giant: gameObject = <null>
  var Top Blocker: gameObject = <null>
  var CamLock Boss: gameObject = <null>
  var Gates: gameObject = <null>
  var Collectable Item Pickup: gameObject = <null>
  var CamLock Intro: gameObject = <null>
}
