# Control
# 2 location(s):
#   Dust_01 / Dustroach FG
#   Dust_03 / Dustroach FG

fsm Control {
  start Init

  state Init {
    RandomInt(min=1, max=57, storeResult=var "Start Frame", inclusiveMax=false, noRepeat=false)
    Tk2dPlayFrame(gameObject=Self, frame=var "Start Frame")
    SetPosition(gameObject=Dustroach FG/Listener, vector=(unset), x=(unset), y=(unset), z=0, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Idle {
    CheckHeroPerformanceRegion(Target=Dustroach FG/Listener, MinReactDelay=0.2, MaxReactDelay=0.4, None=(none), ActiveInner=→"WAKE", ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on WAKE → Wake
  }

  state Wake {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wake", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Wake", emit=0, resetIfPlaying=false)
    PlayRandomAudioClipTable(Table=Dust Roach Bark (sfxdynamic_assets_areadust.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CreateNoise(From=Dustroach FG/Listener, LocalOrigin=(0, 0), Radius=8)
    on FINISHED → Idle 2
  }

  state Idle 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    WaitRandom(timeMin=0.3, timeMax=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bark
  }

  state Bark {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bark")
    WaitRandom(timeMin=1, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=Dust Roach Bark (sfxdynamic_assets_areadust.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Antic
  }

  state Type {
    BoolTest(boolVariable=var "Jump Away", isTrue=→"JUMP", isFalse=→"RUN", everyFrame=false)
    on JUMP → 
    on RUN → Run
  }

  state Run {
    SetVelocityByScale(gameObject=Self, speed=-30, ySpeed=-2.5, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Run")
    Wait(time=8, finishEvent=→"FINISHED", realTime=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Alarm Pause {
    WaitRandom(timeMin=0.1, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake
  }

  state Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Type
  }

  var Jump Away: bool = false

  var Start Frame: int = 0
  var Pt Wake: gameObject = Dustroach Caged/Pt Wake (localpoolprefabs_assets_areadust.bundle)
}
