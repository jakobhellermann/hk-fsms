# Set Colliders
# 5 location(s):
#   Aqueduct_01 / Swamp Barnacle (1)/Tendril
#   Aqueduct_02 / Swamp Barnacle (1)/Tendril
#   Aqueduct_03 / Swamp Barnacle (1)/Tendril
#   Aqueduct_07 / Swamp Barnacle (1)/Tendril
#   Aqueduct_08 / Swamp Barnacle (1)/Tendril

fsm Set Colliders {
  start State 1

  state State 1 {
    on FINISHED → State 2
  }

  state State 2 {
    SetParent(gameObject=var "Tendril Collider Bottom", parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    GetPosition2d(gameObject=var "Tendril Collider Bottom", vector_2d=(unset), x=(unset), y=var "Collider Bot Y", space=1, everyFrame=false)
    FloatOperator(float1=var "Collider Bot Y", float2=-1, operation=Multiply, storeResult=var "Collider Height", everyFrame=false)
    FloatOperator(float1=var "Collider Bot Y", float2=0.5, operation=Multiply, storeResult=var "Mid Point Y", everyFrame=false)
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=(unset), y=var "Self Y", space=1, everyFrame=false)
    FloatOperator(float1=var "Mid Point Y", float2=var "Self Y", operation=Subtract, storeResult=var "Collider Offset", everyFrame=false)
    SetBoxCollider2DSize(gameObject=Self, width=0.3124999, height=var "Collider Height", offsetX=0, offsetY=var "Collider Offset")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Attack Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → State 3
  }

  state State 3 {
  }

  var Collider Bot Y: float = 0
  var Collider Height: float = 0
  var Collider Offset: float = 0
  var Mid Point Y: float = 0
  var Self Y: float = 0
  var Attack Detector: gameObject = Swamp Barnacle (1)/Tendril/Attack Detector
  var Parent: gameObject = Swamp Barnacle (1)
  var Tendril Collider Bottom: gameObject = Swamp Barnacle (1)/Tendril/Tendril Collider Bottom
}
