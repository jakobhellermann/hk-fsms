// component disabled: this FSM does not run until something enables it
fsm Friend Bonk {
  start Idle

  state Idle {
    on FRIEND BONK → Check Dir
  }

  state Check Dir {
    GetEventSender(sentByGameObject=var "Event Sender")
    CheckTargetDirection(gameObject=Self, target=var "Event Sender", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on R → Recoil R
    on L → Recoil L
  }

  state Recoil R {
    BeginRecoil(target=Self, attackDirection=180, attackType=1, attackMagnitude=1)
    on FINISHED → Idle
  }

  state Recoil L {
    BeginRecoil(target=Self, attackDirection=180, attackType=1, attackMagnitude=1)
    on FINISHED → Idle
  }

  var Event Sender: gameObject = <null>
}
