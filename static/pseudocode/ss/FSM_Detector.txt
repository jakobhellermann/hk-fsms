# FSM
# 1 location(s):
#   Greymoor_20c / Battle Scene/Detector

fsm FSM {
  start Next Frame Event

  state Next Frame Event {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Check Finished
  }

  state Check Finished {
    GetPersistentBool(target=Battle Scene, storeValue=var "Battle Finished")
    BoolTest(boolVariable=var "Battle Finished", isTrue=→"TRUE", isFalse=(none), everyFrame=false)
    on TRUE → Disable Self
    on FINISHED → Wait for battle start
  }

  state Disable Self {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait for battle start {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="Player", sendEvent=→"TRUE", storeCollider=var "")
    on TRUE → Wait Trapdoor closed
  }

  state Wait Trapdoor closed {
    WaitTrapdoorState(Target=Battle Scene/greymoor_flip_bridge (1), everyFrame=true, openedEvent=(none), closedEvent=→"TRUE")
    on TRUE → Check Player Inside
  }

  state Check Player Inside {
    WaitTrapdoorState(Target=Battle Scene/greymoor_flip_bridge (1), everyFrame=true, openedEvent=→"FINISHED", closedEvent=(none))
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="Player", sendEvent=→"TRUE", storeCollider=var "")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on TRUE → Start Battle
    on FINISHED → Wait for battle start
  }

  state Start Battle {
    CallMethodProper(gameObject=Battle Scene, behaviour="BattleScene", methodName="StartBattle", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Disable Self
  }

  var Battle Finished: bool = false
}
