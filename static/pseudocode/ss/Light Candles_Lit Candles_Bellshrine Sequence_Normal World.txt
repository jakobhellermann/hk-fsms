# Light Candles
# 1 location(s):
#   Bellshrine_02 / Black Thread States/Normal World/Bellshrine Sequence/Lit Candles

fsm Light Candles {
  start Init

  state Idle {
    on LIGHT CANDLE → Light Candle
    on LIGHT ALL CANDLES → Light All
  }

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    GetChildCount(gameObject=Self, storeResult=var "Child Count")
    SetIntValue(intVariable=var "Current Child Index", intValue=0, everyFrame=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Light
    on LIGHT ALL CANDLES → Wait Frame
  }

  state Light All {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetSceneAmbientLight(Intensity=var "End Ambient Light")
  }

  state Light Candle {
    IntCompare(integer1=var "Current Child Index", integer2=var "Child Count", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    GetChildNum(gameObject=Self, childIndex=var "Current Child Index", store=var "Current Child")
    ActivateGameObject(gameObject=var "Current Child", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorGroupPlay(StateName="Appear", Target=var "Current Child")
    SetIntValue(intVariable=var "Child Count Floor", intValue=var "Child Count", everyFrame=false)
    IntAdd(intVariable=var "Child Count Floor", add=-1, everyFrame=false)
    ConvertIntToFloat(intVariable=var "Current Child Index", floatVariable=var "Child Percent", everyFrame=false)
    FloatDivide(floatVariable=var "Child Percent", divideBy=var "Child Count Floor", everyFrame=false)
    FloatLerp(StartValue=var "Start Ambient Light", EndValue=var "End Ambient Light", Time=var "Child Percent", StoreValue=var "Ambient Light")
    SetSceneAmbientLight(Intensity=var "Ambient Light")
    IntAdd(intVariable=var "Current Child Index", add=1, everyFrame=false)
    on FINISHED → Idle
  }

  state Set Light {
    SetSceneAmbientLight(Intensity=var "Start Ambient Light")
    on FINISHED → Idle
  }

  state Wait Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Light All
  }

  var Ambient Light: float = 0
  var Child Percent: float = 0
  var End Ambient Light: float = 0.8
  var Start Ambient Light: float = 0.2
  var Child Count: int = 0
  var Child Count Floor: int = 0
  var Current Child Index: int = 0
  var Current Child: gameObject = Black Thread States/Normal World/Bellshrine Sequence/Lit Candles/Candle Group 1
  var Self: gameObject = <null>
}
