# Take Damage
# 1 location(s):
#   Belltown_08 / Boss Scene/Wisp Pyre Effigy

fsm Take Damage {
  start Init

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"DAMAGED", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=var "Damage Dealt", storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetCircleCollider(gameObject=var "Collider", active=true)
    FloatAdd(floatVariable=var "Hit Cooldown Timer", add=-1, everyFrame=true, perSecond=true)
    FloatClamp(floatVariable=var "Hit Cooldown Timer", minValue=-5, maxValue=5, everyFrame=true)
    on DAMAGED → Increment Hit Total?
  }

  state Hit {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    PlayRandomAudioClipTable(Table=wisp_lantern_hit (sfxdynamic_assets_areabell.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Core", SpawnPosition=(0, 0, 0))
    PlayAudioEvent(audioClip=wisp_fireball_explode (sfxstatic_assets_shared.bundle), pitchMin=1.5, pitchMax=1.7, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Core", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Haze Hit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lantern Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Hit", emit=0, resetIfPlaying=false)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Core", position=(unset), rotation=(unset), storeObject=var "")
    IntOperator(integer1=var "HP", integer2=var "Damage Dealt", operation=Subtract, storeResult=var "HP", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Total Times Hit", integer2=29, equal=(none), lessThan=(none), greaterThan=→"BREAK", everyFrame=false)
    on FINISHED → Wobble Dir
    on BREAK → Break
  }

  state Wobble Dir {
    CheckTargetDirection(gameObject=Self, target=var "Damager", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Wobble L
    on R → Wobble R
  }

  state Wobble L {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WOBBLE L", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Wobble R {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WOBBLE R", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  state Break {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FINAL BREAK", delay=0, everyFrame=false)
  }

  state Recover {
    SetCircleCollider(gameObject=var "Collider", active=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Init {
    on CORE DAMAGE READY → Idle
  }

  state Increment Hit Total? {
    FloatCompare(float1=var "Hit Cooldown Timer", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    IntAdd(intVariable=var "Total Times Hit", add=1, everyFrame=false)
    SetFloatValue(floatVariable=var "Hit Cooldown Timer", floatValue=0.5, everyFrame=false)
    on FINISHED → Hit
  }

  var Hit Cooldown Timer: float = 0
  var Damage Dealt: int = 0
  var HP: int = 250
  var Total Times Hit: int = 0
  var Core: gameObject = Boss Scene/Wisp Pyre Effigy/Core
  var Damager: gameObject = <null>
  var Haze Hit: gameObject = Boss Scene/Wisp Pyre Effigy/Core/Haze Hit
  var Lantern Flash: gameObject = Boss Scene/Wisp Pyre Effigy/Core/Lantern Flash
  var Pt Hit: gameObject = Boss Scene/Wisp Pyre Effigy/Core/Pt Hit
  var Collider: gameObject = Boss Scene/Wisp Pyre Effigy/Core/Collider
}
