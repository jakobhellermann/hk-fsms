fsm Screen Shake {
  start Init
  on STOP SHAKE → Idle  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on START SHAKE → Shake
  }

  state Shake {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=var "Shake Disatance", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Shake
  }

  var Shake Disatance: float = 7

  var Distance: float = 0
}
