fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Lace Meet")
    ActivateGameObject(gameObject=var "Lace Meet", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Check
  }

  state Check {
    GetPlayerDataBool(boolName="encounteredLace1", storeValue=var "encounteredLace1")
    GetPlayerDataBool(boolName="encounteredLaceBlastedBridge", storeValue=var "encounteredLaceBlastedBridge")
    GetPlayerDataBool(boolName="laceMeetCitadel", storeValue=var "laceMeetCitadel")
    BoolNoneTrue(boolVariables=[var "encounteredLace1", var "encounteredLaceBlastedBridge", var "laceMeetCitadel"], sendEvent=→"LACE", storeResult=(unset), everyFrame=false)
    on LACE → Lace
    on FINISHED → None
  }

  state Lace {
    ActivateGameObject(gameObject=var "Lace Meet", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="LACE PRESENT")
  }

  state None {
  }

  var defeatedLastJudge: bool = false
  var encounteredLace1: bool = false
  var encounteredLaceBlastedBridge: bool = false
  var laceMeetCitadel: bool = false
  var defeatedCoralDrillerSolo: bool = false
  var blackThreadWorld: bool = false
  var Lace Meet: gameObject = <null>
  var Repairing Workers: gameObject = <null>
}
