// component disabled: this FSM does not run until something enables it
fsm Pull Hero {
  start Idle
  on PULL HERO END → Idle  // from any state

  state Idle {
    on PULL HERO → Check
  }

  state Check {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=4.25, tolerance=0.75, equal=→"END", lessThan=→"PUSH", greaterThan=→"PULL", everyFrame=false)
    on PULL → Pull
    on PUSH → Push
    on END → End
  }

  state End {
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Idle
  }

  state Pull {
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    ClampAngle(angleVariable=var "Angle", minValue=0, maxValue=180, everyFrame=false)  // disabled
    FloatClamp(floatVariable=var "Angle", minValue=0, maxValue=180, everyFrame=false)  // disabled
    SetVelocityAsAngle(gameObject=var "Hero", angle=var "Angle", speed=-15, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Push {
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    ClampAngle(angleVariable=var "Angle", minValue=0, maxValue=180, everyFrame=false)  // disabled
    FloatClamp(floatVariable=var "Angle", minValue=0, maxValue=180, everyFrame=false)  // disabled
    SetVelocityAsAngle(gameObject=var "Hero", angle=var "Angle", speed=15, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  var Distance: float = 0
  var Angle: float = 0
}
