# Silk Spool UI
# 1 location(s):
#   silkspoolupgrade_assets_all.bundle / Silk Spool UI

fsm Silk Spool UI {
  start Init
  on DISABLE → State 1  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetHero(storeResult=var "Hero")
    SetAnimator(target=Self, active=false)
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Increase Silk
  }

  state Destroy Self {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(true))
    SetPlayerDataBool(boolName="isInvincible", value=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    SendEventToRegister(eventName="SILK SPOOL UI END")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Hero Anim End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Heart Piece End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Return Control
  }

  state Return Control {
    AwardQueuedAchievements(delay=0.5)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=ResetAnimationDownspikeBounce())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=SetStartWithDownSpikeEnd())
    PlayAudioEvent(audioClip=hornet_sprint_jump_2nd_spin (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=0.8, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControlToIdle", parameters=[], storeResult=(unset var), EveryFrame=false)
    BoolTest(boolVariable=var "In Interaction", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    CheckIsInQuestBoardDepositSequence(trueEvent=→"BOARD", falseEvent=(none), storeValue=(unset))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    on BOARD → Continue Deposit
    on FINISHED → Save Queued
  }

  state Increase Silk {
    AddHeroInputBlocker(Blocker=Self)
    RecordPlayerStory(EventType=SpoolPiece)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=heartpiece_decorative_ui (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    GetPlayerDataVariable(VariableName="silkSpoolParts", StoreValue=var "Spool Parts")
    IntCompare(integer1=var "Spool Parts", integer2=1, equal=→"FULL", lessThan=→"PART", greaterThan=→"FULL", everyFrame=false)
    on PART → Part
    on FULL → Full
  }

  state Part {
    IntOperator(integer1=var "Spool Parts", integer2=1, operation=Add, storeResult=var "Spool Parts", everyFrame=false)
    SetPlayerDataVariable(VariableName="silkSpoolParts", SetValue=var "Spool Parts")
    SendEventToRegister(eventName="SILK SPOOL SAVE")
    UpdateSilkSpoolAchievement()
    SetAnimator(target=Self, active=true)
    AnimatorPlayStateWait(target=Self, stateName="Part", finishEvent=→"FINISHED")
    ListenForAnimationEvent(Target=Self, Response=→"FINISHED")
    on FINISHED → Hero Anim End
  }

  state Full {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddToMaxSilk", parameters=[1], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataVariable(VariableName="silkSpoolParts", SetValue=0)
    SendEventToRegister(eventName="SILK SPOOL SAVE")
    SetAnimator(target=Self, active=true)
    AnimatorPlayStateWait(target=Self, stateName="Full", finishEvent=→"FINISHED")
    on FINISHED → Wait
  }

  state Continue Deposit {
    SendEventToRegister(eventName="CONTINUE QUEST BOARD DEPOSIT")
    on FINISHED → Destroy Self
  }

  state Wait {
    PlayerDataVariableTest(VariableName="IsAnyCursed", ExpectedValue=true, IsExpectedEvent=→"CANCEL", IsNotExpectedEvent=(none))
    on SPOOL MAX UP ENDED → Full End
    on CANCEL → Full End
  }

  state Full End {
    AnimatorPlayStateWait(target=Self, stateName="Full End", finishEvent=→"FINISHED")
    on FINISHED → Hero Anim End
  }

  state State 1 {
    RemoveHeroInputBlocker(Blocker=Self)
  }

  state Save Queued {
    DoQueuedSaveGame()
    on FINISHED → Destroy Self
  }

  var Anim Length: float = 4

  var Silk Max: int = 0
  var Spool Parts: int = 0
  var In Interaction: bool = false
  var Hero: gameObject = <null>
  var Self: gameObject = <null>
}
