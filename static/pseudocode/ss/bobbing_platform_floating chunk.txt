# bobbing_platform
# 3 location(s):
#   Peak_04 / drop pillar_chunk (2)/floating chunk
#   Peak_04 / drop pillar_chunk (1)/floating chunk
#   Peak_05e / drop pillar_chunk/floating chunk

// uses template: bobbing_platform_heavy
fsm bobbing_platform {
  start Init

  state Init {
    FindNamedChild(gameObject=var "Art", storeResult=var "Plat Sprite")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "Current X", y=(unset), z=(unset), space=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Inactive
  }

  state Set Idle Pos {
    GetPosition(gameObject=Self, vector=(unset), x=var "Current X", y=(unset), z=(unset), space=0, everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Drift L
    on R → Drift R
  }

  state Drift L {
    SetFloatValue(floatVariable=var "Target X", floatValue=var "Current X", everyFrame=false)
    FloatAddRandom(floatVariable=var "Target X", addMin=-0.4, addMax=-0.2)
    AccelerateToX(gameObject=Self, accelerationFactor=0.0025, targetSpeed=-0.15)
    CheckXPosition(gameObject=Self, compareTo=var "Target X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"R", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    Collision2dEventLayerV2(collision=OnCollisionStay2D, collideTag="", collideLayer=Player(9), sendEvent=→"LAND", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset), ignoreFloor=false, ignoreRoof=false, onlyAbove=true, onlyLeft=false, onlyRight=false)
    Collision2dEventLayerV2(collision=OnCollisionStay2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"LAND", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset), ignoreFloor=false, ignoreRoof=false, onlyAbove=true, onlyLeft=false, onlyRight=false)
    GetDistance(gameObject=Self, target=var "MainCamera", storeResult=var "Cam Distance", everyFrame=true)
    FloatCompare(float1=var "Cam Distance", float2=var "Active Distance", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"STOP", everyFrame=true)
    on R → Drift R
    on LAND → Land
    on STOP → Inactive
  }

  state Drift R {
    SetFloatValue(floatVariable=var "Target X", floatValue=var "Current X", everyFrame=false)
    FloatAddRandom(floatVariable=var "Target X", addMin=0.2, addMax=0.4)
    AccelerateToX(gameObject=Self, accelerationFactor=0.0025, targetSpeed=0.15)
    CheckXPosition(gameObject=Self, compareTo=var "Target X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"L", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    Collision2dEventLayerV2(collision=OnCollisionStay2D, collideTag="", collideLayer=Player(9), sendEvent=→"LAND", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset), ignoreFloor=false, ignoreRoof=false, onlyAbove=true, onlyLeft=false, onlyRight=false)
    Collision2dEventLayerV2(collision=OnCollisionStay2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"LAND", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset), ignoreFloor=false, ignoreRoof=false, onlyAbove=true, onlyLeft=false, onlyRight=false)
    GetDistance(gameObject=Self, target=var "MainCamera", storeResult=var "Cam Distance", everyFrame=true)
    FloatCompare(float1=var "Cam Distance", float2=var "Active Distance", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"STOP", everyFrame=true)
    on L → Drift L
    on LAND → Land
    on STOP → Inactive
  }

  state Land {
    PlayAudioEventRandom(audioClips=[lilypad_hornet_land_1 (sfxstatic_assets_areaclover.bundle), lilypad_hornet_land_2 (sfxstatic_assets_areaclover.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Collision2dEventLayerV2(collision=OnCollisionExit2D, collideTag="", collideLayer=Player(9), sendEvent=→"LEAVE", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset), ignoreFloor=false, ignoreRoof=false, onlyAbove=false, onlyLeft=false, onlyRight=false)
    Collision2dEventLayerV2(collision=OnCollisionExit2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"LEAVE", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset), ignoreFloor=false, ignoreRoof=false, onlyAbove=false, onlyLeft=false, onlyRight=false)
    on LEAVE → Leave
  }

  state Leave {
    Tk2dPlayAnimation(gameObject=var "Plat Sprite", animLibName="", clipName="Plat Land")
    on FINISHED → Set Idle Pos
  }

  state Inactive {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    GetDistance(gameObject=Self, target=var "MainCamera", storeResult=var "Cam Distance", everyFrame=true)
    FloatCompare(float1=var "Cam Distance", float2=var "Active Distance", tolerance=0, equal=(none), lessThan=→"START", greaterThan=(none), everyFrame=true)
    on START → Restart
  }

  state Restart {
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Drift L
    on R → Drift R
  }

  var Current X: float = 0
  var Target X: float = 0
  var X Speed: float = 0
  var Start Y: float = 0
  var Cam Distance: float = 0
  var Active Distance: float = 45
  var Bop Audio Timer: float = 0
  var Ray Distance: float = 4.2
  var Art: gameObject = <null>
  var Plat Sprite: gameObject = <null>
}
