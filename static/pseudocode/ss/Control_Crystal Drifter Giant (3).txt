# Control
# 1 location(s):
#   Peak_06 / Crystal Drifter Giant (3)

fsm Control {
  start Init
  on ZERO HP → Burst  // from any state

  state Init {
    CallMethodProper(gameObject=Self, behaviour="AmbientSway", methodName="RecordInitialPosition", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    SetAudioLoop(gameObject=Self, loop=true)
    CallMethodProper(gameObject=Self, behaviour="AmbientSway", methodName="ResumeSway", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetAudioClip(gameObject=Self, audioClip=crystal_drifter_giant_fly_loop (sfxstatic_assets_areapeak.bundle), autoPlay=true, stopOnExit=false)
    AudioPlayInState(gameObject=Self, volume=1)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    IdleBuzzV4(gameObject=Self, waitMin=0.25, waitMax=0.5, speedMax=9, deceleration=0.95, accelerationMin=40, accelerationMax=40, roamingRangeX=0.25, roamingRangeY=0.25, manualStartPos=(unset))  // disabled
    TimeLimitSetV2(TimeDelay=1, StoreValue=var "Attack Cooldown", CanSet=(unset), EveryFrame=false)
    TimeLimitCheckV2(StoredValue=var "Attack Cooldown", AboveEvent=(none), BelowEvent=(none), StoreBool=var "Cooldown Finished", EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=var "In Attack Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    BoolAllTrue(boolVariables=[var "In Attack Range", var "Cooldown Finished"], sendEvent=→"ATTACK", storeResult=(unset), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on TOOK DAMAGE → Antic
    on ATTACK → Antic
    on SING → Sing
  }

  state Antic {
    SetAudioLoop(gameObject=Self, loop=false)
    SetAudioClip(gameObject=Self, audioClip=crystal_drifter_giant_antic (sfxstatic_assets_areapeak.bundle), autoPlay=true, stopOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=true)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=true)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashingWhite())
    StopParticleEmittersInChildren(gameObject=var "Pt Idle")
    PlayParticleEmitterChildren(gameObject=var "Pt Broken", resetTimeIfPlaying=false, stopOnStateExit=false)
    Wait(time=0.9, finishEvent=→"FINISHED", realTime=false)
    CallMethodProper(gameObject=Self, behaviour="AmbientSway", methodName="StopSway", parameters=[], storeResult=(unset var), EveryFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    on FINISHED → Spawn Spikes
    on SING → Sing
  }

  state Crystal Burst {
    SendMessageV2(gameObject=var "Spawn Point", delivery=BroadcastMessage, options=1, functionCall=SendEvent("FIRE"), everyFrame=false)  // disabled
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitterChildren(gameObject=var "Pt Fire", resetTimeIfPlaying=false, stopOnStateExit=false)
    on FINISHED → Fired
  }

  state Gone {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reform Antic
  }

  state Reform Antic {
    SetAudioClip(gameObject=Self, audioClip=crystal_drifter_giant_spawn_start (sfxstatic_assets_areapeak.bundle), autoPlay=true, stopOnExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Pt ReformAntic", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt ReformAntic", resetTimeIfPlaying=false, stopOnStateExit=true)
    on FINISHED → Reform
  }

  state Reform {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Regrow", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetMeshRenderer(gameObject=Self, active=true)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RefillHP())
    on FINISHED → Reform 2
  }

  state Reform 2 {
    PlayAudioEvent(audioClip=crystal_drifter_giant_spawn_end (sfxstatic_assets_areapeak.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitterChildren(gameObject=var "Pt Reform", resetTimeIfPlaying=false, stopOnStateExit=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Idle", resetTimeIfPlaying=false, stopOnStateExit=false)
    StopParticleEmittersInChildren(gameObject=var "Pt Broken")
    ActivateGameObject(gameObject=var "Pt ReformAntic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Spawn Spikes {
    SpawnRandomObjectsRadialV2(Prefab=Crystal Drifter Giant Spike (localpoolprefabs_assets_areapeak.bundle), SpawnPoint=var "Spawn Point", SpawnAsChildren=false, MinCount=15, MaxCount=20, MinAngle=0, MaxAngle=360, AngleErrorMax=5, SpawnRadius=4.8)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelFlash())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashWhiteQuick())
    on FINISHED → Wait Antic
  }

  state Wait Antic {
    Wait(time=0.9, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=true)
    on FINISHED → Crystal Burst
  }

  state Fired {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fire", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Burst {
    PlayAudioEvent(audioClip=crystal_drifter_giant_death (sfxstatic_assets_areapeak.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CallMethodProper(gameObject=Self, behaviour="AmbientSway", methodName="StopSway", parameters=[], storeResult=(unset var), EveryFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Form", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelFlash())
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    FreezeMomentV2(FreezeMomentType=EnemyDeath, WaitForFinish=false)
    SendMessageV2(gameObject=var "Spawn Point", delivery=BroadcastMessage, options=1, functionCall=SendEvent("FIRE"), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Burst", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Journal Entry?
  }

  state Journal Entry? {
    HasTakenDamage(target=Self, takenDamageBool=var "Has Taken Damage", eventTarget=Self, takenDamageEvent=(none))
    BoolTest(boolVariable=var "Has Taken Damage", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    RecordJournalKillV2(Record=Crystal Drifter Giant (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle), ShowPopup=true)
    on FINISHED → Gone
  }

  state Sing {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=crystal_drifter_giant_sing (sfxdynamic_assets_areapeak.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SpawnRandomObjectsRadialV2(Prefab=Crystal Drifter Giant Spike (localpoolprefabs_assets_areapeak.bundle), SpawnPoint=var "Spawn Point", SpawnAsChildren=false, MinCount=15, MaxCount=20, MinAngle=0, MaxAngle=360, AngleErrorMax=5, SpawnRadius=4.8)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelFlash())
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=flashWhiteQuick())
    on END → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Attack Cooldown: float = 0
  var Cooldown Finished: bool = false
  var In Attack Range: bool = false
  var Has Taken Damage: bool = false
  var Pt Idle: gameObject = Crystal Drifter Giant (3)/Pt Idle
  var Pt Antic: gameObject = Crystal Drifter Giant (3)/Pt Antic
  var Pt Form: gameObject = Crystal Drifter Giant (3)/Pt Form
  var Pt ReformAntic: gameObject = Crystal Drifter Giant (3)/Pt ReformAntic
  var Pt Reform: gameObject = Crystal Drifter Giant (3)/Pt Reform
  var Pt Broken: gameObject = Crystal Drifter Giant (3)/Pt Broken
  var Spawn Point: gameObject = Crystal Drifter Giant (3)/Spikes Spawn Point
  var Pt Fire: gameObject = Crystal Drifter Giant (3)/Pt Fire
  var Audio Loop Voice: gameObject = Crystal Drifter Giant (3)/Audio Loop Voice
}
