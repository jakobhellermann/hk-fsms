# Notice Reaction
# 1 location(s):
#   Tut_04 / States/Intro Scene/Snail Shamans Set/Shaman Mid

fsm Notice Reaction {
  start Init

  state Init {
    on NOTICE → Notice?
    on CANCEL → Inert
  }

  state Notice? {
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="left1", equalEvent=(none), notEqualEvent=→"FINISHED", storeResult=(unset), everyFrame=false)
    PlayRandomAudioClipTableV3(Table=CM_notice (sfxdynamic_assets_areatutorial.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    on FINISHED → Inert
  }

  state Inert {
    EnableFsmSelf(SetEnabled=false)
  }

  var Entry Gate: string = ""
}
