# Control
# 3 location(s):
#   Abyss_05 / Abyss Tendril Hero Damager
#   Abyss_07 / Abyss Tendril Hero Damager
#   Abyss_08 / Abyss Tendril Hero Damager

fsm Control {
  start Init

  state Idle {
    AudioStop(gameObject=var "Current Clip", fadeTime=0)
    on ABYSS DAMAGE → Hit
  }

  state Hit {
    PlayerDataVariableTest(VariableName="HasWhiteFlower", ExpectedValue=true, IsExpectedEvent=→"CANCEL", IsNotExpectedEvent=(none))
    PlayAudioEvent(audioClip=sand_centipedes_eat_hornet_or_enemy (sfxstatic_assets_areacoral.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Current Clip")
    SpawnObjectFromGlobalPool(gameObject=standard_health_hit (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="TakeQuickDamage", parameters=[1, true], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "MainCamera", behaviour="CameraController", methodName="FreezeInPlace", parameters=[true], storeResult=(unset var), EveryFrame=false)
    SendEventToRegister(eventName="HERO DAMAGED")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="WOUND START", delay=0, everyFrame=false)
    Wait(time=0.6, finishEvent=→"MULTIHIT END", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=true)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Translate(gameObject=var "Hero", vector=(unset), x=0, y=-3.5, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SetPositionToObject2D(gameObject=var "Damage Effects", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    PlayParticleEmitterChildren(gameObject=var "Damage Effects", resetTimeIfPlaying=false, stopOnStateExit=true)
    on CANCEL → Idle
    on MULTIHIT END → Hazard Hit
  }

  state Hazard Hit {
    SendEventToRegister(eventName="ABYSS TENDRIL HAZARD")
    SpawnObjectFromGlobalPool(gameObject=standard_health_hit (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=true)
    SetPositionToObject2D(gameObject=var "Damage Effects", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    PlayParticleEmitterChildren(gameObject=var "Damage Effects", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=var "Hero", vector=(unset), x=0, y=-3.5, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Hazard Wait
    on HAZARD RESPAWNED → Void Silk
  }

  state Init {
    on FINISHED → Idle
  }

  state Void Silk {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="ActivateVoidAcid", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Hazard Wait {
    AudioStop(gameObject=var "Current Clip", fadeTime=0)
    on HAZARD RESPAWNED → Void Silk
  }

  var Damage Effects: gameObject = Abyss Tendril Hero Damager/Damage Effects
  var Current Clip: gameObject = <null>
}
