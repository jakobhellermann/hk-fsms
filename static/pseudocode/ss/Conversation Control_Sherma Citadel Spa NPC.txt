# Conversation Control
# 1 location(s):
#   Song_10 / Sherma Citadel Spa NPC

fsm Conversation Control {
  start Init

  state Init {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"DEACTIVATE", isFalse=(none))
    PlayerDataBoolTest(boolName="shermaCitadelSpa_Visiting", isTrue=(none), isFalse=→"DEACTIVATE")  // disabled
    PlayerDataBoolTest(boolName="shermaCitadelSpa_Left", isTrue=→"DEACTIVATE", isFalse=(none))
    PlayerDataBoolTest(boolName="shermaCitadelEnclave_Seen", isTrue=→"DEACTIVATE", isFalse=(none))
    ActivateGameObject(gameObject=var "Seen Range", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Seen Range", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
    on DEACTIVATE → Deactivate
  }

  state End {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK END", delay=0, everyFrame=false)
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Idle
  }

  state Greet {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_CITADEL_SPA_GREET", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on CONVO_END → End
  }

  state Idle {
    on INTERACT → Convo
  }

  state Convo {
    SetPlayerDataBool(boolName="metSherma", value=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALK START", delay=0, everyFrame=false)
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="SHERMA")
    BoolTest(boolVariable=var "Activated", isTrue=(none), isFalse=→"MEET", everyFrame=false)
    PlayerDataBoolTest(boolName="shermaCitadelSpa_ExtraConvo", isTrue=(none), isFalse=→"EXTRA")
    SendEvent(eventTarget=Self, sendEvent=→"REPEAT", delay=0, everyFrame=false)
    on MEET → Greet
    on EXTRA → Extra
    on REPEAT → Repeat
  }

  state Repeat {
    RunDialogueV4(CustomText="", Sheet="Wanderers", Key="SHERMA_CITADEL_SPA_REPEAT", TextColor=(unset), NpcVoiceTableOverride=sherma_talk_sing (sfxdynamic_assets_sherma.bundle), OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Extra {
    GetPlayerDataBool(boolName="visitedEnclave", storeValue=var "Visited Enclave")
    ConvertBoolToString(boolVariable=var "Visited Enclave", stringVariable=var "Convo Key", falseString="SHERMA_CITADEL_SPA_NOENCLAVE", trueString="SHERMA_CITADEL_SPA_ENCLAVE", everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key=var "Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetPlayerDataBool(boolName="shermaCitadelSpa_ExtraConvo", value=true)
    on CONVO_END → End
  }

  var Activated: bool = false
  var Visited Enclave: bool = false
  var Convo Key: string = ""
  var Seen Range: gameObject = Sherma Citadel Spa NPC/Seen Range
}
