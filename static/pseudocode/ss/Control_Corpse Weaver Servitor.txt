# Control
# 1 location(s):
#   enemycorpses_assets_areaweaver.bundle / Corpse Weaver Servitor

fsm Control {
  start Init
  on SPLASH → Splash  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetParent(gameObject=var "Effect", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Spin
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"BREAK", sendEventSpikes=→"BREAK", sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=true, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"NAIL", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Hit Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on TINK DOWN → Launch D
    on TINK LEFT → Launch L
    on TINK RIGHT → Launch R
    on TINK UP → Launch U
    on BREAK → Break
    on NAIL → Nail Dir
  }

  state Break {
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    PlayAudioEventRandom(audioClips=[shell_shard_rock_hit_1 (sfxstatic_assets_areabone.bundle)], pitchMin=0.85, pitchMax=1.15, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Shell Shard 01 (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=15, speedMax=25, angleMin=0, angleMax=180, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Shell Shard 02 (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=15, speedMax=25, angleMin=0, angleMax=180, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    ActivateGameObject(gameObject=var "Break Fx", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Launch L {
    SetVelocityAsAngle(gameObject=Self, angle=150, speed=25, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.2, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Launch Frame
  }

  state Launched {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=DoSpin())
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    ReceivedDamageStay(Target=Self, collideTag="", sendEvent=→"BREAK", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=true, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    Wait(time=2, finishEvent=→"BREAK", realTime=false)
    on BREAK → Break
  }

  state Launch Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=15, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Speed", everyFrame=false)
    on FINISHED → Launched
  }

  state Launch R {
    SetVelocityAsAngle(gameObject=Self, angle=30, speed=25, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.2, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Launch Frame
  }

  state Launch U {
    SetVelocityAsAngle(gameObject=Self, angle=90, speed=35, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.2, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Launch Frame
  }

  state Launch D {
    SetVelocityAsAngle(gameObject=Self, angle=270, speed=20, everyFrame=false)
    on FINISHED → Launch Frame
  }

  state Spin {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=DoSpin())
    on FINISHED → Idle
  }

  state Splash {
    on TINK DOWN → Break
    on TINK LEFT → Break
    on TINK RIGHT → Break
    on TINK UP → Break
  }

  state Nail Dir {
    PlayAudioEventRandom(audioClips=[shard_region_hit_1 (sfxstatic_assets_areabone.bundle), shard_region_hit_2 (sfxstatic_assets_areabone.bundle), shard_region_hit_3 (sfxstatic_assets_areabone.bundle), shard_region_hit_4 (sfxstatic_assets_areabone.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    FloatCompare(float1=var "Hit Direction", float2=0, tolerance=30, equal=→"TINK RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Hit Direction", float2=90, tolerance=30, equal=→"TINK UP", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Hit Direction", float2=180, tolerance=30, equal=→"TINK LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Hit Direction", float2=270, tolerance=30, equal=→"TINK DOWN", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"TINK UP", delay=0, everyFrame=false)
    on TINK DOWN → Launch D
    on TINK LEFT → Launch L
    on TINK RIGHT → Launch R
    on TINK UP → Launch U
  }

  var Speed: float = 0
  var Hit Direction: float = 0
  var Self: gameObject = Corpse Weaver Servitor
  var Break Fx: gameObject = Corpse Weaver Servitor/Break Fx
  var Effect: gameObject = Corpse Weaver Servitor/Effect
}
