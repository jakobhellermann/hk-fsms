  }

    EnumSwitch(enumVariable=var "Enviro Type", compareTo=[EnvironmentTypes(8)], sendEvent=[→"BELL"], everyFrame=false)
    on BELL → Stamp Bell
    on FINISHED → Stamp Generic
  }

  state Stamp Bell {
    PlayRandomAudioClipTable(Table=beastling_stamp_bell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))

  }

  state Stamp Generic {
