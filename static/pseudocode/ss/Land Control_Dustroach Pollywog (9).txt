# Land Control
# 1 location(s):
#   Shadow_13 / Dustroach Pollywog Control/Dustroach Pollywog Pool/Dustroach Pollywog (9)

fsm Land Control {
  start Unsolid
  on SPAWN → Unsolid  // from any state

  state Unsolid {
    ActivateGameObject(gameObject=var "Land Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on SOLID LAND → Solid
  }

  state Solid {
    ActivateGameObject(gameObject=var "Land Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on UNSOLID LAND → Unsolid
  }

  var Land Collider: gameObject = Dustroach Pollywog Control/Dustroach Pollywog Pool/Dustroach Pollywog (9)/Land Collider
}
