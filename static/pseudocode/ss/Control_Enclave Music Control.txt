# Control
# 2 location(s):
#   Song_Enclave / Black Thread States/Black Thread World/Enclave Act 3/Enclave Music Control
#   Song_Enclave / Black Thread States/Normal World/Enclave States/Enclave Music Control

fsm Control {
  start Pause

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    FindNamedChild(gameObject=Self, storeResult=var "First Visit Range")
    on FINISHED → Check Enclave State
  }

  state Check Enclave State {
    PlayerDataBoolTrueAndFalse(trueBool="blackThreadWorld", falseBool="act3_enclaveWakeSceneCompleted", isTrue=→"ACT 3 INTRO", isFalse=(none))
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="bot1", equalEvent=(none), notEqualEvent=(none), storeResult=var "Entering From Library", everyFrame=false)
    PlayerdataIntCompare(playerdataInt="enclaveLevel", compareTo=0, equal=→"UNESTABLISHED", lessThan=(none), greaterThan=(none))
    PlayerDataBoolTest(boolName="visitedEnclave", isTrue=→"VISITED", isFalse=→"UNVISITED")
    on UNESTABLISHED → Unestablished
    on UNVISITED → Unvisited Check
    on VISITED → Visited Check
    on ACT 3 INTRO → No Action
  }

  state Unestablished {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=6)
  }

  state Unvisited Check {
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="bot1", equalEvent=→"FROM LIBRARY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FROM LIBRARY → Unvisited From Library
    on FINISHED → Unvisited
  }

  state Unvisited From Library {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Music Type
  }

  state Unvisited {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    Trigger2dEvent(gameObject=var "First Visit Range", trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Unvisited Pause
  }

  state Music Normal {
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
  }

  state Unvisited Pause {
    Wait(time=1.5, finishEvent=(none), realTime=false)
    on FINISHED → Music Type
  }

  state Visited Check {
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="bot1", equalEvent=→"FROM LIBRARY", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FROM LIBRARY → Visited From Library
    on FINISHED → Visited Pause
  }

  state Visited Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    StringCompare(stringVariable=var "Gate", compareTo="door1", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Music Type
  }

  state No Action {
  }

  state Music Type {
    ApplyMusicCue(musicCue=Enclave (audiocuesdynamic_assets_areasong.bundle), delayTime=0, transitionTime=0)
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"SUB", isFalse=→"NORMAL")
    on NORMAL → Music Normal
    on SUB → Music Sub
  }

  state Music Sub {
    TransitionToAudioSnapshot(snapshot=Sub Area (audioobjects_assets_all.bundle), transitionTime=1)
  }

  state Visited From Library {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=3)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Music Type
  }

  var Entering From Library: bool = false
  var Gate: string = ""
  var First Visit Range: gameObject = <null>
}
