fsm BG Control {
  start Wait

  state Wait {
    on FINISHED → Idle
  }

  state Wait for open {
    CallMethodProper(gameObject=Self, behaviour="Trapdoor", methodName="CloseDoorCustom", parameters=[], storeResult=(unset var), EveryFrame=false)
    on BG OPEN → Idle
  }

  state Idle {
    CallMethodProper(gameObject=Self, behaviour="Trapdoor", methodName="OpenDoorCustom", parameters=[1], storeResult=(unset var), EveryFrame=false)
    on BG CLOSE → Wait for open
  }
}
