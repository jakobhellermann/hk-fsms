    SetPosition(gameObject=var "Explosion", vector=(unset), x=var "Shift X", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Pt Antic", vector=(unset), x=var "Shift X", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    RandomFloat(min=var "Scale Min", max=var "Scale Max", storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Antic
  }

  state Antic {
    ActivateGameObject(gameObject=var "Pt Antic", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1.15, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=TrobbioFireworkAntic (sfxdynamic_assets_trobbio.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Damaging?
  }

  state Explode {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Explosion", activate=true, recursive=false, resetOnExit=true, everyFrame=false)

    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Damaging? {
    SetBoolValue(boolVariable=var "Do Damage", boolValue=var "Nondamaging", everyFrame=false)
    BoolFlip(boolVariable=var "Do Damage")
    SetDamageHero(Target=var "Damager A", Enabled=var "Do Damage")
    SetDamageHero(Target=var "Damager B", Enabled=var "Do Damage")
    on FINISHED → Explode
  }

  var Scale Min: float = 1.5
  var Scale Max: float = 1.4

  var Shift X: float = 0
  var Scale: float = 0
  var Nondamaging: bool = false
  var Do Damage: bool = false
  var Pt Antic: gameObject = Grand Stage Scene/Boss Scene TormentedTrobbio/Flare Glitter/Flare Glitter 6/Pt Antic
  var Explosion: gameObject = Grand Stage Scene/Boss Scene TormentedTrobbio/Flare Glitter/Flare Glitter 6/Explosion
  var Damager A: gameObject = Grand Stage Scene/Boss Scene TormentedTrobbio/Flare Glitter/Flare Glitter 6/Explosion/Trobbio_dazzle_flash/Damager
  var Damager B: gameObject = Grand Stage Scene/Boss Scene TormentedTrobbio/Flare Glitter/Flare Glitter 6/Explosion/Trobbio_dazzle_flash (1)/Damager
