# Control
# 2 location(s):

fsm Control {
  start Init

  state Init {
    GetGrandparent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Start Range")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "CamLock Standard")
    ActivateGameObject(gameObject=var "Antic Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on FAKE BENCH SIT → Cage
  }

  state Cage {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=AffectedByGravity(false))
    SendEventByName(eventTarget=GameObject(var "Cage"), sendEvent="CAGE", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Roar Lock")
    Tk2dPlayFrame(gameObject=var "Hero", frame=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=true)
    PlayAudioEvent(audioClip=hunter_trap_cage_come_up (sfxstatic_assets_areaant.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=bone_lever_activate (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayRandomAudioClipTable(Table=Wound Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0))
    on FINISHED → Shift Down
  }

  state Step {
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Start Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=bone_trap_button (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Cage
  }

  state Shift Down {
    SendEventByNameUpwards(Target=Self, EventName="BENCH BREAK")
    StopParticleEmittersInChildrenV2(Target=var "Antic Effects", StopBehaviour=1)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Hero", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -73, 0), time=1.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    CameraFollowYInState(Target=Self, KeepOffset=true)
    SetCollider(gameObject=var "CamLock Standard", active=false, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Fall")
    PlayParticleEmitterInState(gameObject=var "Pt Drop")
    AudioPlayInState(gameObject=Self, volume=1)
    PlayAudioEvent(audioClip=misc_rumble_loop (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=0.85, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Rumble PLayer")
    GetPosition2d(gameObject=var "Caged Hero Bound X Min", vector_2d=(unset), x=var "Caged X Min", y=(unset), space=0, everyFrame=true)
    GetPosition2d(gameObject=var "Caged Hero Bound X Max", vector_2d=(unset), x=var "Caged X Max", y=(unset), space=0, everyFrame=true)
    GetPosition2d(gameObject=var "Caged Hero Bound Y Max", vector_2d=(unset), x=var "Caged Y Max", y=(unset), space=0, everyFrame=true)
    GetPosition2d(gameObject=var "Hero", vector_2d=(unset), x=var "Hero X", y=var "Hero Y", space=0, everyFrame=true)
    FloatClamp(floatVariable=var "Hero X", minValue=var "Caged X Min", maxValue=var "Caged X Max", everyFrame=true)
    FloatClamp(floatVariable=var "Hero Y", minValue=0, maxValue=var "Caged Y Max", everyFrame=true)
    SetPosition2d(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=var "Hero Y", space=0, everyFrame=true, lateUpdate=false)
    on FINISHED → Land
  }

  state Land {
    AudioStopV2(gameObject=var "Rumble PLayer", fadeTime=0, cancelOnEarlyExit=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayRandomAudioClipTable(Table=Wound Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0))
    PlayAudioEvent(audioClip=hunter_trap_cage_come_up (sfxstatic_assets_areaant.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=big_fly_wall_hit (sfxstatic_assets_areaslabareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "Cage"), sendEvent="READY", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Get Off", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=false)
    SetCollider(gameObject=var "Cage", active=true, resetOnExit=false)
    SetCollider(gameObject=var "CamLock Standard", active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Regain Control
  }

  state Regain Control {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControlToIdle", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(var "Boss Scene"), sendEvent="ENTER", delay=0, everyFrame=false)
  }

  var Caged X Min: float = 0
  var Caged X Max: float = 0
  var Caged Y Max: float = 0
  var Hero X: float = 0
  var Hero Y: float = 0
  var Cage: gameObject = Group(Split)/Group (4)/TrapBench/Cage
  var Self: gameObject = Group(Split)/Group (4)/TrapBench
  var Boss Scene: gameObject = <null>
  var Start Range: gameObject = <null>
  var CamLock Standard: gameObject = <null>
  var Pt Drop: gameObject = Group(Split)/Group (4)/TrapBench/Pt Drop
  var Slam Effect: gameObject = Group(Split)/Group (4)/TrapBench/Slam Effect
  var Break Effects: gameObject = Group(Split)/Group (4)/TrapBench/Break Effects
  var Antic Effects: gameObject = Group(Split)/Group (4)/TrapBench/Antic Effects
  var Rumble PLayer: gameObject = <null>
  var Caged Hero Bound Y Max: gameObject = Group(Split)/Group (4)/TrapBench/Caged Hero Bound Y Max
  var Caged Hero Bound X Max: gameObject = Group(Split)/Group (4)/TrapBench/Caged Hero Bound X Max
  var Caged Hero Bound X Min: gameObject = Group(Split)/Group (4)/TrapBench/Caged Hero Bound X Min
}
