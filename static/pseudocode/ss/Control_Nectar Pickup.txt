fsm Control {
  start Pause

  state Air {
    FaceAngle(gameObject=Self, angleOffset=0, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=true)
    ProjectileSquash(gameObject=Self, stretchFactor=1.4, stretchMinX=0.6, stretchMaxY=1.75, scaleModifier=1, otherGameObject=<null>, everyFrame=true)
    CheckCollisionSide(collidingObject=Self, topHit=var "Top", rightHit=var "Right", bottomHit=var "Bot", leftHit=var "Left", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=true)
    BoolAnyTrue(boolVariables=[var "Bot"], sendEvent=→"GROUND", storeResult=(unset), everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"COLLECT", storeCollider=var "")
    SetSpriteRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Land Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on GROUND → Ground
    on COLLECT → Collect
  }

  state Ground {
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Bot", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Left", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Right", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Top", boolValue=true, everyFrame=false)
    CheckCollisionSide(collidingObject=Self, topHit=var "Top", rightHit=var "Right", bottomHit=var "Bot", leftHit=var "Left", topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=true)
    BoolNoneTrue(boolVariables=[var "Bot"], sendEvent=→"AIR", storeResult=(unset), everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"COLLECT", storeCollider=var "")
    SetSpriteRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Land Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on AIR → Air
    on COLLECT → Collect
  }

  state Audio {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Pt Trail", storeResult=var "Pt Trail")
    on FINISHED → Air
  }

  state Collect {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Item Get Effect Small (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, -0.005), rotation=(unset), storeObject=var "")
    StopParticleEmitter(gameObject=var "Pt Trail")
    SetParent(gameObject=var "Pt Trail", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    CollectableItemCollect(Amount=1, Item=Shell Flower (dataassets_assets_assets/dataassets/collectables/collectableitems.bundle))
    SetPosition(gameObject=Self, vector=(unset), x=-50, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindNamedChild(gameObject=Self, storeResult=var "Land Sprite")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on FINISHED → Audio
    on ACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Bot: bool = false
  var Left: bool = false
  var Right: bool = false
  var Top: bool = false
  var Pt Trail: gameObject = <null>
  var Self: gameObject = <null>
  var Land Sprite: gameObject = <null>
}
