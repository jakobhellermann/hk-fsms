# Enter Memory
# 1 location(s):
#   Memory_Red / Memory Control/top1

fsm Enter Memory {
  start Pause

  state Pause {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    ActivateGameObject(gameObject=var "Impact Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Fall Audio")
    on START ENTRY → Set Respawn?
    on TEST → Test
  }

  state Regain Control {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Save?
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    ApplyMusicCue(musicCue=<null>, delayTime=0, transitionTime=0)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    CreateObject(gameObject=Hornet_red_memory_start_fall (localpoolprefabs_assets_areamemoryareatutorial.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "Fall Hero")
    SetParent(gameObject=var "Fall Hero", parent=var "Hero", resetLocalPosition=true, resetLocalRotation=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="FaceRight", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateGameObject(gameObject=var "Fall Audio", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Low Graphics?
  }

  state Save? {
    SaveGame()
    on FINISHED → End
  }

  state End {
    SendEventByNameUpwards(Target=Self, EventName="DOOR GET UP FINISHED")
    SetPlayerDataBool(boolName="disablePause", value=false)
    on TEST → Test
  }

  state Blank Screen {
    on FINISH ENTRY → Take Control
  }

  state Set Respawn? {
    ActivateGameObject(gameObject=var "Scenery", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetDeathRespawnMarker(Target=Memory Control/door_wakeInMemory/Death Respawn Marker)
    SetPlayerDataBool(boolName="disablePause", value=true)
    on FINISHED → Blank Screen
  }

  state Weak Fall {
    DoCameraShakeV4(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=true, vibrate=true)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    HeroClampFallVelocity(Hero=var "Hero")
    CameraBlurPlaneFade(From={Spacing=30, Vibrancy=0.5, MaskLerp=0}, To={Spacing=var "Blur Spacing", Vibrancy=var "Blur Vibrancy", MaskLerp=0}, Curve=curve[(time=0, value=0, inSlope=0, outSlope=0), (time=1, value=1, inSlope=0, outSlope=0)], Duration=0)
    CameraBlurPlaneFadeToV3(Spacing=var "Blur Spacing", Vibrancy=0, MaskLerp=1, MaskScale=2, Curve=curve[(time=0, value=0, inSlope=0, outSlope=1), (time=1, value=1, inSlope=1, outSlope=0)], Duration=7)
    ScreenFaderAlpha(EndAlpha=0, Duration=7)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Weak Fall")  // disabled
    CheckIsCharacterGrounded(Target=var "Hero", RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    PlayAudioEvent(audioClip=misc_rumble_impact (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on LAND → Weak Fall Land
  }

  state Weak Fall Land {
    SetPositionToObject2D(gameObject=var "Impact Effect", targetObject=var "Hero", xOffset=0, yOffset=-1, everyFrame=false)
    ActivateGameObject(gameObject=var "Impact Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    DestroyObject(gameObject=var "Fall Hero", delay=0, detachChildren=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Fall Audio", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=hornet_land_hard new (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=false_knight_land_1st_time (sfxstatic_assets_enemyheavy.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="FallToProstrate")
    CreateObject(gameObject=Hard Land Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Title
  }

  state Weak Fall Recover {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Prostrate Rise", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=hornet_jump_ss (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=weak_hornet_exert_short_01 (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayRandomAudioClipTable(Table=hornet_weak_exert_standard (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0))  // disabled
    PlayAudioEvent(audioClip=hornet_misc_foley_move_1 (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventDelayed(audioClip=hornet_misc_foley_move_2 (herosfxstatic_assets_all.bundle), delay=1, pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Regain Control
  }

  state Title {
    SendEventToRegister(eventName="AREA TITLE APPEAR")
    CameraBlurPlaneFadeToV3(Spacing=var "Blur Spacing", Vibrancy=0, MaskLerp=1, MaskScale=4, Curve=curve[(time=0, value=0, inSlope=0, outSlope=1), (time=1, value=1, inSlope=1, outSlope=0)], Duration=5)
    on TITLE DISPLAYED → Weak Fall Recover
  }

  state Test {
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Self", xOffset=0, yOffset=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Scenery", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Take Control
  }

  state Low Graphics? {
    SetFloatValue(floatVariable=var "Blur Spacing", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Blur Vibrancy", floatValue=0, everyFrame=false)
    CheckPlatformGraphicsTier(CheckValue=Low, EqualEvent=→"FINISHED", HigherEvent=(none), LowerEvent=→"FINISHED", NotEqualEvent=(none))
    SetFloatValue(floatVariable=var "Blur Spacing", floatValue=30, everyFrame=false)
    SetFloatValue(floatVariable=var "Blur Vibrancy", floatValue=0.5, everyFrame=false)
    on FINISHED → Weak Fall
  }

  var Scenery: gameObject = Scenery Groups/Entry Scenery

  var Blur Spacing: float = 0
  var Blur Vibrancy: float = 0
  var Self: gameObject = <null>
  var Fall Hero: gameObject = <null>
  var Impact Effect: gameObject = Memory Control/Red_Memory_Ground_Impact
  var Fall Audio: gameObject = <null>
}
