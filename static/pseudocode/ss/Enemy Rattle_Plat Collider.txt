fsm Enemy Rattle {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Plat Dust")
    on FINISHED → Detect
  }

  state Detect {
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag=var "Hopper", collideLayer=Enemies(11), sendEvent=→"COLLIDE", storeCollider=var "Enemy", storeForce=var "Force", contactPoint=(unset), contactNormal=(unset))
    on COLLIDE → Check Force
  }

  state Check Force {
    FloatCompare(float1=var "Force", float2=3, tolerance=0, equal=(none), lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Detect
    on FINISHED → Rattle
  }

  state Rattle {
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    CallMethodProper(gameObject=var "Parent", behaviour="JitterSelf", methodName="StartJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → End
  }

  state End {
    CallMethodProper(gameObject=var "Parent", behaviour="JitterSelf", methodName="StopJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Detect
  }

  var Force: float = 0
  var Hopper: string = ""
  var Parent: gameObject = <null>
  var Plat Dust: gameObject = <null>
  var Enemy: gameObject = <null>
}
