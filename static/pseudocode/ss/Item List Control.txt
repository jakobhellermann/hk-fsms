    GetLanguageString(sheetName="UI", convName="SHOP_PURCHASE_COMPLETE", storeValue=var "Str Prompt Complete")

  }

    on FINISHED → Queue Layout Rebuild
  }

  state Check Sub Items {
    CallMethodProper(gameObject=Self, behaviour="ShopMenuStock", methodName="GetHasSubItems", parameters=[var "Current Item"], storeResult=var "Has Sub Items", EveryFrame=false)
    BoolTest(boolVariable=var "Has Sub Items", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    ActivateGameObject(gameObject=var "Sub Item Group", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ShopMenuStock", methodName="SetupSubItems", parameters=[var "Current Item"], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Sub Item Group", behaviour="ShopSubItemSelection", methodName="SetItem", parameters=[var "Item GameObject", var "Current Item Sub"], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ShopMenuStock", methodName="GetSubItemSelectPrompt", parameters=[var "Current Item"], storeResult=var "Str", EveryFrame=false)
    SetTextMeshProText(gameObject=var "Confirm Msg", textString=var "Str")
    FadeNestedFadeGroup(Target=var "Sub Item Group", ToAlpha=1, FadeTime=0)
    on CANCEL → Activate Yes No
    on TO CONFIRM → To Confirm
  }

  state Activate Confirm {
    FadeNestedFadeGroup(Target=var "Confirm", ToAlpha=1, FadeTime=0)
    on FINISHED → Check Sub Items
  }

  sequence state To Confirm {
    CallMethodProper(gameObject=Self, behaviour="ShopMenuStock", methodName="GetSubItemPurchaseIcon", parameters=[var "Current Item", var "Current Item Sub"], storeResult=var "Sprite", EveryFrame=false)
    SetSpriteRendererSprite(gameObject=var "Confirm Sprite", sprite=var "Sprite")
    FadeNestedFadeGroup(Target=var "Sub Item Group", ToAlpha=0, FadeTime=0.2)
    FadeNestedFadeGroup(Target=var "Confirm Msg", ToAlpha=0, FadeTime=0.2)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Sub Item Group", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetTextMeshProText(gameObject=var "Confirm Msg", textString=var "Str Prompt Confirm")
    FadeNestedFadeGroup(Target=var "Confirm Msg", ToAlpha=1, FadeTime=0.2)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    on FINISHED → Activate Yes No
  }

  state Back to Sub? {
    BoolTest(boolVariable=var "Has Sub Items", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    on CANCEL → Reset
    on FINISHED → Blank Name and Desc
  }

  state Reset {
    SendEventToRegister(eventName="RESET SHOP WINDOW")
  }

  state Queue Layout Rebuild {
    MarkUIForRebuild(gameObject=var "Item Details", markGraphicRebuild=false, markLayoutRebuild=true)
    on FINISHED → Tool Return
  }
