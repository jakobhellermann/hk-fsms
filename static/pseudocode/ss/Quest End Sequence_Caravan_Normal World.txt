# Quest End Sequence
# 1 location(s):
#   Bone_10 / Black Thread States Thread Only Variant/Normal World/Caravan

// uses template: caravan_troupe_travel_handler
fsm Quest End Sequence {
  start Wait

  state Wait {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    on QUEST END TRAVEL → Travel
    on QUEST END NOTRAVEL → No Travel
  }

  state Travel {
    SetBoolValue(boolVariable=var "Do Travel", boolValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=1.5)
    TransitionToAudioSnapshot(snapshot=Off (audioobjects_assets_all.bundle), transitionTime=1.5)
    TransitionToAudioSnapshot(snapshot=at None (audioobjects_assets_all.bundle), transitionTime=1.5)
    on FINISHED → Fade Down
  }

  state No Travel {
    SetBoolValue(boolVariable=var "Do Travel", boolValue=false, everyFrame=false)
    on FINISHED → Fade Down
  }

  state Fade Down {
    SendEventToRegister(eventName="FLEA MUSIC STOP FAST")
    SendEventToRegister(eventName="CARAVAN TRAVEL START")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataVariable(VariableName="disableSaveQuit", SetValue=true)
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=true)
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 1), duration=1.5)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Location
  }

  state Travel? {
    ActivateInteraction(Activate=true)
    BoolTest(boolVariable=var "Do Travel", isTrue=→"TRAVEL", isFalse=→"NO TRAVEL", everyFrame=false)
    on TRAVEL → Sound With
    on NO TRAVEL → Sound Without
  }

  state Set Location {
    SetPlayerDataVariable(VariableName="CaravanTroupeLocation", SetValue=var "Next Location")
    on FINISHED → Travel?
  }

  state Change Scene {
    SetPlayerDataVariable(VariableName="disableSaveQuit", SetValue=false)
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=false)
    BeginSceneTransition(sceneName="Room_Caravan_Interior_Travel", entryGateName="", entryDelay=0, visualization=Default, preventCameraFadeOut=true)
  }

  state Disable {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on FINISHED → Fade Up
  }

  state Sound Without {
    PlayAudioEvent(audioClip=flea_carravan_travel_without_hornet_2d (sfxstatic_assets_fleacaravan.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventToRegister(eventName="AMBIENT FLEA AUDIO STOP")
    Wait(time=7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Disable
  }

  state Fade Up {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=1.5)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return Control
  }

  state Return Control {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataVariable(VariableName="disableSaveQuit", SetValue=false)
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=false)
  }

  state Sound With {
    StartPreloadingScene(SceneName="Room_Caravan_Interior_Travel", LoadMode=LoadSceneMode(1))
    SendEventToRegister(eventName="AMBIENT FLEA AUDIO STOP")
    PlayAudioEvent(audioClip=flea_caravan_leave_with_hornet_2d (sfxstatic_assets_fleacaravan.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Change Scene
  }

  var Next Location: enum = CaravanTroupeLocations(1)

  var Do Travel: bool = false
  var Self: gameObject = <null>
}
