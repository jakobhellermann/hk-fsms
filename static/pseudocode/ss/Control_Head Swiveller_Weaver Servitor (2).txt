fsm Control {
  start Init

  state Init {
    on LOOK START → Swivel L
  }

  state Swivel L {
    AnimateRotationTo(gameObject=Self, fromValue=var "Start Rotation", toValue=-5, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)

    on LOOK STOP → Stop
    on FINISHED → Swivel R
  }

  state Stop {
    AnimateRotationTo(gameObject=Self, fromValue=var "Start Rotation", toValue=0, worldSpace=false, negativeSpace=false, time=0.35, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Start Rotation", floatValue=0, everyFrame=false)
    on LOOK START → Swivel L
  }

  state Swivel R {
    AnimateRotationTo(gameObject=Self, fromValue=var "Start Rotation", toValue=5, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Start Rotation", floatValue=5, everyFrame=false)
    on LOOK STOP → Stop
    on FINISHED → Swivel L
  }

  var Start Rotation: float = 0
}
