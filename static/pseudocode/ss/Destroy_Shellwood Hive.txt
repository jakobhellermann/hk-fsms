# Destroy
# 1 location(s):
#   Shellwood_01 / Wasp Scene/Hive Scene/Shellwood Hive

fsm Destroy {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindChild(gameObject=Self, childName="CamLock", storeResult=var "CamLock")
    FindChild(gameObject=Self, childName="Mask", storeResult=var "Mask")
    FindChild(gameObject=Self, childName="Remasker New", storeResult=var "Remasker New")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"DESTROY", isFalse=(none), everyFrame=false)
    on DESTROY → Spawn Remnant Wasps
    on FINISHED → Idle
  }

  state Idle {
    SendEventByName(eventTarget=GameObject(var "Control"), sendEvent="SPAWN", delay=0, everyFrame=false)
    on DESTROY → Destroy
  }

  state Deactivate {
    ActivateGameObject(gameObject=var "BG Wasps", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Destroy {
    PlayAudioEvent(audioClip=shellwood_hive_break (sfxstatic_assets_areashellwood.bundle), pitchMin=0.85, pitchMax=0.85, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    RecordJournalKill(Record=<null>)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Destroy Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Remasker New", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Art", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Control", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    FadeGroupDown(target=var "Mask", fast=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  state Spawn Remnant Wasps {
    SendRandomEventV4(events=[→"ONE", →"TWO"], weights=[0.7, 0.3], eventMax=[1, 1], missedMax=[1, 1], activeBool=(unset))
    on ONE → 1 Wasp
    on TWO → 2 Wasps
  }

  state 1 Wasp {
    CreateObject(gameObject=Shellwood Wasp (localpoolprefabs_assets_areashellwood.bundle), spawnPoint=var "Wasps - Hive Destroyed", position=(unset), rotation=(unset), storeObject=var "Wasp")
    TranslateRandom(gameObject=var "Wasp", translateMin=(-0.5, -0.5, 0), translateMax=(0.5, 0.5, 0), space=1)
    on FINISHED → Deactivate
  }

  state 2 Wasps {
    CreateObject(gameObject=Shellwood Wasp (localpoolprefabs_assets_areashellwood.bundle), spawnPoint=var "Wasps - Hive Destroyed", position=(unset), rotation=(unset), storeObject=var "Wasp")
    TranslateRandom(gameObject=var "Wasp", translateMin=(-2, -1, 0), translateMax=(-2, 1, 0), space=1)
    CreateObject(gameObject=Shellwood Wasp (localpoolprefabs_assets_areashellwood.bundle), spawnPoint=var "Wasps - Hive Destroyed", position=(unset), rotation=(unset), storeObject=var "Wasp")
    TranslateRandom(gameObject=var "Wasp", translateMin=(2, -1, 0), translateMax=(2, 1, 0), space=1)
    on FINISHED → Deactivate
  }

  var BG Wasps: gameObject = <null>

  var Activated: bool = false
  var Art: gameObject = Wasp Scene/Hive Scene/Shellwood Hive/Art
  var CamLock: gameObject = <null>
  var Control: gameObject = Wasp Scene/Hive Scene/Shellwood Hive/Control
  var Destroy Effects: gameObject = Wasp Scene/Hive Scene/Shellwood Hive/Destroy Effects
  var Mask: gameObject = <null>
  var Remasker New: gameObject = <null>
  var Terrain: gameObject = Wasp Scene/Hive Scene/Shellwood Hive/Terrain
  var Wasps - Hive Destroyed: gameObject = Wasp Scene/Hive Scene/Shellwood Hive/Control/Wasps - Hive Destroyed
  var Wasp: gameObject = <null>
}
