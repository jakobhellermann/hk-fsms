# Control
# 1 location(s):
#   Shellwood_11b_Memory / Boss Scene/Spike Balls P1

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Flower Queen Boss", storeResult=var "Boss")
    GetPosition(gameObject=Self, vector=(unset), x=var "Mid X", y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on SPIKE ATTACK → Check Dir
  }

  state Check Dir {
    SetPosition(gameObject=Self, vector=(unset), x=var "Mid X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    CheckTargetDirection(gameObject=Self, target=var "Boss", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Shift L
    on R → Shift R
  }

  state Shift L {
    Translate(gameObject=Self, vector=(unset), x=-1.75, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Send Events
  }

  state Shift R {
    Translate(gameObject=Self, vector=(unset), x=1.75, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Send Events
  }

  state Send Events {
    ArrayShuffle(array=var "Spike Balls", startIndex=(unset), shufflingRange=(unset))
    ArrayGet(array=var "Spike Balls", index=0, storeValue=var "Next Ball", everyFrame=false, indexOutOfRange=(none))
    SendEventByName(eventTarget=GameObject(var "Next Ball"), sendEvent="ATTACK HIGH", delay=0, everyFrame=false)
    ArrayGet(array=var "Spike Balls", index=1, storeValue=var "Next Ball", everyFrame=false, indexOutOfRange=(none))
    SendEventByName(eventTarget=GameObject(var "Next Ball"), sendEvent="ATTACK HIGH", delay=0, everyFrame=false)
    ArrayGet(array=var "Spike Balls", index=2, storeValue=var "Next Ball", everyFrame=false, indexOutOfRange=(none))
    SendEventByName(eventTarget=GameObject(var "Next Ball"), sendEvent="ATTACK TOP", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Spike Balls: array = var "Spike Balls"

  var Mid X: float = 0
  var Boss: gameObject = <null>
  var Next Ball: gameObject = <null>
  var Parent: gameObject = <null>
}
