# Heal HP
# 1 location(s):
#   Belltown_Room_Spare / furnishings/spa/spa_bucket/Spa Region

fsm Heal HP {
  start Idle
  on STOP HEAL → Idle  // from any state

  state Idle {
    on START HEAL → Heal Start
  }

  state Heal Start {
    SetBoolValue(boolVariable=var "Do Effect", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Heal Silk", boolValue=false, everyFrame=false)
    on FINISHED → Heal HP?
  }

  state Heal HP? {
    GetPlayerDataInt(intName="health", storeValue=var "HP")
    GetPlayerDataVariable(VariableName="CurrentMaxHealth", StoreValue=var "Max HP")
    IntCompare(integer1=var "HP", integer2=var "Max HP", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AddHealth", parameters=[1], storeResult=(unset var), EveryFrame=false)
    SetBoolValue(boolVariable=var "Do Effect", boolValue=true, everyFrame=false)
    on FINISHED → Heal Silk?
  }

  state Heal Silk? {
    GetPlayerDataInt(intName="silk", storeValue=var "Silk")
    GetPlayerDataVariable(VariableName="CurrentSilkMax", StoreValue=var "Max Silk")
    IntCompare(integer1=var "Silk", integer2=var "Max Silk", equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    AddSilk(target=var "Hero", amount=1, playHeroEffect=false)
    SetBoolValue(boolVariable=var "Do Effect", boolValue=true, everyFrame=false)
    SetBoolValue(boolVariable=var "Did Heal Silk", boolValue=true, everyFrame=false)
    on FINISHED → Do Effect?
  }

  state Do Effect? {
    BoolTest(boolVariable=var "Do Effect", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetPositionToObject2D(gameObject=var "Spa Heal HP Burst", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Spa Heal HP Burst", emit=0, resetIfPlaying=false)
    DoSpriteFlash(Target=var "Hero", FlashMethod="flashSoulGet", IsRepeating=false, CancelOnExit=false)
    PlayAudioEvent(audioClip=focus_ready (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Pause 1
  }

  state Pause 2 {
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Heal Start
  }

  state Pause 1 {
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Extra Silk?
  }

  state Extra Silk? {
    BoolTest(boolVariable=var "Did Heal Silk", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    AddSilk(target=var "Hero", amount=1, playHeroEffect=false)
    on FINISHED → Pause 2
  }

  var HP: int = 0
  var Max HP: int = 0
  var Silk: int = 0
  var Max Silk: int = 0
  var Do Effect: bool = false
  var Did Heal Silk: bool = false
  var Spa Heal HP Burst: gameObject = furnishings/spa/spa_bucket/Spa Region/Spa Heal HP Burst
}
