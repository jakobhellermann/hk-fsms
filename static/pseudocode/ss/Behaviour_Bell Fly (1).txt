# Behaviour
# 2 location(s):
#   Belltown_04 / Bell Fly (1)

fsm Behaviour {
  start Init
  on BLOCKED HIT → Bounce  // from any state

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    on FINISHED → Fly
  }

  state Fly {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SING → Sing
  }

  state Bounce {
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[bell_fly_hit_on_head_1 (sfxstatic_assets_areabell.bundle), bell_fly_hit_on_head_2 (sfxstatic_assets_areabell.bundle), bell_fly_hit_on_head_3 (sfxstatic_assets_areabell.bundle), bell_goomba_hit_on_head_4 (sfxstatic_assets_areabell.bundle)], weights=[1, 1, 1, 1], pitchMin=0.85, pitchMax=1.15, stopPreviousSound=true)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Tween"), sendEvent="PAUSE", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, -7), x=(unset), y=(unset), everyFrame=false)
    AudioStop(gameObject=var "Head Bounce Player", fadeTime=0)
    PlayAudioEvent(audioClip=bell_fly_head_bounce (sfxstatic_assets_areabell.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Head Bounce Player")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bounce", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    on FINISHED → Resume
  }

  state Sing {
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=bell_fly_sing (sfxdynamic_assets_areabell.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.7, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Tween"), sendEvent="PAUSE", delay=0, everyFrame=false)
    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStop(gameObject=var "Audio Loop Sing", fadeTime=0.15)
    on FINISHED → Resume
  }

  state Resume {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Tween"), sendEvent="RESUME", delay=0, everyFrame=false)
    on FINISHED → Fly
  }

  var Head Bounce Player: gameObject = <null>
  var Self: gameObject = <null>
  var Audio Loop Sing: gameObject = <null>
  var Audio Loop Voice: gameObject = Bell Fly (1)/Audio Loop Voice
}
