# FSM
# 2 location(s):
#   Coral_19 / Coral_01/Support Plat/rubble_fall_generic_follow_right (1)

fsm FSM {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Follow Parent")
    FindNamedChild(gameObject=var "Follow Parent", storeResult=var "Impact Dust (1)")
    FindNamedChild(gameObject=var "Follow Parent", storeResult=var "Impact Dust (2)")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    StopParticleEmittersInChildren(gameObject=var "Follow Parent")  // disabled
    on ENTER → Follow
  }

  state Follow {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    PlayParticleEmitterChildren(gameObject=var "Follow Parent", resetTimeIfPlaying=false, stopOnStateExit=true)
    GetPosition2d(gameObject=var "Hero", vector_2d=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=true)
    SetPosition2d(gameObject=var "Follow Parent", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=true, lateUpdate=false)
    on EXIT → Idle
  }

  var Hero X: float = 0
  var Follow Parent: gameObject = <null>
  var Impact Dust (1): gameObject = <null>
  var Impact Dust (2): gameObject = <null>
}
