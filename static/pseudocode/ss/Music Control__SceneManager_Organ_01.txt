# Music Control
# 1 location(s):
#   Organ_01 / _SceneManager

fsm Music Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait For Scene Manager
  }

  state Check {
    PlayerDataBoolTest(boolName="defeatedPhantom", isTrue=→"DEFEATED", isFalse=→"UNDEFEATED")
    on DEFEATED → Defeated
    on UNDEFEATED → Undefeated
  }

  state Defeated {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=8)
  }

  state Undefeated {
    ApplyMusicCue(musicCue=MistMaze_Organ (audiocuesdynamic_assets_areaorgan.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal - SlightlyMuffledOrgan (audioobjects_assets_all.bundle), transitionTime=4)
  }

  state Wait For Scene Manager {
    WaitForSceneAudioSnapshotsApplied(SendEvent=→"FINISHED")
    on FINISHED → Check
  }
}
