# Control
# 1 location(s):
#   Ward_07 / Group:1/Junk Hatch

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Lid L")
    FindNamedChild(gameObject=Self, storeResult=var "Lid R")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Junk Fall")
    FindNamedChild(gameObject=Self, storeResult=var "Drop Bed")
    FindNamedChild(gameObject=Self, storeResult=var "Drop Chair")
    FindNamedChild(gameObject=Self, storeResult=var "Corpse")
    FindNamedChild(gameObject=Self, storeResult=var "Collectable Item Pickup")
    FindNamedChild(gameObject=Self, storeResult=var "Return Corpse")
    FindNamedChild(gameObject=Self, storeResult=var "Harpoon Ring Pull Switch")
    FindNamedChild(gameObject=Self, storeResult=var "Song Threaded Husk R")
    FindNamedChild(gameObject=Self, storeResult=var "Song Threaded Husk L")
    FindNamedChild(gameObject=Self, storeResult=var "Jitter Corpses R")
    FindNamedChild(gameObject=Self, storeResult=var "Jitter Corpses L")
    FindNamedChild(gameObject=Self, storeResult=var "Return Enemy")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ACTIVATE → Activated
    on FINISHED → Idle
  }

  state Idle {
    ActivateGameObject(gameObject=var "Return Corpse", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return Enemy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on OPEN → Open Pause
  }

  sequence state Open Pause {
    Wait(time=1, finishEvent=(none), realTime=false)
    PlayAudioEvent(audioClip=ward_slide_hatch_lock_activate (sfxstatic_assets_areaward.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Open
  }

  state Open {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayAudioEvent(audioClip=song_gate_small_open (sfxstatic_assets_areasong.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=junk_chute_ward_07_empty (sfxstatic_assets_areaward.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimateRotationTo(gameObject=var "Lid R", fromValue=0, toValue=90, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=23, reverse=false, finishEvent=(none), realTime=false)
    AnimateRotationTo(gameObject=var "Lid L", fromValue=0, toValue=-90, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=23, reverse=false, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Drop Bed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Junk Fall", resetTimeIfPlaying=false, stopOnStateExit=false)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=true, Delay=0, RepeatDelay=0.2)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObjectDelay(gameObject=var "Corpse", activate=true, resetOnExit=false, delay=1.8)
    ActivateGameObjectDelay(gameObject=var "Drop Chair", activate=true, resetOnExit=false, delay=1)
    on FINISHED → End
  }

  sequence state End {
    ActivateGameObject(gameObject=var "Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    SetPositionToObject2D(gameObject=var "Collectable Item Pickup", targetObject=var "Corpse", xOffset=0, yOffset=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on ITEM COLLECTED → Wake Enemies
  }

  state Activated {
    ActivateGameObject(gameObject=var "Harpoon Ring Pull Switch", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Return Enemy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetRotation(gameObject=var "Lid R", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=90, space=1, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Lid L", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-90, space=1, everyFrame=false, lateUpdate=false)
  }

  sequence state Wake Enemies {
    Wait(time=0.75, finishEvent=(none), realTime=false)
    SendEventByName(eventTarget=GameObject(var "Song Threaded Husk L"), sendEvent="WAKE", delay=0, everyFrame=false)  // disabled
    SendMessage(gameObject=var "Jitter Corpses L", delivery=BroadcastMessage, options=1, functionCall=StartJitter())  // disabled
    Wait(time=2, finishEvent=(none), realTime=false)  // disabled
    SendMessage(gameObject=var "Jitter Corpses L", delivery=BroadcastMessage, options=1, functionCall=StopJitter())  // disabled
    Wait(time=0.5, finishEvent=(none), realTime=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Song Threaded Husk R"), sendEvent="WAKE", delay=0, everyFrame=false)
    SendMessage(gameObject=var "Jitter Corpses R", delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    Wait(time=2, finishEvent=(none), realTime=false)
    SendMessage(gameObject=var "Jitter Corpses R", delivery=BroadcastMessage, options=1, functionCall=StopJitter())
  }

  var Activated: bool = false
  var Collectable Item Pickup: gameObject = <null>
  var Corpse: gameObject = <null>
  var Drop Bed: gameObject = <null>
  var Drop Chair: gameObject = <null>
  var Harpoon Ring Pull Switch: gameObject = <null>
  var Jitter Corpses R: gameObject = <null>
  var Lid L: gameObject = <null>
  var Lid R: gameObject = <null>
  var Pt Junk Fall: gameObject = <null>
  var Return Corpse: gameObject = <null>
  var Return Enemy: gameObject = <null>
  var Song Threaded Husk L: gameObject = <null>
  var Song Threaded Husk R: gameObject = <null>
  var Jitter Corpses L: gameObject = <null>
}
