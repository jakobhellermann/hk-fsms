# Custom Guard Range
# 1 location(s):
#   Ant_02 / Battle Scene/Wave 1/Bone Hunter Throw/Guard Range

fsm Custom Guard Range {
  start State 1

  state State 1 {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → State 2
  }

  state State 2 {
    GetEntryGateName(StoreValue=var "Gate Name")
    StringCompare(stringVariable=var "Gate Name", compareTo="left1", equalEvent=→"SHIFT DOWN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on SHIFT DOWN → State 3
  }

  state State 3 {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  var Gate Name: string = ""
}
