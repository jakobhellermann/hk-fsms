# Msg Collider
# 1 location(s):
#   localpoolprefabs_assets_areadocks.bundle / DF Bomb Rock

fsm Msg Collider {
  start Pause

  state Pause {
    ActivateGameObject(gameObject=var "ESCAPE Message", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Active
  }

  state Active {
    ActivateGameObject(gameObject=var "ESCAPE Message", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var ESCAPE Message: gameObject = DF Bomb Rock/ESCAPE Message
}
