# Control
# 1 location(s):
#   Bone_East_01 / Barrel Plat Lift (2)

fsm Control {
  start Idle

  state Idle {
    FindNamedChild(gameObject=Self, storeResult=var "Drop Message")
    CallMethodProper(gameObject=var "Hero Stick", behaviour="HeroPlatformStick", methodName="set_IsActive", parameters=[false], storeResult=(unset var), EveryFrame=false)
    on ACTIVATE → Activate
    on OPEN → Antic
  }

  state Activate {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Raise Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
  }

  state Antic {
    iTweenMoveBy(gameObject=Self, id="", vector=(0, -2, 0), time=1, delay=0, speed=(unset), easeType=5, loopType=0, space=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)  // disabled
    ObjectJitterLocal(gameObject=var "Plat Sprite", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=30)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Drop Message", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    PlayAudioEvent(audioClip=dock_lift_activate (sfxstatic_assets_areadocks.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Dir
  }

  state Rise Up {
    CallMethodProper(gameObject=var "Hero Stick", behaviour="HeroPlatformStick", methodName="set_IsActive", parameters=[true], storeResult=(unset var), EveryFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=10, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Raise Y", tolerance=0, equal=→"END", lessThan=(none), greaterThan=→"END", everyFrame=true)
    ObjectJitterLocal(gameObject=var "Plat Sprite", x=0.05, y=0.05, z=(unset), resetPosOnStateExit=true, limitFps=30)  // disabled
    on END → End
  }

  state End {
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Raise Y", Space=0, EveryFrame=false)
    ObjectJitterLocal(gameObject=var "Plat Sprite", x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=30)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    AudioStop(gameObject=Self, fadeTime=0)
    PlayAudioEvent(audioClip=bone_carriage_arrive (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Finished
  }

  state Finished {
    CallMethodProper(gameObject=var "Hero Stick", behaviour="HeroPlatformStick", methodName="set_IsActive", parameters=[false], storeResult=(unset var), EveryFrame=false)
  }

  state Dir {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVE START", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Slammer", isTrue=→"SLAM", isFalse=(none), everyFrame=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=false)
    FloatCompare(float1=var "Self Y", float2=var "Raise Y", tolerance=0, equal=→"U", lessThan=→"U", greaterThan=→"D", everyFrame=false)
    on U → Rise Up
    on D → Rise Down
    on SLAM → Slam Down
  }

  state Rise Down {
    CallMethodProper(gameObject=var "Hero Stick", behaviour="HeroPlatformStick", methodName="set_IsActive", parameters=[true], storeResult=(unset var), EveryFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-10, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Raise Y", tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    ObjectJitterLocal(gameObject=var "Plat Sprite", x=0.05, y=0.05, z=(unset), resetPosOnStateExit=true, limitFps=30)  // disabled
    on END → End
  }

  state Slam Down {
    ActivateGameObject(gameObject=var "Crusher", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Drop Message", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-50, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Raise Y", tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    PlayAudioEvent(audioClip=barrel_plat_lift_dropper_break_loose (sfxstatic_assets_arealibraryareashellwoodareaunderstoreareawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on END → Slam
  }

  state Slam {
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Raise Y", Space=0, EveryFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    AudioStop(gameObject=Self, fadeTime=0)
    PlayAudioEvent(audioClip=understore_impact_break_quite_a_few_things (sfxstatic_assets_areaunderstore.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Finished
  }

  var Raise Y: float = 64.53
  var Slammer: bool = false

  var Self Y: float = 0
  var Tween Pos: vector3 = (0, 0, 0)
  var Plat Sprite: gameObject = Barrel Plat Lift (2)/Plat Sprite
  var Crusher: gameObject = Barrel Plat Lift (2)/Crusher
  var Slam Effect: gameObject = Barrel Plat Lift (2)/Slam Effect
  var Drop Message: gameObject = <null>
  var Hero Stick: gameObject = Barrel Plat Lift (2)/Hero Stick
}
