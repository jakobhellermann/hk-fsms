# Control
# 1 location(s):
#   Shadow_18 / Battle Scene/Boss Summon Crew

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Swamp Muckman A")
    FindNamedChild(gameObject=Self, storeResult=var "Swamp Muckman B")
    FindNamedChild(gameObject=Self, storeResult=var "Swamp Muckman Tall A")
    FindNamedChild(gameObject=Self, storeResult=var "Swamp Muckman Tall B")
    FindNamedChild(gameObject=Self, storeResult=var "Centre")
    on FINISHED → Idle
  }

  state Idle {
    on SUMMON → Choice
  }

  state Choice {
    RandomFloat(min=50, max=57, storeResult=var "Spawn X Left")
    RandomFloat(min=64, max=72, storeResult=var "Spawn X Right")
    SendRandomEventV4(events=[→"SHORT", →"TALL", →"BOTH"], weights=[0.25, 0.25, 0.5], eventMax=[1, 1, 2], missedMax=[3, 3, 2], activeBool=(unset))
    on TALL → Both Tall
    on SHORT → Both Short
    on BOTH → Both Types
  }

  state Both Tall {
    SetPosition2d(gameObject=var "Swamp Muckman Tall A", vector=(unset), x=var "Spawn X Left", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition2d(gameObject=var "Swamp Muckman Tall B", vector=(unset), x=var "Spawn X Right", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Swamp Muckman Tall A"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Swamp Muckman Tall B"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Both Short {
    SetPosition2d(gameObject=var "Swamp Muckman A", vector=(unset), x=var "Spawn X Left", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition2d(gameObject=var "Swamp Muckman B", vector=(unset), x=var "Spawn X Right", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Swamp Muckman A"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Swamp Muckman B"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Both Types {
    SetPosition2d(gameObject=var "Swamp Muckman A", vector=(unset), x=var "Spawn X Left", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPosition2d(gameObject=var "Swamp Muckman Tall A", vector=(unset), x=var "Spawn X Right", y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Swamp Muckman A"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Swamp Muckman Tall A"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state State 1 {
  }

  var Spawn X Left: float = 0
  var Spawn X Right: float = 0
  var Swamp Muckman Tall B: gameObject = <null>
  var Swamp Muckman Tall A: gameObject = <null>
  var Swamp Muckman A: gameObject = <null>
  var Swamp Muckman B: gameObject = <null>
  var Centre: gameObject = <null>
}
