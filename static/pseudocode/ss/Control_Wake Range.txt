# Control
# 1 location(s):
#   Hang_14 / Handmaiden Scene/Wake Range

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Handmaiden Scene")
    FindNamedChild(gameObject=var "Handmaiden Scene", storeResult=var "Handmaiden Attackers")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Wake
    on LAUNDRY TINK → Wake
    on DID TAUNT → Dist Check
    on LAUNDRY BONK → Wake
  }

  state Wake {
    SendEventToRegister(eventName="WAKE AWAY")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake Enemies
  }

  state Wake Enemies {
    SendEventToRegister(eventName="WAKE AWAY")  // disabled
    SendEventByName(eventTarget=GameObject(var "Handmaiden Attackers"), sendEvent="WAKE", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="WAKE")
  }

  state Dist Check {
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=35, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"CANCEL", everyFrame=false)
    on FINISHED → Taunt Pause
    on CANCEL → Idle
  }

  state Taunt Pause {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake
  }

  var Distance: float = 0
  var Handmaiden Scene: gameObject = <null>
  var Handmaiden Attackers: gameObject = <null>
}
