  state Crawl Idle {

  }

  }

    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[blade_spider_antic (sfxstatic_assets_areacoral.bundle)], weights=[1], pitchMin=0.95, pitchMax=1.15, stopPreviousSound=false)  // disabled

    SetScale(gameObject=var "Slash Damager", vector=(unset), x=0.75, y=0.75, z=(unset), everyFrame=false, lateUpdate=false)

    SetVelocity2dBool(gameObject=Self, activeBool=var "No Ground Ahead", vector=(unset), x=0, y=(unset), everyFrame=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SLASH START", delay=0, everyFrame=false)
    on FINISHED → Attack Recover
    on MULTI HIT CONNECT → Multihit
  }

  state Cancel Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Crawl Alert
  }

    CheckPassedTarget(Self=Self, Target=var "Hero", IsActive=true, DefaultFacingRight=false, DistancePast=0, StoreResult=(unset), PassedEvent=→"ALERT", NotPassedEvent=(none), EveryFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)

    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=(none), InRangeDelay=0.2, OutOfRangeEvent=→"UNALERT", OutOfRangeDelay=1, everyFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=var "Grounded", GroundedEvent=(none), NotGroundedEvent=(none), EveryFrame=true)

    FloatTestToBool(float1=var "Gravity", float2=0, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Horizontal", everyFrame=true)

    on SING → Sing
    on ATTACK → Attack Antic
    on UNALERT → Start Idle
    on FALL → Fall
  }

    StopCrawler(Target=Self, WaitForTurn=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)

    GetGravity2dScale(gameObject=Self, storeGravity=var "Gravity", everyFrame=true)
    FloatTestToBool(float1=var "Gravity", float2=0, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Horizontal", everyFrame=true)

    StartCrawler(Target=Self, ScheduleTurn=false)

  }

    StopCrawler(Target=Self, WaitForTurn=false)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=(unset), InRangeEvent=→"ALERT", InRangeDelay=0.2, OutOfRangeEvent=→"UNALERT", OutOfRangeDelay=0, everyFrame=false)

    on ALERT → Face Hero?
  }

  }

    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    SetMeshRenderer(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)

    SetMeshRenderer(gameObject=Self, active=true)
    FindNamedChild(gameObject=Self, storeResult=var "Wake Range")

  }

    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")

  }

  state Attack Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SLASH END", delay=0, everyFrame=false)
