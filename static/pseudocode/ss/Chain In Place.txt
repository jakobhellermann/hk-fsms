// uses template: slab_prisoner_chain
fsm Chain In Place {
  start Init

  state Init {
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Init X", y=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    SetPosition2D(GameObject=Self, Vector=(unset), X=var "Init X", Y=(unset), Space=0, EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    CheckXPosition(gameObject=Self, compareTo=var "Init X", compareToOffset=0, tolerance=0.1, equal=(none), equalBool=(unset), lessThan=→"PULL RIGHT", lessThanBool=(unset), greaterThan=→"PULL LEFT", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on PULL LEFT → Pull Left
    on PULL RIGHT → Pull Right
  }

  state Pull Left {
    AccelerateToX(gameObject=Self, accelerationFactor=2, targetSpeed=-15)
    CheckXPosition(gameObject=Self, compareTo=var "Init X", compareToOffset=0, tolerance=0.1, equal=(none), equalBool=(unset), lessThan=→"FINISHED", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → Idle
  }

  state Pull Right {
    AccelerateToX(gameObject=Self, accelerationFactor=2, targetSpeed=15)
    CheckXPosition(gameObject=Self, compareTo=var "Init X", compareToOffset=0, tolerance=0.1, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"FINISHED", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → Idle
  }

  var Init X: float = 0
}
