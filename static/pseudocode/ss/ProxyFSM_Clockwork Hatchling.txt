# ProxyFSM
# 1 location(s):
#   herodynamic_assets_all.bundle / Clockwork Hatchling

fsm ProxyFSM {
  start Idle
  on HIT LANDED → Hit Landed  // from any state
  on HAZARD RELOAD → Hazard Reload  // from any state
  on BENCHREST → Bench Rest  // from any state
  on BENCHREST END → End Bench  // from any state
  on END FOLLOWERS INSTANT → Hazard Reload  // from any state
  on CLEAR EFFECTS → Hazard Reload 2  // from any state
  on ATTACK BLOCKED → Hit Blocked  // from any state

  state Idle {
  }

  state Hit Landed {
    CallMethodProper(gameObject=Self, behaviour="ClockworkHatchling", methodName="FsmHitLanded", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Damager", behaviour="DamageEnemies", methodName="GetLastTargetDamaged", parameters=[], storeResult=var "Last Target Damaged", EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="ClockworkHatchling", methodName="SetLastTargetDamaged", parameters=[var "Last Target Damaged"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Hazard Reload {
    CallMethodProper(gameObject=Self, behaviour="ClockworkHatchling", methodName="FsmHazardReload", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Bench Rest {
    CallMethodProper(gameObject=Self, behaviour="ClockworkHatchling", methodName="FsmBenchRestStart", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state End Bench {
    CallMethodProper(gameObject=Self, behaviour="ClockworkHatchling", methodName="FsmBenchRestEnd", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Hazard Reload 2 {
    CallMethodProper(gameObject=Self, behaviour="ClockworkHatchling", methodName="ClearEffects", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Hit Blocked {
    CallMethodProper(gameObject=Self, behaviour="ClockworkHatchling", methodName="FsmHitBlocked", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  var Damager: gameObject = Clockwork Hatchling/Damager
  var Last Target Damaged: gameObject = <null>
}
