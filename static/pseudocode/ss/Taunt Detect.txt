# Taunt Detect
# 3 location(s):
#   Bone_East_04c / Scene Control/Slab Jailer Scene/Slab Fly Large Cage
#   Greymoor_05 / Scene Control/Slab Jailer Enemy/Slab Fly Large Cage
#   Shadow_21 / Scene Control/Slab Jailer Scene/Slab Fly Large Cage

fsm Taunt Detect {
  start Init

  state Init {
    AddEventRegister(target=Self, eventName="DID TAUNT")
    on FINISHED → Idle
  }

  state Idle {
    on DID TAUNT → Distance Check
  }

  state Distance Check {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=15, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    on CANCEL → Idle
    on FINISHED → Type Check
  }

  state Type Check {
    CheckIfToolEquipped(Tool=Shakra Ring (dataassets_assets_assets/dataassets/tools/toolitems.bundle), RequiredAmountLeft=(unset), trueEvent=→"RING", falseEvent=(none), storeValue=(unset))
    on RING → Set Taunt Ring
    on FINISHED → Set Taunt
  }

  state Set Taunt {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Heard Challenge", setValue=true, everyFrame=false)
  }

  state Set Taunt Ring {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Heard Ring Challenge", setValue=true, everyFrame=false)
  }

  var Distance: float = 0
}
