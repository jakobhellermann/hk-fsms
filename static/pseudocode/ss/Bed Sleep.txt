# Bed Sleep
# 1 location(s):
#   Belltown_Room_Spare / furnishings/bed/hornet

fsm Bed Sleep {
  start Init

  state Idle {
    on BENCHREST START → Sleep Timer
    on BENCHREST INIT → Start Laying Down
  }

  state Sleep Timer {
    WaitRandom(timeMin=4, timeMax=6, finishEvent=→"FINISHED", realTime=false)
    on BENCHREST ENDING → Idle
    on FINISHED → Lay Down
    on MAP OPENED → Quick Map Pause
  }

  state Lay Down {
    SendEventByNameV2(eventTarget=GameObject(furnishings/bed/RestBench), sendEvent="HERO SIT END", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SetMeshRenderer(gameObject=Self, active=true)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hornet Lay Down", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on BENCHREST ENDING → Get Up
    on FINISHED → Listen Open
  }

  state Get Up {
    SetFsmBool(gameObject=var "Bench", fsmName="Bench Control", variableName="Waiting For Control", setValue=true, everyFrame=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hornet Sit Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Return Control
  }

  state Return Control {
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SetMeshRenderer(gameObject=Self, active=false)
    SetFsmBool(gameObject=var "Bench", fsmName="Bench Control", variableName="Waiting For Control", setValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Idle
  }

  sequence state Start Laying Down {
    SetFsmBool(gameObject=var "Bench", fsmName="Bench Control", variableName="Waiting For Control", setValue=true, everyFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hornet Lay Down", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=99)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWait(Target=Self, ClipName="Hornet Sit Up", AnimationCompleteEvent=(none))
    on FINISHED → Return Control
  }

  state Quick Map Pause {
    on CLOSE QUICK MAP → Sleep Timer
  }

  state Listen Close {
    on CLOSE QUICK MAP → Close Map Anim
    on BENCHREST ENDING → Close Map End
  }

  state Listen Open {
    on MAP OPENED → Open Map Anim
    on BENCHREST ENDING → Get Up
  }

  state Close Map End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hornet Lay Map Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Get Up
  }

  state Open Map Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hornet Lay Map Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on BENCHREST ENDING → Open Map Ending
    on FINISHED → Listen Close
    on CLOSE QUICK MAP → Close Map Anim
  }

  state Open Map Ending {
    SetFsmBool(gameObject=var "Bench", fsmName="Bench Control", variableName="Waiting For Control", setValue=true, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hornet Lay Map Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Close Map End
  }

  state Close Map Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hornet Lay Map Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on BENCHREST ENDING → Set Wait for bench control
    on FINISHED → Listen Open
    on MAP OPENED → Open Map Anim
  }

  state Set Wait for bench control {
    SetFsmBool(gameObject=var "Bench", fsmName="Bench Control", variableName="Waiting For Control", setValue=true, everyFrame=false)
    on FINISHED → Close Map End
  }

  var Bench: gameObject = furnishings/bed/RestBench
}
