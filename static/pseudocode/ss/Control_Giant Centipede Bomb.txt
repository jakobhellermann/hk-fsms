# Control
# 1 location(s):
#   localpoolprefabs_assets_areabellway.bundle / Giant Centipede Bomb

fsm Control {
  start Init
  on PROJECTILE BREAK → Explode  // from any state
  on DEALT DAMAGE → Explode  // from any state

  state Init {
    SetBoolValue(boolVariable=var "Fuse Set", boolValue=false, everyFrame=false)
    RandomFloat(min=1.5, max=2, storeResult=var "Timer")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    ActivateGameObject(gameObject=var "Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Idle", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    FloatOperator(float1=var "Bounce Speed", float2=6, operation=Subtract, storeResult=var "Fling Speed", everyFrame=false)
    FloatOperator(float1=var "Bounce Speed", float2=10, operation=Subtract, storeResult=var "Fling Speed", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fling
  }

  state Air {
    RandomFloat(min=200, max=600, storeResult=var "Spin Amount")
    RandomlyFlipFloat(storeResult=var "Spin Amount")
    Rotate(gameObject=var "Sprite", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spin Amount", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"FLOOR", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=→"WALL R", bottomHitEvent=→"FLOOR", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    FloatAdd(floatVariable=var "Timer", add=-1, everyFrame=true, perSecond=true)
    FloatTestToBool(float1=var "Timer", float2=0, tolerance=0, equalBool=(unset), lessThanBool=var "Time Up", greaterThanBool=(unset), everyFrame=true)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=var "Self Y", space=0, everyFrame=true)
    BoolTest(boolVariable=var "Time Up", isTrue=→"TIME UP", isFalse=(none), everyFrame=true)
    FloatInRange(floatVariable=var "Self X", lowerValue=var "Explode X Min", upperValue=var "Explode X Max", boolVariable=var "In Range X", trueEvent=(none), falseEvent=(none), everyFrame=true)  // disabled
    FloatInRange(floatVariable=var "Self Y", lowerValue=var "Explode Y Min", upperValue=var "Explode Y Max", boolVariable=var "In Range Y", trueEvent=(none), falseEvent=(none), everyFrame=true)  // disabled
    BoolAllTrue(boolVariables=[var "In Range X", var "In Range Y", var "Time Up"], sendEvent=→"TIME UP", storeResult=(unset), everyFrame=true)  // disabled
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Bouncer(24), sendEvent=→"EXPLODE", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on WALL L → 
    on WALL R → 
    on FLOOR → Squash
    on TIME UP → Time Up
    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
    on TINK UP → Tink Up
    on TINK DOWN → Tink Down
    on EXPLODE → Explode
    on HEAVY → Explode
    on DEALT DAMAGE → Explode
  }

  state Explode {
    StopParticleEmittersInChildren(gameObject=var "Pt Antic")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=trobbios_projectile_explode (sfxstatic_assets_shared.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=hornet_got_gooped (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Blast", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Recycle
  }

  state Recycle {
    ActivateGameObject(gameObject=var "Blast", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RecycleSelf()
  }

  state Fling {
    RandomFloat(min=0.85, max=1.15, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Bounce Speed", space=0, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    on FINISHED → Bounce Speed Sign
  }

  state Time Up {
    BoolTest(boolVariable=var "Fuse Set", isTrue=→"EXPLODE", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Fuse Set", boolValue=true, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Idle")
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Bomb Antic")
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetFloatValue(floatVariable=var "Timer", floatValue=0.35, everyFrame=false)
    PlayAudioEvent(audioClip=small_fireball_projectile_explode_antic (sfxstatic_assets_areabone.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on EXPLODE → Explode
    on FINISHED → Air
  }

  state Bounce Speed Sign {
    FloatSignTest(floatValue=var "Bounce Speed", isPositive=→"FINISHED", isNegative=(none), everyFrame=false)
    FloatMultiply(floatVariable=var "Bounce Speed", multiplyBy=-1, everyFrame=false)
    on FINISHED → Air
  }

  state Floor {
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=30, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.5, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Air
  }

  state Squash {
    Wait(time=0.085, finishEvent=→"FINISHED", realTime=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), space=0, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Land", emit=0, resetIfPlaying=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-0.25, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetScale(gameObject=Self, vector=(unset), x=1.25, y=0.75, z=(unset), everyFrame=false, lateUpdate=false)
    PlayAudioEventRandom(audioClips=[goop_impact_2 (sfxstatic_assets_areaslab.bundle), goop_impact_3 (sfxstatic_assets_areaslab.bundle), goop_impact_4 (sfxstatic_assets_areaslab.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[bounce_shroom (sfxstatic_assets_areamemory.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Floor
  }

  state Tink Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=20, everyFrame=false)
    on FINISHED → Tink Effect
  }

  state Tink Effect {
    PlayAudioEventRandom(audioClips=[goop_impact_2 (sfxstatic_assets_areaslab.bundle), goop_impact_3 (sfxstatic_assets_areaslab.bundle), goop_impact_4 (sfxstatic_assets_areaslab.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[bounce_shroom (sfxstatic_assets_areamemory.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Time Up
  }

  state Tink Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-10, everyFrame=false)
    on FINISHED → Tink Effect
  }

  state Tink Right {
    SetVelocity2d(gameObject=Self, vector=(unset), x=8, y=(unset), everyFrame=false)
    on FINISHED → Tink Effect
  }

  state Tink Left {
    SetVelocity2d(gameObject=Self, vector=(unset), x=-8, y=(unset), everyFrame=false)
    on FINISHED → Tink Effect
  }

  state Short Fuse {
    StopParticleEmitter(gameObject=var "Pt Idle")
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Bomb Antic")
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetFloatValue(floatVariable=var "Timer", floatValue=0.1, everyFrame=false)  // disabled
    PlayAudioEvent(audioClip=small_fireball_projectile_explode_antic (sfxstatic_assets_areabone.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.1, finishEvent=(none), realTime=false)
    on FINISHED → Explode
  }

  var Spin Amount: float = 0
  var Bounce Speed: float = 0
  var Timer: float = 0
  var Rotation: float = 0
  var Self Y: float = 0
  var Explode Y Min: float = 16
  var Explode Y Max: float = 21
  var Self X: float = 0
  var Explode X Max: float = 86
  var Explode X Min: float = 62
  var Fling Speed: float = 0
  var Pitch: float = 0
  var X Speed: float = 0
  var In Range X: bool = false
  var Time Up: bool = false
  var In Range Y: bool = false
  var Fuse Set: bool = false
  var Sprite: gameObject = Giant Centipede Bomb/Sprite
  var Pt Idle: gameObject = Giant Centipede Bomb/Pt Idle
  var Pt Antic: gameObject = Giant Centipede Bomb/Pt Antic
  var Blast: gameObject = Giant Centipede Bomb/Blast
  var Pt Land: gameObject = Giant Centipede Bomb/Pt Land
  var Self: gameObject = Giant Centipede Bomb
}
