  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "No_clamber_collider")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Lower")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Rubble")
    on OPEN → Lower Pause
    on ACTIVATE → Activate
  }

  state Activate {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Activate Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
  }

  state Lower Pause {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Lower
  }

  state Lower {
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -46.3, 0), time=3, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "No_clamber_collider", activate=true, recursive=false, resetOnExit=true, everyFrame=false)

    SetFsmBool(gameObject=var "Slug", fsmName="Pray Reaction", variableName="Lift Move", setValue=true, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="JitterSelf", methodName="StartJitter", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    PlayParticleEmitterInState(gameObject=var "Pt Lower")
    PlayParticleEmitter(gameObject=var "Pt Rubble", emit=0, resetIfPlaying=false)
    AudioPlayInState(gameObject=var "Audio Loop ChainPlatLower", volume=1)
    on FINISHED → End
  }

  state End {
    PlayAudioEvent(audioClip=ruins_big_lift_arrive (sfxstatic_assets_areaboneareadocks.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    SetFsmBool(gameObject=var "Slug", fsmName="Pray Reaction", variableName="Lift Move", setValue=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="JitterSelf", methodName="StopJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  var Activate Y: float = -46.3
  var Slug: gameObject = Lift Dropper R/Coal Bucket Plat_multi (5)/Plat (4)/Tar Slug (2)

  var No_clamber_collider: gameObject = <null>
  var Pt Lower: gameObject = <null>

}
