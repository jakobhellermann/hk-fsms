# Control
# 1 location(s):
#   herodynamic_assets_all.bundle / Tool Barb

fsm Control {
  start Init
  on TORNADO → Blow Away  // from any state
  on LAVA → Break  // from any state
  on BREAK HERO PROJECTILE → Damage 1  // from any state

  state Init {
    SetLayer(gameObject=Self, layer=Terrain Detector(14))
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Projectile Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetGameObject(variable=var "Projectile", gameObject=<null>, everyFrame=false)
    ActivateGameObject(gameObject=var "Throw Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    RandomVector2(Start=(-7, -7), End=(7, 7), StoreResult=var "Force Vector", EveryFrame=false)  // disabled
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=var "Force Vector", x=(unset), y=(unset), vector3=(unset), everyFrame=false)  // disabled
    on FINISHED → Poison?
  }

  state Fly {
    PlayAudioEvent(audioClip=hornet_tool_barb_throw (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"TERRAIN", storeCollider=var "", storeForce=(unset))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barb Fly")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"SPRING", storeCollider=var "")  // disabled
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Attack(17), sendEvent=→"SPRING", storeCollider=var "")  // disabled
    on FINISHED → Stick
    on TERRAIN → Stick
    on SPRING → Stick Spring
  }

  state Stick {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Set", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    ActivateGameObject(gameObject=var "Throw Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=hornet_tool_barb_arrive (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.77, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Poison Set", emit=0, resetIfPlaying=false)
    on FINISHED → Wait To Set Attack Layer
  }

  state Recycle {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    RecycleSelf()
  }

  state Ready {
    PlayAudioEvent(audioClip=hornet_tool_barb_arm (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barb Idle")
    GameObjectIsNull(gameObject=var "Projectile", isNull=(none), isNotNull=→"BREAK", storeResult=(unset), everyFrame=false)
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"SPRING", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Enemy Attack(22), sendEvent=→"SPRING", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Attack(17), sendEvent=→"TINK", storeCollider=var "")
    ActivateGameObject(gameObject=var "Projectile Detector", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Trigger2dEventV2(gameObject=var "Projectile Detector", trigger=0, collideTag="", collideLayer=Enemy Attack(22), sendEvent=→"PROJECTILE", storeCollider=var "Projectile")
    Trigger2dEventV2(gameObject=var "Projectile Detector", trigger=0, collideTag="Nail Attack", collideLayer=Attack(17), sendEvent=→"BREAK", storeCollider=var "Projectile")
    Trigger2dEventV2(gameObject=var "Projectile Detector", trigger=0, collideTag="", collideLayer=Attack(17), sendEvent=→"BREAK", storeCollider=var "Projectile")
    on SPRING → Damage 1
    on TINK → Burst Antic Short
    on TINK DOWN → Burst Antic Short
    on BREAK → Burst Antic Short
    on PROJECTILE → Projectile Break
  }

  state Damage 1 {
    ActivateGameObject(gameObject=var "Throw Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Burst", animationTriggerEvent=→"FINISHED", animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=hornet_tool_barb_spikey_attack (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitter(gameObject=var "Pt Poison Burst", emit=0, resetIfPlaying=false)
    Wait(time=0.27, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Damage 2
  }

  state Burst {
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=5)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Radial Burst Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SpawnRandomObjectsRadial(Prefab=var "Spine Object", SpawnPoint=Self, MinCount=6, MaxCount=6, MinAngle=0, MaxAngle=360, AngleErrorMax=20)
    PlayAudioEvent(audioClip=hornet_tool_barb_explode_into_projectiles (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Recycle
  }

  state Burst Antic Short {
    FreezeMoment(FreezeMomentType=EnemyDeath)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Hit Burst Short", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    ActivateGameObject(gameObject=var "Throw Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=hero_parry (herosfxstatic_assets_all.bundle), pitchMin=1.15, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitter(gameObject=var "Pt Poison Burst", emit=0, resetIfPlaying=false)
    on FINISHED → Burst
  }

  state Damage 2 {
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=→"FINISHED")
    Wait(time=0.27, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Damage 3
  }

  state Damage 3 {
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Recycle
  }

  state Stick Spring {
    SetLayer(gameObject=Self, layer=Attack(17))
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Set Q", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetRandomRotation(gameObject=Self, x=false, y=false, z=true)
    PlayAudioEvent(audioClip=hornet_tool_barb_arrive (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Damage 1
  }

  state Projectile Break {
    SendEventByName(eventTarget=GameObject(var "Projectile"), sendEvent="PROJECTILE BREAK", delay=0, everyFrame=false)
    on FINISHED → Damage 1
  }

  state Blow Away {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Barb Fly")
    PlayAudioEvent(audioClip=hornet_tool_barb_arrive (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Throw Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObject(flungObject=Self, speedMin=30, speedMax=30, angleMin=50, angleMax=130)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    on BREAK → Break
  }

  state Break {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Barb Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Nail Terrain Hit Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Recycle
  }

  state Wait To Set Attack Layer {
    SetLayer(gameObject=Self, layer=Attack(17))
    Trigger2dEventV2(gameObject=Self, trigger=0, collideTag="", collideLayer=Attack(17), sendEvent=(none), storeCollider=var "Projectile")
    Tk2dIsPlaying(gameObject=Self, clipName="Barb Set", isPlaying=(unset), isPlayingEvent=(none), isNotPlayingEvent=→"FINISHED", everyframe=true)
    PlayAudioEvent(audioClip=hornet_tool_barb_arm (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Ready
  }

  state Poison? {
    ActivateGameObject(gameObject=var "Pt Poison Set", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Poison Burst", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetGameObject(variable=var "Spine Object", gameObject=Tool Barb Spine, everyFrame=false)
    CheckIfToolEquipped(Tool=Poison Pouch (dataassets_assets_assets/dataassets/tools/toolitems.bundle), RequiredAmountLeft=(unset), trueEvent=(none), falseEvent=→"FINISHED", storeValue=(unset))
    PlayParticleEmitter(gameObject=var "Pt Poison Trail", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Pt Poison Set", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Poison Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetGameObject(variable=var "Spine Object", gameObject=Tool Barb Spine Poison Variant, everyFrame=false)
    on FINISHED → Fly
  }

  var Force Vector: vector2 = (0, 0)
  var Damager: gameObject = Tool Barb/Damager
  var Projectile: gameObject = <null>
  var Projectile Detector: gameObject = Tool Barb/Projectile Detector
  var Pt Poison Burst: gameObject = Tool Barb/Pt Poison Burst
  var Pt Poison Set: gameObject = Tool Barb/Pt Poison Set
  var Pt Poison Trail: gameObject = Tool Barb/Pt Poison Trail
  var Self: gameObject = <null>
  var Spine Object: gameObject = <null>
  var Throw Damager: gameObject = Tool Barb/Throw Damager
}
