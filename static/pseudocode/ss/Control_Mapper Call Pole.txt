# Control
# 1 location(s):
#   Greymoor_08_mapper / Mapper Call Pole

fsm Control {
  start Init
  on MAPPER SPAR START → Inert  // from any state
  on MAPPER SPAR END → Reactivate  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT → Hit
    on DID TAUNT POSHANKA → Poshanka Call
  }

  state Hit {
    PlayAudioEvent(audioClip=shakra_entrance_shing (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    ActivateGameObject(gameObject=var "Ring Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pole Hit")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Event
  }

  state Send Event {
    BoolTest(boolVariable=var "Sent Event", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SendEventToRegister(eventName="MAPPER CALL")
    SetBoolValue(boolVariable=var "Sent Event", boolValue=true, everyFrame=false)
    BoolTest(boolVariable=var "z_Do Not Set Location", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Set Away", boolValue=var "Set In Town", everyFrame=false)
    BoolFlip(boolVariable=var "Set Away")
    SetPlayerDataBool(boolName="mapperAway", value=var "Set Away")
    on FINISHED → Idle
  }

  state Inert {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  state Reactivate {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SetBoolValue(boolVariable=var "Sent Event", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Poshanka Call {
    CheckTrackTriggerCountV2(Target=var "Poshanka Range", Count=0, Test=Equal, EveryFrame=false, SetBool=(unset), SuccessEvent=→"CANCEL", FailEvent=(none))
    on CANCEL → Idle
    on FINISHED → Poshanka Summon
  }

  state Poshanka Summon {
    SendEventToRegister(eventName="MAPPER POSHANKA CALL")
    on FINISHED → Send Event
  }

  var Set In Town: bool = false
  var z_Do Not Set Location: bool = false

  var Sent Event: bool = false
  var Set Away: bool = false
  var Ring Flash: gameObject = Mapper Call Pole/Ring Flash
  var Poshanka Range: gameObject = Mapper Call Pole/Poshanka Range
}
