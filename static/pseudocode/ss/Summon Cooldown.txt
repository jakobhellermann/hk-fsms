# Summon Cooldown
# 2 location(s):
#   Ant_19 / Boss Control/Boss Scene/Bone Flyer Giant
#   Bone_East_08_boss_beastfly / Boss Scene Beastfly/Beastfly States/Active/Bone Flyer Giant

fsm Summon Cooldown {
  start Idle

  state Idle {
    SetBoolValue(boolVariable=var "Summon Cooling Down", boolValue=false, everyFrame=false)
    on SUMMON → Summon Cooldown
  }

  state Summon Cooldown {
    SetBoolValue(boolVariable=var "Summon Cooling Down", boolValue=true, everyFrame=false)
    Wait(time=10, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Summon Cooling Down: bool = false
}
