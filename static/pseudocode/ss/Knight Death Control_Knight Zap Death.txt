# Knight Death Control
# 1 location(s):
#   localpoolprefabs_assets_areaarboriumareacoralshared.bundle / Knight Zap Death

fsm Knight Death Control {
  start Pause
  on HAZARD RESPAWN RESET → Recycle  // from any state
  on HORNET DEATH → Recycle  // from any state

  state Damage {
    ScreenFlash(flashColour=(1, 0.61764705, 0.94198793, 1))
    SpawnObjectFromGlobalPool(gameObject=standard_health_hit (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="HAZARD DAMAGE", delay=0, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(1))
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Hero Damage Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    MatchScaleSign(Target=Self, MatchTo=var "Hero", EveryFrame=false)
    SendEventToRegister(eventName="KNIGHT SPIKE HIT")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Extra Damage
  }

  state Recycle {
    RecycleSelf()
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Damage
  }

  state Extra Damage {
    DamageHeroDirectly(damager=Self, damageAmount=1, spikeHazard=false, sinkHazard=true)
  }

  var Hero Scale: float = 0
  var Self: gameObject = Knight Zap Death
}
