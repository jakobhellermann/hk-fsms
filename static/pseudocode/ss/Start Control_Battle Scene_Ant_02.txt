# Start Control
# 1 location(s):
#   Ant_02 / Battle Scene

fsm Start Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on ANT GUARD WOKE → Wait Til In Start Range
  }

  state Wait Til In Start Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Can Start Range", storeResult=false, sendEvent="START", outOfRangeEvent="", everyFrame=true)
    on START → Start
  }

  state Start {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartBattle())
    FindChild(gameObject=Self, childName="Wave 1/Bone Hunter Throw", storeResult=var "Hunter")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 1
  }

  state State 1 {
    SetFsmBool(gameObject=var "Hunter", fsmName="Control", variableName="Endless Guard Range", setValue=true, everyFrame=false)
  }

  var Hunter: gameObject = <null>
}
