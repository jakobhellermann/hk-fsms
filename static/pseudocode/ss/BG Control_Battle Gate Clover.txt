# BG Control
# 1 location(s):
#   Clover_10 / Boss Scene/Battle Gates/Battle Gate Clover

fsm BG Control {
  start Opened
  on BG QUICK OPEN → Quick Open  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close Pause
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=gate_slam (sharedeffects_assets_all.bundle))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Clover Gate Close", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Close 2
  }

  state Close 2 {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BG OPEN → Open
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=gate_open (sharedeffects_assets_all.bundle), pitchMin=1, pitchMax=1, volume=0.85, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Plat Stream Dust", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Clover Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Finish Open
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Clover Gate Closed")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BG OPEN → Open
    on FINISHED → Double Close
  }

  state Double Close {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Clover Gate Closed")
    on BG OPEN → Open
  }

  state Quick Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Finish Open {
    SetMeshRenderer(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close Pause {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Close 1
  }

  var Start Closed: bool = false

  var Plat Dust: gameObject = Boss Scene/Battle Gates/Battle Gate Clover/Plat Dust
  var Plat Stream Dust: gameObject = Boss Scene/Battle Gates/Battle Gate Clover/Plat Stream Dust
  var Raise Dust: gameObject = <null>
  var Self: gameObject = <null>
  var Slam Effect: gameObject = Boss Scene/Battle Gates/Battle Gate Clover/Close Effect
  var Wall Collider: gameObject = Boss Scene/Battle Gates/Battle Gate Clover/Wall Collider
}
