// uses template: rosary_thief_group
fsm FSM {
  start Init

  state Init {
    AddEventRegister(target=Self, eventName="ALERT THIEVES")
    on ALERT THIEVES → Set Alerted
  }

  state Set Alerted {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  var Activated: bool = false
}
