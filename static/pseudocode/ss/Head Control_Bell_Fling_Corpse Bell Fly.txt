# Head Control
# 1 location(s):
#   enemycorpses_assets_areabell.bundle / Corpse Bell Fly/Fling/Bell

fsm Head Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Spawn Effects
  }

  state Idle {
    on TINK RIGHT → Right
    on TINK LEFT → Left
    on TINK UP → Up
    on TINK DOWN → Down
  }

  state Check Direction {
    GetDamagerDirection(target=var "Damager", storeDirection=var "Attack Direction")
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Left
    on RIGHT → Right
    on UP → Up
    on DOWN → Down
  }

  state Left {
    RandomFloat(min=130, max=140, storeResult=var "Angle")
    on FINISHED → Hit Effects
  }

  state Right {
    RandomFloat(min=40, max=50, storeResult=var "Angle")
    on FINISHED → Hit Effects
  }

  state Spin {
    RandomFloat(min=3, max=7, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    GetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=10, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Speed", everyFrame=false)
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Up {
    RandomFloat(min=85, max=95, storeResult=var "Angle")
    on FINISHED → Hit Effects
  }

  state Down {
    RandomFloat(min=250, max=290, storeResult=var "Angle")
    on FINISHED → Hit Effects
  }

  state Hit Effects {
    PlayAudioEvent(audioClip=bell_fly_bell_long_ver (sfxstatic_assets_areabell.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Spin
  }

  state Spawn Effects {
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Idle
  }

  var Angle: float = 0
  var Attack Direction: float = 0
  var Speed: float = 0
  var Damage Dealt: int = 0
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
