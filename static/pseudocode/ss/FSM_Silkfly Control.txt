# FSM
# 1 location(s):
#   Dust_Maze_09_entrance / Silkfly Control

fsm FSM {
  start Wait

  state Wait {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check Gate
  }

  state Check Gate {
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="right1", equalEvent=(none), notEqualEvent=→"DEACTIVATE", storeResult=(unset), everyFrame=false)
    on DEACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Entry Gate: string = ""
}
