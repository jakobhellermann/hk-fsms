fsm Wave Control {
  start Idle
  on WAVE STOP → Idle  // from any state

  state Idle {
    on WAVE START → Start
  }

  state Start {
    RandomFloat(min=15, max=14, storeResult=var "Top Y")
    RandomFloat(min=9, max=10, storeResult=var "Bot Y")
    SetFloatValue(floatVariable=var "Tween Time", floatValue=40, everyFrame=false)
    FloatDivide(floatVariable=var "Tween Time", divideBy=var "Fly Speed", everyFrame=false)
    RandomFloat(min=0.4, max=0.7, storeResult=var "Tween Time Multiplier")  // disabled
    FloatMultiply(floatVariable=var "Tween Time", multiplyBy=0.2, everyFrame=false)
    FloatCompare(float1=var "Entry Y", float2=12, tolerance=0, equal=→"START LOW", lessThan=→"START LOW", greaterThan=→"START HIGH", everyFrame=false)
    on START HIGH → Tween Down
    on START LOW → Tween Up
  }

  state Tween Down {
    AnimateYPositionTo(GameObject=Self, ToValue=var "Bot Y", localSpace=true, time=var "Tween Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tween Up
  }

  state Tween Up {
    AnimateYPositionTo(GameObject=Self, ToValue=var "Top Y", localSpace=true, time=var "Tween Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tween Down
  }

  var Bot Y: float = 0
  var Entry Y: float = 0
  var Fly Speed: float = 0
  var Top Y: float = 0
  var Tween Time: float = 0
  var Tween Time Multiplier: float = 0
}
