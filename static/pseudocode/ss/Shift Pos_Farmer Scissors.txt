fsm Shift Pos {
  start Pause

  state Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Check {
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="right1", equalEvent=→"SHIFT", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on SHIFT → Shift
  }

  state Shift {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=86.12, z=(unset), space=1, everyFrame=false, lateUpdate=false)
  }

  var Gate: string = ""
}
