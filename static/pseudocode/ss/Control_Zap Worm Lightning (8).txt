    on ATTACK → Antic
  }

  }

  }

  state Explode {

    Wait(time=1, finishEvent=(none), realTime=false)
    PlayAudioEventRandom(audioClips=[lightning_zap (sfxstatic_assets_shared.bundle), lightning_zap_2 (sfxstatic_assets_shared.bundle), lightning_zap_3 (sfxstatic_assets_shared.bundle)], pitchMin=0.85, pitchMax=1.15, volume=0.8, audioPlayerPrefab=Audio Player Actor NoReverb@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

  }

    on FINISHED → Antic
    on EXIT → Idle
    on INACTIVE → Idle
  }

    SetPosition2d(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=var "Translate X Min", max=var "Translate X Max", storeResult=var "Shift")
    Translate(gameObject=Self, vector=(unset), x=var "Shift", y=(unset), z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
