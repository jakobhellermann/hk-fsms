# Control
# 1 location(s):
#   localpoolprefabs_assets_areaant.bundle / Hunter Child Sickle

fsm Control {
  start Init
  on DEALT DAMAGE → Dealt Damage  // from any state

  state Flying {
    AudioPlayInState(gameObject=Self, volume=1)
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=3.2, vector3=(unset), everyFrame=false)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset), contactPoint=var "Contact Point", contactNormal=var "Contact Normal")
    Wait(time=5, finishEvent=→"END", realTime=false)
    ClampSpeed(gameObject=Self, speedMin=30, speedMax=(unset), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Target", collideLayer=(unset), sendEvent=→"TARGET BREAK", storeCollider=var "")
    on END → Break
    on COLLIDE → Thunk
    on TINK RIGHT → Tink R
    on TINK LEFT → Tink L
    on TINK UP → Tink U
    on TINK DOWN L → Tink DL
    on TINK DOWN R → Tink DR
    on PARRIED → Deflect
    on TARGET BREAK → Target Break
    on PROJECTILE BREAK → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Thunk {
    PlayAudioEvent(audioClip=hunter_child_sickle_wall_hit (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCircleCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="SmallShake", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=var "Sickle", active=true)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ConvertNormalToAngle(contactNormal=var "Contact Normal", storeAngle=var "Angle", everyFrame=false)
    SetPosition(gameObject=var "Sickle", vector=var "Contact Point", x=(unset), y=(unset), z=0.0052, space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=50, max=130, storeResult=var "Random")
    FloatAdd(floatVariable=var "Angle", add=var "Random", everyFrame=false, perSecond=false)
    SetRotation(gameObject=var "Sickle", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    Wait(time=6, finishEvent=→"END", realTime=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"DMG", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetCollider(gameObject=var "Sickle", active=true, resetOnExit=false)
    RayCast2dV2(fromGameObject=var "Sickle Ray Pt", fromPosition=(unset), direction=(0, 1), space=1, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"BREAK PROJECTILE", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on END → Break
    on DMG → Damaged
    on BREAK PROJECTILE → Break
  }

  state Break {
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Sickle", active=false)
    SetCollider(gameObject=var "Sickle", active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=4, finishEvent=→"END", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    StopPlayParticleEffectsInChildren(Target=Self)
    on END → Recycle
  }

  state Init {
    SetMeshRenderer(gameObject=var "Sickle", active=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Break Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0.175)
    SetCircleCollider(gameObject=Self, active=true)
    SetCollider(gameObject=var "Sickle", active=false, resetOnExit=false)
    SetDamageHero(Target=Self, Enabled=true)
    FindNamedChild(gameObject=var "Sickle", storeResult=var "Sickle Ray Pt")
    on FINISHED → Flying
  }

  state Damaged {
    FlingObjectsFromGlobalPool(gameObject=Shell Shard 01 (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Shard Point", position=(unset), spawnMin=1, spawnMax=1, speedMin=15, speedMax=15, angleMin=20, angleMax=160, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")  // disabled
    FlingObjectsFromGlobalPool(gameObject=Shell Shard 02 (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Shard Point", position=(unset), spawnMin=1, spawnMax=1, speedMin=15, speedMax=15, angleMin=20, angleMax=160, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")  // disabled
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Break
  }

  state Tink R {
    SetVelocityAsAngle(gameObject=Self, angle=5, speed=45, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tinked {
    SetDamageHero(Target=Self, Enabled=false)
    on FINISHED → Flying
  }

  state Tink L {
    SetVelocityAsAngle(gameObject=Self, angle=175, speed=45, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tink U {
    SetVelocityAsAngle(gameObject=Self, angle=90, speed=45, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tink DL {
    SetVelocityAsAngle(gameObject=Self, angle=225, speed=45, everyFrame=false)
    on FINISHED → Tinked
  }

  state Tink DR {
    SetVelocityAsAngle(gameObject=Self, angle=315, speed=45, everyFrame=false)
    on FINISHED → Tinked
  }

  state Deflect {
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=10, vector3=(unset), everyFrame=false)
    SetDamageHero(Target=Self, Enabled=false)
    on FINISHED → Flying
  }

  state Target Break {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)  // disabled
    Wait(time=0.025, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Dealt Damage {
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Sickle", active=false)
    SetCollider(gameObject=var "Sickle", active=false, resetOnExit=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  var Angle: float = 0
  var Angle Mod: float = 0
  var Random: float = 0
  var Speed: float = 0
  var Contact Normal: vector3 = (0, 0, 0)
  var Contact Point: vector3 = (0, 0, 0)
  var Break Effect: gameObject = Hunter Child Sickle/Sickle/Break Effect
  var Enemy Damager: gameObject = Hunter Child Sickle/Enemy Damager
  var Pt Break: gameObject = Hunter Child Sickle/Sickle/Pt Break
  var Shard Point: gameObject = Hunter Child Sickle/Sickle/Shard Point
  var Sickle: gameObject = Hunter Child Sickle/Sickle
  var Thunk Effect: gameObject = Hunter Child Sickle/Sickle/Thunk Effect
  var Sickle Ray Pt: gameObject = <null>
}
