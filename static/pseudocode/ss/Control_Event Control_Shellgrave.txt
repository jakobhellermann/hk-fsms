# Control
# 1 location(s):
#   Shellgrave / Event Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Pilgrim Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Plinney Rescue Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Rosary Pilgrim Scene")
    ActivateGameObject(gameObject=var "Plinney Rescue Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rosary Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Plinney?
  }

  state Plinney? {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=(none), isFalse=→"FINISHED")
    PlayerDataBoolTest(boolName="savedPlinney", isTrue=→"NONE", isFalse=(none))
    PlayerDataBoolTest(boolName="sawPlinneyLeft", isTrue=→"PLINNEY", isFalse=→"NONE")
    on PLINNEY → Plinney
    on FINISHED → Rosary?
    on NONE → None
  }

  state Plinney {
    ActivateGameObject(gameObject=var "Plinney Rescue Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rosary Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Rosary? {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"PILGRIMS", isFalse=(none))
    PlayerDataBoolTest(boolName="defeatedShellwoodRosaryPilgrim", isTrue=→"PILGRIMS", isFalse=→"ROSARY PILGRIM")
    on ROSARY PILGRIM → Rosary Pilgrim
    on PILGRIMS → Pilgrims
  }

  state Rosary Pilgrim {
    ActivateGameObject(gameObject=var "Plinney Rescue Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rosary Pilgrim Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pilgrims {
    ActivateGameObject(gameObject=var "Plinney Rescue Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rosary Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state None {
    ActivateGameObject(gameObject=var "Plinney Rescue Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rosary Pilgrim Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Nail Upgrades: int = 0
  var Oils Held: int = 0
  var Pilgrim Scene: gameObject = <null>
  var Plinney Rescue Scene: gameObject = <null>
  var Rosary Pilgrim Scene: gameObject = <null>
}
