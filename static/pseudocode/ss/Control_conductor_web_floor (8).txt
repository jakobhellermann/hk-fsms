fsm Control {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    FloatCompare(float1=var "Z Pos", float2=-1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Jitter", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    CheckTrackTriggerCount(target=Self, count=0, test=MoreThan, everyFrame=true, successEvent=→"ENTER")
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StopJitter())
    on ENTER → Jitter
  }

  state Jitter {
    CheckTrackTriggerCount(target=Self, count=0, test=LessThanOrEqual, everyFrame=true, successEvent=→"EXIT")
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    PlayParticleEmitterInState(gameObject=var "Pt Jitter")
    on EXIT → Idle
  }

  var Z Pos: float = 0
  var Pt Jitter: gameObject = conductor_web_floor (8)/Pt Jitter
}
