# Control
# 1 location(s):
#   localpoolprefabs_assets_areashellwood.bundle / Fisher Projectile

fsm Control {
  start Init

  state Init {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pole Throw")
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    RandomFloat(min=0.0053, max=0.0059, storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    SetDamageEnemyAmount(target=Self, damageDealt=15)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Fly
  }

  state Fly {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"HIT TERRAIN", storeCollider=var "")  // disabled
    Collision2dEvent(gameObject=var "Terrain Detector", collision=0, collideTag="", sendEvent=→"HIT TERRAIN", storeCollider=var "", storeForce=(unset))
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer=(unset), sendEvent=→"WATER", storeCollider=var "Surface")
    FaceAngle(gameObject=Self, angleOffset=0, angleOffsetIfMirrored=180, otherGameObject=<null>, everyFrame=true)
    on HIT TERRAIN → Stick
    on PARRIED → Deflect
    on WATER → Splash
    on PROJECTILE BREAK → Break
  }

  state Stick {
    PlayAudioEvent(audioClip=fisher_projectile_impact (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Pole Thunk")
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=4, finishEvent=→"WAIT", realTime=false)
    SetDamageEnemyAmount(target=Self, damageDealt=0)
    on WAIT → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Recycle
  }

  state Deflect {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    RandomFloat(min=5, max=10, storeResult=var "Angle Mod")
    RandomlyFlipFloat(storeResult=var "Angle Mod")
    FloatAdd(floatVariable=var "Angle", add=var "Angle Mod", everyFrame=false, perSecond=false)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    GetVelocity2d(gameObject=Self, vector=var "Velocity", x=(unset), y=(unset), space=0, everyFrame=false)
    Vector2Add(vector2Variable=var "Velocity", addVector=(0, 10), everyFrame=false, perSecond=false)
    SetVelocity2d(gameObject=Self, vector=var "Velocity", x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Fly
  }

  state Splash {
    SpawnObjectFromGlobalPool(gameObject=Lil Splash (localpoolprefabs_assets_water.bundle), spawnPoint=var "Thunk Effect", position=(unset), rotation=(unset), storeObject=var "Splash")  // disabled
    GetPosition(gameObject=var "Surface", vector=(unset), x=(unset), y=var "Surface Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Splash", vector=(unset), x=(unset), y=var "Surface Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Fly
  }

  var Angle: float = 0
  var Angle Mod: float = 0
  var Speed: float = 0
  var Surface Y: float = 0
  var Z Pos: float = 0
  var Velocity: vector2 = (0, 0)
  var Break Effect: gameObject = Fisher Projectile/Break Effect
  var Pt Break: gameObject = Fisher Projectile/Pt Break
  var Splash: gameObject = <null>
  var Surface: gameObject = <null>
  var Thunk Effect: gameObject = Fisher Projectile/Thunk Effect
  var Terrain Detector: gameObject = Fisher Projectile/Terrain Detector
}
