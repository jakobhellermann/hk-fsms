# Pin Thunk Tracker
# 1 location(s):
#   Bone_12 / Black Thread States/Normal World/Pin Gallery States/Here/Pin Challenge

fsm Pin Thunk Tracker {
  start Idle

  state Idle {
    SetIntValue(intVariable=var "Pin Fails", intValue=0, everyFrame=false)
    on PIN CHALLENGE START → In Challenge
  }

  state In Challenge {
    on PIN CHALLENGE END → Idle
    on TOOL PIN THUNKED → Pin Failed
  }

  state Pin Failed {
    IntAdd(intVariable=var "Pin Fails", add=1, everyFrame=false)
    FormatString(format="PIN CHALLENGE FAIL {0}", variables=[var "Pin Fails"], storeResult=var "Event", everyFrame=false)
    SendEventToRegister(eventName=var "Event")
    SendEventToRegister(eventName="LADYBUG MISS")
    IntCompare(integer1=var "Pin Fails", integer2=3, equal=→"FAIL", lessThan=(none), greaterThan=→"FAIL", everyFrame=false)
    PlayerDataBoolTest(boolName="PinGalleryLastChallengeOpen", isTrue=(none), isFalse=→"FINISHED")
    IntCompare(integer1=var "Pin Fails", integer2=1, equal=→"FAIL", lessThan=(none), greaterThan=→"FAIL", everyFrame=false)
    on FINISHED → In Challenge
    on FAIL → Fail
  }

  state Fail {
    SendEventToRegister(eventName="PIN CHALLENGE FAIL")
    on FINISHED → Idle
  }

  var Pin Fails: int = 0
  var Event: string = ""
}
