    FindNamedChild(gameObject=var "Corpse Parent", storeResult=var "Pull Crack")
    FindNamedChild(gameObject=var "Corpse Parent", storeResult=var "Blood Burst")
    ActivateGameObject(gameObject=var "Hornet Dummy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hornet Leap", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetScale(gameObject=var "Corpse Parent", vector=(unset), xScale=var "Parent X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Parent X Scale", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)

    on FINISHED → Idle
  }

  state Leap Up {
    PlayAudioEvent(audioClip=hornet_jump_ss (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    on FINISHED → Grab Antic
  }

  state Grab Antic {
    PlayRandomAudioClipTable(Table=Attack Heavy Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    ActivateGameObject(gameObject=var "Hornet Leap", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hornet Dummy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hornet Dummy", clipName="Heart Grab", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Hornet Dummy", xOffset=0, yOffset=0, everyFrame=true)
    on FINISHED → Grab
  }

  state Grab {

    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Blood Stream", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Blood Particle", emit=0, resetIfPlaying=false)
    on FINISHED → Grab Idle
  }

  state Pull Out {
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1.15, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    PlayParticleEmitter(gameObject=var "Blood Burst", emit=0, resetIfPlaying=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Tk2dPlayAnimation(gameObject=var "Corpse Parent", animLibName="", clipName="Death Collapse")
    Tk2dPlayAnimation(gameObject=var "Hornet Dummy", animLibName="", clipName="Pull")
    SetParent(gameObject=var "Hornet Dummy", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetIsKinematic2d(gameObject=var "Hornet Dummy", isKinematic=false)
    SetVelocityByScale(gameObject=var "Hornet Dummy", speed=30, ySpeed=5, everyFrame=false)
    DecelerateXY(gameObject=var "Hornet Dummy", decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)

    RayCast2dV2(fromGameObject=var "Hornet Dummy", fromPosition=(unset), direction=(0, -1), space=0, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"LAND", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)

  }

    ActivateGameObject(gameObject=var "Hornet Dummy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetMeshRenderer(gameObject=var "Hero", active=true)  // disabled
    SetIsKinematic2d(gameObject=var "Hero", isKinematic=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())  // disabled
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SendEventToRegister(eventName="BOSS DEFEAT")
  }

  state Hornet Land {
    PlayAudioEvent(audioClip=hornet_ground_land (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    Tk2dPlayAnimation(gameObject=var "Hornet Dummy", animLibName="", clipName="Land")
    Wait(time=1, finishEvent=→"NEXT", realTime=false)
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Hornet Dummy", xOffset=0, yOffset=0, everyFrame=true)
    on NEXT → Seth Check
  }

  state Get Item {
    SetPlayerDataVariable(VariableName="defeatedAntQueen", SetValue=true)
    AwardAchievementProgress(Key="DEFEATED_HUNTER_QUEEN", CurrentValue=(unset), MaxValue=(unset))

    Wait(time=1.7, finishEvent=→"NEXT", realTime=false)
    on NEXT → End
  }

  state Hornet End {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hornet Dummy", clipName="End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → End
  }

  state Grab Idle {
    ListenForAttack(eventTarget=Self, wasPressed=→"YANK", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    on YANK → Yank
  }

  state Yank {
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    IntAdd(intVariable=var "Heart Yanks", add=-1, everyFrame=false)
    IntCompare(integer1=var "Heart Yanks", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)

    Tk2dPlayAnimation(gameObject=var "Hornet Dummy", animLibName="", clipName="Yank")

    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on END → Pull Out
    on FINISHED → Grab Idle
  }

  state Seth Check {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="SethTravelCheck", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Get Item
  }

  var Parent X Scale: float = 0

  var Self: gameObject = Corpse Hunter Queen/NPC
