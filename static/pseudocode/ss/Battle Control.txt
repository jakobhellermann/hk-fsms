# Battle Control
# 1 location(s):
#   Shellwood_18 / Boss Scene Parent/Boss Scene

fsm Battle Control {
  start Pause

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Battle Gates")
    FindNamedChild(gameObject=Self, storeResult=var "Splinter Queen")
    FindNamedChild(gameObject=Self, storeResult=var "Battle Start Range")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Boss")
    PlayerDataBoolTest(boolName="defeatedSplinterQueen", isTrue=→"DEFEATED", isFalse=(none))
    on FINISHED → Idle
    on DEFEATED → Defeated
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Battle Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Battle Start
    on FIRST SPIKES → Start Tension
  }

  state Battle Start {
    ActivateGameObject(gameObject=var "CamLock Boss", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Battle Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Splinter Queen"), sendEvent="BATTLE START", delay=0.5, everyFrame=false)
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=4)  // disabled
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=3)
    on BOSS DEFEAT → Defeat Pause
  }

  state Defeat Pause {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    PlayAudioEvent(audioClip=shellwood_twig_wall_break (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventToRegister(eventName="GATE SPIKES DESTROY")
    SendEventToRegisterDelay(EventName="GATE SPIKES REMOVE", delay=0.25, ExcludeTarget=Self)
    ActivateGameObjectDelay(gameObject=var "CamLock Boss", activate=false, resetOnExit=false, delay=0.5)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Defeated {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Start Tension {
    Trigger2dEvent(gameObject=var "Battle Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    ApplyMusicCue(musicCue=Creepy Boss Ambient (audiocuesdynamic_assets_areabellareashellwoodareaward.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=3)
    SendEventToRegister(eventName="TENSION STOP")
    on ENTER → Battle Start
  }

  var Battle Gates: gameObject = <null>
  var Splinter Queen: gameObject = <null>
  var Battle Start Range: gameObject = <null>
  var CamLock Boss: gameObject = <null>
}
