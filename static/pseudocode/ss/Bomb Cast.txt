  }

    SetPosition(gameObject=var "Bomb", vector=(unset), x=var "Cast X L", y=var "Cast Y", z=var "Cast Z", space=0, everyFrame=false, lateUpdate=false)
    IntAdd(intVariable=var "Bomb Pairs", add=-1, everyFrame=false)
    Wait(time=var "Bomb Pause", finishEvent=→"FINISHED", realTime=false)
    on END → Idle
    on FINISHED → Bomb Pairs
