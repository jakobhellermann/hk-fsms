# Control

  start Init

  state Init {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetCircleCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    SetIntValue(intVariable=var "Bounces", intValue=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Spark", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetFloatValue(floatVariable=var "Lifetime", floatValue=6, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFloatValue(floatVariable=var "Fire Timer", floatValue=1.5, everyFrame=false)

    ActivateGameObject(gameObject=var "Antic Ring", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Antic Pusher", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Warning Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Air
  }

  state Recycle {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    RecycleSelf()
  }

  state Air {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)

    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=Attack(17), sendEvent=→"NAIL", storeCollider=var "Nail Object")  // disabled
    FloatAdd(floatVariable=var "Lifetime", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Lifetime", float2=0, tolerance=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer=(unset), sendEvent=→"WATER", storeCollider=var "")
    CheckCollisionSide(collidingObject=var "Terrain Box", topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"ROOF", rightHitEvent=→"WALL R", bottomHitEvent=→"GROUND", leftHitEvent=→"WALL L", otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)

    SetFloatSign(floatValue=var "Sprite Spin", setPositive=true, everyFrame=true)
    FloatMultiply(floatVariable=var "Sprite Spin", multiplyBy=120, everyFrame=true)
    Rotate(gameObject=var "Ball Sprite", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Sprite Spin", space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    FloatAdd(floatVariable=var "Fire Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Fire Timer", float2=0, tolerance=0, equal=→"FIRE", lessThan=→"FIRE", greaterThan=(none), everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXPLODE", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Explosion", collideLayer=Attack(17), sendEvent=→"COMBO EXPLODE", storeCollider=var "")
    on WALL L → Wall L
    on WALL R → Wall R
    on GROUND → Ground
    on ROOF → Roof
    on NAIL → Nail Hit
    on WATER → Explode
    on FIRE → Explode Antic
    on EXPLODE → Check Parry
    on COMBO EXPLODE → Combo Explode
    on PROJECTILE BREAK → Explode
  }

  state Ground {
    IntCompare(integer1=var "Bounces", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=18.8, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetPosition(gameObject=var "Pt Bounce", vector=(unset), x=0, y=-0.5, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Bounce
  }

  state Break {
    StopParticleEmitter(gameObject=var "Pt Dust")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Spark", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    PlayAudioEvent(audioClip=small_bell_projectile_break (sfxstatic_assets_shared.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Bounce {
    PlayAudioEventRandom(audioClips=[small_bell_projectile_bounce (sfxstatic_assets_shared.bundle), small_bell_projectile_bounce_2 (sfxstatic_assets_shared.bundle), small_bell_projectile_bounce_3 (sfxstatic_assets_shared.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    PlayParticleEmitter(gameObject=var "Pt Bounce", emit=0, resetIfPlaying=false)
    on FINISHED → Air
  }

  state Wall L {
    IntCompare(integer1=var "Bounces", integer2=-1, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)  // disabled

    Translate(gameObject=Self, vector=(unset), x=0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)

    on FINISHED → Bounce
  }

  state Wall R {
    IntCompare(integer1=var "Bounces", integer2=-1, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=-10, y=(unset), everyFrame=false)

  }

    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)

  }

    PlayAudioEvent(audioClip=projectile_zap_antic (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    SetGravity2dScaleV2(gameObject=Self, gravityScale=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Antic Ring", activate=true, recursive=false, resetOnExit=false, everyFrame=false)

  state Combo Explode {
    Tk2dPlayAnimation(gameObject=var "Ball Sprite", animLibName="", clipName="Ball Antic")
    SetGravity2dScaleV2(gameObject=Self, gravityScale=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Antic Ring", activate=true, recursive=false, resetOnExit=false, everyFrame=false)

  }

  state Check Parry {
    HeroControllerMethods(method=IsParryingActive, parameters=[], everyFrame=false, storeValue=(unset var), isTrue=→"CROSS STITCH", isFalse=(none))
    on FINISHED → Explode
    on CROSS STITCH → Parried
  }

  }
