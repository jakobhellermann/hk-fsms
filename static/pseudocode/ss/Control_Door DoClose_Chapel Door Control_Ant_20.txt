fsm Control {
  start Pause

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FindNamedChild(gameObject=Self, storeResult=var "Pt LandDust")
    FindNamedChild(gameObject=Self, storeResult=var "black_fader_moon")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "gate_glows Do Close")
    ActivateGameObject(gameObject=var "gate_glows Do Close", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Close
  }

  state Close {
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -9, 0), time=0.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "gate_glows Do Close", ToAlpha=0, FadeTime=0.5)  // disabled
    PlayAudioEvent(audioClip=weaver_gate_close_start (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -11), SpawnedPlayerRef=var "")
    on FINISHED → Slam
  }

  state Slam {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventToRegister(eventName="CHAPEL DOOR SLAM")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt LandDust", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "black_fader_moon", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "gate_glows Do Close", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    PlayAudioEvent(audioClip=song_gate_opened (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -11), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=weaver_gate_close_end (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -11), SpawnedPlayerRef=var "")
    on FINISHED → Save
  }

  state Save {
    SaveGame()
  }

  var Pt LandDust: gameObject = <null>
  var black_fader_moon: gameObject = <null>
  var gate_glows Do Close: gameObject = <null>
  var Parent: gameObject = <null>
}
