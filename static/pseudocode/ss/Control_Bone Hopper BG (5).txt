# Control
# 6 location(s):
#   Bone_East_24 / Bone Hopper FG Group/Bone Hopper BG (5)
#   Bone_East_24 / Bone Hopper FG Group/Bone Hopper BG (4)
#   Bone_East_24 / Bone Hopper FG Group/Bone Hopper BG (2)
#   Bone_East_24 / Bone Hopper FG Group/Bone Hopper BG
#   Bone_East_24 / Bone Hopper FG Group/Bone Hopper BG (1)
#   Bone_East_24 / Bone Hopper FG Group/Bone Hopper BG (3)

fsm Control {
  start Init

  state Spawn {
    SetMeshRenderer(gameObject=Self, active=true)
    RandomFloat(min=178, max=183, storeResult=var "Start X")
    RandomFloat(min=var "Z Min", max=var "Z Max", storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Ground Y", z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Spawn Jump
  }

  state Spawn Jump {
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=20, z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    SetVelocity2d(gameObject=Self, vector=(unset), x=-9, y=18, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Rising
  }

  state Rising {
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Velocity Y", float2=0, tolerance=0, equal=(none), lessThan=→"FALLING", greaterThan=(none), everyFrame=true)
    on FALLING → Falling
  }

  state Falling {
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAND → Land Sfx
  }

  state Land {
    CheckXPosition(gameObject=Self, compareTo=var "End X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"END", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Jump Sfx
    on END → End Fall
  }

  state Jump Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Jump Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
  }

  state Jump {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    RandomFloat(min=12, max=18, storeResult=var "Jump Y")
    SetVelocity2d(gameObject=Self, vector=(unset), x=-9, y=var "Jump Y", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Rising
  }

  state End Fall {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Ready
  }

  state Ready {
    SetMeshRenderer(gameObject=Self, active=false)
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on SPAWN → Spawn
    on MANUAL SPAWN → Manual Spawn
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Ready
  }

  state Manual Spawn {
    SetMeshRenderer(gameObject=Self, active=true)
    RandomFloat(min=var "Z Min", max=var "Z Max", storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Jump Sfx
  }

  state Jump Sfx {
    BoolTest(boolVariable=var "Play Audio", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayAudioEvent(audioClip=bone_hopper_jump (sfxstatic_assets_areawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Jump
  }

  state Land Sfx {
    BoolTest(boolVariable=var "Play Audio", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayAudioEventRandom(audioClips=[hornet_land_grass_1 (sfxstatic_assets_areacoral.bundle), hornet_land_grass_2 (sfxstatic_assets_areawilds.bundle), hornet_land_grass_3 (sfxstatic_assets_areacoral.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Land
  }

  var Z Min: float = -16
  var Z Max: float = -8
  var Play Audio: bool = true

  var Start X: float = 0
  var Velocity Y: float = 0
  var Ground Y: float = 24.2
  var End X: float = 80
  var Z Pos: float = 0
  var Jump Y: float = 0
  var Parent: gameObject = <null>
}
