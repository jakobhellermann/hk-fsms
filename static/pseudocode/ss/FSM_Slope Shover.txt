# FSM
# 1 location(s):
#   Memory_First_Sinner / Slope Shover

fsm FSM {
  start State 1

  state State 1 {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"SHOVE", storeCollider=var "")
    on SHOVE → State 2
  }

  state State 2 {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="AffectedByGravity", parameters=[true], storeResult=(unset var), EveryFrame=false)
    Translate(gameObject=var "Hero", vector=(unset), x=-0.3, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → State 1
  }
}
