# Game Specific Control
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Flea Game - Juggling

fsm Game Specific Control {
  start Init
  on GAME END PLAY → Reset Game  // from any state
  on GAME FAILED → Game Failed  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Bellfleas")
    FindNamedChild(gameObject=Self, storeResult=var "Bellfleas Quickened")
    FindNamedChild(gameObject=Self, storeResult=var "Grishkin JuggleGame")
    FindNamedChild(gameObject=Self, storeResult=var "Varga JuggleGame")
    FindNamedChild(gameObject=Self, storeResult=var "Bellfleas Giant")
    FindNamedChild(gameObject=var "Bellfleas Giant", storeResult=var "Bellflea Juggler Giant")
    on FINISHED → Inactive
  }

  state Inactive {
    on GAME BEGIN PLAY → Start Pause
    on PREPARE FLEA GAME → Prepare Game
  }

  state Start Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Quad Bounce Time", floatValue=3, everyFrame=false)
    SetIntValue(intVariable=var "Sets Completed", intValue=0, everyFrame=false)
    SetIntValue(intVariable=var "Strikes", intValue=0, everyFrame=false)
    SetIntValue(intVariable=var "Grishkin Appearances", intValue=0, everyFrame=false)
    SetIntValue(intVariable=var "Tinks Until Set Complete", intValue=0, everyFrame=false)
    SetParent(gameObject=var "Bellflea Juggler Giant", parent=var "Bellfleas Giant", resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Bellflea Juggler Giant", parent=var "Bellfleas Giant", resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Stage Check
  }

  state Single {
    SetIntValue(intVariable=var "Fleas To Spawn", intValue=1, everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2, everyFrame=false)
    on FINISHED → Set Entry Pause
  }

  state Set In Progress {
    SetIntValue(intVariable=var "Special Set Type", intValue=0, everyFrame=false)
    IntCompare(integer1=var "Tinks Until Set Complete", integer2=1, equal=→"END SET", lessThan=→"END SET", greaterThan=(none), everyFrame=true)
    on END SET → Next Set Pause
    on FLEA TINKED → Decrement Set Tinks
    on FLEA FAIL → Strike
  }

  state Next Set Pause {
    IntAdd(intVariable=var "Sets Completed", add=1, everyFrame=false)
    SetBoolValue(boolVariable=var "Next Set Is Special", boolValue=false, everyFrame=false)
    on FINISHED → Add Variants?
  }

  state Double {
    SetIntValue(intVariable=var "Fleas To Spawn", intValue=2, everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.5, everyFrame=false)  // disabled
    on FINISHED → Set Entry Pause
  }

  state Triple {
    SetIntValue(intVariable=var "Fleas To Spawn", intValue=3, everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=3, everyFrame=false)  // disabled
    on FINISHED → Set Entry Pause
  }

  state Quad {
    SetIntValue(intVariable=var "Fleas To Spawn", intValue=4, everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=3, everyFrame=false)  // disabled
    on FINISHED → Set Entry Pause
  }

  state Quick Flea {
    SetIntValue(intVariable=var "Special Set Type", intValue=1, everyFrame=false)
    GetRandomChild(gameObject=var "Bellfleas Quickened", storeResult=var "Next Flea")
    SetFsmFloat(gameObject=var "Next Flea", fsmName="Control", variableName="Bounce Time", setValue=1, everyFrame=false)
    SetFsmBool(gameObject=var "Next Flea", fsmName="Control", variableName="Quickened", setValue=true, everyFrame=false)
    SetFsmFloat(gameObject=var "Next Flea", fsmName="Control", variableName="Entry Pause", setValue=1.25, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Next Flea"), sendEvent="ENTER", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Tinks Until Set Complete", add=3, everyFrame=false)
    on FINISHED → Set In Progress
  }

  state Reset Game {
    SendEventToRegister(eventName="RESET FLEA GAMES")
    SendEventToRegister(eventName="JUGGLE GAME END")
    on FINISHED → Inactive
  }

  state Grishkin {
    SetIntValue(intVariable=var "Special Set Type", intValue=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Grishkin JuggleGame"), sendEvent="ENTER", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Tinks Until Set Complete", add=4, everyFrame=false)
    IntAdd(intVariable=var "Grishkin Appearances", add=1, everyFrame=false)
    on FINISHED → Set In Progress
  }

  state Spawn Flea {
    GetRandomChild(gameObject=var "Bellfleas", storeResult=var "Next Flea")
    SetFsmFloat(gameObject=var "Next Flea", fsmName="Control", variableName="Bounce Time", setValue=var "Bounce Time", everyFrame=false)
    SetFsmFloat(gameObject=var "Next Flea", fsmName="Control", variableName="Entry Pause", setValue=var "Entry Pause", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Next Flea"), sendEvent="ENTER", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Next Flea", fsmName="Control", variableName="Enter Left", setValue=var "Spawn Left", everyFrame=false)
    BoolFlip(boolVariable=var "Spawn Left")
    IntAdd(intVariable=var "Fleas To Spawn", add=-1, everyFrame=false)
    IntAdd(intVariable=var "Tinks Until Set Complete", add=3, everyFrame=false)
    FloatAdd(floatVariable=var "Entry Pause", add=var "Entry Pause Increment", everyFrame=false, perSecond=false)
    IntCompare(integer1=var "Fleas To Spawn", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"REPEAT", everyFrame=false)
    on REPEAT → Spawn Flea
    on END → Set Score Til Complete
  }

  state Set Score Til Complete {
    IntOperator(integer1=var "Tinks Until Set Complete", integer2=3, operation=Multiply, storeResult=var "Tinks Until Set Complete", everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "Next Set Is Special", boolValue=false, everyFrame=false)
    on FINISHED → Set In Progress
  }

  state Set Entry Pause {
    SetFloatValue(floatVariable=var "Entry Pause", floatValue=0, everyFrame=false)
    ConvertIntToFloat(intVariable=var "Fleas To Spawn", floatVariable=var "Fleas To Spawn Float", everyFrame=false)
    FloatOperator(float1=var "Bounce Time", float2=var "Fleas To Spawn Float", operation=Divide, storeResult=var "Entry Pause Increment", everyFrame=false)
    on FINISHED → Spawn Flea
  }

  state Stage Check {
    RandomBool(storeResult=var "Spawn Left")
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=1.75, everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"SPECIAL", delay=0, everyFrame=false)  // disabled
    IntCompare(integer1=var "Sets Completed", integer2=0, equal=→"SINGLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.25, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=1, equal=→"DOUBLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.1, everyFrame=false)
    SetBoolValue(boolVariable=var "Next Set Is Special", boolValue=true, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=2, equal=→"DOUBLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=3, equal=→"SPECIAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=1.95, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=4, equal=→"DOUBLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=1.85, everyFrame=false)
    SetBoolValue(boolVariable=var "Next Set Is Special", boolValue=true, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=5, equal=→"DOUBLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=6, equal=→"SPECIAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.4, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=7, equal=→"TRIPLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.3, everyFrame=false)
    SetBoolValue(boolVariable=var "Next Set Is Special", boolValue=true, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=8, equal=→"TRIPLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=9, equal=→"SPECIAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.2, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=10, equal=→"TRIPLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.15, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=11, equal=→"TRIPLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=2.1, everyFrame=false)
    SetBoolValue(boolVariable=var "Next Set Is Special", boolValue=true, everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=12, equal=→"TRIPLE", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=13, equal=→"SPECIAL", lessThan=(none), greaterThan=(none), everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=var "Quad Bounce Time", everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=14, equal=→"QUAD", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendRandomEventV4(events=[→"SPECIAL", (none)], weights=[0.3, 0.7], eventMax=[1, 2], missedMax=[2, 1], activeBool=(unset))
    FloatAdd(floatVariable=var "Quad Bounce Time", add=-0.125, everyFrame=false, perSecond=false)
    FloatClamp(floatVariable=var "Quad Bounce Time", minValue=2, maxValue=4, everyFrame=false)
    SetFloatValue(floatVariable=var "Bounce Time", floatValue=var "Quad Bounce Time", everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"QUAD", delay=0, everyFrame=false)
    on SINGLE → Single
    on DOUBLE → Double
    on TRIPLE → Triple
    on QUAD → Quad
    on SPECIAL → Special
  }

  state Special {
    SendEvent(eventTarget=Self, sendEvent=→"TRAVELLER", delay=0, everyFrame=false)  // disabled
    SendRandomEventV4(events=[→"QUICK FLEA", →"TRAVELLER"], weights=[1, 1], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on QUICK FLEA → Quick Flea
    on TRAVELLER → Traveller Type
  }

  state Decrement Set Tinks {
    IntAdd(intVariable=var "Tinks Until Set Complete", add=-1, everyFrame=false)
    on FINISHED → Set In Progress
  }

  state TEST top difficulty {
    SetIntValue(intVariable=var "Sets Completed", intValue=15, everyFrame=false)
    SetFloatValue(floatVariable=var "Quad Bounce Time", floatValue=2.2, everyFrame=false)  // disabled
    on FINISHED → Stage Check
  }

  state Game End {
    SendEventToRegister(eventName="JUGGLE GAME END")
  }

  state Strike {
    IntAdd(intVariable=var "Strikes", add=1, everyFrame=false)
    FormatString(format="JUGGLE GAME FAIL {0}", variables=[var "Strikes"], storeResult=var "Event", everyFrame=false)
    SendEventToRegister(eventName=var "Event")
    IntSwitch(intVariable=var "Special Set Type", compareTo=[1, 2], sendEvent=[→"END SET", →"END SET"], everyFrame=false)  // disabled
    IntCompare(integer1=var "Strikes", integer2=3, equal=→"END", lessThan=→"FINISHED", greaterThan=→"END", everyFrame=false)
    on FINISHED → Set In Progress
    on END → End Game
  }

  state End Game {
    SendEventToRegister(eventName="GAME END")
    on FINISHED → Game End
  }

  state Traveller Type {
    BoolTest(boolVariable=var "Varga Active", isTrue=(none), isFalse=→"GRISHKIN", everyFrame=false)  // disabled
    IntCompare(integer1=var "Grishkin Appearances", integer2=3, equal=→"VARGA", lessThan=→"GRISHKIN", greaterThan=(none), everyFrame=false)
    SendRandomEventV4(events=[→"GRISHKIN", →"VARGA"], weights=[1, 1], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on GRISHKIN → Grishkin
    on VARGA → Varga
  }

  state Varga {
    SetIntValue(intVariable=var "Special Set Type", intValue=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Varga JuggleGame"), sendEvent="ENTER", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Tinks Until Set Complete", add=4, everyFrame=false)
    on FINISHED → Set In Progress
  }

  state Add Variants? {
    BoolTest(boolVariable=var "Added Variants", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    IntCompare(integer1=var "Sets Completed", integer2=16, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetParent(gameObject=var "Bellflea Juggler Giant", parent=var "Bellfleas", resetLocalPosition=false, resetLocalRotation=false)
    SetBoolValue(boolVariable=var "Varga Active", boolValue=true, everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "Added Variants", boolValue=true, everyFrame=false)
    on FINISHED → Stage Check
  }

  state Prepare Game {
    ActivateGameObject(gameObject=var "Bellflea Juggler Giant", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bellfleas", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Grishkin JuggleGame", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Varga JuggleGame", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inactive
  }

  state Game Failed {
    on FINISHED → Inactive
  }

  var Tinks Until Set Complete: int = 0

  var Bounce Time: float = 0
  var Entry Pause: float = 0
  var Fleas To Spawn Float: float = 0
  var Entry Pause Increment: float = 0
  var Quad Bounce Time: float = 3
  var Fleas To Spawn: int = 0
  var Sets Completed: int = 0
  var Strikes: int = 0
  var Special Set Type: int = 0
  var Grishkin Appearances: int = 0
  var Spawn Left: bool = false
  var Next Set Is Special: bool = false
  var Varga Active: bool = false
  var Added Variants: bool = false
  var Event: string = ""
  var Bellfleas: gameObject = <null>
  var Next Flea: gameObject = <null>
  var Grishkin JuggleGame: gameObject = <null>
  var Bellfleas Quickened: gameObject = <null>
  var Varga JuggleGame: gameObject = <null>
  var Bellflea Juggler Giant: gameObject = <null>
  var Bellfleas Giant: gameObject = <null>
}
