# BG Control
# 1 location(s):
#   Shadow_18 / Battle Scene/Gates/Battle Gate Swamp (3)

fsm BG Control {
  start Pause
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state
  on ACTIVATE → Destroy  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Swamp Gate Opened", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on BG CLOSE → Close Pause
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Swamp Gate Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimpleV2(gameObject=Self, volume=(unset), startTime=0, oneShotClip=mantis_lords_gate_close (sfxstatic_assets_battle.bundle))
    on FINISHED → Close 2
  }

  state Close 2 {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Slam", emit=0, resetIfPlaying=false)
    on BG OPEN → Open
    on OPEN → Open
  }

  state Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Swamp Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlaySimpleV2(gameObject=Self, volume=(unset), startTime=0, oneShotClip=col_cage_open (sfxstatic_assets_areabellareaswamp.bundle))
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Opened Effects
  }

  state Quick Close {
    SetBoolValue(boolVariable=var "Start Closed", boolValue=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Swamp Gate Closed")
    on BG OPEN → Open
    on OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Swamp Gate Opened", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close Pause {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    WaitRandom(timeMin=0, timeMax=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Close 1
  }

  state Pause {
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Opened
  }

  state Opened Effects {
    PlayParticleEmitter(gameObject=var "Pt Open", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Rocks", emit=0, resetIfPlaying=false)
    on FINISHED → Opened
  }

  var Start Closed: bool = false

  var Dust: gameObject = <null>
  var Pt Open: gameObject = <null>
  var Pt Rocks: gameObject = Battle Scene/Gates/Battle Gate Swamp (3)/Pt Rocks
  var Pt Slam: gameObject = Battle Scene/Gates/Battle Gate Swamp (3)/Pt Slam
  var Raise Dust: gameObject = Battle Scene/Gates/Battle Gate Swamp (3)/Plat Dust
  var Self: gameObject = <null>
  var Slam Effect: gameObject = Battle Scene/Gates/Battle Gate Swamp (3)/Close Effect
}
