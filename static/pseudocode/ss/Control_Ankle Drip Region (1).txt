# Control
# 2 location(s):
#   Aqueduct_01 / Ankle Drip Region (1)
#   Aqueduct_03 / Ankle Drip Region (1)

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Do Drip
  }

  state Do Drip {
    PlayParticleEmitter(gameObject=var "Dripper pt", emit=0, resetIfPlaying=true)
    on FINISHED → Idle
  }

  var Dripper pt: gameObject = Ankle Drip Region (1)/Dripper/Dripper pt
}
