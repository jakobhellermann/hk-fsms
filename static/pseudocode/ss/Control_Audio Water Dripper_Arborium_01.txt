fsm Control {
  start Pause

  state Pause {
    WaitRandom(timeMin=var "Wait Min", timeMax=var "Wait Max", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Drip
  }

  state Drip {
    RandomFloat(min=5, max=30, storeResult=var "X Offset")
    RandomlyFlipFloat(storeResult=var "X Offset")
    SetPosition2d(gameObject=var "Audio Player", vector=(unset), x=var "X Offset", y=(unset), space=1, everyFrame=false, lateUpdate=false)
    GetRandomAudioClipFromTable(Table=water_drip (sfxdynamic_assets_water.bundle), ForcePlay=false, StoreClip=var "Clip")
    RandomFloat(min=0.5, max=0.6, storeResult=var "Volume")
    RandomFloat(min=0.9, max=1.1, storeResult=var "Pitch")
    SetAudioPitch(gameObject=var "Audio Player", pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Player", volume=var "Volume", oneShotClip=var "Clip")
    on FINISHED → Pause
  }

  var Wait Min: float = 0.5
  var Wait Max: float = 2

  var X Offset: float = 0
  var Volume: float = 0
  var Pitch: float = 0
  var Audio Player: gameObject = Audio Water Dripper/Audio Player
  var Clip: object = <null>
}
