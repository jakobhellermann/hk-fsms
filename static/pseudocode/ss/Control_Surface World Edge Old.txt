# Control
# 1 location(s):
#   Abandoned_town / Surface World Edge Old

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock")
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Dormant
  }

  state Dormant {
    Trigger2dEvent(gameObject=var "Start Range", trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Take Control
  }

  state Take Control {
    SetPlayerDataBool(boolName="disablePause", value=true)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    ActivateGameObject(gameObject=var "CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RunFSM(fsmTemplateControl=template=6159435039333135483 in[Clamp X<-0], finishEvent=→"LAND", everyFrame=false)
    on LAND → Start Pause
  }

  state Meet {
    RunDialogue(Sheet="City", Key="SURFACE_EDGE", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=true, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetPlayerDataBool(boolName="encounteredSurfaceEdge", value=true)
    on CONVO_END → End Dialogue
  }

  state End Dialogue {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    EndDialogue(ReturnControl=false, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move Back
  }

  state End {
    SetPlayerDataBool(boolName="disablePause", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    on FINISHED → Dormant
  }

  state Start Pause {
    PlayAudioEvent(audioClip=soft_landing_sand (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Convo Choice
  }

  state Convo Choice {
    PlayerDataBoolTest(boolName="encounteredSurfaceEdge", isTrue=→"REPEAT", isFalse=→"MEET")
    on MEET → Meet
    on REPEAT → Repeat
  }

  state Repeat {
    RunDialogue(Sheet="City", Key="SURFACE_EDGE_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=true, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Walk Back {
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Walk")
    Wait(time=10, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=5, y=(unset), everyFrame=true)
    CheckXPosition(gameObject=var "Hero", compareTo=var "Walk To X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"FINISHED", greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → End
  }

  state Turn {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    SetVelocity2d(gameObject=var "Hero", vector=(unset), x=5, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="TurnWalk", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Walk Back
  }

  state Move Back {
    MoveHeroToPosX(Hero=var "Hero", PositionTarget=var "", PositionX=var "Walk To X", FinishEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on FINISHED → End
  }

  var Walk To X: float = 61

  var CamLock: gameObject = <null>
  var Self: gameObject = <null>
  var Start Range: gameObject = Surface World Edge Old/Start Range
}
