# Control
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/Bonetown Resident Corpse

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="pilbyKilled", isTrue=(none), isFalse=→"INACTIVE")
    PlayerDataBoolTrueAndFalse(trueBool="skullKingKilled", falseBool="skullKingBenchMended", isTrue=→"ACTIVE", isFalse=(none))
    PlayerDataBoolTest(boolName="skullKingInvaded", isTrue=→"ACTIVE", isFalse=→"INACTIVE")
    on ACTIVE → Active
    on INACTIVE → Inactive
  }

  state Active {
  }

  state Inactive {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
