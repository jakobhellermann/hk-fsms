# Control
# 1 location(s):
#   Ant_08 / Battle Scene/Wave 1/Bone Hunter Tiny (3)

fsm Control {
  start Init

  state Init {
    BoolTest(boolVariable=var "Battler", isTrue=→"BATTLE", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "c_Ambusher", isTrue=→"AMBUSHER", isFalse=(none), everyFrame=false)
    on FINISHED → Walk
    on BATTLE → Dormant
    on AMBUSHER → Ambush Ready
  }

  state Walk {
    BoolTest(boolVariable=var "Battler", isTrue=→"BATTLE", isFalse=(none), everyFrame=false)
    SetAudioClip(gameObject=Self, audioClip=bone_hunter_tiny_walk (sfxstatic_assets_areaant.bundle), autoPlay=true, stopOnExit=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Chase Range", storeResult=false, sendEvent="CHASE", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    WalkLeftRight(gameObject=Self, walkSpeed=4, spriteFacesLeft=true, groundLayer="Terrain", turnDelay=0.1, wallLookAhead=0.5, walkAnimName="Walk", turnAnimName="Turn", moveWhileTurning=true, startLeft=false, startRight=false, keepDirection=true, flipBeforeTurn=true, turnAudioClipTable=<null>)
    on CHASE → Chase
    on SING → Sing
    on BATTLE → Battle Walk
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    SetAudioSource(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    on BATTLE START → Start Pause
  }

  state Start Pause {
    WaitRandom(timeMin=1, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pos Check
  }

  state Emerge 1 {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Dust", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Rock", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Walk Dust", emit=0, resetIfPlaying=false)
    PlayRandomAudioClipTable(Table=enemy_dig_in_antic (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Emerge 2
  }

  state Emerge 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetMeshRenderer(gameObject=Self, active=true)
    PlayRandomAudioClipTable(Table=enemy_dig_in_emerge (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ant_tiny_yelp (sfxdynamic_assets_areaant.bundle), pitchOffset=0.1, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Emerge 3
  }

  state Emerge 3 {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopParticleEmitter(gameObject=var "Pt Dust")
    StopParticleEmitter(gameObject=var "Pt Rock")
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Battle Walk
  }

  state Battle Walk {
    SetAudioSource(gameObject=Self, active=true)
    SetAudioClip(gameObject=Self, audioClip=bone_hunter_tiny_walk (sfxstatic_assets_areaant.bundle), autoPlay=true, stopOnExit=false)
    BoolTest(boolVariable=var "c_Ambusher", isTrue=→"NORMAL", isFalse=(none), everyFrame=false)
    WalkLeftRight(gameObject=Self, walkSpeed=6, spriteFacesLeft=true, groundLayer="Terrain", turnDelay=0.25, wallLookAhead=0, walkAnimName="Walk", turnAnimName="Turn", moveWhileTurning=true, startLeft=false, startRight=false, keepDirection=true, flipBeforeTurn=true, turnAudioClipTable=<null>)
    WaitRandom(timeMin=1, timeMax=3, finishEvent=→"SHIFT", realTime=false)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Chase Range", storeResult=false, sendEvent="CHASE", outOfRangeEvent="", everyFrame=true)
    on SHIFT → Submerge 1
    on TOOK DAMAGE → Dmg Response
    on NORMAL → Walk
    on CHASE → Chase
  }

  state Submerge 1 {
    SetAudioSource(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Disappear", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    StopParticleEmitter(gameObject=var "Walk Dust")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ant_tiny_yelp (sfxdynamic_assets_areaant.bundle), pitchOffset=0.1, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Submerge 2
  }

  state Submerge 2 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitterInState(gameObject=var "Pt Dust")
    PlayParticleEmitterInState(gameObject=var "Pt Rock")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Submerge 3
  }

  state Submerge 3 {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Aim
  }

  state Aim {
    SendRandomEventV2(events=[→"HERO", →"RANDOM"], weights=[0.4, 0.6], trackingInts=[var "Ct Hero", var "Ct Random"], eventMax=[1, 2])
    on RANDOM → Random
    on HERO → Hero
  }

  state Random {
    GetRandomObject(withTag="Teleplane", storeResult=var "Teleplane", everyFrame=false)
    on FINISHED → Set Pos
  }

  state Hero {
    FindClosest(gameObject=var "Hero", withTag="Teleplane", ignoreOwner=true, mustBeVisible=false, storeObject=var "Teleplane", storeDistance=(unset), everyFrame=false)
    on FINISHED → Set Pos
  }

  state Set Pos {
    GetColliderRange(gameObject=var "Teleplane", minX=var "X Min", maxX=var "X Max", minY=(unset), maxY=var "Tele Y", everyFrame=false)
    RandomFloat(min=var "X Min", max=var "X Max", storeResult=var "Tele X")
    SetPosition(gameObject=Self, vector=(unset), x=var "Tele X", y=var "Tele Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Terrain Box", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emerge Ready
  }

  state Emerge Ready {
    ActivateGameObject(gameObject=var "Terrain Box", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    on FINISHED → Emerge 1
  }

  state Dmg Response {
    BoolTest(boolVariable=var "z_No Dig", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    SendRandomEventV2(events=[→"SHIFT", →"SPEED UP"], weights=[1, 1], trackingInts=[var "Ct Shift", var "Ct SpeedUp"], eventMax=[1, 2])
    on SHIFT → Submerge 1
    on SPEED UP → Speed Up
    on CANCEL → Battle Walk
  }

  state Speed Up {
    SetAudioClip(gameObject=Self, audioClip=bone_hunter_tiny_run (sfxstatic_assets_areaant.bundle), autoPlay=true, stopOnExit=false)
    WaitRandom(timeMin=0.75, timeMax=1.5, finishEvent=→"SHIFT", realTime=false)
    WalkLeftRight(gameObject=Self, walkSpeed=9, spriteFacesLeft=true, groundLayer="Terrain", turnDelay=0.25, wallLookAhead=0, walkAnimName="Run", turnAnimName="Turn Q", moveWhileTurning=false, startLeft=false, startRight=false, keepDirection=false, flipBeforeTurn=false, turnAudioClipTable=<null>)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ant_tiny_yelp (sfxdynamic_assets_areaant.bundle), pitchOffset=0.1, stopPreviousSound=true, forcePlay=false)
    on SHIFT → Submerge 1
  }

  state Chase {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=ant_tiny_yelp (sfxdynamic_assets_areaant.bundle), pitchOffset=0.1, stopPreviousSound=true, forcePlay=false)
    SetAudioClip(gameObject=Self, audioClip=bone_hunter_tiny_run (sfxstatic_assets_areaant.bundle), autoPlay=true, stopOnExit=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Chase Range", storeResult=false, sendEvent="", outOfRangeEvent="SHIFT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    WalkLeftRight(gameObject=Self, walkSpeed=8, spriteFacesLeft=true, groundLayer="Terrain", turnDelay=0.1, wallLookAhead=0.5, walkAnimName="Run", turnAnimName="Turn", moveWhileTurning=true, startLeft=false, startRight=false, keepDirection=true, flipBeforeTurn=true, turnAudioClipTable=<null>)
    on SHIFT → Walk
    on SING → Sing
  }

  state Sing {
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=ant_tiny_sing (sfxdynamic_assets_areaant.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Ambush Ready {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    SetMeshRenderer(gameObject=Self, active=false)
    SetAudioSource(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    on WAKE → Emerge 1
  }

  state Pos Check {
    GetXDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=6, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=false)
    on FINISHED → Emerge 1
    on SHIFT → Spawn Pos Alt
  }

  state Spawn Pos Alt {
    SetPositionToObject2D(gameObject=Self, targetObject=var "Spawn Pos Alt", xOffset=0, yOffset=0, everyFrame=false)
    on FINISHED → Emerge 1
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Walk
  }

  var Battler: bool = true
  var c_Ambusher: bool = false
  var z_No Dig: bool = true

  var Tele X: float = 0
  var Tele Y: float = 0
  var X Max: float = 0
  var X Min: float = 0
  var Distance: float = 0
  var Ct Hero: int = 0
  var Ct Random: int = 0
  var Ct Shift: int = 0
  var Ct SpeedUp: int = 0
  var Pt Dust: gameObject = Battle Scene/Wave 1/Bone Hunter Tiny (3)/Pt Dust
  var Pt Rock: gameObject = Battle Scene/Wave 1/Bone Hunter Tiny (3)/Pt Rock
  var Teleplane: gameObject = <null>
  var Terrain Box: gameObject = Battle Scene/Wave 1/Bone Hunter Tiny (3)/Terrain Box
  var Walk Dust: gameObject = Battle Scene/Wave 1/Bone Hunter Tiny (3)/Walk Dust
  var Spawn Pos Alt: gameObject = Battle Scene/Wave 1/Bone Hunter Tiny (3)/Spawn Pos Alt
  var Audio Loop Voice: gameObject = Battle Scene/Wave 1/Bone Hunter Tiny (3)/Audio Loop Voice
}
