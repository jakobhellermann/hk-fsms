    SetParent(gameObject=var "Chain Dust", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on ACTIVATE → Activate
    on OPEN → Antic
  }

  state Antic {
    SendEventByName(eventTarget=GameObject(var "Cog 1"), sendEvent="START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Cog 2"), sendEvent="START", delay=0, everyFrame=false)

    PlayParticleEmitter(gameObject=var "Antic Dust 2", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Move Dust", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Chain Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Drop
  }

  state Activate {
    SetParent(gameObject=var "Cog 1", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Cog 2", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Hero Land", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)

  }

  state Drop {
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -29, 0), time=2.5, speed=(unset), delay=(unset), easeType=3, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Slam
  }

  state Slam {
    AudioPlayV2(gameObject=var "Audio Source", volume=1, oneShotClip=chain_plat_bone_04_centre_land (sfxstatic_assets_areaboneareaunderstore.bundle), finishedEvent=(none))
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)

    StopParticleEmitter(gameObject=var "Chain Dust")
  }

  var Antic Dust: gameObject = Chain Plat Lever/Antic Dust
  var Antic Dust 2: gameObject = <null>
  var Cog 1: gameObject = <null>
  var Cog 2: gameObject = <null>
  var Hero Land: gameObject = Chain Plat Lever/Hero Land
  var Chain Dust: gameObject = Chain Plat Lever/Chain Dust
  var Move Dust: gameObject = Chain Plat Lever/Move Dust
  var Audio Source: gameObject = Chain Plat Lever/Plat Sprite/Audio Source
}
