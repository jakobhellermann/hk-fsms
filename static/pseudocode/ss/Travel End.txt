# Travel End
# 3 location(s):
#   Aqueduct_05_caravan / Caravan_States/Fleatopia/door_caravanTravelEnd
#   Coral_Judge_Arena / door_caravanTravelEnd
#   Greymoor_08_caravan / Caravan/door_caravanTravelEnd

fsm Travel End {
  start Pause

  state Pause {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    SetTransformParent(gameObject=Self, parent=<null>, worldPositionStays=true)
    on START ENTRY → Wait
  }

  state Send Event {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="GetSceneNameString", parameters=[], storeResult=var "Scene Name", EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetBenchRespawn", parameters=["Death Respawn Marker", var "Scene Name", 0, true], storeResult=(unset var), EveryFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=0, functionCall=TimePasses())
    QueueSaveGame()
    RemoveHeroInputBlocker(Blocker=Self)
    SetPlayerDataBool(boolName="disablePause", value=false)
    SendEventToRegister(eventName="CARAVAN TRAVEL ENDED")
  }

  state Wait {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    on FINISH ENTRY → Wait More
  }

  sequence state Fade Up {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=1.5)
    Wait(time=2, finishEvent=(none), realTime=false)
    on FINISHED → Send Event
  }

  state Wait More {
    AddHeroInputBlocker(Blocker=Self)
    SetPlayerDataBool(boolName="disablePause", value=true)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Up
  }

  var Scene Name: string = ""
  var Self: gameObject = <null>
}
