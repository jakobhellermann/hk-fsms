# FSM
# 2 location(s):
#   Song_Enclave / Black Thread States/Normal World/Enclave States/States/Level 1/City Merchant Enclave/Seen Range
#   Song_Enclave / Black Thread States/Black Thread World/City Merchant Enclave/Seen Range

fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Leave Library?
  }

  state Set Seen {
    SetPlayerDataBool(boolName="cityMerchantRecentlySeenInEnclave", value=true)
  }

  state Leave Library? {
    PlayerDataBoolTest(boolName="cityMerchantInLibrary03Seen", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="cityMerchantInLibrary03Left", value=true)
    on FINISHED → Leave Forum?
  }

  state Leave Forum? {
    PlayerDataBoolTest(boolName="cityMerchantInGrandForumSeen", isTrue=(none), isFalse=→"FINISHED")
    SetPlayerDataBool(boolName="cityMerchantInGrandForumLeft", value=true)
    on FINISHED → Set Seen
  }
}
