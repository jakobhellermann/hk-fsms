# FSM
# 4 location(s):
#   Abyss_01 / Weaver_super_jump_pad/Plate
#   Abyss_09 / Weaver_super_jump_pad/Plate
#   Abyss_09 / Weaver_super_jump_pad (1)/Plate
#   Abyss_11 / Weaver_super_jump_pad/Plate

fsm FSM {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Player(9), sendEvent=→"COLLIDE", storeCollider=var "", storeForce=var "Force", contactPoint=(unset), contactNormal=(unset))
    on COLLIDE → Check Force
  }

  state Check Force {
    FloatCompare(float1=var "Force", float2=30, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"DEPRESS", everyFrame=false)
    on FINISHED → Idle
    on DEPRESS → Depress
  }

  state Depress {
    PlayAudioEvent(audioClip=song_gate_hit (sfxstatic_assets_areasong.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=weaver_gate_open_end (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatePositionTo(gameObject=Self, toValue=(0, -0.31, -0.14), localSpace=true, time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Return
  }

  state Return {
    AnimatePositionTo(gameObject=Self, toValue=(0, 0, -0.14), localSpace=true, time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  var Force: float = 0
}
