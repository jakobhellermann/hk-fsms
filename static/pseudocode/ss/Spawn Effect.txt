# Spawn Effect
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/Charm Effects/Fractured Mask Break

fsm Spawn Effect {
  start Init

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init Check
  }

  state Idle {
    SetBoolValue(boolVariable=var "Was Tool Equipped", boolValue=var "Is Tool Equipped", everyFrame=false)
    EnableFsmSelf(SetEnabled=false)
    on TOOL EQUIPS CHANGED → Init Check
    on HERO DAMAGED → Check
  }

  state Init Check {
    CheckIfToolEquipped(Tool=var "Tool", RequiredAmountLeft=1, trueEvent=(none), falseEvent=(none), storeValue=var "Is Tool Equipped")
    on FINISHED → Idle
  }

  sequence state Check {
    NextFrameEvent(sendEvent=(none))
    CheckIfToolEquipped(Tool=var "Tool", RequiredAmountLeft=1, trueEvent=→"FINISHED", falseEvent=(none), storeValue=var "Is Tool Equipped")
    BoolTestMulti(boolVariables=[var "Is Tool Equipped", var "Was Tool Equipped"], boolStates=[false, true], trueEvent=→"SHATTER", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Idle
    on SHATTER → Instantiate Effect
  }

  state Instantiate Effect {
    CreateObject(gameObject=bone_mask_shatter (localpoolprefabs_assets_shared.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Spawned Effect")
    MoveGameObjectToScene(gameObject=var "Spawned Effect", findRootIfNecessary=false, success=(unset), successEvent=(none), failureEvent=(none), sceneReference=0, sceneAtIndex=0, sceneByName="", sceneByPath="", sceneByGameObject=Self, sceneFound=(unset), sceneFoundEvent=(none), sceneNotFoundEvent=(none))
    on FINISHED → Idle
  }

  var Is Tool Equipped: bool = false
  var Was Tool Equipped: bool = false
  var Self: gameObject = <null>
  var Spawned Effect: gameObject = <null>
  var Tool: object = Fractured Mask (dataassets_assets_assets/dataassets/tools/toolitems.bundle)
}
