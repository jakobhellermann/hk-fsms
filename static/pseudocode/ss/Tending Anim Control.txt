# Tending Anim Control
# 1 location(s):
#   Dust_11 / Steel Soul States/Regular/NPC Control/Grub Farmer NPC

fsm Tending Anim Control {
  start Init
  on TEND ANIM STOP → Inactive  // from any state

  state Init {
    on FINISHED → Inactive
  }

  state Inactive {
    AudioStop(gameObject=var "Audio Loop Tending", fadeTime=0)
    on TEND ANIM START → Start Audio
  }

  state Start Audio {
    AudioPlaySimple(gameObject=var "Audio Loop Tending", volume=1, oneShotClip=<null>)
    on FINISHED → Work Loop
  }

  state Work Loop {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Work Loop", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    ObjectJitterOnRender(Target=var "Cocoon Crt", X=0.075, Y=0.075, Z=(unset), LimitFps=15)
    on FINISHED → Reset?
  }

  state Reset? {
    SendRandomEventV4(events=[→"FINISHED", →"RESET"], weights=[0.75, 0.25], eventMax=[3, 1], missedMax=[1, 3], activeBool=(unset))
    on FINISHED → Work Loop
    on RESET → Reset Anim
  }

  state Reset Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Work Reset", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Work Loop
  }

  var Audio Loop Tending: gameObject = Steel Soul States/Regular/NPC Control/Grub Farmer NPC/Audio Loop Tending
  var Cocoon Crt: gameObject = <null>
}
