# Area Title Control
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Area Title
#   Menu_Title / _GameCameras/HudCamera/In-game/Area Title

fsm Area Title Control {
  start First Start
  on LEVEL LOADED → Destroy  // from any state
  on DISABLE → Destroy  // from any state

  state Init {
    SetFloatValue(floatVariable=var "Small Title X Pos", floatValue=-13.68, everyFrame=false)
    SetFloatValue(floatVariable=var "Small Title X Pos R", floatValue=13, everyFrame=false)
    on FINISHED → Other
  }

  state Visited Check {
    BoolTest(boolVariable=var "NPC Title", isTrue=→"VISITED", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Visited", isTrue=→"VISITED", isFalse=→"UNVISITED", everyFrame=false)
    on UNVISITED → Set Large
    on VISITED → Set Text Small
  }

  state Fleurs Up {
    PlayAudioEvent(audioClip=ui_area_title_appear (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateInteraction(Activate=false)
    FadeColorFader(target=var "Current Fluer Bot", fadeType=UP, useChildren=false)
    FadeColorFader(target=var "Current Fluer Top", fadeType=UP, useChildren=false)
    ActivateGameObject(gameObject=var "Current Title Folder", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Current Title Folder", ToAlpha=1, FadeTime=var "Large Fade In Time")
    Tk2dPlayAnimation(gameObject=var "Current Fluer Top", animLibName="", clipName=var "Fluer Anim Top Appear")
    Tk2dPlayAnimation(gameObject=var "Current Fluer Bot", animLibName="", clipName=var "Fluer Anim Bot Appear")
    AnimatorPlay(gameObject=var "Current Title Main", stateName="Glow", layer=(unset), normalizedTime=(unset), everyFrame=false)
    AnimatorPlay(gameObject=var "Current Title Sub", stateName="Glow", layer=(unset), normalizedTime=(unset), everyFrame=false)
    AnimatorPlay(gameObject=var "Current Title Super", stateName="Glow", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=var "Unvisited Wait", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Titles Down
    on DIALOGUE BOX APPEARING → Big Title End
    on INVENTORY OPENED → Big Title End
  }

  state Titles Up {
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fleurs Up
    on NPC CONVO START → Fleurs Up
    on DIALOGUE BOX APPEARING → Big Title End
    on INVENTORY OPENED → Big Title End
  }

  state Titles Down {
    FadeNestedFadeGroup(Target=var "Current Title Folder", ToAlpha=0, FadeTime=var "Large Fade Out Time")
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    FadeColorFader(target=var "Current Fluer Bot", fadeType=DOWN, useChildren=false)
    FadeColorFader(target=var "Current Fluer Top", fadeType=DOWN, useChildren=false)
    on FINISHED → Big Title End
    on DIALOGUE BOX APPEARING → Big Title End
    on INVENTORY OPENED → Big Title End
  }

  state Done {
    SetBoolValue(boolVariable=var "NPC Title Waiting", boolValue=false, everyFrame=false)
    UIMsgPopupSetMinY(value=-9999)
    SendEventByName(eventTarget=BroadcastAll, sendEvent="TITLE DISPLAYED", delay=0, everyFrame=false)
    SendEventToRegister(eventName="TITLE DISPLAYED")
    SetBoolValue(boolVariable=var "NPC Title", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "City Title", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Display Right", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Subscript", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Superscript", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Visited", boolValue=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    SetBoolValue(boolVariable=var "NPC Title Waiting", boolValue=false, everyFrame=false)
    UIMsgPopupSetMinY(value=-9999)
    ActivateGameObject(gameObject=var "Large Title Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Large City Title Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Small Title Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Bot", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Top", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init all
    on NPC TITLE DOWN → Done
  }

  state Visited Appear {
    SendEventByName(eventTarget=GameObject(var "Small Title Folder"), sendEvent="UP", delay=0, everyFrame=false)  // disabled
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Small Title Folder", ToAlpha=1, FadeTime=var "Small Fade In Time")
    BoolTest(boolVariable=var "NPC Title", isTrue=→"NPC", isFalse=(none), everyFrame=false)
    on FINISHED → Visited Disappear
    on NPC CONVO START → Quick Disappear
    on NPC → NPC Wait for Event
  }

  state Visited Disappear {
    SendEventByName(eventTarget=GameObject(var "Small Title Folder"), sendEvent="DOWN", delay=0, everyFrame=false)  // disabled
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Small Title Folder", ToAlpha=0, FadeTime=var "Small Fade Out Time")
    on FINISHED → Done
  }

  state Big Title End {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BIG TITLE END", delay=0, everyFrame=false)
    ActivateInteraction(Activate=true)
    ActivateGameObject(gameObject=var "Current Fluer Bot", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Top", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Done
  }

  state Set Text Small {
    SetTextMeshProText(gameObject=var "Small Title Main", textString=var "Title Main")
    SetTextMeshProText(gameObject=var "Small Title Sub", textString=var "Title Sub")
    SetTextMeshProText(gameObject=var "Small Title Sup", textString=var "Title Sup")
    FadeNestedFadeGroup(Target=var "Small Title Folder", ToAlpha=0, FadeTime=0)
    ActivateGameObject(gameObject=var "Small Title Folder", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=-13.68, y=-12.41, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetFloatValue(floatVariable=var "Longest Width", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Margin X", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Right X", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Width Main", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Width Sub", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Width Super", floatValue=0, everyFrame=false)
    on FINISHED → Pause Frame
  }

  state Super only 2 {
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=(unset), y=-13.15, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Visited Appear
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Bounds Main
  }

  state Set Margin {
    FloatMultiply(floatVariable=var "Width Main", multiplyBy=0.5, everyFrame=false)
    GetMeshRendererBounds(gameObject=var "Small Title Sub", width=var "Width Sub", height=(unset), widthMax=200, heightMax=(unset))  // disabled
    FloatMultiply(floatVariable=var "Width Sub", multiplyBy=0.5, everyFrame=false)
    GetMeshRendererBounds(gameObject=var "Small Title Sup", width=var "Width Super", height=(unset), widthMax=200, heightMax=(unset))  // disabled
    FloatMultiply(floatVariable=var "Width Super", multiplyBy=0.5, everyFrame=false)
    SetFloatToHighest(floatVariable=var "Longest Width", value1=var "Width Main", value2=var "Width Sub", everyFrame=false)
    SetFloatToHighest(floatVariable=var "Longest Width", value1=var "Longest Width", value2=var "Width Super", everyFrame=false)
    BoolTest(boolVariable=var "Display Right", isTrue=→"RIGHT", isFalse=(none), everyFrame=false)
    FloatAdd(floatVariable=var "Longest Width", add=var "Margin X", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Longest Width", add=var "Small Title X Pos", everyFrame=false, perSecond=false)
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=var "Longest Width", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    GetConstantsValue(variableName="AREA_TITLE_UI_MSG_Y", storeValue=var "Value")
    UIMsgPopupSetMinY(value=var "Value")
    on FINISHED → Sub or Super Active? 2
    on RIGHT → Set Margin R
  }

  state Set Text Large {
    ActivateGameObject(gameObject=var "Current Fluer Bot", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Current Title Folder", ToAlpha=0, FadeTime=0)
    ActivateGameObject(gameObject=var "Current Fluer Top", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetTextMeshProText(gameObject=var "Current Title Main", textString=var "Title Main")
    SetTextMeshProText(gameObject=var "Current Title Sub", textString=var "Title Sub")
    SetTextMeshProText(gameObject=var "Current Title Super", textString=var "Title Sup")
    Tk2dPlayAnimation(gameObject=var "Current Fluer Top", animLibName="", clipName="Idle")
    Tk2dPlayAnimation(gameObject=var "Current Fluer Bot", animLibName="", clipName="Idle")
    on FINISHED → Titles Up
  }

  state Check Sub {
    StringCompare(stringVariable=var "Title Sub", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Subscript", boolValue=true, everyFrame=false)
    on FINISHED → Check Super
  }

  state Reset Sub & Sup {
    SetBoolValue(boolVariable=var "Title Has Subscript", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Superscript", boolValue=false, everyFrame=false)
    on FINISHED → Check Sub
  }

  state Check Super {
    StringCompare(stringVariable=var "Title Sup", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Superscript", boolValue=true, everyFrame=false)
    on FINISHED → Visited Check
  }

  state Sub or Super Active? {
    BoolAllTrue(boolVariables=[var "Title Has Subscript", var "Title Has Superscript"], sendEvent=→"BOTH", storeResult=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Subscript", isTrue=→"SUB", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Superscript", isTrue=→"SUP", isFalse=(none), everyFrame=false)
    on BOTH → Set Text Large 2
    on SUB → Sub Only
    on SUP → Sup Only
    on FINISHED → Neither
  }

  state Sub Only {
    SetPosition(gameObject=var "Current Fluer Bot", vector=(unset), x=(unset), y=-5.24, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Current Title Folder", vector=(unset), x=(unset), y=-2.24, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    RectTransformSetLocalPosition(gameObject=var "Large Title Main", position2d=(unset), position=(unset), x=(unset), y=0.25, z=(unset), everyFrame=false, updateType=0)
    on FINISHED → Set Text Large 2
  }

  state Neither {
    SetPosition(gameObject=var "Current Fluer Bot", vector=(unset), x=(unset), y=-4.38, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Current Title Folder", vector=(unset), x=(unset), y=-2.27, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    RectTransformSetLocalPosition(gameObject=var "Large Title Main", position2d=(unset), position=(unset), x=(unset), y=0.18, z=(unset), everyFrame=false, updateType=0)
    on FINISHED → Set Text Large 2
  }

  state Sup Only {
    SetPosition(gameObject=var "Current Fluer Bot", vector=(unset), x=(unset), y=-5.24, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Current Title Folder", vector=(unset), x=(unset), y=-3.26, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    RectTransformSetLocalPosition(gameObject=var "Large Title Main", position2d=(unset), position=(unset), x=(unset), y=0.12, z=(unset), everyFrame=false, updateType=0)
    on FINISHED → Set Text Large 2
  }

  state NPC Wait for Event {
    SetBoolValue(boolVariable=var "NPC Title Waiting", boolValue=true, everyFrame=false)
    on NPC TITLE DOWN → Quick Disappear
    on NPC CONVO START → Quick Disappear
  }

  state Other {
    BuildString(stringParts=[var "Area Event", "_MAIN"], separator="", addToEnd=true, storeResult=var "Main String", everyFrame=false)
    BuildString(stringParts=[var "Area Event", "_SUB"], separator="", addToEnd=true, storeResult=var "Sub String", everyFrame=false)
    BuildString(stringParts=[var "Area Event", "_SUPER"], separator="", addToEnd=true, storeResult=var "Super String", everyFrame=false)
    GetLanguageStringProcessed(sheetName="Titles", convName=var "Main String", storeValue=var "Title Main", fontSource=Trajan)
    GetLanguageStringProcessed(sheetName="Titles", convName=var "Sub String", storeValue=var "Title Sub", fontSource=Trajan)
    GetLanguageStringProcessed(sheetName="Titles", convName=var "Super String", storeValue=var "Title Sup", fontSource=Trajan)
    on FINISHED → Reset Sub & Sup
  }

  state Sub or Super Active? 2 {
    BoolAllTrue(boolVariables=[var "Title Has Subscript", var "Title Has Superscript"], sendEvent=→"BOTH", storeResult=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Subscript", isTrue=→"SUB", isFalse=(none), everyFrame=false)
    BoolTest(boolVariable=var "Title Has Superscript", isTrue=→"SUP", isFalse=(none), everyFrame=false)
    SendEventByName(eventTarget=Self, sendEvent="NEITHER", delay=0, everyFrame=false)
    on BOTH → Visited Appear
    on NEITHER → Super only 2
    on SUB → Visited Appear
    on SUP → Super only 2
  }

  state Quick Disappear {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Small Title Folder", ToAlpha=0, FadeTime=var "Small Fade Out Quick Time")
    on NPC TITLE DOWN CANCEL → Visited Appear
    on FINISHED → Done
  }

  state Destroy {
    CallStaticMethod(className="UIMsgPopupBaseBase", methodName="set_MinYPos", parameters=[-9999], storeResult=(unset var), everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Bot", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Top", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "NPC Title", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "City Title", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Display Right", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Subscript", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Superscript", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Visited", boolValue=false, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Small Title Folder", ToAlpha=0, FadeTime=0)
    FadeNestedFadeGroup(Target=var "Current Title Folder", ToAlpha=0, FadeTime=0)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Disabled Interaction", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    ActivateInteraction(Activate=true)
    SetBoolValue(boolVariable=var "Disabled Interaction", boolValue=false, everyFrame=false)
    on FINISHED → Empty
  }

  state Set Margin R {
    FloatSubtract(floatVariable=var "Longest Width", subtract=var "Margin X", everyFrame=false, perSecond=false)
    FloatSubtract(floatVariable=var "Small Title X Pos R", subtract=var "Longest Width", everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Longest Width", add=var "Small Title X Pos R", everyFrame=false, perSecond=false)  // disabled
    SetPosition(gameObject=var "Small Title Folder", vector=(unset), x=var "Small Title X Pos R", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Sub or Super Active? 2
  }

  state Bounds Main {
    GetMeshRendererBounds(gameObject=var "Small Title Main", width=var "Width Main", height=(unset), widthMax=200, heightMax=(unset))
    on FINISHED → Bounds Sub
  }

  state Bounds Sub {
    StringCompare(stringVariable=var "Title Sub", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    GetMeshRendererBounds(gameObject=var "Small Title Sub", width=var "Width Sub", height=(unset), widthMax=200, heightMax=(unset))
    on FINISHED → Bounds Super
  }

  state Bounds Super {
    StringCompare(stringVariable=var "Title Sup", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    GetMeshRendererBounds(gameObject=var "Small Title Sup", width=var "Width Super", height=(unset), widthMax=200, heightMax=(unset))
    on FINISHED → Set Margin
  }

  state City? {
    BoolTest(boolVariable=var "City Title", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Set Large
    on TRUE → Set Large City
  }

  state Set Large {
    SetGameObject(variable=var "Current Title Folder", gameObject=var "Large Title Folder", everyFrame=false)
    SetGameObject(variable=var "Current Title Main", gameObject=var "Large Title Main", everyFrame=false)
    SetGameObject(variable=var "Current Title Sub", gameObject=var "Large Title Sub", everyFrame=false)
    SetGameObject(variable=var "Current Title Super", gameObject=var "Large Title Super", everyFrame=false)
    SetGameObject(variable=var "Current Fluer Bot", gameObject=var "Large Fleur Bot", everyFrame=false)
    SetGameObject(variable=var "Current Fluer Top", gameObject=var "Large Fleur Top", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Top Appear", stringValue="Top Appear", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Top Disappear", stringValue="Top Disappear", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Bot Appear", stringValue="Bottom Appear", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Bot Disappear", stringValue="Bottom Disappear", everyFrame=false)
    on FINISHED → Sub or Super Active?
  }

  state Set Large City {
    SetGameObject(variable=var "Current Title Folder", gameObject=var "Large City Title Folder", everyFrame=false)
    SetGameObject(variable=var "Current Title Main", gameObject=var "Large City Title Main", everyFrame=false)
    SetGameObject(variable=var "Current Title Sub", gameObject=var "Large City Title Sub", everyFrame=false)
    SetGameObject(variable=var "Current Title Super", gameObject=var "Large City Title Super", everyFrame=false)
    SetGameObject(variable=var "Current Fluer Bot", gameObject=var "Large City Fluer Bot", everyFrame=false)
    SetGameObject(variable=var "Current Fluer Top", gameObject=var "Large City Fluer Top", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Top Appear", stringValue="City Top Appear", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Top Disappear", stringValue="City Top Disappear", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Bot Appear", stringValue="City Bottom Appear", everyFrame=false)
    SetStringValue(stringVariable=var "Fluer Anim Bot Disappear", stringValue="City Bottom Disappear", everyFrame=false)
    on FINISHED → Sub or Super Active?
  }

  state Fleurs Up 2 {
    AnimatorPlay(gameObject=_GameCameras/HudCamera/In-game/Area Title/Title Large, stateName="Appear", layer=(unset), normalizedTime=(unset), everyFrame=false)
    ActivateInteraction(Activate=false)
    SetBoolValue(boolVariable=var "Disabled Interaction", boolValue=true, everyFrame=false)
    Wait(time=var "Unvisited Wait", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Titles Down 2
    on DIALOGUE BOX APPEARING → Big Title End 2
    on INVENTORY OPENED → Big Title End 2
  }

  state Titles Up 2 {
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fleurs Up 2
    on NPC CONVO START → Fleurs Up 2
    on DIALOGUE BOX APPEARING → Big Title End 2
    on INVENTORY OPENED → Big Title End 2
  }

  state Titles Down 2 {
    AnimatorPlay(gameObject=_GameCameras/HudCamera/In-game/Area Title/Title Large, stateName="Hide", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Big Title End 2
    on DIALOGUE BOX APPEARING → Big Title End 2
    on INVENTORY OPENED → Big Title End 2
  }

  state Big Title End 2 {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="BIG TITLE END", delay=0, everyFrame=false)
    ActivateInteraction(Activate=true)
    SetBoolValue(boolVariable=var "Disabled Interaction", boolValue=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Bot", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Top", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Done
  }

  state Set Text Large 2 {
    ActivateGameObject(gameObject=var "Current Fluer Bot", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Current Fluer Top", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetTextMeshProText(gameObject=var "Current Title Main", textString=var "Title Main")
    SetTextMeshProText(gameObject=var "Current Title Sub", textString=var "Title Sub")
    SetTextMeshProText(gameObject=var "Current Title Super", textString=var "Title Sup")
    on FINISHED → Titles Up 2
  }

  state First Start {
    BoolTest(boolVariable=var "Initialised", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Initialised", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Pause
  }

  state Init all {
    SetFloatValue(floatVariable=var "Small Title X Pos", floatValue=-13.68, everyFrame=false)
    SetFloatValue(floatVariable=var "Small Title X Pos R", floatValue=13, everyFrame=false)
    BuildString(stringParts=[var "Area Event", "_MAIN"], separator="", addToEnd=true, storeResult=var "Main String", everyFrame=false)
    BuildString(stringParts=[var "Area Event", "_SUB"], separator="", addToEnd=true, storeResult=var "Sub String", everyFrame=false)
    BuildString(stringParts=[var "Area Event", "_SUPER"], separator="", addToEnd=true, storeResult=var "Super String", everyFrame=false)
    GetLanguageStringProcessed(sheetName="Titles", convName=var "Main String", storeValue=var "Title Main", fontSource=Trajan)
    GetLanguageStringProcessed(sheetName="Titles", convName=var "Sub String", storeValue=var "Title Sub", fontSource=Trajan)
    GetLanguageStringProcessed(sheetName="Titles", convName=var "Super String", storeValue=var "Title Sup", fontSource=Trajan)
    SetBoolValue(boolVariable=var "Title Has Subscript", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Title Has Superscript", boolValue=false, everyFrame=false)
    on FINISHED → Check Sub
  }

  state Empty {
  }

  var Large Fade In Time: float = 0.3
  var Large Fade Out Time: float = 0.75
  var Small Fade In Time: float = 1
  var Small Fade Out Quick Time: float = 0.5
  var Small Fade Out Time: float = 1
  var Large Fleur Bot: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Large/Title_Fleur_Bot
  var Large Fleur Top: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Large/Title_Fleur_Top
  var Large Title Folder: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Large/Title Large Folder
  var Large Title Main: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Large/Title Large Folder/Title Large Main
  var Large Title Sub: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Large/Title Large Folder/Title Large Sub
  var Large Title Super: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Large/Title Large Folder/Title Large Super
  var Small Title Folder: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Small
  var Small Title Main: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Small/Title Small Main
  var Small Title Sub: gameObject = _GameCameras/HudCamera/In-game/Area Title/Title Small/Title Small Sub
