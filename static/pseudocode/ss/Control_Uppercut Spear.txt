# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Uppercut Spear

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Coral King")
    FindNamedChild(gameObject=Self, storeResult=var "Coral_spear_long")
    on FINISHED → Idle
  }

  state Idle {
    on SPEAR → Dir
  }

  state Attack {
    ActivateGameObject(gameObject=var "Coral_spear_long", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Dir {
    ActivateGameObject(gameObject=var "Coral_spear_long", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByScale(gameObject=var "Coral King", eventTarget=Self, xScale=true, positiveEvent=→"R", negativeEvent=→"L", space=0)
    on L → Set L
    on R → Set R
  }

  state Set L {
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Hero X", add=8.21, everyFrame=false, perSecond=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Attack
  }

  state Set R {
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Hero X", add=-8.21, everyFrame=false, perSecond=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Attack
  }

  var Hero X: float = 0
  var X Scale: float = 0
  var Boss Scene: gameObject = <null>
  var Coral King: gameObject = <null>
  var Coral_spear_long: gameObject = <null>
}
