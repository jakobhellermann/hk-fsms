# Control
# 1 location(s):
#   Ant_21 / Hunter Trap Plate

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)  // disabled
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Player(9), sendEvent=→"STEP", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"STEP", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Corpse(26), sendEvent=→"STEP", storeCollider=var "")
    on STEP → Cam Check
    on ACTIVATED → Activated
  }

  state Activated {
    SendEventByName(eventTarget=GameObject(var "Device"), sendEvent="ACTIVATED", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Step {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayAudioEvent(audioClip=bone_trap_button (sfxstatic_assets_areaant.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Plate Press", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Step Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Activate Device
  }

  state Activate Device {
    SendEventByName(eventTarget=GameObject(var "Device"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Reset Timer
  }

  state Reset Timer {
    FloatCompare(float1=var "Reset Time", float2=0, tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    Wait(time=var "Reset Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait For Hero Clear
    on END → End
  }

  state Come Up {
    SetMeshRenderer(gameObject=Self, active=true)
    PlayAudioEvent(audioClip=bone_trap_button_rise (sfxstatic_assets_areaant.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Plate Reset", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Reset {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Step Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Delay {
    Wait(time=var "Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate Device
  }

  state End {
  }

  state Wait For Hero Clear {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    on FINISHED → Come Up
  }

  state Cam Check {
    BoolTest(boolVariable=var "Check In Camera", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CheckOutOfCamera(gameObject=Self, margin=3, outsideEvent=→"CANCEL", insideEvent=(none), insideBool=false, outsideBool=false, everyFrame=false)
    on FINISHED → Step
    on CANCEL → Idle
  }

  var Reset Time: float = 0
  var Device: gameObject = Hunter Trap Cage

  var Delay: float = 0.5
  var Distance: float = 0
  var Activated: bool = false
  var Check In Camera: bool = false
  var Step Effects: gameObject = Hunter Trap Plate/Step Effects
}
