# FSM
# 1 location(s):
#   Coral_02 / shell_plat shortcut/shell_plat_hang_adjustable shortcut

// uses template: mechanism_move_to_y
fsm FSM {
  start Idle

  state Idle {
    on ACTIVATE → Activate Pause
    on OPEN → Antic
  }

  state Activate {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Destination Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SWAY PLAT START", delay=0, everyFrame=false)
  }

  state Antic {
    GetParent(gameObject=Self, storeResult=var "")
    ObjectJitterLocal(gameObject=Self, x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SWAY PLAT STOP", delay=0, everyFrame=false)
    on FINISHED → Dir
  }

  state Move Up {
    SetVelocity2d(gameObject=Self, vector=(0, 10), x=(unset), y=(unset), everyFrame=false)  // disabled
    Translate(gameObject=Self, vector=(unset), x=(unset), y=10, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Destination Y", tolerance=0, equal=→"END", lessThan=(none), greaterThan=→"END", everyFrame=true)
    on END → Arrive Sfx
  }

  state Arrive {
    AudioStop(gameObject=Self, fadeTime=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Destination Y", Space=0, EveryFrame=false)
    ObjectJitterLocal(gameObject=Self, x=0.1, y=0.1, z=(unset), resetPosOnStateExit=true, limitFps=0)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SWAY PLAT START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="MOVE STOP", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Dir {
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="MOVE START", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MOVE START", delay=0, everyFrame=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=false)
    FloatCompare(float1=var "Self Y", float2=var "Destination Y", tolerance=0, equal=→"U", lessThan=→"U", greaterThan=→"D", everyFrame=false)
    on U → Move Up
    on D → Move Down
  }

  state Move Down {
    SetVelocity2d(gameObject=Self, vector=(0, -10), x=(unset), y=(unset), everyFrame=false)  // disabled
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-10, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Destination Y", tolerance=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=true)
    on END → Arrive Sfx
  }

  state End {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MECHANISM MOVE END", delay=0, everyFrame=false)
  }

  state Arrive Sfx {
    PlayAudioEvent(audioClip=dock_chain_rock_plat_end_move (sfxstatic_assets_areadocks.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    BoolTest(boolVariable=var "Plat Sfx", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Arrive
  }

  state Activate Pause {
    Wait(time=var "Activate Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  var Destination Y: float = 27.29
  var Activate Pause: float = 1
  var Event Receiver: gameObject = shell_plat shortcut

  var Self Y: float = 0
  var Plat Sfx: bool = false
  var Plat Sprite: gameObject = Barrel Plat Lift/Plat Sprite (localpoolprefabs_assets_areacoral.bundle)
}
