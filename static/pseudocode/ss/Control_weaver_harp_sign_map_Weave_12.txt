# Control
# 1 location(s):
#   Weave_12 / weaver_harp_sign_map

fsm Control {
  start Pause
  on MAP TAKEN → Map Taken  // from any state
  on MAP TAKE → Map Take  // from any state

  state Pause {
    ActivateGameObject(gameObject=var "Inspect Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Active?
  }

  state Active? {
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0)
    FadeNestedFadeGroup(Target=var "Glow Text", ToAlpha=0, FadeTime=0)
    GameObjectCompare(gameObjectVariable=var "Read Persistent", compareTo=<null>, equalEvent=→"TRUE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    GetPersistentBool(target=var "Read Persistent", storeValue=var "Activated")
    BoolTest(boolVariable=var "Activated", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Inactive
    on TRUE → Inspectable
  }

  state Inactive {
    AudioStop(gameObject=Self, fadeTime=0)
    on POWER CHAMBER UNLOCK → Activate Burst
  }

  state Activate Burst {
    PlayAudioEvent(audioClip=weaver_harp_sign_burst (sfxstatic_assets_areaweaver.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Flashes", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ScreenFlash(flashColour=(1, 1, 1, 0.516))
    FadeColorFader(target=var "Summon Glow", fadeType=DOWN, useChildren=true)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    SendEventByName(eventTarget=GameObject(var "Notify"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0.2)
    on FINISHED → Text Up
  }

  state Text Up {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FadeNestedFadeGroup(Target=var "Glow Text", ToAlpha=1, FadeTime=0.5)
    ActivateGameObject(gameObject=var "Glow Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "text_bg", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Wisp", emit=0, resetIfPlaying=false)
    on FINISHED → Inspectable
  }

  state Readable {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.3, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on NEEDOLIN → Idle Strum
  }

  state Inspectable {
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0)
    FadeNestedFadeGroup(Target=var "Glow Text", ToAlpha=1, FadeTime=0)
    ActivateGameObject(gameObject=var "Glow Text", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "text_bg", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → Readable
  }

  state Map Taken {
    ActivateGameObject(gameObject=var "Glow Text", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Wisp", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Summon Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Strings", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Map Take {
    FadeColorFader(target=var "Glow Text", fadeType=DOWN, useChildren=true)
    FadeNestedFadeGroup(Target=var "Glow Text", ToAlpha=0, FadeTime=0.2)
    StopParticleEmitter(gameObject=var "Pt Wisp")
    FadeColorFader(target=var "Summon Glow", fadeType=DOWN, useChildren=true)
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=0, FadeTime=2)
  }

  state Idle Strum {
    ActivateInteractible(Target=var "Inspect Region", Activate=false, AllowQueueing=false, UseChildren=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=1, FadeTime=2)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.3, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumStart())
    on CANCEL → Not In Range
    on END → Idle Strum End
  }

  state Not In Range {
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Readable
  }

  state Idle Strum End {
    ActivateInteractible(Target=var "Inspect Region", Activate=true, AllowQueueing=false, UseChildren=false)
    SendMessage(gameObject=var "Strings", delivery=SendMessage, options=1, functionCall=StrumEnd())
    FadeNestedFadeGroup(Target=var "Strings", ToAlpha=var "Strings Group Idle Alpha", FadeTime=0.5)
    on FINISHED → Readable
  }

  var Inspect Region: gameObject = weaver_harp_sign_map/Get Map Inspect
  var Notify: gameObject = <null>
  var Read Persistent: gameObject = weaver_lift_power_chamber

  var Strings Group Idle Alpha: float = 0.25
  var Activated: bool = false
  var Flashes: gameObject = weaver_harp_sign_map/flashes
  var Glow Text: gameObject = weaver_harp_sign_map/glow_text
  var Pt Wisp: gameObject = weaver_harp_sign_map/Pt Wisp
  var Strings: gameObject = weaver_harp_sign_map/Strings
  var Summon Glow: gameObject = weaver_harp_sign_map/Summon Glow
  var text_bg: gameObject = weaver_harp_sign_map/text_bg
}
