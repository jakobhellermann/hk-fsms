  start Init

  state Init {
    on LOOK START → Swivel L
  }

  state Swivel L {

  }

    AnimateRotationTo(gameObject=Self, fromValue=var "Start Rotation", toValue=5, worldSpace=false, negativeSpace=false, time=1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
