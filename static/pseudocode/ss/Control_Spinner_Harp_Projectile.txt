# Control
# 1 location(s):
#   localpoolprefabs_assets_areabell.bundle / Spinner_Harp_Projectile

fsm Control {
  start Init
  on ROOF → Roof Dissipate  // from any state

  state DL {
    SetVelocityAsAngle(gameObject=Self, angle=225, speed=var "Speed", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=225, space=0, everyFrame=false, lateUpdate=false)

    SetBoolValue(boolVariable=var "Vert Up", boolValue=false, everyFrame=false)
    CheckYPosition(gameObject=Self, compareTo=var "Floor Y", compareToOffset=0, tolerance=0, equal=→"FLOOR", lessThan=→"FLOOR", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))

    on FLOOR → Floor Impact
    on WALL L → Wall Impact L
  }

  state Init {
    SendEventByName(eventTarget=Self, sendEvent="DL", delay=0, everyFrame=false)  // disabled
    on DL → DL
    on DR → DR
  }

  state Roof Dissipate {
    RecycleSelf()
  }

    SetPosition(gameObject=var "Impact", vector=(unset), x=(unset), y=var "Floor Y", z=0.008, space=0, everyFrame=false, lateUpdate=false)
    BoolTest(boolVariable=var "Hor Right", isTrue=→"UR", isFalse=→"UL", everyFrame=false)
    on UL → UL
    on UR → UR
  }

  state UL {

  }

    on UR → UR
    on DR → DR
  }
