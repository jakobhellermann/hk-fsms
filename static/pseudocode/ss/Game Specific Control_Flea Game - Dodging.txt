# Game Specific Control
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Flea Game - Dodging

fsm Game Specific Control {
  start Init
  on GAME END PLAY → Reset Game  // from any state
  on GAME FAILED → Stop Spawning  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Bellfleas")
    FindNamedChild(gameObject=Self, storeResult=var "Bellflea Positions Horizontal")
    FindNamedChild(gameObject=Self, storeResult=var "Bellflea Positions Horizontal High")
    FindNamedChild(gameObject=Self, storeResult=var "Bellflea Positions Top")
    FindNamedChild(gameObject=Self, storeResult=var "Bellflea Positions Stomp")
    FindNamedChild(gameObject=Self, storeResult=var "Giant Flea DodgeGame")
    FindNamedChild(gameObject=Self, storeResult=var "Flea Hunter DodgeGame")
    FindNamedChild(gameObject=Self, storeResult=var "Active While Playing")
    FindNamedChild(gameObject=var "Active While Playing", storeResult=var "Dodge Plat L")
    FindNamedChild(gameObject=var "Active While Playing", storeResult=var "Dodge Plat R")
    ActivateGameObject(gameObject=var "Timer Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=var "Timer Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIntValue(intVariable=var "Score", intValue=0, everyFrame=false)
    on GAME BEGIN PLAY → Start Pause
    on PREPARE FLEA GAME → Prepare Fleas
  }

  state Start Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Flea Speed", floatValue=var "Flea Speed Initial", everyFrame=false)
    SetIntValue(intVariable=var "Score", intValue=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Started Timer", boolValue=false, everyFrame=false)
    on FINISHED → Horizontal High?
  }

  sequence state Reset Game {
    SendEventToRegister(eventName="RESET FLEA GAMES")
    NextFrameEvent(sendEvent=(none))
    ActivateGameObject(gameObject=var "Bellfleas", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inactive
  }

  state Stop Spawning {
    ActivateGameObject(gameObject=var "Timer Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="FLEA GAME FAILED")
  }

  state Pause {
    FloatAdd(floatVariable=var "Spawn Set Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Spawn Set Timer", float2=var "Spawn Set Pause", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SPAWN", everyFrame=true)
    on SPAWN → Plat Shift?
  }

  state Choice {
    SetStringValue(stringVariable=var "Event Choice 1", stringValue="CHARGE L", everyFrame=false)
    SetStringValue(stringVariable=var "Event Choice 2", stringValue="CHARGE R", everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"SPECIAL", delay=0, everyFrame=false)  // disabled
    IntCompare(integer1=var "Score", integer2=6, equal=(none), lessThan=→"ANY 3", greaterThan=(none), everyFrame=false)
    IntTestToBool(int1=var "Score", int2=30, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Spawn More Fleas", everyFrame=false)
    IntTestToBool(int1=var "Score", int2=60, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Spawn Most Fleas", everyFrame=false)
    SendRandomEventV4(events=[→"ANY 5", →"ANY 5", →"DIAG", →"DOWN", →"SPECIAL"], weights=[1, 1, 1, 1, 0.25], eventMax=[2, 2, 2, 1, 1], missedMax=[5, 5, 5, 6, 6], activeBool=var "Spawn Most Fleas")
    SendRandomEventV4(events=[→"ANY 4", →"ANY 5", →"DIAG", →"DOWN", →"SPECIAL"], weights=[1, 1, 1, 1, 0.25], eventMax=[2, 2, 2, 1, 1], missedMax=[5, 5, 5, 6, 6], activeBool=var "Spawn More Fleas")
    SendRandomEventV4(events=[→"ANY 3", →"ANY 4", →"DIAG", →"DOWN", →"SPECIAL"], weights=[1, 1, 1, 1, 0.25], eventMax=[2, 2, 2, 1, 1], missedMax=[5, 5, 5, 6, 6], activeBool=(unset))
    on ANY 3 → Set Any 3
    on ANY 4 → Set Any 4
    on ANY 5 → Set Any 5
    on DIAG → Set Diag
    on DOWN → Set Down
    on SPECIAL → Special
  }

  state Set Any 3 {
    SetIntValue(intVariable=var "Fleas To Spawn", intValue=3, everyFrame=false)
    SendRandomEventV4(events=[→"TIMING EVEN", →"TIMING EARLY", →"TIMING LATE"], weights=[1, 1, 1], eventMax=[2, 1, 1], missedMax=[3, 4, 4], activeBool=(unset))
    on TIMING EVEN → 3 Even
    on TIMING EARLY → 3 Early
    on TIMING LATE → 3 Late
  }

  state 3 Even {
    SetFloatValue(floatVariable=var "Spawn Pause 1", floatValue=0.75, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 2", floatValue=0.75, everyFrame=false)
    on FINISHED → Reset Positions
  }

  state Reset Positions {
    SendEventToRegister(eventName="RESET DODGE POSITIONS")
    on FINISHED → Spawn Fleas
  }

  state 3 Early {
    SetFloatValue(floatVariable=var "Spawn Pause 1", floatValue=0.5, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 2", floatValue=1, everyFrame=false)
    on FINISHED → Reset Positions
  }

  state 3 Late {
    SetFloatValue(floatVariable=var "Spawn Pause 1", floatValue=1, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 2", floatValue=0.5, everyFrame=false)
    on FINISHED → Reset Positions
  }

  state Set Any 4 {
    SetIntValue(intVariable=var "Fleas To Spawn", intValue=4, everyFrame=false)
    SendRandomEventV4(events=[→"TIMING EVEN", →"TIMING SPLIT"], weights=[1, 1], eventMax=[2, 1], missedMax=[3, 4], activeBool=(unset))
    on TIMING EVEN → 4 Even
    on TIMING SPLIT → 4 Split
  }

  state 4 Even {
    SetFloatValue(floatVariable=var "Spawn Pause 1", floatValue=0.75, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 2", floatValue=0.75, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 3", floatValue=0.75, everyFrame=false)
    on FINISHED → Reset Positions
  }

  state 4 Split {
    SetFloatValue(floatVariable=var "Spawn Pause 1", floatValue=0.5, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 2", floatValue=1.25, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 3", floatValue=0.5, everyFrame=false)
    on FINISHED → Reset Positions
  }

  sequence state Spawn Fleas {
    GetRandomChild(gameObject=var "Bellflea Positions Crt", storeResult=var "Next Pos")
    GetPosition2D(gameObject=var "Next Pos", vector=var "Spawn Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetParent(gameObject=var "Next Pos", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRandomChild(gameObject=var "Bellfleas", storeResult=var "Next Flea")
    SetPosition2D(GameObject=var "Next Flea", Vector=var "Spawn Pos", X=(unset), Y=var "Spawn Y", Space=1, EveryFrame=false)
    SelectRandomString(strings=[var "Event Choice 1", var "Event Choice 2"], weights=[1, 1], storeString=var "Flea Event")
    SendEventByName(eventTarget=GameObject(var "Next Flea"), sendEvent=var "Flea Event", delay=0, everyFrame=false)
    Wait(time=var "Spawn Pause 1", finishEvent=(none), realTime=false)
    GetRandomChild(gameObject=var "Bellflea Positions Crt", storeResult=var "Next Pos")
    GetPosition2D(gameObject=var "Next Pos", vector=var "Spawn Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetParent(gameObject=var "Next Pos", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRandomChild(gameObject=var "Bellfleas", storeResult=var "Next Flea")
    SetFsmBool(gameObject=var "Next Flea", fsmName="Control", variableName="Quickened", setValue=var "Will Quicken", everyFrame=false)
    SetPosition2D(GameObject=var "Next Flea", Vector=var "Spawn Pos", X=(unset), Y=var "Spawn Y", Space=1, EveryFrame=false)
    SelectRandomString(strings=[var "Event Choice 1", var "Event Choice 2"], weights=[1, 1], storeString=var "Flea Event")
    SendEventByName(eventTarget=GameObject(var "Next Flea"), sendEvent=var "Flea Event", delay=0, everyFrame=false)
    Wait(time=var "Spawn Pause 2", finishEvent=(none), realTime=false)
    GetRandomChild(gameObject=var "Bellflea Positions Crt", storeResult=var "Next Pos")
    GetPosition2D(gameObject=var "Next Pos", vector=var "Spawn Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetParent(gameObject=var "Next Pos", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRandomChild(gameObject=var "Bellfleas", storeResult=var "Next Flea")
    SetPosition2D(GameObject=var "Next Flea", Vector=var "Spawn Pos", X=(unset), Y=var "Spawn Y", Space=1, EveryFrame=false)
    SelectRandomString(strings=[var "Event Choice 1", var "Event Choice 2"], weights=[1, 1], storeString=var "Flea Event")
    SendEventByName(eventTarget=GameObject(var "Next Flea"), sendEvent=var "Flea Event", delay=0, everyFrame=false)
    IntCompare(integer1=var "Fleas To Spawn", integer2=4, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    Wait(time=var "Spawn Pause 3", finishEvent=(none), realTime=false)
    GetRandomChild(gameObject=var "Bellflea Positions Crt", storeResult=var "Next Pos")
    GetPosition2D(gameObject=var "Next Pos", vector=var "Spawn Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetParent(gameObject=var "Next Pos", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRandomChild(gameObject=var "Bellfleas", storeResult=var "Next Flea")
    SetPosition2D(GameObject=var "Next Flea", Vector=var "Spawn Pos", X=(unset), Y=var "Spawn Y", Space=1, EveryFrame=false)
    SelectRandomString(strings=[var "Event Choice 1", var "Event Choice 2"], weights=[1, 1], storeString=var "Flea Event")
    SendEventByName(eventTarget=GameObject(var "Next Flea"), sendEvent=var "Flea Event", delay=0, everyFrame=false)
    IntCompare(integer1=var "Fleas To Spawn", integer2=5, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    Wait(time=var "Spawn Pause 4", finishEvent=(none), realTime=false)
    GetRandomChild(gameObject=var "Bellflea Positions Crt", storeResult=var "Next Pos")
    GetPosition2D(gameObject=var "Next Pos", vector=var "Spawn Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetParent(gameObject=var "Next Pos", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRandomChild(gameObject=var "Bellfleas", storeResult=var "Next Flea")
    SetPosition2D(GameObject=var "Next Flea", Vector=var "Spawn Pos", X=(unset), Y=var "Spawn Y", Space=1, EveryFrame=false)
    SelectRandomString(strings=[var "Event Choice 1", var "Event Choice 2"], weights=[1, 1], storeString=var "Flea Event")
    SendEventByName(eventTarget=GameObject(var "Next Flea"), sendEvent=var "Flea Event", delay=0, everyFrame=false)
    on FINISHED → Set Pause Time
  }

  state Set Diag {
    SetStringValue(stringVariable=var "Event Choice 1", stringValue="DIAG", everyFrame=false)
    SetStringValue(stringVariable=var "Event Choice 2", stringValue="DIAG", everyFrame=false)
    SetGameObject(variable=var "Bellflea Positions Crt", gameObject=var "Bellflea Positions Top", everyFrame=false)
    on FINISHED → Set Any 4
  }

  state Set Down {
    SetStringValue(stringVariable=var "Event Choice 1", stringValue="DOWN", everyFrame=false)
    SetStringValue(stringVariable=var "Event Choice 2", stringValue="DOWN", everyFrame=false)
    SetGameObject(variable=var "Bellflea Positions Crt", gameObject=var "Bellflea Positions Stomp", everyFrame=false)
    on FINISHED → Set Any 4
  }

  state Plat Shift? {
    IntAdd(intVariable=var "Sets Until Plat Shift", add=-1, everyFrame=false)
    IntCompare(integer1=var "Sets Until Plat Shift", integer2=0, equal=→"SHIFT", lessThan=→"SHIFT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Quicken?
    on SHIFT → Plat Config Check
  }

  state Plat Config Check {
    SetIntValue(intVariable=var "Sets Until Plat Shift", intValue=3, everyFrame=false)
    IntSwitch(intVariable=var "Plat Config", compareTo=[1, 2, 3], sendEvent=[→"BOTH UP", →"LEFT UP", →"RIGHT UP"], everyFrame=false)
    on BOTH UP → Both Up
    on LEFT UP → Left Up
    on RIGHT UP → Right Up
  }

  state Both Up {
    SendRandomEventV4(events=[→"LEFT UP", →"RIGHT UP"], weights=[0.5, 0.5], eventMax=[1, 1], missedMax=[1, 1], activeBool=(unset))
    on LEFT UP → To Left Up
    on RIGHT UP → To Right Up
  }

  state To Left Up {
    SetIntValue(intVariable=var "Plat Config", intValue=2, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Dodge Plat R"), sendEvent="LOWER", delay=0, everyFrame=false)
    SelectRandomString(strings=["RAISE HIGH", "RAISE LOW"], weights=[1, 1], storeString=var "Plat Event")
    SendEventByName(eventTarget=GameObject(var "Dodge Plat L"), sendEvent=var "Plat Event", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Sets Until Plat Shift", intValue=2, everyFrame=false)
    on FINISHED → Plat Shift End
  }

  state Plat Shift End {
    on FINISHED → Quicken?
  }

  state To Right Up {
    SetIntValue(intVariable=var "Plat Config", intValue=3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Dodge Plat L"), sendEvent="LOWER", delay=0, everyFrame=false)
    SelectRandomString(strings=["RAISE HIGH", "RAISE LOW"], weights=[1, 1], storeString=var "Plat Event")
    SendEventByName(eventTarget=GameObject(var "Dodge Plat R"), sendEvent=var "Plat Event", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Sets Until Plat Shift", intValue=2, everyFrame=false)
    on FINISHED → Plat Shift End
  }

  state To Both Up {
    SetIntValue(intVariable=var "Plat Config", intValue=1, everyFrame=false)
    SelectRandomString(strings=["RAISE HIGH", "RAISE LOW"], weights=[1, 1], storeString=var "Plat Event")
    SendEventByName(eventTarget=GameObject(var "Dodge Plat L"), sendEvent=var "Plat Event", delay=0, everyFrame=false)
    SelectRandomString(strings=["RAISE HIGH", "RAISE LOW"], weights=[1, 1], storeString=var "Plat Event")
    SendEventByName(eventTarget=GameObject(var "Dodge Plat R"), sendEvent=var "Plat Event", delay=0, everyFrame=false)
    SetIntValue(intVariable=var "Sets Until Plat Shift", intValue=1, everyFrame=false)
    on FINISHED → Plat Shift End
  }

  state Left Up {
    SendRandomEventV4(events=[→"BOTH UP", →"RIGHT UP"], weights=[0.5, 0.5], eventMax=[1, 1], missedMax=[1, 1], activeBool=(unset))
    on BOTH UP → To Both Up
    on RIGHT UP → To Right Up
  }

  state Right Up {
    SendRandomEventV4(events=[→"LEFT UP", →"BOTH UP"], weights=[0.5, 0.5], eventMax=[1, 1], missedMax=[1, 1], activeBool=(unset))
    on BOTH UP → To Both Up
    on LEFT UP → To Left Up
  }

  state Quicken? {
    SetBoolValue(boolVariable=var "Will Quicken", boolValue=false, everyFrame=false)
    IntCompare(integer1=var "Score", integer2=12, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.8, 0.2], eventMax=[4, 1], missedMax=[1, 4], activeBool=(unset))
    SetBoolValue(boolVariable=var "Will Quicken", boolValue=true, everyFrame=false)
    on FINISHED → Horizontal High?
  }

  state Set Pause Time {
    GetFsmInt(gameObject=Self, fsmName="flea_game_master_control", variableName="Score", storeValue=var "Score", everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Set Timer", floatValue=0, everyFrame=false)
    ConvertIntToFloat(intVariable=var "Score", floatVariable=var "Score Float", everyFrame=false)
    FloatCompare(float1=var "Score Float", float2=0, tolerance=0, equal=→"MIN", lessThan=→"MIN", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Score Float", float2=var "Max Difficulty Score", tolerance=0, equal=→"MAX", lessThan=(none), greaterThan=→"MAX", everyFrame=false)
    FloatOperator(float1=var "Score Float", float2=var "Max Difficulty Score", operation=Divide, storeResult=var "Pause Decrement", everyFrame=false)
    FloatMultiply(floatVariable=var "Pause Decrement", multiplyBy=1.5, everyFrame=false)
    FloatOperator(float1=1.75, float2=var "Pause Decrement", operation=Subtract, storeResult=var "Spawn Set Pause", everyFrame=false)
    on FINISHED → Pause
    on MAX → Max Difficulty
    on MIN → Min Difficulty
  }

  sequence state Giant {
    Wait(time=1, finishEvent=(none), realTime=false)
    SendEventByName(eventTarget=GameObject(var "Giant Flea DodgeGame"), sendEvent="ENTER", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    SetIntValue(intVariable=var "Sets Until Plat Shift", intValue=2, everyFrame=false)
    on FINISHED → Set Pause Time
  }

  state Special {
    SendEvent(eventTarget=Self, sendEvent=→"HUNTER", delay=0, everyFrame=false)  // disabled
    SendRandomEventV4(events=[→"GIANT", →"HUNTER"], weights=[0.5, 0.5], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on GIANT → Giant
    on HUNTER → Hunter
  }

  sequence state Hunter {
    Wait(time=0.75, finishEvent=(none), realTime=false)
    SendEventByName(eventTarget=GameObject(var "Flea Hunter DodgeGame"), sendEvent="ENTER", delay=0, everyFrame=false)
    on FINISHED → Set Pause Time
  }

  state Max Difficulty {
    SetFloatValue(floatVariable=var "Spawn Set Pause", floatValue=0.25, everyFrame=false)
    on FINISHED → Pause
  }

  state Min Difficulty {
    SetFloatValue(floatVariable=var "Spawn Set Pause", floatValue=1.75, everyFrame=false)
    on FINISHED → Pause
  }

  state Start Timer {
    BoolTest(boolVariable=var "Started Timer", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Timer Obj", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Started Timer", boolValue=true, everyFrame=false)
    on FINISHED → Horizontal High?
  }

  state Set Any 5 {
    SetIntValue(intVariable=var "Fleas To Spawn", intValue=5, everyFrame=false)
    SendRandomEventV4(events=[→"TIMING EVEN", →"TIMING SPLIT"], weights=[1, 1], eventMax=[2, 1], missedMax=[3, 4], activeBool=(unset))
    on TIMING EVEN → 5 Even
    on TIMING SPLIT → 5 Split
  }

  state 5 Even {
    SetFloatValue(floatVariable=var "Spawn Pause 1", floatValue=0.6, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 2", floatValue=0.6, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 3", floatValue=0.6, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 4", floatValue=0.6, everyFrame=false)
    on FINISHED → Reset Positions
  }

  state 5 Split {
    SetFloatValue(floatVariable=var "Spawn Pause 1", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 2", floatValue=0.9, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 3", floatValue=0.4, everyFrame=false)
    SetFloatValue(floatVariable=var "Spawn Pause 4", floatValue=0.9, everyFrame=false)
    on FINISHED → Reset Positions
  }

  state Horizontal High? {
    SetGameObject(variable=var "Bellflea Positions Crt", gameObject=var "Bellflea Positions Horizontal", everyFrame=false)
    GetFsmBool(gameObject=var "Dodge Plat L", fsmName="Control", variableName="Is Low", storeValue=var "Plat L Is Low", everyFrame=false)
    GetFsmBool(gameObject=var "Dodge Plat R", fsmName="Control", variableName="Is Low", storeValue=var "Plat R Is Low", everyFrame=false)
    BoolNoneTrue(boolVariables=[var "Plat L Is Low", var "Plat R Is Low"], sendEvent=(none), storeResult=var "No Low Plats", everyFrame=false)
    BoolTest(boolVariable=var "No Low Plats", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetGameObject(variable=var "Bellflea Positions Crt", gameObject=var "Bellflea Positions Horizontal High", everyFrame=false)
    on FINISHED → Choice
  }

  state Prepare Fleas {
    ActivateGameObject(gameObject=var "Bellfleas", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bellflea Positions Horizontal", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bellflea Positions Horizontal High", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bellflea Positions Stomp", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bellflea Positions Top", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flea Hunter DodgeGame", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Giant Flea DodgeGame", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inactive
  }

  var Flea Speed: float = 4
  var Flea Speed Initial: float = 4
  var Spawn Pause 1: float = 0
  var Spawn Pause 2: float = 0
  var Spawn Pause 3: float = 0
  var Spawn Y: float = 0
  var Spawn Set Pause: float = 0.5
  var Spawn Set Timer: float = 0
  var Max Difficulty Score: float = 70
  var Score Float: float = 0
  var Pause Decrement: float = 0
  var Spawn Pause 4: float = 0
  var Fleas To Spawn: int = 0
  var Plat Config: int = 1
  var Score: int = 0
  var Sets Until Plat Shift: int = 3
  var Will Quicken: bool = false
  var Spawn More Fleas: bool = false
  var Spawn Most Fleas: bool = false
  var Plat L Is Low: bool = false
  var Plat R Is Low: bool = false
  var No Low Plats: bool = false
  var Started Timer: bool = false
  var Flea Event: string = ""
  var Event Choice 1: string = ""
  var Event Choice 2: string = ""
  var Plat Event: string = ""
  var Spawn Pos: vector2 = (0, 0)
  var Bellflea Positions Crt: gameObject = <null>
  var Bellflea Positions Horizontal: gameObject = <null>
  var Bellflea Positions Stomp: gameObject = <null>

  var Bellflea Positions Horizontal High: gameObject = <null>
}
