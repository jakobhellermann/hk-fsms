# Idle Anim
# 1 location(s):
#   Sprintmaster_Cave / Race Group/Sprintmaster Runner

fsm Idle Anim {
  start Init
  on RESTART → Idle  // from any state
  on STOP → Inert  // from any state
  on WIN RACE → Win Race  // from any state
  on RESTART TO STRETCH → To Stretch  // from any state
  on FADE VOICE → Fade Voice  // from any state

  state Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle Jog", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetAudioClip(gameObject=Self, audioClip=sprintmaster_dial_light_jog_on_spot_loop (sfxstatic_assets_areaaqueduct.bundle), autoPlay=true, stopOnExit=true)
    WaitRandom(timeMin=2, timeMax=5, finishEvent=→"STRETCH", realTime=false)
    SetRandomAudioClipFromTable(Table=SM_idle_jog_loop (sfxdynamic_assets_sprintmaster.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0.35)
    SelectRandomFloat(floats=[0, 2.7, 5.57], weights=[1, 1, 1], storeFloat=var "Audio Time")
    SetProperty(targetProperty=AudioSource.time on Race Group/Sprintmaster Runner/Audio Loop Voice@AudioSource, everyFrame=false)
    on STRETCH → Stretch
  }

  state Inert {
  }

  sequence state Stretch {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=sprintmaster_land_and_get_down_to_stretch (sfxstatic_assets_areaaqueduct.bundle))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=sprintmaster_dial_stretch (sfxstatic_assets_areaaqueduct.bundle))
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=SM_stretch (sfxdynamic_assets_sprintmaster.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SelectRandomString(strings=["Down", "Up"], weights=[0.5, 0.5], storeString=var "Stretch Anim Dir")
    SelectRandomString(strings=["L", "R"], weights=[0.5, 0.5], storeString=var "Stretch Anim Facing")
    FormatString(format="Stretch {0} {1}", variables=[var "Stretch Anim Dir", var "Stretch Anim Facing"], storeResult=var "Stretch Anim", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Stretch Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    WaitRandom(timeMin=2, timeMax=3, finishEvent=(none), realTime=false)
    BuildString(stringParts=[var "Stretch Anim", " End"], separator="", addToEnd=false, storeResult=var "Stretch Anim", everyFrame=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName=var "Stretch Anim", AnimationCompleteEvent=(none))
    on FINISHED → Idle
  }

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    on FINISHED → Idle
  }

  sequence state Win Race {
    Tk2dPlayAnimationWait(Target=Self, ClipName="Antic Skid", AnimationCompleteEvent=(none))
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=SM_race_win (sfxdynamic_assets_sprintmaster.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Victory Pose", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=2, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName="Victory Pose End", AnimationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Jog")
  }

  sequence state To Stretch {
    Wait(time=0.25, finishEvent=(none), realTime=false)
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → Stretch
  }

  state Fade Voice {
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=1, cancelOnEarlyExit=false)
    Wait(time=1, finishEvent=→"NEXT", realTime=false)
    on NEXT → Inert
  }

  var Audio Time: float = 0
  var Stretch Anim Dir: string = ""
  var Stretch Anim Facing: string = ""
  var Stretch Anim: string = ""
  var Self: gameObject = <null>
  var Audio Loop Voice: gameObject = Race Group/Sprintmaster Runner/Audio Loop Voice
}
