# Repairing Scene
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/RestBench Control

fsm Repairing Scene {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Bench Getting Repaired")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="skullKingBenchMended", isTrue=→"INACTIVE", isFalse=(none))
    PlayerDataBoolTest(boolName="skullKingKilled", isTrue=→"ACTIVE", isFalse=→"INACTIVE")
    on ACTIVE → Active
    on INACTIVE → Inactive
  }

  state Active {
    ActivateGameObject(gameObject=var "Bench Getting Repaired", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="SPECIAL FIXER")
    SendEventToRegister(eventName="BENCH REPAIRING")
  }

  state Inactive {
    ActivateGameObject(gameObject=var "Bench Getting Repaired", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Bench Getting Repaired: gameObject = <null>
}
