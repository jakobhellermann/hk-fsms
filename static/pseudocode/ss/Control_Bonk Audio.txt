# Control
# 1 location(s):
#   Peak_Mask_Maker / Peak Mask Maker/Bonk Audio

fsm Control {
  start Idle

  state Idle {
    on BONK → Masked?
    on BIG BONK → Big Bonk Audio
  }

  state Masked? {
    GetFsmBool(gameObject=var "Parent", fsmName="Dialogue", variableName="Is Naked", storeValue=var "Is Naked", everyFrame=false)
    BoolTest(boolVariable=var "Is Naked", isTrue=→"UNMASKED", isFalse=→"MASKED", everyFrame=false)
    on MASKED → Masked
    on UNMASKED → Unmasked
  }

  state Masked {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=MM_masked_exclaim (sfxdynamic_assets_areapeak.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Idle
  }

  state Unmasked {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=MM_unmasked_hit (sfxdynamic_assets_areapeak.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Idle
  }

  state Big Bonk Audio {
    SetAudioVolume(gameObject=var "Audio Loop Voice", volume=1, everyFrame=false)
    PlayOneShotOnAudioSource(gameObject=var "Audio Loop Voice", audioClip=<null>, audioClipTable=MM_unmasked_hit (sfxdynamic_assets_areapeak.bundle), pitchOffset=0, volumeScale=2.5, stopPreviousSound=true, forcePlay=true)
    Wait(time=0.1, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  var Is Naked: bool = false
  var Parent: gameObject = Peak Mask Maker
  var Audio Loop Voice: gameObject = Peak Mask Maker/Audio Loop Voice
}
