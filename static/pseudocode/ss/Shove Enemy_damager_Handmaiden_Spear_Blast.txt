# Shove Enemy
# 1 location(s):
#   localpoolprefabs_assets_areasong.bundle / Handmaiden_Spear_Blast/damager

fsm Shove Enemy {
  start Idle

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"SHOVE", storeCollider=var "Enemy")
    on SHOVE → Shove
  }

  state Shove {
    GetAngleToTarget2D(gameObject=Self, target=var "Enemy", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    AddForce2dAsAngle(gameObject=var "Enemy", forceMode=1, atPosition=(unset), angle=var "Angle", speed=100, maxSpeed=0, maxSpeedX=0, maxSpeedY=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Angle: float = 0
  var Enemy: gameObject = <null>
}
