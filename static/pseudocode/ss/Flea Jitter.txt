# Flea Jitter
# 1 location(s):
#   Slab_Cell / Flea Slab Cage/lamp

fsm Flea Jitter {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Choice
  }

  state Shaking {
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    RandomFloat(min=250, max=500, storeResult=var "Speed")
    AddForce2dAsAngle(gameObject=Self, forceMode=0, atPosition=(unset), angle=var "Angle", speed=var "Speed", maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choice
  }

  state Shove {
    PlayAudioEventRandom(audioClips=[environment_chain_move_01 (sfxstatic_assets_chain.bundle), environment_chain_move_02 (sfxstatic_assets_chain.bundle), environment_chain_move_03 (sfxstatic_assets_chain.bundle), environment_chain_move_04 (sfxstatic_assets_chain.bundle), environment_chain_move_05 (sfxstatic_assets_chain.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    RandomFloatEither(value1=20, value2=160, storeResult=var "Angle")
    RandomFloat(min=2500, max=5000, storeResult=var "Speed")
    AddForce2dAsAngle(gameObject=Self, forceMode=0, atPosition=(unset), angle=var "Angle", speed=var "Speed", maxSpeed=(unset), maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    on FINISHED → Choice
  }

  state Choice {
    SendRandomEventV4(events=[→"SHOVE", →"FINISHED"], weights=[0.05, 0.95], eventMax=[1, 25], missedMax=[25, 1], activeBool=(unset))
    on SHOVE → Shove
    on FINISHED → Shaking
  }

  var Angle: float = 0
  var Speed: float = 0
}
