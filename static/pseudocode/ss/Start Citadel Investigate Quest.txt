# Start Citadel Investigate Quest
# 1 location(s):
#   Coral_10 / Citadel Quest Starter

fsm Start Citadel Investigate Quest {
  start Wait

  state Wait {
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Quest Started?
  }

  state Start Quest? {
    ArrayGet(array=var "Entry Gates", index=var "Index", storeValue=var "Testing Gate", everyFrame=false, indexOutOfRange=→"FALSE")
    GetName(gameObject=var "Testing Gate", storeName=var "Testing Gate Name", everyFrame=false)
    StringCompare(stringVariable=var "Testing Gate Name", compareTo=var "Entry Gate", equalEvent=→"TRUE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    IntAdd(intVariable=var "Index", add=1, everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"LOOP", delay=0, everyFrame=false)
    on LOOP → Start Quest?
    on FALSE → Inert
    on TRUE → Start Wait
  }

  state Inert {
    on AREA TITLE DISPLAYED → Start Quest
  }

  sequence state Start Quest {
    MenuStyleUnlockActionV2(UnlockKey="ENDING_B", ForceChange=false)
    SetPlayerDataVariable(VariableName="visitedCitadel", SetValue=true)
    BeginQuestV2(ShowPrompt=true, Quest=var "Quest")
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="TimePasses", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    PlayerDataBoolTest(boolName="act2Started", isTrue=→"TRUE", isFalse=(none))
    SetPlayerDataBool(boolName="act2Started", value=true)
    QueueSaveGameV2(createAutoSave=true, nameEnum=ACT_2)
    on TRUE → Normal Save
  }

  state Get Entry Gate {
    GetEntryGateName(StoreValue=var "Entry Gate")
    SetIntValue(intVariable=var "Index", intValue=0, everyFrame=false)
    on FINISHED → Start Quest?
  }

  state Quest Started? {
    CheckQuestState(NotTrackedEvent=→"FALSE", TrackedEvent=→"TRUE", CompletedEvent=→"TRUE", Quest=var "Quest")
    on FALSE → Get Entry Gate
    on TRUE → Inert 2
  }

  state Inert 2 {
  }

  state Start Wait {
    BoolTest(boolVariable=var "Wait For Display Event", isTrue=(none), isFalse=→"FALSE", everyFrame=false)
    on AREA TITLE DISPLAYED → Start Quest
    on FALSE → Delay
  }

  state Delay {
    Wait(time=2, finishEvent=(none), realTime=false)
    on FINISHED → Start Quest
  }

  state Normal Save {
    QueueSaveGameV2(createAutoSave=false, nameEnum=NONE)
  }

  var Wait For Display Event: bool = false
  var Entry Gates: array = var "Entry Gates"

  var Index: int = 0
  var Entry Gate: string = ""
  var Testing Gate Name: string = ""
  var Testing Gate: gameObject = <null>
  var Quest: object = Citadel Investigate (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle)
}
