# Death Control
# 1 location(s):
#   Organ_01 / Boss Scene/Phantom

fsm Death Control {
  start Pause

  state Idle {
    on ZERO HP → Ready
  }

  state Ready {
    GetEventSender(sentByGameObject=var "Event Sender")
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Final Phase", setValue=true, everyFrame=false)
    on BLOCKED HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FINAL BLOCK", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Event Sender: gameObject = <null>
}
