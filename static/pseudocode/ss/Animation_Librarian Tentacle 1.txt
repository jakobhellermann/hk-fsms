# Animation
# 3 location(s):
#   Belltown_Room_Relic / relic_dealer/tendrils/Librarian Tentacle 1
#   Library_08 / Librarian Tentacle 1 (1)
#   Library_08 / Librarian Tentacle 1

// uses template: librarian_tentancles_animation
fsm Animation {
  start Init

  state Init {
    AddEventRegister(target=Self, eventName="LIBRARIAN TENTACLE IDLE")
    AddEventRegister(target=Self, eventName="LIBRARIAN TENTACLE MOVE")
    GetAudioVolume(gameObject=Self, storeVolume=var "Loop Volume", everyFrame=false)
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    on FINISHED → Idling
  }

  state Moving {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Move Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetAudioVolume(gameObject=Self, volume=var "Loop Volume", everyFrame=false)
    on LIBRARIAN TENTACLE IDLE → Keep Moving Random
  }

  state Idling {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Idle Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetAudioVolume(gameObject=Self, volume=0, everyFrame=false)
    on LIBRARIAN TENTACLE MOVE → Keep Idle Random
  }

  state Keep Idle Random {
    RandomFloat(min=0, max=0.3, storeResult=var "Wait Time")
    WaitTimeAndTk2dFrame(Tk2dAnimator=Self, Time=var "Wait Time", FinishEvent=→"FINISHED")
    on LIBRARIAN TENTACLE IDLE → Idling
    on FINISHED → Moving
  }

  state Keep Moving Random {
    RandomFloat(min=0.2, max=0.5, storeResult=var "Wait Time")
    WaitTimeAndTk2dFrame(Tk2dAnimator=Self, Time=var "Wait Time", FinishEvent=→"FINISHED")
    on FINISHED → Idling
    on LIBRARIAN TENTACLE MOVE → Moving
  }

  var Idle Anim: string = "T1 Idle"
  var Move Anim: string = "T1 Move"

  var Wait Time: float = 0
  var Loop Volume: float = 0
}
