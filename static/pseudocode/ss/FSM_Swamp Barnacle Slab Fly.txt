# 1 location(s):
#   Aqueduct_04 / Swamp Barnacle Slab Fly

fsm FSM {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Hit")
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT → Hit
  }

  state Hit {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, -0.75, 0), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=enemy_damage_old (sfxstatic_assets_areabellshared.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatorPlay(gameObject=Self, stateName="", layer=(unset), normalizedTime=(unset), everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Hit", emit=0, resetIfPlaying=false)
    on FINISHED → Idle
  }

  var Self: gameObject = <null>
  var Pt Hit: gameObject = <null>
}
