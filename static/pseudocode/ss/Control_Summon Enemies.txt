# Control
# 2 location(s):
#   Ant_19 / Boss Control/Boss Scene/Summon Enemies
#   Bone_East_08_boss_beastfly / Boss Scene Beastfly/Summon Enemies

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on SUMMON → Summon
  }

  state Summon {
    GetRandomChild(gameObject=Self, storeResult=var "Enemy")
    SetPositionToObject(gameObject=var "Enemy", targetObject=var "Summon Point", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Enemy"), sendEvent="SUMMON", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Summon Point: gameObject = <null>
  var Enemy: gameObject = <null>
}
