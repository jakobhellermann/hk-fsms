# Control
# 2 location(s):
#   Clover_06 / White Cloverstag Encounter
#   Clover_16 / White Cloverstag Encounter

fsm Control {
  start Init

  state Init {
    FadeNestedFadeGroup(Target=var "Light", ToAlpha=1, FadeTime=0)
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle F")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="LEAVE", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=var "In Needolin Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=→"LEAVE", IgnoreNeedolinRange=true, useActiveBool=true, ActiveBool=var "In Needolin Range", StoreState=None, EveryFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=(none), ActiveOuter=→"LEAVE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"LEAVE", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    WaitRandom(timeMin=2, timeMax=4, finishEvent=→"CALL", realTime=false)
    on LEAVE → Leave Delay
    on CALL → Call Antic
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Leave Antic {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=white_cloverstag_short_cry (sfxdynamic_assets_areaclover.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)  // disabled
    SetPlayerDataBool(boolName="encounteredWhiteCloverstagMid", value=true)
    ActivateGameObject(gameObject=White Cloverstag Encounter/Orb Source, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Jump
  }

  state Jump {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump Away")
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SetVelocityByScale(gameObject=Self, speed=-15, ySpeed=18, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SetRendererSortingLayer(Target=Self, SortingLayerName="Over", SortingOrder=0)
    PlayAudioEventRandom(audioClips=[cloverstag_jump_1 (sfxstatic_assets_areaclover.bundle), cloverstag_jump_2 (sfxstatic_assets_areaclover.bundle), cloverstag_jump_3 (sfxstatic_assets_areaclover.bundle), cloverstag_jump_4 (sfxstatic_assets_areaclover.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Tele Out 1
  }

  state Tele Out 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    PlayAudioEvent(audioClip=radiance_teleport (sfxstatic_assets_areaclover.bundle), pitchMin=1.2, pitchMax=1.4, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → Tele Out 2
  }

  state Tele Out 2 {
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    DecelerateV2(gameObject=Self, deceleration=0.825, brakeOnExit=true)
    on FINISHED → Tele Out End
  }

  state Tele Out End {
    SpawnObjectFromGlobalPool(gameObject=white_stag_tele_flash (localpoolprefabs_assets_areaclover.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetMeshRenderer(gameObject=Self, active=false)
    FadeNestedFadeGroup(Target=var "Light", ToAlpha=0, FadeTime=0.2)
    on FINISHED → 
  }

  state Leave Delay {
    Wait(time=var "Leave Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Leave Antic
  }

  state Call Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Call", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="LEAVE", outOfRangeEvent="", everyFrame=true)
    on FINISHED → Call
    on LEAVE → Leave Antic
  }

  state Call {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="LEAVE", outOfRangeEvent="", everyFrame=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=white_cloverstag_short_cry (sfxdynamic_assets_areaclover.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Idle
    on LEAVE → Leave Antic
  }

  var Leave Delay: float = 0

  var In Needolin Range: bool = false
  var Self: gameObject = White Cloverstag Encounter
  var Light: gameObject = White Cloverstag Encounter/Light
  var Audio Loop Voice: gameObject = White Cloverstag Encounter/Audio Loop Voice
}
