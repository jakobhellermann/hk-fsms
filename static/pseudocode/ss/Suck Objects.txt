# Suck Objects
# 1 location(s):
#   Shadow_18 / Battle Scene/Wave 6 - Boss/Swamp Shaman/Suck Zone Object

fsm Suck Objects {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Suck Point")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"TRIGGER", storeCollider=var "Colliding Object")
    on TRIGGER → Suck
  }

  state Suck {
    HasComponent(gameObject=var "Colliding Object", component="Rigidbody2D", removeOnExit=false, trueEvent=(none), falseEvent=→"FINISHED", store=(unset), everyFrame=false)
    CustomTagIsSuckable(target=var "Colliding Object", isSuckableEvent=(none), isNotSuckableEvent=→"FINISHED", storeIsSuckable=false)
    GetAngleToTarget2D(gameObject=var "Colliding Object", target=var "Suck Point", offsetX=0, offsetY=0, storeAngle=var "Force Angle", pause=0, everyFrame=false)
    AddForce2dAsAngle(gameObject=var "Colliding Object", forceMode=0, atPosition=(unset), angle=var "Force Angle", speed=75, maxSpeed=50, maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=false)
    on FINISHED → Detect
  }

  var Force Angle: float = 0
  var Suck Point: gameObject = <null>
  var Colliding Object: gameObject = <null>
}
