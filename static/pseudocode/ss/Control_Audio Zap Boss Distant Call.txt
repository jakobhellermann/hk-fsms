# Control
# 1 location(s):
#   Coral_29 / Boss Scene/Audio Zap Boss Distant Call

fsm Control {
  start Out

  state Out {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    WaitRandom(timeMin=4, timeMax=6, finishEvent=→"CALL", realTime=false)
    on EXIT → Out
    on CALL → Call
  }

  state Call {
    AudioPlayRandom(gameObject=var "Self", audioClips=[coral_zap_core_vox_long_1_maybe_during_attack (sfxstatic_assets_areacoral.bundle), coral_zap_core_vox_long_2_maybe_during_attack (sfxstatic_assets_areacoral.bundle)], weights=[1, 1], pitchMin=1, pitchMax=1)  // disabled
    FadeAudio(gameObject=Self, startVolume=0, endVolume=0.3, time=1)
    WaitRandom(timeMin=4, timeMax=5, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on FINISHED → Call End
    on EXIT → Call End
  }

  state Call End {
    FadeAudio(gameObject=Self, startVolume=0.3, endVolume=0, time=1)
    on FINISHED → Out
  }

  var Self: gameObject = Boss Scene/Audio Zap Boss Distant Call
}
