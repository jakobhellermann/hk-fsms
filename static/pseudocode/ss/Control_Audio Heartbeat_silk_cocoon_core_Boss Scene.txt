# Control
# 1 location(s):
#   Cradle_03 / Boss Scene/silk_cocoon_core/Audio Heartbeat

fsm Control {
  start Start Pause
  on MEMORY ORB COLLECT → Stop - Silk Heart Memory  // from any state

  state Idle {
    on PLAY → Pause
  }

  state Pause {
    Wait(time=var "Play Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Volume Type
  }

  state Play {
    AudioPlayRandom(gameObject=var "Self", audioClips=[silk_cradle_beat_1 (sfxstatic_assets_areacradle.bundle), silk_cradle_beat_2 (sfxstatic_assets_areacradle.bundle), silk_cradle_beat_3 (sfxstatic_assets_areacradle.bundle), silk_cradle_beat_4 (sfxstatic_assets_areacradle.bundle), silk_cradle_beat_5 (sfxstatic_assets_areacradle.bundle)], weights=[1, 1, 1, 1, 1], pitchMin=1, pitchMax=1)
    on FINISHED → Idle
  }

  state Start Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Volume Type
  }

  state Volume Type {
    GetYDistance(gameObject=Self, target=var "Hero", storeResult=var "Y Distance", everyFrame=false, allowNegatives=false)
    FloatCompare(float1=var "Y Distance", float2=25, tolerance=0, equal=→"FULL", lessThan=→"FULL", greaterThan=→"DISTANCE", everyFrame=false)
    on FULL → Full
    on DISTANCE → Distance
  }

  state Full {
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    on FINISHED → Play
  }

  state Distance {
    FloatOperator(float1=var "Y Distance", float2=120, operation=Divide, storeResult=var "Y Distance", everyFrame=false)
    FloatOperator(float1=1, float2=var "Y Distance", operation=Subtract, storeResult=var "Volume", everyFrame=false)
    SetAudioVolume(gameObject=Self, volume=var "Volume", everyFrame=false)
    on FINISHED → Play
  }

  state Stop - Silk Heart Memory {
  }

  var Play Pause: float = 1.3
  var Y Distance: float = 0
  var Volume: float = 0
  var Self: gameObject = Boss Scene/silk_cocoon_core/Audio Heartbeat
}
