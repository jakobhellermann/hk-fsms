# Tubes Reaction
# 1 location(s):
#   Weave_10 / Crest Upgrade Shrine/Breathing Loop Audio

fsm Tubes Reaction {
  start Idle

  state Idle {
    AddEventRegister(target=Self, eventName="TUBES CHANGE")
    on TUBES CHANGE → Check
  }

  state Check {
    GetFsmInt(gameObject=var "Tubes Parent", fsmName="Record Active Tubes", variableName="Tubes Active", storeValue=var "Tubes Active", everyFrame=false)
    IntSwitch(intVariable=var "Tubes Active", compareTo=[0, 1, 2, 3], sendEvent=[→"NONE", →"1", →"2", →"3"], everyFrame=false)
    on NONE → None
    on 1 → 1
    on 2 → 2
    on 3 → 3
  }

  state None {
    SetAudioVolume(gameObject=Self, volume=1, everyFrame=false)
    on FINISHED → Idle
  }

  state 1 {
    SetAudioVolume(gameObject=Self, volume=0.9, everyFrame=false)
    on FINISHED → Idle
  }

  state 2 {
    SetAudioVolume(gameObject=Self, volume=0.75, everyFrame=false)
    on FINISHED → Idle
  }

  state 3 {
    SetAudioVolume(gameObject=Self, volume=0.5, everyFrame=false)
    on FINISHED → Idle
  }

  var Tubes Parent: gameObject = Crest Upgrade Shrine/crest_shrine_break_tube parent

  var Tubes Active: int = 0
}
