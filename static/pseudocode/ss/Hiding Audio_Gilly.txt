fsm Hiding Audio {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on START HIDE → Hiding
  }

  state Hiding {
    WaitRandom(timeMin=2.2, timeMax=6, finishEvent=→"CALL", realTime=false)
    on END HIDE → Idle
    on CALL → Call
  }

  state Call {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Gilly_rest_murmur (sfxdynamic_assets_areawilds.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Hiding
  }

  var Audio Loop Voice: gameObject = Gilly/Audio Loop Voice
}
