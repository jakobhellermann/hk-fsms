# Tick
# 1 location(s):
#   Dust_10 / Spike Pendulum Controller

fsm Tick {
  start Pause

  state Tick {
    SendEventToRegister(eventName="TICK")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegisterDelay(EventName="TICK HALF", delay=0.5, ExcludeTarget=Self)
    on FINISHED → Tick
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tick
  }
}
