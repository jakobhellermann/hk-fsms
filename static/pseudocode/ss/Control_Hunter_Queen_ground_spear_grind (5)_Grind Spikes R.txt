# Control
# 1 location(s):
#   Memory_Ant_Queen / Boss Scene/Grind Spikes R/Hunter_Queen_ground_spear_grind (5)

fsm Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Spear", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Antic
  }

  state Activate {
    PlayRandomAudioClipTable(Table=carmelita_large_spikes_kkck_up (sfxdynamic_assets_areaantareamemory.bundle), AudioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    ActivateGameObject(gameObject=var "Spear", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    RandomFloat(min=-0.2, max=0.2, storeResult=var "Y Shift")
    SetPosition2d(gameObject=var "Spear", vector=(unset), x=(unset), y=var "Y Shift", space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → End
    on HEAVY → Heavy Break
  }

  state End {
  }

  state Antic {
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    on FINISHED → Activate
  }

  state Heavy Break {
    PlayAudioEvent(audioClip=hunter_cage_break (sfxstatic_assets_areaant.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.5, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitter(gameObject=var "Pt Heavy Break", emit=0, resetIfPlaying=false)
    on FINISHED → End
  }

  var Y Shift: float = 0
  var Spear: gameObject = Boss Scene/Grind Spikes R/Hunter_Queen_ground_spear_grind (5)/Hunter_Queen_ground_spear_single
  var Pt Antic: gameObject = Boss Scene/Grind Spikes R/Hunter_Queen_ground_spear_grind (5)/Pt Antic
  var Pt Heavy Break: gameObject = Boss Scene/Grind Spikes R/Hunter_Queen_ground_spear_grind (5)/Pt Heavy Break
}
