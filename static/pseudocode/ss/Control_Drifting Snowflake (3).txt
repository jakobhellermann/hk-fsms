# Control
# 1 location(s):
#   Peak_02 / Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake (3)

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Start X", float2=var "Drift X", operation=Add, storeResult=var "Drift R", everyFrame=false)
    FloatOperator(float1=var "Start X", float2=var "Drift X", operation=Subtract, storeResult=var "Drift L", everyFrame=false)
    on FINISHED → Dormant
  }

  state Dormant {
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Water Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Sprites", activate=false)
    Wait(time=1, finishEvent=→"SPAWN", realTime=false)  // disabled
    SetParent(gameObject=Self, parent=var "Parent", resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on SPAWN → Spawn
  }

  state Spawn {
    PlayParticleEmitterChildren(gameObject=var "Pt Idle", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRandomChild(gameObject=var "Sprites", storeResult=var "Sprite")
    RandomFloat(min=0, max=1, storeResult=(unset))
    ActivateGameObject(gameObject=var "Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RandomFloat(min=-2.5, max=-3, storeResult=var "Y Speed")
    RandomFloat(min=0.8, max=1.1, storeResult=var "Scale")
    SetScale(gameObject=var "Sprite", vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    RandomFloat(min=1.25, max=2, storeResult=var "Drift Speed")
    RandomFloatEither(value1=-20, value2=20, storeResult=var "Rotation Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Speed", everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="START", delay=0, everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Drift L
    on R → Drift R
  }

  state Drift L {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation Speed", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    EaseFloat(fromValue=var "X Pos", toValue=var "Drift L", floatVariable=var "X Tween", time=var "Drift Speed", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"R", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Tween", y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"NEEDLE BREAK", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer=(unset), sendEvent=→"WATER", storeCollider=var "")
    Collision2dEvent(gameObject=var "Terrain Collider", collision=0, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"BREAK", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=true, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on R → Drift R
    on BREAK → Break
    on NEEDLE BREAK → Shake
    on WATER → Water
  }

  state Drift R {
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation Speed", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=Self, vector=(unset), x=var "X Pos", y=(unset), z=(unset), space=0, everyFrame=false)
    EaseFloat(fromValue=var "X Pos", toValue=var "Drift R", floatVariable=var "X Tween", time=var "Drift Speed", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"L", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Tween", y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"NEEDLE BREAK", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer=(unset), sendEvent=→"WATER", storeCollider=var "")
    Collision2dEvent(gameObject=var "Terrain Collider", collision=0, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"BREAK", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=true, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on L → Drift L
    on BREAK → Break
    on NEEDLE BREAK → Shake
    on WATER → Water
  }

  state Break {
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Sprites", activate=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    StopParticleEmittersInChildren(gameObject=var "Pt Idle")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Dormant
  }

  state Shake {
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Strike Nail", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Break
  }

  state Water {
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Water Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateAllChildren(gameObject=var "Sprites", activate=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    StopParticleEmittersInChildren(gameObject=var "Pt Idle")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Dormant
  }

  var Drift X: float = 2

  var Start Y: float = 0
  var Start X: float = 0
  var Drift R: float = 0
  var Drift L: float = 0
  var Y Speed: float = 0
  var Rotation Speed: float = 0
  var X Tween: float = 0
  var X Pos: float = 0
  var Drift Speed: float = 0
  var Scale: float = 0
  var Sprites: gameObject = Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake (3)/Sprites
  var Terrain Collider: gameObject = Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake (3)/Terrain Collider
  var Sprite: gameObject = <null>
  var Parent: gameObject = <null>
  var Pt Idle: gameObject = Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake (3)/Pt Idle
  var Break Effects: gameObject = Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake (3)/Break Effects
  var Strike Nail: gameObject = Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake (3)/Strike Nail
  var Water Effects: gameObject = Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake (3)/Water Effects
}
