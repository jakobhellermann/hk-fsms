  on BUBBLE UP → Bubble Up  // from any state

    FindNamedChild(gameObject=Self, storeResult=var "Return Fx")
    FindNamedChild(gameObject=Self, storeResult=var "ReturnAntic Fx")
    FindNamedChild(gameObject=Self, storeResult=var "Submerge Fx")
    FindNamedChild(gameObject=Self, storeResult=var "Hero Shover")
    FindNamedChild(gameObject=var "Sprite", storeResult=var "Crack Glow")
    FindNamedChild(gameObject=var "Sprite", storeResult=var "Crack Glow Reform")
    ActivateGameObject(gameObject=var "Hero Shover", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetPosition(gameObject=var "Sprite", vector=var "Sprite World Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)

    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.5)
    on COLLIDE → Drop Antic
    on AWAY → Drop Antic
  }

  state Drop Antic {
    PlayAudioEvent(audioClip=magnet_platform_v2_antic (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1.3, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Crack Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Antic Fx", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetPosition(gameObject=var "Sprite", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    ObjectJitter(gameObject=var "Sprite", x=0.15, y=(unset), z=(unset), allowMovement=false, limitFps=0)

    on FINISHED → Drop
    on HAZARD RESPAWNED → Hazard Reset

  state Drop {

    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    AnimatePositionBy(gameObject=var "Sprite", shiftBy=(0, -5, 0), time=0.45, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)

    on FINISHED → Return Antic
    on HAZARD RESPAWNED → Hazard Reset
  }

  state Return Antic {
    BoolTest(boolVariable=var "Go Away", isTrue=→"AWAY", isFalse=(none), everyFrame=true)
    PlayParticleEmitterChildren(gameObject=var "ReturnAntic Fx", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)

    on AWAY → State 1
    on HAZARD RESPAWNED → Hazard Reset
  }

    PlayAudioEvent(audioClip=moss_crumble_plat_returned (sfxstatic_assets_areamoss.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    on FINISHED → Idle
  }

  state Bubble Up {
    SetPosition(gameObject=var "Sprite", vector=(unset), x=(unset), y=-5, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Return
  }

  state State 1 {
  }

  state Hazard Reset {
    SetPosition(gameObject=var "Sprite", vector=(unset), x=0, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    ActivateGameObject(gameObject=var "Crack Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

    on FINISHED → Idle
  }

  var Return Time: float = 1.8

  var Sprite World Pos: vector3 = (0, 0, 0)
  var Sprite: gameObject = lava_crumble_plat/Sprite (localpoolprefabs_assets_water.bundle)
  var Crack Glow: gameObject = <null>
  var Antic Fx: gameObject = lava_crumble_plat/Antic Fx (localpoolprefabs_assets_water.bundle)
  var Submerge Fx: gameObject = lava_crumble_plat/Submerge Fx (localpoolprefabs_assets_water.bundle)

  var Return Fx: gameObject = lava_crumble_plat/Return Fx (localpoolprefabs_assets_water.bundle)
  var Crack Glow Reform: gameObject = <null>
  var Hero Shover: gameObject = <null>
}
