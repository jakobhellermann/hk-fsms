# Low Health FX
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Health
#   Menu_Title / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Health

fsm Low Health FX {
  start Pause

  state Init {
    GetPlayerDataInt(intName="health", storeValue=var "HP")
    IntCompare(integer1=var "HP", integer2=1, equal=→"LOW", lessThan=(none), greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on LOW → Low Health On Entry
  }

  state Idle {
    SetParticleEmission(gameObject=var "Low Health Particles", emission=false, resetOnExit=false)
    SetParticleEmission(gameObject=var "Low Health Leak", emission=false, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Light"), sendEvent="DOWN", delay=0, everyFrame=false)
    SetPosition(gameObject=var "Health 1", vector=var "H1 Initial Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Joni Health", vector=var "H1 Initial Pos", x=(unset), y=(unset), z=-3, space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="DOWN", delay=0, everyFrame=false)
    on HERO DAMAGED → Check Health Pause
    on HERO DAMAGED EXTRA → Check Health Pause
  }

  state Check Health {
    GetPlayerDataInt(intName="health", storeValue=var "HP")
    GetPlayerDataInt(intName="healthBlue", storeValue=var "Blue HP")
    IntCompare(integer1=var "Blue HP", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    IntCompare(integer1=var "HP", integer2=1, equal=→"LOW", lessThan=(none), greaterThan=(none), everyFrame=false)
    IntCompare(integer1=var "HP", integer2=2, equal=→"CAUTION", lessThan=(none), greaterThan=(none), everyFrame=false)
    on LOW → Wait
    on CAUTION → Caution
    on FINISHED → Idle
  }

  state Low Health {
    SetFloatValue(floatVariable=var "Wait Time", floatValue=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "AudioManager"), sendEvent="HERO LOW HEALTH", delay=0, everyFrame=false)
    CheckIfToolEquipped(Tool=Fractured Mask (dataassets_assets_assets/dataassets/tools/toolitems.bundle), RequiredAmountLeft=1, trueEvent=(none), falseEvent=(none), storeValue=var "Do Low Health Particles")
    BoolFlip(boolVariable=var "Do Low Health Particles")
    SetParticleEmission(gameObject=var "Low Health Particles", emission=var "Do Low Health Particles", resetOnExit=false)
    SetParticleEmission(gameObject=var "Low Health Leak", emission=var "Do Low Health Particles", resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Light"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventToRegister(eventName="HEALTH FIRST JITTER START")
    SendEventToRegisterOnExit(EventName="HEALTH ALL JITTER STOP", ExcludeTarget=Self)
    on HERO HEALED → Check Health Pause
    on HERO HEALED FULL → Idle
    on LOW HEALTH END → Idle
    on ADD BLUE HEALTH → Idle
    on HERO DAMAGED → Check Health Pause
    on ENTERING SCENE → Low Health Repeat
    on HERO DAMAGED EXTRA → Check Health Pause
    on ADD PURPLE HEALTH → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Pre Init
  }

  state Low Health On Entry {
    SetParticleEmission(gameObject=var "Low Health Particles", emission=true, resetOnExit=false)
    SetParticleEmission(gameObject=var "Low Health Leak", emission=true, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Light"), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Low Health Repeat
  }

  state Init Wait {
    Wait(time=2.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Init
  }

  state Pre Init {
    FindChild(gameObject=Self, childName="Health 1", storeResult=var "Health 1")
    FindChild(gameObject=Self, childName="Joni Health", storeResult=var "Joni Health")
    FindChild(gameObject=Self, childName="Low Health Particles", storeResult=var "Low Health Particles")
    FindChild(gameObject=Self, childName="Low Health Light", storeResult=var "Low Health Light")
    FindChild(gameObject=var "Hero", childName="Low Health Leak", storeResult=var "Low Health Leak")
    GetPosition(gameObject=var "Health 1", vector=var "H1 Initial Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    on FINISHED → Init Wait
  }

  state Check Health Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Health
  }

  state Wait {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Low Health
  }

  state Caution {
    SetParticleEmission(gameObject=var "Low Health Particles", emission=false, resetOnExit=false)
    SetParticleEmission(gameObject=var "Low Health Leak", emission=false, resetOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="CAUTION", delay=0, everyFrame=false)
    on HERO HEALED → Check Health Pause
    on HERO HEALED FULL → Idle
    on LOW HEALTH END → Idle
    on ADD BLUE HEALTH → Idle
    on HERO DAMAGED → Check Health Pause
    on HERO DAMAGED EXTRA → Check Health Pause
    on ADD PURPLE HEALTH → Idle
  }

  state Caution Repeat On Scene Entry {
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="CAUTION REPEAT", delay=0, everyFrame=false)
    on HERO HEALED → Check Health Pause
    on HERO HEALED FULL → Idle
    on LOW HEALTH END → Idle
    on ADD BLUE HEALTH → Idle
    on HERO DAMAGED → Check Health Pause
    on ENTERING SCENE → Caution Repeat On Scene Entry
    on HERO DAMAGED EXTRA → Check Health Pause
    on ADD PURPLE HEALTH → Idle
  }

  state Low Health Repeat {
    SendEventByName(eventTarget=GameObject(var "Low Health Vignette"), sendEvent="LOW HEALTH REPEAT", delay=0, everyFrame=false)
    SendEventToRegister(eventName="HEALTH FIRST JITTER START")
    SendEventToRegisterOnExit(EventName="HEALTH ALL JITTER STOP", ExcludeTarget=Self)
    on HERO HEALED → Check Health Pause
    on HERO HEALED FULL → Idle
    on LOW HEALTH END → Idle
    on ADD BLUE HEALTH → Idle
    on HERO DAMAGED → Check Health Pause
    on ENTERING SCENE → Low Health Repeat
    on HERO DAMAGED EXTRA → Check Health Pause
    on ADD PURPLE HEALTH → Idle
  }

  var Wait Time: float = 0
  var Blue HP: int = 0
  var HP: int = 0
  var Do Low Health Particles: bool = false
  var H1 Initial Pos: vector3 = (0, 0, 0)
  var Health 1: gameObject = <null>
  var Joni Health: gameObject = <null>
  var Low Health Leak: gameObject = <null>
  var Low Health Light: gameObject = <null>
  var Low Health Particles: gameObject = <null>
  var Low Health Vignette: gameObject = _GameCameras/HudCamera/In-game/Low Health Vignette
}
