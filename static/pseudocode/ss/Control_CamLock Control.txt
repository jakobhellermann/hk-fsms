# Control
# 1 location(s):
#   Song_04 / CamLock Control

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Enter Folder")
    FindNamedChild(gameObject=Self, storeResult=var "Exit Folder")
    FindNamedChild(gameObject=Self, storeResult=var "Enter Range")
    FindNamedChild(gameObject=Self, storeResult=var "Exit Range")
    on FINISHED → Out
  }

  state Out {
    ActivateGameObject(gameObject=var "Enter Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Exit Folder", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=var "Enter Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    ActivateGameObject(gameObject=var "Enter Folder", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Exit Folder", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=var "Exit Range", trigger=0, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → Out
  }

  var Enter Folder: gameObject = <null>
  var Exit Folder: gameObject = <null>
  var Enter Range: gameObject = <null>
  var Exit Range: gameObject = <null>
}
