# Curtain Jiggle
# 1 location(s):
#   Song_09b / Gourmand/Great Gourmand Scene

fsm Curtain Jiggle {
  start Wait
  on INERT → State 2  // from any state

  state Wait {
    WaitRandom(timeMin=15, timeMax=25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Jiggle
  }

  state State 2 {
  }

  state Jiggle {
    AnimatorPlayStateWait(target=Self, stateName="Jiggle", finishEvent=→"FINISHED")
    on FINISHED → Wait
  }
}
