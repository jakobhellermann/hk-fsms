# FSM
# 1 location(s):
#   Library_13b / Desk Inspect and Quill/Interactive Quills/Collectable Item Pickup - Quill Red

fsm FSM {
  start Wait

  state Wait {
    on INTERACT → Pick Up?
  }

  state Pick Up? {
    DialogueYesNoItemV6(Prompt={Sheet="UI", Key="TAKE_ITEM_PROMPT"}, PromptFormat=WillGetItemName, BoxLabel="", RequiredItems=[], RequiredAmounts=[0], CurrencyCost=0, CurrencyType=Money, ShowCounter=false, ConsumeItem=false, TakeDisplay=Silent, DisplayType=WillGetItems, WillGetItem=Trobbio Quill Red (dataassets_assets_assets/dataassets/collectables/fakecollectables.bundle), YesEvent=→"YES", NoEvent=→"NO", ReturnHUDAfter=false)
    on YES → Do Pickup
    on NO → Run Dialogue
  }

  state Do Pickup {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=DoPickup())
    Wait(time=1.45, finishEvent=(none), realTime=false)
    on FINISHED → Deactivate
  }

  state Run Dialogue {
    RunDialogue(Sheet="Inspect", Key="LIBRARY_THEATRE_LINES", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=Top, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state End Dialogue {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Wait
  }

  state Deactivate {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    on FINISHED → 
  }
}
