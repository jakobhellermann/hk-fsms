# BG Control
# 1 location(s):
#   Bone_East_18b / Boss Scene/Gates/Battle Gate Ant Straight (2)

fsm BG Control {
  start Pause
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state
  on ACTIVATE → Destroy  // from any state

  state Opened {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    AnimatorPlay(gameObject=Self, stateName="Open", layer=0, normalizedTime=1, everyFrame=false)
    on BG CLOSE → Close Pause
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    SetCollider(gameObject=var "Wall Collider", active=true, resetOnExit=false)
    AnimatorPlayStateWait(target=Self, stateName="Close", finishEvent=→"FINISHED")
    ListenForAnimationEvent(Target=Self, Response=→"FINISHED")
    on FINISHED → Close 2
  }

  state Close 2 {
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Slam", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=var "Wall Collider", active=true, resetOnExit=false)
    on BG OPEN → Open
    on OPEN → Open
  }

  state Open {
    PlayAudioEvent(audioClip=mantis_lords_gate_close (sfxstatic_assets_battle.bundle), pitchMin=0.85, pitchMax=1.2, volume=0.75, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Wall Collider", active=false, resetOnExit=false)
    AnimatorPlayStateWait(target=Self, stateName="Open", finishEvent=→"FINISHED")
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Opened Effects
  }

  state Quick Close {
    SetBoolValue(boolVariable=var "Start Closed", boolValue=false, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    AnimatorPlay(gameObject=Self, stateName="Close", layer=0, normalizedTime=1, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetCollider(gameObject=var "Wall Collider", active=true, resetOnExit=false)
    on BG OPEN → Open
    on OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Raise Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Wall Collider", active=false, resetOnExit=false)
    AnimatorPlay(gameObject=Self, stateName="Open", layer=0, normalizedTime=1, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close Pause {
    WaitRandom(timeMin=0, timeMax=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Close 1
  }

  state Pause {
    ActivateGameObject(gameObject=var "Raise Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Wall Collider", active=false, resetOnExit=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Opened
  }

  state Opened Effects {
    PlayParticleEmitter(gameObject=var "Pt Open", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Rocks", emit=0, resetIfPlaying=false)
    on FINISHED → Opened
  }

  var Start Closed: bool = false

  var Pt Open: gameObject = Boss Scene/Gates/Battle Gate Ant Straight (2)/Pt Open
  var Pt Rocks: gameObject = Boss Scene/Gates/Battle Gate Ant Straight (2)/Pt Rocks
  var Pt Slam: gameObject = Boss Scene/Gates/Battle Gate Ant Straight (2)/Pt Slam
  var Raise Dust: gameObject = <null>
  var Self: gameObject = <null>
  var Wall Collider: gameObject = Boss Scene/Gates/Battle Gate Ant Straight (2)/Wall Collider
}
