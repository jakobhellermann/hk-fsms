# Control
# 1 location(s):
#   Shadow_05 / Swamp Trap Wire (3)

fsm Control {
  start Pause

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATED → Activated
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"CUT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"TOUCH", storeCollider=var "")
    on TOUCH → Touch
    on CUT → Check On Camera
  }

  state Touch {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wire Hit", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Snap
  }

  state Snap {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Snap", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    DoCameraShakeV2(Target=<null>, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    SendEventByName(eventTarget=GameObject(var "Trap Folder"), sendEvent="TRAP", delay=0.1, everyFrame=false)
    PlayAudioEvent(audioClip=swamp_trap_wire (sfxstatic_assets_areaswamp.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Activated {
    SendEventByName(eventTarget=GameObject(var "Trap Folder"), sendEvent="ACTIVATED", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Check On Camera {
    CheckOutOfCamera(gameObject=Self, margin=6, outsideEvent=→"CANCEL", insideEvent=→"FINISHED", insideBool=false, outsideBool=false, everyFrame=false)
    on FINISHED → Snap
    on CANCEL → Idle
  }

  var Trap Folder: gameObject = stake_trap_swing (4)

  var Activated: bool = false
  var Pt Snap: gameObject = Swamp Trap Wire (3)/Pt Snap
}
