# Control
# 2 location(s):
#   Coral_44 / Coral Crust Wall Mid
#   Coral_44 / Coral Crust Wall Mid (1)

// uses template: coral_crust_wall
fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetOwner(storeGameObject=var "Self")
    SetBoolValue(boolVariable=var "Fully Break", boolValue=true, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "CameraLockArea")
    FindNamedChild(gameObject=Self, storeResult=var "Damager")
    FindNamedChild(gameObject=Self, storeResult=var "Terrain Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Strike Marker")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Hit")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Break")
    FindNamedChild(gameObject=Self, storeResult=var "Smash Effect")
    FindNamedChild(gameObject=Self, storeResult=var "Damageable Range")
    FindNamedChild(gameObject=Self, storeResult=var "Wall Parts")
    FindNamedChild(gameObject=Self, storeResult=var "Extra Art")
    FadeNestedFadeGroup(Target=var "Fader", ToAlpha=1, FadeTime=0)
    ActivateGameObject(gameObject=var "z1 Active If Destroyed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    StringCompare(stringVariable=var "PD Bool", compareTo="", equalEvent=→"FINISHED", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName=var "PD Bool", isTrue=→"ACTIVATED", isFalse=(none))
    on FINISHED → Idle
    on ACTIVATED → Set Broken
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"DAMAGE", sendEventHeavy=→"BREAK", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damage Collider", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=true)
    on DAMAGE → Range Check
    on FORCE DAMAGE → Force Damage
    on BREAK → Range Check 2
  }

  state Hit {
    PlayAudioEvent(audioClip=sword_terrain_temp (sfxstatic_assets_shared.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "z2 Hit Receiver"), sendEvent="CORAL DAMAGED", delay=0, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetPositionToObject(gameObject=var "Strike Marker", targetObject=var "Damage Collider", xOffset=0, yOffset=0, zOffset=0, overrideZ=0, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Hit", resetTimeIfPlaying=false, stopOnStateExit=false)
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    SendMessage(gameObject=var "Extra Art", delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Strike Effect
    on BREAK → Heavy Break
  }

  state Strike Effect {
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Strike Marker", position=(unset), rotation=(unset), storeObject=var "")
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    PlayRandomAudioClipTableV2(Table=var "Hit Sounds", AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false)
    on FINISHED → Idle
    on BREAK → Break Pause
  }

  state Break Pause {
    Wait(time=0.085, finishEvent=→"FINISHED", realTime=false)  // disabled
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Break
  }

  state Break {
    PlayRandomAudioClipTableV2(Table=var "Break Sounds", AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0), ForcePlay=false)
    PlayAudioEvent(audioClip=coral_wall_break (sfxstatic_assets_areacoral.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FadeNestedFadeGroup(Target=var "Fader", ToAlpha=0, FadeTime=0.3)
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="BREAK", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "z2 Hit Receiver"), sendEvent="CORAL DESTROYED", delay=0, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Break", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendMessage(gameObject=var "Notify Group", delivery=SendMessage, options=1, functionCall=UpdateCrustUnlock())
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Smash Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Crust Mask 1"), sendEvent="BREAK", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Crust Mask 2"), sendEvent="BREAK", delay=0, everyFrame=false)
    on FINISHED → Broken
  }

  state Broken {
    SetPlayerDataBool(boolName=var "PD Bool", value=true)
    SetCollider(gameObject=var "CameraLockArea", active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "z1 Active If Destroyed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Wall Parts", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Extra Art", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    BoolTest(boolVariable=var "Fully Break", isTrue=→"BREAK", isFalse=(none), everyFrame=false)
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")
    BuildString(stringParts=[var "Clip", " Broken"], separator="", addToEnd=true, storeResult=var "Break Clip", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Break Clip")
    on BREAK → Fully Broken
  }

  state Range Check {
    GameObjectIsNull(gameObject=var "Damageable Range", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Damageable Range", storeResult=false, sendEvent="FINISHED", outOfRangeEvent="CANCEL", everyFrame=false)
    on FINISHED → Hit
    on CANCEL → Idle
  }

  state Fully Broken {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Force Damage {
    SetGameObject(variable=var "Damage Collider", gameObject=var "Self", everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=zombie_guard_club (sfxstatic_assets_areawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Hit
  }

  state Set Broken {
    SendEventByName(eventTarget=GameObject(var "Event Receiver"), sendEvent="BROKEN", delay=0, everyFrame=false)
    FadeNestedFadeGroup(Target=var "Fader", ToAlpha=0, FadeTime=0)
    on FINISHED → Broken
  }

  state Heavy Break {
    SetIntValue(intVariable=var "Hits", intValue=0, everyFrame=false)
    on FINISHED → Hit
  }

  state Range Check 2 {
    GameObjectIsNull(gameObject=var "Damageable Range", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Damageable Range", storeResult=false, sendEvent="FINISHED", outOfRangeEvent="CANCEL", everyFrame=false)
    on FINISHED → Heavy Break
    on CANCEL → Idle
  }

  var Hits: int = 3
  var Fully Break: bool = false
  var PD Bool: string = ""
  var Crust Mask 1: gameObject = <null>
  var Crust Mask 2: gameObject = <null>
  var Event Receiver: gameObject = Secret Mask New Sharp Ultra
  var Fader: gameObject = <null>
  var z1 Active If Destroyed: gameObject = <null>
  var z2 Hit Receiver: gameObject = <null>
  var Hit Sounds: object = choral_or_other_break (sfxdynamic_assets_areacoral.bundle)
  var Break Sounds: object = <null>

  var Activated: bool = false
  var Clip: string = ""
  var Break Clip: string = ""
  var CameraLockArea: gameObject = <null>
  var Damage Collider: gameObject = <null>
  var Damageable Range: gameObject = <null>
  var Damager: gameObject = <null>
  var Extra Art: gameObject = <null>
  var Notify Group: gameObject = <null>
  var Pt Break: gameObject = <null>
  var Pt Hit: gameObject = <null>
  var Self: gameObject = <null>
  var Smash Effect: gameObject = <null>
  var Strike Marker: gameObject = <null>
  var Terrain Collider: gameObject = <null>
  var Wall Parts: gameObject = <null>
}
