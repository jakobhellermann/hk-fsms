# Control
# 2 location(s):
#   Bone_18 / Quest Scene/Final Encounter/Battle Scene/Battle End
#   Bone_18 / Black Thread States/Black Thread World/Battle Scene/Battle End

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Gates")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Silk Spool")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Collectable Item Pickup")
    on FINISHED → Idle
  }

  state Idle {
    on BATTLE END → Activate Locket
  }

  state Complete Quest {
    EndQuest(ConsumeCurrency=(unset), Quest=<null>)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataBool(boolName="isInvincible", value=true)
    on WISH PROMISED END → Activate Item
  }

  state Open Gates {
    SendEventByName(eventTarget=GameObject(var "Gates"), sendEvent="BG OPEN", delay=0, everyFrame=false)
  }

  state Activate Item {
    FindGameObject(objectName="", withTag="Boss Corpse", store=var "Boss Corpse")
    SetPositionToObject(gameObject=var "Silk Spool", targetObject=var "Boss Corpse", xOffset=0, yOffset=0, zOffset=0, overrideZ=0.001, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Silk Spool", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="isInvincible", value=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Open Gates
  }

  state Activate Locket {
    ActivateGameObject(gameObject=var "Collectable Item Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Battle Scene: gameObject = <null>
  var Gates: gameObject = <null>
  var Boss Corpse: gameObject = <null>
  var Collectable Item Pickup: gameObject = <null>
  var Silk Spool: gameObject = <null>
}
