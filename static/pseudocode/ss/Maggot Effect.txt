  start Init
  on MAGGOT RESET → Inert  // from any state

  state Init {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    CreateObject(gameObject=Maggot_Bind_Burst_UI (localpoolprefabs_assets_shared.bundle), spawnPoint=var "", position=(unset), rotation=(unset), storeObject=var "Burst Effect")
    ActivateGameObject(gameObject=var "Burst Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetTransformParent(gameObject=var "Burst Effect", parent=var "Self", worldPositionStays=false)
    SetPosition(gameObject=var "Burst Effect", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Inert
  }

  state Inert {
    SetMeshRenderer(gameObject=Self, active=false)
    EnableFsmSelf(SetEnabled=false)
    on MAGGOT CHECK → Is Maggoted?
  }

  state Maggot {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Maggot Orb Loop", animationTriggerEvent=(none), animationCompleteEvent=(none))
    EnableFsmSelf(SetEnabled=false)
    on MAGGOT CHECK → Is Maggoted? 2
  }

  state Maggot Appear {
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Maggot Orb Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Maggot
  }

  state Maggot Disappear {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Maggot Orb Disappear", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Maggot Orb Burst
  }

  state Is Maggoted? {
    GetHeroCState(VariableName="isMaggoted", StoreValue=var "Is Maggoted", EveryFrame=false)
    BoolTest(boolVariable=var "Is Maggoted", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Inert
    on TRUE → Maggot Appear
  }

  state Is Maggoted? 2 {
    GetHeroCState(VariableName="isMaggoted", StoreValue=var "Is Maggoted", EveryFrame=false)
    BoolTest(boolVariable=var "Is Maggoted", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Maggot Disappear
    on TRUE → Maggot
  }

  state Maggot Orb Burst {
    ActivateGameObject(gameObject=var "Burst Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Burst Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Inert
  }

  var Is Maggoted: bool = false
  var Self: gameObject = <null>
  var Burst Effect: gameObject = <null>
}
