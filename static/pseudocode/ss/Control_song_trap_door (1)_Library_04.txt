# Control
# 3 location(s):
#   Library_04 / song_trap_door (1)
#   Library_13 / song_trap_door (1)
#   Slab_03 / song_trap_door (1)

fsm Control {
  start Idle

  state Idle {
    FindChild(gameObject=Self, childName="Door/Tilt Plat/terrain collider", storeResult=var "Terrain Collider")
    on OPEN → Disable As Opens?
    on ACTIVATE → Activate
  }

  state Open {
    SetAnimatorTrigger(gameObject=var "Door", trigger="open")
    SetPosition(gameObject=var "Door", vector=(unset), x=(unset), y=(unset), z=0.001, space=1, everyFrame=false, lateUpdate=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Enviro Region Simple", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Deactivate Collider
  }

  state Activate {
    SetAnimatorTrigger(gameObject=var "Door", trigger="opened")
    SetPosition(gameObject=var "Door", vector=(unset), x=(unset), y=(unset), z=0.001, space=1, everyFrame=false, lateUpdate=false)
    FindChild(gameObject=Self, childName="Opened Collider", storeResult=var "Open Collider")
    ActivateGameObject(gameObject=var "Open Collider", activate=var "Disable Collider On Open", recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enviro Region Simple", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolFlip(boolVariable=var "Disable Collider On Open")
    ActivateGameObject(gameObject=var "Terrain Collider", activate=var "Disable Collider On Open", recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Deactivate Collider {
    FindChild(gameObject=Self, childName="Opened Collider", storeResult=var "Open Collider")
    ActivateGameObject(gameObject=var "Open Collider", activate=var "Disable Collider On Open", recursive=false, resetOnExit=false, everyFrame=false)
    BoolFlip(boolVariable=var "Disable Collider On Open")
    ActivateGameObject(gameObject=var "Terrain Collider", activate=var "Disable Collider On Open", recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Disable As Opens? {
    BoolTest(boolVariable=var "z Disable Collider As It Opens", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Open
  }

  var Disable Collider On Open: bool = true
  var z Disable Collider As It Opens: bool = true

  var Door: gameObject = song_trap_door (1)/Door
  var Terrain Collider: gameObject = <null>
  var Open Collider: gameObject = <null>
  var Enviro Region Simple: gameObject = song_trap_door (1)/Enviro Region Simple
}
