# Needolin Scene
# 1 location(s):
#   Coral_Judge_Arena / Boss Scene

fsm Needolin Scene {
  start Inert
  on BEGIN → Begin  // from any state
  on END → End  // from any state

  state Inert {
  }

  state Begin {
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=true)
    RunFSM(fsmTemplateControl=template=-6849325147229470167 in[Skip Begin<-false, Skip End<-true, Timeout Override<-true], finishEvent=(none), everyFrame=false)
    on NEEDOLIN LOCK → Inert
  }

  sequence state End {
    RunFSM(fsmTemplateControl=template=-6879378911247425464 in[Is In Needolin Proper<-false], finishEvent=(none), everyFrame=false)
    RemoveUsingSilk(Amount=1, UsingType=Normal, DidAddTracker=var "Is In Needolin Proper")
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=false)
    ControlNeedolin(target=var "Hero", isPlaying=false)
    Tk2dPlayAnimationWait(Target=var "Hero", ClipName="Needolin End", AnimationCompleteEvent=(none))  // disabled
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControlToIdle", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignTest(floatValue=var "Hero X Scale", isPositive=→"LEFT", isNegative=(none), everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="FlipSprite", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="TurnToChallengeTalk")
    PlayAudioEvent(audioClip=zombie_shield_sword_1 (sfxstatic_assets_areacoralareadustareashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on LEFT → No Turn
  }

  sequence state No Turn {
    Tk2dPlayAnimationWait(Target=var "Hero", ClipName="Needolin End", AnimationCompleteEvent=(none))  // disabled
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="ToChallengeTalk")
    PlayAudioEvent(audioClip=zombie_shield_sword_1 (sfxstatic_assets_areacoralareadustareashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
  }

  var Is In Needolin Proper: bool = true

  var Hero X Scale: float = 0
  var Facing Left: bool = false
  var Challenge Anim: string = ""
}
