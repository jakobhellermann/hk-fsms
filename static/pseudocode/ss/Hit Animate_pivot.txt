# Hit Animate
# 1 location(s):
#   Song_09b / Gourmand/pale_oil_holder/pivot

fsm Hit Animate {
  start Idle

  state Idle {
    on TINK UP → Choose Dir
    on TINK DOWN → Choose Dir
    on TINK LEFT → Hit L
    on TINK RIGHT → Hit R
  }

  state Hit L {
    CallMethodProper(gameObject=Self, behaviour="CurveRotationAnimation", methodName="StartAnimation", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Choose Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"TINK LEFT", leftEvent=→"TINK RIGHT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=true, everyFrame=false)
    on TINK LEFT → Hit L
    on TINK RIGHT → Hit R
  }

  state Hit R {
    CallMethodProper(gameObject=Self, behaviour="CurveRotationAnimation", methodName="StartAnimation", parameters=[false], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }
}
