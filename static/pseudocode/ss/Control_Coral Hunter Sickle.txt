# Control
# 1 location(s):
#   localpoolprefabs_assets_areaantareacloverareacoralareamemory.bundle / Coral Hunter Sickle

fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Bubble Popper", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Bubble Popper", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Heavy Attack Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sickle")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Dir
    on GROUND → G Pause
  }

  state Fly {
    SetVelocityAsAngle(gameObject=Self, angle=var "Travel Angle", speed=25, everyFrame=true)
    FloatAdd(floatVariable=var "Travel Angle", add=var "Angle Shift", everyFrame=true, perSecond=true)
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=true)
    RandomFloat(min=0.9, max=1.1, storeResult=var "Audio Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Audio Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"BREAK", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on BREAK → Break
    on PARRIED → Deflect
    on DAMAGED HERO → Break
  }

  state Break {
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Heavy Attack Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=spikes_break (herostatic_assets_all.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Bubble Popper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sickle Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sickle Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Check Dir {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Travel Angle", everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Velocity X", float2=0, tolerance=0, equal=(none), lessThan=→"L", greaterThan=→"R", everyFrame=false)
    on L → L
    on R → R
  }

  state R {
    SetFloatValue(floatVariable=var "Angle Shift", floatValue=60, everyFrame=false)
    on FINISHED → Fly
  }

  state L {
    SetFloatValue(floatVariable=var "Angle Shift", floatValue=-60, everyFrame=false)
    on FINISHED → Fly
  }

  state Deflect {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sickle Deflected")
    ActivateGameObject(gameObject=var "Bubble Popper", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Heavy Attack Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=50, vector3=(unset), everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Travel Angle", speed=30, everyFrame=true)
    FloatAdd(floatVariable=var "Travel Angle", add=var "Angle Shift", everyFrame=true, perSecond=true)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=true)
    RandomFloat(min=0.4, max=0.5, storeResult=var "Audio Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Audio Pitch", everyFrame=false)
    on BREAK → Break
  }

  state G Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Dir 2
  }

  state Check Dir 2 {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Travel Angle", everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "Velocity X", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Velocity X", float2=0, tolerance=0, equal=(none), lessThan=→"L", greaterThan=→"R", everyFrame=false)
    on L → L 2
    on R → R 2
  }

  state R 2 {
    SetFloatValue(floatVariable=var "Angle Shift", floatValue=10, everyFrame=false)
    on FINISHED → Fly
  }

  state L 2 {
    SetFloatValue(floatVariable=var "Angle Shift", floatValue=-10, everyFrame=false)
    on FINISHED → Fly
  }

  state Double Hit {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    on FINISHED → Break
  }

  var Velocity X: float = 0
  var Travel Angle: float = 0
  var Angle Shift: float = 0
  var Audio Pitch: float = 0
  var Bubble Popper: gameObject = <null>
  var Pt Break: gameObject = Coral Hunter Sickle/Pt Break
  var Heavy Attack Detector: gameObject = <null>
}
