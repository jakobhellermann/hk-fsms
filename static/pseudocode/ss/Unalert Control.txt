fsm Unalert Control {
  start Init

  state Init {
    SetParent(gameObject=var "Unalert Range Obj", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Idle {
    on FIGHT STARTED → In Range
  }

  state In Range {
    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=→"RANGE OUT", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    SetFsmBool(gameObject=Self, fsmName="Attack Enemies", variableName="In Unalert Range", setValue=false, everyFrame=false)
    on RANGE OUT → Out of Range
  }

  state Out of Range {
    CheckAlertRange(alertRange=var "Unalert Range", storeResult=(unset), InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=→"RANGE IN", OutOfRangeDelay=0, everyFrame=true)
    TransitionToAudioSnapshot(snapshot=Normal Soft (audioobjects_assets_all.bundle), transitionTime=2)
    Wait(time=1, finishEvent=(none), realTime=false)
    SetFsmBool(gameObject=Self, fsmName="Attack Enemies", variableName="In Unalert Range", setValue=true, everyFrame=false)
    Wait(time=3, finishEvent=→"END", realTime=false)
    on RANGE IN → Return Music
    on END → End Battle
  }

  state Return Music {
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
    on FINISHED → In Range
  }

  state End Battle {
    SetFsmBool(gameObject=Self, fsmName="Attack Enemies", variableName="Unalert End", setValue=true, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=2)
    on FINISHED → Idle
  }

  var Unalert Range Obj: gameObject = Mapper Spar NPC/Unalert Range
  var Unalert Range: object = Mapper Spar NPC/Unalert Range@AlertRange
}
