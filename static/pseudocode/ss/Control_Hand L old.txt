// uses template: silk_boss_hand
fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Silk Boss")
    FindChild(gameObject=var "Silk Boss", childName=var "Point Name", storeResult=var "Hand Point")
    on FINISHED → Follow Near
  }

  state Follow Near {
    SetPositionToObject2D(gameObject=Self, targetObject=var "Hand Point", xOffset=0, yOffset=0, everyFrame=false)  // disabled
    ChaseObjectGround(gameObject=Self, target=var "Hand Point", xOffset=0, speedMax=2, acceleration=0.5, animateTurnAndRun=false, runAnimation="", turnAnimation="", turnRange=0.2, snapTo=false, snapSpeedOnly=false, onlyOnStateEntry=false)
    GetPosition(gameObject=var "Hand Point", vector=(unset), x=(unset), y=var "Hand Y", z=(unset), space=0, everyFrame=true)
    SilkBossIdleBuzz(gameObject=Self, targetY=var "Hand Y", targetRange=0.1, acceleration=5, speedMax=2, fuzzTargetRange=false)
    GetDistance(gameObject=Self, target=var "Hand Point", storeResult=var "Distance From Hand Point", everyFrame=true)
    FloatCompare(float1=var "Distance From Hand Point", float2=3, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FAR", everyFrame=true)
    on FAR → Chase Pause
  }

  state Chase Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Chase
  }

  state Chase {
    ChaseObjectV2(gameObject=Self, target=var "Hand Point", speedMax=20, accelerationForce=15, offsetX=0, offsetY=0)
    ChaseObjectV3(gameObject=Self, target=var "Hand Point", accelerationForce=1, offsetX=0, offsetY=0, speedMax=20, speedMin=0)  // disabled
    GetDistance(gameObject=Self, target=var "Hand Point", storeResult=var "Distance From Hand Point", everyFrame=true)
    FloatCompare(float1=var "Distance From Hand Point", float2=1.5, tolerance=0, equal=(none), lessThan=→"NEAR", greaterThan=(none), everyFrame=true)
    on NEAR → Follow Near
  }

  var Point Name: string = "Hand L Idle Point"

  var Idle Y: float = 138.5
  var Hand Y: float = 0
  var Distance From Hand Point: float = 0
  var Boss Scene: gameObject = <null>
  var Silk Boss: gameObject = <null>
  var Hand Point: gameObject = <null>
}
