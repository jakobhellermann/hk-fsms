# Behaviour
# 1 location(s):
#   Crawl_08 / Area_States/Uninfected/Scientist_set/Blue Scientist

fsm Behaviour {
  start Init
  on INTERACT TALK → Activate  // from any state

  state Init {
    SetTransformParent(gameObject=var "Patrol Point L", parent=<null>, worldPositionStays=true)
    SetTransformParent(gameObject=var "Patrol Point R", parent=<null>, worldPositionStays=true)
    WaitForHeroInPosition(sendEvent=→"POSITION", skipIfAlreadyPositioned=true)
    on POSITION → Deactivate 2
  }

  state Walk Left {
    SetGameObject(variable=var "Patrol Point Current", gameObject=var "Patrol Point L", everyFrame=false)
    on FINISHED → Walk
  }

  state Deactivate {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dSpriteSetScale(gameObject=Self, scale=(1, 1, 1), everyframe=false)
    on FINISHED → Dir?
  }

  state Activate {
    GetProperty(targetProperty=tk2dLookAnimNPC.State on Area_States/Uninfected/Scientist_set/Blue Scientist@tk2dLookAnimNPC, everyFrame=false)
    EnumCompare(enumVariable=var "Look Anim State", compareTo=Disabled, equalEvent=(none), notEqualEvent=→"FINISHED", storeResult=(unset), everyFrame=false)
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    on FINISHED → Meet Corpse?
  }

  state End Dlg {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Wait
  }

  state Wait {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
    on SING → Sing
    on STARTLE → Startle
  }

  state Walk Right {
    SetGameObject(variable=var "Patrol Point Current", gameObject=var "Patrol Point R", everyFrame=false)
    on FINISHED → Walk
  }

  state Walk {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Walk", animationTriggerEvent=(none), animationCompleteEvent=(none))
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "Patrol Point Current", vectorPosition=(unset), time=(unset), delay=0, speed=var "Walk Speed", space=0, easeType=21, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"STOP", realTime=false, stopOnExit=true, loopDontFinish=false)
    AudioPlayInState(gameObject=Self, volume=1)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on STOP → Observe
    on INTERACT START → Stop Observe
    on SING → Sing
    on STARTLE → Startle
  }

  sequence state Observe {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Alchemist_tank_observe (sfxdynamic_assets_areacrawl.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayAudioEvent(audioClip=alchemist_look_up_and_ponder (sfxstatic_assets_areacrawl.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    WaitRandom(timeMin=2, timeMax=7, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName="Sing End", AnimationCompleteEvent=(none))
    on FINISHED → Turn
    on INTERACT START → Stop Observe
    on SING → Sing
    on STARTLE → Startle
  }

  sequence state Turn {
    PlayAudioEvent(audioClip=alchemist_turn (sfxstatic_assets_areacrawl.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWait(Target=Self, ClipName="Look Back Turn", AnimationCompleteEvent=(none))
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Dir?
    on INTERACT START → Cancel Turn
    on STARTLE → Startle
  }

  state Dir? {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignTest(floatValue=var "X Scale", isPositive=→"L", isNegative=→"R", everyFrame=false)
    on L → Walk Left
    on R → Walk Right
  }

  state Cancel Turn {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
  }

  sequence state Stop Observe {
    PlayAudioEvent(audioClip=alchemist_look_up_and_ponder (sfxstatic_assets_areacrawl.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWait(Target=Self, ClipName="Sing End", AnimationCompleteEvent=(none))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
  }

  state Start Pos? {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"L", leftEvent=→"R", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Start L
    on R → Start R
  }

  state Start L {
    GetPosition2D(gameObject=var "Patrol Point L", vector=var "Start Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=var "Start Pos", X=0, Y=0, Space=0, EveryFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Observe
  }

  state Start R {
    GetPosition2D(gameObject=var "Patrol Point R", vector=var "Start Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=var "Start Pos", X=0, Y=0, Space=0, EveryFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Observe
  }

  state Deactivate 2 {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    on FINISHED → Start Pos?
  }

  state Common {
    GetPlayerDataVariable(VariableName="BlueScientistTalkedCorpse", StoreValue=var "Chooser")
    ConvertBoolToString(boolVariable=var "Chooser", stringVariable=var "Offer Cell", falseString="BLUE_SCIENTIST_OFFER", trueString="BLUE_SCIENTIST_OFFER_ZANGO", everyFrame=false)
    ConvertBoolToString(boolVariable=var "Chooser", stringVariable=var "Accept Cell", falseString="BLUE_SCIENTIST_ACCEPT", trueString="BLUE_SCIENTIST_ACCEPT_ZANGO", everyFrame=false)
    GetPlayerDataVariable(VariableName="BlueAssistantCorpseFound", StoreValue=var "Chooser")
    ConvertBoolToString(boolVariable=var "Chooser", stringVariable=var "Meet Cell", falseString="BLUE_SCIENTIST_MEET", trueString="BLUE_SCIENTIST_MEET_ZANGO", everyFrame=false)
    RunFSM(fsmTemplateControl=template=-669518168033599682 in[Meet Cell<-var "Meet Cell", Met Bool<-"BlueScientistMet", Q1 Accept Cell<-var "Accept Cell", Q1 Can Complete Cell<-"BLUE_SCIENTIST_CANCOMPLETE", Q1 Decline Cell<-"BLUE_SCIENTIST_DECLINE", Q1 Offer Cell<-var "Offer Cell", Q1 Post Complete Cell<-"BLUE_SCIENTIST_QUEST_COMPLETE_2", Q1 Pre Complete Cell<-"BLUE_SCIENTIST_QUEST_COMPLETE_1", Q1 Refuse Cell<-"BLUE_SCIENTIST_REFUSE", Q1 Reoffer Cell<-"BLUE_SCIENTIST_REOFFER", Q1 Repeat Cell<-"BLUE_SCIENTIST_QUEST_REPEAT", Q2 Accept Cell<-"", Q2 Can Complete Cell<-"", Q2 Decline Cell<-"", Q2 Offer Cell<-"", Q2 Post Complete Cell<-"", Q2 Reoffer Cell<-"", Q2 Repeat Cell<-"", Repeat Cell<-"BLUE_SCIENTIST_REPEAT", Repeat Done Cell<-"", Q2 Pre Complete Cell<-"", Q2 Refuse Cell<-"", Refill Offer Cell<-"BLUE_SCIENTIST_REFILL_OFFER", Refill Refuse Cell<-"BLUE_SCIENTIST_REFILL_REFUSE", Refill Accept Cell<-"BLUE_SCIENTIST_REFILL_ACCEPT"], finishEvent=→"FINISHED", everyFrame=false)
    on FINISHED → End Dlg
  }

  state Meet Corpse? {
    PlayerDataVariableTest(VariableName="BlueScientistMet", ExpectedValue=true, IsExpectedEvent=→"FINISHED", IsNotExpectedEvent=(none))
    GetPlayerDataVariable(VariableName="BlueAssistantCorpseFound", StoreValue=var "Chooser")
    SetPlayerDataVariable(VariableName="BlueScientistTalkedCorpse", SetValue=var "Chooser")
    on FINISHED → Common
  }

  state Sing {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Alchemist_sing (sfxdynamic_assets_areacrawl.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.75, MaxReactDelay=0.75, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Sing Finish
  }

  state Sing Finish {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Wait
  }

  state Startle {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Alchemist_startle (sfxdynamic_assets_areacrawl.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Wait(time=0, finishEvent=→"END", realTime=true)
    on END → Startle End
  }

  state Startle End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Startle Finish
  }

  state Startle Finish {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Wait
  }

  var Walk Speed: float = 5
  var X Scale: float = 0
  var Chooser: bool = false
  var Offer Cell: string = ""
  var Accept Cell: string = ""
  var Meet Cell: string = ""
  var Start Pos: vector2 = (0, 0)
  var Patrol Point Current: gameObject = <null>
  var Patrol Point L: gameObject = Area_States/Uninfected/Scientist_set/Blue Scientist/Patrol Point L
  var Patrol Point R: gameObject = Area_States/Uninfected/Scientist_set/Blue Scientist/Patrol Point R
  var Audio Loop Voice: gameObject = Area_States/Uninfected/Scientist_set/Blue Scientist/Audio Loop Voice
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
