# Msg Control
# 1 location(s):
#   prompts_assets_all.bundle / UI Msg Crest Evolve

fsm Msg Control {
  start Init

  state Detect {
    ListenForPromptContinue(EventTarget=Self, ContinuePressed=→"PRESS")
    on PRESS → Down
  }

  state Done {
    CallMethodProper(gameObject=Self, behaviour="UIMsgProxy", methodName="SetIsInMsg", parameters=[false], storeResult=(unset var), EveryFrame=false)
    SendEventToRegister(eventName="GET ITEM MSG END")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Init {
    ActivateGameObject(gameObject=var "Combo 1 Evolve", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Combo 2 Evolve", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Socket Evolve 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Socket Evolve 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Cursed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fleur", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Stop", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="UIMsgProxy", methodName="SetIsInMsg", parameters=[true], storeResult=(unset var), EveryFrame=false)
    StringSwitch(stringVariable=var "Item", compareTo=["Hunter Combo 1", "Hunter Combo 2", "Socket 0", "Socket 1", "Cursed"], sendEvent=[→"H COMBO 1", →"H COMBO 2", →"SOCKET 1", →"SOCKET 2", →"CURSED"], everyFrame=false)
    on H COMBO 1 → Set H Combo 1
    on H COMBO 2 → Set H Combo 2
    on SOCKET 1 → Set Socket 1
    on SOCKET 2 → Set Socket 2
    on CURSED → Set Cursed
  }

  state Top Up {
    AudioPlayerOneShotSingle(audioPlayer=var "Audio Player", spawnPoint=var "Hero", audioClip=ui_ability_get_from_weavers (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0.2, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Item Name"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name Prefix"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Fleur", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.65, everyFrame=false)
    on FINISHED → Bot Up
  }

  state Bot Up {
    SendEventByName(eventTarget=GameObject(var "Msg 1"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Up
  }

  state Stop Up {
    ActivateGameObject(gameObject=var "Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClip=button (sfxstatic_assets_shared.bundle), pitchMin=0.8, pitchMax=0.8, volume=0.3, delay=0, storePlayer=<null>)
    FadeNestedFadeGroup(Target=var "Icon", ToAlpha=0, FadeTime=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Done
  }

  state Audio Player Actor {
    SetGameObject(variable=var "Audio Player", gameObject=Audio Player NoDestroy (audioobjects_assets_all.bundle), everyFrame=false)
    on FINISHED → Icon Evolve
  }

  state Set H Combo 1 {
    SetGameObject(variable=var "Icon", gameObject=var "Combo 1 Evolve", everyFrame=false)
    GetLanguageString(sheetName="Prompts", convName="UI_MSG_HCOMBO_1_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="UI_MSG_HCOMBO_1_DESC", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Msg 1", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="UI_MSG_HCOMBO_PREFIX", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Audio Player Actor
  }

  state Set H Combo 2 {
    SetGameObject(variable=var "Icon", gameObject=var "Combo 2 Evolve", everyFrame=false)
    GetLanguageString(sheetName="Prompts", convName="UI_MSG_HCOMBO_2_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="UI_MSG_HCOMBO_2_DESC", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Msg 1", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="UI_MSG_HCOMBO_PREFIX", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Audio Player Actor
  }

  state Icon Evolve {
    ActivateGameObject(gameObject=var "Icon", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Top Up
  }

  state BG Up {
    SendEventByName(eventTarget=GameObject(var "BG"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Icon Evolve
  }

  state Set Socket 1 {
    SetGameObject(variable=var "Icon", gameObject=var "Socket Evolve 1", everyFrame=false)
    GetLanguageString(sheetName="Tools", convName="UI_MSG_TITLE_EXTRASLOT_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Tools", convName="UI_MSG_EXTRA_SLOT", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Msg 1", textString=var "Game Text")
    GetLanguageString(sheetName="Tools", convName="UI_MSG_TITLE_EXTRASLOT_PREFIX_1", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Audio Player Actor
  }

  state Set Socket 2 {
    SetGameObject(variable=var "Icon", gameObject=var "Socket Evolve 2", everyFrame=false)
    GetLanguageString(sheetName="Tools", convName="UI_MSG_TITLE_EXTRASLOT_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Tools", convName="UI_MSG_EXTRA_SLOT", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Msg 1", textString=var "Game Text")
    GetLanguageString(sheetName="Tools", convName="UI_MSG_TITLE_EXTRASLOT_PREFIX_2", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Audio Player Actor
  }

  state Set Cursed {
    SetGameObject(variable=var "Icon", gameObject=var "Cursed", everyFrame=false)
    GetLanguageString(sheetName="Tools", convName="CREST_WITCH_NAME", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Tools", convName="CREST_WITCH_DESC", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Msg 1", textString=var "Game Text")
    GetLanguageString(sheetName="Tools", convName="UI_MSG_TITLE_CREST", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    on FINISHED → Audio Player Actor
  }

  var Item: string = ""

  var Game Text: string = ""
  var Audio Player: gameObject = <null>
  var BG: gameObject = UI Msg Crest Evolve/BG
  var Fleur: gameObject = UI Msg Crest Evolve/Fleur
  var Icon: gameObject = <null>
  var Item Name: gameObject = UI Msg Crest Evolve/Item Name
  var Item Name Prefix: gameObject = UI Msg Crest Evolve/Item Name Prefix
  var Msg 1: gameObject = UI Msg Crest Evolve/Msg 1
  var Stop: gameObject = UI Msg Crest Evolve/Stop
  var Combo 1 Evolve: gameObject = UI Msg Crest Evolve/Hunter_Crest_Evolve
  var Combo 2 Evolve: gameObject = UI Msg Crest Evolve/Hunter_Crest_Evolve Final
  var Socket Evolve 1: gameObject = UI Msg Crest Evolve/Tool_Socket_Evolve_lvl1
  var Socket Evolve 2: gameObject = UI Msg Crest Evolve/Tool_Socket_Evolve_lvl2
  var Cursed: gameObject = UI Msg Crest Evolve/Cursed_Crest_Evolve
}
