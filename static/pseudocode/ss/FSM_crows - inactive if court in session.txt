# FSM
# 2 location(s):
#   Greymoor_01 / crows - inactive if court in session
#   Greymoor_02 / crowmen - inactive if crow court is active

// uses template: craw_summons_got_active
fsm FSM {
  start Test

  state Test {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=(none), isFalse=→"FINISHED")
    PlayerDataBoolTest(boolName="CrowCourtInSession", isTrue=→"COURT IN SESSION", isFalse=(none))
    on COURT IN SESSION → Court In Session
    on FINISHED → Not In Session
  }

  state Court In Session {
    ActivateGameObject(gameObject=Self, activate=var "Set Active", recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Not In Session {
    SetBoolValue(boolVariable=var "Activate", boolValue=var "Set Active", everyFrame=false)
    BoolFlip(boolVariable=var "Activate")
    ActivateGameObject(gameObject=Self, activate=var "Activate", recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Set Active: bool = false

  var Activate: bool = false
}
