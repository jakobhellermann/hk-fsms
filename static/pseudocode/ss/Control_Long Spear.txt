# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Long Spear

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Coral_spear_long L")
    FindNamedChild(gameObject=Self, storeResult=var "Coral_spear_long R")
    on FINISHED → Idle
  }

  state Idle {
    on GROUND R → Ground R
    on GROUND L → Ground L
  }

  state Ground R {
    ActivateGameObject(gameObject=var "Coral_spear_long R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Spear Attack
  }

  state Spear Attack {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Ground L {
    ActivateGameObject(gameObject=var "Coral_spear_long L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Spear Attack
  }

  var Coral_spear_long L: gameObject = <null>
  var Coral_spear_long R: gameObject = <null>
}
