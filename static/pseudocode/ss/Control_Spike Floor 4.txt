# Control
# 1 location(s):
#   Cradle_03 / Boss Scene/Spike Floors/Spike Floor 4

fsm Control {
  start Init
  on HERO SPIKED → Spikes Away  // from any state

  state Init {
    ActivateGameObject(gameObject=var "Art", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Silk_yank", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=var "Art", storeResult=var "Spike Collider")
    on FINISHED → Idle
  }

  state Idle {
    Wait(time=1, finishEvent=→"ATTACK", realTime=false)  // disabled
    AudioStopV2(gameObject=var "Audio Loop Rumble", fadeTime=0, cancelOnEarlyExit=false)
    on ATTACK → Attack Delay
    on SPIKE INTRO → Attack Delay
  }

  state Antic {
    ActivateGameObject(gameObject=var "Silk_yank", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    Wait(time=1.9, finishEvent=→"FINISHED", realTime=false)
    SetFloatValue(floatVariable=var "Spike Time", floatValue=8, everyFrame=false)
    SetIntValue(intVariable=var "Hits", intValue=2, everyFrame=false)
    PlayAudioEvent(audioClip=silk_boss_tendrils_fly_down_grab_spikes (sfxstatic_assets_areacradle.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 10, 0), SpawnedPlayerRef=var "")
    AudioPlaySimple(gameObject=var "Audio Loop Rumble", volume=0.75, oneShotClip=<null>)
    on FINISHED → Yank
    on SILK STUNNED → Spikes End
    on ATTACK CLEAR → Spikes End
  }

  state Yank {
    ActivateGameObject(gameObject=var "Art", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=(unset), y=-10, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionTo(gameObject=var "Art", toValue=(0, 0, 0), localSpace=true, time=0.2, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    RandomlyFlipScale(gameObject=var "Art")
    ActivateGameObject(gameObject=var "Spike Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioStopV2(gameObject=var "Audio Loop Rumble", fadeTime=0.2, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=false_knight_jump (sfxstatic_assets_enemyheavy.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 10, 0), SpawnedPlayerRef=var "")
    on FINISHED → Slam
    on SILK STUNNED → Spikes Away
    on ATTACK CLEAR → Spikes Away
  }

  state Slam {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    ActivateGameObject(gameObject=var "Slam Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    RandomFloat(min=-5.5, max=5.5, storeResult=var "Rotation")
    SetRotation(gameObject=var "Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    PlayAudioEvent(audioClip=white_spikes_appear_01 (sfxstatic_assets_areacradle.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 10, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=coral_final_judge_small_land (sfxstatic_assets_areacoral.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Spikes Out
  }

  state Spikes Out {
    FloatAdd(floatVariable=var "Spike Time", add=-1, everyFrame=true, perSecond=true)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Spike Time", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Spikes Away
    on TINK → Tink
    on SILK STUNNED → Spikes Away
    on ATTACK CLEAR → Spike Deactivate
    on TINK HEAVY → Spikes Away
  }

  state Spikes Away {
    AnimatePositionTo(gameObject=var "Art", toValue=(0, -5, 0), localSpace=true, time=0.75, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=→"FINISHED", realTime=false)
    AnimateRotationTo(gameObject=var "Art", fromValue=var "Rotation", toValue=0, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)
    AudioStopV2(gameObject=var "Audio Loop Rumble", fadeTime=0.2, cancelOnEarlyExit=false)
    on FINISHED → Spikes End
  }

  state Spikes End {
    ActivateGameObject(gameObject=var "Art", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetTag(gameObject=Self, tag="Extra Tag")
    on FINISHED → Idle
  }

  state Tink {
    FloatAdd(floatVariable=var "Spike Time", add=-2.5, everyFrame=false, perSecond=false)
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    RandomFloatEither(value1=-1, value2=1, storeResult=var "Rotation Add")
    FloatAdd(floatVariable=var "Rotation", add=var "Rotation Add", everyFrame=false, perSecond=false)
    FloatClamp(floatVariable=var "Rotation", minValue=-9, maxValue=9, everyFrame=false)
    SetRotation(gameObject=var "Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    Translate(gameObject=var "Art", vector=(unset), x=(unset), y=-0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Spikes Out
    on SILK STUNNED → Spikes Away
    on ATTACK CLEAR → Spike Deactivate
    on TINK HEAVY → Spikes Away
  }

  state Attack Delay {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    SetTag(gameObject=Self, tag="Untagged")
    on FINISHED → Antic
    on ATTACK CLEAR → Spikes End
  }

  state Spike Deactivate {
    AudioStopV2(gameObject=var "Audio Loop Rumble", fadeTime=0, cancelOnEarlyExit=false)
    ActivateGameObject(gameObject=var "Spike Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Spikes Away
  }

  var Rotation: float = 0
  var Spike Time: float = 0
  var Rotation Add: float = 0
  var Hits: int = 0
  var Silk_yank: gameObject = Boss Scene/Spike Floors/Spike Floor 4/Silk_yank
  var Art: gameObject = Boss Scene/Spike Floors/Spike Floor 4/Art
  var Pt Antic: gameObject = Boss Scene/Spike Floors/Spike Floor 4/Pt Antic
  var Slam Effects: gameObject = Boss Scene/Spike Floors/Spike Floor 4/Slam Effects
  var Hazard Respawn Marker: gameObject = Boss Scene/Spike Floors/Spike Floor 4/Hazard Respawn Marker
  var Spike Collider: gameObject = <null>
  var Audio Loop Rumble: gameObject = Boss Scene/Spike Floors/Spike Floor 4/Audio Loop Rumble
}
