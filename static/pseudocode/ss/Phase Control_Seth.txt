# Phase Control
# 1 location(s):
#   Shellwood_22 / Boss Scene/Seth

fsm Phase Control {
  start Init

  state Init {
    GetHP(target=Self, storeValue=var "P2 HP")
    MultiplyIntByFloat(integer=var "P2 HP", multiplyFloat=0.75, storeResult=var "P2 HP", everyFrame=false, forceRoundUp=false)
    GetHP(target=Self, storeValue=var "P3 HP")
    MultiplyIntByFloat(integer=var "P3 HP", multiplyFloat=0.4, storeResult=var "P3 HP", everyFrame=false, forceRoundUp=false)
    on FINISHED → P1
  }

  state P1 {
    on TOOK DAMAGE → Check 1
  }

  state Check 1 {
    CompareHP(enemy=var "Self", integer2=var "P2 HP", equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P1
    on NEXT → Set P2
  }

  state P2 {
    on TOOK DAMAGE → Check 2
  }

  state Set P2 {
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time", setValue=0.5, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="P2", setValue=true, everyFrame=false)
    on FINISHED → P2
  }

  state Check 2 {
    CompareHP(enemy=var "Self", integer2=var "P3 HP", equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P2
    on NEXT → Set P3
  }

  state Set P3 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="P3 STUN", delay=0, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Idle Time", setValue=0.3, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="P2", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="P3", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Do Rage Stomp", setValue=true, everyFrame=false)
    on FINISHED → 
  }

  var P2 HP: int = 0
  var P3 HP: int = 0
  var Self: gameObject = Boss Scene/Seth
}
