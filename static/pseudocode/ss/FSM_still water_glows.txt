# FSM
# 1 location(s):
#   Abyss_Cocoon / Boss Control/still water_glows

fsm FSM {
  start State 1
  on EMERGE WAVES → State 3  // from any state

  state State 1 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=1)
    on ABYSS WAVES → State 2
  }

  state State 2 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=1)
    Wait(time=8, finishEvent=(none), realTime=false)
    on FINISHED → State 1
  }

  state State 3 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=1)
    Wait(time=5, finishEvent=(none), realTime=false)
    on FINISHED → State 1
  }
}
