# Control
# 1 location(s):
#   Bone_05_boss / Boss Scene/BoneBeast AtkBell R

fsm Control {
  start Init

  state Init {
    SetParent(gameObject=var "Pt Rock", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Dormant
  }

  state Dormant {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    StopParticleEmitter(gameObject=var "Pt Dust")
    on LAUNCH → Launch
  }

  state Launch {
    ArrayGetRandom(array=var "Sprites", storeValue=var "Sprite Crt", index=(unset), noRepeat=false, everyFrame=false)
    SetSpriteRendererSprite(gameObject=Self, sprite=var "Sprite Crt")
    SetFloatValue(floatVariable=var "End Timer", floatValue=0, everyFrame=false)
    SetCircleCollider(gameObject=Self, active=true)
    SetFloatValue(floatVariable=var "Speed Crt", floatValue=var "X Speed", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed Crt", y=38, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    FloatOperator(float1=var "X Speed", float2=-100, operation=Multiply, storeResult=var "Speed Crt", everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Speed Crt", everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetSpriteRenderer(gameObject=Self, active=true)
    PlayParticleEmitter(gameObject=var "Pt Dust", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt DustStart", emit=0, resetIfPlaying=false)
    on FINISHED → Launched Up
  }

  state Launched Up {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=1, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Ground Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    on LAND → First Bounce
  }

  state First Bounce {
    AudioPlayRandom(gameObject=var "Self", audioClips=[bell_ball_bounce_1 (sfxstatic_assets_areabell.bundle), bell_ball_bounce_2 (sfxstatic_assets_areabell.bundle)], weights=[0.5, 0.5], pitchMin=1.2, pitchMax=0.8)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    FloatOperator(float1=var "X Speed", float2=12, operation=Multiply, storeResult=var "Speed Crt", everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Speed Crt", y=14, everyFrame=false)
    FloatOperator(float1=var "Speed Crt", float2=-50, operation=Multiply, storeResult=var "Speed Crt", everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Speed Crt", everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Rock", emit=0, resetIfPlaying=false)
    on FINISHED → In Air
  }

  state In Air {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=1, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Ground Y", tolerance=0, equal=→"LAND", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    FloatAdd(floatVariable=var "End Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "End Timer", float2=2, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"END", everyFrame=true)
    on LAND → Bounce
    on END → Dormant
  }

  state Bounce {
    AudioPlayRandom(gameObject=var "Self", audioClips=[bell_impact_new_bell_1 (sfxstatic_assets_areabell.bundle), bell_impact_new_bell_2 (sfxstatic_assets_areabell.bundle), bell_impact_new_bell_3 (sfxstatic_assets_areabell.bundle), bell_impact_new_bell_4 (sfxstatic_assets_areabell.bundle)], weights=[0.25, 0.25, 0.25, 0.25], pitchMin=1.1, pitchMax=0.9)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Ground Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=14, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Rock", emit=0, resetIfPlaying=false)
    on FINISHED → In Air
  }

  var X Speed: float = 1

  var End Timer: float = 0
  var Ground Y: float = -4.32
  var Self Y: float = 0
  var Speed Crt: float = 0
  var Boss Scene: gameObject = <null>
  var Long Spear: gameObject = <null>
  var Pt Dust: gameObject = Boss Scene/BoneBeast AtkBell R/Pt Dust
  var Pt DustStart: gameObject = Boss Scene/BoneBeast AtkBell R/Pt DustStart
  var Pt Rock: gameObject = Boss Scene/BoneBeast AtkBell R/Pt Rock
  var Self: gameObject = <null>
  var Sprite Crt: object = <null>
  var Sprites: array = var "Sprites"
}
