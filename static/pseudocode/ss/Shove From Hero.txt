# Shove From Hero
# 1 location(s):
#   localpoolprefabs_assets_areawisp.bundle / Wisp Fireball

fsm Shove From Hero {
  start Idle
  on HERO SHOVE STOP → Idle  // from any state

  state Idle {
    on HERO SHOVE START → Range Out
  }

  state Range Out {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Hero Shove Range", storeResult=false, sendEvent="RANGE IN", outOfRangeEvent="", everyFrame=true)
    on RANGE IN → Range In
  }

  state Range In {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Hero Shove Range", storeResult=false, sendEvent="", outOfRangeEvent="RANGE OUT", everyFrame=true)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=true)
    FloatAdd(floatVariable=var "Angle", add=180, everyFrame=true, perSecond=false)
    AddForce2dAsAngle(gameObject=Self, forceMode=0, atPosition=(unset), angle=var "Angle", speed=40, maxSpeed=9, maxSpeedX=(unset), maxSpeedY=(unset), everyFrame=true)
    on RANGE OUT → Range Out
  }

  var Angle: float = 0
}
