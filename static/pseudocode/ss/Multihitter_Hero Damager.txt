fsm Multihitter {
  start Init

  state Init {
    GetGrandparent(gameObject=Self, storeResult=var "Grandparent")
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    on TRIGGER STAY 2D → Start Hit?
  }

  state Start Hit? {
    CanHeroTakeDamage(eventTarget=Self, canTakeDmgEvent=→"HIT", cannotTakeDmgEvent=→"HIT")
    on CANCEL → Idle
    on HIT → Hit
  }

  state Hit {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="TakeQuickDamage", parameters=[1, true], storeResult=(unset var), EveryFrame=false)  // disabled
    SendEventToRegister(eventName="HERO DAMAGED")
    PlayAudioEvent(audioClip=hornet_fell_into_gears (sfxstatic_assets_areacog.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="WOUND START", delay=0, everyFrame=false)
    Wait(time=0.75, finishEvent=→"MULTIHIT END", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="MULTIHIT", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Grandparent"), sendEvent="MULTIHIT", delay=0, everyFrame=false)
    on MULTIHIT END → Idle
  }

  var Parent: gameObject = <null>
  var Grandparent: gameObject = <null>
}
