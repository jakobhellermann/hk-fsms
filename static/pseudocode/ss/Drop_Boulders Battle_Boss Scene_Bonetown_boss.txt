# Drop
# 1 location(s):
#   Bonetown_boss / Boss Scene/Boulders Battle

fsm Drop {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    SetIntValue(intVariable=var "Loop", intValue=0, everyFrame=false)
    on DROP → Drop 1
  }

  state Drop 1 {
    GetRandomChild(gameObject=Self, storeResult=var "Boulder")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Hero X", float2=10, operation=Add, storeResult=var "Max X", everyFrame=false)
    FloatOperator(float1=var "Hero X", float2=10, operation=Subtract, storeResult=var "Min X", everyFrame=false)
    GetPosition(gameObject=var "Boulder", vector=(unset), x=var "Boulder X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Boulder X", lowerValue=var "Min X", upperValue=var "Max X", boolVariable=(unset), trueEvent=(none), falseEvent=→"RETRY", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Boulder"), sendEvent="DROP", delay=0, everyFrame=false)
    on RETRY → Retry 1
    on FINISHED → Pause
  }

  state Retry 1 {
    IntAdd(intVariable=var "Loop", add=1, everyFrame=false)
    IntCompare(integer1=var "Loop", integer2=900, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetIntValue(intVariable=var "Loop", intValue=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Drop 1
  }

  state Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)  // disabled
    WaitRandom(timeMin=0.1, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    SetIntValue(intVariable=var "Loop", intValue=0, everyFrame=false)
    on FINISHED → Drop 2
  }

  state Drop 2 {
    GetRandomChild(gameObject=Self, storeResult=var "Boulder")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Hero X", float2=16, operation=Add, storeResult=var "Max X", everyFrame=false)
    FloatOperator(float1=var "Hero X", float2=16, operation=Subtract, storeResult=var "Min X", everyFrame=false)
    GetPosition(gameObject=var "Boulder", vector=(unset), x=var "Boulder X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Boulder X", lowerValue=var "Min X", upperValue=var "Max X", boolVariable=(unset), trueEvent=(none), falseEvent=→"RETRY", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Boulder"), sendEvent="DROP", delay=0, everyFrame=false)
    on RETRY → Retry 2
    on FINISHED → Idle
  }

  state Retry 2 {
    IntAdd(intVariable=var "Loop", add=1, everyFrame=false)
    IntCompare(integer1=var "Loop", integer2=900, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    SetIntValue(intVariable=var "Loop", intValue=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Drop 2
  }

  var Hero X: float = 0
  var Boulder X: float = 0
  var Max X: float = 0
  var Min X: float = 0
  var Loop: int = 0
  var Boulder: gameObject = <null>
}
