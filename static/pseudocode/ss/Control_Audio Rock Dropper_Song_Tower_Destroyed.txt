# Control
# 1 location(s):
#   Song_Tower_Destroyed / Audio Rock Dropper

fsm Control {
  start Pause

  state Pause {
    WaitRandom(timeMin=0.5, timeMax=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Drop
  }

  state Drop {
    RandomFloat(min=5, max=18, storeResult=var "X Offset")
    RandomlyFlipFloat(storeResult=var "X Offset")
    SetVector3XYZ(vector3Variable=var "Audio Offset", vector3Value=(unset), x=var "X Offset", y=(unset), z=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Use Own AudioSource", isTrue=→"SELF", isFalse=(none), everyFrame=false)
    PlayRandomAudioClipTable(Table=rock_fall (sfxdynamic_assets_areabone.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=var "Audio Offset")
    on FINISHED → Pause
    on SELF → Self Audio
  }

  state Self Audio {
    SetPosition2d(gameObject=var "Audio Child", vector=(unset), x=var "X Offset", y=(unset), space=1, everyFrame=false, lateUpdate=false)
    GetRandomAudioClipFromTable(Table=rock_fall (sfxdynamic_assets_areabone.bundle), ForcePlay=false, StoreClip=var "Clip")
    AudioPlaySimple(gameObject=var "Audio Child", volume=(unset), oneShotClip=var "Clip")
    on FINISHED → Pause
  }

  var Use Own AudioSource: bool = true

  var X Offset: float = 0
  var Audio Offset: vector3 = (0, 0, 0)
  var Self: gameObject = Audio Rock Dropper
  var Audio Child: gameObject = Audio Rock Dropper/Audio Child
  var Clip: object = <null>
}
