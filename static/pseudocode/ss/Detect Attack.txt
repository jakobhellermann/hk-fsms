fsm Detect Attack {
  start Pause

  state Pause {
    GetParent(gameObject=Self, storeResult=var "Tendril")
    GetParent(gameObject=var "Tendril", storeResult=var "Barnacle")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Collider
  }

  state Set Collider {
    GetBoxCollider2DSizeVector(gameObject1=var "Tendril", size=var "Box Size", offset=var "Box Offset")
    SetBoxCollider2DSizeVector(gameObject1=Self, size=var "Box Size", offset=var "Box Offset")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEventV2(gameObject=Self, trigger=0, collideTag="", collideLayer=Attack(17), sendEvent=→"TRIGGER", storeCollider=var "Attack Object")
    on TRIGGER → Check Damager
  }

  state Check Damager {
    GameObjectCompareTag(gameObject=var "Attack Object", tag="Breaker", trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    HasComponent(gameObject=var "Attack Object", component="DamageEnemies", removeOnExit=false, trueEvent=(none), falseEvent=→"CANCEL", store=(unset), everyFrame=false)
    CallMethodProper(gameObject=var "Attack Object", behaviour="DamageEnemies", methodName="GetDamageDealt", parameters=[], storeResult=var "DamageDealt", EveryFrame=false)
    IntCompare(integer1=var "DamageDealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    on CANCEL → Recover
    on FINISHED → Send Attack Event
  }

  state Recover {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Send Attack Event {
    GetPosition2D(gameObject=var "Attack Object", vector=var "Grabbed Object Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self Pos X", y=(unset), space=0, everyFrame=false)
    SetVector2XY(vector2Variable=var "Grabbed Object Pos", vector2Value=(unset), x=var "Self Pos X", y=(unset), everyFrame=false)
    SetFsmVector2(gameObject=var "Barnacle", fsmName="Control", variableName="Grabbed Object Pos", setValue=var "Grabbed Object Pos", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Barnacle"), sendEvent="TENDRIL ATTACKED", delay=0, everyFrame=false)
    on FINISHED → Recover
  }

  var Self Pos X: float = 0
  var DamageDealt: int = 0
  var Attack Object Tag: string = ""
  var Box Size: vector2 = (0, 0)
  var Box Offset: vector2 = (0, 0)
  var Grabbed Object Pos: vector2 = (0, 0)
  var Tendril: gameObject = <null>
  var Barnacle: gameObject = <null>
  var Attack Object: gameObject = <null>
}
