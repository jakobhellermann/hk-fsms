# Flea Control
# 1 location(s):
#   Slab_Cell / Flea Slab Cage

fsm Flea Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Flea", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Rescue {
    ActivateGameObject(gameObject=var "Flea", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "lamp")
    EnableFSM(gameObject=var "lamp", fsmName="Flea Jitter", enable=false, resetOnExit=false)
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
  }

  state Idle {
    CheckQuestPdSceneBool(QuestTarget=var "Quest Tracker", ExpectedValue=true, trueEvent=→"ACTIVATE", falseEvent=(none), storeValue=(unset))
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    AudioPlayInState(gameObject=var "Audio Loop Voice", volume=1)
    on ACTIVATE → Activated
    on BREAK → Set Flea Dir
  }

  state Activated {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set Flea Dir {
    CheckTargetDirection(gameObject=var "Flea", target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"SWING L", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"SWING R", delay=0, everyFrame=false)
    on SWING L → Swing L
    on SWING R → Swing R
  }

  state Swing L {
    SetFsmBool(gameObject=var "Flea", fsmName="Control", variableName="Emerge Left", setValue=true, everyFrame=false)
    on FINISHED → Rescue
  }

  state Swing R {
    SetFsmBool(gameObject=var "Flea", fsmName="Control", variableName="Emerge Right", setValue=true, everyFrame=false)
    on FINISHED → Rescue
  }

  var Break Effects: gameObject = <null>
  var Flea: gameObject = Flea Slab Cage/Flea Rescue Activation
  var lamp: gameObject = <null>
  var Audio Loop Voice: gameObject = Flea Slab Cage/Audio Loop Voice
  var Quest Tracker: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
