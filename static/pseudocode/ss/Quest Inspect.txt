# Quest Inspect
# 1 location(s):
#   Room_Pinstress / Pinstress States/Room Regular Parent/Peak_Summons/Inspect Region

fsm Quest Inspect {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Peak_Summons")
    FindNamedChild(gameObject=var "Peak_Summons", storeResult=var "Rag Pickup Effect")
    FindNamedChild(gameObject=var "Peak_Summons", storeResult=var "inspect plink")
    CheckQuestState(NotTrackedEvent=(none), TrackedEvent=→"QUEST TAKEN", CompletedEvent=(none), Quest=Pinstress Battle (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    on QUEST TAKEN → Quest Taken
    on FINISHED → Idle
  }

  state Quest Taken {
    SetSpriteRenderer(gameObject=var "Peak_Summons", active=false)
    ActivateGameObject(gameObject=var "inspect plink", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    on INTERACT → Inspect
  }

  state Inspect {
    RunDialogue(Sheet="Inspect", Key="PINSTRESS_SUMMONS", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=Top, OffsetY=0, Target=Self)
    on CONVO_END → Close Dialogue
  }

  state Close Dialogue {
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=Self, UseChildren=false)
    Wait(time=0.3, finishEvent=(none), realTime=false)
    on FINISHED → Quest Prompt
  }

  state Quest Prompt {
    CheckQuestStateV2(NotTrackedEvent=(none), IncompleteEvent=→"ALREADY ACCEPTED", CompletedEvent=(none), Quest=Pinstress Battle Pre (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    QuestYesNoV2(Quest=Pinstress Battle (dataassets_assets_assets/dataassets/questsystem/quests.bundle), BeginQuest=false, YesEvent=→"ACCEPT", NoEvent=→"DECLINE", ReturnHUDAfter=false)
    on DECLINE → End
    on ACCEPT → Hornet Reach
    on ALREADY ACCEPTED → Already Accepted
  }

  state End {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Hornet Reach {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    PlayAudioEvent(audioClip=hornet_misc_foley_move_2 (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StopAnimationControl())
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Stand 1")
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Take Summons
  }

  state Take Summons {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetSpriteRenderer(gameObject=var "Peak_Summons", active=false)
    ActivateGameObject(gameObject=var "inspect plink", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rag Pickup Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Stand 3", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Start Quest
  }

  sequence state Start Quest {
    BoolTest(boolVariable=var "Already Accepted", isTrue=→"ALREADY ACCEPTED", isFalse=(none), everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    BeginQuest(Quest=Pinstress Battle (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    on FINISHED → End
    on ALREADY ACCEPTED → Start Quest Quick
  }

  state Already Accepted {
    SetBoolValue(boolVariable=var "Already Accepted", boolValue=true, everyFrame=false)
    on FINISHED → Hornet Reach
  }

  state Start Quest Quick {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    BeginQuest(Quest=Pinstress Battle (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    on FINISHED → Idle
  }

  var Already Accepted: bool = false
  var Peak_Summons: gameObject = <null>
  var Rag Pickup Effect: gameObject = <null>
  var inspect plink: gameObject = <null>
}
