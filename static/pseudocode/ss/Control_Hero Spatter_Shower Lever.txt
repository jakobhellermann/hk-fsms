# Control
# 1 location(s):
#   Song_10 / Shower System/Shower Lever/Hero Spatter

fsm Control {
  start Init
  on SHOWER OFF → End Pause  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Splash")
    ActivateGameObject(gameObject=<null>, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Dormant
  }

  sequence state Dormant {
    StopParticleEmitter(gameObject=var "Pt Splash")
    Wait(time=0.8, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=<null>, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on SHOWER ON → Pause
  }

  state Pause {
    Wait(time=2.45, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Pt
  }

  state Out {
    SetBoolValue(boolVariable=var "Will Drip", boolValue=false, everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Shower Range", storeResult=false, sendEvent="IN", outOfRangeEvent="", everyFrame=true)
    SetParticleEmission(gameObject=var "Pt Splash", emission=false, resetOnExit=false)
    StopParticleEmitter(gameObject=var "Pt Flame Steam")
    AudioStopV2(gameObject=var "Audio Loop Steam", fadeTime=0, cancelOnEarlyExit=false)
    SetBoolValue(boolVariable=var "Burning", boolValue=false, everyFrame=false)
    on IN → In
  }

  state In {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Shower Range", storeResult=false, sendEvent="", outOfRangeEvent="OUT", everyFrame=true)
    SetParticleEmission(gameObject=var "Pt Splash", emission=true, resetOnExit=false)
    SetPositionToObject(gameObject=var "Pt Splash", targetObject=var "Hero", xOffset=0, yOffset=0, zOffset=0, overrideZ=0.002, everyFrame=true)
    Wait(time=0.5, finishEvent=→"CLEAN", realTime=false)
    CheckHeroNailImbuement(checkType=Fire, trueEvent=(none), falseEvent=(none), currentElementResult=None, storeResult=var "Is Flame Imbued", everyFrame=true)
    BoolTestMulti(boolVariables=[var "Is Flame Imbued", var "Burning"], boolStates=[true, false], trueEvent=→"START BURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Is Flame Imbued", var "Burning"], boolStates=[false, true], trueEvent=→"END BURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on OUT → Drip?
    on CLEAN → Reduce Odours
    on START BURN → Start Burn
    on END BURN → End Burn
  }

  state Start Pt {
    PlayParticleEmitter(gameObject=var "Pt Splash", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=<null>, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Out
  }

  state End Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    SetPositionToObject(gameObject=var "Pt Splash", targetObject=var "Hero", xOffset=0, yOffset=0, zOffset=0, overrideZ=0.002, everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Shower Range", storeResult=false, sendEvent="", outOfRangeEvent="FINISHED", everyFrame=true)
    on FINISHED → Dormant
  }

  state Reduce Odours {
    SetBoolValue(boolVariable=var "Will Drip", boolValue=true, everyFrame=false)
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=ReduceOdours(10))
    on FINISHED → In
  }

  state Drip? {
    BoolTest(boolVariable=var "Will Drip", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Dripper White (globalpoolprefabs_assets_all.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "Dripper")
    CallMethodProper(gameObject=var "Dripper", behaviour="Dripper", methodName="StartDripper", parameters=[var "Hero"], storeResult=(unset var), EveryFrame=false)
    SetBoolValue(boolVariable=var "Will Drip", boolValue=false, everyFrame=false)
    on FINISHED → Out
  }

  state Start Burn {
    PlayParticleEmitter(gameObject=var "Pt Flame Steam", emit=0, resetIfPlaying=false)
    SetBoolValue(boolVariable=var "Burning", boolValue=true, everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Loop Steam", volume=1, oneShotClip=<null>)
    on FINISHED → In
  }

  state End Burn {
    StopParticleEmitter(gameObject=var "Pt Flame Steam")
    SetBoolValue(boolVariable=var "Burning", boolValue=false, everyFrame=false)
    AudioStopV2(gameObject=var "Audio Loop Steam", fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → In
  }

  var Is Flame Imbued: bool = false
  var Will Drip: bool = false
  var Burning: bool = false
  var Pt Splash: gameObject = <null>
  var Dripper: gameObject = <null>
  var Pt Flame Steam: gameObject = Shower System/Shower Lever/Hero Spatter/Pt Flame Steam
  var Audio Loop Steam: gameObject = Shower System/Audio Loop Steam
}
