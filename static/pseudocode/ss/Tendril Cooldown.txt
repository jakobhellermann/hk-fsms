# Tendril Cooldown
# 1 location(s):
#   Abyss_Cocoon / Boss Control/Lost Lace Boss

fsm Tendril Cooldown {
  start Idle

  state Idle {
    FloatCompare(float1=var "Cooldown", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"COUNTDOWN", everyFrame=true)
    on COUNTDOWN → Countdown
  }

  state Countdown {
    FloatAdd(floatVariable=var "Cooldown", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Cooldown", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    on FINISHED → Idle
  }

  var Cooldown: float = 0
}
