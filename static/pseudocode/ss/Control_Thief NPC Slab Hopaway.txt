# Control
# 1 location(s):
#   Slab_14 / Thief NPC Scene/Thief NPC Slab Hopaway

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="HasSlabKeyA", isTrue=→"AWAY", isFalse=(none))
    CheckIfCrestEquipped(Crest=Cloakless (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=(none), falseEvent=→"AWAY", storeValue=(unset))
    on AWAY → Away
    on FINISHED → Init
  }

  state Away {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Waiting {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wave Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Leave Range", storeResult=false, sendEvent="LEAVE", outOfRangeEvent="", everyFrame=true)
    on ALERT → Wave Pause
    on LEAVE → Leave
  }

  state Leave {
    ActivateInteractible(Target=var "Slab Lock L", Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Away
  }

  state Wave Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wave
    on STARTLE → Startle
  }

  state Wave {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Wave")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Grindle_call_out (sfxdynamic_assets_areabone.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SetPlayerDataInt(intName="grindleSlabSequence", value=1)
    on FINISHED → Wave End
    on STARTLE → Startle
  }

  state Wave End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wave End Q", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
    on STARTLE → Startle
  }

  state Leap Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Leap", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=(unset), brakeOnExit=false)
    CheckXPosition(gameObject=Self, compareTo=105, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"LEAVE", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    PlayAudioEvent(audioClip=bone_hopper_land (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Leap
    on LEAVE → Leave
  }

  state Leap {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=25, y=(unset), everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.95, decelerationY=(unset), brakeOnExit=false)
    ActivateInteractible(Target=var "Slab Lock L", Activate=true, AllowQueueing=false, UseChildren=false)
    PlayAudioEvent(audioClip=bone_hopper_jump (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Land
  }

  state Turn {
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[Grindle_laugh_01 (sfxstatic_assets_areasong.bundle)], weights=[1], pitchMin=1, pitchMax=1, stopPreviousSound=false)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Leap Antic
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Slab Lock L")
    ActivateInteractible(Target=var "Slab Lock L", Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    on FINISHED → Waiting
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Leave Range", storeResult=false, sendEvent="LEAVE", outOfRangeEvent="", everyFrame=true)
    on INTERACT → Convo Choice
    on LEAVE → Leave
    on STARTLE → Startle
  }

  state Convo Choice {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    PlayerDataBoolTest(boolName="grindleReleasedFromBonejail", isTrue=→"MET", isFalse=→"UNMET")
    on MET → Met
    on UNMET → Unmet
  }

  state Met {
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=true, npcTitle="GRINDLE")
    RunDialogue(Sheet="Wanderers", Key="GRINDLE_SLAB_EARLY", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Leave Pause
  }

  state Unmet {
    RunDialogue(Sheet="Wanderers", Key="GRINDLE_SLAB_EARLY_NOTMET", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Leave Pause
  }

  state Leave Pause {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    on FINISHED → Turn
  }

  state Land {
    PlayAudioEvent(audioClip=bone_hopper_land (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Leap Antic
  }

  sequence state Startle {
    SetPlayerDataInt(intName="grindleSlabSequence", value=1)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    PlayAudioEvent(audioClip=bone_hopper_jump (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Grindle_alert (sfxdynamic_assets_areasong.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName="Startle", AnimationCompleteEvent=(none))
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  var Activated: bool = false
  var Parent: gameObject = <null>
  var Slab Lock L: gameObject = <null>
  var Slab Lock R: gameObject = <null>
  var Audio Loop Voice: gameObject = Thief NPC Scene/Thief NPC Slab Hopaway/Audio Loop Voice
}
