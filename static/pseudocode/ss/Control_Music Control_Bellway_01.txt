# Control
# 1 location(s):
#   Bellway_01 / Music Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"SILENT", isFalse=(none))
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="left1", equalEvent=(none), notEqualEvent=→"SILENT", storeResult=(unset), everyFrame=false)
    PlayerDataBoolTest(boolName="skullKingInvaded", isTrue=→"BATTLE", isFalse=→"SILENT")
    on SILENT → Silent
    on BATTLE → Battle
  }

  state Silent {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=8)
  }

  state Battle {
    TransitionToAudioSnapshot(snapshot=Normal Soft (audioobjects_assets_all.bundle), transitionTime=4)
  }

  var Entry Gate: string = ""
}
