# FSM
# 1 location(s):
#   Bone_12 / Black Thread States/Normal World/Jail Scene 2

fsm FSM {
  start Pre-Init

  state Init {
    ActivateGameObject(gameObject=var "Thief NPC Save", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BREAK → Break
    on BROKEN → Broken
  }

  state Break {
    ActivateGameObject(gameObject=var "Thief Dummy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thief NPC Save", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Broken {
    ActivateGameObject(gameObject=var "Thief Dummy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thief NPC Save", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Thief NPC Saved", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pre-Init {
    FindChild(gameObject=Self, childName="Thief Dummy", storeResult=var "Thief Dummy")
    FindChild(gameObject=Self, childName="Thief NPC Save", storeResult=var "Thief NPC Save")
    FindChild(gameObject=Self, childName="Thief NPC Saved", storeResult=var "Thief NPC Saved")
    ActivateGameObject(gameObject=var "Thief NPC Save", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Init
    on BROKEN → Broken
  }

  var Thief Dummy: gameObject = <null>
  var Thief NPC Save: gameObject = <null>
  var Thief NPC Saved: gameObject = <null>
}
