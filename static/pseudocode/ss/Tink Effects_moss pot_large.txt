# Tink Effects
# 1 location(s):
#   Mosstown_02c / moss_hut_set/NPC Scene/moss pot_large

fsm Tink Effects {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Hit")
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Moss Creep NPC")
    on FINISHED → Idle
  }

  state Idle {
    on TINK → Effects
  }

  state Effects {
    PlayParticleEmitterChildren(gameObject=var "Pt Hit", resetTimeIfPlaying=false, stopOnStateExit=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Moss Creep NPC"), sendEvent="TINK POT", delay=0, everyFrame=false)
    PlayAudioEvent(audioClip=acid_hiss (sfxstatic_assets_shared.bundle), pitchMin=0.85, pitchMax=1.15, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Idle
  }

  var Pt Hit: gameObject = <null>
  var Parent: gameObject = <null>
  var Moss Creep NPC: gameObject = <null>
}
