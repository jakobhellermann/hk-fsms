# Atmos Pan Control
# 2 location(s):
#   coremanagers_assets__gamemanager.bundle / _GameManager/AudioManager/Atmos
#   Menu_Title / _GameManager/AudioManager/Atmos

fsm Atmos Pan Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    EnableFsmSelf(SetEnabled=false)
    on LEAVING SCENE → Reset Pan
  }

  state Reset Pan {
    PanAudio(gameObject=var "Layer 1", targetPan=0, overTime=0.5)
    PanAudio(gameObject=var "Layer 2", targetPan=0, overTime=0.5)
    PanAudio(gameObject=var "Layer 3", targetPan=0, overTime=0.5)
    PanAudio(gameObject=var "Layer 4", targetPan=0, overTime=0.5)
    PanAudio(gameObject=var "Layer 5", targetPan=0, overTime=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  var Layer 1: gameObject = _GameManager/AudioManager/Atmos/Layer 1
  var Layer 2: gameObject = _GameManager/AudioManager/Atmos/Layer 2
  var Layer 3: gameObject = _GameManager/AudioManager/Atmos/Layer 3
  var Layer 4: gameObject = _GameManager/AudioManager/Atmos/Layer 4
  var Layer 5: gameObject = _GameManager/AudioManager/Atmos/Layer 5
}
