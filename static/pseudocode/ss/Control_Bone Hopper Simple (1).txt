# Control
# 1 location(s):
#   Bone_East_03 / Bone Hopper Simple (1)

fsm Control {
  start Init

  state Init {
    BoolTest(boolVariable=var "Wait For Hero", isTrue=→"WAIT", isFalse=(none), everyFrame=false)
    RandomlyFlipScale(gameObject=Self)
    on FINISHED → Next
    on WAIT → Wait For Hero
  }

  state Jump Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Jump Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → Jump Voice?
    on SING → Sing
    on FALL → Fall
  }

  state Jump Launch {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    NextFrameEvent(sendEvent=→"FINISHED")
    RandomFloat(min=14.5, max=16, storeResult=var "Jump Y")
    SetVelocityByScale(gameObject=Self, speed=var "X Speed", ySpeed=var "Jump Y", everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Jump", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=bone_hopper_jump (sfxstatic_assets_areawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Jump Air
  }

  state Jump Air {
    SetVelocityByScale(gameObject=Self, speed=var "X Speed", ySpeed=(unset), everyFrame=true)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Land {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Land", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=bone_hopper_land (sfxstatic_assets_areawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AdjustColliderPosToEdge(Target=Self, Width=0.2)
    on FINISHED → Follow Leader?
  }

  state Next {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignToBool(Value=var "X Scale", StoreIsPositive=var "Facing L", EveryFrame=false)
    BoolTestMulti(boolVariables=[var "Facing L", var "Force Turn R"], boolStates=[true, true], trueEvent=→"TURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing L", var "Force Turn L"], boolStates=[false, true], trueEvent=→"TURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    RayCast2dV2(fromGameObject=var "Ray Pt", fromPosition=(unset), direction=(-1, 0), space=1, distance=4, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "Wall Ahead", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    RayCast2dV2(fromGameObject=var "Ray Pt", fromPosition=(0, 1), direction=(-1, 0), space=1, distance=4, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "High Wall Ahead", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTestMulti(boolVariables=[var "High Wall Ahead", var "Wall Ahead"], boolStates=[true, true], trueEvent=→"TURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "High Wall Ahead", var "Wall Ahead"], boolStates=[false, true], trueEvent=→"HIGH JUMP", falseEvent=(none), storeResult=(unset), everyFrame=false)
    RayCast2dV2(fromGameObject=var "Ray Pt Ahead", fromPosition=(unset), direction=(0, -1), space=1, distance=var "Drop Distance", minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"TURN", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolTest(boolVariable=var "Startled", isTrue=→"JUMP", isFalse=(none), everyFrame=false)
    SendRandomEventV4(events=[→"JUMP", →"IDLE"], weights=[0.8, 0.2], eventMax=[4, 1], missedMax=[1, 4], activeBool=(unset))
    on TURN → Turn
    on IDLE → Set Idle Time
    on JUMP → Jump Antic
    on HIGH JUMP → High Jump Antic
    on SING → Sing
  }

  state Turn {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlipScaleOnExit(gameObject=Self, flipHorizontally=true, flipVertically=false)  // disabled
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Jump Antic
  }

  state Set Idle Time {
    RandomFloat(min=1, max=4, storeResult=var "Idle Time")
    RandomFloat(min=1, max=3, storeResult=var "Turn Time")
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle F")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=var "Idle Time", finishEvent=→"END", realTime=false)
    Wait(time=var "Turn Time", finishEvent=→"TURN", realTime=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on END → Idle Turn?
    on TURN → Look Back
    on COAL → Next
    on TOOK DAMAGE → Follow Leader?
    on SING → Sing
    on FALL → Fall
  }

  state Idle Turn? {
    SendRandomEventV4(events=[→"END", →"TURN"], weights=[0.65, 0.35], eventMax=[2, 1], missedMax=[1, 2], activeBool=(unset))
    on TURN → Idle Turn
    on END → Follow Leader?
  }

  state Idle Turn {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FlipScaleOnExit(gameObject=Self, flipHorizontally=true, flipVertically=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → Follow Leader?
    on FALL → Fall
  }

  state Look Back {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Back")
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"END", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[bone_hopper_simple_alert_1 (sfxstatic_assets_areawilds.bundle), bone_hopper_simple_alert_2 (sfxstatic_assets_areawilds.bundle), bone_hopper_simple_alert_3 (sfxstatic_assets_areawilds.bundle)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.15, stopPreviousSound=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on END → Look Return
    on COAL → Follow Leader?
    on TOOK DAMAGE → Follow Leader?
    on SING → Sing
    on FALL → Fall
  }

  state Look Return {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle Turn?
  }

  state High Jump Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Jump Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    on FINISHED → High Jump Launch
    on FALL → Fall
  }

  state High Jump Launch {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    NextFrameEvent(sendEvent=→"FINISHED")
    RandomFloat(min=15, max=17, storeResult=var "Jump Y")  // disabled
    SetVelocityByScale(gameObject=Self, speed=var "X Speed", ySpeed=21, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Jump", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=bone_hopper_jump (sfxstatic_assets_areawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → High Jump Air
  }

  state High Jump Air {
    SetVelocityByScale(gameObject=Self, speed=var "X Speed", ySpeed=(unset), everyFrame=true)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Wait For Hero {
    BoolTest(boolVariable=var "No Cam Wake", isTrue=→"NO CAM WAKE", isFalse=(none), everyFrame=false)
    CheckOutOfCamera(gameObject=Self, margin=6, outsideEvent=(none), insideEvent=→"WAKE", insideBool=false, outsideBool=false, everyFrame=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle B")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on WAKE → Jump Antic
    on TOOK DAMAGE → Jump Antic
    on NO CAM WAKE → Wait For Hero 2
  }

  state Follow Leader? {
    GameObjectIsNull(gameObject=var "Leader", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetXDistance(gameObject=Self, target=var "Leader", storeResult=var "Distance From Leader", everyFrame=false)
    FloatCompare(float1=var "Distance From Leader", float2=6, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Leader", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Leader Is R", leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignToBool(Value=var "X Scale", StoreIsPositive=var "Facing L", EveryFrame=false)
    BoolTestMulti(boolVariables=[var "Leader Is R", var "Facing L"], boolStates=[true, true], trueEvent=→"TURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Leader Is R", var "Facing L"], boolStates=[false, false], trueEvent=→"TURN", falseEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Next
    on TURN → Turn
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"FALL", EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=<null>, noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
    on FALL → Fall
    on SING DURATION END → Sing End
    on COAL → Jump Launch
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStop(gameObject=var "Audio Loop Voice", fadeTime=0.15)
    on FINISHED → Next
  }

  state Fall {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Jump")
    on LAND → Land
  }

  state Jump Voice? {
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.75, 0.25], eventMax=[5, 1], missedMax=[1, 5], activeBool=(unset))
    AudioPlayRandomVoice(gameObject=var "Audio Loop Voice", audioClips=[bone_hopper_simple_alert_1 (sfxstatic_assets_areawilds.bundle), bone_hopper_simple_alert_2 (sfxstatic_assets_areawilds.bundle), bone_hopper_simple_alert_3 (sfxstatic_assets_areawilds.bundle)], weights=[1, 1, 1], pitchMin=0.85, pitchMax=1.15, stopPreviousSound=true)
    on FINISHED → Jump Launch
  }

  state Wait For Hero 2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle B")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on WAKE → Jump Antic
    on TOOK DAMAGE → Jump Antic
  }

  var Wait For Hero: bool = true
  var Leader: gameObject = <null>

  var Jump Y: float = 0
  var X Speed: float = -7.5
  var Idle Time: float = 0
  var Turn Time: float = 0
  var X Scale: float = 0
  var Distance From Leader: float = 0
  var Drop Distance: float = 3
  var Wall Ahead: bool = false
  var High Wall Ahead: bool = false
  var Force Turn L: bool = false
  var Force Turn R: bool = false
  var Facing L: bool = false
  var Leader Is R: bool = false
  var Startled: bool = false
  var No Cam Wake: bool = false
  var Audio Loop Voice: gameObject = Bone Hopper Simple (1)/Audio Loop Voice
  var Pt Jump: gameObject = Bone Hopper Simple (1)/Pt Jump
  var Pt Land: gameObject = Bone Hopper Simple (1)/Pt Land
  var Ray Pt: gameObject = Bone Hopper Simple (1)/Ray Pt
  var Ray Pt Ahead: gameObject = Bone Hopper Simple (1)/Ray Pt Ahead
}
