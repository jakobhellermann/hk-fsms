# Control
# 1 location(s):
#   Cradle_03 / Boss Scene/Strand Patterns/Pattern 2/Silk Boss WebStrand (8)

fsm Control {
  start Init
  on ATTACK CLEAR → Inactive  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=var "web_strand_single", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on APPEAR → Appear Pause
  }

  state Appear Pause {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Appear
    on CANCEL → Inactive
  }

  state Appear {
    ActivateGameObject(gameObject=var "web_strand_single", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=var "web_strand_single", stateName="Silk_Beast_Web", layer=(unset), normalizedTime=(unset), everyFrame=false)
    on BURST → Burst Start
    on CANCEL → Inactive
  }

  state Burst Start {
    AnimatorPlay(gameObject=var "web_strand_single", stateName="Silk_Beast_Web_Burst", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=0.65, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(gameObject=var "hero_catcher", trigger=0, collideTag="", sendEvent=→"CATCH HERO", storeCollider=var "")
    Trigger2dEvent(gameObject=var "hero_catcher", trigger=1, collideTag="", sendEvent=→"CATCH HERO", storeCollider=var "")
    on FINISHED → Check Roof?
    on CATCH HERO → Catch Hero
    on CANCEL → Inactive
  }

  state Catch Hero {
    CanHeroTakeDamage(eventTarget=Self, canTakeDmgEvent=(none), cannotTakeDmgEvent=→"CANCEL")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="STRAND CATCH", delay=0, everyFrame=false)
    SendEventToRegister(eventName="HORNET CAUGHT")
    on CANCEL → Catch Cancel
    on FINISHED → Catch Cancel
  }

  state Catch Cancel {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Inactive
  }

  state Burst {
    Trigger2dEvent(gameObject=var "hero_catcher", trigger=0, collideTag="", sendEvent=→"CATCH HERO", storeCollider=var "")
    Trigger2dEvent(gameObject=var "hero_catcher", trigger=1, collideTag="", sendEvent=→"CATCH HERO", storeCollider=var "")
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Inactive
    on CATCH HERO → Catch Hero
    on CANCEL → Inactive
  }

  state Check Roof? {
    BoolTest(boolVariable=var "Check Roof", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    CheckYPosition(gameObject=var "Hero", compareTo=147.2, compareToOffset=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CATCH HERO", everyFrame=false, space=0, activeBool=(unset))
    on FINISHED → Burst
    on CATCH HERO → Catch Hero
  }

  var Check Roof: bool = false

  var web_strand_single: gameObject = Boss Scene/Strand Patterns/Pattern 2/Silk Boss WebStrand (8)/web_strand_single
  var hero_catcher: gameObject = Boss Scene/Strand Patterns/Pattern 2/Silk Boss WebStrand (8)/web_strand_single/hair/hero_catcher
  var Parent: gameObject = <null>
}
