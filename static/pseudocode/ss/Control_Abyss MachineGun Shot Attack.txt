# Control
# 1 location(s):
#   blackthread_assets_all.bundle / Abyss MachineGun Shot Attack

fsm Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Audio ShotExtra", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spit Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Check Hero Dir
  }

  state Antic Follow {
    PlayAudioEvent(audioClip=radiance_burst_white_stuff_move_up (sfxstatic_assets_blackthread.bundle), pitchMin=1, pitchMax=1, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Audio Player Start")
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Antic Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Fire Start
  }

  state Fire Start {
    StopParticleEmitter(gameObject=var "Pt Antic")
    ActivateGameObject(gameObject=var "Antic Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIntValue(intVariable=var "Shots", intValue=5, everyFrame=false)
    on FINISHED → Fire
  }

  state Fire {
    FadeAudio(gameObject=var "Audio Player Start", startVolume=(unset), endVolume=0, time=0.2)
    ActivateGameObject(gameObject=var "Spit Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Audio ShotExtra", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Audio ShotExtra", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Abyss Bullet (localpoolprefabs_assets_areaabyss.bundle), spawnPoint=var "Shot Point", position=(unset), rotation=(unset), storeObject=var "Projectile")
    SetRotation(gameObject=var "Projectile", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    SetVelocityAsAngle(gameObject=var "Projectile", angle=var "Angle", speed=var "Shot Speed", everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=mini_mawlek_spit (sfxstatic_assets_areasong.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FloatAdd(floatVariable=var "Angle", add=var "Angle Add", everyFrame=false, perSecond=false)
    on FINISHED → Repeat?
  }

  state Repeat? {
    IntAdd(intVariable=var "Shots", add=-1, everyFrame=false)
    IntCompare(integer1=var "Shots", integer2=0, equal=→"END", lessThan=→"END", greaterThan=→"REPEAT", everyFrame=false)
    on REPEAT → Fire
    on END → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state test {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic Follow
  }

  state Check Hero Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Aim L
    on R → Aim R
  }

  state Aim L {
    SetFloatValue(floatVariable=var "Angle", floatValue=180, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Add", floatValue=-5, everyFrame=false)
    on FINISHED → Antic Follow
  }

  state Aim R {
    SetFloatValue(floatVariable=var "Angle", floatValue=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Angle Add", floatValue=5, everyFrame=false)
    on FINISHED → Antic Follow
  }

  var Angle: float = 0
  var Shot Speed: float = 30
  var Angle Add: float = 0
  var Shots: int = 0
  var Antic Sprite: gameObject = Abyss MachineGun Shot Attack/Antic Sprite
  var Audio Player Start: gameObject = <null>
  var Audio ShotExtra: gameObject = Abyss MachineGun Shot Attack/Audio ShotExtra
  var Projectile: gameObject = <null>
  var Pt Antic: gameObject = Abyss MachineGun Shot Attack/Pt Antic
  var Shot Point: gameObject = Abyss MachineGun Shot Attack/Shot Point
  var Spit Effect: gameObject = Abyss MachineGun Shot Attack/Spit Effect
  var Audio Loop Void: gameObject = Abyss MachineGun Shot Attack/Audio Loop Void
}
