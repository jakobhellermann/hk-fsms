# Control
# 1 location(s):
#   Cog_05 / Battle Scene/Wave 2/Song Automaton 02

fsm Control {
  start Init
  on SPAWN → Spawn  // from any state
  on FRIEND BONK → Friend Bonk  // from any state
  on ZERO HP → Death  // from any state
  on COG DAMAGE → Cog Damage  // from any state

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    RandomlyFlipScale(gameObject=Self)
    on FINISHED → Dormant
  }

  state Walk {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK START", delay=0, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    CheckOutOfCamera(gameObject=Self, margin=12, outsideEvent=→"SLEEP", insideEvent=(none), insideBool=false, outsideBool=false, everyFrame=true)
    on ATTACK → Attack Antic
    on TOOK DAMAGE → Attack Antic
    on SING → Sing
    on SLEEP → Dormant
  }

  state Attack Antic {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK STOP", delay=0, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=song_automaton_fly_antic (sfxstatic_assets_areacog.bundle), pitchMin=0.9, pitchMax=0.9, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Automaton_mid_small_move (sfxdynamic_assets_areacog.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Steam Antic
  }

  state Steam Antic {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=song_automaton_02_antic (sfxstatic_assets_areacog.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Steam")
    on FINISHED → Steam Attack
  }

  state Steam Attack {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=song_automaton_02_steam_release (sfxstatic_assets_areacog.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    PlayParticleEmitterInState(gameObject=var "Pt Steam")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObjectDelay(gameObject=var "Steam Hit", activate=true, resetOnExit=true, delay=0.3)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Automaton_Steamer_attack (sfxdynamic_assets_areacog.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Recover
  }

  state Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Walk Away
  }

  state Walk Away {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK START", delay=0, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    on ATTACK → Attack Antic
    on TOOK DAMAGE → Attack Antic
    on SING → Sing
  }

  state Sing {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK STOP", delay=0, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pray", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Automaton_mid_sing (sfxdynamic_assets_areacog.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pray End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Walk Away
  }

  state Bonk {
    SetAudioClip(gameObject=Self, audioClip=<null>, autoPlay=false, stopOnExit=false)
    SetBoolValue(boolVariable=var "Fell", boolValue=false, everyFrame=false)
    SetVelocityByScale(gameObject=Self, speed=-5, ySpeed=10, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    NextFrameEvent(sendEvent=→"FINISHED")
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Tiny Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateAllChildren(gameObject=var "Wall Bump", activate=true)
    on FINISHED → Bonk Air
  }

  state Bonk Air {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Automaton_mid_stun_hit (sfxdynamic_assets_areacog.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on LAND → Land
  }

  state Land {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Recover")
    Wait(time=0.33, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Turn
  }

  state Turn {
    BoolTest(boolVariable=var "Fell", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Walk
  }

  state Friend Bonk {
    GetEventSender(sentByGameObject=var "Bonk Sender")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK STOP", delay=0, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Bonk Sender", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    on FINISHED → Bonk
  }

  state Spawn {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK STOP", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spawn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Walk
  }

  state Death {
    SetParent(gameObject=var "Pt Antic", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Pt Steam", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
  }

  state Cog Damage {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    SetBoolValue(boolVariable=var "Fell", boolValue=true, everyFrame=false)
    on FINISHED → Bonk Air
  }

  state Dormant {
    CheckOutOfCamera(gameObject=Self, margin=5, outsideEvent=(none), insideEvent=→"WAKE", insideBool=false, outsideBool=false, everyFrame=true)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK STOP", delay=0, everyFrame=false)
    on WAKE → Walk
  }

  var Fell: bool = false
  var Pt Antic: gameObject = Battle Scene/Wave 2/Song Automaton 02/Pt Antic
  var Pt Steam: gameObject = Battle Scene/Wave 2/Song Automaton 02/Pt Steam
  var Steam Hit: gameObject = Battle Scene/Wave 2/Song Automaton 02/Steam Hit
  var Bonk Sender: gameObject = <null>
  var Wall Bump: gameObject = Song Automaton 01/Wall Bump (localpoolprefabs_assets_areacog.bundle)
  var Audio Loop Voice: gameObject = Battle Scene/Wave 2/Song Automaton 02/Audio Loop Voice
}
