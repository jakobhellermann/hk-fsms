# Control
# 1 location(s):
#   Cog_05 / Song Automaton Fly Spike BG (1)

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Deparent
  }

  state To Patrol Antic {
    FireAtTarget(gameObject=Self, target=var "Destination", speed=-6, position=(0, 0, 0), spread=0, everyFrame=false)
    Decelerate(gameObject=Self, deceleration=0.9)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → To Patrol V
  }

  state To Patrol V {
    FireAtTarget(gameObject=Self, target=var "Destination", speed=8, position=(0, 0, 0), spread=0, everyFrame=true)
    GetDistance(gameObject=Self, target=var "Destination", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Velocity", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"H", everyFrame=true)
    FloatCompare(float1=var "X Velocity", float2=-1, tolerance=0, equal=(none), lessThan=→"H", greaterThan=(none), everyFrame=true)
    on FINISHED → Patrol Antic
    on H → To Patrol H
  }

  state Patrol Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=true)
    on FINISHED → Spikes Out 1
  }

  state Spikes Out 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spikes Out", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Spikes Out 2
  }

  state Pick Destination {
    BoolTest(boolVariable=var "Returning", isTrue=→"START", isFalse=→"PATROL", everyFrame=false)
    on START → Start
    on PATROL → Patrol
  }

  state Start {
    SetGameObject(variable=var "Destination", gameObject=var "Start Point", everyFrame=false)
    on FINISHED → To Patrol Antic
  }

  state Patrol {
    SetGameObject(variable=var "Destination", gameObject=var "Patrol Point", everyFrame=false)
    on FINISHED → To Patrol Antic
  }

  state Shift {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    BoolFlip(boolVariable=var "Returning")
    on FINISHED → Pick Destination
  }

  state Spikes Out 2 {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0, resetOnStateExit=true)
    on FINISHED → Spikes In
  }

  state Spikes In {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spikes In", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Shift
  }

  state Deparent {
    SetParent(gameObject=var "Patrol Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Start Point", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Pick Destination
  }

  state To Patrol H {
    FireAtTarget(gameObject=Self, target=var "Destination", speed=8, position=(0, 0, 0), spread=0, everyFrame=true)
    GetDistance(gameObject=Self, target=var "Destination", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=1, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0)
    GetVelocity2d(gameObject=Self, vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Velocity", float2=0, tolerance=1, equal=→"V", lessThan=(none), greaterThan=(none), everyFrame=true)
    on FINISHED → Patrol Antic
    on V → To Patrol V
  }

  var Distance: float = 0
  var X Velocity: float = 0
  var Returning: bool = false
  var Start Point: gameObject = Song Automaton Fly Spike BG (1)/Start Point
  var Patrol Point: gameObject = Song Automaton Fly Spike BG (1)/Patrol Point
  var Destination: gameObject = <null>
}
