fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Walk")
    on FINISHED → Out
  }

  state Out {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    PlayParticleEmitterInState(gameObject=var "Pt Walk")
    on EXIT → Out
  }

  var Pt Walk: gameObject = <null>
}
