# FSM
# 1 location(s):
#   Clover_18 / Fountains Group

fsm FSM {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Plaque Up", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Plaque Down", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Plates Init
  }

  state Inert {
    EnableFsmSelf(SetEnabled=false)
    on SHRINE PLATE DROPPED → Check Plates Init
  }

  state Check Plates Init {
    GetPersistentBool(target=var "Plate L", storeValue=var "Plate L Complete")
    GetPersistentBool(target=var "Plate TR", storeValue=var "Plate TR Complete")
    GetPersistentBool(target=var "Plate BR", storeValue=var "Plate BR Complete")
    BoolAllTrue(boolVariables=[var "Plate BR Complete", var "Plate L Complete", var "Plate TR Complete"], sendEvent=(none), storeResult=var "Complete", everyFrame=false)
    BoolTest(boolVariable=var "Complete", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Inert
    on TRUE → Complete Fountains
  }

  state Complete Fountains {
    ActivateGameObject(gameObject=var "Plaque Up", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Plaque Down", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    EnableFsmSelf(SetEnabled=false)
  }

  var Plate BR Complete: bool = false
  var Plate L Complete: bool = false
  var Plate TR Complete: bool = false
  var Complete: bool = false
  var Plate L: gameObject = Fountains Group/clover_lake_shrine_small_room/Shrine Pressure Plate L
  var Plate TR: gameObject = Fountains Group/clover_lake_shrine_small_room (1)/Shrine Pressure Plate TR
  var Plate BR: gameObject = Fountains Group/clover_lake_shrine_small_room (2)/Shrine Pressure Plate BR
  var Plaque Up: gameObject = Fountains Group/Plaque/Up
  var Plaque Down: gameObject = Fountains Group/Plaque/Down
  var Needolin Plinth Inspect: gameObject = Fountains Group/Shrine Group/Needolin Plinth/Plinth Inspect
}
