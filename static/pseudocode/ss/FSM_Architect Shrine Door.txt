# FSM
# 1 location(s):
#   Under_17 / Architect Shrine Door

fsm FSM {
  start Inert

  state Inert {
    on UNLOCK → Unlock
    on UNLOCKED → Entry from this gate?
  }

  state Unlocked {
    SetAnimator(target=Self, active=true)
    AnimatorPlay(gameObject=Self, stateName="Unlock", layer=0, normalizedTime=1, everyFrame=false)
    on FINISH ENTRY → Got Crest?
  }

  state Unlock {
    EnableFsmSelf(SetEnabled=false)
  }

  state Lock {
    SetDeathRespawnMarker(Target=Architect Shrine Door/door1/Death Respawn Marker Architect Door)
    ActivateInteractible(Target=Architect Shrine Door/door1, Activate=false, AllowQueueing=false, UseChildren=false)
    AnimatorPlay(gameObject=Self, stateName="Lock", layer=0, normalizedTime=0, everyFrame=false)
    EnableFsmSelf(SetEnabled=false)
    SaveGame()
  }

  state Got Crest? {
    GetIsCrestUnlocked(Crest=Toolmaster (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"TRUE", falseEvent=→"FALSE", storeValue=(unset))
    on FALSE → Stay
    on TRUE → Lock
  }

  state Stay {
    EnableFsmSelf(SetEnabled=false)
  }

  state Entry from this gate? {
    GetEntryGateName(StoreValue=var "Entry Gate")
    GetName(gameObject=Architect Shrine Door/door1, storeName=var "Door Name", everyFrame=false)
    StringCompare(stringVariable=var "Entry Gate", compareTo=var "Door Name", equalEvent=→"TRUE", notEqualEvent=→"FALSE", storeResult=(unset), everyFrame=false)
    on TRUE → Unlocked
    on FALSE → Got Crest? 2
  }

  state Got Crest? 2 {
    GetIsCrestUnlocked(Crest=Toolmaster (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"TRUE", falseEvent=→"FALSE", storeValue=(unset))
    on FALSE → Unlocked
    on TRUE → Locked
  }

  state Locked {
    SetAnimator(target=Self, active=true)
    AnimatorPlay(gameObject=Self, stateName="Lock", layer=0, normalizedTime=1, everyFrame=false)
    ActivateInteractible(Target=Architect Shrine Door/door1, Activate=false, AllowQueueing=false, UseChildren=false)
    EnableFsmSelf(SetEnabled=false)
    SetDeathRespawnMarker(Target=<null>)
  }

  var Entry Gate: string = ""
  var Door Name: string = ""
}
