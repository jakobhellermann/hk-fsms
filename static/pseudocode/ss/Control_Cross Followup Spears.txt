# Control
# 2 location(s):
#   Memory_Coral_Tower / Boss Scene/Cross Followup Spears
#   Memory_Coral_Tower / Boss Scene/Cross Spears

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Coral King")
    FindNamedChild(gameObject=Self, storeResult=var "Coral_spear_long")
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    on SPEAR → Pause
  }

  state Attack {
    SetPositionToObject(gameObject=Self, targetObject=var "Coral King", xOffset=0, yOffset=0, zOffset=-0.006, overrideZ=(unset), everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateAllChildren(gameObject=var "Self", activate=true)
    on FINISHED → Idle
  }

  state Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Attack
  }

  var X Scale: float = 0
  var Boss Scene: gameObject = <null>
  var Coral King: gameObject = <null>
  var Coral_spear_long: gameObject = <null>
  var Self: gameObject = <null>
}
