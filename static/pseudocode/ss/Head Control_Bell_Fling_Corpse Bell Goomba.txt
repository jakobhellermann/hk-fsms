# Head Control
# 2 location(s):
#   enemycorpses_assets_areabell.bundle / Corpse Bell Goomba/Fling/Bell
#   enemycorpses_assets_areabell.bundle / Corpse Bell Goomba Shiny Variant/Fling/Bell

fsm Head Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"TAKE DAMAGE", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=var "Damage Dealt", storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on TAKE DAMAGE → Hit Effects
    on SPIKES → Hazard
  }

  state Check Direction {
    GetDamagerDirection(target=var "Damager", storeDirection=var "Attack Direction")
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 361], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on LEFT → Left
    on RIGHT → Right
    on UP → Up
    on DOWN → Down
  }

  state Left {
    RandomFloat(min=130, max=140, storeResult=var "Angle")
    RandomFloat(min=17, max=22, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Right {
    RandomFloat(min=40, max=50, storeResult=var "Angle")
    RandomFloat(min=17, max=22, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Spin {
    GetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=10, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Speed", everyFrame=false)
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Up {
    RandomFloat(min=85, max=95, storeResult=var "Angle")
    RandomFloat(min=23, max=27, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Down {
    RandomFloat(min=60, max=120, storeResult=var "Angle")
    RandomFloat(min=10, max=15, storeResult=var "Speed")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    on FINISHED → Spin
  }

  state Hit Effects {
    IntCompare(integer1=var "Damage Dealt", integer2=0, equal=→"CANCEL", lessThan=→"CANCEL", greaterThan=(none), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=bell_goomba_bell_long_ver (sfxstatic_assets_areabell.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Check Direction
    on CANCEL → Idle
  }

  state Hazard {
    CallMethodProper(gameObject=Self, behaviour="BreakOnHazard", methodName="Break", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  var Angle: float = 0
  var Attack Direction: float = 0
  var Speed: float = 0
  var Damage Dealt: int = 0
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
