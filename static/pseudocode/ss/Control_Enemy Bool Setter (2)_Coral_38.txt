# Control
# 4 location(s):
#   Coral_38 / Enemy Bool Setter (2)
#   Coral_38 / Enemy Bool Setter (3)
#   Coral_38 / Enemy Bool Setter (1)
#   Coral_38 / Enemy Bool Setter

fsm Control {
  start Detect

  state Detect {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "Enemy")
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "Enemy")
    on ENTER → Enter
    on EXIT → Exit
  }

  state Enter {
    SetFsmBool(gameObject=var "Enemy", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Do Parent", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    GetParent(gameObject=var "Enemy", storeResult=var "Enemy Parent")
    SetFsmBool(gameObject=var "Enemy Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=true, everyFrame=false)
    on FINISHED → Detect
  }

  state Exit {
    SetFsmBool(gameObject=var "Enemy", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=false, everyFrame=false)
    BoolTest(boolVariable=var "Do Parent", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    GetParent(gameObject=var "Enemy", storeResult=var "Enemy Parent")
    SetFsmBool(gameObject=var "Enemy Parent", fsmName=var "FSM Name", variableName=var "Bool Name", setValue=false, everyFrame=false)
    on FINISHED → Detect
  }

  var Do Parent: bool = true
  var Bool Name: string = "No Dig"
  var FSM Name: string = "Control"

  var Enemy: gameObject = <null>
  var Enemy Parent: gameObject = <null>
}
