# Proxy
# 1 location(s):
#   Dock_03 / Harpoon Ring Pull Switch/Harpoon Ring Chain

fsm Proxy {
  start State 1
  on CONNECT → CONNECT  // from any state
  on DISCONNECT → DISCONNECT  // from any state

  state State 1 {
  }

  state CONNECT {
    SendEventByNameV2(eventTarget=GameObject(var "Event Target"), sendEvent="CONNECT", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Is Connected", boolValue=true, everyFrame=false)
  }

  state DISCONNECT {
    SendEventByNameV2(eventTarget=GameObject(var "Event Target"), sendEvent="DISCONNECT", delay=0, everyFrame=false)
    SetBoolValue(boolVariable=var "Is Connected", boolValue=false, everyFrame=false)
  }

  var Event Target: gameObject = Harpoon Ring Pull Switch

  var Is Connected: bool = false
}
