# Control
# 1 location(s):
#   Tut_01 / Tutorial Intro Cutscene/Attack Reminder

fsm Control {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Pause
  }

  state Reminder {
    SendEventToRegister(eventName="REMINDER ATTACK")
  }

  state Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"PERFORMED", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    on FINISHED → Reminder
    on PERFORMED → Inactive
  }

  state Inactive {
  }

  var Control Reminder: gameObject = <null>
}
