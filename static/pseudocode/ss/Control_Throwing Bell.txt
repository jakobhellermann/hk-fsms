# Control
# 1 location(s):
#   localpoolprefabs_assets_areasong.bundle / Throwing Bell

fsm Control {
  start Init
  on PROJECTILE BREAK → Break  // from any state
  on DEALT DAMAGE → Dealt Damage  // from any state

  state Init {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetCircleCollider(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    SetIntValue(intVariable=var "Bounces", intValue=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Spark", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetFloatValue(floatVariable=var "Lifetime", floatValue=6, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Air
  }

  state Recycle {
    SetDamageHeroAmount(target=Self, damageDealt=1)
    RecycleSelf()
  }

  state Air {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    FloatAdd(floatVariable=var "Lifetime", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Lifetime", float2=0, tolerance=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"NAIL", sendEventHeavy=→"BREAK", sendEventSpikes=→"BREAK", sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Water Surface", collideLayer=(unset), sendEvent=→"WATER", storeCollider=var "")
    on WALL L → Wall L
    on WALL R → Wall R
    on GROUND → Ground
    on ROOF → Roof
    on NAIL → Nail Hit
    on BREAK → Break
    on WATER → Rotate Check
    on TINK UP → Tink U
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK DOWN → Tink D
  }

  state Ground {
    IntCompare(integer1=var "Bounces", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=18.8, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Bounce
    on BREAK → Break
  }

  state Break {
    ActivateGameObject(gameObject=var "Spark", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=small_bell_projectile_break (sfxstatic_assets_shared.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Deactivate
  }

  state Bounce {
    PlayAudioEventRandom(audioClips=[small_bell_projectile_bounce (sfxstatic_assets_shared.bundle), small_bell_projectile_bounce_2 (sfxstatic_assets_shared.bundle), small_bell_projectile_bounce_3 (sfxstatic_assets_shared.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    IntAdd(intVariable=var "Bounces", add=-1, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Air
    on BREAK → Break
  }

  state Wall L {
    IntCompare(integer1=var "Bounces", integer2=-1, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=10, y=(unset), everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Bounce
    on BREAK → Break
  }

  state Wall R {
    IntCompare(integer1=var "Bounces", integer2=-1, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-10, y=(unset), everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=-0.1, y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Bounce
    on BREAK → Break
  }

  state Roof {
    IntCompare(integer1=var "Bounces", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-0.1, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Bounce
    on BREAK → Break
  }

  state Nail Hit {
    FloatAddRandom(floatVariable=var "Direction", addMin=-5, addMax=5)
    SetVelocityAsAngle(gameObject=Self, angle=var "Direction", speed=30, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Air
    on BREAK → Break
  }

  state Water {
    PlayAudioEventRandom(audioClips=[splash_small_object_1 (sfxstatic_assets_water.bundle), splash_small_object_2 (sfxstatic_assets_water.bundle), splash_small_object_3 (sfxstatic_assets_water.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    StopParticleEmitter(gameObject=var "Pt Dust")
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    DecelerateXY(gameObject=Self, decelerationX=0.82, decelerationY=0.6, brakeOnExit=false)
    SetCircleCollider(gameObject=Self, active=false)
    FlingObjectsFromGlobalPool(gameObject=Spatter White R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0.5, 0), spawnMin=4, spawnMax=5, speedMin=15, speedMax=20, angleMin=40, angleMax=140, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    Tk2dStopAnimation(gameObject=Self)
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Deactivate
  }

  state Rotate Check {
    SetFloatValue(floatVariable=var "Rotation", floatValue=360, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"FINISHED", negativeEvent=(none), space=0)
    SetFloatValue(floatVariable=var "Rotation", floatValue=-360, everyFrame=false)
    on FINISHED → Water
  }

  state Dealt Damage {
    on FINISHED → Break
  }

  state Deactivate {
    StopParticleEmitter(gameObject=var "Pt Dust")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Tink U {
    SetFloatValue(floatVariable=var "Direction", floatValue=90, everyFrame=false)
    on FINISHED → Nail Hit
  }

  state Tink L {
    SetFloatValue(floatVariable=var "Direction", floatValue=180, everyFrame=false)
    on FINISHED → Nail Hit
  }

  state Tink R {
    SetFloatValue(floatVariable=var "Direction", floatValue=0, everyFrame=false)
    on FINISHED → Nail Hit
  }

  state Tink D {
    SetFloatValue(floatVariable=var "Direction", floatValue=270, everyFrame=false)
    on FINISHED → Nail Hit
  }

  var Direction: float = 0
  var Lifetime: float = 0
  var Rotation: float = 0
  var Bounces: int = 1
  var Nail Object: gameObject = <null>
  var Pt Dust: gameObject = Throwing Bell/Pt Dust
  var Pt Shatter: gameObject = Throwing Bell/Pt Shatter
  var Self: gameObject = Throwing Bell
  var Spark: gameObject = Throwing Bell/Spark
  var Enemy Damager: gameObject = Throwing Bell/Enemy Damager
}
