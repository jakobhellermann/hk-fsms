# FSM
# 1 location(s):
#   localpoolprefabs_assets_shared.bundle / Revenge Crystal Hornet

fsm FSM {
  start Init Wait
  on CLEAR EFFECTS → Recycle  // from any state

  state Init Wait {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Appear Antic 1
  }

  state Appear Begin {
    CallMethodProper(gameObject=var "Form Antic Pt", behaviour="PlayParticleEffects", methodName="StopParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Appear Begin", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=crystal_reform_end (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Appear
  }

  state Appear {
    CallMethodProper(gameObject=var "Form Pt", behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    RandomInt(min=1, max=4, storeResult=var "Num", inclusiveMax=true, noRepeat=false)
    FormatString(format="Appear {0}", variables=[var "Num"], storeResult=var "Anim", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → Idle
  }

  state Appear Antic 1 {
    AddSpawnedToQueueGroup(Group=Revenge Crystal Hornet (dataassets_assets_assets/dataassets/spawnqueuegroups.bundle), Target=Self)
    Translate(gameObject=Self, vector=(unset), x=0.7, y=0.7, z=(unset), space=1, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)  // disabled
    MatchScaleSign(Target=Self, MatchTo=var "Hero", EveryFrame=false)
    RandomFloat(min=-30, max=30, storeResult=var "Rotation")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=1, everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Form Antic Pt", behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetPositionToObject2D(gameObject=Self, targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Appear Antic 2
  }

  state Recycle {
    RecycleSelf()
  }

  state Crystal Burst {
    SetMeshRenderer(gameObject=Self, active=false)
    CallMethodProper(gameObject=var "Form Pt", behaviour="PlayParticleEffects", methodName="ClearParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Form Pt", behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    SpawnRandomObjectsRadialV2(Prefab=Revenge Crystal Spikes, SpawnPoint=Self, SpawnAsChildren=false, MinCount=8, MaxCount=8, MinAngle=0, MaxAngle=360, AngleErrorMax=2, SpawnRadius=4)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=crystal_break (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Spike Break
  }

  state Spike Break {
    PlayAudioEvent(audioClip=crystal_drifter_damage_2 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Recycle
  }

  state Idle {
    SetCollider(gameObject=Self, active=true, resetOnExit=true)
    CallMethodProper(gameObject=var "Idle Pt", behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=(unset), InRangeEvent=→"ATTACK", InRangeDelay=0.1, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    Wait(time=8, finishEvent=→"FINISHED", realTime=false)
    on ATTACK → Burst Antic
    on NAIL HIT → Take Damage
    on SPAWN DEQUEUED → Burst Antic
    on FINISHED → Auto Break Wait
  }

  state Burst Antic {
    CallMethodProper(gameObject=var "Idle Pt", behaviour="PlayParticleEffects", methodName="StopParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    FormatString(format="Break {0}", variables=[var "Num"], storeResult=var "Anim", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Crystal Burst
  }

  state Auto Break Wait {
    ObjectJitterLocal(gameObject=Self, x=0.05, y=0.05, z=(unset), resetPosOnStateExit=true, limitFps=30)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=(unset), InRangeEvent=→"ATTACK", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on ATTACK → Burst Antic
    on NAIL HIT → Burst Antic
    on SPAWN DEQUEUED → Burst Antic
    on FINISHED → Burst Antic
  }

  state Appear Antic 2 {
    Wait(time=0.9, finishEvent=(none), realTime=false)
    on FINISHED → Appear Begin
  }

  state Take Damage {
    PlayAudioEventRandom(audioClips=[crystal_break_new_1 (sfxstatic_assets_shared.bundle), crystal_break_new_2 (sfxstatic_assets_shared.bundle), crystal_break_new_3 (sfxstatic_assets_shared.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Burst Antic
  }

  var Rotation: float = 0
  var Num: int = 0
  var Active Count: int = 0
  var Anim: string = ""
  var Form Antic Pt: gameObject = Revenge Crystal Hornet/Pt ReformAntic
  var Form Pt: gameObject = Revenge Crystal Hornet/Pt Form
  var Idle Pt: gameObject = Revenge Crystal Hornet/Pt Idle
  var Enemy Damager: gameObject = Revenge Crystal Hornet/Enemy Damager
  var Alert Range: object = Revenge Crystal Hornet/Attack Range@AlertRange
}
