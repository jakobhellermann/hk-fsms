# Control
# 1 location(s):
#   Dock_11 / Tar Slug Huge (7)

fsm Control {
  start Init

  state Crawling {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.15, MaxReactDelay=0.3, None=(none), ActiveInner=→"PRAY", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="", everyFrame=true)
    AudioPlayInState(gameObject=Self, volume=1)
    on PRAY → Stop Climber
    on ATTACK → Attack Antic
  }

  state Praying {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Pray", animationTriggerEvent=(none), animationCompleteEvent=(none))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=<null>, noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Start Crawler
    on SING DURATION END → Start Crawler
  }

  state Stop Climber {
    StopCrawler(Target=Self, WaitForTurn=true)
    on FINISHED → Praying
  }

  state Attack Antic {
    StopCrawler(Target=Self, WaitForTurn=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Antic")
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=coral_final_judge_ring_of_flame_appear (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=0.7, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=hatcher_give_birth (sfxstatic_assets_areadocks.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Attack
  }

  state Attack {
    PlayAudioEvent(audioClip=enemy_death_in_lava (sfxstatic_assets_shared.bundle), pitchMin=0.8, pitchMax=0.9, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CheckIsBlackThreaded(Target=Self, TrueEvent=(none), FalseEvent=(none), StoreValue=var "Is Black Threaded", EveryFrame=false)
    BoolTestToGameObject(Test=var "Is Black Threaded", ExpectedValue=true, TrueGameObject=var "Attack Fx Void", FalseGameObject=var "Attack Fx", StoreResult=var "Attack Fx Crt", EveryFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Attack Fx Crt", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObjectDelay(gameObject=var "Attack Collider Circle", activate=true, resetOnExit=true, delay=0.1)  // disabled
    SetCollider(gameObject=var "Attack Collider Column", active=true, resetOnExit=true)  // disabled
    SetCollider(gameObject=var "Corpse Collider Column", active=true, resetOnExit=true)
    SetCircleCollider(gameObject=var "Corpse Collider Circle", active=true)
    ActivateGameObject(gameObject=var "Attack Collider Column", activate=true, recursive=false, resetOnExit=true, everyFrame=false)  // disabled
    EaseFloat(fromValue=1, toValue=12, floatVariable=var "Column Scale", time=0.37, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetScale(gameObject=var "Attack Collider Column", vector=(unset), x=(unset), y=var "Column Scale", z=(unset), everyFrame=true, lateUpdate=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    AudioPlayInState(gameObject=var "Audio Loop AttackSpurt", volume=1)
    on FINISHED → Attack End
  }

  state Attack End {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=var "Corpse Collider Circle", active=false)
    PlayAudioEvent(audioClip=coral_final_judge_vertical_flame_antic (sfxstatic_assets_areacoral.bundle), pitchMin=0.85, pitchMax=0.85, volume=0.75, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Attack Cooldown 1
  }

  state Attack Cooldown 1 {
    StartCrawler(Target=Self, ScheduleTurn=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    AudioPlayInState(gameObject=Self, volume=1)
    on FINISHED → Attaack Cooldown 2
  }

  state Start Crawler {
    StartCrawler(Target=Self, ScheduleTurn=false)
    on FINISHED → Crawling
  }

  state Init {
    FindChild(gameObject=var "Attack Collider Column", childName="Corpse Collider", storeResult=var "Corpse Collider Column")
    on FINISHED → Start Crawler
  }

  state Attaack Cooldown 2 {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.3, None=(none), ActiveInner=→"PRAY", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Crawling
    on PRAY → Stop Climber
  }

  var Column Scale: float = 0
  var Is Black Threaded: bool = false
  var Attack Collider Circle: gameObject = <null>
  var Attack Collider Column: gameObject = Tar Slug Huge (7)/Attack Collider Column
  var Attack Fx: gameObject = Tar Slug Huge (7)/Attack Fx
  var Attack Fx Crt: gameObject = <null>
  var Attack Fx Void: gameObject = Tar Slug Huge (7)/Attack Fx Void
  var Audio Loop AttackSpurt: gameObject = Tar Slug Huge (7)/Audio Loop AttackSpurt
  var Audio Loop Voice: gameObject = Tar Slug Huge (7)/Audio Loop Voice
  var Corpse Collider Circle: gameObject = Tar Slug Huge (7)/Corpse Collider Circle
  var Corpse Collider Column: gameObject = <null>
  var Pt Antic: gameObject = Tar Slug Huge (7)/Pt Antic
}
