# Control
# 1 location(s):
#   localpoolprefabs_assets_areacloverareamoss.bundle / SilkAcid BurstCloud

fsm Control {
  start Idle

  state Idle {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=var "Acid Collider", active=true)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    RandomFloat(min=0.07, max=0.25, storeResult=var "Gravity")
    RandomFloat(min=0.8, max=1.35, storeResult=var "Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    SetGravity2dScale(gameObject=Self, gravityScale=var "Gravity")
    on FINISHED → End
  }

  state End {
    SetCircleCollider(gameObject=var "Acid Collider", active=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  var Gravity: float = 0
  var Scale: float = 0
  var Acid Collider: gameObject = SilkAcid BurstCloud/Silk Acid Region
}
