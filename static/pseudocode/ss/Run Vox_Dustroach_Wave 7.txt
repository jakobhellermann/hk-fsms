# Run Vox
# 1 location(s):
#   Dust_03 / Battle Control/Battle Scene/Wave 7/Dustroach

fsm Run Vox {
  start Init

  state Init {
    on FINISHED → Quiet
  }

  state Quiet {
    AudioStop(gameObject=var "Audio Loop RunVox", fadeTime=0)
    on RUN VOICE START → Pick Clip
  }

  state Pick Clip {
    SendRandomEvent(events=[→"1", →"2"], weights=[0.5, 0.5], delay=0)
    on 1 → Clip 1
    on 2 → Clip 2
  }

  state Clip 1 {
    SetAudioClip(gameObject=var "Audio Loop RunVox", audioClip=dust_roach_new_shout_variants_long_panting_1 (sfxstatic_assets_areadust.bundle), autoPlay=false, stopOnExit=false)
    SetFloatValue(floatVariable=var "Start Time", floatValue=0, everyFrame=false)
    SendRandomEvent(events=[→"FINISHED", →"FINISHED"], weights=[0.5, 0.5], delay=0)
    SetFloatValue(floatVariable=var "Start Time", floatValue=1.9, everyFrame=false)
    on FINISHED → Start Voice
  }

  state Clip 2 {
    SetAudioClip(gameObject=var "Audio Loop RunVox", audioClip=dust_roach_new_shout_variants_long_panting_2 (sfxstatic_assets_areadust.bundle), autoPlay=false, stopOnExit=false)
    SetFloatValue(floatVariable=var "Start Time", floatValue=0, everyFrame=false)
    SendRandomEvent(events=[→"FINISHED", →"FINISHED"], weights=[0.5, 0.5], delay=0)
    SetFloatValue(floatVariable=var "Start Time", floatValue=1.9, everyFrame=false)
    on FINISHED → Start Voice
  }

  state Start Voice {
    AudioPlaySimpleV2(gameObject=var "Audio Loop RunVox", volume=1, startTime=var "Start Time", oneShotClip=<null>)
    on RUN VOICE STOP → Fade Down
  }

  state Fade Down {
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.25)
    on FINISHED → Quiet
  }

  var Start Time: float = 0
  var Audio Loop RunVox: gameObject = Battle Control/Battle Scene/Wave 7/Dustroach/Audio Loop RunVox
}
