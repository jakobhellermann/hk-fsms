# Do Explosion
# 1 location(s):
#   enemycorpses_assets_areaslab.bundle / Corpse Slab Fly BabyExploder

fsm Do Explosion {
  start Fly Out

  state Antic {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=swamp_bounce_pod_antic_loop (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=0.7, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Looping Audio")
    PlayAudioEvent(audioClip=swamp_bounce_pod_antic_loop (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Looping Audio Voice")
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air 2")  // disabled
    DecelerateV2(gameObject=Self, deceleration=0.925, brakeOnExit=false)
    on FINISHED → Slow Down
  }

  state Explode {
    AudioStop(gameObject=var "Looping Audio", fadeTime=0)
    AudioStop(gameObject=var "Looping Audio Voice", fadeTime=0)
    PlayAudioEvent(audioClip=swamp_bounce_pod_explode (sfxstatic_assets_areaswamp.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Explode", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SpawnObjectFromGlobalPool(gameObject=Fresh_Fly_pop_explosion (localpoolprefabs_assets_areaslab.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Fly Out {
    SetGravity2dScale(gameObject=Self, gravityScale=0.75)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    PlayRandomAudioClipTable(Table=SFS_death (sfxdynamic_assets_areaslab.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Antic
  }

  state Slow Down {
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    DecelerateV2(gameObject=Self, deceleration=0.87, brakeOnExit=false)
    Wait(time=0.45, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air 3")  // disabled
    on FINISHED → Explode
  }

  var Self: gameObject = Corpse Slab Fly BabyExploder
  var Looping Audio: gameObject = <null>
  var Antic Effects: gameObject = Corpse Slab Fly BabyExploder/Antic Effects
  var Looping Audio Voice: gameObject = <null>
}
