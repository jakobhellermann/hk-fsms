# Hit Animate
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/caravan_signpost

fsm Hit Animate {
  start Check Scale
  on BREAK → Break  // from any state

  state Idle {
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=var "Force")  // disabled
    on HIT_UP → Choose Dir
    on HIT_DOWN → Choose Dir
    on HIT_LEFT → Hit L
    on HIT_RIGHT → Hit R
    on TINK UP → Choose Dir
    on TINK DOWN → Choose Dir
    on TINK LEFT → Hit L
    on TINK RIGHT → Hit R
    on COLLIDE → Break Check
  }

  state Hit L {
    AnimatorPlay(gameObject=Self, stateName=var "Anim Hit Left", layer=0, normalizedTime=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=-2, y=(unset), everyFrame=false)
    RayCast2dV2(fromGameObject=var "Ground Ray A", fromPosition=(unset), direction=(0, -1), space=0, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"LAUNCH", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    RayCast2dV2(fromGameObject=var "Ground Ray B", fromPosition=(unset), direction=(0, -1), space=0, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"LAUNCH", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Idle
    on LAUNCH → Launch L
  }

  state Choose Dir {
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"HIT_LEFT", leftEvent=→"HIT_RIGHT", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=true, everyFrame=false)
    on HIT_LEFT → Hit L
    on HIT_RIGHT → Hit R
  }

  state Hit R {
    AnimatorPlay(gameObject=Self, stateName=var "Anim Hit Right", layer=0, normalizedTime=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=2, y=(unset), everyFrame=false)
    RayCast2dV2(fromGameObject=var "Ground Ray A", fromPosition=(unset), direction=(0, -1), space=0, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"LAUNCH", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    RayCast2dV2(fromGameObject=var "Ground Ray B", fromPosition=(unset), direction=(0, -1), space=0, distance=1, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"LAUNCH", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on FINISHED → Idle
    on LAUNCH → Launch R
  }

  state Launch L {
    SetVelocityAsAngle(gameObject=Self, angle=130, speed=17, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=120, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Collider Wheel Right", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Launched
  }

  state Launched {
    BoolTest(boolVariable=var "Belltown", isTrue=→"BELLTOWN", isFalse=(none), everyFrame=false)  // disabled
    CheckYPositionV2(gameObject=Self, compareTo=9, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=var "Near Pit", greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    BoolAllTrue(boolVariables=[var "Belltown", var "Near Pit"], sendEvent=→"BELLTOWN", storeResult=(unset), everyFrame=false)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset))  // disabled
    SetGravity2dScale(gameObject=Self, gravityScale=2)
    on BREAK → Break
    on BELLTOWN → Belltown Pit
  }

  state Break {
    PlayAudioEvent(audioClip=hunter_cage_break (sfxstatic_assets_areaant.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Collider Body", position=(unset), rotation=(unset), storeObject=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "break_chunks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "break_chunks", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Launch R {
    SetVelocityAsAngle(gameObject=Self, angle=50, speed=17, everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=-120, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Collider Wheel Left", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Launched
  }

  state Check Scale {
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignToBool(Value=var "X Scale", StoreIsPositive=var "Scale Positive", EveryFrame=false)
    BoolTest(boolVariable=var "Scale Positive", isTrue=→"POS", isFalse=→"NEG", everyFrame=false)
    on POS → Pos
    on NEG → Neg
  }

  state Pos {
    SetGameObject(variable=var "Collider Wheel Left", gameObject=var "Collider Wheel A", everyFrame=false)
    SetGameObject(variable=var "Collider Wheel Right", gameObject=var "Collider Wheel B", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Hit Left", stringValue="Hit Left", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Hit Right", stringValue="Hit Right", everyFrame=false)
    on FINISHED → Idle
  }

  state Neg {
    SetGameObject(variable=var "Collider Wheel Left", gameObject=var "Collider Wheel B", everyFrame=false)
    SetGameObject(variable=var "Collider Wheel Right", gameObject=var "Collider Wheel A", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Hit Left", stringValue="Hit Right", everyFrame=false)
    SetStringValue(stringVariable=var "Anim Hit Right", stringValue="Hit Left", everyFrame=false)
    on FINISHED → Idle
  }

  state Launch Cancel {
    SetBoolValue(boolVariable=(unset), boolValue=false, everyFrame=false)
  }

  state Break Check {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Belltown Pit {
    SetGravity2dScale(gameObject=Self, gravityScale=5)
    on TINK LEFT → Pit Hit
    on TINK RIGHT → Pit Hit
    on TINK UP → Pit Hit
    on TINK DOWN → Pit Hit
  }

  state Pit Hit {
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=5, everyFrame=false)
    on FINISHED → Belltown Pit
  }

  var Belltown: bool = false

  var X Scale: float = 0
  var Force: float = 0
  var Rotation: float = 0
  var Angle: float = 0
  var Scale Positive: bool = false
  var No Fall Break: bool = false
  var Near Pit: bool = false
  var Anim: string = ""
  var Anim Hit Left: string = ""
  var Anim Hit Right: string = ""
  var Collider Body: gameObject = Caravan_States/Flea Festival/caravan_signpost/Collider Body
  var Collider Wheel A: gameObject = Caravan_States/Flea Festival/caravan_signpost/Collider Wheel A
  var Collider Wheel B: gameObject = Caravan_States/Flea Festival/caravan_signpost/Collider Wheel B
  var Ground Ray A: gameObject = Caravan_States/Flea Festival/caravan_signpost/Ground Ray A
  var Ground Ray B: gameObject = Caravan_States/Flea Festival/caravan_signpost/Ground Ray B
  var break_chunks: gameObject = Caravan_States/Flea Festival/caravan_signpost/break_chunks
  var Collider Wheel Left: gameObject = <null>
  var Collider Wheel Right: gameObject = <null>
}
