fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    FindChild(gameObject=Self, childName="Fruit Shine", storeResult=var "Shine")
    FindChild(gameObject=Self, childName="Break Effects", storeResult=var "Break Effects")
    CreateObject(gameObject=Mossberry Pickup (localpoolprefabs_assets_areamoss.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "Fruit")
    SetFsmGameObject(gameObject=var "Fruit", fsmName="Control", variableName="Fruit Parent", setValue=var "Self", everyFrame=false)
    SetPosition(gameObject=var "Fruit", vector=(unset), x=(unset), y=(unset), z=0.002, space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Fruit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
    on ACTIVATE → Activated
  }

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"DAMAGED", storeCollider=var "Damager")
    on DAMAGED → Break
  }

  state Activated {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Break {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=moss_vine_cut (sfxstatic_assets_areatutorial.bundle), pitchMin=1.1, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    ActivateGameObject(gameObject=var "Fruit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shine", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Damager", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    FloatAdd(floatVariable=var "Angle", add=180, everyFrame=false, perSecond=false)
    FlingObject(flungObject=var "Fruit", speedMin=8, speedMax=8, angleMin=var "Angle", angleMax=var "Angle")
    AddForce2d(gameObject=var "Fruit", forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=15, vector3=(unset), everyFrame=false)
    SetSpriteRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
  }

  var Angle: float = 0
  var Activated: bool = false
  var Break Effects: gameObject = <null>
  var Damager: gameObject = <null>
  var Fruit: gameObject = <null>
  var Self: gameObject = moss_berry_vine/moss_berry_fruit
  var Shine: gameObject = <null>
}
