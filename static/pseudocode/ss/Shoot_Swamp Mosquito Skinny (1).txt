# Shoot
# 2 location(s):
#   Aqueduct_01 / Swamp Mosquito Skinny (1)
#   Aqueduct_02 / Swamp Mosquito Skinny (1)

fsm Shoot {
  start Idle

  state Idle {
    on DO SHOOT → Shoot Antic
  }

  state Shoot Antic {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    PlayAudioEvent(audioClip=slab_fly_large_spit_antic (sfxstatic_assets_areaslab.bundle), pitchMin=1.1, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=swamp_mosquito_antic (sfxstatic_assets_areaswamp.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.85, finishEvent=→"FINISHED", realTime=false)
    SetIntValue(intVariable=var "Shots", intValue=3, everyFrame=false)
    on FINISHED → Shoot Start
  }

  state Shoot Aim {
    PlayAudioEvent(audioClip=spitting_zombie_spit (sfxstatic_assets_areaaqueduct.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Spit Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Shoot Speed", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    CheckIsBlackThreaded(Target=Self, TrueEvent=(none), FalseEvent=(none), StoreValue=var "Is Black Threaded", EveryFrame=false)
    BoolTestToGameObject(Test=var "Is Black Threaded", ExpectedValue=true, TrueGameObject=Skinny Mosquito Bullet Void (localpoolprefabs_assets_areaaqueduct.bundle), FalseGameObject=Skinny Mosquito Bullet (localpoolprefabs_assets_areaaqueduct.bundle), StoreResult=var "Bullet Crt", EveryFrame=false)
    SpawnObjectFromGlobalPool(gameObject=var "Bullet Crt", spawnPoint=var "Shoot Point", position=(unset), rotation=(unset), storeObject=var "Bullet")
    GetDistanceV2(gameObject=var "Shoot Point", target=var "Hero", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=false)
    FloatOperator(float1=var "Distance", float2=3.5, operation=Divide, storeResult=var "Y Offset", everyFrame=false)
    FloatClamp(floatVariable=var "Y Offset", minValue=0, maxValue=7, everyFrame=false)
    GetAngleToTarget2D(gameObject=var "Shoot Point", target=var "Hero", offsetX=0, offsetY=var "Y Offset", storeAngle=var "Angle", pause=0, everyFrame=false)
    FloatAddRandom(floatVariable=var "Angle", addMin=-5, addMax=5)
    ClampAngleByScale(gameObject=Self, space=0, angleVariable=var "Angle", positiveMinValue=-45, positiveMaxValue=35, negativeMinValue=145, negativeMaxValue=225, everyFrame=false)
    RandomFloat(min=22, max=24, storeResult=var "Shoot Speed")
    on FINISHED → Shoot
  }

  state Repeat? {
    IntAdd(intVariable=var "Shots", add=-1, everyFrame=false)
    IntCompare(integer1=var "Shots", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"REPEAT", everyFrame=false)
    on REPEAT → Shoot Aim
    on FINISHED → Idle
  }

  state Shoot Start {
    on FINISHED → Shoot Aim
  }

  state Clamp To Prev {
    IntCompare(integer1=var "Shots", integer2=3, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatOperator(float1=var "Prev Angle", float2=25, operation=Add, storeResult=var "Angle Max", everyFrame=false)
    FloatOperator(float1=var "Prev Angle", float2=25, operation=Subtract, storeResult=var "Angle Min", everyFrame=false)
    FloatClamp(floatVariable=var "Angle", minValue=var "Angle Min", maxValue=var "Angle Max", everyFrame=false)
    SetFloatValue(floatVariable=var "Prev Angle", floatValue=var "Angle", everyFrame=false)
    on FINISHED → Shoot
  }

  state Shoot {
    SetVelocityAsAngle(gameObject=var "Bullet", angle=var "Angle", speed=var "Shoot Speed", everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Repeat?
  }

  var Shoot Speed: float = 0
  var Speed Multiplier: float = 0
  var Angle: float = 0
  var Distance: float = 0
  var Y Offset: float = 0
  var Prev Angle: float = 0
  var Angle Max: float = 0
  var Angle Min: float = 0
  var Shots: int = 0
  var Is Black Threaded: bool = false
  var Shoot Point: gameObject = Swamp Mosquito Skinny (1)/Shoot Point
  var Bullet: gameObject = <null>
  var Spit Effect: gameObject = Swamp Mosquito Skinny (1)/Spit Effect
  var Bullet Crt: gameObject = <null>
}
