# Dust Effects
# 1 location(s):
#   localpoolprefabs_assets_vampiregnat.bundle / Vampire Gnat Sickle

fsm Dust Effects {
  start Init
  on EFFECT STOP → Effect Stop  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(0, -1), space=0, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=→"PLAY", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on PLAY → Play
  }

  state Play {
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(0, -1), space=0, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"END", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Hit Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    SetPosition2D(GameObject=var "Pt Dust", Vector=var "Hit Point", X=0, Y=0, Space=0, EveryFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Dust")
    on END → Idle
  }

  state Effect Stop {
  }

  var Hit Point: vector2 = (0, 0)
  var Pt Dust: gameObject = Vampire Gnat Sickle/Pt Dust
}
