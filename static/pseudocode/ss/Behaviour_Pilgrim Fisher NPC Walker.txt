# Behaviour
# 3 location(s):
#   Belltown / Pilgrim Fisher NPC Walker
#   Belltown_07 / Pilgrim Fisher NPC Walker
#   Shellwood_01 / Pilgrim Fisher NPC Walker

fsm Behaviour {
  start Walker

  state Walker {
    on DLG STARTED → Talking
    on SING STARTED → Singing
  }

  state Talking {
    CallMethodProper(gameObject=Self, behaviour="CrossSceneWalker", methodName="PauseWalker", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=(none))
    FaceObjectV3(ObjectA=Self, ObjectB=var "Hero", SpriteFacesRight=true, NewAnimationClip="TurnToIdle", ResetFrame=true, PauseBetweenTurns=0, StoreDuration=(unset), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK STOP", delay=0, everyFrame=false)
    on DLG ENDED → End Talk Wait
  }

  state End Talk Wait {
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    on DLG STARTED → Talking
    on SING STARTED → Singing
    on FINISHED → Unpause
  }

  state Unpause {
    CallMethodProper(gameObject=Self, behaviour="CrossSceneWalker", methodName="ResumeWalker", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK START", delay=0, everyFrame=false)
    on FINISHED → Walker
  }

  state Singing {
    CallMethodProper(gameObject=Self, behaviour="CrossSceneWalker", methodName="PauseWalker", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WALK STOP", delay=0, everyFrame=false)
    on SING ENDED → End Talk Wait
  }
}
