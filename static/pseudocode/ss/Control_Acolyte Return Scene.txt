# Control
# 1 location(s):
#   Library_15 / Acolyte Return Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="seenScholarAcolytes", isTrue=→"SPAWN", isFalse=(none))
    on SPAWN → Spawn
  }

  state Spawn {
    GetRandomChild(gameObject=Self, storeResult=var "To Spawn")
    ActivateGameObject(gameObject=var "To Spawn", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var To Spawn: gameObject = <null>
}
