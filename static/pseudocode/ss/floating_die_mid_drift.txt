// uses template: floating_die_mid_drift
fsm floating_die_mid_drift {
  start Init

  state Init {
    GetGrandParent(gameObject=Self, storeResult=var "Dice Game Parent")
    on FINISHED → Idle
  }

  state Idle {
    on MID DRIFT → Float R
  }

  state Float R {
    CheckTargetDirection(gameObject=Self, target=var "Dice Game Parent", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=true)
    GetXDistance(gameObject=Self, target=var "Dice Game Parent", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=var "Mid Range", tolerance=0, equal=(none), lessThan=→"STOP", greaterThan=(none), everyFrame=true)
    AccelerateToX(gameObject=Self, accelerationFactor=0.035, targetSpeed=1)
    on L → Float L
    on STOP → Stop
  }

  state Stop {
    GetXDistance(gameObject=Self, target=var "Dice Game Parent", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=var "Mid Range", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"R", everyFrame=true)
    DecelerateXY(gameObject=Self, decelerationX=0.965, decelerationY=(unset), brakeOnExit=false)
    on R → Float R
  }

  state Float L {
    CheckTargetDirection(gameObject=Self, target=var "Dice Game Parent", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=true)
    GetXDistance(gameObject=Self, target=var "Dice Game Parent", storeResult=var "Distance", everyFrame=true)
    FloatCompare(float1=var "Distance", float2=var "Mid Range", tolerance=0, equal=(none), lessThan=→"STOP", greaterThan=(none), everyFrame=true)
    AccelerateToX(gameObject=Self, accelerationFactor=0.035, targetSpeed=-1)
    on R → Float R
    on STOP → Stop
  }

  var Angular Velocity: float = 0
  var Float Height: float = 0
  var Distance: float = 0
  var Mid Range: float = 0.5
  var Dice Game Parent: gameObject = <null>
}
