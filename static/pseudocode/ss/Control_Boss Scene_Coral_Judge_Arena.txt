# Control
# 1 location(s):
#   Coral_Judge_Arena / Boss Scene

fsm Control {
  start Pause
  on TEST_GATE → Gate Open Scene  // from any state

  sequence state Pause {
    FindNamedChild(gameObject=Self, storeResult=var "Great Gate")
    FindNamedChild(gameObject=var "Great Gate", storeResult=var "Gate")
    ActivateGameObject(gameObject=var "Gate", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate Quest Ender", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=true)
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pilgrims")
    FindNamedChild(gameObject=var "Pilgrims", storeResult=var "Gate BG Pilgrim")
    FindNamedChild(gameObject=var "Pilgrims", storeResult=var "Gate NPC")
    FindNamedChild(gameObject=var "Gate NPC", storeResult=var "Audio Loop Voice")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock B")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Intro 1")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Intro 2")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Pre")
    FindNamedChild(gameObject=Self, storeResult=var "Gate L")
    FindNamedChild(gameObject=Self, storeResult=var "Intro Judge")
    FindNamedChild(gameObject=Self, storeResult=var "Last Judge")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Flame Leap")
    FindNamedChild(gameObject=Self, storeResult=var "Silk Heart")
    FindNamedChild(gameObject=var "Intro Judge", storeResult=var "Sound Point")
    ActivateGameObject(gameObject=var "Gate", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="defeatedLastJudge", isTrue=→"DEFEATED", isFalse=(none))
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"BLACK THREAD", isFalse=(none))
    PlayerDataBoolTest(boolName="encounteredLastJudge", isTrue=→"ENCOUNTERED", isFalse=→"UNENCOUNTERED")
    on UNENCOUNTERED → Music
    on ENCOUNTERED → Encountered
    on DEFEATED → Defeated
    on BLACK THREAD → Set Black Thread
  }

  state Out of Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → In Range
  }

  state In Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on EXIT → Out of Range
    on NEEDOLIN → NPCs Start Singing
    on DO NEEDOLIN SCENE → Needolin Scene
  }

  state NPCs Singing Wait {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.5, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    WaitRandom(timeMin=1, timeMax=1.2, finishEvent=→"FINISHED", realTime=false)
    on END → Needolin End
    on FINISHED → Start Gate
  }

  state Needolin End {
    Tk2dPlayAnimation(gameObject=var "Gate BG Pilgrim", animLibName="", clipName="Round SingEndToLookRight")  // disabled
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0.15, cancelOnEarlyExit=false)
    SendEventByName(eventTarget=GameObject(var "Gate BG Pilgrim"), sendEvent="SING END", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Gate Lock", behaviour="BellShrineGateLock", methodName="StoppedPlaying", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Needolin End 2
  }

  state Intro Flare {
    SetCollider(gameObject=var "Gate NPC", active=false, resetOnExit=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Sound Point", audioClip=last_judge_bg_vvwwhoooffff (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetFsmBool(gameObject=var "Hero", fsmName="Roar and Wound States", variableName="No Roar Shove", setValue=true, everyFrame=false)
    StartRoarEmitter(spawnPoint=Self, delay=0, stunHero=false, roarBurst=false, isSmall=false, noVisualEffect=true, forceThroughBind=false, stopOnExit=false)
    SendEventToRegister(eventName="CENSER SHAKE SMALL")
    ActivateGameObject(gameObject=var "Intro Judge", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro 1", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Pre", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=2)
    on FINISHED → Judge Leap
  }

  state Judge Leap {
    Tk2dPlayAnimationWithEvents(gameObject=var "Intro Judge", clipName="Intro Leap", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Flame Leap", emit=0, resetIfPlaying=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Sound Point", audioClip=last_judge_bg_vvwwhoooffff_dissapear (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Lept
  }

  state Lept {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Big Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventToRegister(eventName="CENSER SHAKE SMALL")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Intro Judge", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End Needolin Scene
  }

  state Start Fall {
    StopRoarEmitter(delay=0)
    SendEventByName(eventTarget=GameObject(var "Last Judge"), sendEvent="INTRO FALL", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro 1", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Gate NPC", animLibName="", clipName="Stand Sing")
    Tk2dPlayFrame(gameObject=var "Gate NPC", frame=0)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=gate_pilgrim_alert (sfxdynamic_assets_areacoral.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SetScale(gameObject=var "Gate NPC", vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Gate BG Pilgrim"), sendEvent="SCARED", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Gate BG Pilgrim", animLibName="", clipName="Round SingToLookLeft")  // disabled
    on BATTLE START → Battle
  }

  state Battle {
    ActivateGameObject(gameObject=var "CamLock B", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro 2", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on JUDGE DEFEATED → Defeated Pause
  }

  state Encountered {
    ActivateGameObject(gameObject=var "Gate BG Pilgrim", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Pre", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Refight Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    ApplyMusicCue(musicCue=Coral_Gorge (audiocuesdynamic_assets_areacoral.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Main Only JudgeArena (audioobjects_assets_all.bundle), transitionTime=6)
    on ENTER → Start Refight
  }

  state Start Refight {
    SendEventByName(eventTarget=GameObject(var "Last Judge"), sendEvent="REFIGHT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Intro 2", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Pre", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=2)
    on BATTLE START → Battle
  }

  state Spawn Silkheart {
    ActivateGameObject(gameObject=var "Silk Heart", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Defeated Pause {
    GetHeroCState(VariableName="dead", StoreValue=var "Hero Dead", EveryFrame=false)
    BoolTest(boolVariable=var "Hero Dead", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait for enter 2
    on CANCEL → Hero Dead
  }

  sequence state Gate Open Scene {
    ActivateGameObject(gameObject=var "Gate L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataVariable(VariableName="SeenLastJudgeGateOpen", SetValue=true)
    GetPosition2D(gameObject=var "Gate", vector=var "Self Pos 2D", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Roar Wave Emitter", Vector=var "Self Pos 2D", X=0, Y=0, Space=0, EveryFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Roar and Wound States", variableName="No Roar Shove", setValue=true, everyFrame=false)
    StartRoarEmitter(spawnPoint=var "Hero", delay=0, stunHero=true, roarBurst=false, isSmall=false, noVisualEffect=true, forceThroughBind=false, stopOnExit=false)
    AnimatorPlayStateWait(target=var "Gate", stateName="Open", finishEvent=(none))
    Wait(time=1, finishEvent=(none), realTime=false)
    StopRoarEmitter(delay=0)
    Wait(time=1, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Gate Quest Ender", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Defeated {
    ActivateGameObject(gameObject=var "Gate L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate BG Pilgrim", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Pre", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=8)
    on FINISHED → Seen Gate Open?
  }

  sequence state Gate Cinematic {
    StartPreloadingScene(SceneName=var "Cinematic End Scene", LoadMode=LoadSceneMode(1))
    StartCinematic(CinematicPlayer=var "Cinematic Player")
    on CINEMATIC START → Cinematic Started
  }

  state Cinematic Ended {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataVariable(VariableName="disablePause", SetValue=false)
    SetPlayerDataVariable(VariableName="SeenLastJudgeGateOpen", SetValue=true)
    SetPlayerDataVariable(VariableName="visitedGrandGate", SetValue=false)
    StopVibrationV2(tag="", fadeTime=0, waitUntilFinish=false)
    on FINISHED → Next Scene
  }

  state Next Scene {
    BeginSceneTransition(sceneName=var "Cinematic End Scene", entryGateName="left_firstEntrance", entryDelay=0, visualization=Default, preventCameraFadeOut=true)
  }

  state Visited City? {
    PlayerDataVariableTest(VariableName="visitedCitadel", ExpectedValue=true, IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on TRUE → Wait for enter
    on FALSE → Wait for enter 3
  }

  state Wait for enter {
    CheckTrackTriggerCount(target=Boss Scene/Gate Open Trigger, count=0, test=MoreThan, everyFrame=true, successEvent=→"FINISHED")
    on FINISHED → Gate Open Scene
  }

  state Seen Gate Open? {
    PlayerDataVariableTest(VariableName="SeenLastJudgeGateOpen", ExpectedValue=true, IsExpectedEvent=→"TRUE", IsNotExpectedEvent=→"FALSE")
    on TRUE → Gate Opened
    on FALSE → Wait for enter 2
  }

  state Gate Opened {
    AnimatorPlay(gameObject=var "Gate", stateName="Opened", layer=0, normalizedTime=1, everyFrame=false)
    ActivateGameObject(gameObject=var "Gate Quest Ender", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Wait for enter 2 {
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"BLACK THREAD", isFalse=(none))
    CheckTrackTriggerCount(target=Boss Scene/Gate Open Trigger, count=0, test=MoreThan, everyFrame=true, successEvent=→"FINISHED")
    on FINISHED → Visited City?
    on BLACK THREAD → Black Thread End
  }

  state Needolin End 2 {
    Tk2dPlayAnimation(gameObject=var "Gate NPC", animLibName="", clipName="Stand Sing End")
    ActivateInteractible(Target=var "Gate NPC", Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → NPC Start React
  }

  state Wait for enter 3 {
    CheckTrackTriggerCount(target=Boss Scene/Gate Open Trigger, count=0, test=MoreThan, everyFrame=true, successEvent=→"FINISHED")
    on FINISHED → Gate Open To Cinematic
  }

  state Gate Open To Cinematic {
    SetPlayerDataVariable(VariableName="disablePause", SetValue=true)
    ActivateGameObject(gameObject=var "Gate L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetPosition2D(gameObject=var "Gate", vector=var "Self Pos 2D", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Roar Wave Emitter", Vector=var "Self Pos 2D", X=0, Y=0, Space=0, EveryFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Roar and Wound States", variableName="No Roar Shove", setValue=true, everyFrame=false)
    StartRoarEmitter(spawnPoint=var "Hero", delay=0, stunHero=true, roarBurst=false, isSmall=false, noVisualEffect=true, forceThroughBind=false, stopOnExit=false)
    AnimatorPlay(gameObject=var "Gate", stateName="Open", layer=0, normalizedTime=0, everyFrame=false)
    ListenForAnimationEvent(Target=var "Gate", Response=→"FINISHED")
    on FINISHED → Gate Cinematic
  }

  state Music {
    ApplyMusicCue(musicCue=Coral_Gorge (audiocuesdynamic_assets_areacoral.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Main Only JudgeArena (audioobjects_assets_all.bundle), transitionTime=6)
    on FINISHED → Out of Range
  }

  sequence state Cinematic Started {
    StopRoarEmitter(delay=0)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    AnimatorPlay(gameObject=var "Gate", stateName="Open", layer=0, normalizedTime=1, everyFrame=false)
    Wait(time=6, finishEvent=(none), realTime=false)
    PlayVibrationV3(vibrationDataAsset=camera_rumble_medium (vibrationstatic_assets_vibrationdataasset.bundle), motors=None, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)  // disabled
    StopVibrationV2(tag="", fadeTime=0, waitUntilFinish=false)
    PlayVibrationV3(vibrationDataAsset=very_large_shake (vibrationstatic_assets_vibrationdataasset.bundle), motors=None, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)  // disabled
    on CINEMATIC END → Cinematic Ended
  }

  state Hero Dead {
  }

  state Needolin Lock {
    SendEventToRegister(eventName="NEEDOLIN LOCK")  // disabled
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=true)
    on GATE LOCKS COMPLETE → Intro Flare
  }

  state Start Gate {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CallMethodProper(gameObject=var "Gate Lock", behaviour="BellShrineGateLock", methodName="StartedPlaying", parameters=[], storeResult=(unset var), EveryFrame=false)
    on END → Needolin End
    on GATE LOCKS PRE COMPLETE → Needolin Lock
  }

  sequence state NPCs Start Singing {
    CallMethodProper(gameObject=var "Gate Lock", behaviour="BellShrineGateLock", methodName="GetIsAllUnlocked", parameters=[], storeResult=var "All Unlocked", EveryFrame=false)
    BoolTest(boolVariable=var "All Unlocked", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gate NPC"), sendEvent="STOP REACT", delay=0, everyFrame=false)
    ActivateInteractible(Target=var "Gate NPC", Activate=false, AllowQueueing=false, UseChildren=false)
    WaitRandom(timeMin=0.1, timeMax=0.2, finishEvent=(none), realTime=false)
    SetRandomAudioClipFromTable(Table=gate_pilgrim_sing (sfxdynamic_assets_areacoral.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    Tk2dPlayAnimation(gameObject=var "Gate NPC", animLibName="", clipName="Stand Sing")
    WaitRandom(timeMin=0.1, timeMax=0.2, finishEvent=(none), realTime=false)
    SendEventByName(eventTarget=GameObject(var "Gate BG Pilgrim"), sendEvent="SING", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Gate BG Pilgrim", animLibName="", clipName="Round Sing")  // disabled
    on FINISHED → NPCs Singing Wait
    on CANCEL → Wait For Cancel
  }

  state Needolin Scene {
    SetBoolValue(boolVariable=var "In Needolin Scene", boolValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Needolin Scene"), sendEvent="BEGIN", delay=0, everyFrame=false)
    on FINISHED → In Range
  }

  state End Needolin Scene {
    BoolTest(boolVariable=var "In Needolin Scene", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "In Needolin Scene", boolValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Needolin Scene"), sendEvent="END", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on FINISHED → Start Fall
  }

  state Wait For Cancel {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.5, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → In Range
  }

  state Set Black Thread {
    SetPlayerDataBool(boolName="encounteredLastJudge", value=true)
    SetPlayerDataBool(boolName="defeatedLastJudge", value=true)
    SetPlayerDataBool(boolName="SeenLastJudgeGateOpen", value=true)
    on FINISHED → Defeated
  }

  state Black Thread End {
    Comment(comment="In Act 3, we ignore all logic related to opening the grand gate - it's busted")
    SendEventToRegister(eventName="BLACK THREAD GATE OPEN")
  }

  state NPC Start React {
    SendEventByName(eventTarget=GameObject(var "Gate NPC"), sendEvent="START REACT", delay=0, everyFrame=false)
    on FINISHED → In Range
  }

  var Cinematic Player: gameObject = Cinematic Player

  var Hero Dead: bool = false
  var All Unlocked: bool = false
  var In Needolin Scene: bool = false
  var Cinematic End Scene: string = "Coral_10"
  var Self Pos 2D: vector2 = (0, 0)
  var CamLock B: gameObject = <null>
  var CamLock Intro 1: gameObject = <null>
  var CamLock Intro 2: gameObject = <null>
  var CamLock Pre: gameObject = <null>
  var Gate: gameObject = <null>
  var Gate BG Pilgrim: gameObject = <null>
  var Gate L: gameObject = <null>
  var Gate Lock: gameObject = Boss Scene/Great Gate/Gate/Gate/Gate Bells
  var Gate NPC: gameObject = <null>
  var Gate Quest Ender: gameObject = Boss Scene/Gate Quest Ender
  var Great Gate: gameObject = <null>
  var Intro Judge: gameObject = <null>
  var Last Judge: gameObject = <null>
  var Pilgrims: gameObject = <null>
  var Pt Flame Leap: gameObject = <null>
  var Silk Heart: gameObject = <null>
  var Sound Point: gameObject = <null>
  var Audio Loop Voice: gameObject = <null>
}
