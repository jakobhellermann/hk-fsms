# Control
# 1 location(s):
#   Belltown / Music Control

fsm Control {
  start Pause

  state Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="spinnerDefeated", isTrue=→"NORMAL", isFalse=→"CURSED")
    on NORMAL → Normal
    on CURSED → Cursed
  }

  state Normal {
    GetCurrentMusicCueName(musicCueName=var "Current Music Cue")
    StringCompare(stringVariable=var "Current Music Cue", compareTo="Belltown", equalEvent=(none), notEqualEvent=→"START", storeResult=(unset), everyFrame=false)
    ApplyMusicCue(musicCue=Belltown (audiocuesdynamic_assets_areabell.bundle), delayTime=0, transitionTime=0)
    ApplyMusicSnapshot(musicSnapshot=Normal (audioobjects_assets_all.bundle), delayTime=0, transitionTime=3, skipMarkerBlock=true)
    on START → First Visit?
  }

  sequence state Cursed {
    WaitForSceneAudioSnapshotsApplied(SendEvent=(none))
    ApplyMusicCue(musicCue=Belltown_Cursed (audiocuesdynamic_assets_areabell.bundle), delayTime=0, transitionTime=0)
    ApplyMusicSnapshot(musicSnapshot=Normal (audioobjects_assets_all.bundle), delayTime=0, transitionTime=3, skipMarkerBlock=true)
  }

  state Start Visited {
    ApplyMusicCue(musicCue=Belltown (audiocuesdynamic_assets_areabell.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=1)
    ApplyMusicSnapshot(musicSnapshot=Normal (audioobjects_assets_all.bundle), delayTime=0, transitionTime=1, skipMusicMarkerBlock=true)
  }

  state First Visit? {
    PlayerDataBoolTest(boolName="visitedBellhartSaved", isTrue=→"VISITED", isFalse=→"UNVISITED")
    on VISITED → Visited Pause
    on UNVISITED → Unvisited Pause
  }

  state Unvisited Pause {
    Wait(time=2.5, finishEvent=(none), realTime=false)
    on FINISHED → Start Visited
  }

  state Visited Pause {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Start Visited
  }

  var Current Music Cue: string = ""
}
