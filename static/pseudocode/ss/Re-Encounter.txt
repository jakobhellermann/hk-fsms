# Re-Encounter
# 1 location(s):
#   Tut_03 / Black Thread States/Normal World/Battle Scene

fsm Re-Encounter {
  start State 1

  state State 1 {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → State 2
  }

  state State 2 {
    PlayerDataBoolTest(boolName="encounteredMossMother", isTrue=→"RE ENCOUNTER", isFalse=(none))
    on RE ENCOUNTER → State 3
  }

  state State 3 {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
  }
}
