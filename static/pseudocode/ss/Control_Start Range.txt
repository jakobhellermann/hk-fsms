# Control
# 1 location(s):
#   Hang_04_boss / Battle Scene/Start Range

// uses template: silk_thread_battle_wake
fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "CamLocks")
    FindNamedChild(gameObject=Self, storeResult=var "Wake Threads")
    FindNamedChild(gameObject=Self, storeResult=var "Pt WakeSlam")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Tension")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Boss Tension")
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tension?
  }

  state State {
    SendEventByName(eventTarget=GameObject(var "Audio Boss Tension"), sendEvent="TENSION END", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"WOKEN", isFalse=→"UNWOKEN", everyFrame=false)
    on WOKEN → Woken
    on UNWOKEN → Roar Lock
  }

  state Woken {
    CallMethodProper(gameObject=var "Battle Scene", behaviour="BattleScene", methodName="StartBattle", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  state Wake {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl())
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=RegainControl())
    CallMethodProper(gameObject=var "Battle Scene", behaviour="BattleScene", methodName="StartBattle", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)  // disabled
    PlayParticleEmitter(gameObject=var "Pt WakeSlam", emit=0, resetIfPlaying=false)  // disabled
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT", delay=0, everyFrame=false)  // disabled
    PlayAudioEvent(audioClip=silk_threads_pull_up_ground (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
  }

  state Roar Lock {
    AudioStopV2(gameObject=var "Audio Loop Tension", fadeTime=0.5, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=silk_scream_mid (sfxstatic_assets_arealibraryareasong.bundle), pitchMin=1, pitchMax=1, volume=0.8, audioPlayerPrefab=Audio Player Actor Lowpass@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=misc_rumble_impact (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLocks", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR WOUND CANCEL", delay=0, everyFrame=false)
    SetFsmBool(gameObject=var "Hero", fsmName="Roar and Wound States", variableName="No Roar Shove", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR ENTER", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake Threads
  }

  state Wake Threads {
    SendEventByName(eventTarget=GameObject(var "Wake Threads"), sendEvent="IMPACT", delay=0, everyFrame=false)
    on SILK STRAND START → Start
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    FadeAudio(gameObject=var "Audio Loop Tension", startVolume=0, endVolume=1, time=4)
    on ENTER → State
  }

  state Start {
    PlayAudioEvent(audioClip=silk_threads_tighten (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on SILK STRAND PULL → Pull Quake
  }

  state Pull Quake {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    PlayParticleEmitter(gameObject=var "Pt WakeSlam", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    PlayAudioEvent(audioClip=silk_threads_pull_up_ground (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="ROAR EXIT SCENE", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 1 Q")
    SetVelocityByScale(gameObject=var "Hero", speed=15, ySpeed=(unset), everyFrame=false)
    DecelerateXY(gameObject=var "Hero", decelerationX=0.85, decelerationY=(unset), brakeOnExit=true)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    CallMethodProper(gameObject=var "Battle Scene", behaviour="BattleScene", methodName="StartBattle", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Hero Rise
  }

  state Hero Rise {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Collect Normal 3 Q", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Wake
  }

  state Tension? {
    ActivateGameObject(gameObject=var "Audio Boss Tension", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    AudioPlaySimple(gameObject=var "Audio Loop Tension", volume=0, oneShotClip=<null>)
    ActivateGameObject(gameObject=var "Audio Boss Tension", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Activated: bool = false
  var Wake Scene: gameObject = <null>
  var Woken Scene: gameObject = <null>
  var Wake Threads: gameObject = <null>
  var Pt WakeSlam: gameObject = <null>
  var Battle Scene: gameObject = <null>
  var CamLocks: gameObject = <null>
  var CamLock: gameObject = <null>
  var Wake Enemies: gameObject = <null>
  var Audio Loop Tension: gameObject = <null>
  var Audio Boss Tension: gameObject = <null>
}
