# Scene Control
# 1 location(s):
#   Greymoor_05 / Scene Control

fsm Scene Control {
  start Pause

  state Pause {
    FindNamedChild(gameObject=Self, storeResult=var "Farmer Enemies")
    FindNamedChild(gameObject=Self, storeResult=var "Vampire Gnat Boss Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Slab Jailer Scene")
    FindNamedChild(gameObject=Self, storeResult=var "Slab Jailer Enemy")
    FindNamedChild(gameObject=Self, storeResult=var "Perch")
    FindNamedChild(gameObject=Self, storeResult=var "Vampire Gnat Roosting")
    ActivateGameObject(gameObject=var "Farmer Enemies", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vampire Gnat Boss Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Vampire Gnat Roosting", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slab Jailer Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Slab Jailer Enemy", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CheckStartedOnThisScene(trueEvent=(none), falseEvent=→"FINISHED", storeValue=(unset))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Perch Angle
  }

  state Boss Check {
    PlayerDataBoolTest(boolName="greymoor05_clearedOut", isTrue=→"CLEARED OUT", isFalse=(none))
    PlayerDataBoolTest(boolName="defeatedVampireGnatBoss", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataVariableTest(VariableName="VampireGnatBossInAltLoc", ExpectedValue=true, IsExpectedEvent=→"BOSS", IsNotExpectedEvent=(none))
    on BOSS → Boss
    on FINISHED → Enemy Suite
    on CLEARED OUT → Cleared Out
  }

  state Boss {
    ActivateGameObject(gameObject=var "Vampire Gnat Boss Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Farmer Enemies", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Jail Cart?
  }

  state Perch Angle {
    PlayerDataBoolTest(boolName="encounteredVampireGnat_05", isTrue=→"ANGLED", isFalse=(none))
    on ANGLED → Angle Perch
    on FINISHED → Roosting?
  }

  state Angle Perch {
    SetRotation(gameObject=var "Perch", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-10, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Roosting?
  }

  state Enemy Suite {
    PlayerDataBoolTest(boolName="hasWalljump", isTrue=(none), isFalse=→"FARMERS")
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"FARMERS", isFalse=(none))
    PlayerDataBoolTest(boolName="citadelWoken", isTrue=(none), isFalse=→"FARMERS")
    PlayerDataBoolTest(boolName="greymoor05_killedJailer", isTrue=→"FARMERS", isFalse=(none))
    PlayerDataBoolTest(boolName="previouslyVisitedGreymoor_05", isTrue=(none), isFalse=→"FARMERS")  // disabled
    PlayerDataBoolTest(boolName="visitedUpperSlab", isTrue=→"FARMERS", isFalse=(none))
    PlayerDataBoolTest(boolName="slab_cloak_battle_completed", isTrue=→"FARMERS", isFalse=→"JAILER")
    on FARMERS → Farmers
    on JAILER → Jailer
  }

  state Jail Cart? {
    PlayerDataBoolTest(boolName="citadelWoken", isTrue=(none), isFalse=→"FINISHED")
    PlayerDataBoolTest(boolName="previouslyVisitedGreymoor_05", isTrue=(none), isFalse=→"FINISHED")
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(boolName="visitedUpperSlab", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(boolName="slab_cloak_battle_completed", isTrue=→"FINISHED", isFalse=→"CART PRESENT")
    on FINISHED → End
    on CART PRESENT → Cart Present
  }

  state Cleared Out {
    on FINISHED → Jail Cart?
  }

  state End {
  }

  state Cart Present {
    ActivateGameObject(gameObject=var "Slab Jailer Scene", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Farmers {
    ActivateGameObject(gameObject=var "Farmer Enemies", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Jail Cart?
  }

  state Jailer {
    ActivateGameObject(gameObject=var "Slab Jailer Enemy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Jail Cart?
  }

  state Roosting? {
    PlayerDataVariableTest(VariableName="VampireGnatBossInAltLoc", ExpectedValue=true, IsExpectedEvent=→"FINISHED", IsNotExpectedEvent=(none))
    PlayerDataBoolTest(boolName="encounteredVampireGnatBoss", isTrue=→"FINISHED", isFalse=(none))
    PlayerDataBoolTest(boolName="encounteredVampireGnat_05", isTrue=→"FINISHED", isFalse=→"ROOSTING")
    on ROOSTING → Roosting
    on FINISHED → Boss Check
  }

  state Roosting {
    ActivateGameObject(gameObject=var "Vampire Gnat Roosting", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Roosting Enemy", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Roosting Enemy", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → End
  }

  state Editor test {
    CheckIsInEditor(trueEvent=→"TEST", falseEvent=(none), storeValue=(unset))
    on FINISHED → Perch Angle
    on TEST → Boss
  }

  var Roosting Enemy: gameObject = Scene Control/Farmer Enemies/Roosting Enemies

  var Farmer Enemies: gameObject = <null>
  var Vampire Gnat Boss Scene: gameObject = <null>
  var Slab Jailer Scene: gameObject = <null>
  var Perch: gameObject = <null>
  var Vampire Gnat Roosting: gameObject = <null>
  var Slab Jailer Enemy: gameObject = <null>
}
