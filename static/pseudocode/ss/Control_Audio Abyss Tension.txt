# Control
# 1 location(s):
#   Cog_10_Destroyed / Audio Abyss Tension

fsm Control {
  start Init

  state Pause {
    Wait(time=var "Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Tension
  }

  state Start Tension {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Trigger2dEvent(gameObject=var "End Range", trigger=1, collideTag="", sendEvent=→"TENSION END", storeCollider=var "")
    on LEAVING SCENE → Leave Scene
    on TENSION END → Fade Out
  }

  state Init {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    FindNamedChild(gameObject=Self, storeResult=var "End Range")
    FindNamedChild(gameObject=Self, storeResult=var "Restart Range")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Pause
  }

  state Leave Scene {
    AudioStop(gameObject=Self, fadeTime=1)
  }

  state Fade Out {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    AudioStop(gameObject=Self, fadeTime=var "Fade Out Time")
    Trigger2dEvent(gameObject=var "Restart Range", trigger=1, collideTag="", sendEvent=→"RESTART", storeCollider=var "")
    on RESTART → Restart
  }

  state Restart {
    Trigger2dEvent(gameObject=var "End Range", trigger=1, collideTag="", sendEvent=→"TENSION END", storeCollider=var "")
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=3)
    on LEAVING SCENE → Leave Scene
    on TENSION END → Fade Out
  }

  var Fade Out Time: float = 0.5
  var Pause: float = 0

  var Activated: bool = false
  var End Range: gameObject = <null>
  var Restart Range: gameObject = <null>
}
