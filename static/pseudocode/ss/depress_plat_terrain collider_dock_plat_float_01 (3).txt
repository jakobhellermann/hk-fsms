// uses template: depress_plat
fsm depress_plat {
  start Init
  on READY → Idle  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=var "Parent", childName="Plat Dust", storeResult=var "Plat Dust")
    FindChild(gameObject=var "Parent", childName="Water Drips", storeResult=var "Water Drips")
    FindChild(gameObject=var "Parent", childName="Art", storeResult=var "Art")
    on FINISHED → Idle
  }

  state Idle {
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Player(9), sendEvent=→"COLLIDE", storeCollider=var "Collider", storeForce=(unset), contactPoint=(unset), contactNormal=var "Normal")
    on COLLIDE → Check
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink Up
  }

  state Check {
    GetVector3XYZ(vector3Variable=var "Normal", storeX=(unset), storeY=var "Normal Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=-1, tolerance=0.01, equal=→"LAND", lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on LAND → Effects?
  }

  state Down {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=var "Shift By", time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Collision2dEventLayer(collision=OnCollisionExit2D, collideTag="", collideLayer=Player(9), sendEvent=→"UNCOLLIDE", storeCollider=var "Collider", storeForce=(unset), contactPoint=(unset), contactNormal=var "Normal")
    on UNCOLLIDE → Uncollide Frame
  }

  state Up {
    AnimatePositionTo(gameObject=var "Art", toValue=(0, 0, 0), localSpace=true, time=0.15, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  state Tink L {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(-0.1, 0, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink R {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(0.1, 0, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink Up {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(0, 0.1, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink Return {
    AnimatePositionTo(gameObject=var "Art", toValue=(0, 0, 0), localSpace=true, time=0.1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  state Effects? {
    PlayRandomAudioClipTable(Table=var "Land Audio Clip Table", AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CheckCurrentMapZoneEnum(MapZone=RED_CORAL_GORGE, trueEvent=→"CORAL", falseEvent=→"FINISHED", storeValue=(unset))  // disabled
    on CORAL → Coral Effects
    on FINISHED → Reg Effects
  }

  state Reg Effects {
    PlayRandomAudioClipTable(Table=var "Land Audio Clip Table", AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Down
  }

  state Coral Effects {
    PlayParticleEmitter(gameObject=var "Water Drips", emit=0, resetIfPlaying=false)
    on FINISHED → Down
  }

  state Uncollide Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Player(9), sendEvent=→"COLLIDE", storeCollider=var "Collider", storeForce=(unset), contactPoint=(unset), contactNormal=var "Normal")
    on FINISHED → Up
    on STAY → Stay Down
  }

  state Stay Down {
    Collision2dEventLayer(collision=OnCollisionExit2D, collideTag="", collideLayer=Player(9), sendEvent=→"UNCOLLIDE", storeCollider=var "Collider", storeForce=(unset), contactPoint=(unset), contactNormal=var "Normal")
    on UNCOLLIDE → Uncollide Frame
  }

  var Shift By: vector3 = (0, -0.1, 0)
  var Land Audio Clip Table: object = <null>

  var Normal Y: float = 0
  var Normal: vector3 = (0, 0, 0)
  var Art: gameObject = <null>
  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
  var Plat Dust: gameObject = <null>
  var Water Drips: gameObject = <null>
}
