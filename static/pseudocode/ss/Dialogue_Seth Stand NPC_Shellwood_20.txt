# Dialogue
# 1 location(s):
#   Shellwood_20 / Seth Stand NPC

fsm Dialogue {
  start Init

  state Init {
    on FINISHED → Look Up
  }

  state Look Up {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Up")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT → Look Up End
    on SING → Observe Start
    on STARTLE → Observe Start
  }

  state Look Up End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Look Up End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Face Hero
  }

  state Face Hero {
    CheckFacingTarget(facingObject=Self, target=var "Hero", spriteFacesRight=false, everyFrame=false, facingEvent=→"FORWARD", notFacingEvent=→"BACK", facingBool=(unset), notFacingBool=(unset))
    on FORWARD → Forward
    on BACK → Back
  }

  state Convo Check {
    SetPlayerDataBool(boolName="MetSethNPC", value=true)
    CheckBackerCreditsSetting(trueEvent=(none), falseEvent=(none), storeValue=var "Show Backer Credits")
    ConvertBoolToString(boolVariable=var "Show Backer Credits", stringVariable=var "Title", falseString="SETH_NPC", trueString="SETH_NPC_BC", everyFrame=false)
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle=var "Title")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TALKED", delay=0, everyFrame=false)
    BoolTest(boolVariable=var "Activated", isTrue=→"REPEAT", isFalse=→"MEET", everyFrame=false)
    on MEET → Meet
    on REPEAT → Repeat
  }

  state Meet {
    RunDialogue(Sheet="Wanderers", Key=var "Meet Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on CONVO_END → End Dialogue
  }

  state End Dialogue {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Watch Hero
  }

  state Repeat {
    RunDialogue(Sheet="Wanderers", Key=var "Repeat Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Start Watch Hero {
    SetFloatValue(floatVariable=var "Watch Time", floatValue=2, everyFrame=false)  // disabled
    CheckFacingTarget(facingObject=Self, target=var "Hero", spriteFacesRight=false, everyFrame=false, facingEvent=→"FORWARD", notFacingEvent=→"BACK", facingBool=(unset), notFacingBool=(unset))  // disabled
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    CheckFacingTarget(facingObject=Self, target=var "Hero", spriteFacesRight=false, everyFrame=true, facingEvent=(none), notFacingEvent=(none), facingBool=var "Facing Hero", notFacingBool=(unset))
    BoolTestMulti(boolVariables=[var "Looking Back", var "Facing Hero"], boolStates=[true, true], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "Looking Back", var "Facing Hero"], boolStates=[false, false], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Turn Back To Look Up?
    on INTERACT → Face Hero
    on STARTLE → Startle
    on SING → Sing
  }

  state Turn Back To Look Up? {
    BoolTest(boolVariable=var "Looking Back", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle Back End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Look Up
    on INTERACT → Face Hero
  }

  state Forward {
    SetBoolValue(boolVariable=var "Looking Back", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Forward")
    on FINISHED → Convo Check
  }

  state Back {
    SetBoolValue(boolVariable=var "Looking Back", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Back")
    on FINISHED → Convo Check
  }

  sequence state Observe Start {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    CheckFacingTarget(facingObject=Self, target=var "Hero", spriteFacesRight=false, everyFrame=false, facingEvent=(none), notFacingEvent=(none), facingBool=var "Facing Hero", notFacingBool=(unset))
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWait(Target=Self, ClipName="Look Up End", AnimationCompleteEvent=(none))
    ConvertBoolToString(boolVariable=var "Facing Hero", stringVariable=var "Anim", falseString="Idle Back", trueString="Idle Forward", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim")
    on FINISHED → Observe
  }

  state Observe {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on FINISHED → Observe End
    on INTERACT → Face Hero
    on SING → Sing
    on STARTLE → Startle
  }

  state Observe End {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"CANCEL", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Look Up
    on CANCEL → Observe
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Look Up")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=Seth_npc_sing (sfxdynamic_assets_seth.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Observe Start
  }

  state Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Seth_NPC_startle (sfxdynamic_assets_seth.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Observe Start
  }

  var Meet Convo Key: string = "SETH_SHELLWOOD_MEET"
  var Repeat Convo Key: string = "SETH_SHELLWOOD_REPEAT"

  var Watch Time: float = 0
  var Activated: bool = false
  var Looking Back: bool = false
  var Show Backer Credits: bool = false
  var Facing Hero: bool = false
  var Title: string = ""
  var Anim: string = ""
  var Audio Loop Voice: gameObject = Seth Stand NPC/Audio Loop Voice
}
