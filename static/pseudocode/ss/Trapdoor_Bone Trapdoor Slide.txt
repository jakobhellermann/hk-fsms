# Trapdoor
# 1 location(s):
#   Mosstown_01 / Bone Trapdoor Slide

fsm Trapdoor {
  start State 1

  state State 1 {
    on OPEN → Jitter
    on ACTIVATE → Activate
  }

  state Jitter {
    ObjectJitterLocal(gameObject=var "Graphic", x=0.05, y=0.05, z=(unset), resetPosOnStateExit=true, limitFps=0)
    PlayParticleEffectsInState(Target=var "Rumble Particles", StartDelay=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=bone_trapdoor_rumble_loop (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Spawned Player")
    on FINISHED → Move
  }

  state Move {
    AudioStop(gameObject=var "Spawned Player", fadeTime=0)
    PlayAudioEvent(audioClip=bone_trapdoor_move_open (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Slide Dust", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetTransformParent(gameObject=var "Slide Dust", parent=<null>, worldPositionStays=true)
    iTweenMoveBy(gameObject=Self, id="", vector=var "Move Offset", time=0.3, delay=0, speed=(unset), easeType=21, loopType=0, space=1, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=0, axis=none, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Move End
  }

  state Activate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Move End {
    ActivateGameObject(gameObject=var "Slide Dust", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
    on FINISHED → Activate
  }

  var Move Offset: vector3 = (4.22, 0, 0)
  var Graphic: gameObject = Bone Trapdoor Slide/Graphic
  var Rumble Particles: gameObject = Bone Trapdoor Slide/Rumble Particles
  var Slide Dust: gameObject = Bone Trapdoor Slide/Slide Dust
  var Spawned Player: gameObject = <null>
}
