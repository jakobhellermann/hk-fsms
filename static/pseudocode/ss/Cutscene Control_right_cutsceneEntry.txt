# Cutscene Control
# 1 location(s):
#   Song_20 / right_cutsceneEntry

fsm Cutscene Control {
  start Init

  state Init {
    on START ENTRY → State 1
  }

  state Take Control {
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait for in position
  }

  state Give Control {
    EndFreeCameraMode()
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=false)
    PlayerDataBoolTest(boolName="act2Started", isTrue=→"FINISHED", isFalse=(none))
    SetPlayerDataBool(boolName="act2Started", value=true)
    QueueSaveGameV2(createAutoSave=true, nameEnum=ACT_2)
    on FINISHED → End
  }

  state Fade In {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=2)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Move Camera Down
  }

  state Wait for title {
    on TITLE DISPLAYED → Move Camera Down
  }

  state Looking Up {
    GetPosition2D(gameObject=right_cutsceneEntry/Scene Entry Pos, vector=var "Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    PositionColliderOnGroundV2(Target=var "Hero", GroundPos=var "Pos")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    HeroPlayLookUpAnim()
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="LookUp", animationTriggerEvent=(none), animationCompleteEvent=(none))  // disabled
    on FINISHED → Position Camera Up
  }

  sequence state Position Camera Up {
    NextFrameEvent(sendEvent=(none))
    CameraRepositionToHeroV2(forceDirect=false, timeOut=2, onFinish=(none))
    StartFreeCameraMode(StoreObject=var "Camera Target")
    NextFixedFrameEvent(SendEvent=(none))
    NextFrameEvent(sendEvent=(none))
    NextFixedFrameEvent(SendEvent=(none))
    GetPosition(gameObject=var "Camera Target", vector=var "Initial Camera Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    Translate(gameObject=var "Camera Target", vector=(unset), x=(unset), y=1.5, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Fade In
  }

  state Move Camera Down {
    iTweenMoveTo(gameObject=var "Camera Target", id="", transformPosition=var "", vectorPosition=var "Initial Camera Pos", time=2.5, delay=0, speed=(unset), space=0, easeType=2, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    on FINISHED → Give Control
  }

  state State 1 {
    ActivateGameObject(gameObject=Area Title Controller, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetPlayerDataBool(boolName="disablePause", value=true)
    on FINISH ENTRY → Take Control
  }

  state Wait for in position {
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=true)
    on FINISHED → Looking Up
  }

  state End {
  }

  var Pos: vector2 = (0, 0)
  var Initial Camera Pos: vector3 = (0, 0, 0)
  var Camera Target: gameObject = <null>
}
