# Control
# 1 location(s):
#   localpoolprefabs_assets_areacrawl.bundle / Bone Worm BlueTurret

  start Init
  on EXTRACT → Extract Start  // from any state

  state Init {
    SetParent(gameObject=var "Animator Group", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Submerged
    on FIND GROUP → 
  }

  state Submerged {
    AnimatorGroupPlay(StateName="Rest", Target=var "Animator Group")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Submerge")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=var "In Alert Range", sendEvent="", outOfRangeEvent="", everyFrame=true)

    BoolAllTrue(boolVariables=[var "Can Alert", var "In Alert Range"], sendEvent=→"ALERT", storeResult=(unset), everyFrame=true)
    on TOOK DAMAGE → Emerge
    on ALERT → Emerge
  }

  state Emerge {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Emerge")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=broodmother_spit (sfxstatic_assets_areaslab.bundle), pitchMin=0.75, pitchMax=1.25, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatorGroupPlay(StateName="Idle", Target=var "Animator Group")
    on FINISHED → Shoot
  }

  state Shoot {
    PlayAudioEvent(audioClip=broodmother_spit_antic (sfxstatic_assets_areaslab.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shoot")
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterInState(gameObject=var "Pt SpitAntic")
    FlingObjectsFromGlobalPoolVelTime(gameObject=Lifeblood Projectile (localpoolprefabs_assets_lifeblood.bundle), spawnPoint=var "Shot Point L", position=(0, 0, 0), frequency=0.3, spawnMin=1, spawnMax=1, speedMinX=-5, speedMaxX=-12, speedMinY=0, speedMaxY=10, originVariationX=0.2, originVariationY=1, startDelay=0.6)

    AudioPlayInState(gameObject=Self, volume=1)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Shot Point B", audioClip=mini_mawlek_spit (sfxstatic_assets_areasong.bundle), pitchMin=0.7, pitchMax=1.3, volume=1, delay=0.6, storePlayer=<null>)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Shot Point B", audioClip=mini_mawlek_spit (sfxstatic_assets_areasong.bundle), pitchMin=0.7, pitchMax=1.3, volume=1, delay=0.8, storePlayer=<null>)

  }

  state Recover {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Recover")
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Unalert?
  }

  state Unalert? {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Unalert Range", storeResult=false, sendEvent="", outOfRangeEvent="UNALERT", everyFrame=false)
    on FINISHED → Shoot
    on UNALERT → Unalert
  }

  state Unalert {
    PlayAudioEvent(audioClip=broodmother_give_birth (sfxstatic_assets_areaslab.bundle), pitchMin=0.75, pitchMax=1.25, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

  }

    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=true)

    on EXTRACT FINISH → Extract Kill
    on HERO DAMAGED → Unextract
  }

  state Extract Kill {
    InstaDeath(target=Self, direction=(unset))  // disabled
    SetParent(gameObject=var "Extract Corpse", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=var "Extract Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetDead())

    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
  }

  state Extract Start {
    AudioStop(gameObject=var "Audio Loop Sing", fadeTime=0)
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    PlayAudioEventRandom(audioClips=[bone_worm_blue_turret_death_1 (sfxstatic_assets_lifeblood.bundle), bone_worm_blue_turret_death_2 (sfxstatic_assets_lifeblood.bundle), bone_worm_blue_turret_death_3 (sfxstatic_assets_lifeblood.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shoot")
    SetDamageHero(Target=Self, Enabled=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=FlashingBluebloodExtract())
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=SetIsLifebloodHealing(false))
    GetPosition2d(gameObject=var "Hero", vector_2d=(unset), x=var "Hero X", y=var "Hero Y", space=0, everyFrame=false)

    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Recover
  }

  var Animator Group: gameObject = Bone Worm BlueTurret/blueblood_worm_growths

  var Hero Y: float = 0
  var Hero X: float = 0
  var Hero Offset: float = 0
