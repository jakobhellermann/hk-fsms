# Control
# 1 location(s):
#   Bone_Steel_Servant / Steel Servant Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    CheckQuestStateV2(NotTrackedEvent=(none), IncompleteEvent=→"COMPLETED", CompletedEvent=→"COMPLETED", Quest=Steel Sentinel Pt2 (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    FindNamedChild(gameObject=Self, storeResult=var "Start Range")
    FindNamedChild(gameObject=Self, storeResult=var "Steel Servant NPC")
    FindNamedChild(gameObject=Self, storeResult=var "Abyss Mass Summon")
    FindNamedChild(gameObject=Self, storeResult=var "Scene NPC")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Encounter")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Battle")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Player Summon")
    FindNamedChild(gameObject=Self, storeResult=var "Battle Scene")  // disabled
    on FINISHED → Idle
    on COMPLETED → Completed
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Capture
  }

  sequence state Capture {
    RunFSM(fsmTemplateControl=template=6159435039333135483 in[Clamp X<-29], finishEvent=(none), everyFrame=false)
    HeroTurnToFace(Target=var "Steel Servant NPC")
    on FINISHED → Flourish
  }

  state Flourish {
    SendEventToRegister(eventName="TENSION END")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=StopAnimationControl())
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Challenge Talk Start")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Hero", audioClip=hornet_sword_1 (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0.25, storePlayer=<null>)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "CamLock Encounter", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Battle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Convo
  }

  state Convo {
    RunDialogue(Sheet="Coral", Key="STEEL_SERVANT_RITUAL", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=var "Scene NPC")
    on CONVO_END → End Dialogue
  }

  state End Dialogue {
    EndDialogue(ReturnControl=false, ReturnHUD=false, Target=var "Scene NPC", UseChildren=false)
    on FINISHED → Summon Start
  }

  state Summon Start {
    AnimatorPlay(gameObject=var "Abyss Mass Summon", stateName="Abyss_Mass_Summon_Appear", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=1.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Summon Burst
  }

  state Summon Burst {
    ActivateGameObject(gameObject=var "Audio Player Summon", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="SUMMON BURST")
    SendEventToRegister(eventName="FLAMES OFF")
    SendEventByName(eventTarget=GameObject(var "Steel Servant NPC"), sendEvent="HIDE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Abyss Mass"), sendEvent="WAKE", delay=0, everyFrame=false)
    on BATTLE START → Battle
  }

  state Battle {
    SetPlayerDataBool(boolName="EncounteredSummonedSaviour", value=true)
    ActivateGameObject(gameObject=var "CamLock First Entry", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock Encounter", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Completed {
    SendEventToRegister(eventName="FLAMES OFF")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Setup Encountered {
    PlayerDataBoolTest(boolName="EncounteredSummonedSaviour", isTrue=(none), isFalse=→"FINISHED")
    SetPosition2d(gameObject=var "Start Range", vector=(unset), x=32, y=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Init
  }

  var Abyss Mass Summon: gameObject = <null>
  var Audio Player Summon: gameObject = <null>
  var CamLock Battle: gameObject = <null>
  var CamLock Encounter: gameObject = <null>
  var Scene NPC: gameObject = <null>
  var Start Range: gameObject = <null>
  var Steel Servant NPC: gameObject = <null>
  var Abyss Mass: gameObject = Steel Servant Scene/Battle Scene/Wave 1/Abyss Mass
  var Battle Scene: gameObject = <null>
  var CamLock First Entry: gameObject = Steel Servant Scene/CamLock First Entry
}
