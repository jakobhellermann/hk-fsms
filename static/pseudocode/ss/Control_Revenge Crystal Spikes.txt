# Control
# 1 location(s):
#   localpoolprefabs_assets_shared.bundle / Revenge Crystal Spikes

fsm Control {
  start Init

  state Init {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=var "Sprite", active=true)
    GetOwner(storeGameObject=var "Self")
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Spike Form", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Active
  }

  state Active {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Spawn Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=var "Spawn Terrain Detector", trigger=1, collideTag="", sendEvent=→"BREAK", storeCollider=var "")
    Tk2dWatchAnimationEvents(gameObject=var "Sprite", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on BREAK → Break
    on FINISHED → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    PlayParticleEmitter(gameObject=var "Pt Shatter", emit=0, resetIfPlaying=false)
    WaitParticleSystem(target=var "Pt Shatter", stop=false, withChildren=false, timeOut=3)
    on FINISHED → Recycle
  }

  var Rotation: float = 0
  var Pt Shatter: gameObject = Revenge Crystal Spikes/Pt Shatter
  var Self: gameObject = <null>
  var Sprite: gameObject = Revenge Crystal Spikes/Sprite
  var Spawn Terrain Detector: gameObject = Revenge Crystal Spikes/Spawn Terrain Detector
}
