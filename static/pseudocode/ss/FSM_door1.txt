# FSM
# 1 location(s):
#   Shellwood_25 / door1

fsm FSM {
  start Wait

  state Wait {
    on START ENTRY → Set Respawn
  }

  state Set Respawn {
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="GetSceneNameString", parameters=[], storeResult=var "Scene Name", EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetBenchRespawn", parameters=["Death Respawn Marker", var "Scene Name", 0, true], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="SetCurrentMapZoneAsRespawn", parameters=[], storeResult=var "Scene Name", EveryFrame=false)
    SaveGame()
    EnableFSM(gameObject=Self, fsmName="FSM", enable=false, resetOnExit=false)
  }

  var Scene Name: string = ""
}
