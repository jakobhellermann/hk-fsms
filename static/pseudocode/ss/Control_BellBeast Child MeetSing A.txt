# Control
# 2 location(s):
#   Bellway_Centipede_Arena / Boss Scene/Bell Beast Children/BellBeast Child MeetSing A
#   Bellway_Centipede_Arena / Boss Scene/Bell Beast Children/BellBeast Child MeetSing C

fsm Control {
  start Init

  state Init {
    on FINISHED → Roll Out
  }

  state Roll Out {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Insta Roll")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayAudioEvent(audioClip=bell_beast_underground_emerge (sfxstatic_assets_areabellway.bundle), pitchMin=1.1, pitchMax=1.5, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    RandomFloat(min=20, max=25, storeResult=var "Launch Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Launch Speed", everyFrame=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Velocity Y", float2=-0.01, tolerance=0, equal=(none), lessThan=→"FALL", greaterThan=(none), everyFrame=true)
    on FALL → Fall
  }

  state Fall {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="RollToDive")
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Land {
    PlayAudioEventRandom(audioClips=[bell_particle_impact_1 (sfxstatic_assets_areabell.bundle), bell_particle_impact_2 (sfxstatic_assets_areabell.bundle), bell_particle_impact_3 (sfxstatic_assets_areabell.bundle), bell_particle_impact_4 (sfxstatic_assets_areabell.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LandToIdle")
    PlayVibrationV3(vibrationDataAsset=tiny_impact (vibrationstatic_assets_vibrationdataasset.bundle), motors=None, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)
    on CHILDREN SING → Sing
  }

  sequence state Sing {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=(none), realTime=false)
    AudioPlaySimple(gameObject=Self, volume=(unset), oneShotClip=<null>)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Sing Anim")
  }

  var Sing Anim: string = "Sing"

  var Launch Speed: float = 0
  var Velocity Y: float = 0
}
