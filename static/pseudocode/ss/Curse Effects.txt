# Curse Effects

#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Thread/Spool/Bind Orb
#   Menu_Title / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Thread/Spool/Bind Orb

fsm Curse Effects {
  start Inert
  on DEACTIVATE → Inert  // from any state

  state Inert {
    on ACTIVATE WITCH → Set Cursed
    on ACTIVATE FWITCH → Set Full
  }

  state Witch Wiggle Loop {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Wiggle Loop Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on SILK CURSED ORB DISAPPEAR → Witch Idle
    on HEALTH CURSE HEALTH BROKEN → Witch Idle
  }

  state Witch Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Idle Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on SILK CURSED ORB DISAPPEAR ANTIC → Witch Wiggle Loop
    on HEALTH CURSE ANTIC → Witch Wiggle Loop
    on SILK CURSED UPDATE → Witch Wiggle Once
    on SILK CURSED BOB → Witch Wiggle Once
  }

  sequence state Witch Wiggle Once {
    WaitRandom(timeMin=0.05, timeMax=0.15, finishEvent=(none), realTime=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Wiggle Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Witch Idle
    on SILK CURSED ORB DISAPPEAR ANTIC → Witch Wiggle Loop
    on HEALTH CURSE ANTIC → Witch Wiggle Loop
    on SILK CURSED BOB → Witch Wiggle Once
  }

  state Set Cursed {
    SetStringValue(stringVariable=var "Idle Anim", stringValue="HUD Frame Witch Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Wiggle Anim", stringValue="HUD Frame Witch Wiggle", everyFrame=false)
    SetStringValue(stringVariable=var "Wiggle Loop Anim", stringValue="HUD Frame Witch Wiggle Loop", everyFrame=false)
    on FINISHED → Witch Wiggle Once
  }

  state Set Full {
    SetStringValue(stringVariable=var "Idle Anim", stringValue="HUD Frame FWitch Idle", everyFrame=false)
    SetStringValue(stringVariable=var "Wiggle Anim", stringValue="HUD Frame FWitch Wiggle", everyFrame=false)
    SetStringValue(stringVariable=var "Wiggle Loop Anim", stringValue="HUD Frame FWitch Wiggle Loop", everyFrame=false)
    on FINISHED → Witch Wiggle Once
  }

  var Wiggle Anim: string = ""
  var Idle Anim: string = ""
  var Wiggle Loop Anim: string = ""
}
