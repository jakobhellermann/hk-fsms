# Control
# 1 location(s):
#   Abyss_05 / Abyss Crawler Large (1)

fsm Control {
  start Init

  state Init {
    StartCrawler(Target=Self, ScheduleTurn=false)
    on FINISHED → Crawl
  }

  state Crawl {
    SetAudioVolume(gameObject=var "Audio Crawl", volume=1, everyFrame=false)
    SetAudioClip(gameObject=var "Audio Crawl", audioClip=abyss_giant_goomba_walk_loop (sfxstatic_assets_areaabyss.bundle), autoPlay=true, stopOnExit=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    on ALERT → Stop Crawler
    on TOOK DAMAGE → Stop Crawler
  }

  state Charge Antic {
    AudioStop(gameObject=Self, fadeTime=0)
    PlayAudioEvent(audioClip=abyss_giant_goomba_antic (sfxstatic_assets_areaabyss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Charge Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Charge
  }

  state Charge {
    SetCrawlerAnim(CrawlAnim="Charge", TurnAnim="TurnQuick", Target=Self)
    StartCrawler(Target=Self, ScheduleTurn=false)
    SetCrawlerSpeed(Speed=10, Target=Self)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    SetAudioClip(gameObject=var "Audio Crawl", audioClip=abyss_giant_goomba_charge_loop (sfxstatic_assets_areaabyss.bundle), autoPlay=true, stopOnExit=false)
    PlayParticleEmitter(gameObject=var "Pt Charge", emit=0, resetIfPlaying=false)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(25, 20), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=true, Delay=0, RepeatDelay=0.15)
    on FINISHED → Can End
  }

  state Can End {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Charge Range", storeResult=false, sendEvent="", outOfRangeEvent="UNALERT", everyFrame=true)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(25, 20), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=true, Delay=0, RepeatDelay=0.15)
    on UNALERT → Charge End
  }

  state Charge End {
    PlayAudioEvent(audioClip=skull_king_jump (sfxstatic_assets_areabone.bundle), pitchMin=1.5, pitchMax=1.5, volume=0.4, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    StopCrawler(Target=Self, WaitForTurn=false)
    AudioStop(gameObject=var "Audio Crawl", fadeTime=0.15)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Charge End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocityByScale(gameObject=Self, speed=-10, ySpeed=(unset), everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.95, decelerationY=(unset), brakeOnExit=true)
    RayCast2dV2(fromGameObject=var "Ray Pt Ground", fromPosition=(unset), direction=(0, -1), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=(none), storeDidHit=var "No Ground Below", storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    BoolFlipEveryFrame(boolVariable=var "No Ground Below", everyFrame=true)
    SetVelocity2dBool(gameObject=Self, activeBool=var "No Ground Below", vector=(unset), x=0, y=(unset), everyFrame=true)
    on FINISHED → Charge Recover
  }

  state Charge Recover {
    StopParticleEmitter(gameObject=var "Pt Charge")
    SetCrawlerAnim(CrawlAnim="Crawl", TurnAnim="Turn", Target=Self)
    SetCrawlerSpeed(Speed=2, Target=Self)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    StartCrawler(Target=Self, ScheduleTurn=false)
    on FINISHED → Crawl
  }

  state Stop Crawler {
    StopCrawler(Target=Self, WaitForTurn=true)
    on FINISHED → Charge Antic
  }

  var No Ground Below: bool = false
  var Pt Charge: gameObject = Abyss Crawler Large (1)/Pt Charge
  var Ray Pt Ground: gameObject = Abyss Crawler Large (1)/Ray Pt Ground
  var Audio Crawl: gameObject = Abyss Crawler Large (1)/Audio Crawl
}
