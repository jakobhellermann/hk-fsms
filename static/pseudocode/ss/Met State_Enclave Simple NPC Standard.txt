  }

    on CANCEL → Disable FSM
    on FINISHED → Wait for dialogue
  }

  }

  state Disable FSM {
    EnableFsmSelf(SetEnabled=false)
  }

  state Set Met {
    SetPlayerDataBool(boolName="MetEnclaveScaredPilgrim", value=true)
    on FINISHED → Disable FSM
  }

  var Value: int = 0
}
