# FSM
# 2 location(s):
#   Song_01 / Breakable Floor_basic Song_01/Jitter Bits
#   Tube_Hub / Breakable Floor_basic/Jitter Bits

// uses template: floor_jitter
fsm FSM {
  start Init
  on EXIT → End  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Jitter Floor")
    on FINISHED → Idle
  }

  state Idle {
    on ENTER → Move Jitter
  }

  state Jitter Check {
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Velocity", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"MOVE", everyFrame=true)
    FloatCompare(float1=var "X Velocity", float2=-1, tolerance=0, equal=(none), lessThan=→"MOVE", greaterThan=(none), everyFrame=true)
    on EXIT → Idle
    on MOVE → Move Jitter
  }

  state End {
    SendMessage(gameObject=var "Jitter Floor", delivery=BroadcastMessage, options=1, functionCall=StopJitter())
    on FINISHED → Idle
  }

  state Move Jitter {
    SendMessage(gameObject=var "Jitter Floor", delivery=BroadcastMessage, options=1, functionCall=StartJitter())
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=var "X Velocity", y=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "X Velocity", float2=0, tolerance=1, equal=→"STOP", lessThan=(none), greaterThan=(none), everyFrame=true)
    on STOP → Stop Pause
  }

  state Stop Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop
  }

  state Stop {
    SendMessage(gameObject=var "Jitter Floor", delivery=BroadcastMessage, options=1, functionCall=StopJitter())
    on FINISHED → Jitter Check
  }

  var X Velocity: float = 0
  var Jitter Floor: gameObject = <null>
}
