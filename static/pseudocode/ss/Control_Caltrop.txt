# Control
# 1 location(s):
#   localpoolprefabs_assets_areadust.bundle / Caltrop

fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Fx", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetDamageEnemyAmount(target=Self, damageDealt=0)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    TranslateRandom(gameObject=Self, translateMin=(0, 0, -0.001), translateMax=(0, 0, 0), space=1)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on TINK DOWN → Launch D
    on TINK LEFT → Launch L
    on TINK RIGHT → Launch R
    on TINK UP → Launch U
    on START TINK → Launch Frame
  }

  state Idle {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=DoSpin())
    WaitRandom(timeMin=8, timeMax=9, finishEvent=→"BREAK", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"BREAK", storeCollider=var "")
    GetSpeed2d(gameObject=Self, storeResult=var "Speed Crt", everyFrame=true)
    FloatCompare(float1=var "Speed Crt", float2=0, tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"PROJECTILE BREAK", sendEventSpikes=→"PROJECTILE BREAK", sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on TINK DOWN → Launch D
    on TINK LEFT → Launch L
    on TINK RIGHT → Launch R
    on TINK UP → Launch U
    on BREAK → Break
    on FINISHED → Break Pause
    on PROJECTILE BREAK → Break
    on DEALT DAMAGE → Break
  }

  state Break {
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Fx", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=spikes_break (herostatic_assets_all.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Launch L {
    SetVelocityAsAngle(gameObject=Self, angle=150, speed=25, everyFrame=false)
    on FINISHED → Launch Frame
  }

  state Launched {
    SetDamageEnemyAmount(target=Self, damageDealt=5)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=DoSpin())
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))  // disabled
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on BREAK → Break
    on DAMAGED ENEMY → Break
    on FINISHED → Launched - Tink Breakable
  }

  state Launch Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=15, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Speed", everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    on FINISHED → Launched
  }

  state Launch R {
    SetVelocityAsAngle(gameObject=Self, angle=30, speed=25, everyFrame=false)
    on FINISHED → Launch Frame
  }

  state Launch U {
    SetVelocityAsAngle(gameObject=Self, angle=90, speed=35, everyFrame=false)
    on FINISHED → Launch Frame
  }

  state Launch D {
    SetVelocityAsAngle(gameObject=Self, angle=270, speed=20, everyFrame=false)
    on FINISHED → Launch Frame
  }

  state Break Pause {
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"PROJECTILE BREAK", sendEventSpikes=→"PROJECTILE BREAK", sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Break
    on TINK DOWN → Launch D
    on TINK LEFT → Launch L
    on TINK RIGHT → Launch R
    on TINK UP → Launch U
    on BREAK → Break
    on PROJECTILE BREAK → Break
    on DEALT DAMAGE → Break
  }

  state Launched - Tink Breakable {
    SetDamageEnemyAmount(target=Self, damageDealt=5)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=DoSpin())
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))  // disabled
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))  // disabled
    Collision2dEventLayer(collision=OnCollisionStay2D, collideTag="", collideLayer=Enemies(11), sendEvent=→"BREAK", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))  // disabled
    Wait(time=1.75, finishEvent=→"BREAK", realTime=false)
    on BREAK → Break
    on TINK DOWN → Break
    on TINK LEFT → Break
    on TINK RIGHT → Break
    on TINK UP → Break
    on DAMAGED ENEMY → Break
  }

  var Spin Factor: float = 10

  var Speed: float = 0
  var Speed Crt: float = 0
  var Break Fx: gameObject = Caltrop/Break Fx
  var Collider: gameObject = <null>
  var Self: gameObject = <null>
  var Terrain: gameObject = Caltrop/Terrain
}
