# Placement Control
# 1 location(s):
#   Ant_04_mid / Black Thread States Thread Only Variant/Normal World/Battle Scene/Mapper NPC

fsm Placement Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Ant Battle
  }

  state Check Ant Battle {
    PlayerDataBoolTest(boolName="ant04_battleCompleted", isTrue=(none), isFalse=→"WAIT FOR BATTLE")
    PlayerDataBoolTest(boolName="mapperMetInAnt04", isTrue=→"PRESENT", isFalse=→"AWAY")
    on PRESENT → Idle
    on WAIT FOR BATTLE → Waiting
    on AWAY → Away
  }

  state Idle {
  }

  state Waiting {
    EnableFSM(gameObject=Self, fsmName="Rest Animation", enable=false, resetOnExit=false)  // disabled
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Right")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=70, z=(unset), space=0, everyFrame=true, lateUpdate=false)
    on SHOP NPC START → Activate
  }

  state Activate {
    SetPlayerDataBool(boolName="mapperMetInAnt04", value=true)
    SetPlayerDataBool(boolName="SeenMapperHuntersNest", value=true)
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=(unset), Target=Self)
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Activate Rest After Delay
  }

  state Away {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Activate Rest After Delay {
    EnableFSM(gameObject=Self, fsmName="Rest Animation", enable=true, resetOnExit=false)
  }
}
