# chapel_door_control
# 1 location(s):
#   Bonegrave / Chapel Door Control

// uses template: chapel_door_control
fsm chapel_door_control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "door1")
    FindNamedChild(gameObject=Self, storeResult=var "Death Respawn Marker")
    FindNamedChild(gameObject=Self, storeResult=var "Door Open")
    FindNamedChild(gameObject=Self, storeResult=var "Door Closed")
    FindNamedChild(gameObject=Self, storeResult=var "Door DoClose")
    SendMessage(gameObject=var "door1", delivery=SendMessage, options=1, functionCall=SetIsInactive(true))
    ActivateGameObject(gameObject=var "Door Closed", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Door DoClose", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Door Open", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Crest Check
  }

  state Crest Check {
    EnumSwitch(enumVariable=var "Crest Type", compareTo=[CrestType(1), CrestType(2), CrestType(3), CrestType(4), CrestType(5), CrestType(6)], sendEvent=[→"REAPER", →"WANDERER", →"WARRIOR", →"WITCH", →"TOOLMASTER", →"SPINNER"], everyFrame=false)
    on REAPER → Reaper
    on WANDERER → Wanderer
    on WARRIOR → Beast
    on WITCH → Witch
    on TOOLMASTER → Toolmaster
    on SPINNER → Shaman
  }

  state Reaper {
    SetObjectValue(objectVariable=var "Crest Object", objectValue=Reaper (dataassets_assets_assets/dataassets/tools/crestitems.bundle), everyFrame=false)
    SetStringValue(stringVariable=var "Closed PD Bool", stringValue="chapelClosed_reaper", everyFrame=false)
    on FINISHED → State Check
  }

  state Wanderer {
    SetObjectValue(objectVariable=var "Crest Object", objectValue=Wanderer (dataassets_assets_assets/dataassets/tools/crestitems.bundle), everyFrame=false)
    SetStringValue(stringVariable=var "Closed PD Bool", stringValue="chapelClosed_wanderer", everyFrame=false)
    on FINISHED → State Check
  }

  state Beast {
    SetObjectValue(objectVariable=var "Crest Object", objectValue=Warrior (dataassets_assets_assets/dataassets/tools/crestitems.bundle), everyFrame=false)
    SetStringValue(stringVariable=var "Closed PD Bool", stringValue="chapelClosed_beast", everyFrame=false)
    on FINISHED → State Check
  }

  state Witch {
    SetObjectValue(objectVariable=var "Crest Object", objectValue=Witch (dataassets_assets_assets/dataassets/tools/crestitems.bundle), everyFrame=false)
    SetStringValue(stringVariable=var "Closed PD Bool", stringValue="chapelClosed_witch", everyFrame=false)
    on FINISHED → State Check
  }

  state Toolmaster {
    SetObjectValue(objectVariable=var "Crest Object", objectValue=Toolmaster (dataassets_assets_assets/dataassets/tools/crestitems.bundle), everyFrame=false)
    SetStringValue(stringVariable=var "Closed PD Bool", stringValue="chapelClosed_toolmaster", everyFrame=false)
    on FINISHED → State Check
  }

  state Shaman {
    SetObjectValue(objectVariable=var "Crest Object", objectValue=Spell (dataassets_assets_assets/dataassets/tools/crestitems.bundle), everyFrame=false)
    SetStringValue(stringVariable=var "Closed PD Bool", stringValue="chapelClosed_shaman", everyFrame=false)
    on FINISHED → State Check
  }

  state State Check {
    GetIsCrestUnlocked(Crest=var "Crest Object", trueEvent=(none), falseEvent=→"OPEN", storeValue=(unset))
    PlayerDataBoolTest(boolName=var "Closed PD Bool", isTrue=→"CLOSED", isFalse=→"DO CLOSE")
    on OPEN → Open
    on DO CLOSE → Do Close
    on CLOSED → Closed
  }

  state Open {
    SendMessage(gameObject=var "door1", delivery=SendMessage, options=1, functionCall=SetIsInactive(false))
    ActivateGameObject(gameObject=var "Door Open", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Closed {
    SendMessage(gameObject=var "door1", delivery=SendMessage, options=1, functionCall=SetIsInactive(true))
    ActivateGameObject(gameObject=var "Door Closed", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Do Close {
    SendMessage(gameObject=var "door1", delivery=SendMessage, options=1, functionCall=SetIsInactive(true))
    SetPlayerDataBool(boolName=var "Closed PD Bool", value=true)
    FindNamedChild(gameObject=Self, storeResult=var "Death Respawn Marker")
    GameObjectIsNull(gameObject=var "Death Respawn Marker", isNull=(none), isNotNull=→"SET_RESPAWN", storeResult=(unset), everyFrame=false)
    SetDeathRespawnV2(RespawnMarkerName="Death Respawn Marker", RespawnType=0, RespawnFacingRight=true, RespawnMapZone=NONE, RespawnExtraRestZone=None)
    SetDeathRespawn(gameManager=var "GameManager", respawnMarkerName="Death Respawn Marker", respawnType=0, respawnFacingRight=true)  // disabled
    on FINISHED → Activate Door Close
    on SET_RESPAWN → Set Respawn
  }

  state Activate Door Close {
    ActivateGameObject(gameObject=var "Door DoClose", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Set Respawn {
    SetDeathRespawnMarker(Target=var "Death Respawn Marker")
    on FINISHED → Activate Door Close
  }

  var Crest Type: enum = CrestType(2)

  var Closed PD Bool: string = ""
  var door1: gameObject = <null>
  var Door Art: gameObject = <null>
  var Close Effects: gameObject = <null>
  var Death Respawn Marker: gameObject = <null>
  var Door Closed: gameObject = <null>
  var Door DoClose: gameObject = <null>
  var Door Open: gameObject = <null>
  var Crest Object: object = <null>
}
