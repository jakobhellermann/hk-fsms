#   Hang_09 / Soft Waterfall Region

fsm Hero Effects {
  start Init

  state Init {
    on FINISHED → Out
  }

  state Out {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"ENTER", storeCollider=var "")
    AudioStop(gameObject=var "Pt Hero Splash", fadeTime=0)
    on ENTER → In
  }

  state In {
    Trigger2dEventLayer(trigger=OnTriggerExit2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"EXIT", storeCollider=var "")
    PlayParticleEmitterChildren(gameObject=var "Pt Hero Splash", resetTimeIfPlaying=false, stopOnStateExit=true)
    SetPositionToObject2D(gameObject=var "Pt Hero Splash", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=true)
    GetPlayerDataVariable(VariableName="CloakFlySmell", StoreValue=var "Cloak Smells")  // disabled
    Wait(time=0.5, finishEvent=→"CLEAN", realTime=false)
    BoolTestDelay(boolVariable=var "Cloak Smells", isTrue=→"SET CLEAN", isFalse=(none), delay=0.5)  // disabled
    AudioPlaySimple(gameObject=var "Pt Hero Splash", volume=1, oneShotClip=<null>)
    on EXIT → Out
    on CLEAN → Reduce Odours
  }

  state Reduce Odours {
    SetPlayerDataVariable(VariableName="CloakFlySmell", SetValue=false)  // disabled
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=ReduceOdours(10))
    on FINISHED → In
  }

  var Cloak Smells: bool = false
  var Pt Hero Splash: gameObject = Soft Waterfall Region/Pt Hero Splash
}
