# Rest Animation
# 4 location(s):
#   Bone_04 / Mapper NPC
#   Bone_East_01 / Mapper NPC
#   Crawl_01 / Mapper NPC
#   Peak_02 / Mapper NPC

fsm Rest Animation {
  start Wait For Hero In Position
  on END ANIM → End Anim  // from any state

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Mapper NPC@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Turn Right
    on RIGHT → Start Singing
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Turn Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Start Singing
  }

  state Start Singing {
    SetRandomAudioClipFromTable(Table=Shakra_call (sfxdynamic_assets_mapper.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEventsV2(gameObject=Self, clipName="Idle to Sing Right", animationTriggerEvent=(none), animationCompleteEvent=(none), animationInterruptedEvent=→"BREAK")
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    WaitRandom(timeMin=var "Min Turn Delay", timeMax=var "Max Turn Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sing Left
    on CANCEL → Sing to Idle
    on BREAK → End Anim
  }

  state Sing Left {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing Right to Left", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    WaitRandom(timeMin=var "Min Turn Delay", timeMax=var "Max Turn Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sing Right
    on CANCEL → Sing Right
  }

  state Sing Right {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing Left to Right", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    WaitRandom(timeMin=var "Min Turn Delay", timeMax=var "Max Turn Delay", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Sing Left
    on CANCEL → Sing to Idle
  }

  state Sing to Idle {
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=shakra_notice (sfxdynamic_assets_mapper.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing Right to Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Reset
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[false], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state End Anim {
    SetBoolValue(boolVariable=var "Resting", boolValue=false, everyFrame=false)
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    on FINISHED → Idle
  }

  state Wait For Hero In Position {
    WaitForHeroInPosition(sendEvent=(none), skipIfAlreadyPositioned=true)
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Idle
  }

  var Max Turn Delay: float = 5
  var Min Turn Delay: float = 3

  var Resting: bool = false
  var Audio Loop Voice: gameObject = Mapper NPC/Audio Loop Voice
  var Call Audio Clip: object = <null>
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
