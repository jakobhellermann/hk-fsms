# Control
# 1 location(s):
#   localpoolprefabs_assets_laceboss.bundle / Lost Lace Ground Tendril

fsm Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Tendrils", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Antic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "tendril antic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Antic
  }

  state Antic {
    PlayRandomAudioClipTable(Table=lost_lace_ground_tendrils_antic (sfxdynamic_assets_areaabyss.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Self X", lowerValue=5, upperValue=54, boolVariable=(unset), trueEvent=(none), falseEvent=→"ARENA OUT", everyFrame=false)
    SetFloatValue(floatVariable=var "Antic Y", floatValue=-6.7, everyFrame=false)
    ActivateGameObject(gameObject=var "Antic", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition2d(gameObject=var "Antic", vector=(unset), x=(unset), y=var "Antic Y", space=1, everyFrame=true, lateUpdate=false)
    EaseFloat(fromValue=-6.7, toValue=-4.7, floatVariable=var "Antic Y", time=0.2, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    AnimatePositionBy(gameObject=var "Antic", shiftBy=(0, 2, 0), time=0.2, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)  // disabled
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "tendril antic", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Attack
    on ARENA OUT → Bump R?
  }

  state Attack {
    PlayRandomAudioClipTable(Table=lost_lace_ground_tendrils_whip_up (sfxdynamic_assets_laceboss.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    ActivateGameObject(gameObject=var "Tendrils", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    AnimatePositionBy(gameObject=var "Antic", shiftBy=(0, -3, 0), time=0.5, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    SetPosition2d(gameObject=var "Antic", vector=(unset), x=(unset), y=-6.7, space=1, everyFrame=false, lateUpdate=false)
    RecycleSelf()
  }

  state Bump R? {
    FloatInRange(floatVariable=var "Self X", lowerValue=4, upperValue=5, boolVariable=(unset), trueEvent=(none), falseEvent=→"FINISHED", everyFrame=false)
    SetPosition2d(gameObject=Self, vector=(unset), x=5.1, y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEvent(eventTarget=Self, sendEvent=→"BUMPED", delay=0, everyFrame=false)
    on FINISHED → Bump L?
    on BUMPED → Antic
  }

  state Bump L? {
    FloatInRange(floatVariable=var "Self X", lowerValue=54, upperValue=55, boolVariable=(unset), trueEvent=(none), falseEvent=→"FINISHED", everyFrame=false)
    SetPosition2d(gameObject=Self, vector=(unset), x=53.9, y=(unset), space=0, everyFrame=false, lateUpdate=false)
    SendEvent(eventTarget=Self, sendEvent=→"BUMPED", delay=0, everyFrame=false)
    on FINISHED → Recycle
    on BUMPED → Antic
  }

  var Antic Y: float = 0
  var Self X: float = 0
  var Antic: gameObject = Lost Lace Ground Tendril/Lost_Lace_Spot_Antic
  var Tendrils: gameObject = Lost Lace Ground Tendril/Lost_Lace_tendril_set
  var tendril antic: gameObject = Lost Lace Ground Tendril/tendril antic
}
