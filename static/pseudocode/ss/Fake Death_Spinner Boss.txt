# Fake Death
# 1 location(s):
#   Belltown_Shrine / Black Thread States Thread Only Variant/Normal World/Boss Scene/Spinner Boss

fsm Fake Death {
  start Init

  state Init {
    GetFsmInt(gameObject=Self, fsmName="Control", variableName="P3 HP", storeValue=var "P3 HP", everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on TOOK DAMAGE → Check
  }

  state Check {
    CompareHP(enemy=var "Self", integer2=1, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    CompareHP(enemy=var "Self", integer2=var "P3 HP", equal=(none), lessThan=→"DEATH", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on DEATH → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FAKE DEATH", delay=0, everyFrame=false)
  }

  var P3 HP: int = 0
  var Self: gameObject = Black Thread States Thread Only Variant/Normal World/Boss Scene/Spinner Boss
}
