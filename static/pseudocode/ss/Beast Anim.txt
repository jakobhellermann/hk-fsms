# Beast Anim
# 1 location(s):
#   Bone_05_boss / Boss Scene/Thick Silk Vines Beast/Beast

fsm Beast Anim {
  start Init
  on FREE → Free  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Thin 1", storeResult=var "Thin 1")
    FindChild(gameObject=Self, childName="Thin 2", storeResult=var "Thin 2")
    FindChild(gameObject=Self, childName="Thin 3", storeResult=var "Thin 3")
    FindChild(gameObject=Self, childName="Thin 4", storeResult=var "Thin 4")
    FindChild(gameObject=Self, childName="Slam Effect", storeResult=var "Slam Effect")
    FindChild(gameObject=Self, childName="Roar Point", storeResult=var "Roar Point")
    FindChild(gameObject=Self, childName="Pt LandBell", storeResult=var "Pt LandBell")
    FindChild(gameObject=Self, childName="Pt LandDust", storeResult=var "Pt LandDust")
    FindChild(gameObject=Self, childName="Pt ThreadRoar", storeResult=var "Pt ThreadRoar")
    on FINISHED → Idle
  }

  state Idle {
    WaitRandom(timeMin=0.5, timeMax=1.5, finishEvent=→"STRUGGLE", realTime=false)
    FloatAdd(floatVariable=var "Sfx Timer", add=-1, everyFrame=true, perSecond=true)
    on STRUGGLE → Struggle
  }

  state Struggle {
    Tk2dPlayFrame(gameObject=var "Thin 1", frame=0)
    Tk2dPlayFrame(gameObject=var "Thin 2", frame=0)
    Tk2dPlayFrame(gameObject=var "Thin 3", frame=0)
    Tk2dPlayFrame(gameObject=var "Thin 4", frame=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Strung Up New", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventToRegister(eventName="STRUGGLE")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Sfx?
  }

  state Free {
    SetPosition(gameObject=Self, vector=(unset), x=4.29, y=-2.33, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    PlayAudioEvent(audioClip=bellbeast_movement_grunt_1 (sfxstatic_assets_areabellway.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=<null>)
    ActivateAllChildren(gameObject=var "Self", activate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Intro Land")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Intro Land", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Intro CamLock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Land
  }

  state Land {
    SendEventByName(eventTarget=GameObject(var "Gate 1"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gate 2"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    Wait(time=1.25, finishEvent=→"NEXT", realTime=false)
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=bell_beast_land_from_jump (sfxstatic_assets_areabell.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on NEXT → Roar
  }

  state Roar {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Roar")
    PlayAudioEvent(audioClip=bellbeast_roar (sfxstatic_assets_areabellway.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=<null>)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=false, bossTitle="BELLBEAST")
    PlayParticleEmitter(gameObject=var "Pt DropBells", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Pt ThreadRoar", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt ThreadRoar", emit=0, resetIfPlaying=false)
    StartRoarEmitter(spawnPoint=var "Roar Point", delay=0, stunHero=true, roarBurst=false, isSmall=false, noVisualEffect=false, forceThroughBind=false, stopOnExit=false)
    on FINISHED → Roar End
  }

  state Roar End {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Intro Submerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=bell_beast_dive_under (sfxstatic_assets_areabell.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    ActivateGameObject(gameObject=var "Intro CamLock", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Boss Camlock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Boss Camlock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt LandBell", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt LandDust", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt LandBell", emit=0, resetIfPlaying=false)
    StopRoarEmitter(delay=0)
    on FINISHED → Battle Start
  }

  state Battle Start {
    SetPlayerDataBool(boolName="encounteredBellBeast", value=true)
    SetMeshRenderer(gameObject=Self, active=false)
    SendEventByName(eventTarget=GameObject(var "Boss"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    ApplyMusicCue(musicCue=BellBattle (audiocuesdynamic_assets_areabell.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.25)
  }

  state Sfx? {
    FloatCompare(float1=var "Sfx Timer", float2=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Sfx Timer", floatValue=0.5, everyFrame=false)
    PlayAudioEventRandom(audioClips=[bell_beast_trapped_distressed_1 (sfxstatic_assets_areabone.bundle), bell_beast_trapped_distressed_2 (sfxstatic_assets_areabone.bundle), bell_beast_trapped_distressed_3 (sfxstatic_assets_areabone.bundle), bell_beast_trapped_distressed_4 (sfxstatic_assets_areabone.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Anim End
  }

  state Anim End {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Boss: gameObject = Boss Scene/Bone Beast
  var Boss Camlock: gameObject = Boss Scene/CamLock Battle
  var Gate 1: gameObject = Boss Scene/Threaded Bell Gate
  var Gate 2: gameObject = Boss Scene/Threaded Bell Gate (1)
  var Intro CamLock: gameObject = Boss Scene/CamLock Intro
  var Pt DropBells: gameObject = Boss Scene/Pt DropBells Long

  var Sfx Timer: float = 0
  var Pt LandBell: gameObject = <null>
  var Pt LandDust: gameObject = <null>
  var Pt ThreadRoar: gameObject = <null>
  var Roar Emitter: gameObject = <null>
  var Roar Point: gameObject = <null>
  var Self: gameObject = <null>
  var Slam Effect: gameObject = <null>
  var Thin 1: gameObject = <null>
  var Thin 2: gameObject = <null>
  var Thin 3: gameObject = <null>
  var Thin 4: gameObject = <null>
}
