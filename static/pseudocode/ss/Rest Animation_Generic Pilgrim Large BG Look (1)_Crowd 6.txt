# Rest Animation
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/Pilgrim Crowds/Crowds/Crowd 6/Generic Pilgrim Large BG Look (1)

fsm Rest Animation {
  start Idle
  on DEACTIVATED → Idle  // from any state
  on DEACTIVATED_INSTANT → Idle  // from any state

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Black Thread States/Normal World/Pilgrim Crowds/Crowds/Crowd 6/Generic Pilgrim Large BG Look (1)@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Rest From Left
    on RIGHT → Rest From Right
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "Should Face Left"], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    on FINISHED → Idle
  }

  state Rest From Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest Start Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))  // disabled
    on FINISHED → Rest Idle
  }

  state Rest Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    SetRandomAudioClipFromTable(Table=pilgrim_weary_idle_loop (sfxdynamic_assets_areabellareaboneareagreymoorareamossareawilds.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0.5)
    on CANCEL → Wake Up
  }

  state Rest From Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest Start Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))  // disabled
    on FINISHED → Rest Idle
  }

  state Wake Up {
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Pilgrim_weary_female_notice (sfxdynamic_assets_areabellareaboneareagreymoorareamossareawilds.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="ShouldFaceLeft", parameters=[], storeResult=var "Should Face Left", EveryFrame=false)
    BoolTest(boolVariable=var "Should Face Left", isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    on LEFT → Rest to Left
    on RIGHT → Rest to Right
  }

  state Rest to Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest End Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Rest to Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Rest End Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  var Resting: bool = false
  var Should Face Left: bool = false
  var Audio Loop Voice: gameObject = Black Thread States/Normal World/Pilgrim Crowds/Crowds/Crowd 6/Generic Pilgrim Large BG Look (1)/Audio Loop Voice
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
