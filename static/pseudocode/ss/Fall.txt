# Fall
# 7 location(s):
#   Abyss_01 / Abyss Crawler (2)
#   Abyss_02 / Abyss Crawler (2)
#   Abyss_02b / Abyss Crawler (1)
#   Abyss_03 / Abyss Crawler
#   Abyss_05 / Abyss Crawler (2)
#   Abyss_05 / Abyss Crawler (4)
#   Abyss_13 / Abyss Crawler

fsm Fall {
  start Check

  state Check {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true)
    FloatTestToBool(float1=var "Rotation", float2=0, tolerance=0.1, equalBool=var "Ground Crawler", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=var "Grounded", GroundedEvent=(none), NotGroundedEvent=(none), EveryFrame=true)
    BoolTestMulti(boolVariables=[var "Ground Crawler", var "Grounded"], boolStates=[true, false], trueEvent=→"FALL", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on FALL → Fall
  }

  state Fall {
    StopCrawler(Target=Self, WaitForTurn=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    DecelerateXY(gameObject=Self, decelerationX=0.98, decelerationY=(unset), brakeOnExit=false)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    on LAND → Land
  }

  state Land {
    PlayRandomAudioClipTable(Table=corpse_land_med (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Restart Crawler
  }

  state Restart Crawler {
    StartCrawler(Target=Self, ScheduleTurn=false)
    on FINISHED → Check
  }

  var Rotation: float = 0
  var Ground Crawler: bool = false
  var Grounded: bool = false
}
