# Control
# 1 location(s):
#   Coral_33 / Black Thread States/Normal World/Dice Game Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Dice Game")
    FindNamedChild(gameObject=Self, storeResult=var "Dice Game Corpse")
    GetPlayerDataInt(intName="dicePilgrimState", storeValue=var "Dice Pilgrim State")
    IntSwitch(intVariable=var "Dice Pilgrim State", compareTo=[0, 1, 2], sendEvent=[→"FINISHED", →"DEAD", →"DEPARTED"], everyFrame=false)
    on DEPARTED → Departed
    on DEAD → Dead
    on FINISHED → Active
  }

  state Departed {
    ActivateGameObject(gameObject=var "Dice Game", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dice Game Corpse", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Dead {
    ActivateGameObject(gameObject=var "Dice Game", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dice Game Corpse", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Active {
    ActivateGameObject(gameObject=var "Dice Game", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Dice Game Corpse", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Dice Pilgrim State: int = 0
  var Pilgrim Defeated: bool = false
  var Visited Citadel: bool = false
  var Dice Game: gameObject = <null>
  var Dice Game Corpse: gameObject = <null>
}
