# FSM
# 2 location(s):
#   Weave_03 / Boss Scene/Stals B
#   Weave_03 / Boss Scene/Stals

// uses template: moss_mother_double_stal_control
fsm FSM {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "Mossbone Stalactite L")
    FindNamedChild(gameObject=Self, storeResult=var "Mossbone Stalactite R")
    FindNamedChild(gameObject=Self, storeResult=var "Mossbone Stalactite ML")
    FindNamedChild(gameObject=Self, storeResult=var "Mossbone Stalactite MR")
    on FINISHED → Idle
  }

  state Idle {
    on FALL → Type
  }

  state Type {
    SendRandomEventV4(events=[→"TRIPLE", →"QUAD"], weights=[1, 1], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on TRIPLE → Triple
    on QUAD → Quad
  }

  state Quad {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FALL", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Triple {
    GetRandomChild(gameObject=Self, storeResult=var "Omitted Stal")
    SetParent(gameObject=var "Omitted Stal", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FALL", delay=0, everyFrame=false)
    SetParent(gameObject=var "Omitted Stal", parent=var "Self", resetLocalPosition=false, resetLocalRotation=false)
    Wait(time=0.8, finishEvent=(none), realTime=false)
    on FINISHED → Delayed Drop
  }

  state Delayed Drop {
    SendEventByName(eventTarget=GameObject(var "Omitted Stal"), sendEvent="FALL", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Mossbone Stalactite L: gameObject = <null>
  var Mossbone Stalactite R: gameObject = <null>
  var Mossbone Stalactite ML: gameObject = <null>
  var Mossbone Stalactite MR: gameObject = <null>
  var Omitted Stal: gameObject = <null>
  var Self: gameObject = <null>
}
