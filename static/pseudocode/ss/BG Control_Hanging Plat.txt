  start Quick Close
  on ACTIVATE → Activated  // from any state

  state Quick Close {

  }

  }

    SendMessageV2(gameObject=var "Send Play Message To", delivery=SendMessage, options=1, functionCall=Play(), everyFrame=false)
    on FINISHED → Activated
  }

  }

  var Start Closed: bool = false
  var Send Play Message To: gameObject = <null>

  var Camera: gameObject = <null>
  var Dust: gameObject = <null>
  var Fallen Plat: gameObject = Hanging Plat/Activated Plat
  var Hanging Plat: gameObject = <null>
  var Raise Dust: gameObject = <null>
  var Self: gameObject = <null>
  var Slam Effect: gameObject = <null>
