# Control
# 1 location(s):
#   Room_CrowCourt_02 / Battle Scene/Wave 5/Wave Starter

fsm Control {
  start Idle

  state Idle {
    GetParent(gameObject=Self, storeResult=var "Wave")
    on BATTLE START → Crowd Roar
  }

  state Start {
    SendEventByName(eventTarget=GameObject(var "Wave"), sendEvent="DO BATTLE START", delay=0, everyFrame=false)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.25)
    DestroySelf(detachChildren=false)
  }

  state Crowd Roar {
    SendEventToRegister(eventName="CROWD ROAR")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0.2)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Crowd Idle
  }

  state Crowd Idle {
    SendEventToRegister(eventName="CROWD IDLE")
    Wait(time=1.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start
  }

  var Encountered Crow Court: bool = false
  var Wave: gameObject = <null>
}
