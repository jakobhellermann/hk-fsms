# Control
# 2 location(s):
#   Shadow_02 / Swamp Goomba (2)
#   Shadow_20 / Swamp Goomba (2)

fsm Control {
  start Init
  on DROWN → Drown  // from any state
  on EXTRACT → Extract  // from any state

  state Crawl {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.5, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SING → Sing
  }

  state Sing {
    StopCrawler(Target=Self, WaitForTurn=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.7, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=<null>, noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Start Crawl
  }

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")  // disabled
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wake Range?
  }

  state Start Crawl {
    StartCrawler(Target=Self, ScheduleTurn=false)
    on FINISHED → Crawl
  }

  state Wake Range? {
    FindNamedChild(gameObject=Self, storeResult=var "Wake Range")
    GameObjectIsNull(gameObject=var "Wake Range", isNull=→"WAKE", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Wake Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on WAKE → Start Crawl
    on TOOK DAMAGE → Start Crawl
  }

  state Drown {
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air 1")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Wait(time=0.75, finishEvent=(none), realTime=false)
    RecordJournalKill(Record=Swamp Goomba (dataassets_assets_assets/dataassets/enemyjournal/journalrecords.bundle))
    on FINISHED → Drown End
  }

  state Drown End {
    DestroySelf(detachChildren=false)
  }

  state Extract {
    EnemyDeathEffectsRegular+SimulateDeath(target=Self)
    SetPlayerDataBool(boolName="hasKilled", value=true)
    StopCrawler(Target=Self, WaitForTurn=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air 1")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Bum Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Rotation", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Offset", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Rotation", multiplyBy=-90, everyFrame=false)
    FloatMultiply(floatVariable=var "Offset", multiplyBy=0.25, everyFrame=false)
    FindChild(gameObject=var "Hero", childName="Tool Effects/Extract Point", storeResult=var "Extract Point")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Extract Point", xOffset=var "Offset", yOffset=0.2, zOffset=(unset), overrideZ=0.003, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=FlashingSwampExtract())
    on EXTRACT FINISH → Extract Kill
  }

  state Extract Kill {
    ActivateGameObject(gameObject=var "Swamp Extract Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelFlash())
  }

  var Offset: float = 0
  var Rotation: float = 0
  var Z Pos: float = 0
  var Bum Collider: gameObject = Swamp Goomba (2)/Bum Collider
  var Extract Point: gameObject = <null>
  var Wake Range: gameObject = <null>
  var Swamp Extract Burst: gameObject = Swamp Goomba (2)/Swamp Extract Burst
  var Audio Loop Voice: gameObject = Swamp Goomba (2)/Audio Loop Voice
}
