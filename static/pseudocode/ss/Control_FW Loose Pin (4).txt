fsm Control {
  start Init
  on STUN CANCEL → Stun Cancel  // from any state
  on BOOM → Fall  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Loose Pins Folder")
    RandomFloat(min=-0.001, max=0.001, storeResult=var "Z Pos")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    RandomFloat(min=0.01, max=0.015, storeResult=var "Z Pos")
    on FINISHED → Idle
  }

  state Idle {
    Tk2dSpriteTweenColor(gameObject=Self, color=(0.5372549, 0.5372549, 0.5372549, 1), duration=1)
    on PULL → Pull Raise
    on CAM SHAKE → Dislodge Chance
  }

  state Pull Raise {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StopBounce())
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    DecelerateV2(gameObject=Self, deceleration=0.92, brakeOnExit=false)
    RandomFloat(min=10, max=20, storeResult=var "Speed")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Speed", everyFrame=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation Start", space=0, everyFrame=false)
    RandomFloat(min=-15, max=15, storeResult=var "Offset")
    FloatOperator(float1=var "Rotation Start", float2=var "Offset", operation=Add, storeResult=var "Rotation End", everyFrame=false)
    AnimateRotationTo(gameObject=Self, fromValue=var "Rotation Start", toValue=var "Rotation End", worldSpace=false, negativeSpace=false, time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pull To
  }

  state Pull To {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    GetPosition(gameObject=var "Projectile Target", vector=var "Target Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Target Pos", vector3Value=(unset), x=(unset), y=(unset), z=var "Z Pos", everyFrame=false)
    GetRotation(gameObject=var "Projectile Target", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tween Rotation", space=0, everyFrame=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Pin Rotation", space=0, everyFrame=false)
    AnimatePositionTo(gameObject=Self, toValue=var "Target Pos", localSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=2, reverse=false, finishEvent=(none), realTime=false)
    AnimateRotationTo(gameObject=Self, fromValue=var "Pin Rotation", toValue=var "Tween Rotation", worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=2, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.65, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fire Ready
  }

  state Fire Ready {
    SetMeshRenderer(gameObject=Self, active=false)
    on THUNK → Thunk
  }

  state Thunk {
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetMeshRenderer(gameObject=Self, active=true)
    SetParent(gameObject=Self, parent=var "Loose Pins Folder", resetLocalPosition=false, resetLocalRotation=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendRandomEventV4(events=[→"STICK", →"FALL"], weights=[0.5, 0.5], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on STICK → Idle
    on FALL → Fall
  }

  state Fall {
    Wait(time=0.25, finishEvent=(none), realTime=false)  // disabled
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartBounce())
    on FINISHED → Idle
  }

  state Stun Cancel {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetMeshRenderer(gameObject=Self, active=true)
    SetParent(gameObject=Self, parent=var "Loose Pins Folder", resetLocalPosition=false, resetLocalRotation=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartBounce())
    on FINISHED → Idle
  }

  state Dislodge Chance {
    SendRandomEvent(events=[→"FINISHED", →"FALL"], weights=[0.9, 0.1], delay=0)
    on FINISHED → Idle
    on FALL → Jostle
  }

  state Jostle {
    RandomFloat(min=-360, max=360, storeResult=var "Angular Velocity")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=var "Angular Velocity", everyFrame=false)
    on FINISHED → Fall
  }

  var Speed: float = 0
  var Rotation Start: float = 0
  var Offset: float = 0
  var Rotation End: float = 0
  var Pin Rotation: float = 0
  var Tween Rotation: float = 0
  var Z Pos: float = 0
  var Angle: float = 0
  var Angular Velocity: float = 0
  var Target Pos: vector3 = (0, 0, 0)
  var Projectile Target: gameObject = <null>
  var Loose Pins Folder: gameObject = <null>
}
