# Control
# 1 location(s):
#   Bellway_Centipede_Arena / Boss Scene/BodyWall L

fsm Control {
  start Init
  on CENTIPEDE EXPLODING → Exploding  // from any state
  on WALL EXPLODE → Wall Explode  // from any state

  state Init {
    SetPosition(gameObject=Self, vector=(unset), x=var "Tight X", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Dormant
  }

  state Dormant {
    ActivateAllChildren(gameObject=var "Self", activate=false)
    on APPEAR → Appear Start
  }

  state Appear Start {
    ActivateAllChildren(gameObject=var "Self", activate=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "Tight X", y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Segs"), sendEvent="MOVE", delay=0, everyFrame=false)
    SetPosition(gameObject=var "Segs", vector=(unset), x=(unset), y=-30, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt Dig Bot", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Top Delay", emit=0, resetIfPlaying=false)
    on FINISHED → Appear Up
  }

  state Appear Up {
    Translate(gameObject=var "Segs", vector=(unset), x=(unset), y=30, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckYPosition(gameObject=var "Segs", compareTo=0, compareToOffset=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true, space=1, activeBool=(unset))
    on FINISHED → Appear Loop
    on APPEAR END → Appear End
  }

  state Appear Loop {
    SetPosition(gameObject=var "Segs", vector=(unset), x=(unset), y=-6.89, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Appear Up
  }

  state Appear End {
    SetPosition(gameObject=var "Segs", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Segs"), sendEvent="IDLE", delay=0, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt Dig Bot")
    StopParticleEmitter(gameObject=var "Pt Top Delay")
    on WIDE → Wide
    on COIL REVERSE → Coil Move R
  }

  state Wide {
    SendEventByName(eventTarget=GameObject(var "Segs"), sendEvent="MOVE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SHIFT", delay=0, everyFrame=false)
    EasePosX(GameObject=Self, toValue=var "Wide X", localSpace=false, time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)  // disabled
    Translate(gameObject=var "Segs", vector=(unset), x=(unset), y=15, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckYPosition(gameObject=var "Segs", compareTo=0, compareToOffset=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true, space=1, activeBool=(unset))
    PlayParticleEmitterInState(gameObject=var "Pt Dig Top")
    PlayParticleEmitterInState(gameObject=var "Pt Dig Bot")
    on SHIFT END → Wide Shift End
    on FINISHED → Appear Loop 2
  }

  state Idle {
    on COIL START → Coil Move
    on COIL REVERSE → Coil Move R
  }

  state Exploding {
  }

  state Wide Shift End {
    SetPosition(gameObject=var "Segs", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Segs"), sendEvent="IDLE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Appear Loop 2 {
    SetPosition(gameObject=var "Segs", vector=(unset), x=(unset), y=-6.89, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Wide
  }

  state Coil Move {
    Translate(gameObject=var "Segs", vector=(unset), x=(unset), y=15, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckYPosition(gameObject=var "Segs", compareTo=0, compareToOffset=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true, space=1, activeBool=(unset))
    PlayParticleEmitterInState(gameObject=var "Pt Dig Top")
    PlayParticleEmitterInState(gameObject=var "Pt Dig Bot")
    on FINISHED → Appear Loop 3
    on COIL END → Idle
  }

  state Appear Loop 3 {
    SetPosition(gameObject=var "Segs", vector=(unset), x=(unset), y=-6.89, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Coil Move
  }

  state Wall Explode {
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Hero Blocker", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Segs"), sendEvent="EXPLODE", delay=0, everyFrame=false)
  }

  state Coil Move R {
    Translate(gameObject=var "Segs", vector=(unset), x=(unset), y=-30, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    CheckYPosition(gameObject=var "Segs", compareTo=0, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true, space=1, activeBool=(unset))
    PlayParticleEmitterInState(gameObject=var "Pt Dig Top")
    PlayParticleEmitterInState(gameObject=var "Pt Dig Bot")
    on FINISHED → Appear Loop R
    on COIL END → Idle
  }

  state Appear Loop R {
    SetPosition(gameObject=var "Segs", vector=(unset), x=(unset), y=6.89, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Coil Move R
  }

  var Wide X: float = 121.5
  var Tight X: float = 126.5
  var Tiny X: float = 0
  var Explode Pause: float = 0

  var Segs Pos Y: float = 0
  var Time: float = 0
  var Damager: gameObject = Boss Scene/BodyWall L/Damager
  var Hero Blocker: gameObject = Boss Scene/BodyWall L/Hero Blocker
  var Pt Dig Bot: gameObject = Boss Scene/BodyWall L/Pt Dig Bot
  var Pt Dig Top: gameObject = Boss Scene/BodyWall L/Pt Dig Top
  var Pt Top Delay: gameObject = Boss Scene/BodyWall L/Pt Dig Top Delay
  var Segs: gameObject = Boss Scene/BodyWall L/Segs
  var Self: gameObject = Boss Scene/BodyWall L
}
