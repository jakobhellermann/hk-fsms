# Control
# 1 location(s):
#   Song_Tower_01 / State Control/song_tower_right_gate

fsm Control {
  start Init

  state Init {
    PlayerDataBoolTest(boolName="laceTowerDoorOpened", isTrue=→"OPENED", isFalse=(none))
    PlayerDataBoolTest(boolName="defeatedLaceTower", isTrue=→"DEFEATED LACE", isFalse=(none))
    on OPENED → Opened
    on DEFEATED LACE → Defeated Lace
  }

  state Opened {
    CallMethodProper(gameObject=Self, behaviour="Gate", methodName="Opened", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  state Defeated Lace {
    FindNamedChild(gameObject=Self, storeResult=var "Open Trigger")
    Trigger2dEvent(gameObject=var "Open Trigger", trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Open
  }

  state Open {
    SetPlayerDataBool(boolName="laceTowerDoorOpened", value=true)
    CallMethodProper(gameObject=Self, behaviour="Gate", methodName="Open", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  var Open Trigger: gameObject = <null>
}
