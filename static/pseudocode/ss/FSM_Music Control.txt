# FSM
# 1 location(s):
#   Slab_22 / Music Control

fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    CheckIfCrestEquipped(Crest=Cloakless (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"CLOAKLESS", falseEvent=→"HAS CLOAK", storeValue=(unset))
    on HAS CLOAK → Has Cloak
    on CLOAKLESS → Cloakless
  }

  state Has Cloak {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
  }

  state Cloakless {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
  }
}
