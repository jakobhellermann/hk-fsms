# FSM
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Festival_Feast/Flea Brew Scene/Flea Festival Brew Pot/Refill Inspect

fsm FSM {
  start Idle

  state Idle {
    on INTERACT → Brew?
  }

  state Brew? {
    GetToolEquipInfo(Tool=Flea Brew (dataassets_assets_assets/dataassets/tools/toolitems.bundle), StoreIsEquipped=(unset), StoreIsUnlocked=var "Brew unlocked", StoreAmountLeft=(unset), StoreMaxAmount=(unset), SomeLeftEquippedEvent=(none), NoneLeftEquippedEvent=(none), EveryFrame=false)
    BoolTest(boolVariable=var "Brew unlocked", isTrue=(none), isFalse=→"BREW", everyFrame=false)
    on BREW → Take?
    on FINISHED → Brew Full?
  }

  sequence state End Dialogue {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Idle
  }

  state Brew Full Dlg {
    RunDialogue(Sheet="Caravan", Key="FESTIVAL_POT_REFILL_BREW_FULL", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  state Brew Full? {
    GetProperty(targetProperty=ToolItemStatesLiquid.IsFull on Flea Brew (dataassets_assets_assets/dataassets/tools/toolitems.bundle), everyFrame=false)
    BoolTest(boolVariable=var "Chooser", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Refill?
    on TRUE → Brew Inspect
  }

  state Refill? {
    GetLocalisedString(Cell={Sheet="Caravan", Key="FESTIVAL_POT_INSPECT"}, StoreString=var "Inspect Text")
    GetLocalisedString(Cell={Sheet="Caravan", Key="BREW_REFILL_PROMPT"}, StoreString=var "Text")
    FormatString(format="{0}\\n(1}", variables=[(unset var), (unset var)], storeResult=var "Text", everyFrame=false)  // disabled
    DialogueYesNoV3(Text=var "Text", TranslationSheet="", TranslationKey="", UseCurrency=false, CurrencyCost=0, CurrencyType=None, ConsumeCurrency=false, WillGetItem=<null>, YesEvent=→"TRUE", NoEvent=→"FALSE", ReturnHUDAfter=false)
    on FALSE → End Dialogue
    on TRUE → Refill.
  }

  state Refill. {
    SetToolUnlocked(Tool=Flea Brew (dataassets_assets_assets/dataassets/tools/toolitems.bundle), WaitForTutorialMsgEnd=true)
    on FINISHED → End Dialogue
  }

  state Give Brew {
    SetToolUnlocked(Tool=Flea Brew (dataassets_assets_assets/dataassets/tools/toolitems.bundle), WaitForTutorialMsgEnd=true)
    on FINISHED → End Dialogue
  }

  state Take? {
    GetLocalisedString(Cell={Sheet="Caravan", Key="FESTIVAL_POT_INSPECT"}, StoreString=var "Inspect Text")
    GetLocalisedString(Cell={Sheet="Caravan", Key="FESTIVAL_POT_TAKE_BREW"}, StoreString=var "Text")
    FormatString(format="{0}\\n(1}", variables=[(unset var), (unset var)], storeResult=var "Text", everyFrame=false)
    DialogueYesNoV3(Text=var "Text", TranslationSheet="", TranslationKey="", UseCurrency=false, CurrencyCost=0, CurrencyType=None, ConsumeCurrency=false, WillGetItem=<null>, YesEvent=→"TRUE", NoEvent=→"FALSE", ReturnHUDAfter=false)
    on FALSE → End Dialogue
    on TRUE → Give Brew
  }

  state Brew Inspect {
    RunDialogue(Sheet="Caravan", Key="FESTIVAL_POT_INSPECT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=Top, OffsetY=0, Target=Self)
    on CONVO_END → End Dialogue
  }

  var Area Meet Cell: string = ""
  var Area Repeat Cell: string = ""

  var Refill Cost: int = 0
  var Activated: bool = false
  var Brew unlocked: bool = false
  var Chooser: bool = false
  var Text: string = ""
  var Inspect Text: string = ""
  var Track Trigger Exit: gameObject = <null>
}
