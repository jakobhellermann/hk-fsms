# Fade Control
# 6 location(s):
#   Ant_19 / Crest Get Shrine/Crest Rune
#   Chapel_Wanderer / Group (1)/Crest Get Shrine/Crest Rune
#   Greymoor_20c / Crest Get Shrine/Crest Rune
#   Tut_04 / States/Intro Scene/Snail Shamans Set/Crest Get Shrine/Crest Rune
#   Tut_05 / Crest Get Shrine/Crest Rune
#   Under_20 / Crest Get Shrine/Crest Rune

fsm Fade Control {
  start Init
  on STAB FLASH FINAL → Stab Flash Final  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Crest Get Shrine")
    FindNamedChild(gameObject=var "Crest Get Shrine", storeResult=var "Crest Rune Needolin Range")
    on FINISHED → Idle
  }

  state Idle {
    CheckTrackTriggerCount(target=var "Crest Rune Needolin Range", count=1, test=MoreThanOrEqual, everyFrame=true, successEvent=→"RANGE IN")
    on STAB FLASH → Stab Flash 1
    on RANGE IN → Needolin Range
  }

  state Stab Flash 1 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=0.05)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stab Flash 2
  }

  state Stab Flash 2 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0.75)
    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Needolin Range
    on STAB FLASH → Stab Flash 1
  }

  state Stab Flash Final {
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=0.05)
  }

  state Needolin Range {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=2, MaxReactDelay=2, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckTrackTriggerCount(target=var "Crest Rune Needolin Range", count=0, test=LessThanOrEqual, everyFrame=true, successEvent=→"RANGE OUT")
    on RANGE OUT → Idle
    on STAB FLASH → Stab Flash 1
    on SING → Sing Up
  }

  state Sing Up {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"SING END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=Self, ToAlpha=0.25, FadeTime=2)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on SING END → Stab Flash 2
    on FINISHED → Sing Down
  }

  state Sing Down {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"SING END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=2)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on SING END → Stab Flash 2
    on FINISHED → Sing Up
  }

  var Crest Get Shrine: gameObject = <null>
  var Crest Rune Needolin Range: gameObject = <null>
}
