# Hero Fling Out
# 2 location(s):
#   Bellway_Aqueduct / Bone Beast NPC
#   Shellwood_19 / Bone Beast NPC

fsm Hero Fling Out {
  start Idle

  state Idle {
    GetHero(storeResult=var "Hero")
    on POSITION → Position
  }

  state Hero Fing Out {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    SetPosition2D(GameObject=var "Hero", Vector=var "Fling Start Pos", X=0, Y=0, Space=0, EveryFrame=false)
    CalculateProjectileVelocity(Target=var "Hero", TargetPosition=var "Hero Pos", FireAngle=var "Hero Enter Angle", ImpulseVelocity=var "Fling Velocity", EveryFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=var "Fling Velocity", x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Fling Anim 1
  }

  state Land {
    SetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=(unset), z=var "Hero Z", space=0, everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControlToIdle", parameters=[], storeResult=(unset var), EveryFrame=false)
    PlayVibrationV3(vibrationDataAsset=quick_travel_soft_land_vibration (vibrationstatic_assets_vibrationdataasset.bundle), motors=None, loopTime=(unset), loopAuto=false, tag="", stopOnStateExit=false)
    on FINISHED → End
  }

  state End {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    GetFsmGameObject(gameObject=Self, fsmName="Interaction", variableName="Temp Warmth Region", storeValue=var "Temp Warmth Region", everyFrame=false)
    ActivateGameObject(gameObject=var "Temp Warmth Region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End Quest?
  }

  state Fling Anim 1 {
    CheckIsCharacterGrounded(Target=var "Hero", RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    SelectRandomString(strings=["DownSpikeBounce 1", "DownSpikeBounce 2"], weights=[1, 1], storeString=var "Bounce Anim")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=(unset), z=var "Hero Z", space=0, everyFrame=false)
    SetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=(unset), z=0.007, space=0, everyFrame=true, lateUpdate=true)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName=var "Bounce Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    GetPlayerDataBool(boolName="hasDash", storeValue=var "Has Dash")
    ListenForDashV2(EventTarget=Self, WasPressed=→"DASH", WasReleased=(none), IsPressed=(none), IsNotPressed=(none), DelayBeforeActive=0, IsActive=var "Has Dash")
    GetPlayerDataBool(boolName="hasHarpoonDash", storeValue=var "Has Harpoon Dash")
    ListenForSuperdashV3(Target=Self, WasPressed=→"HARPOON", WasReleased=(none), IsPressed=(none), IsNotPressed=(none), isPressedBool=false, ActiveBool=var "Has Harpoon Dash")
    on FINISHED → Fling Anim Loop
    on DASH → Dash Queued
    on LAND → Land
    on HARPOON → Harpoon Queued
  }

  state Fling Anim Loop {
    SetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=(unset), z=var "Hero Z", space=0, everyFrame=false, lateUpdate=false)
    CheckIsCharacterGrounded(Target=var "Hero", RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Airborne")
    Tk2dPlayFrame(gameObject=var "Hero", frame=9)
    ListenForDashV2(EventTarget=Self, WasPressed=→"DASH", WasReleased=(none), IsPressed=(none), IsNotPressed=(none), DelayBeforeActive=0, IsActive=var "Has Dash")
    GetPlayerDataBool(boolName="hasHarpoonDash", storeValue=var "Has Harpoon Dash")
    ListenForSuperdashV3(Target=Self, WasPressed=→"HARPOON", WasReleased=(none), IsPressed=(none), IsNotPressed=(none), isPressedBool=false, ActiveBool=var "Has Harpoon Dash")
    on DASH → Dash Cancel
    on LAND → Land
    on HARPOON → Harpoon Cancel
  }

  state Position {
    SetMeshRenderer(gameObject=var "Hero", active=false)
    GetPosition2D(gameObject=Bone Beast NPC/Travel Interaction/Fling Out Land Pos, vector=var "Hero Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    GetPosition2D(gameObject=Bone Beast NPC/Hero Fling Start, vector=var "Fling Start Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    SetPosition2D(GameObject=var "Hero", Vector=var "Fling Start Pos", X=0, Y=0, Space=0, EveryFrame=false)
    TryPositionColliderOnGround(Target=var "Hero", GroundPos=(unset), MaxCorrection=0, ClampOnFailure=false, SuccessEvent=(none), FailedEvent=(none))
    CallMethodProper(gameObject=var "MainCamera", behaviour="CameraController", methodName="PositionToHeroInstant", parameters=[false], storeResult=(unset var), EveryFrame=false)
    on FLING → Hero Fing Out
  }

  state Dash Queued {
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Vel Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Vel Y", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    Tk2dWatchAnimationEvents(gameObject=var "Hero", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Dash Cancel
  }

  state Dash Cancel {
    SetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=(unset), z=var "Hero Z", space=0, everyFrame=false, lateUpdate=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="SetStartWithDashKeepFacing", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → End
  }

  sequence state End Quest? {
    PlayerDataVariableTest(VariableName="UnlockedFastTravelTeleport", ExpectedValue=true, IsExpectedEvent=(none), IsNotExpectedEvent=→"FINISHED")
    CheckQuestState(NotTrackedEvent=→"FINISHED", TrackedEvent=(none), CompletedEvent=→"FINISHED", Quest=Bellbeast Rescue (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    AddHeroInputBlocker(Blocker=Self)
    Wait(time=1, finishEvent=(none), realTime=false)
    RemoveHeroInputBlocker(Blocker=Self)
    TryEndQuestV2(ConsumeCurrency=false, FailEvent=→"FINISHED", SuccessEvent=→"FINISHED", Quest=Bellbeast Rescue (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    on FINISHED → Idle
  }

  state Harpoon Cancel {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=0, functionCall=FlipSprite())
    SendEventByName(eventTarget=GameObjectFSM(var "Hero", fsm="Harpoon Dash"), sendEvent="DO MOVE FIXED", delay=0, everyFrame=false)
    on FINISHED → End
  }

  state Harpoon Queued {
    GetVelocity2d(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Vel Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Hero Vel Y", float2=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    Tk2dWatchAnimationEvents(gameObject=var "Hero", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Harpoon Cancel
  }

  var Hero Enter Angle: float = 57.5

  var Hero Z: float = 0
  var Hero Vel Y: float = 0
  var Has Dash: bool = false
  var Has Harpoon Dash: bool = false
  var Bounce Anim: string = ""
  var Fling Start Pos: vector2 = (0, 0)
  var Fling Velocity: vector2 = (0, 0)
  var Hero Pos: vector2 = (0, 0)
  var Hero: gameObject = <null>
  var Temp Warmth Region: gameObject = <null>
}
