# FSM
# 1 location(s):
#   Cinematic_Stag_travel / Screen Fader

fsm FSM {
  start Clear Fader

  state Clear Fader {
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 1), duration=0)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    on DISABLE → Full Fader
    on VIDEO LOADED → Fade In
  }

  state Idle {
    on DISABLE → Full Fader
  }

  state Full Fader {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
  }

  state Fade In {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=0.7)
    on FINISHED → Idle
  }
}
