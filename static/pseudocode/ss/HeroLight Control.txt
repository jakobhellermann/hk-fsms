# HeroLight Control
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/HeroLight

fsm HeroLight Control {
  start Init
  on DEATH → Death  // from any state
  on DAMAGE → Damage  // from any state

  state Init {
    SetSpriteRenderer(gameObject=Self, active=true)
    on FINISHED → Idle
  }

  state Idle {
  }

  state Damage {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Damage Scale", time=0.25, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Damage Pause
  }

  state Damage Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recover
  }

  state Recover {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Idle Scale", time=0.7, delay=0, speed=(unset), easeType=1, loopType=0, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=true)
    on FINISHED → Idle
  }

  state Death {
    iTweenScaleTo(gameObject=Self, id="", transformScale=var "", vectorScale=var "Damage Scale", time=2, delay=0, speed=(unset), easeType=21, loopType=0, startEvent=(none), finishEvent=(none), realTime=false, stopOnExit=true, loopDontFinish=true)
  }

  var Damage Scale: vector3 = (1.5, 1.5, 1.5)
  var Idle Scale: vector3 = (3, 3, 3)
}
