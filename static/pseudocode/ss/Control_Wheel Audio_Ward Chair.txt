fsm Control {
  start Init
  on WHEEL START → Play Audio  // from any state
  on WHEEL FINISH → Finish  // from any state

  state Init {
    GetAudioVolume(gameObject=Self, storeVolume=var "Play Volume", everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Always Check Velocity", isTrue=→"ALWAYS CHECK", isFalse=(none), everyFrame=false)
    on WHEEL START → Play Audio
    on ALWAYS CHECK → Idle Checking
  }

  state Play Audio {
    AudioPlaySimple(gameObject=Self, volume=var "Play Volume", oneShotClip=<null>)
    GetVelocity2d(gameObject=var "Parent", vector=(unset), x=var "Parent Velocity X", y=(unset), space=0, everyFrame=true)
    SetFloatSign(floatValue=var "Parent Velocity X", setPositive=true, everyFrame=true)
    FloatCompare(float1=var "Parent Velocity X", float2=0.5, tolerance=0, equal=(none), lessThan=→"PAUSE", greaterThan=(none), everyFrame=true)
    on PAUSE → Pause Audio
  }

  state Pause Audio {
    AudioStopV2(gameObject=Self, fadeTime=0.3, cancelOnEarlyExit=false)
    GetVelocity2d(gameObject=var "Parent", vector=(unset), x=var "Parent Velocity X", y=(unset), space=0, everyFrame=true)
    SetFloatSign(floatValue=var "Parent Velocity X", setPositive=true, everyFrame=true)
    FloatCompare(float1=var "Parent Velocity X", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"PLAY", everyFrame=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on PLAY → Play Audio
    on FINISHED → Idle
  }

  state Finish {
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
  }

  state Idle Checking {
    GetVelocity2d(gameObject=var "Parent", vector=(unset), x=var "Parent Velocity X", y=(unset), space=0, everyFrame=true)
    SetFloatSign(floatValue=var "Parent Velocity X", setPositive=true, everyFrame=true)
    FloatCompare(float1=var "Parent Velocity X", float2=1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"WHEEL START", everyFrame=true)
    on WHEEL START → Play Audio
  }

  var Always Check Velocity: bool = false

  var Parent Velocity X: float = 0
  var Play Volume: float = 0
  var Parent: gameObject = <null>
}
