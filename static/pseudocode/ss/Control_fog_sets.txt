# Control
# 1 location(s):
#   Dust_Maze_Last_Hall / fog_sets

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Check
  }

  state Check {
    GetEntryGateName(StoreValue=var "Gate")
    PlayerDataVariableTest(VariableName="PreviousMazeScene", ExpectedValue="Dust_09", IsExpectedEvent=→"CANCEL", IsNotExpectedEvent=(none))
    StringCompare(stringVariable=var "Gate", compareTo="left1", equalEvent=→"LEFT", notEqualEvent=→"RIGHT", storeResult=(unset), everyFrame=false)
    on CANCEL → Check Flipped
    on LEFT → Entered From L
    on RIGHT → Entered From R
  }

  state Entered From L {
    ActivateGameObject(gameObject=var "Entered From L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Entered From R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Entered From R {
    ActivateGameObject(gameObject=var "Entered From L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Entered From R", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Check Flipped {
    StringCompare(stringVariable=var "Gate", compareTo="left1", equalEvent=→"RIGHT", notEqualEvent=→"LEFT", storeResult=(unset), everyFrame=false)
    on LEFT → Entered From L
    on RIGHT → Entered From R
  }

  var Gate: string = ""
  var Entered From L: gameObject = fog_sets/travelling right group
  var Entered From R: gameObject = fog_sets/travelling left group
}
