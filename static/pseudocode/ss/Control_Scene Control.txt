fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindNamedChild(gameObject=Self, storeResult=var "Slab Jailer Scene")
    ActivateGameObject(gameObject=var "Slab Jailer Scene", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataVariableTest(VariableName="MapperLeftShellwood", ExpectedValue=true, IsExpectedEvent=(none), IsNotExpectedEvent=→"MAPPER")
    PlayerDataVariableTest(VariableName="slab_cloak_battle_completed", ExpectedValue=true, IsExpectedEvent=→"MAPPER", IsNotExpectedEvent=(none))
    PlayerDataVariableTest(VariableName="blackThreadWorld", ExpectedValue=true, IsExpectedEvent=→"MAPPER", IsNotExpectedEvent=(none))
    PlayerDataVariableTest(VariableName="slabFlyInShellwood16", ExpectedValue=true, IsExpectedEvent=→"SLAB FLY", IsNotExpectedEvent=→"EMPTY")
    on MAPPER → Mapper
    on EMPTY → Empty
    on SLAB FLY → Empty
  }

  state Mapper {
  }

  state Empty {
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Seen Empty Room
  }

  state Set Seen Empty Room {
    SetPlayerDataBool(boolName="seenEmptyShellwood16", value=true)
  }

  var Slab Jailer Scene: gameObject = <null>
}
