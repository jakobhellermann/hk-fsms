# sherma_anim
# 2 location(s):
#   Bone_05_boss / Sherma Bell Beast NPC
#   Bone_East_10 / Black Thread States Thread Only Variant/Normal World/Pilgrim NPC Scene/2 Brolly Revisit/Sherma Docks NPC

// uses template: sherma_anim
fsm sherma_anim {
  start Init
  on EXCITED → Excite Antic  // from any state
  on GO IDLE → Idle  // from any state
  on EXCITED LOOK → Excite Look Start  // from any state
  on EXCITED STARTLE → Excited Startle  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Chiming", isTrue=→"CHIME", isFalse=(none), everyFrame=true)
    BoolTest(boolVariable=var "Excited", isTrue=→"EXCITED", isFalse=(none), everyFrame=true)
    on CHIME → Chime Anim
  }

  state Chime Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Chime", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Chime Tone
    on CHIME END → Chime Anim Wait
  }

  state Chime Repeat {
    on FINISHED → Chime Anim
    on END → Idle
    on CHIME END → Idle
  }

  state Idle {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="READY", delay=0, everyFrame=false)
    on CHIME → Chime Anim
  }

  state Chime Tone {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor ShortRange (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClips=[sherma_bell_ring_single_1 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_2 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_3 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_4 (sfxstatic_assets_sherma.bundle)], weights=[1, 1, 1, 1], pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Chime Idle
    on CHIME END → Chime Anim Wait
  }

  state Chime Anim Wait {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Chime Idle {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on FINISHED → Chime Repeat
    on CHIME END → Idle
  }

  state Excite Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Excite Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Excited Anim
  }

  state Excited Anim {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Excited", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Excited Tone
    on EXCITED END → Excite Anim Wait
  }

  state Excited Tone {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor ShortRange (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClips=[sherma_bell_ring_single_1 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_2 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_3 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_4 (sfxstatic_assets_sherma.bundle)], weights=[1, 1, 1, 1], pitchMin=0.9, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Excited Anim
    on EXCITED END → Excite Anim Wait
  }

  state Excite Anim Wait {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Excited Recover
  }

  state Excited Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Excite Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Idle
  }

  state Excite Look Start {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ExcitedLook")
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Excite Look
  }

  state Excite Look {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Excited Tone 2
  }

  state Excited Tone 2 {
    AudioPlayerOneShot(audioPlayer=Audio Player Actor ShortRange (audioobjects_assets_all.bundle), spawnPoint=var "Self", audioClips=[sherma_bell_ring_single_1 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_2 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_3 (sfxstatic_assets_sherma.bundle), sherma_bell_ring_single_4 (sfxstatic_assets_sherma.bundle)], weights=[1, 1, 1, 1], pitchMin=0.9, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Excite Look
  }

  state Excited Startle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Excite Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Excited Anim
  }

  var Chiming: bool = false
  var Excited: bool = true

  var Cam Distance: float = 0
  var Loop Player: gameObject = <null>
  var Self: gameObject = <null>
}
