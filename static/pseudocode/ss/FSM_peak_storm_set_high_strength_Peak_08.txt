# FSM
# 1 location(s):
#   Peak_08 / peak_storm_set_high_strength

// uses template: follow_camera_bound
fsm FSM {
  start Pause

  state Follow {
    SetGameObject(variable=var "Camera", gameObject=var "MainCamera", everyFrame=false)
    GetVector2XY(vector2Variable=var "Offset", storeX=var "Offset X", storeY=var "Offset Y", everyFrame=false)
    GetPosition(gameObject=var "Camera", vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Pos X", add=var "Offset X", everyFrame=true, perSecond=false)
    FloatAdd(floatVariable=var "Pos Y", add=var "Offset Y", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Pos X", minValue=var "X Min", maxValue=var "X Max", everyFrame=true)
    FloatClamp(floatVariable=var "Pos Y", minValue=var "Y Min", maxValue=var "Y Max", everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=true, lateUpdate=false)
  }

  state Pause {
    BoolTest(boolVariable=var "Self Space", isTrue=→"SELF", isFalse=(none), everyFrame=false)
    on FINISHED → Follow
    on SELF → Follow Self
  }

  state Follow Self {
    SetGameObject(variable=var "Camera", gameObject=var "MainCamera", everyFrame=false)
    GetVector2XY(vector2Variable=var "Offset", storeX=var "Offset X", storeY=var "Offset Y", everyFrame=false)
    GetPosition(gameObject=var "Camera", vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=true)
    FloatAdd(floatVariable=var "Pos X", add=var "Offset X", everyFrame=true, perSecond=false)
    FloatAdd(floatVariable=var "Pos Y", add=var "Offset Y", everyFrame=true, perSecond=false)
    FloatClamp(floatVariable=var "Pos X", minValue=var "X Min", maxValue=var "X Max", everyFrame=true)
    FloatClamp(floatVariable=var "Pos Y", minValue=var "Y Min", maxValue=var "Y Max", everyFrame=true)
    SetPosition(gameObject=Self, vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=1, everyFrame=true, lateUpdate=false)
  }

  var X Max: float = 244.8
  var X Min: float = 98.39
  var Y Max: float = 2000
  var Y Min: float = 0
  var Self Space: bool = false
  var Offset: vector2 = (0, 0)

  var Offset X: float = 0
  var Offset Y: float = 0
  var Pos X: float = 0
  var Pos Y: float = 0
  var Camera: gameObject = <null>
}
