fsm Motion {
  start Init
  on RESTART → Init  // from any state
  on STOP → Stop  // from any state

  state Init {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → L
    on R → R
  }

  state L {
    RandomFloat(min=1, max=1.2, storeResult=var "Time")
    RandomFloat(min=-0.5, max=-0.6, storeResult=var "X Target")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Start", y=(unset), z=(unset), space=1, everyFrame=false)
    EaseFloat(fromValue=var "X Start", toValue=var "X Target", floatVariable=var "X Tween", time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Tween", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToIdle", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    on FINISHED → R
  }

  state R {
    RandomFloat(min=1, max=1.2, storeResult=var "Time")
    RandomFloat(min=0.5, max=0.6, storeResult=var "X Target")
    GetPosition(gameObject=Self, vector=(unset), x=var "X Start", y=(unset), z=(unset), space=1, everyFrame=false)
    EaseFloat(fromValue=var "X Start", toValue=var "X Target", floatVariable=var "X Tween", time=var "Time", speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Tween", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToIdle", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    on FINISHED → L
  }

  state Stop {
    GetPosition(gameObject=Self, vector=(unset), x=var "X Start", y=(unset), z=(unset), space=1, everyFrame=false)
    EaseFloat(fromValue=var "X Start", toValue=0, floatVariable=var "X Tween", time=0.15, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "X Tween", y=(unset), z=(unset), space=1, everyFrame=true, lateUpdate=false)
  }

  var Time: float = 0
  var Pos: float = 0
  var Current: float = 0
  var X Target: float = 0
  var X Start: float = 0
  var X Tween: float = 0
}
