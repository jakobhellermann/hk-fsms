// uses template: sway_plat_dropper
fsm Detect Landing {
  start Init
  on READY → Idle  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetGrandparent(gameObject=Self, storeResult=var "Grandparent")
    FindChild(gameObject=var "Parent", childName="Plat Dust", storeResult=var "Plat Dust")
    FindChild(gameObject=var "Parent", childName="Water Drips", storeResult=var "Water Drips")
    FindChild(gameObject=var "Parent", childName="Art Parent", storeResult=var "Art Parent")
    FindChild(gameObject=var "Parent", childName="Damager", storeResult=var "Damager")
    FindChild(gameObject=var "Art Parent", childName="Art", storeResult=var "Art")
    FindChild(gameObject=var "Art", childName="plat", storeResult=var "Plat Art")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Slam Effect")
    FindNamedChild(gameObject=var "Art", storeResult=var "chain")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Pt Trail")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Break Effects")
    GetPosition(gameObject=var "Plat Art", vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    GetPosition(gameObject=var "Art Parent", vector=var "Art World Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Fall Cage Setup
  }

  state Idle {
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Player(9), sendEvent=→"COLLIDE", storeCollider=var "Collider", storeForce=(unset), contactPoint=(unset), contactNormal=var "Normal")
    SetPosition(gameObject=var "Plat Art", vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Plat Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=1, everyFrame=false, lateUpdate=false)
    on COLLIDE → Check
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink Up 1
  }

  state Check {
    GetVector3XYZ(vector3Variable=var "Normal", storeX=(unset), storeY=var "Normal Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=-1, tolerance=0.01, equal=→"LAND", lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on LAND → Effects?
  }

  state Down {
    PlayAudioEvent(audioClip=cage_droppable_land_on (sfxstatic_assets_areadustareadustmazeareaslabareaswamp.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Slam Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventByName(eventTarget=GameObject(var "Art"), sendEvent="LAND", delay=0, everyFrame=false)
    SetPosition(gameObject=var "Art Parent", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art Parent", shiftBy=(0, -0.05, 0), time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break Antic
  }

  state Tink L {
    SendEventByNameV3(eventTarget=GameObject(var "Shake Notify"), sendEvent="SHAKE", delay=0, everyFrame=false, activeBool=(unset))
    SendEventByName(eventTarget=GameObject(var "Art"), sendEvent="TINK LEFT", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Idle
  }

  state Tink R {
    SendEventByNameV3(eventTarget=GameObject(var "Shake Notify"), sendEvent="SHAKE", delay=0, everyFrame=false, activeBool=(unset))
    SendEventByName(eventTarget=GameObject(var "Art"), sendEvent="TINK RIGHT", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Idle
  }

  state Tink Up 1 {
    SendEventByNameV3(eventTarget=GameObject(var "Shake Notify"), sendEvent="SHAKE", delay=0, everyFrame=false, activeBool=(unset))
    SetPosition(gameObject=var "Art Parent", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art Parent", shiftBy=(0, 0.1, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Up 2
  }

  state Tink Up 2 {
    AnimatePositionBy(gameObject=var "Art Parent", shiftBy=(0, -0.1, 0), time=0.1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tink Up End
  }

  state Tink Up End {
    SetPosition(gameObject=var "Art Parent", vector=(unset), x=(unset), y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  state Effects? {
    CheckCurrentMapZoneEnum(MapZone=RED_CORAL_GORGE, trueEvent=→"CORAL", falseEvent=→"FINISHED", storeValue=(unset))
    on CORAL → Coral Effects
    on FINISHED → Reg Effects
  }

  state Reg Effects {
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Down
  }

  state Coral Effects {
    PlayParticleEmitter(gameObject=var "Water Drips", emit=0, resetIfPlaying=false)
    on FINISHED → Down
  }

  state Break Antic {
    CallMethodProper(gameObject=var "Parent", behaviour="JitterSelf", methodName="StartJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Break {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    PlayAudioEvent(audioClip=cage_droppable_drop (sfxstatic_assets_areadustmaze.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CallMethodProper(gameObject=var "Parent", behaviour="JitterSelf", methodName="StopJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "chain", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AccelerateToY(gameObject=var "Parent", accelerationFactor=5, targetSpeed=-40)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt Trail", emit=0, resetIfPlaying=false)
    RayCast2d(fromGameObject=var "Parent", fromPosition=(unset), direction=(0, -1), space=1, distance=4, minDepth=(unset), maxDepth=(unset), hitEvent=→"SPLASH", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Splash Point", storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=1, layerMask=[Soft Terrain(25)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    CheckYPositionV2(gameObject=var "Parent", compareTo=var "Ground Smash Y", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=var "Below Ground Smash Y", greaterThan=(none), greaterThanBool=(unset), everyFrame=true, space=0, activeBool=(unset))
    BoolAllTrue(boolVariables=[var "Below Ground Smash Y", var "Ground Smasher"], sendEvent=→"SMASH", storeResult=(unset), everyFrame=true)
    on FINISHED → End
    on SPLASH → Splash
    on SMASH → Smash
  }

  state End {
    SetVelocity2d(gameObject=var "Parent", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    SetFsmBool(gameObject=var "Parent", fsmName="Persistence", variableName="Activated", setValue=true, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Splash {
    PlayAudioEvent(audioClip=enemy_splash_in_water (sfxstatic_assets_water.bundle), pitchMin=0.85, pitchMax=0.85, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetVector2XY(vector2Variable=var "Splash Point", storeX=var "Splash X", storeY=var "Splash Y", everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Splash Pos", vector3Value=(unset), x=var "Splash X", y=var "Splash Y", z=0.001, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Splash Effect Big (globalpoolprefabs_assets_all.bundle), spawnPoint=<null>, position=var "Splash Pos", rotation=(unset), storeObject=var "")
    AccelerateToY(gameObject=var "Parent", accelerationFactor=5, targetSpeed=-40)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state Fall Cage Setup {
    BoolTest(boolVariable=var "Ground Smasher", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    FindNamedChild(gameObject=var "Grandparent", storeResult=var "Fall Cage Remnant")
    FindNamedChild(gameObject=var "Grandparent", storeResult=var "Fall Cage Smash Effects")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on FINISHED → Idle
    on ACTIVATED → Activated
  }

  state Activated {
    ActivateGameObject(gameObject=var "Fall Cage Remnant", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Parent", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Smash {
    ActivateGameObject(gameObject=var "Fall Cage Remnant", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fall Cage Smash Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Parent", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Ground Smash Y: float = 0
  var Ground Smasher: bool = false
  var Shake Notify: gameObject = <null>

  var Normal Y: float = 0
  var Splash X: float = 0
  var Splash Y: float = 0
  var Activated: bool = false
  var Below Ground Smash Y: bool = false
  var Splash Point: vector2 = (0, 0)
  var Start Pos: vector3 = (0, 0, 0)
  var Normal: vector3 = (0, 0, 0)
  var Art World Pos: vector3 = (0, 0, 0)
  var Splash Pos: vector3 = (0, 0, 0)
  var Art: gameObject = <null>
  var Art Parent: gameObject = <null>
  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
  var Plat Art: gameObject = <null>
  var Plat Dust: gameObject = <null>
  var Slam Effect: gameObject = <null>
  var Water Drips: gameObject = <null>
  var chain: gameObject = <null>
  var Pt Trail: gameObject = <null>
  var Break Effects: gameObject = <null>
  var Damager: gameObject = <null>
  var Grandparent: gameObject = <null>
  var Fall Cage Remnant: gameObject = <null>
  var Fall Cage Smash Effects: gameObject = <null>
}
