# Death Control
# 1 location(s):
#   Abyss_Cocoon / Boss Control/Lost Lace Boss

fsm Death Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    GetHP(target=Self, storeValue=var "P2 HP")
    GetHP(target=Self, storeValue=var "P3 HP")
    MultiplyIntByFloat(integer=var "P2 HP", multiplyFloat=0.75, storeResult=var "P2 HP", everyFrame=false, forceRoundUp=false)
    MultiplyIntByFloat(integer=var "P3 HP", multiplyFloat=0.5, storeResult=var "P3 HP", everyFrame=false, forceRoundUp=false)
    on FINISHED → P1
  }

  state P1 {
    on TOOK DAMAGE → HP Check 1
  }

  sequence state HP Check 1 {
    BoolTest(boolVariable=var "Prevent Death", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    CompareHP(enemy=var "Self", integer2=var "P3 HP", equal=→"NEXT", lessThan=→"NEXT", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → P1
    on NEXT → Mid Death Stagger
  }

  state Mid Death Stagger {
    ApplyMusicCue(musicCue=None (audiocuesdynamic_assets_shared.bundle), delayTime=0, transitionTime=0)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP", delay=0, everyFrame=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=EndConstrainY())
    SetMeshRenderer(gameObject=Self, active=true)
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Self Y", minValue=6.7, maxValue=100, everyFrame=false)
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=false, lateUpdate=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Stun Air")
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    CreateObject(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=Lace_defeat_02 (sfxstatic_assets_laceboss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SetInvincible(target=Self, Invincible=true, InvincibleFromDirection=0, resetOnStateExit=false)
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    SendMessage(gameObject=Self, delivery=SendMessage, options=0, functionCall=CancelRecoil())
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SendEventToRegister(eventName="STUN EFFECT END")
    SendEventToRegister(eventName="CLEAN UP PROJECTILES")
    on FINISHED → Mid Death Fling
  }

  state Mid Death Fling {
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=false)
    FloatOperator(float1=var "Land Y", float2=0.1, operation=Add, storeResult=var "Min Y", everyFrame=false)
    FloatClamp(floatVariable=var "Self Y", minValue=var "Min Y", maxValue=99999, everyFrame=false)
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    PlayAudioEvent(audioClip=boss_death_pt_3 (sfxstatic_assets_boss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    CreateObject(gameObject=Boss Death Explosion Black (localpoolprefabs_assets_areacloverareamosslaceboss.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    SetVelocityByScale(gameObject=Self, speed=-8, ySpeed=18, everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=1)
    CheckYPosition(gameObject=Self, compareTo=var "Land Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"LAND", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAND → Mid Death Splash
  }

  state Mid Death Splash {
    SendMessage(gameObject=var "HeroLight", delivery=SendMessage, options=0, functionCall=FadeOut())
    PlayAudioEvent(audioClip=big splash (sfxstatic_assets_water.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=Lost_Lace_dive_in (localpoolprefabs_assets_laceboss.bundle), spawnPoint=var "Self", position=(0, 0, -0.001), rotation=(unset), storeObject=var "Dive In Splash")
    SetPosition2d(gameObject=var "Dive In Splash", vector=(unset), x=(unset), y=var "Dive In Splash Y", space=0, everyFrame=false, lateUpdate=false)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    GetPosition2d(gameObject=Self, vector_2d=(unset), x=(unset), y=var "Pos Y", space=1, everyFrame=true)
    FloatClamp(floatVariable=var "Pos Y", minValue=-10, maxValue=100, everyFrame=true)
    SetPosition2d(gameObject=Self, vector=(unset), x=(unset), y=var "Pos Y", space=1, everyFrame=true, lateUpdate=false)
    Wait(time=4, finishEvent=→"NEXT", realTime=false)
    SetPositionToObject2D(gameObject=var "Cross Slash Splash", targetObject=var "Self", xOffset=0, yOffset=0, everyFrame=false)
    SetPosition2d(gameObject=var "Cross Slash Splash", vector=(unset), x=(unset), y=5, space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Cross Slash Splash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on NEXT → P3 Start
  }

  state P3 Start {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TO P3", delay=0, everyFrame=false)
    on P3 STARTED → P3
  }

  state P3 {
    on ZERO HP → Can Die?
  }

  state Allow Death {
    SetSpecialDeath(target=Self, hasSpecialDeath=false)
  }

  state Can Die? {
    BoolTest(boolVariable=var "Prevent Death", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    on FINISHED → Allow Death
    on CANCEL → P3
  }

  state TEST {
    on FINISHED → Allow Death
  }

  var Self Y: float = 0
  var Land Y: float = 6.4
  var Dive In Splash Y: float = 7.5
  var Pos Y: float = 0
  var Min Y: float = 0
  var P2 HP: int = 0
  var P3 HP: int = 0
  var Prevent Death: bool = false
  var Self: gameObject = <null>
  var Dive In Splash: gameObject = <null>
  var HeroLight: gameObject = Boss Control/Lost Lace Boss/HeroLight
  var Cross Slash Splash: gameObject = Boss Control/Lost Lace Boss/Cross Slash Splash
}
