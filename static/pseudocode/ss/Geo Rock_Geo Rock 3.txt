# Geo Rock
# 6 location(s):
#   Aqueduct_08 / Geo Rock 3
#   Bone_01c / Geo Rock 3
#   Bone_04 / Geo Rock 3
#   Shadow_19 / Geo Rock 3
#   Shadow_20 / Geo Rock 3
#   Shellwood_26 / Geo Rock 3

// uses template: geo_rock
fsm Geo Rock {
  start Pause Frame Init

  state Idle {
    SetBoolValue(boolVariable=var "Skip Effects", boolValue=false, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Idle Anim", animationTriggerEvent=(none), animationCompleteEvent=(none))
    RandomFloat(min=1.5, max=4, storeResult=var "Waiter")
    Wait(time=var "Waiter", finishEvent=→"WAIT", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"TAKE DAMAGE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Attack Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on WAIT → Gleam
    on TAKE DAMAGE → Hit
    on HEAVY → Heavy Break
    on HIT SKIP EFFECTS → Hit Skip Effects
  }

  state Gleam {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"TAKE DAMAGE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Gleam Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
    on TAKE DAMAGE → Hit
    on HEAVY → Heavy Break
    on HIT SKIP EFFECTS → Hit Skip Effects
  }

  state Hit {
    FlingObjectsFromGlobalPool(gameObject=Geo Small (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Per Hit", spawnMax=var "Geo Per Hit", speedMin=23, speedMax=35, angleMin=80, angleMax=100, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Geo Med (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), spawnMin=var "Medium Per Hit", spawnMax=var "Medium Per Hit", speedMin=23, speedMax=35, angleMin=80, angleMax=100, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    IntOperator(integer1=var "Hits", integer2=1, operation=Subtract, storeResult=var "Hits", everyFrame=false)
    BoolTest(boolVariable=var "Skip Effects", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Strike Nail R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=8)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"HIT", lessThan=→"HIT", greaterThan=(none), everyFrame=false)
    on FINISHED → Check Direction
    on HIT → Pause Frame
  }

  state Initiate {
    GetOwner(storeGameObject=var "Self")
    IntCompare(integer1=var "Hits", integer2=1, equal=(none), lessThan=→"BROKEN", greaterThan=(none), everyFrame=false)
    GetPosition(gameObject=var "Self", vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    GetRotation(gameObject=var "Self", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    on FINISHED → Idle
    on BROKEN → Broken
  }

  state Check Direction {
    BoolTest(boolVariable=var "Skip Effects", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    PlayRandomAudioClipTable(Table=var "Hit Audio Table", AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    FloatSwitch(floatVariable=var "Attack Direction", lessThan=[45, 135, 225, 360], sendEvent=[→"RIGHT", →"UP", →"LEFT", →"DOWN"], everyFrame=false)
    on RIGHT → Hit Right
    on UP → Hit Down
    on LEFT → Hit Left
    on DOWN → Hit Down
    on FINISHED → Idle
  }

  state Destroy {
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=8)
    AudioPlayRandom(gameObject=var "Self", audioClips=[breakable_wall_hit_1 (sfxstatic_assets_shared.bundle), breakable_wall_hit_2 (sfxstatic_assets_shared.bundle)], weights=[1, 1], pitchMin=1, pitchMax=1)
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Geo Small (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), spawnMin=var "Final Payout", spawnMax=var "Final Payout", speedMin=23, speedMax=30, angleMin=80, angleMax=100, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Geo Med (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), spawnMin=var "Medium On Break", spawnMax=var "Medium On Break", speedMin=23, speedMax=35, angleMin=80, angleMax=100, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Broken Anim")
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med Down R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(-72.5, -180, -180), storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(unset), storeObject=var "")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(true))
    GameObjectCompare(gameObjectVariable=var "Alt Hit Effects", compareTo=<null>, equalEvent=→"FINISHED", notEqualEvent=→"ALT", storeResult=(unset), everyFrame=false)
    on ALT → Alt Hit Effects
    on FINISHED → State 1
  }

  state Hit Right {
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false, limitFps=0)
    FloatCompare(float1=var "Angle", float2=270, tolerance=10, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(0, 90, 270), storeObject=var "")
    GameObjectCompare(gameObjectVariable=var "Alt Hit Effects", compareTo=<null>, equalEvent=(none), notEqualEvent=→"ALT", storeResult=(unset), everyFrame=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"TAKE DAMAGE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Attack Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Return Left
    on ALT → Alt Hit Effects 2
    on TAKE DAMAGE → Hit Loop
    on HEAVY → Heavy Break
  }

  state Hit Left {
    FloatCompare(float1=var "Angle", float2=90, tolerance=10, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(180, 90, 270), storeObject=var "")
    GameObjectCompare(gameObjectVariable=var "Alt Hit Effects", compareTo=<null>, equalEvent=(none), notEqualEvent=→"ALT", storeResult=(unset), everyFrame=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"TAKE DAMAGE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Attack Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Return Left
    on ALT → Alt Hit Effects 2
    on TAKE DAMAGE → Hit Loop
    on HEAVY → Heavy Break
  }

  state Return Left {
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"TAKE DAMAGE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Attack Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Idle
    on TAKE DAMAGE → Hit Loop
    on HEAVY → Heavy Break
  }

  state Hit Down {
    FloatCompare(float1=var "Angle", float2=180, tolerance=10, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    ObjectJitter(gameObject=Self, x=0.05, y=0.05, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=var "Recoil Time", finishEvent=→"FINISHED", realTime=false)
    SpawnObjectFromGlobalPool(gameObject=Dust Hit Med R (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0, 0.1), rotation=(-72.5, -180, -180), storeObject=var "")
    GameObjectCompare(gameObjectVariable=var "Alt Hit Effects", compareTo=<null>, equalEvent=(none), notEqualEvent=→"ALT", storeResult=(unset), everyFrame=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"TAKE DAMAGE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "Damager", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Attack Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Return Left
    on ALT → Alt Hit Effects 2
    on TAKE DAMAGE → Hit Loop
    on HEAVY → Heavy Break
  }

  state Pause Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Destroy
  }

  state Broken {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Broken Anim")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=SetActive(true))
  }

  state Pause Frame Init {
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Idle Anim")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Initiate
  }

  state State 1 {
    SpawnObjectFromGlobalPool(gameObject=Break Rock Effect Small (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
  }

  state Alt Hit Effects {
    SendMessageV2(gameObject=var "Alt Hit Effects", delivery=SendMessage, options=0, functionCall=PlayParticles(), everyFrame=false)
  }

  state Alt Hit Effects 2 {
    SendMessageV2(gameObject=var "Alt Hit Effects", delivery=SendMessage, options=0, functionCall=PlayParticles(), everyFrame=false)
    on FINISHED → Return Left
  }

  state Heavy Break {
    IntOperator(integer1=var "Geo Per Hit", integer2=var "Hits", operation=Multiply, storeResult=var "Geo Per Hit", everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Geo Small (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), spawnMin=var "Geo Per Hit", spawnMax=var "Geo Per Hit", speedMin=23, speedMax=35, angleMin=80, angleMax=100, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    SetIntValue(intVariable=var "Hits", intValue=0, everyFrame=false)
    on FINISHED → Pause Frame
  }

  state Hit Skip Effects {
    SetBoolValue(boolVariable=var "Skip Effects", boolValue=true, everyFrame=false)
    on FINISHED → Hit
  }

  state Hit Loop {
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Hit
  }

  var Final Payout: int = 6
  var Geo Per Hit: int = 3
  var Hits: int = 3
  var Medium On Break: int = 2
  var Medium Per Hit: int = 1
  var Broken Anim: string = "Broken 3"
  var Gleam Anim: string = "Gleam 3"
  var Alt Hit Effects: gameObject = <null>
  var Hit Audio Table: object = rosary_rock_hit (sfxdynamic_assets_shared.bundle)

  var Angle: float = 0
  var Attack Direction: float = 0
  var Recoil Time: float = 0.1
  var Start X: float = 0
  var Start Y: float = 0
  var Waiter: float = 0
  var Skip Effects: bool = false
  var Idle Anim: string = ""
  var Damager: gameObject = <null>
  var Self: gameObject = <null>
}
