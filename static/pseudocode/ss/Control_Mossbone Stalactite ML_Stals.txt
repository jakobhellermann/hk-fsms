# Control
# 1 location(s):
#   Weave_03 / Boss Scene/Stals/Mossbone Stalactite ML

fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Antic Dust", storeResult=var "Antic Dust")
    FindChild(gameObject=Self, childName="Antic Dust 2", storeResult=var "Antic Dust 2")
    FindChild(gameObject=Self, childName="Antic Rocks", storeResult=var "Antic Rocks")
    FindChild(gameObject=Self, childName="Break Rocks", storeResult=var "Break Rocks")
    FindChild(gameObject=Self, childName="Break Rocks 2", storeResult=var "Break Rocks 2")
    FindChild(gameObject=Self, childName="Dust Trail", storeResult=var "Dust Trail")
    FindChild(gameObject=Self, childName="Sprite", storeResult=var "Sprite")
    FindChild(gameObject=Self, childName="Thunk Effect", storeResult=var "Thunk Effect")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=1, everyFrame=false)
    on FINISHED → Dormant
  }

  state Dormant {
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FALL → Position
  }

  state Position {
    RandomFloat(min=var "Min X", max=var "Max X", storeResult=var "Start X")
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Start Y", z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Antic
  }

  state Antic {
    PlayParticleEmitterInState(gameObject=var "Antic Dust")
    PlayParticleEmitterInState(gameObject=var "Antic Rocks")
    PlayParticleEmitterInState(gameObject=var "Antic Dust 2")
    RandomWait(min=0.75, max=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic Pause
  }

  state Fall {
    ActivateGameObject(gameObject=var "Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Crusher", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    PlayParticleEmitterInState(gameObject=var "Dust Trail")
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-25, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Self Y", z=(unset), space=0, everyFrame=true)
    FloatCompare(float1=var "Self Y", float2=var "Bot Y", tolerance=0, equal=→"FINISHED", lessThan=→"LAND", greaterThan=(none), everyFrame=true)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"LAND", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on LAND → Land Pos
    on SLASH → Land
  }

  state Land {
    AudioPlayRandom(gameObject=var "Self", audioClips=[stalactite_rock_impact_1 (sfxstatic_assets_areabone.bundle), stalactite_rock_impact_2 (sfxstatic_assets_areabone.bundle), stalactite_rock_impact_3 (sfxstatic_assets_areabone.bundle)], weights=[1, 1, 1], pitchMin=1, pitchMax=1)
    PlayAudioEvent(audioClip=bone_boulder_ground_impact (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1.25, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayParticleEmitter(gameObject=var "Break Rocks", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Break Rocks 2", emit=0, resetIfPlaying=false)
    on FALL → Position
  }

  state Antic Pause {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fall
  }

  state Land Pos {
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Bot Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Land
  }

  var Bot Y: float = 19
  var Max X: float = -0.9
  var Min X: float = -4.41

  var Self Y: float = 0
  var Start X: float = 0
  var Start Y: float = 0
  var Antic Dust: gameObject = <null>
  var Antic Dust 2: gameObject = <null>
  var Antic Rocks: gameObject = <null>
  var Break Rocks: gameObject = <null>
  var Break Rocks 2: gameObject = <null>
  var Dust Trail: gameObject = <null>
  var Self: gameObject = <null>
  var Sprite: gameObject = <null>
  var Thunk Effect: gameObject = <null>
  var Enemy Crusher: gameObject = Boss Scene/Stals/Mossbone Stalactite ML/Enemy Crusher
}
