# hero_binding_check
# 3 location(s):
#   Bone_East_18b / Boss Scene/Bone Hunter Trapper
#   Coral_39 / Coral Warrior Grey
#   Organ_01 / Boss Scene/Phantom

// uses template: hero_binding_check
fsm hero_binding_check {
  start Init

  state Init {
    AddEventRegister(target=Self, eventName="STARTED BIND")
    on FINISHED → Idle
  }

  state Idle {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Hero Binding", setValue=false, everyFrame=false)
    on STARTED BIND → Exceptions
  }

  state Hero Binding {
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Hero Binding", setValue=true, everyFrame=false)
    Wait(time=var "Awareness Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Exceptions {
    CheckIfCrestEquipped(Crest=Witch (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"CANCEL", falseEvent=(none), storeValue=(unset))
    CheckIfCrestEquipped(Crest=Cursed (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"CANCEL", falseEvent=(none), storeValue=(unset))
    CheckIfCrestEquipped(Crest=Warrior (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=→"CANCEL", falseEvent=(none), storeValue=(unset))
    on CANCEL → Idle
    on FINISHED → Hero Binding
  }

  var Awareness Time: float = 2
}
