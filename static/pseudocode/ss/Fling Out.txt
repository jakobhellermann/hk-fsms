# Fling Out
# 1 location(s):
#   Coral_39 / Coral Warrior Item Spitter/Pit Pickup

fsm Fling Out {
  start Fling

  state Fling {
    SetVelocityAsAngle(gameObject=Self, angle=60, speed=30, everyFrame=true)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rising
  }

  state Rising {
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Velocity Y", float2=0, tolerance=0, equal=(none), lessThan=→"FALLING", greaterThan=(none), everyFrame=false)
    on FALLING → Falling
  }

  state Falling {
    CheckYPosition(gameObject=Self, compareTo=1.5, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"CATCH", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on CATCH → State 1
  }

  state State 1 {
    SetPosition2d(gameObject=Self, vector=(unset), x=6.2, y=2.5, space=0, everyFrame=false, lateUpdate=false)
  }

  var Velocity Y: float = 0
}
