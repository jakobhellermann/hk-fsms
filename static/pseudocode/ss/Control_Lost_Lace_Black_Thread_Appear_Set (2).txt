# Control
# 1 location(s):
#   Abyss_Cocoon / Boss Control/Black Thread Attack (deprecated)/Lost_Lace_Black_Thread_Appear_Set (2)

fsm Control {
  start Init

  state Init {
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Init X", y=(unset), space=1, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    ActivateGameObject(gameObject=var "Thread Parent", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on ATTACK → Attack Pause
  }

  state Attack {
    SetPosition2d(gameObject=Self, vector=(unset), x=var "Init X", y=(unset), space=1, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=Self, translateMin=(-4, 0, 0), translateMax=(4, 0, 0), space=1)
    RandomFloat(min=20, max=-40, storeResult=var "Rotation")
    SetRotation(gameObject=var "Thread Parent", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Thread Parent", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=4.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Attack Pause {
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Attack
  }

  var Init X: float = 0
  var Rotation: float = 0
  var Thread Parent: gameObject = Boss Control/Black Thread Attack (deprecated)/Lost_Lace_Black_Thread_Appear_Set (2)/Thread_Parent
}
