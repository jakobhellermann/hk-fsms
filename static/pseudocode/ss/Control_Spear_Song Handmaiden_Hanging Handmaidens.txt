# Control
# 1 location(s):
#   Hang_14 / Handmaiden Scene/Hanging Handmaidens/Song Handmaiden/Spear

fsm Control {
  start Init

  state Init {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RandomFloat(min=0.003, max=0.004, storeResult=var "Pos Z")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Pos Z", space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Appear Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    FadeColorFader(target=var "Idle Glow", fadeType=UP, useChildren=false)
    on FINISHED → Idle
  }

  state Idle {
    on FIRE → Get Antic Angle
    on HANDMAIDEN KILLED → Deactivate
  }

  state Antic {
    PlayAudioEvent(audioClip=song_handmaiden_projectile_aim (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FloatOperator(float1=var "Angle", float2=180, operation=Add, storeResult=var "Antic Angle", everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Antic Angle", speed=30, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.875, brakeOnExit=false)
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Facing Angle", worldSpace=false, time=0.3, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=0.7, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Pos Z", space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Fire
  }

  state Get Antic Angle {
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0.25, storeAngle=var "Angle", pause=0, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"LEFT", negativeEvent=→"RIGHT", space=0)
    on LEFT → Fire L
    on RIGHT → Fire R
  }

  state Fire {
    PlayAudioEvent(audioClip=song_handmaiden_projectile_fire (sfxstatic_assets_areasong.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=37, everyFrame=false)
    ChaseObjectWisp(gameObject=Self, target=var "Hero", accelerationForce=45, offsetX=0, offsetY=0, speedMax=50, speedMin=37)
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=true)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObjectDelay(gameObject=var "Terrain Detector", activate=true, resetOnExit=false, delay=0.1)
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=→"COLLIDE", storeCollider=var "", storeForce=(unset))
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Pos Z", space=0, everyFrame=false, lateUpdate=false)
    on COLLIDE → Explode
    on DAMAGED HERO → Explode
  }

  state Explode {
    PlayAudioEvent(audioClip=song_handmaiden_projectile_impact (sfxstatic_assets_areasong.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=handmaiden_spear_silk_explosion (localpoolprefabs_assets_areasong.bundle), spawnPoint=var "Explosion Origin", position=(unset), rotation=(unset), storeObject=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spear Break")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Terrain Detector", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1, finishEvent=(none), realTime=false)
    FadeColorFader(target=var "Idle Glow", fadeType=DOWN, useChildren=false)
    on FINISHED → Deactivate
  }

  state Fire L {
    FloatClamp(floatVariable=var "Angle", minValue=135, maxValue=225, everyFrame=false)
    SetFloatValue(floatVariable=var "Facing Angle", floatValue=var "Angle", everyFrame=false)
    ReflectAngle(angle=var "Facing Angle", reflectHorizontally=true, reflectVertically=true, disallowNegative=false, storeResult=var "Facing Angle")
    on FINISHED → Antic Pause
  }

  state Fire R {
    FloatClamp(floatVariable=var "Angle", minValue=315, maxValue=405, everyFrame=false)  // disabled
    ClampAngle(angleVariable=var "Angle", minValue=-45, maxValue=45, everyFrame=false)
    SetFloatValue(floatVariable=var "Facing Angle", floatValue=var "Angle", everyFrame=false)
    on FINISHED → Antic Pause
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Antic Pause {
    ActivateGameObject(gameObject=var "Appear Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.15, finishEvent=(none), realTime=false)
    on FINISHED → Antic
  }

  var Angle: float = 0
  var Facing Angle: float = 0
  var Antic Angle: float = 0
  var Pos Z: float = 0
  var Terrain Detector: gameObject = Handmaiden Scene/Hanging Handmaidens/Song Handmaiden/Spear/Terrain Detector
  var Break Effects: gameObject = Handmaiden Scene/Hanging Handmaidens/Song Handmaiden/Spear/Break Effects
  var Appear Glow: gameObject = Handmaiden Scene/Hanging Handmaidens/Song Handmaiden/Spear/Appear Glow
  var Explosion Origin: gameObject = Handmaiden Scene/Hanging Handmaidens/Song Handmaiden/Spear/Explosion Origin
  var Idle Glow: gameObject = Handmaiden Scene/Hanging Handmaidens/Song Handmaiden/Spear/Idle Glow
}
