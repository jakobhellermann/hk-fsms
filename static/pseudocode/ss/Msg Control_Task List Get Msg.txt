# Msg Control
# 1 location(s):
#   localpoolprefabs_assets_areamoss.bundle / Task List Get Msg

fsm Msg Control {
  start Start

  state Detect {
    ListenForCast(eventTarget=Self, wasPressed=→"PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForJump(eventTarget=Self, wasPressed=→"PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), queueBool=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForAttack(eventTarget=Self, wasPressed=→"PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), delayBeforeActive=0)
    on PRESS → Down
  }

  state Done {
    SendEventToRegister(eventName="GET ITEM MSG END")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Top Up {
    SendEventByName(eventTarget=GameObject(var "BG"), sendEvent="UP", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=var "Audio Player", spawnPoint=var "Hero", audioClip=H179_new_wish_granted redo (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0.5, storePlayer=<null>)
    SendEventByName(eventTarget=GameObject(var "Icon"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name Prefix"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Fleur", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Button", delivery=SendMessage, options=1, functionCall=RefreshButtonIcon())
    ActivateGameObjectDelay(gameObject=var "Icon Flash", activate=true, resetOnExit=false, delay=0.65)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.65, everyFrame=false)
    on FINISHED → Bot Up
  }

  state Bot Up {
    SendEventByName(eventTarget=GameObject(var "Press"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Msg 1"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Msg 2"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Button"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Up
  }

  state Stop Up {
    ActivateGameObject(gameObject=var "Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Down {
    SendEventByName(eventTarget=GameObject(var "Icon"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Fleur"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name Prefix"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Msg 1"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Msg 2"), sendEvent="DOWN", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Stop"), sendEvent="DOWN", delay=0, everyFrame=false)
    PlayAudioEvent(audioClip=button (sfxstatic_assets_shared.bundle), pitchMin=0.95, pitchMax=1.05, volume=0.3, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → BG Down
  }

  state Audio Player Actor {
    SetGameObject(variable=var "Audio Player", gameObject=Audio Player NoDestroy (audioobjects_assets_all.bundle), everyFrame=false)
    on FINISHED → Top Up
  }

  state Start {
    SetPosition(gameObject=Self, vector=(unset), x=0, y=0, z=0, space=0, everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Audio Player Actor
  }

  state BG Down {
    SendEventByName(eventTarget=GameObject(var "BG"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Done
  }

  var Audio Player: gameObject = <null>
  var BG: gameObject = Task List Get Msg/BG
  var Button: gameObject = <null>
  var Fleur: gameObject = Task List Get Msg/Fleur
  var Icon: gameObject = Task List Get Msg/Icon
  var Icon Flash: gameObject = Task List Get Msg/Icon Flash
  var Item Name: gameObject = Task List Get Msg/Item Name
  var Item Name Prefix: gameObject = Task List Get Msg/Item Name Prefix
  var Msg 1: gameObject = Task List Get Msg/Msg 1
  var Msg 2: gameObject = Task List Get Msg/Msg 2
  var Press: gameObject = <null>
  var Stop: gameObject = Task List Get Msg/Stop
}
