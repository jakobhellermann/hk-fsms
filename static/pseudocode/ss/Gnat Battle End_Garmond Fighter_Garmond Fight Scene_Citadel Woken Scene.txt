fsm Gnat Battle End {
  start Idle

  state Idle {
    on VAMPIRE GNAT DEFEATED → Set
  }

  state Set {
    FindGameObject(objectName="", withTag="Boss Corpse", store=var "Target")
    SetFsmBool(gameObject=Self, fsmName="Control", variableName="Boss Defeated", setValue=true, everyFrame=false)
    GetPosition(gameObject=var "Target", vector=(unset), x=var "Target X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Target X", minValue=var "End Min X", maxValue=var "End Max X", everyFrame=false)
    SetParent(gameObject=var "Battle End Marker", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPosition(gameObject=var "Battle End Marker", vector=(unset), x=var "Target X", y=(unset), z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetFsmGameObject(gameObject=Self, fsmName="Control", variableName="Target", setValue=var "Battle End Marker", everyFrame=false)
    on VAMPIRE GNAT END → End Pause
  }

  state End Pause {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate NPC
  }

  state Activate NPC {
    ActivateGameObject(gameObject=var "Victory NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var End Min X: float = 16
  var End Max X: float = 88

  var Target X: float = 0
  var Battle End Marker: gameObject = Citadel Woken Scene/Garmond Fight Scene/Garmond Fighter/Battle End Marker
  var Target: gameObject = <null>
  var Victory NPC: gameObject = Citadel Woken Scene/Garmond Fight Scene/Garmond Fighter/Victory NPC
}
