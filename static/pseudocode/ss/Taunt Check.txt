# Taunt Check
# 1 location(s):
#   Bone_East_10 / Rhino Scene/Gate Smashing

fsm Taunt Check {
  start Idle

  state Idle {
    on DID TAUNT → Check Range
  }

  state Check Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Challenge Range", storeResult=false, sendEvent="", outOfRangeEvent="CANCEL", everyFrame=false)
    on FINISHED → Send Event
    on CANCEL → Idle
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TAUNT REACT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }
}
