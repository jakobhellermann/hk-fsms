# Control
# 2 location(s):
#   Mosstown_02 / Black Thread States Thread Only Variant/Normal World/Reminder Silk Skill
#   Mosstown_02 / Black Thread States Thread Only Variant/Normal World/Reminder Silk Skill (1)

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ACTIVATE → Activated
    on ENTER → Check Silk
  }

  state Activated {
  }

  state Check Silk {
    PlayerDataBoolTest(boolName="hasNeedleThrow", isTrue=(none), isFalse=→"CANCEL")
    GetPlayerDataInt(intName="silk", storeValue=var "Silk")
    IntCompare(integer1=var "Silk", integer2=2, equal=(none), lessThan=(none), greaterThan=→"PLAY REMINDER", everyFrame=false)
    Wait(time=0.5, finishEvent=→"RECHECK", realTime=false)
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"CANCEL", storeCollider=var "")
    on FINISHED → Send
    on CANCEL → Idle
    on RECHECK → Check Silk
    on PLAY REMINDER → Send
  }

  state Send {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventToRegisterDelay(EventName="REMINDER SKILL", delay=0, ExcludeTarget=Self)
  }

  var Silk: int = 0
  var Activated: bool = false
}
