# Control
# 1 location(s):
#   Shellwood_11b_Memory / Boss Scene/Respawn Plat

fsm Control {
  start Init
  on KNIGHT SPIKE HIT → Appear Pause  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Marker")
    FindNamedChild(gameObject=Self, storeResult=var "Plat")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Emerge")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Appear")
    on FINISHED → Dormant
  }

  state Dormant {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Plat", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Is Active", boolValue=false, everyFrame=false)
  }

  state Appear Pause {
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Appear
  }

  state Appear {
    SetBoolValue(boolVariable=var "Is Active", boolValue=true, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Appear", resetTimeIfPlaying=false, stopOnStateExit=true)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Plat", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPosition(gameObject=var "Plat", vector=(unset), x=(unset), y=-4, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Plat", shiftBy=(0, 4, 0), time=0.25, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    Wait(time=2, finishEvent=→"HERO RESPAWNED", realTime=false)
    on HERO RESPAWNED → Disappear Pause
  }

  state Disappear Antic {
    ObjectJitter(gameObject=var "Plat", x=0.1, y=0.1, z=(unset), allowMovement=false, limitFps=0)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Emerge", resetTimeIfPlaying=false, stopOnStateExit=true)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Plat", audioClip=vine_ball_antic_for_emerge (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Disappear
  }

  state Disappear {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    AnimatePositionBy(gameObject=var "Plat", shiftBy=(0, -4, 0), time=0.35, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Plat", audioClip=flower_side_dash_antic_foley (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Dormant
  }

  state Disappear Pause {
    Wait(time=1, finishEvent=(none), realTime=false)
    on FINISHED → Disappear Antic
  }

  var Is Active: bool = false
  var Marker: gameObject = <null>
  var Plat: gameObject = <null>
  var Pt Emerge: gameObject = <null>
  var Pt Appear: gameObject = <null>
}
