# Control
# 1 location(s):
#   Mosstown_02 / Vine Platform/Activated Plat/Dream Gate Set Lock

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Gate Allowed", setValue=false, everyFrame=false)
    on EXIT → Out
  }

  state Out {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    SetFsmBool(gameObject=var "Hero", fsmName="Dream Nail", variableName="Dream Gate Allowed", setValue=true, everyFrame=false)
    on ENTER → In
  }
}
