  start Init

  state Idle {
    ActivateGameObject(gameObject=var "Timer Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    on GAME BEGIN PLAY → Activate Timer
  }

  state Init {
    on FINISHED → Idle
  }

  sequence state Activate Timer {
    Wait(time=2, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Timer Obj", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Timer Obj: gameObject = Caravan_States/Flea Festival/Flea Game - Dodging/Score Timer
}
