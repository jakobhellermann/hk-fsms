# Control
# 1 location(s):
#   Library_02 / GameObject/Upper Scene/steam_vent_single (2)

fsm Control {
  start Init
  on ANTIC → Antic SFX?  // from any state
  on ATTACK → Attack SFX?  // from any state
  on IDLE → Re-Idle  // from any state

  state Init {
    FindNamedChild(gameObject=var "Pt Attack", storeResult=var "Pt Rubble")
    SetCollider(gameObject=var "Damager", active=false, resetOnExit=false)
    SetCollider(gameObject=var "Air Blow Region", active=false, resetOnExit=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false)
    SetBoolValue(boolVariable=var "In Vent Range", boolValue=var "z Start In Vent Range", everyFrame=false)
    FloatCompare(float1=var "Rotation", float2=0, tolerance=0.1, equal=→"FINISHED", lessThan=(none), greaterThan=(none), everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Rubble", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Pusher", active=false, resetOnExit=false)
    FindNamedChild(gameObject=var "Damager", storeResult=var "Pusher")
    on FINISHED → Idle
  }

  state Idle {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP SHAKE", delay=0, everyFrame=false)
    on FINISHED → Antic SFX?
  }

  state Antic {
    PlayParticleEmitterChildren(gameObject=var "Pt Antic", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Buddy"), sendEvent="ANTIC", delay=0, everyFrame=false)
    on FINISHED → Attack SFX?
  }

  state Attack {
    Wait(time=var "Attack Time", finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetCollider(gameObject=var "Damager", active=true, resetOnExit=true)
    SetCollider(gameObject=var "Air Blow Region", active=true, resetOnExit=true)
    SetCollider(gameObject=var "Pusher", active=true, resetOnExit=true)
    ActivateGameObjectDelay(gameObject=var "Damager", activate=true, resetOnExit=false, delay=0.1)
    StopParticleEmitter(gameObject=var "Pt Attack")
    PlayParticleEmitterInState(gameObject=var "Pt Attack")
    SendEventByName(eventTarget=GameObject(var "Buddy"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Re-Idle
  }

  state Re-Idle {
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    SendEventByName(eventTarget=GameObject(var "Buddy"), sendEvent="IDLE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STOP SHAKE", delay=0, everyFrame=false)
    on FINISHED → Antic SFX?
  }

  state Antic SFX? {
    BoolTest(boolVariable=var "In Vent Range", isTrue=→"SFX", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Antic
    on SFX → Antic SFX
  }

  state Antic SFX {
    AudioPlay(gameObject=Self, volume=0.7, oneShotClip=steam_vent_antic (sfxstatic_assets_areadocks.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    on FINISHED → Antic
  }

  state Attack SFX? {
    BoolTest(boolVariable=var "In Vent Range", isTrue=→"SFX", isFalse=→"FINISHED", everyFrame=false)
    on FINISHED → Attack
    on SFX → Attack SFX
  }

  state Attack SFX {
    AudioPlay(gameObject=Self, volume=0.7, oneShotClip=steam_vent_release (sfxstatic_assets_areadocks.bundle), WaitForEndOfClip=false, finishedEvent=(none))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="START SHAKE", delay=0, everyFrame=false)
    on FINISHED → Attack
  }

  var Wait Time: float = 0.75
  var Attack Time: float = 0.5
  var z Start In Vent Range: bool = false
  var Buddy: gameObject = <null>

  var Rotation: float = 0
  var Trigger Count: int = 0
  var In Vent Range: bool = false
  var Pt Antic: gameObject = GameObject/Upper Scene/steam_vent_single (2)/Pt Antic
  var Pt Attack: gameObject = GameObject/Upper Scene/steam_vent_single (2)/Pt Attack
  var Damager: gameObject = GameObject/Upper Scene/steam_vent_single (2)/Damager
  var Pt Rubble: gameObject = <null>
  var Vent Range: gameObject = <null>
  var Air Blow Region: gameObject = GameObject/Upper Scene/steam_vent_single (2)/Damager/Air Blow Region
  var Pusher: gameObject = <null>
}
