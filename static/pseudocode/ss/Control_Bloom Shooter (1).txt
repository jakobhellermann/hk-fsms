# Control
# 2 location(s):
#   Arborium_03 / Bloom Shooter (1)
#   Shellwood_03 / Bloom Shooter (1)

fsm Control {
  start Init

  state Init {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle Mid", space=0, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"L", negativeEvent=→"R", space=0)
    on L → Set L
    on R → Set R
  }

  state Idle {
    CheckIsBlackThreaded(Target=Self, TrueEvent=(none), FalseEvent=(none), StoreValue=var "Is Black Threaded", EveryFrame=false)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=var "In Alert Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    BoolTestMulti(boolVariables=[var "In Alert Range", var "Is Black Threaded"], boolStates=[true, false], trueEvent=→"ATTACK", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on TOOK DAMAGE → Antic
    on ATTACK → Antic
    on SING → Sing
  }

  state Antic {
    CheckIsBlackThreaded(Target=Self, TrueEvent=→"BLACK THREADED", FalseEvent=(none), StoreValue=(unset), EveryFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot Start", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    StopParticleEmitter(gameObject=var "Pt PollenShot")
    PlayParticleEmitter(gameObject=var "Pt PollenShot", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=bloom_shooter_activate (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Shoot Retract 1
    on BLACK THREADED → Void Attack
  }

  state Shoot 1 {
    PlayAudioEvent(audioClip=bloom_shooter_shoot_1 (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Pollen Shot (localpoolprefabs_assets_areaarboriumareashellwoodareaunderstore.bundle), spawnPoint=var "Shoot Point", position=(unset), rotation=(unset), storeObject=var "Shot")
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle Mid", pause=0, everyFrame=false)
    FloatAddRandom(floatVariable=var "Angle Mid", addMin=-5, addMax=5)
    ClampAngle(angleVariable=var "Angle Mid", minValue=var "Angle Low", maxValue=var "Angle High", everyFrame=false)
    SetVelocityAsAngle(gameObject=var "Shot", angle=var "Angle Mid", speed=var "Shot Speed", everyFrame=false)
    Wait(time=var "Shoot Pause", finishEvent=→"NEXT", realTime=false)
    on NEXT → Shoot Retract 2
  }

  state Shoot End {
    Wait(time=1.3, finishEvent=→"FINISHED", realTime=false)
    WaitRandom(timeMin=0.8, timeMax=1, finishEvent=→"FINISHED", realTime=false)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Shoot End")
    on FINISHED → Idle
    on TOOK DAMAGE → Idle
  }

  state Set Angle {
    on FINISHED → Idle
  }

  state Shoot 2 {
    PlayAudioEvent(audioClip=bloom_shooter_shoot_2 (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Pollen Shot (localpoolprefabs_assets_areaarboriumareashellwoodareaunderstore.bundle), spawnPoint=var "Shoot Point", position=(unset), rotation=(unset), storeObject=var "Shot")
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle Mid", pause=0, everyFrame=false)
    FloatAddRandom(floatVariable=var "Angle Mid", addMin=-5, addMax=5)
    ClampAngle(angleVariable=var "Angle Mid", minValue=var "Angle Low", maxValue=var "Angle High", everyFrame=false)
    SetVelocityAsAngle(gameObject=var "Shot", angle=var "Angle Mid", speed=var "Shot Speed", everyFrame=false)
    Wait(time=var "Shoot Pause", finishEvent=→"NEXT", realTime=false)
    on NEXT → Shoot Retract 3
  }

  state Shoot 3 {
    PlayAudioEvent(audioClip=bloom_shooter_shoot_3 (sfxstatic_assets_areashellwood.bundle), pitchMin=0.95, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Spit Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Pollen Shot (localpoolprefabs_assets_areaarboriumareashellwoodareaunderstore.bundle), spawnPoint=var "Shoot Point", position=(unset), rotation=(unset), storeObject=var "Shot")
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle Mid", pause=0, everyFrame=false)
    FloatAddRandom(floatVariable=var "Angle Mid", addMin=-5, addMax=5)
    ClampAngle(angleVariable=var "Angle Mid", minValue=var "Angle Low", maxValue=var "Angle High", everyFrame=false)
    SetVelocityAsAngle(gameObject=var "Shot", angle=var "Angle Mid", speed=var "Shot Speed", everyFrame=false)
    Wait(time=var "Shoot Pause", finishEvent=→"NEXT", realTime=false)
    StopParticleEmitter(gameObject=var "Pt PollenShot")
    on NEXT → Sing?
  }

  state Set L {
    FloatAdd(floatVariable=var "Angle Mid", add=180, everyFrame=false, perSecond=false)
    FloatOperator(float1=var "Angle Mid", float2=45, operation=Add, storeResult=var "Angle High", everyFrame=false)
    FloatOperator(float1=var "Angle Mid", float2=45, operation=Subtract, storeResult=var "Angle Low", everyFrame=false)
    on FINISHED → Set Angle
  }

  state Set R {
    FloatAdd(floatVariable=var "Angle Mid", add=360, everyFrame=false, perSecond=false)
    FloatOperator(float1=var "Angle Mid", float2=45, operation=Add, storeResult=var "Angle High", everyFrame=false)
    FloatOperator(float1=var "Angle Mid", float2=45, operation=Subtract, storeResult=var "Angle Low", everyFrame=false)
    on FINISHED → Set Angle
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=flower_drifter_singing (sfxdynamic_assets_areashellwood.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    PlayParticleEmitterInState(gameObject=var "Pt Sing")
    on CANCEL → Shoot End
    on SING DURATION END → Shoot End
  }

  state Sing? {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=false)
    on SING → Sing
    on FINISHED → Shoot End
  }

  state Shoot Retract 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot", animationTriggerEvent=→"NEXT", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on NEXT → Shoot 1
  }

  state Shoot Retract 2 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot", animationTriggerEvent=→"NEXT", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on NEXT → Shoot 2
  }

  state Shoot Retract 3 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Shoot", animationTriggerEvent=→"NEXT", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on NEXT → Shoot 3
  }

  state Void Attack {
    BlackThreadSetAttackQueued(Target=Self, SetAttackQueued=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SING → Sing
  }

  var Angle High: float = 0
  var Angle Low: float = 0
  var Angle Max: float = 0
  var Angle Mid: float = 0
  var Shot Speed: float = 15
  var Shoot Pause: float = 0.6
  var Is Black Threaded: bool = false
  var In Alert Range: bool = false
  var Pt PollenShot: gameObject = Bloom Shooter (1)/Pt PollenShot
  var Shoot Point: gameObject = Bloom Shooter (1)/Shoot Point
  var Shot: gameObject = <null>
  var Spit Effect: gameObject = Bloom Shooter (1)/Spit Effect
  var Audio Loop Voice: gameObject = Bloom Shooter (1)/Audio Loop Voice
  var Pt Sing: gameObject = Bloom Shooter (1)/Pt Sing
}
