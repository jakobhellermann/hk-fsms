    FindChild(gameObject=Self, childName="boulder/Sprites", storeResult=var "Sprites")
    FindChild(gameObject=Self, childName="boulder/break_effects", storeResult=var "break_effects")
    FindChild(gameObject=Self, childName="Pt Trail", storeResult=var "Trail")
    ActivateGameObject(gameObject=var "Trail", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GetActiveChild(gameObject=var "Sprites", storeResult=var "Sprite Crt")

    SetPolygonCollider(gameObject=var "Sprite Crt", active=false, resetOnExit=false)

  }

  state Idle {

    on CANCEL → Idle Inert
    on DROP → Drop Antic
    on BOULDER HIT → Drop Antic
    on SHAKE → Shake
    on TRAP → Drop Antic
  }

  state Drop Antic {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SetParent(gameObject=var "Pt Idle", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)

    Wait(time=0.8, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=bone_boulder_shake_n_break (sfxstatic_assets_areabone.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    on FINISHED → Drop
  }

  state Drop {
    ActivateGameObject(gameObject=var "Trail", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmittersInChildren(gameObject=var "Pt Antic")
    CallMethodProper(gameObject=var "Sprite Crt", behaviour="NonBouncer", methodName="SetActive", parameters=[false], storeResult=(unset var), EveryFrame=false)
    AccelerateToY(gameObject=Self, accelerationFactor=4, targetSpeed=-40)
    ActivateGameObjectDelay(gameObject=var "Terrain Detector", activate=true, resetOnExit=false, delay=0.2)
    Collision2dEventLayer(collision=OnCollisionEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"LAND", storeCollider=var "", storeForce=(unset), contactPoint=(unset), contactNormal=(unset))

    SetDamageEnemyAmount(target=var "Sprite Crt", damageDealt=80)
    SetPolygonCollider(gameObject=var "Sprite Crt", active=true, resetOnExit=false)
    ReceivedDamage(Target=Self, collideTag="NonBouncer", sendEvent=(none), sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=→"LAND", storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=true, ignoreTraps=true, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=1, storeDirection=-40, storeMagnitudeMultiplier=(unset), firstHitOnly=false)

    BoolAllTrue(boolVariables=[], sendEvent=(none), storeResult=false, everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Default(0), sendEvent=→"RANGE OUT", storeCollider=var "")
    CheckYPosition(gameObject=Self, compareTo=0, compareToOffset=0, tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false, space=0, activeBool=false)
    on LAND → Break
    on SPLASH → Splash
    on LAVA → Lava Splash
    on RANGE OUT → Deactivate
  }

  state Break {
    PlayAudioEvent(audioClip=bone_boulder_ground_impact (sfxstatic_assets_areabone.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=Audio Player Actor Distant V2@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    ActivateGameObject(gameObject=var "Pt Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite Crt", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "break_effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Impact Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)

  }

  state Shake {
    CheckOutOfCamera(gameObject=Self, margin=6, outsideEvent=→"FINISHED", insideEvent=(none), insideBool=false, outsideBool=false, everyFrame=false)
    WaitRandom(timeMin=0.25, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    ObjectJitter(gameObject=var "Sprite Crt", x=0.1, y=0.05, z=(unset), allowMovement=false, limitFps=15)
    Trigger2dEvent(gameObject=var "Alert Range", trigger=0, collideTag="", sendEvent=→"DROP", storeCollider=var "")
    on DROP → Drop Antic
    on BOULDER HIT → Drop Antic
    on FINISHED → Idle
    on TRAP → Drop Antic
  }

  state Splash {
    PlayAudioEvent(audioClip=enemy_splash_in_water (sfxstatic_assets_water.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    on FINISHED → Deactivate
  }

  state Deactivate {

  }

  state Drop Skip {
    SetBoolValue(boolVariable=var "Skip Antic", boolValue=true, everyFrame=false)
    on FINISHED → Drop Antic
  }

  state Idle Inert {
    ActivateGameObject(gameObject=var "Pt Idle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Pt Antic", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on DROP → Drop Skip
  }

  state Lava Splash {
    SpawnObjectFromGlobalPool(gameObject=Enemy Lava Death (localpoolprefabs_assets_areabone.bundle), spawnPoint=var "Sprites", position=(0, -4, 0), rotation=(unset), storeObject=var "")

    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
  }

  var Soft Terrain Below: bool = false

  var Impact Damager: gameObject = Boss Scene/Boulders Init/Bone_Boulder/Impact Damager
