# Control
# 1 location(s):
#   localpoolprefabs_assets_areadocks.bundle / Tar Trail

fsm Control {
  start Active

  state Active {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    WaitRandom(timeMin=1.7, timeMax=1.8, finishEvent=→"NEXT", realTime=false)
    SetMeshRenderer(gameObject=var "Sprite", active=true)
    RandomFloatV2(min=0.6, max=1.4, storeResult=var "Scale", everyFrame=false)
    SetFloatValue(floatVariable=var "X Scale", floatValue=var "Scale", everyFrame=false)
    RandomlyFlipFloat(storeResult=var "X Scale")
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=var "Scale", z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayFrame(gameObject=var "Sprite", frame=0)
    Tk2dPlayAnimation(gameObject=var "Sprite", animLibName="", clipName="Trail")
    on NEXT → End
  }

  state End {
    StopParticleEmitter(gameObject=var "Centre")
    StopParticleEmitter(gameObject=var "Pt Idle")
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Tar Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → State 1
  }

  state Recycle {
    RecycleSelf()
  }

  state State 1 {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=var "Sprite", active=false)
    on FINISHED → Recycle
  }

  var Scale: float = 0
  var X Scale: float = 0
  var Centre: gameObject = Tar Trail/Pt Centre
  var Pt Idle: gameObject = Tar Trail/Pt Idle
  var Sprite: gameObject = Tar Trail/Sprite
}
