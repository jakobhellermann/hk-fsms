    AudioPlaySimple(gameObject=var "Audio Loop LavaFall Mid", volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Loop LavaFall Mid", startVolume=0, endVolume=1, time=0.25)
    on FINISHED → Lava 2
  }

  state Lava 2 {
    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)

    Wait(time=0.35, finishEvent=→"FINISHED", realTime=false)

    SendEventByName(eventTarget=GameObject(var "Heat Glow"), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Lava Fall
  }

    PlayParticleEmitter(gameObject=var "particle_lava_blobs", emit=0, resetIfPlaying=false)

    StopParticleEmitter(gameObject=var "particle_lava_blobs (1)")
    AudioStopV2(gameObject=var "Audio Loop LavaFall Mid", fadeTime=0.5, cancelOnEarlyExit=false)
  }

  state Idle {
    WaitRandom(timeMin=3, timeMax=3.5, finishEvent=→"LAVA START", realTime=false)
    SetFsmBool(gameObject=var "Dock Guard Slasher", fsmName="Control", variableName="Lava Coming", setValue=false, everyFrame=false)
    SetFsmBool(gameObject=var "Dock Guard Thrower", fsmName="Control", variableName="Lava Coming", setValue=false, everyFrame=false)
    AudioStopV2(gameObject=var "Audio Loop LavaFall Mid", fadeTime=0.5, cancelOnEarlyExit=false)

  }

  state Dormant {
    on LAVA START → Idle
  }

  state Lava Fall {

    on FINISHED → Idle
  }

  state Lava Antic2 {
    SetFsmBool(gameObject=var "Dock Guard Slasher", fsmName="Control", variableName="Lava Coming", setValue=true, everyFrame=false)
    SetFsmBool(gameObject=var "Dock Guard Thrower", fsmName="Control", variableName="Lava Coming", setValue=true, everyFrame=false)
