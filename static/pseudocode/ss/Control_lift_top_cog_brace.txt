# Control
# 3 location(s):
#   Organ_01 / lift_top_cog_brace
#   Slab_03 / lift_top_cog_brace
#   Under_02 / lift_top_cog_brace

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    SendMessage(gameObject=var "Cog Large", delivery=SendMessage, options=1, functionCall=StopAtCurrentPoint())
    SendMessage(gameObject=var "Cog Small 1", delivery=SendMessage, options=1, functionCall=StopAtCurrentPoint())
    SendMessage(gameObject=var "Cog Small 2", delivery=SendMessage, options=1, functionCall=StopAtCurrentPoint())
    on MOVING DOWN → Moving Down
    on MOVING UP → Moving Up
  }

  state Moving Down {
    SendMessage(gameObject=var "Cog Large", delivery=SendMessage, options=1, functionCall=StartAnimation())
    SendMessage(gameObject=var "Cog Small 1", delivery=SendMessage, options=1, functionCall=StartAnimation())
    SendMessage(gameObject=var "Cog Small 2", delivery=SendMessage, options=1, functionCall=StartAnimation())
    on MOVE STOP → Idle
  }

  state Moving Up {
    SendMessage(gameObject=var "Cog Large", delivery=SendMessage, options=1, functionCall=StartAnimation(true))  // disabled
    SendMessage(gameObject=var "Cog Small 1", delivery=SendMessage, options=1, functionCall=StartAnimation(true))  // disabled
    SendMessage(gameObject=var "Cog Small 2", delivery=SendMessage, options=1, functionCall=StartAnimation(true))  // disabled
    CallMethodProper(gameObject=var "Cog Large", behaviour="CurveRotationAnimation", methodName="StartAnimation", parameters=[true], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Cog Small 1", behaviour="CurveRotationAnimation", methodName="StartAnimation", parameters=[true], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Cog Small 2", behaviour="CurveRotationAnimation", methodName="StartAnimation", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on MOVE STOP → Idle
  }

  var Cog Large: gameObject = lift_top_cog_brace/bell_bench_cog_01_small
  var Cog Small 1: gameObject = lift_top_cog_brace/bell_bench_cog_01_small (1)
  var Cog Small 2: gameObject = lift_top_cog_brace/bell_bench_cog_01_small (2)
}
