# Control
# 2 location(s):
#   Peak_02 / Weaver Servitor Large/Legs Container/Legs/Head Swiveller
#   Peak_04d / Weaver Servitor Large/Legs Container/Legs/Head Swiveller

fsm Control {
  start Init
  on LOOK STOP → Stop  // from any state
  on AIM START → Aim Pop  // from any state

  state Init {
    on LOOK START → Pop L?
  }

  state Swivel L {
    RandomFloat(min=-10, max=-7, storeResult=var "Target Rotation")
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Target Rotation", worldSpace=false, time=0.3, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pop R?
  }

  state Stop {
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=0, worldSpace=false, time=0.35, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on LOOK START → Swivel L
  }

  state Swivel R {
    RandomFloat(min=7, max=10, storeResult=var "Target Rotation")
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Target Rotation", worldSpace=false, time=0.3, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Pop L?
  }

  state Pop L? {
    SendRandomEventV3(events=[→"SKIP", →"POP"], weights=[0.75, 0.25], trackingInts=[var "Ct Skip", var "Ct Pop"], eventMax=[3, 2], trackingIntsMissed=[var "Ms Skip", var "Ms Pop"], missedMax=[2, 3])
    on SKIP → Swivel L
    on POP → Pop L Anim
  }

  state Pop R? {
    SendRandomEventV3(events=[→"SKIP", →"POP"], weights=[0.75, 0.25], trackingInts=[var "Ct Skip", var "Ct Pop"], eventMax=[3, 2], trackingIntsMissed=[var "Ms Skip", var "Ms Pop"], missedMax=[2, 3])
    on SKIP → Swivel R
    on POP → Pop R Anim
  }

  state Pop R {
    PlayParticleEmitter(gameObject=var "Pt SwivelPop", emit=0, resetIfPlaying=false)
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=10, worldSpace=false, time=0.175, speed=(unset), delay=(unset), easeType=28, reverse=false, finishEvent=→"NEXT", realTime=false)
    PlayAudioEventRandom(audioClips=[weaver_servitor_broken_spark_glitch_1 (sfxstatic_assets_areaweaver.bundle), weaver_servitor_broken_spark_glitch_2 (sfxstatic_assets_areaweaver.bundle), weaver_servitor_broken_spark_glitch_3 (sfxstatic_assets_areaweaver.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on NEXT → Pop L?
  }

  state Pop L {
    PlayAudioEventRandom(audioClips=[weaver_servitor_broken_spark_glitch_1 (sfxstatic_assets_areaweaver.bundle), weaver_servitor_broken_spark_glitch_2 (sfxstatic_assets_areaweaver.bundle), weaver_servitor_broken_spark_glitch_3 (sfxstatic_assets_areaweaver.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitter(gameObject=var "Pt SwivelPop", emit=0, resetIfPlaying=false)
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=-10, worldSpace=false, time=0.175, speed=(unset), delay=(unset), easeType=28, reverse=false, finishEvent=→"NEXT", realTime=false)
    on NEXT → Pop R?
  }

  state Aim Pop {
    RandomFloatEither(value1=-15, value2=15, storeResult=var "Target Rotation")
    PlayParticleEmitter(gameObject=var "Pt SwivelPop", emit=0, resetIfPlaying=false)
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Target Rotation", worldSpace=false, time=0.15, speed=(unset), delay=(unset), easeType=28, reverse=false, finishEvent=→"NEXT", realTime=false)
    on NEXT → Stop
  }

  state Pop R Anim {
    BoolTest(boolVariable=var "Pop Anim", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Head", animLibName="", clipName="Head Spark")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Pop R
  }

  state Pop L Anim {
    BoolTest(boolVariable=var "Pop Anim", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Head", animLibName="", clipName="Head Spark")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SendEvent(eventTarget=Self, sendEvent=→"FINISHED", delay=0, everyFrame=false)
    on FINISHED → Pop L
  }

  var Target Rotation: float = 0
  var Ct Pop: int = 0
  var Ct Skip: int = 0
  var Ms Pop: int = 0
  var Ms Skip: int = 0
  var Pop Anim: bool = false
  var Pt SwivelPop: gameObject = Weaver Servitor Large/Legs Container/Legs/Head Swiveller/Head/Pt SwivelPop
  var Head: gameObject = Weaver Servitor Large/Legs Container/Legs/Head Swiveller/Head
}
