# Control
# 2 location(s):
#   Shadow_01 / Bloat Roach (2)
#   Shadow_02 / Bloat Roach (2)

fsm Control {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    FindAlertRange(target=Self, storeResult=var "Attack Range", childName="Attack Range")
    SetParent(gameObject=var "Attack Range Obj", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=2, speedMax=2, accelerationMin=5, accelerationMax=15, roamingRangeX=0.5, roamingRangeY=1.5, manualStartPos=var "Start Pos")
    CheckAlertRange(alertRange=var "Attack Range", storeResult=var "In Attack Range", InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    SetBoolValueAtTime(BoolVariable=var "Attack Ready", BoolValue=true, Time=0.25, SetOppositeOnStateEntry=true)
    BoolTestMulti(boolVariables=[var "Attack Ready", var "In Attack Range"], boolStates=[true, true], trueEvent=→"ATTACK", falseEvent=(none), storeResult=(unset), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ATTACK → Evade?
    on SING → Sing
  }

  state Belch Antic {
    CheckIsBlackThreaded(Target=Self, TrueEvent=(none), FalseEvent=(none), StoreValue=var "Is Black Threaded", EveryFrame=false)
    BoolTestToGameObject(Test=var "Is Black Threaded", ExpectedValue=true, TrueGameObject=var "Pt Spit Void", FalseGameObject=var "Pt Spit", StoreResult=var "Pt Spit Crt", EveryFrame=false)
    BoolTestToGameObject(Test=var "Is Black Threaded", ExpectedValue=true, TrueGameObject=var "Spit Effect Void", FalseGameObject=var "Spit Effect", StoreResult=var "Spit Effect Crt", EveryFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Belch Antic")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnTo Belch Antic", resetFrame=true, everyFrame=false, pauseBetweenTurns=0.5)
    Wait(time=0.9, finishEvent=→"FINISHED", realTime=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=true)
    PlayAudioEvent(audioClip=bloat_roach_antic (sfxstatic_assets_areaswamp.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioStop(gameObject=Self, fadeTime=0)
    SetRotation(gameObject=var "Pt Spit Crt", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Belch
  }

  state Belch {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Belch")
    ActivateGameObject(gameObject=var "Spit Effect Crt", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Spit Crt", resetTimeIfPlaying=false, stopOnStateExit=true)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.2)
    Wait(time=1.25, finishEvent=→"FINISHED", realTime=false)
    SetVelocityByScale(gameObject=Self, speed=10, ySpeed=3, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    PlayAudioEventRandom(audioClips=[bloat_roach_attack_1 (sfxstatic_assets_areaswamp.bundle), bloat_roach_attack_2 (sfxstatic_assets_areaswamp.bundle), bloat_roach_attack_3 (sfxstatic_assets_areaswamp.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=bloat_roach_attack_burst (sfxstatic_assets_areaswamp.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlaySimple(gameObject=var "Audio SpitLoop", volume=1, oneShotClip=<null>)
    EaseFloat(fromValue=5, toValue=-5, floatVariable=var "Spit Angle", time=1.25, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    SetRotation(gameObject=var "Pt Spit Crt", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Spit Angle", space=1, everyFrame=true, lateUpdate=false)
    on FINISHED → Belch End
  }

  state Belch End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Belch End", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FadeAudio(gameObject=var "Audio SpitLoop", startVolume=1, endVolume=0, time=0.2)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    on FINISHED → Belch Recover
  }

  state Belch Recover {
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=2, speedMax=4, accelerationMin=50, accelerationMax=50, roamingRangeX=0.5, roamingRangeY=1, manualStartPos=var "Start Pos")
    AudioStop(gameObject=var "Audio SpitLoop", fadeTime=0)
    on FINISHED → Idle
  }

  state Sing {
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Sing", singAudioTable=<null>, noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.6, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    IdleBuzzV4(gameObject=Self, waitMin=0.5, waitMax=0.75, speedMax=2, deceleration=0.8, accelerationMin=5, accelerationMax=5, roamingRangeX=0.15, roamingRangeY=0.15, manualStartPos=(unset))
    on CANCEL → Sing End
    on SING DURATION END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStop(gameObject=var "Audio Loop Sing", fadeTime=0.15)
    on FINISHED → Idle
  }

  state Evade? {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Above Range", storeResult=false, sendEvent="", outOfRangeEvent="FINISHED", everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=var "Hero Is Right", leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    ConvertBoolToFloat(boolVariable=var "Hero Is Right", floatVariable=var "Evade Velocity", falseValue=20, trueValue=-20, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "Evade Velocity", y=(unset), everyFrame=false)
    on FINISHED → Belch Antic
  }

  var Evade Velocity: float = 0
  var Spit Angle: float = 0
  var In Attack Range: bool = false
  var Attack Ready: bool = false
  var Hero Is Right: bool = false
  var Is Black Threaded: bool = false
  var Start Pos: vector3 = (0, 0, 0)
  var Attack Range Obj: gameObject = Bloat Roach (2)/Attack Range
  var Audio Loop Sing: gameObject = Bloat Roach (2)/Audio Loop Sing
  var Audio SpitLoop: gameObject = Bloat Roach (2)/Audio SpitLoop
  var Pt Spit: gameObject = Bloat Roach (2)/Pt Spit
  var Pt Spit Crt: gameObject = Bloat Roach (2)/Pt Spit
  var Pt Spit Void: gameObject = Bloat Roach (2)/Pt Spit Void
  var Spit Effect Crt: gameObject = Bloat Roach (2)/Spit Effect
  var Spit Effect Void: gameObject = Bloat Roach (2)/Spit Effect Void
  var Spit Effect: gameObject = Bloat Roach (2)/Spit Effect
  var Attack Range: object = <null>
}
