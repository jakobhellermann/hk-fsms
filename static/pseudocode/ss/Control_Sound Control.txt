# Control
# 1 location(s):
#   Tut_01 / Remasker New thumper control/Sound Control

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Set
  }

  state Set {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Remasker New thumper")
    SendMessage(gameObject=var "Remasker New thumper", delivery=SendMessage, options=1, functionCall=SetPlaySound(false))
    SetAudioVolume(gameObject=var "Remasker New thumper", volume=0, everyFrame=false)
    SetAudioClip(gameObject=var "Remasker New thumper", audioClip=<null>, autoPlay=false, stopOnExit=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
  }

  var Activated: bool = false
  var Parent: gameObject = <null>
  var Remasker New thumper: gameObject = <null>
}
