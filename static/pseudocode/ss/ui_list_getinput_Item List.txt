# ui_list_getinput
# 1 location(s):
#   shopui_assets_all.bundle / Shop Menu/Item List Group/Item List

// uses template: ui_list_getinput
fsm ui_list_getinput {
  start Init
  on UI INACTIVE → Inactive  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Inactive", isTrue=(none), isFalse=→"UI ACTIVE", everyFrame=false)
    on UI ACTIVE → Activate
  }

  state Idle {
    SetFloatValue(floatVariable=var "List Scroll Speed", floatValue=0.15, everyFrame=false)
    CheckStaticBool(variableName="IsUIListInputBlocked", trueEvent=→"CANCEL", falseEvent=(none), EveryFrame=true)
    CheckIsCheatManagerOpen(TrueEvent=→"CANCEL 1", FalseEvent=(none), StoreValue=(unset), EveryFrame=true)
    SetBoolValue(boolVariable=var "Repeating", boolValue=false, everyFrame=false)
    SetIntValue(intVariable=var "Repeats", intValue=0, everyFrame=false)
    ListenForRight(eventTarget=Self, wasPressed=→"RIGHT PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    ListenForLeft(eventTarget=Self, wasPressed=→"LEFT PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    ListenForUp(eventTarget=Self, wasPressed=→"UP PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForDown(eventTarget=Self, wasPressed=→"DOWN PRESS", wasReleased=(none), isPressed=(none), isNotPressed=(none), isPressedBool=(unset), queueBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    ListenForMenuActionsV2(EventTarget=Self, SubmitPressed=→"SELECTION MADE", CancelPressed=→"PRESS CANCEL", ExtraPressed=→"EXTRA PRESS", SuperPressed=(none), IgnoreAttack=var "Ignore Attack")
    ListenForInventory(eventTarget=Self, wasPressed=→"PRESS CANCEL", wasReleased=(none), isPressed=(none), isNotPressed=(none))
    ListenForRsDown(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"RS DOWN PRESS", isNotPressed=(none), isPressedBool=(unset), activeBool=(unset))
    ListenForRsUp(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"RS UP PRESS", isNotPressed=(none), isPressedBool=(unset), activeBool=(unset))
    on UP PRESS → Select Up
    on DOWN PRESS → Select Down
    on LEFT PRESS → Select Left
    on RIGHT PRESS → Select Right
    on SELECTION MADE → Confirm
    on RS UP PRESS → RS Up
    on RS DOWN PRESS → RS Down
    on PRESS CANCEL → Cancel
    on CANCEL → Disabled
    on CANCEL 1 → Disabled 1
    on EXTRA PRESS → Extra
  }

  state Select Up {
    BoolTest(boolVariable=var "Allow Vertical Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)  // disabled
    SetIntValue(intVariable=var "Direction Pressed", intValue=2, everyFrame=false)
    IntOperator(integer1=var "Current Item", integer2=1, operation=Subtract, storeResult=var "Current Item", everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI UP", delay=0, everyFrame=false)
    SendInventoryPaneInputEvent(InputEvent=Up, Target=Self)
    on CANCEL → Idle
    on FINISHED → Update
  }

  state Select Left {
    BoolTest(boolVariable=var "Allow Horizontal Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)  // disabled
    SetIntValue(intVariable=var "Direction Pressed", intValue=3, everyFrame=false)
    IntOperator(integer1=var "Current Item", integer2=1, operation=Subtract, storeResult=var "Current Item", everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI LEFT", delay=0, everyFrame=false)
    SendInventoryPaneInputEvent(InputEvent=Left, Target=Self)
    on CANCEL → Idle
    on FINISHED → Update
  }

  state Select Down {
    BoolTest(boolVariable=var "Allow Vertical Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)  // disabled
    SetIntValue(intVariable=var "Direction Pressed", intValue=4, everyFrame=false)
    IntOperator(integer1=var "Current Item", integer2=1, operation=Add, storeResult=var "Current Item", everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI DOWN", delay=0, everyFrame=false)
    SendInventoryPaneInputEvent(InputEvent=Down, Target=Self)
    on CANCEL → Idle
    on FINISHED → Update
  }

  state Select Right {
    SetIntValue(intVariable=var "Direction Pressed", intValue=1, everyFrame=false)
    BoolTest(boolVariable=var "Allow Horizontal Selection", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)  // disabled
    IntOperator(integer1=var "Current Item", integer2=1, operation=Add, storeResult=var "Current Item", everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI RIGHT", delay=0, everyFrame=false)
    SendInventoryPaneInputEvent(InputEvent=Right, Target=Self)
    on CANCEL → Idle
    on FINISHED → Update
  }

  state Update {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SELECT UPDATE", delay=0, everyFrame=false)  // disabled
    IntSwitch(intVariable=var "Direction Pressed", compareTo=[1, 2, 3, 4], sendEvent=[→"RIGHT PRESS", →"UP PRESS", →"LEFT PRESS", →"DOWN PRESS"], everyFrame=false)
    on UP PRESS → Still Up?
    on DOWN PRESS → Still Down?
    on RIGHT PRESS → Still Right?
    on LEFT PRESS → Still Left?
  }

  state Activate {
    SetIntValue(intVariable=var "Current Item", intValue=var "Initial Item", everyFrame=false)  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="SELECT UPDATE", delay=0, everyFrame=false)  // disabled
    on FINISHED → Idle
  }

  state Still Up? {
    ListenForUp(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Repeat Up
  }

  state Repeat Up {
    BoolTest(boolVariable=var "Allow Repeat", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    IntAdd(intVariable=var "Repeats", add=1, everyFrame=false)
    ListenForUp(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Select Up
  }

  state Still Down? {
    ListenForDown(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, queueBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Repeat Down
  }

  state Repeat Down {
    BoolTest(boolVariable=var "Allow Repeat", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    IntAdd(intVariable=var "Repeats", add=1, everyFrame=false)
    ListenForDown(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=false, queueBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false)
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Select Down
  }

  state Still Right? {
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Repeat Right
  }

  state Repeat Right {
    BoolTest(boolVariable=var "Allow Repeat", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    IntAdd(intVariable=var "Repeats", add=1, everyFrame=false)
    ListenForRight(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfLeftPressed=false)
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Select Right
  }

  state Still Left? {
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    BoolTest(boolVariable=var "Repeating", isTrue=→"REPEAT", isFalse=(none), everyFrame=false)
    Wait(time=0.25, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Repeat Left
  }

  state Repeat Left {
    BoolTest(boolVariable=var "Allow Repeat", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetBoolValue(boolVariable=var "Repeating", boolValue=true, everyFrame=false)
    IntAdd(intVariable=var "Repeats", add=1, everyFrame=false)
    ListenForLeft(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"CANCEL", isPressedBool=(unset), pressedTimer=(unset), activeBool=(unset), stateEntryOnly=false, ignoreIfRightPressed=false)
    Wait(time=0.15, finishEvent=→"REPEAT", realTime=true)
    on CANCEL → Idle
    on REPEAT → Select Left
  }

  state Inactive {
    on UI ACTIVE → Activate
  }

  state Confirm {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI CONFIRM", delay=0, everyFrame=false)
    ListenForMenuActionsContinuous(EventTarget=Self, SubmitPressed=(none), CancelPressed=→"FINISHED", NonePressed=→"FINISHED", IgnoreAttack=var "Ignore Attack")
    on FINISHED → Released Confirm
    on CONFIRM CANCEL → Idle
  }

  state RS Up {
    BoolTest(boolVariable=var "Right Stick Speed", isTrue=→"SPEED", isFalse=(none), everyFrame=false)
    ListenForRsUp(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"FINISHED", isPressedBool=(unset), activeBool=(unset))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI RS UP", delay=0, everyFrame=false)
    on FINISHED → Idle
    on SPEED → RS Up Speed
  }

  state Cancel {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI CANCEL", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Idle
    on CANCEL CANCEL → Idle
  }

  state Released Confirm {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI CONFIRM RELEASED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Disabled {
    CheckStaticBool(variableName="IsUIListInputBlocked", trueEvent=(none), falseEvent=→"CANCEL", EveryFrame=true)
    on CANCEL → Idle
  }

  state Disabled 1 {
    CheckIsCheatManagerOpen(TrueEvent=(none), FalseEvent=→"CANCEL", StoreValue=(unset), EveryFrame=true)
    on CANCEL → Idle
  }

  state Extra {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI EXTRA", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state RS Up Speed {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI UP", delay=0, everyFrame=false)
    SendInventoryPaneInputEvent(InputEvent=Up, Target=Self)
    Wait(time=0.03, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Idle
  }

  state RS Down {
    BoolTest(boolVariable=var "Right Stick Speed", isTrue=→"SPEED", isFalse=(none), everyFrame=false)
    ListenForRsDown(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"FINISHED", isPressedBool=(unset), activeBool=(unset))
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI RS DOWN", delay=0, everyFrame=false)
    on FINISHED → Idle
    on SPEED → RS Down Speed
  }

  state RS Down Speed {
    SetFloatValue(floatVariable=var "List Scroll Speed", floatValue=0.03, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="UI DOWN", delay=0, everyFrame=false)
    SendInventoryPaneInputEvent(InputEvent=Down, Target=Self)
    Wait(time=0.03, finishEvent=→"FINISHED", realTime=true)
    on FINISHED → Idle
  }

  var Current Item: int = 0
  var Initial Item: int = 1
  var Allow Horizontal Selection: bool = true
  var Allow Vertical Selection: bool = false
  var Ignore Attack: bool = false
  var Right Stick Speed: bool = false
  var Start Inactive: bool = true
  var Allow Repeat: bool = true

  var List Scroll Speed: float = 0
  var Direction Pressed: int = 0
  var Repeats: int = 0
  var Repeating: bool = false
  var Self: gameObject = <null>
}
