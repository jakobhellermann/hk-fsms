fsm Control {
  start Hidden

  state Hidden {
    SetMeshRenderer(gameObject=Self, active=false)
    AudioStop(gameObject=Self, fadeTime=0)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Needolin Wait
  }

  state Emerge {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Up", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetMeshRenderer(gameObject=Self, active=true)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    PlayAudioEvent(audioClip=abyss_tendrils_emerge (sfxstatic_assets_areaabyss.bundle), pitchMin=1.2, pitchMax=0.8, volume=0.85, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Idle
  }

  state Idle {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.1)
    GetHeroCState(VariableName="swimming", StoreValue=var "Is Hero Swimming", EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=var "In Attack Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    BoolTestMulti(boolVariables=[var "In Attack Range", var "Is Hero Swimming", var "Force On"], boolStates=[true, true, false], trueEvent=→"FINISHED", falseEvent=(none), storeResult=(unset), everyFrame=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Down?
    on ABYSS ROAR STOP → Off
  }

  state Down? {
    BoolTest(boolVariable=var "Force On", isTrue=→"ENTER", isFalse=(none), everyFrame=false)
    GetHeroCState(VariableName="swimming", StoreValue=var "Is Hero Swimming", EveryFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=var "In Attack Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    BoolTestMulti(boolVariables=[var "In Attack Range", var "Is Hero Swimming"], boolStates=[true, false, false], trueEvent=→"ENTER", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "In Attack Range", var "Is Hero Swimming"], boolStates=[false, true], trueEvent=→"ENTER", falseEvent=(none), storeResult=(unset), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"ENTER", IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeAudio(gameObject=Self, startVolume=1, endVolume=0, time=0.25)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Down", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hidden
    on ENTER → Idle
    on ABYSS ROAR STOP → Off
  }

  state Needolin Wait {
    GetHeroCState(VariableName="swimming", StoreValue=var "Is Hero Swimming", EveryFrame=true)
    GetPlayerDataVariable(VariableName="HasWhiteFlower", StoreValue=var "Has White Flower")
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=var "In Attack Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    BoolTestMulti(boolVariables=[var "In Attack Range", var "Is Hero Swimming", var "Has White Flower"], boolStates=[true, false, false], trueEvent=→"ENTER", falseEvent=(none), storeResult=(unset), everyFrame=true)
    BoolTestMulti(boolVariables=[var "In Attack Range", var "Is Hero Swimming"], boolStates=[false, true], trueEvent=→"ENTER", falseEvent=(none), storeResult=(unset), everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="", outOfRangeEvent="END", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.7, None=(none), ActiveInner=(none), ActiveOuter=→"ENTER", IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ENTER → Emerge
    on END → Hidden
    on ABYSS ROAR START → Force On
  }

  state Force On {
    SetBoolValue(boolVariable=var "Force On", boolValue=true, everyFrame=false)
    WaitRandom(timeMin=0.1, timeMax=0.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Emerge
  }

  state Off {
    SetBoolValue(boolVariable=var "Force On", boolValue=false, everyFrame=false)
    on FINISHED → Down?
  }

  var Alert Range: bool = false
  var Force On: bool = false
  var Is Hero Swimming: bool = false
  var In Attack Range: bool = false
  var Has White Flower: bool = false
}
