# Pulse Reaction
# 1 location(s):
#   Under_20 / Crest Get Shrine/Active Before Crest Get/Shrine_Pre/Needolin Reaction

fsm Pulse Reaction {
  start Init

  state In Range {
    CheckTrackTriggerCount(target=Self, count=0, test=LessThanOrEqual, everyFrame=true, successEvent=→"EXIT")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.7, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on SING → Fade Up All
    on EXIT → Idle
  }

  state Fade Up All {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=3)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Down Some
    on CANCEL → Fade Down All
  }

  state Fade Down Some {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=Self, ToAlpha=0.7, FadeTime=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Up Some
    on CANCEL → Fade Down All
  }

  state Fade Down All {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → In Range
    on SING → Fade Up All
  }

  state Idle {
    CheckTrackTriggerCount(target=Self, count=0, test=MoreThan, everyFrame=true, successEvent=→"ENTER")
    on ENTER → In Range
  }

  state Init {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0)
    on FINISHED → Idle
  }

  state Fade Up Some {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.4, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Down Some
    on CANCEL → Fade Down All
  }
}
