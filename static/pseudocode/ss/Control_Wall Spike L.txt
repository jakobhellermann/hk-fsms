# Control
# 1 location(s):
#   Shellwood_11b_Memory / Boss Scene/Wall Spike L

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Damager")
    FindNamedChild(gameObject=Self, storeResult=var "Brambles S")
    FindNamedChild(gameObject=Self, storeResult=var "Brambles L")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Point")
    on FINISHED → Start
  }

  state Start {
    BoolTest(boolVariable=var "Start Active", isTrue=→"UP", isFalse=→"DOWN", everyFrame=false)
    on UP → Up
    on DOWN → Down
    on SPIKES DOWN → Down
  }

  state Up {
    ActivateGameObject(gameObject=var "Brambles S", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Brambles L", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObjectDelay(gameObject=var "Damager", activate=true, resetOnExit=false, delay=0.15)
    Tk2dPlayAnimationChildren(gameObject=var "Brambles S", clipName="Small Bramble Up")
    Tk2dPlayAnimationChildren(gameObject=var "Brambles L", clipName="Large Bramble Up")
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Audio Point", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Audio Point", audioClip=vine_spikes_wall_appear (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on DOWN → Down
    on SPIKES DOWN → Down
  }

  state Down {
    ActivateGameObject(gameObject=var "Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dPlayAnimationChildren(gameObject=var "Brambles S", clipName="Small Bramble Down")
    Tk2dPlayAnimationChildren(gameObject=var "Brambles L", clipName="Large Bramble Down")
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Audio Point", vector=(unset), x=(unset), y=var "Hero Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Audio Point", audioClip=vine_spikes_wall_disappear (sfxstatic_assets_areashellwood.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Down 2
  }

  state Down 2 {
    ActivateGameObject(gameObject=var "Brambles S", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Brambles L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on UP → Up
  }

  var Start Active: bool = true

  var Hero Y: float = 0
  var Brambles L: gameObject = <null>
  var Brambles S: gameObject = <null>
  var Damager: gameObject = <null>
  var Audio Point: gameObject = <null>
}
