# Hair Events
# 1 location(s):
#   Cradle_03 / Boss Scene/Silk Boss Hair

fsm Hair Events {
  start Inert
  on HAIR CALM → Hair Calm  // from any state
  on HAIR WILD → Hair Wild  // from any state
  on HAIR UP → Hair Up  // from any state
  on HAIR DOWN → Pause  // from any state

  state Inert {
  }

  state Hair Up {
    AnimatorGroupSetBool(BoolName="Is Awake", SetValue=true, Target=Self)
  }

  state Hair Down {
    AnimatorGroupSetBool(BoolName="Is Awake", SetValue=false, Target=Self)
    AnimatorGroupSetBool(BoolName="Is Intense", SetValue=false, Target=Self)
  }

  state Hair Wild {
    AnimatorGroupSetBool(BoolName="Is Intense", SetValue=true, Target=Self)
    AnimatorGroupSetBool(BoolName="Is Awake", SetValue=true, Target=Self)
  }

  state Hair Calm {
    AnimatorGroupSetBool(BoolName="Is Intense", SetValue=false, Target=Self)
    AnimatorGroupSetBool(BoolName="Is Awake", SetValue=true, Target=Self)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Hair Down
  }
}
