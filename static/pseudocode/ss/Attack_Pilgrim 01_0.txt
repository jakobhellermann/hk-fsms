# Attack
# 1 location(s):
#   Bone_04 / Pilgrims/Pilgrim Folder/Pilgrim 01:0

fsm Attack {
  start Pause

  state Attack Antic {
    SetBoolValue(boolVariable=var "Attack Instantly", boolValue=false, everyFrame=false)
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Ground Check
  }

  state Attack Launch {
    PlayAudioEvent(audioClip=pilgrim_01_jump (sfxstatic_assets_pilgrimslower.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetFloatValue(floatVariable=var "X Speed", floatValue=var "X Scale", everyFrame=false)
    FloatMultiply(floatVariable=var "X Speed", multiplyBy=-9, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=var "X Speed", y=18, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Air", animationTriggerEvent=(none), animationCompleteEvent=(none))
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Attack Air
  }

  state Attack Roll {
    PlayAudioEvent(audioClip=pilgrim_01_song_land_and_roll (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Attack Land", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    DecelerateXY(gameObject=Self, decelerationX=0.95, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Attack Land
  }

  state Attack Air {
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)  // disabled
    SetVelocity2d(gameObject=Self, vector=(unset), x=var "X Speed", y=(unset), everyFrame=true)
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=→"LAND", NotGroundedEvent=(none), EveryFrame=true)
    on LAND → Roll Start
  }

  state Idle {
    on ATTACK → Choice
  }

  state Return {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="pilgrim_behaviour"), sendEvent="ATTACK END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Attack Land {
    CheckIsCharacterGrounded(Target=Self, RayCount=3, GroundDistance=0.2, SkinWidth=-0.05, SkinHeight=0.1, StoreResult=(unset), GroundedEvent=(none), NotGroundedEvent=→"AIR", EveryFrame=true)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    DecelerateXY(gameObject=Self, decelerationX=0.975, decelerationY=(unset), brakeOnExit=false)
    on FINISHED → Return
    on AIR → Re-Air
  }

  state Roll Start {
    SetVelocityByScale(gameObject=Self, speed=-15, ySpeed=(unset), everyFrame=false)
    on FINISHED → Attack Roll
  }

  state Re-Air {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Re-Air")
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Roll Start
  }

  state Choice {
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=pilgrim_leaper_attack (sfxdynamic_assets_pilgrimslower.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    BoolTest(boolVariable=var "Attack Instantly", isTrue=→"FORWARD", isFalse=(none), everyFrame=false)
    RayCast2d(fromGameObject=var "Ray Pt", fromPosition=(unset), direction=(1, 0), space=1, distance=3.25, minDepth=(unset), maxDepth=(unset), hitEvent=→"FORWARD", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeHitFraction=(unset), repeatInterval=0, layerMask=[Terrain(8)], invertMask=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)

    SendRandomEventV4(events=[→"FORWARD", →"BACK"], weights=[1, 1], eventMax=[2, 1], missedMax=[1, 2], activeBool=(unset))
    on FORWARD → Attack Antic
    on BACK → Roll Antic
  }

  state Roll Antic {
    StopWalker(target=Self, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roll Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Roll Back
  }

  state Roll Back {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Roll Back", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocityByScale(gameObject=Self, speed=13, ySpeed=(unset), everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=true)
    RayCast2dV2(fromGameObject=var "Ray Pt", fromPosition=(unset), direction=(0, -2), space=1, distance=2, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"ATTACK", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)  // disabled

    on ATTACK → 
  }

  state Ground Check {

    on AIR → Fall
  }

  state Fall {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    SetFloatValue(floatVariable=var "X Speed", floatValue=0, everyFrame=false)
    on FINISHED → Attack Air
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Ambush
    on ATTACK → Choice
  }

  state Check Ambush {
    GetFsmBool(gameObject=Self, fsmName="pilgrim_behaviour", variableName="Attack Instantly", storeValue=var "Attack Instantly", everyFrame=false)
    on FINISHED → Idle
  }

  var X Scale: float = 0
  var X Speed: float = 0
  var Attack Instantly: bool = false
  var Self: gameObject = Pilgrims/Pilgrim Folder/Pilgrim 01:0
  var Voice Loop: gameObject = Pilgrims/Pilgrim Folder/Pilgrim 01:0/Voice Loop
  var Ray Pt: gameObject = <null>
  var Ray Pt Edge: gameObject = Pilgrims/Pilgrim Folder/Pilgrim 01:0/Ray Pt Edge
  var Audio Loop Voice: gameObject = Pilgrims/Pilgrim Folder/Pilgrim 01:0/Audio Loop Voice
}
