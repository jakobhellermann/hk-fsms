# Control
# 1 location(s):
#   Dust_10 / bell_bench Group/bell_bench_underside

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "cogs_mg complete")
    FindNamedChild(gameObject=Self, storeResult=var "cogs_mg")
    FindNamedChild(gameObject=var "cogs_mg", storeResult=var "main cog")
    FindNamedChild(gameObject=var "cogs_mg", storeResult=var "top cog")
    PlayerDataBoolTest(boolName="FixedDustBellBench", isTrue=→"ACTIVATED", isFalse=(none))
    on ACTIVATED → Activated
    on FINISHED → Idle
  }

  state Activated {
    ActivateGameObject(gameObject=var "Main Root", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "cogs_mg", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "cogs_mg complete", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetCircleCollider(gameObject=var "main cog", active=true)
  }

  state Idle {
    ActivateGameObject(gameObject=var "Main Root", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BREAK → Break
  }

  state Break {
    PlayAudioEvent(audioClip=shellwood_twig_wall_break (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetPlayerDataBool(boolName="FixedDustBellBench", value=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rise Sfx?
  }

  state Rise Sfx? {
    GetFsmBool(gameObject=var "bell_toll_machine", fsmName="Unlock Behaviour", variableName="Activated", storeValue=var "Toll Machine paid", everyFrame=false)
    BoolTest(boolVariable=var "Toll Machine paid", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    PlayAudioEvent(audioClip=bell_shrine_rise_up (sfxstatic_assets_areabellshrine.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetCircleCollider(gameObject=var "top cog", active=false)
    ActivateGameObject(gameObject=bell_bench Group/bell_bench_underside/cog end audio, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SetCircleCollider(gameObject=var "main cog", active=false)
    ActivateGameObject(gameObject=var "cogs_mg", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "cogs_mg complete", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var bell_toll_machine: gameObject = bell_bench Group/bell_bench/bell_toll_machine

  var Toll Machine paid: bool = false
  var Main Root: gameObject = bell_bench Group/bell_bench_underside/main root
  var cogs_mg: gameObject = <null>
  var cogs_mg complete: gameObject = <null>
  var main cog: gameObject = <null>
  var top cog: gameObject = <null>
}
