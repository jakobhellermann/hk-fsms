# Control
# 1 location(s):
#   Bone_East_10 / Rhino Scene/Gate Smashing

fsm Control {
  start Pause
  on TAUNT REACT → Taunt React  // from any state

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    PlayerDataBoolTest(boolName="rhinoChurchUnlocked", isTrue=→"DORMANT", isFalse=(none))
    PlayerDataBoolTest(boolName="churchRhinoKilled", isTrue=→"DORMANT", isFalse=(none))
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"DORMANT", isFalse=(none))
    FindNamedChild(gameObject=Self, storeResult=var "Slam Particles")
    FindNamedChild(gameObject=Self, storeResult=var "Song_Gate_small")
    FindNamedChild(gameObject=Self, storeResult=var "Sign_post_hang_interactive_1")
    FindNamedChild(gameObject=var "Sign_post_hang_interactive_1", storeResult=var "sign_board")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    on DORMANT → Dormant
    on FINISHED → Far
  }

  state Dormant {
  }

  state Far {
    GetDistance(gameObject=var "MainCamera", target=var "Song_Gate_small", storeResult=var "Distance", everyFrame=true)  // disabled
    FloatCompare(float1=var "Distance", float2=70, tolerance=0, equal=(none), lessThan=→"NEAR", greaterThan=(none), everyFrame=true)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Near Range", storeResult=false, sendEvent="NEAR", outOfRangeEvent="", everyFrame=true)
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=2, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SHAKE", everyFrame=true)
    WaitBool(boolTest=var "Raging", time=0.5, finishEvent=→"SHAKE", realTime=false)
    on NEAR → Near
    on SHAKE → Roar?
  }

  state Near {
    GetDistance(gameObject=var "Song_Gate_small", target=var "MainCamera", storeResult=var "Distance", everyFrame=true)  // disabled
    FloatCompare(float1=var "Distance", float2=50, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FAR", everyFrame=true)  // disabled
    FloatAdd(floatVariable=var "Timer", add=1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Timer", float2=2, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"SHAKE", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Near Range", storeResult=false, sendEvent="", outOfRangeEvent="FAR", everyFrame=true)
    WaitBool(boolTest=var "Raging", time=0.5, finishEvent=→"SHAKE", realTime=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Sing Range", storeResult=var "In Sing Range", sendEvent="", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1, MaxReactDelay=1, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=true, ActiveBool=var "In Sing Range", StoreState=None, EveryFrame=true)
    on FAR → Far
    on SHAKE → Roar? 2
    on SING → Sing Pause
  }

  state Near Shake {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitterChildren(gameObject=var "Slam Particles", resetTimeIfPlaying=false, stopOnStateExit=false)
    FloatAddRandom(floatVariable=var "Timer", addMin=-3, addMax=-2)
    ObjectJitter(gameObject=var "Song_Gate_small", x=0.15, y=(unset), z=(unset), allowMovement=false, limitFps=0)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    RandomFloatEither(value1=-6, value2=6, storeResult=var "Force")
    AddForce2d(gameObject=var "sign_board", forceMode=1, atPosition=(unset), vector=(unset), x=var "Force", y=(unset), vector3=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=zombie_guard_club (sfxstatic_assets_areawilds.bundle), pitchMin=0.75, pitchMax=0.85, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Sfx Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[rhino_grunt_1 (sfxstatic_assets_areabone.bundle), rhino_grunt_2 (sfxstatic_assets_areabone.bundle), rhino_grunt_3 (sfxstatic_assets_areabone.bundle), rhino_grunt_4 (sfxstatic_assets_areabone.bundle), rhino_grunt_5 (sfxstatic_assets_areabone.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Sfx Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Dirt Sfx
  }

  state Far Shake {
    FloatAddRandom(floatVariable=var "Timer", addMin=-3, addMax=-2)
    RandomFloatEither(value1=-6, value2=6, storeResult=var "Force")
    AddForce2d(gameObject=var "sign_board", forceMode=1, atPosition=(unset), vector=(unset), x=var "Force", y=(unset), vector3=(unset), everyFrame=false)
    PlayAudioEvent(audioClip=zombie_guard_club (sfxstatic_assets_areawilds.bundle), pitchMin=0.75, pitchMax=0.85, volume=0.7, audioPlayerPrefab=var "", spawnPoint=var "Sfx Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventRandom(audioClips=[rhino_grunt_1 (sfxstatic_assets_areabone.bundle), rhino_grunt_2 (sfxstatic_assets_areabone.bundle), rhino_grunt_3 (sfxstatic_assets_areabone.bundle), rhino_grunt_4 (sfxstatic_assets_areabone.bundle), rhino_grunt_5 (sfxstatic_assets_areabone.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Sfx Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Reset Rage?
  }

  state Roar? {
    BoolTest(boolVariable=var "Raging", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SendRandomEventV4(events=[→"FINISHED", →"ROAR"], weights=[0.75, 0.25], eventMax=[4, 1], missedMax=[1, 4], activeBool=(unset))
    on ROAR → Roar
    on FINISHED → Far Shake
  }

  state Roar {
    PlayAudioEventRandom(audioClips=[rhino_alert_1 (sfxstatic_assets_areabone.bundle), rhino_alert_2 (sfxstatic_assets_areabone.bundle), rhino_alert_3 (sfxstatic_assets_areabone.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor Lowpass@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Sfx Point", spawnPosition=(8, 0, 0), SpawnedPlayerRef=var "")
    FloatAddRandom(floatVariable=var "Timer", addMin=-2, addMax=-1)
    on FINISHED → Far
  }

  state Roar? 2 {
    BoolTest(boolVariable=var "Raging", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    SendRandomEventV4(events=[→"FINISHED", →"ROAR"], weights=[0.75, 0.25], eventMax=[4, 1], missedMax=[1, 4], activeBool=(unset))
    on ROAR → Roar
    on FINISHED → Near Shake
  }

  state Dirt Sfx {
    PlayAudioEventRandom(audioClips=[platform_dirt_fall_1 (sfxstatic_assets_areabone.bundle), platform_dirt_fall_2 (sfxstatic_assets_areabone.bundle), platform_dirt_fall_3 (sfxstatic_assets_areabone.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Sfx Point", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Reset Rage?
  }

  sequence state Taunt React {
    PlayerDataBoolTest(boolName="rhinoChurchUnlocked", isTrue=→"DORMANT", isFalse=(none))
    PlayerDataBoolTest(boolName="churchRhinoKilled", isTrue=→"DORMANT", isFalse=(none))
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"DORMANT", isFalse=(none))
    Wait(time=1.5, finishEvent=(none), realTime=false)
    PlayAudioEventRandom(audioClips=[rhino_alert_1 (sfxstatic_assets_areabone.bundle), rhino_alert_2 (sfxstatic_assets_areabone.bundle), rhino_alert_3 (sfxstatic_assets_areabone.bundle)], pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor Lowpass@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Sfx Point", spawnPosition=(8, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.5, finishEvent=(none), realTime=false)
    SetBoolValue(boolVariable=var "Raging", boolValue=true, everyFrame=false)
    SetIntValue(intVariable=var "Rage Attacks", intValue=6, everyFrame=false)
    on FINISHED → Near
    on DORMANT → Dormant
  }

  state Reset Rage? {
    BoolTest(boolVariable=var "Raging", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    IntAdd(intVariable=var "Rage Attacks", add=-1, everyFrame=false)
    IntCompare(integer1=var "Rage Attacks", integer2=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Raging", boolValue=false, everyFrame=false)
    SetFloatValue(floatVariable=var "Timer", floatValue=0, everyFrame=false)
    on FINISHED → Far
  }

  state Sing Pause {
    SetFloatValue(floatVariable=var "Timer", floatValue=0, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on END → Near
    on FINISHED → Sing
  }

  state Sing {
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=rhino_sing (sfxdynamic_assets_areawilds.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.5, MaxReactDelay=0.5, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    DoCameraShakeV4(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Super Tiny Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=true, vibrate=true)
    on END → Near
  }

  var Distance: float = 0
  var Force: float = 0
  var Timer: float = 0
  var Rage Attacks: int = 0
  var Raging: bool = false
  var In Sing Range: bool = false
  var Sign_post_hang_interactive_1: gameObject = <null>
  var Slam Particles: gameObject = <null>
  var Song_Gate_small: gameObject = <null>
  var sign_board: gameObject = <null>
  var Sfx Point: gameObject = Rhino Scene/Gate Smashing/Sfx Point
  var Audio Loop Voice: gameObject = <null>
}
