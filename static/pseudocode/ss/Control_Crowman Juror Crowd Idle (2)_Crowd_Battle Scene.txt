# Control
# 1 location(s):
#   Room_CrowCourt_02 / Battle Scene/Crowd/Crowman Juror Crowd Idle (2)

fsm Control {
  start Pause
  on CRAWFATHER DEATH END → Leave Start  // from any state

  state Init {
    SetPosition(gameObject=var "Audio Loop Voice", vector=(unset), x=(unset), y=(unset), z=0, space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Crowd Start
  }

  state Crowd Idle {
    on CROWD ROAR → Roar Pause
  }

  state Roar Pause {
    SetStringValue(stringVariable=var "Return Event", stringValue="ROAR", everyFrame=false)
    WaitRandom(timeMin=0, timeMax=0.35, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Call Type
  }

  state Roar {
    SetStringValue(stringVariable=var "Return Event", stringValue="ROAR", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Crowd Roar")
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"CALL", realTime=false)
    on CROWD IDLE → Idle Pause
    on CALL → Call Type
  }

  state Idle Pause {
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar End
  }

  state Crowd Start {
    RandomInt(min=1, max=4, storeResult=var "Start Frame", inclusiveMax=true, noRepeat=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Crowd Roar")
    Tk2dPlayFrame(gameObject=Self, frame=var "Start Frame")
    on FINISHED → Crowd Init Ruckus
  }

  state Roar End {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Crowd Roar End")
    on FINISHED → Crowd Idle
  }

  state Crowd Init Ruckus {
    SetStringValue(stringVariable=var "Return Event", stringValue="RUCKUS", everyFrame=false)
    WaitRandom(timeMin=0.5, timeMax=2, finishEvent=→"CALL", realTime=false)
    on CROWD IDLE → Crowd Init Pause
    on CALL → Call Type
  }

  state Crowd Init Pause {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Roar End
  }

  state Leave Pause {
    WaitRandom(timeMin=2, timeMax=4, finishEvent=→"FINISHED", realTime=false)  // disabled
    FloatAdd(floatVariable=var "Death Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Death Timer", float2=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    SetStringValue(stringVariable=var "Return Event", stringValue="LEAVE", everyFrame=false)
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"CALL", realTime=false)
    on FINISHED → Leave
    on CALL → Call Type
  }

  state Leave {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Flapping")
    PlayAudioEventRandom(audioClips=[crow_fly_away_fg_bg_1 (sfxstatic_assets_areagreymoor.bundle), crow_fly_away_fg_bg_2 (sfxstatic_assets_areagreymoor.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=crawman_med_yelp (sfxdynamic_assets_areagreymoor.bundle), pitchOffset=0, stopPreviousSound=false, forcePlay=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Y Target", z=var "Z Target", space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Z Target", add=-8, everyFrame=false, perSecond=false)
    FloatAdd(floatVariable=var "Y Target", add=20, everyFrame=false, perSecond=false)
    AnimateYPositionTo(GameObject=Self, ToValue=var "Y Target", localSpace=false, time=0.6, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)
    AnimateZPositionTo(GameObject=Self, ToValue=var "Z Target", localSpace=false, time=0.6, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(0, 0, 0, 1), duration=0.4)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Left
  }

  sequence state Left {
    SetMeshRenderer(gameObject=Self, active=false)
    Wait(time=2, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Call Type {
    SendRandomEventV4(events=[→"CRAWMAN SMALL", →"CRAWMAN MED", →"CRAWMAN BIG", →"CRAW"], weights=[1, 1, 1, 1], eventMax=[2, 2, 2, 2], missedMax=[4, 4, 4, 2], activeBool=(unset))
    on CRAWMAN SMALL → Crawman Small
    on CRAWMAN MED → Crawman Med
    on CRAWMAN BIG → Crawman Big
    on CRAW → Craw
  }

  state Crawman Small {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=crawman_crowd_small_yelp (sfxdynamic_assets_areadocks.bundle), pitchOffset=0, stopPreviousSound=false, forcePlay=false)
    on FINISHED → Return
  }

  state Crawman Med {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=crawman_crowd_med_yelp (sfxdynamic_assets_areadocks.bundle), pitchOffset=0, stopPreviousSound=false, forcePlay=false)
    on FINISHED → Return
  }

  state Crawman Big {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=crawman_crowd_big_yelp (sfxdynamic_assets_areadocks.bundle), pitchOffset=0, stopPreviousSound=false, forcePlay=false)
    on FINISHED → Return
  }

  state Craw {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=craw_crowd_small_big_yelp (sfxdynamic_assets_areadocks.bundle), pitchOffset=0, stopPreviousSound=false, forcePlay=false)
    on FINISHED → Return
  }

  state Return {
    SendEventByName(eventTarget=Self, sendEvent=var "Return Event", delay=0, everyFrame=false)
    on RUCKUS → Crowd Init Ruckus
    on ROAR → Roar
    on LEAVE → Leave Pause
  }

  state Leave Start {
    RandomFloat(min=2, max=4, storeResult=var "Death Timer")
    on FINISHED → Leave Pause
  }

  var Y Target: float = 0
  var Z Target: float = 0
  var Death Timer: float = 0
  var Start Frame: int = 0
  var Return Event: string = ""
  var Audio Loop Voice: gameObject = Battle Scene/Crowd/Crowman Juror Crowd Idle (2)/Audio Loop Voice
}
