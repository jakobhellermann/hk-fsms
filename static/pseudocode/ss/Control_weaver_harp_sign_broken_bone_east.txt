# Control
# 1 location(s):
#   Bone_East_03 / weaver_harp_sign_broken_bone_east

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Inactive
  }

  state Inactive {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="ENTER", outOfRangeEvent="", everyFrame=true)
    on ENTER → Needolin Range
  }

  state Needolin Range {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Needolin Range", storeResult=false, sendEvent="", outOfRangeEvent="EXIT", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=5, MaxReactDelay=5, None=(none), ActiveInner=→"NEEDOLIN", ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on EXIT → Inactive
    on NEEDOLIN → Show Text
  }

  state Show Text {
    PlayParticleEmitterInState(gameObject=var "Pt Wisp")
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.75, MaxReactDelay=0.75, None=→"EXIT", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    ActivateGameObject(gameObject=var "glow_text Lv 1", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on EXIT → Needolin Range
  }

  var glow_text Lv 1: gameObject = weaver_harp_sign_broken_bone_east/glow_text Lv 1
  var Pt Wisp: gameObject = weaver_harp_sign_broken_bone_east/Pt Wisp
}
