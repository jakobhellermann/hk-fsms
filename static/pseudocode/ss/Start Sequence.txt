# Start Sequence
# 1 location(s):
#   Bone_East_LavaChallenge / Battle Scene

fsm Start Sequence {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Lava Plats")
    FindNamedChild(gameObject=Self, storeResult=var "CamLock Arena Entry")
    ActivateGameObject(gameObject=var "Top Smoke", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava Plats", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    ActivateGameObject(gameObject=var "Lava Plats", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayerDataBoolTest(boolName="completedLavaChallenge", isTrue=→"COMPLETED", isFalse=(none))
    on BATTLE FLOOR EXPLODE → Plats Up
    on COMPLETED → Completed
  }

  state Plats Up {
    ActivateGameObject(gameObject=var "CamLock Arena Entry", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Lava Plats", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Lava Plats"), sendEvent="BUBBLE UP", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Battle
  }

  state Start Battle {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=StartBattle())
    ActivateGameObject(gameObject=var "Top Smoke", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Completed {
    FindNamedChild(gameObject=Self, storeResult=var "Rising Lava")
    FindNamedChild(gameObject=var "Rising Lava", storeResult=var "Pt Updraft")
    FindNamedChild(gameObject=Self, storeResult=var "Updraft Region")
    SetPosition(gameObject=var "Rising Lava", vector=(unset), x=(unset), y=243.13, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    PlayParticleEmitter(gameObject=var "Pt Updraft", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Updraft Region", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Top Smoke", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Rising Lava"), sendEvent="COMPLETED", delay=0, everyFrame=false)
  }

  var Lava Plats: gameObject = <null>
  var Rising Lava: gameObject = <null>
  var Pt Updraft: gameObject = <null>
  var Updraft Region: gameObject = <null>
  var Top Smoke: gameObject = Battle Scene/top smoke
  var CamLock Arena Entry: gameObject = <null>
}
