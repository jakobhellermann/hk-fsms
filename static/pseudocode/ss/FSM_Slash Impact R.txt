# FSM
# 1 location(s):
#   globalpoolprefabs_assets_all.bundle / Slash Impact R

fsm FSM {
  start Vibrate

  state Choose Anim {
    BoolTest(boolVariable=var "Chooser", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    RandomBool(storeResult=var "Chooser")
    on TRUE → Anim 1
    on FALSE → Anim 2
  }

  state Anim 1 {
    SetStringValue(stringVariable=var "Anim", stringValue="Slash Impact 1\n", everyFrame=false)
    on FINISHED → Anim
  }

  state Anim 2 {
    SetStringValue(stringVariable=var "Anim", stringValue="Slash Impact 2", everyFrame=false)
    on FINISHED → Anim
  }

  state Anim {
    SetScale(gameObject=Self, vector=(unset), x=0.6, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Wait(time=0.17, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Vibrate {
    CheckSceneName(sceneName="Menu_Title", equalEvent=→"FINISHED", notEqualEvent=(none))
    VibrationPlayerPlay(target=Self)
    on FINISHED → Choose Anim
  }

  var Chooser: bool = false
  var Anim: string = ""
}
