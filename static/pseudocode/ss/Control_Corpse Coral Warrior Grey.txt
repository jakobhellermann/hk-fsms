# Control
# 1 location(s):
#   enemycorpses_assets_areacoral.bundle / Corpse Coral Warrior Grey

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Set Defeated
  }

  state Air {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=CWG_stun (sfxdynamic_assets_areacoral.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SetVelocityByScale(gameObject=Self, speed=10, ySpeed=25, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Air")
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CreateObject(gameObject=Boss Death FinalHit (sharedeffects_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    PlayAudioEvent(audioClip=boss_death_pt_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    CheckYPosition(gameObject=Self, compareTo=5.5, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"CENTIPEDE PIT", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on LAND → Land
    on CENTIPEDE PIT → Drop Into Pit
  }

  state Land {
    CheckYPosition(gameObject=Self, compareTo=5.5, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"CENTIPEDE PIT", greaterThan=(none), everyFrame=false, space=0, activeBool=(unset))
    GetPosition(gameObject=Self, vector=(unset), x=var "Death X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatClamp(floatVariable=var "Death X", minValue=110, maxValue=171, everyFrame=false)
    SetPlayerDataFloat(gameObject=var "GameManager", floatName="greyWarriorDeathX", value=var "Death X")
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=(unset), brakeOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Land", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=2, finishEvent=→"NEXT", realTime=false)
    PlayAudioEvent(audioClip=hornet_land_sand_2 (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on NEXT → Roar Antic
    on CENTIPEDE PIT → Drop Into Pit
  }

  state Roar Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Death Roar", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayRandomVoiceFromTable(gameObject=Corpse Coral Warrior Grey/Audio Loop Voice, audioClipTable=CWG_roar_death (sfxdynamic_assets_areacoral.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Roar
  }

  state Roar {
    Wait(time=4.5, finishEvent=→"FINISHED", realTime=false)
    StartRoarEmitter(spawnPoint=Self, delay=0, stunHero=false, roarBurst=false, isSmall=false, noVisualEffect=false, forceThroughBind=false, stopOnExit=true)
    on FINISHED → Crust Pause
  }

  state Crust Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Death Crust")
    PlayRandomAudioClipTable(Table=small_movement_shell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Crust Up
  }

  state Crust Up {
    PlayParticleEmitterChildren(gameObject=var "Pt Crust Up", resetTimeIfPlaying=false, stopOnStateExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    ObjectJitterOnRender(Target=Self, X=0.1, Y=0.1, Z=(unset), LimitFps=30)
    PlayAudioEvent(audioClip=grey_warrior_stone_turn_into (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fade Away
  }

  state Fade Away {
    ActivateGameObject(gameObject=var "Sword", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 0), duration=1)
    Wait(time=2.7, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=grey_warrior_stone_break_away (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, delay=1.25, storePlayer=<null>)
    on FINISHED → Drop Sword
  }

  state Drop Sword {
    SetIsKinematic2d(gameObject=var "Sword", isKinematic=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate Item
  }

  state Set Defeated {
    SetPlayerDataBool(boolName="defeatedGreyWarrior", value=true)
    on FINISHED → Air
  }

  state Activate Item {
    ActivateGameObject(gameObject=Corpse Coral Warrior Grey/Collectable Item Pickup, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Drop Into Pit {
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Eaten
  }

  state Eaten {
    SendEventToRegister(eventName="WARRIOR EATEN")
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Death X: float = 0
  var Pt Crust Up: gameObject = Corpse Coral Warrior Grey/Pt Crust Up
  var Self: gameObject = Corpse Coral Warrior Grey
  var Sword: gameObject = Corpse Coral Warrior Grey/Sword
  var Collectable Item Pickup: gameObject = Corpse Coral Warrior Grey/Collectable Item Pickup
  var Audio Loop Voice: gameObject = Corpse Coral Warrior Grey/Audio Loop Voice
}
