# Control
# 3 location(s):
#   Wisp_02 / Wisp Farmers/Wisp Fireball Master
#   Wisp_07 / Wisp Farmers (1)/Wisp Fireball Master
#   Wisp_08 / Wisp Farmers (1)/Wisp Fireball Master

fsm Control {
  start Init

  state Init {
    on FINISHED → Can Attack
  }

  state Can Attack {
    on ATTACK REQUEST → Approve Attack
  }

  state Approve Attack {
    GetEventSender(sentByGameObject=var "Requesting Wisp")
    SendEventByName(eventTarget=GameObject(var "Requesting Wisp"), sendEvent="APPROVED", delay=0, everyFrame=false)
    SetFloatValue(floatVariable=var "Inactive Timer", floatValue=var "Time Between Attacks", everyFrame=false)
    on FINISHED → Can't Attack
  }

  state Can't Attack {
    FloatAdd(floatVariable=var "Inactive Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Inactive Timer", float2=0, tolerance=0, equal=→"READY", lessThan=→"READY", greaterThan=(none), everyFrame=true)
    on READY → Can Attack
    on ATTACK REQUEST → Deny Attack
  }

  state Deny Attack {
    GetEventSender(sentByGameObject=var "Requesting Wisp")
    SendEventByName(eventTarget=GameObject(var "Requesting Wisp"), sendEvent="DENIED", delay=0, everyFrame=false)
    on FINISHED → Can't Attack
  }

  var Inactive Timer: float = 0
  var Time Between Attacks: float = 1
  var Requesting Wisp: gameObject = <null>
}
