# Control
# 1 location(s):

  on GO UP → Go Up  // from any state
  on TEST ROOST → Rise  // from any state

  state Init {

    FindChild(gameObject=Self, childName="Burst Effects", storeResult=var "Burst Effects")

    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)

    BoolTest(boolVariable=var "Alert", isTrue=→"CHASE", isFalse=(none), everyFrame=true)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="ReAlert Range", storeResult=false, sendEvent="CHASE", outOfRangeEvent="", everyFrame=true)
    on CHASE → Resume Chase
    on SING → Sing
    on RE HANG → Rise
  }

  state Chase {
    SetBoolValue(boolVariable=var "Can Return to Roof", boolValue=true, everyFrame=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=var "Alert", InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    GetDistance(gameObject=var "Hero", target=var "Home", storeResult=var "Distance", everyFrame=true)
    FloatTestToBool(float1=var "Distance", float2=var "Chase Distance", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Range Out", everyFrame=true)

    ChaseObject(gameObject=Self, target=var "Hero", speedMax=4, acceleration=0.1, targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0, offsetX=0, offsetY=1)
    BoolTestMulti(boolVariables=[var "Alert", var "Range Out"], boolStates=[false, true], trueEvent=→"RANGE OUT", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on RANGE OUT → Pursue
    on SING → Sing
  }

  state Return {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)

    IdleBuzzV3(gameObject=Self, waitMin=0.75, waitMax=0.75, speedMax=5, accelerationMin=25, accelerationMax=25, roamingRangeX=1, roamingRangeY=1, manualStartPos=var "Start Pos")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=var "Alert", InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)  // disabled
    BoolTest(boolVariable=var "Alert", isTrue=→"CHASE", isFalse=(none), everyFrame=true)  // disabled
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="ReAlert Range", storeResult=false, sendEvent="CHASE", outOfRangeEvent="", everyFrame=true)
    on FINISHED → Idle
    on CHASE → Resume Chase
    on SING → Sing
  }

  state Pursue {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=false, StoreState=None, EveryFrame=true)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=var "Alert", InRangeEvent=→"CHASE", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    GetDistance(gameObject=var "Hero", target=var "Home", storeResult=var "Distance", everyFrame=true)
    FloatTestToBool(float1=var "Distance", float2=var "Chase Distance", tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Range Out", everyFrame=true)
    FaceObject(objectA=var "Self", objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Flyer TurnToFly", resetFrame=false, everyFrame=true)
    ChaseObject(gameObject=Self, target=var "Hero", speedMax=4, acceleration=0.1, targetSpread=0, spreadResetTimeMin=0, spreadResetTimeMax=0, offsetX=0, offsetY=0)
    BoolTestMulti(boolVariables=[var "Range Out"], boolStates=[false], trueEvent=→"CHASE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    Wait(time=2, finishEvent=→"RANGE OUT", realTime=false)
    on CHASE → Chase
    on RANGE OUT → Return
    on SING → Sing
  }

  state Burst Out {
    PlayAudioEvent(audioClip=shellwood_goomba_flyer_break_off_ceiling (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Burst Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetAudioSource(gameObject=Self, active=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flyer Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-15, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Current Y Speed", space=0, everyFrame=true)
    FloatCompare(float1=var "Current Y Speed", float2=-0.5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Chase
  }

  state Hang {
    SetAudioSource(gameObject=Self, active=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flyer Hang", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckAlertRange(alertRange=var "Alert Range", storeResult=(unset), InRangeEvent=→"IN RANGE", InRangeDelay=0.1, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.2, None=(none), ActiveInner=→"IN RANGE", ActiveOuter=→"IN RANGE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    CheckIsBlackThreaded(Target=Self, TrueEvent=→"BLACK THREADED", FalseEvent=(none), StoreValue=(unset), EveryFrame=false)
    on IN RANGE → Burst Antic
    on TOOK DAMAGE → Detach From Sway Plat
    on BLACK THREADED → Force From Roof
  }

  state Burst Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flyer Antic", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=shellwood_goomba_flyer_grounded_shake (sfxstatic_assets_areashellwood.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=shellwood_goomba_cry (sfxdynamic_assets_areashellwood.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Detach From Sway Plat
    on TOOK DAMAGE → Detach From Sway Plat
  }

  state Battle Pause {
    SetDamageHeroAmount(target=Self, damageDealt=0)
    WaitRandom(timeMin=0.5, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Burst Antic
  }

  state Sing {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flyer Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)

    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=shellwood_goomba_sing (sfxdynamic_assets_areashellwood.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Resume Flying
    on SING DURATION END → Resume Flying
  }

  state Resume Flying {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flyer Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Chase
  }

  state Start Flying {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flyer Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetDamageHeroAmount(target=Self, damageDealt=1)
    on FINISHED → Idle
  }

  state Rise {
    SetBoolValue(boolVariable=var "Can Return to Roof", boolValue=true, everyFrame=false)
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="Flyer TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=2, everyFrame=true)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=→"HIT", rightHitEvent=(none), bottomHitEvent=(none), leftHitEvent=(none), otherLayer=true, otherLayerNumber=8, ignoreTriggers=false)
    CheckAlertRange(alertRange=var "Alert Range", storeResult=(unset), InRangeEvent=→"CHASE", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=true)
    on HIT → Stick To Sway Plat?
    on CHASE → Chase
    on TOOK DAMAGE → Chase
  }

  state Re Hang {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Flyer ReHang", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Hang
    on TOOK DAMAGE → Detach From Sway Plat
  }

  state Go Up {
    SetVelocity2d(gameObject=Self, vector=(0, 6), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Pursue
  }

  state Stick To Sway Plat? {
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(0, 1), space=0, distance=1.5, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"FINISHED", storeDidHit=(unset), storeHitObject=var "Collide Object", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=0, layerMask=[Hero Only(28)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    GetName(gameObject=var "Collide Object", storeName=var "Collide Object Name", everyFrame=false)
    StringCompare(stringVariable=var "Collide Object Name", compareTo="Art", equalEvent=(none), notEqualEvent=→"FINISHED", storeResult=(unset), everyFrame=false)
    SetParent(gameObject=Self, parent=var "Collide Object", resetLocalPosition=false, resetLocalRotation=false)
    SetBoolValue(boolVariable=var "Roosting on Sway Plat", boolValue=true, everyFrame=false)
    on FINISHED → Re Hang
  }

  state Detach From Sway Plat {
    BoolTest(boolVariable=var "Roosting on Sway Plat", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetBoolValue(boolVariable=var "Roosting on Sway Plat", boolValue=false, everyFrame=false)
    on FINISHED → Burst Out
  }

  state Resume Chase {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=shellwood_goomba_cry (sfxdynamic_assets_areashellwood.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Chase
  }

  state Hang Init Pause {
    Wait(time=0.1, finishEvent=(none), realTime=false)
    on FINISHED → Hang
  }

  state Force From Roof {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-3, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    GetPosition(gameObject=Self, vector=var "Start Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    on FINISHED → Start Flying
  }

  var Chase Distance: float = 12
  var Current Y Speed: float = 0

  var Alert: bool = false

  var Home: gameObject = <null>
  var Self: gameObject = <null>
  var Pt Emerge: gameObject = Shellwood Goomba Flyer (3)/Pt Emerge
  var Collide Object: gameObject = <null>
  var Audio Loop Voice: gameObject = Shellwood Goomba Flyer (3)/Audio Loop Voice
  var Alert Range: object = <null>
}
