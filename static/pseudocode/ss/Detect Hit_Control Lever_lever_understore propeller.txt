# Detect Hit
# 3 location(s):
#   Library_07 / understore propeller/lever/Control Lever
#   Song_11 / propeller understore propeller/lever/Control Lever
#   Under_17 / understore propeller/lever/Control Lever

fsm Detect Hit {
  start Init
  on ACTIVATED → Activated  // from any state

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")
    on HIT → Send
  }

  state Send {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="LEVER HIT", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    GetGrandparent(gameObject=Self, storeResult=var "Parent")
    on FINISHED → Idle
  }

  state Activated {
    GetNextChild(gameObject=Self, storeNextChild=var "Child", loopEvent=→"LOOP", finishedEvent=→"FINISHED", resetFlag=(unset))
    on LOOP → Set Material
    on FINISHED → Inert
  }

  state Inert {
  }

  state Set Material {
    SetMaterial(gameObject=var "Child", materialIndex=0, material=Crossroads_Material (materials_assets_shared.bundle))
    on FINISHED → Activated
  }

  var Parent: gameObject = <null>
  var Child: gameObject = <null>
}
