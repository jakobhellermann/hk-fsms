# Control
# 2 location(s):
#   Wisp_02 / Farmer Wisp BG
#   Wisp_07 / Farmer Wisp BG

fsm Control {
  start Dormant

  state Dormant {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Move
  }

  state Move {
    Translate(gameObject=Self, vector=(unset), x=-1.5, y=(unset), z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    Wait(time=20, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End
  }

  state End {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }
}
