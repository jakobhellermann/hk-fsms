# Knight Death Control
# 1 location(s):
#   localpoolprefabs_assets_shared.bundle / Knight Spike Death

fsm Knight Death Control {
  start Pause
  on HAZARD RESPAWN RESET → Recycle  // from any state

  state Stab {
    SendEventToRegister(eventName="KNIGHT SPIKE HIT")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="HAZARD DAMAGE", delay=0, everyFrame=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=FreezeMoment(9))
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Hero Damage Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fling
  }

  state Fling {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spike Death")
    SetGravity2dScale(gameObject=Self, gravityScale=0.7)
    RandomFloat(min=80, max=100, storeResult=var "Hero Angle")  // disabled
    RandomFloatEither(value1=80, value2=100, storeResult=var "Hero Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Hero Angle", speed=17, everyFrame=false)
    FaceDirection(gameObject=Self, spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", everyFrame=true, pauseBetweenTurns=false, pauseTime=0)
  }

  state Recycle {
    RecycleSelf()
  }

  state Pause {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Spike Death Antic")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetGravity2dScale(gameObject=Self, gravityScale=0)
    GetScale(gameObject=var "Hero", vector=(unset), xScale=var "Hero Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Hero Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Stab
  }

  var Hero Angle: float = 0
  var Hero Scale: float = 0
  var Pierces: gameObject = Knight Spike Death/Pierces
  var Self: gameObject = Knight Spike Death
}
