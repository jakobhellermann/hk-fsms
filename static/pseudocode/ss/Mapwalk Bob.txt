# Mapwalk Bob
# 1 location(s):
#   maps_assets_all.bundle / Game_Map_Hornet/Compass Icon

fsm Mapwalk Bob {
  start Start

  state Idle {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    on MAPWALK START → Mapwalk
  }

  state Mapwalk {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Walk")
    on MAPWALK END → Idle
  }

  state Start {
    BoolTest(boolVariable=var "IsWalking", isTrue=→"MAPWALK START", isFalse=(none), everyFrame=false)
    on MAPWALK START → Mapwalk
    on FINISHED → Idle
  }

  var IsWalking: bool = false
}
