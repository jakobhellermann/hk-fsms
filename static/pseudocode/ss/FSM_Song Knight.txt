# FSM
# 1 location(s):
#   Hang_17b / Boss Scene - To Additive Load/Song Knight

// uses template: song_knight_catch_hero
fsm FSM {
  start Init

  state Idle {
    on CATCH HERO → Catch
    on MULTI HIT CONNECT → Listen for MultiHit?
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Catch Dummy")
    on FINISHED → Idle
  }

  state Catch {
    SetPositionToObject2D(gameObject=var "Catch Dummy", targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    FindClosest(gameObject=var "Catch Dummy", withTag="Orb Target", ignoreOwner=true, mustBeVisible=false, storeObject=var "Catch Target", storeDistance=(unset), everyFrame=false)
    GetPosition(gameObject=var "Catch Target", vector=var "Catch Pos", x=(unset), y=(unset), z=(unset), space=1, everyFrame=false)
    AnimatePositionTo(gameObject=var "Catch Dummy", toValue=var "Catch Pos", localSpace=true, time=0.15, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    MovePosition2dV2(gameObject=var "Hero", targetPosition=var "Catch Dummy", vector=(unset), x=(unset), y=(unset), space=0, everyFrame=true)
    SetPositionToObject2D(gameObject=var "Hero", targetObject=var "Catch Dummy", xOffset=0, yOffset=0, everyFrame=true)  // disabled
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Finish
  }

  state Listen for MultiHit? {
    BoolTest(boolVariable=var "Listen For MultiHit", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    on FINISHED → Catch
    on CANCEL → Idle
  }

  state Finish {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MULTIHIT CATCH END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  var Listen For MultiHit: bool = true

  var Catch Pos: vector3 = (0, 0, 0)
  var Catch Dummy: gameObject = <null>
  var Catch Target: gameObject = <null>
}
