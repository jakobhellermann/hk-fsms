# Music Control
# 1 location(s):
#   Tut_02 / _SceneManager

fsm Music Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check Tute Completion
  }

  state Check Tute Completion {
    PlayerDataBoolTest(boolName="completedTutorial", isTrue=→"COMPLETED", isFalse=→"UNCOMPLETED")
    on COMPLETED → State 1
    on UNCOMPLETED → Unc Pause
  }

  state Unc Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Unc Music
  }

  state Unc Music {
    ApplyMusicCue(musicCue=MossCave (audiocuesdynamic_assets_areatutorial.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Sub Area (audioobjects_assets_all.bundle), transitionTime=2)
    on FINISHED → 
  }

  state State 1 {
    ApplyMusicCue(musicCue=MossCave (audiocuesdynamic_assets_areatutorial.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=4)
  }
}
