    SetRotation(gameObject=var "Effect", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    AnimateRotationTo(gameObject=var "Sprite", fromValue=0, toValue=-90, worldSpace=false, negativeSpace=false, time=0.75, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=(none), realTime=false)

    on SKIP → Subsequent
  }

  state Subsequent {

  }

  state Start On {

    SendEventByName(eventTarget=GameObject(var "Audio Loop Piped Water"), sendEvent="START", delay=0, everyFrame=false)
    on FINISHED → Subsequent

  var Value: int = 0
  var Pt FirstShower: gameObject = <null>
  var Pt Shower: gameObject = <null>
  var Sprite: gameObject = <null>
  var Self: gameObject = Shower System (1)/Shower Lever 2
  var Effect: gameObject = <null>
