# Control
# 1 location(s):
#   Cog_10_Destroyed / Battle Scene/Wake Scene/Wake Threads/black_thread_impacter_intro (2)

fsm Control {
  start Init

  state Init {
    on FINISHED → Inactive
  }

  state Inactive {
    ActivateGameObject(gameObject=var "silk_strand_impact", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on IMPACT → Impact Pause
  }

  state Impact Pause {
    RandomFloat(min=0, max=1.5, storeResult=var "Wait Time")
    FloatSubtract(floatVariable=var "Idle Time", subtract=var "Wait Time", everyFrame=false, perSecond=false)
    Wait(time=var "Wait Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Impact
  }

  state Impact {
    ActivateGameObject(gameObject=var "silk_strand_impact", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    AnimatorPlay(gameObject=var "silk_strand_impact", stateName="Silk_strand_bg_impact", layer=(unset), normalizedTime=(unset), everyFrame=false)
    ListenForAnimationEvent(Target=var "silk_strand_impact", Response=→"FINISHED")
    on FINISHED → Impacted
  }

  state Idle {
    ActivateGameObject(gameObject=var "silk_strand_impact", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FLAIL → Flail Pause
  }

  state Leave {
    AnimatorPlay(gameObject=var "silk_strand_impact", stateName="Silk_strand_bg_exit", layer=(unset), normalizedTime=(unset), everyFrame=false)
    SendEventToRegisterDelay(EventName="SILK STRAND PULL", delay=0.3, ExcludeTarget=Self)
  }

  state Flail Pause {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flail
  }

  state Flail {
    SendEventToRegisterDelay(EventName="ABYSS ROAR START", delay=0, ExcludeTarget=Self)
    on BATTLE END → Burst
  }

  sequence state Impacted {
    DoCameraShakeV4(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false, vibrate=true)
    ActivateGameObject(gameObject=var "black_thread_strand", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=(none))
    CallMethodProper(gameObject=var "black_thread_strand", behaviour="BlackThreadStrand", methodName="RageForTime", parameters=[1], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.3, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  sequence state Burst {
    WaitRandom(timeMin=0, timeMax=2, finishEvent=(none), realTime=false)
    SetAnimatorCullingMode(gameObject=var "black_thread_strand", alwaysAnimate=true, cullCompletely=false)
    CallMethodProper(gameObject=var "black_thread_strand", behaviour="BlackThreadStrand", methodName="Deactivate", parameters=[], storeResult=(unset var), EveryFrame=false)
    AnimatorPlay(gameObject=var "black_thread_strand", stateName="Burst", layer=(unset), normalizedTime=(unset), everyFrame=false)
    Wait(time=5, finishEvent=(none), realTime=false)
    SetAnimatorCullingMode(gameObject=var "black_thread_strand", alwaysAnimate=false, cullCompletely=true)
  }

  var Idle Time: float = 2
  var Wait Time: float = 0
  var silk_strand_impact: gameObject = Battle Scene/Wake Scene/Wake Threads/black_thread_impacter_intro (2)/blac_thread_impact
  var black_thread_strand: gameObject = Battle Scene/Wake Scene/Wake Threads/black_thread_impacter_intro (2)/black_thread_strand
}
