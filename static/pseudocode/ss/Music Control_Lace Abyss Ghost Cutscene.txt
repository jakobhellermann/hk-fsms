# Music Control
# 1 location(s):
#   Abyss_05 / Cutscene Options/Lace Abyss Ghost Cutscene

fsm Music Control {
  start Idle

  state Idle {
    on START MUSIC → Start Music
  }

  sequence state Start Music {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=LastDivePrologue (audiocuesdynamic_assets_areaabyss.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Sub Area (audioobjects_assets_all.bundle), transitionTime=4)
    on STOP MUSIC → Fade To Abyss
  }

  sequence state Fade To Abyss {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=8)
    Wait(time=10, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=Abyss (audiocuesdynamic_assets_areaabyss.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Main Only (audioobjects_assets_all.bundle), transitionTime=4)
  }
}
