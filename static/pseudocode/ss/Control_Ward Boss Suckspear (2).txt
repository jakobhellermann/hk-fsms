    on FINISHED → Idle
  }

  state Idle {

  }

  state Pause {
    Wait(time=var "Suck Pause", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Suck Start
  }

    AnimateRotationTo(gameObject=var "Sprite Pivot", fromValue=var "Pivot Rotate", toValue=0, worldSpace=false, negativeSpace=false, time=0.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=(none), realTime=false)

    on FINISHED → Suck In
  }

  state Suck In {
    PlayAudioEvent(audioClip=unravelled_minion_ground_emerge (sfxstatic_assets_areaward.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")

}
