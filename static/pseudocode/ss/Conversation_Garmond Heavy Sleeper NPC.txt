# Conversation
# 1 location(s):
#   Hang_04 / Black Thread States/Normal World/Aftermath Control/Battle Aftermath/Garmond Heavy Sleeper NPC

fsm Conversation {
  start Pause

  state Idle {
    on INTERACT → Speak
  }

  state End {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Idle
  }

  state Init {
    SetAudioClip(gameObject=var "Audio Loop Voice", audioClip=Garmond_sleep_loop (sfxstatic_assets_garmond.bundle), autoPlay=true, stopOnExit=false)
    on FINISHED → Idle
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Speak {
    RunDialogue(Sheet="Wanderers", Key=var "Speak Convo", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End
  }

  var Speak Convo: string = "GARMOND_GRAND_FORUM_ASLEEP"

  var Spoken: bool = false
  var Audio Loop Voice: gameObject = Black Thread States/Normal World/Aftermath Control/Battle Aftermath/Garmond Heavy Sleeper NPC/Audio Loop Voice
}
