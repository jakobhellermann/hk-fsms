# Control
# 1 location(s):
#   Chapel_Wanderer / Pilgrim Trap Spikes/Pilgrim Trap Spike (1)

fsm Control {
  start Init

  state Init {
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → Dormant
  }

  state Dormant {
    on TRAP → Pause
  }

  state Antic {
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEventRandom(audioClips=[bone_worm_antic_1 (sfxstatic_assets_areabone.bundle), bone_worm_antic_2 (sfxstatic_assets_areabone.bundle), bone_worm_antic_3 (sfxstatic_assets_areabone.bundle)], pitchMin=0.8, pitchMax=1.2, volume=0.8, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Burst
    on RETRACT → Retract
  }

  state Burst {
    PlayParticleEmitterChildren(gameObject=var "Pt Burst", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetMeshRenderer(gameObject=Self, active=true)
    PlayAudioEvent(audioClip=trap_spike_plate_rise (sfxstatic_assets_areadustmaze.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spike Up", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Attack
    on DAMAGED HERO → Damaged Hero
    on RETRACT → Damage Retract Pause
    on TINKED → Retract
  }

  state Attack {
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on DAMAGED HERO → Damaged Hero
    on RETRACT → Damage Retract Pause
    on TINKED → Retract
    on FINISHED → Retract
  }

  state Retract {
    PlayAudioEvent(audioClip=spike_trap_new_spikes_rise (sfxstatic_assets_areaboneareamossareaslabareawilds.bundle), pitchMin=0.95, pitchMax=1.05, volume=0.4, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Spike Down", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → End
  }

  state End {
    SetMeshRenderer(gameObject=Self, active=false)
  }

  state Pause {
    WaitRandom(timeMin=0, timeMax=0.2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  state Damaged Hero {
    GetParent(gameObject=Self, storeResult=var "Parent")
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="RETRACT", delay=0, everyFrame=false)
    on FINISHED → Damage Retract Pause
  }

  state Damage Retract Pause {
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Retract
  }

  var Pt Antic: gameObject = Pilgrim Trap Spikes/Pilgrim Trap Spike (1)/Pt Antic
  var Pt Burst: gameObject = Pilgrim Trap Spikes/Pilgrim Trap Spike (1)/Pt Burst
  var Parent: gameObject = <null>
}
