# Control
# 1 location(s):
#   Bone_01 / Battle Scene/Wave 2/Bone Goomba (1)

fsm Control {
  start Pause

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Emerge")  // disabled
    BoolTest(boolVariable=var "Hiding", isTrue=(none), isFalse=→"ACTIVE", everyFrame=false)
    on ACTIVE → Start Walker
    on FINISHED → Hiding
  }

  state Hiding {
    SetAudioSource(gameObject=Self, active=false)
    SetDamageHeroAmount(target=Self, damageDealt=0)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Hiding")
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    FloatCompare(float1=var "Z Pos", float2=0.5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"BG", everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="", storeResult=var "Alert", sendEvent="", outOfRangeEvent="", everyFrame=true)
    BoolTest(boolVariable=var "Alert", isTrue=→"WAKE", isFalse=(none), everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=(none), ActiveOuter=→"WAKE", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on TOOK DAMAGE → Emerge
    on WAKE → Emerge Antic
    on SING → Start Sing Hiding
    on BG → BG Hiding
  }

  state Start Walker {
    SetAudioSource(gameObject=Self, active=true)
    StartCrawler(Target=Self, ScheduleTurn=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.4, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Velocity Y", float2=-0.1, tolerance=0, equal=(none), lessThan=→"FALL", greaterThan=(none), everyFrame=true)
    on SING → Start Sing Walking
    on FALL → Fall
  }

  state Emerge {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    PlayAudioEvent(audioClip=bone_goomba_emerge (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHeroAmount(target=Self, damageDealt=1)
    ActivateGameObject(gameObject=var "Emerge Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetLayer(gameObject=Self, layer=Enemies(11))
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Start Walker
  }

  state Emerge Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=bone_goomba_head_shake_in_ground (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Emerge
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Start Sing Hiding {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Hiding To Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Sing
  }

  sequence state Start Sing Walking {
    StopCrawler(Target=Self, WaitForTurn=true)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Walk To Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    SetAudioSource(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    on FINISHED → Sing
  }

  state Sing {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.3, MaxReactDelay=0.5, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=mossbone_crawler_sing (sfxdynamic_assets_areamoss.bundle), noThreadEffects=false, noPuppetString=false, randomSingStartTime=true, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Emerge Quick
    on SING DURATION END → Emerge Quick
  }

  state BG Hiding {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetLayer(gameObject=Self, layer=Ignore Raycast(2))
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=CancelSetZ())
    on BATTLE START → Emerge Pause
  }

  state BG Emerge Antic {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Emerge Antic Loop")
    RandomFloat(min=0.006, max=0.0069, storeResult=var "Z Pos")
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false, lateUpdate=false)
    SetLayer(gameObject=Self, layer=Terrain Detector(14))
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayRandomAudioClipTable(Table=enemy_emerge_antic (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayAudioEvent(audioClip=bone_goomba_head_shake_in_ground (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocity2d(gameObject=Self, vector=(0, 10), x=(unset), y=(unset), everyFrame=false)
    PlayParticleEmitterInState(gameObject=var "Pt WakeAntic")
    PlayParticleEmitterInState(gameObject=var "Pt Emerge")
    on FINISHED → Emerge
  }

  state Emerge Pause {
    WaitRandom(timeMin=0, timeMax=0.3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → BG Emerge Antic
  }

  state Fall {
    StopCrawler(Target=Self, WaitForTurn=true)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fall")
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    CheckCollisionSideEnter(topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    CheckCollisionSide(collidingObject=Self, topHit=(unset), rightHit=(unset), bottomHit=(unset), leftHit=(unset), topHitEvent=(none), rightHitEvent=(none), bottomHitEvent=→"LAND", leftHitEvent=(none), otherLayer=false, otherLayerNumber=0, ignoreTriggers=false)
    on LAND → Land
  }

  state Land {
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Start Walker
  }

  state Emerge Quick {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    PlayAudioEvent(audioClip=bone_goomba_emerge (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=0.75, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Emerge", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetDamageHeroAmount(target=Self, damageDealt=1)
    ActivateGameObject(gameObject=var "Emerge Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    SetLayer(gameObject=Self, layer=Enemies(11))
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on FINISHED → Start Walker
  }

  var Hiding: bool = true

  var Z Pos: float = 0
  var Velocity Y: float = 0
  var Alert: bool = false
  var Alert Range: gameObject = <null>
  var Emerge Effect: gameObject = Battle Scene/Wave 2/Bone Goomba (1)/Emerge Effect
  var Pt WakeAntic: gameObject = Battle Scene/Wave 2/Bone Goomba (1)/Pt WakeAntic
  var Pt Emerge: gameObject = Battle Scene/Wave 2/Bone Goomba (1)/Pt Emerge
  var Audio Loop Voice: gameObject = Battle Scene/Wave 2/Bone Goomba (1)/Audio Loop Voice
}
