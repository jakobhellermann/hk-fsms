fsm Rest Animation {
  start Idle
  on REST RESET → Rest Reset  // from any state

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Black Thread States Thread Only Variant/Normal World/_NPCs/Dust Traders/Dust Trader Large@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Rest From Left
    on RIGHT → Rest From Right
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest From Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Turn Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Rest From Right
  }

  state Rest From Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Large Idle Right", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Rest Idle
  }

  state Rest Idle {
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    SelectRandomFloat(floats=[0, 0, 11, 11, 22, 22], weights=[1, 1, 1, 1, 1, 1], storeFloat=var "Clip Start Position")
    SetAudioClip(gameObject=var "Audio Loop Voice Shared", audioClip=Benjin_and_Crull_idle_talk_loop (sfxstatic_assets_areadust.bundle), autoPlay=false, stopOnExit=false)

    AudioPlayDelay(gameObject=var "Audio Loop Voice Shared", volume=1, oneShotClip=<null>, delay=1)

  }

  state Rest Reset {
    SetBoolValue(boolVariable=var "Resting", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  var Clip Start Position: float = 0
  var Resting: bool = false
  var Audio Loop Voice Shared: gameObject = Black Thread States Thread Only Variant/Normal World/_NPCs/Dust Traders/Audio Loop Voice Shared

}
