# Control
# 1 location(s):
#   localpoolprefabs_assets_shared.bundle / Wish Granted Prompt New

fsm Control {
  start Init

  state Init {
    FindChild(gameObject=Self, childName="UI Silk Swish", storeResult=var "UI Silk Swish")
    FindChild(gameObject=var "UI Silk Swish", childName="Explainer", storeResult=var "Explainer")
    FindChild(gameObject=var "UI Silk Swish", childName="Continue Glow", storeResult=var "Continue Glow")
    FindChild(gameObject=var "UI Silk Swish", childName="Quest Icon Glow", storeResult=var "Quest Icon Glow")
    FindChild(gameObject=var "UI Silk Swish", childName="Quest Icon", storeResult=var "Quest Icon")
    FindChild(gameObject=var "UI Silk Swish", childName="UI Stop", storeResult=var "UI Stop")
    ActivateGameObject(gameObject=var "Continue Glow", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Close Inventory
  }

  state Idle {
    TransitionToAudioSnapshot(snapshot=Wish Promised (audioobjects_assets_all.bundle), transitionTime=1)
    ActivateGameObject(gameObject=var "Explainer", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=0.9, z=(unset), space=1, everyFrame=false, lateUpdate=false)  // disabled
    AudioPlayerOneShot(audioPlayer=Audio Player UI (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClips=[<null>], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0.4, storePlayer=<null>)  // disabled
    AudioPlayerOneShot(audioPlayer=Audio Player UI (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClips=[H179_new_wish_promised redo (sfxstatic_assets_shared.bundle)], weights=[1], pitchMin=1, pitchMax=1, volume=1, delay=0.4, storePlayer=<null>)
    on FINISHED → Wait for Input
  }

  state First? {
    PlayerDataBoolTest(boolName="promisedFirstWish", isTrue=(none), isFalse=→"FIRST")
    on FIRST → Explainer Up
    on FINISHED → Idle
  }

  state Fade Down {
    AwardQueuedAchievements(delay=1)
    RemoveHeroInputBlocker(Blocker=Self)
    PlayAudioEvent(audioClip=ui_wish_splash_screen_leave (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetAnimatorTrigger(gameObject=var "UI Silk Swish", trigger="end")
    TransitionToAudioSnapshot(snapshot=Not Playing (audioobjects_assets_all.bundle), transitionTime=3)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Deactivate
  }

  state Explainer Up {
    ActivateGameObject(gameObject=var "Explainer", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=1.9, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Wait for Input
  }

  state Deactivate {
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=false)
    SendMessage(gameObject=var "GameManager", delivery=SendMessage, options=1, functionCall=QueueSaveGame())
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="WISH PROMISED END")
  }

  state Wait for Input {
    ActivateGameObject(gameObject=var "UI Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ListenForPromptContinue(EventTarget=Self, ContinuePressed=→"PRESS")
    on PRESS → Press
  }

  state Press {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Continue Glow", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "UI Stop", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=button (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")  // disabled
    on FINISHED → Fade Down
  }

  state Close Inventory {
    AddHeroInputBlocker(Blocker=Self)
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=true)
    PlayerDataVariableTest(VariableName="isInventoryOpen", ExpectedValue=true, IsExpectedEvent=(none), IsNotExpectedEvent=→"FINISHED")
    SendEventToRegister(eventName="INVENTORY CANCEL")
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  var Continue Glow: gameObject = <null>
  var Explainer: gameObject = <null>
  var Quest Icon: gameObject = <null>
  var Quest Icon Glow: gameObject = <null>
  var UI Silk Swish: gameObject = <null>
  var UI Stop: gameObject = <null>
}
