# Control
# 1 location(s):
#   Cradle_03 / Boss Scene/Soul_Snare/Snare/Rune Particle Follow Hornet

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Walk")
    FindNamedChild(gameObject=Self, storeResult=var "white_light")
    on FINISHED → Out
  }

  state Out {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → In
  }

  state In {
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    PlayParticleEmitterInState(gameObject=var "Pt Walk")
    SendMessage(gameObject=var "white_light", delivery=SendMessage, options=1, functionCall=UnpauseFade())
    SendMessage(gameObject=var "white_light", delivery=SendMessage, options=1, functionCall=FadeIn())
    on EXIT → End
  }

  state End {
    SendMessage(gameObject=var "white_light", delivery=SendMessage, options=1, functionCall=PauseFade())
    EaseSpriteColor(target=var "white_light", fromValue=(unset), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.15, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on FINISHED → Out
    on ENTER → In
  }

  var Pt Walk: gameObject = <null>
  var white_light: gameObject = <null>
}
