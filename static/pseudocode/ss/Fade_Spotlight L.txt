fsm Fade {
  start Down
  on FADE DOWN → Down  // from any state
  on SPOTLIGHTS DOWN → Down  // from any state

  state Down {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=0.5)
    on FADE UP → Up
  }

  state Up {
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=0.5)
    on FADE DOWN → Down
  }
}
