# 1 location(s):
#   Slab_10b / Boss Scene

fsm Outro {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "pre_appear_loop")
    FindNamedChild(gameObject=Self, storeResult=var "Shrine First Weaver NPC")
    ActivateGameObject(gameObject=var "Song Region - Boss Active", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on SPIRE NPC ACTIVATE → Pause
  }

  state Antic {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=silk_heart_crescendo (sfxstatic_assets_silkshared.bundle), pitchMin=1.1, pitchMax=1.1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Player")
    ActivateGameObject(gameObject=var "pre_appear_loop", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Appear
  }

  state Appear {
    ActivateGameObject(gameObject=var "Shrine First Weaver NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Player", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    on FINISHED → Activate
  }

  state Pause {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
  }

  state Activate {
    SetCollider(gameObject=var "Shrine First Weaver NPC", active=true, resetOnExit=false)
  }

  var Pt Appear Antic: gameObject = <null>
  var Shrine First Weaver NPC: gameObject = <null>
  var Player: gameObject = <null>
  var pre_appear_loop: gameObject = <null>
  var Song Region - Boss Active: gameObject = <null>
}
