# Fatigue Control
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Flea Game - Dodging

fsm Fatigue Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    SetFloatValue(floatVariable=var "Fatigue", floatValue=0, everyFrame=false)
    on HORNET BONKED → Bonk Dmg
    on HORNET BONKED HEAVY → Bonk Heavy
  }

  state Bonk Dmg {
    FloatAdd(floatVariable=var "Fatigue", add=20, everyFrame=false, perSecond=false)
    FloatClamp(floatVariable=var "Fatigue", minValue=0, maxValue=100, everyFrame=false)
    on FINISHED → Bonk Recover
  }

  state Bonk Recover {
    FloatAdd(floatVariable=var "Fatigue", add=-5, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Fatigue", float2=0, tolerance=0, equal=(none), lessThan=→"END", greaterThan=(none), everyFrame=true)
    on HORNET BONKED → Bonk Dmg
    on HORNET BONKED HEAVY → Bonk Heavy
    on END → Idle
  }

  state Bonk Heavy {
    FloatAdd(floatVariable=var "Fatigue", add=50, everyFrame=false, perSecond=false)
    FloatClamp(floatVariable=var "Fatigue", minValue=0, maxValue=100, everyFrame=false)
    on FINISHED → Bonk Recover
  }

  var Fatigue: float = 0
}
