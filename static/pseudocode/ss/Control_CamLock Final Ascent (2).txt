# Control
# 1 location(s):
#   Abyss_09 / Group:36/CamLock Final Ascent (2)

fsm Control {
  start Inactive

  state Inactive {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on SUPER JUMP LAUNCH → Active
  }

  state Active {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    on SUPER JUMP ENDED → Inactive
  }
}
