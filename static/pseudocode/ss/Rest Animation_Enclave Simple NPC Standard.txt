# Rest Animation
# 3 location(s):
#   Song_Enclave / Black Thread States/Black Thread World/Enclave Act 3/Town/Enclave Simple NPC Standard
#   Song_Enclave / Black Thread States/Black Thread World/Enclave Act 3 basic talk NPC/Present/Enclave Simple NPC Standard
#   Song_Enclave / Black Thread States/Normal World/Enclave States/States/Level 3/Enclave Simple NPC Standard

fsm Rest Animation {
  start Idle
  on DEACTIVATED_INSTANT → End Rest Instant  // from any state

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="get_PreviousState", parameters=[var "Look Anim State"], storeResult=var "Look Anim State", EveryFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Rest From Left
    on RIGHT → Rest Idle
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Rest From Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Standard Turn Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Rest Idle
  }

  state Rest Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Standard Resting", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    on CANCEL → Wake Up
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "Should Face Left"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest to Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Standard Turn Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Rest to Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Standard Rest to Idle Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Wake Up {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="ShouldFaceLeft", parameters=[], storeResult=var "Should Face Left", EveryFrame=false)
    BoolTest(boolVariable=var "Should Face Left", isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    on LEFT → Rest to Left
    on RIGHT → Rest to Right
  }

  state End Rest Instant {
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="ShouldFaceLeft", parameters=[], storeResult=var "Should Face Left", EveryFrame=false)
    on FINISHED → Reset
  }

  var Resting: bool = false
  var Should Face Left: bool = false
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
