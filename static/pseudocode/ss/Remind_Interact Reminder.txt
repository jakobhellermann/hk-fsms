# Remind
# 1 location(s):
#   Tut_03 / Interact Reminder

fsm Remind {
  start Pause
  on HIDE INTERACT REMINDER → Activate  // from any state

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    on ENTER → State 1
    on ACTIVATE → Activate
  }

  state State 1 {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventToRegister(eventName="REMINDER INTERACT")
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    on EXIT → State 2
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
    on ACTIVATE → Set Active
  }

  state Activate {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventToRegister(eventName="REMINDER INTERACT END")
  }

  state Set Active {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state State 2 {
    SendEventToRegister(eventName="REMINDER INTERACT END")
    on FINISHED → Activate
  }

  var Activated: bool = false
  var Visited Dirtmouth: bool = false
}
