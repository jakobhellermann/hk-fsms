# Control
# 1 location(s):
#   Library_04 / Music Control

fsm Control {
  start Pause

  state Pause {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=false)
    on FINISHED → Check
  }

  state Check {
    PlayerdataIntCompare(playerdataInt="enclaveLevel", compareTo=0, equal=→"FINISHED", lessThan=(none), greaterThan=(none))
    GetEntryGateName(StoreValue=var "Gate")
    StringCompare(stringVariable=var "Gate", compareTo="top1", equalEvent=→"TOP", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on TOP → Top
    on FINISHED → Normal
  }

  state Normal {
    ApplyMusicCue(musicCue=Vaults (audiocuesdynamic_assets_areabonearealibrary.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=4)
  }

  sequence state Top {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=4)
    Wait(time=2, finishEvent=(none), realTime=false)
    ApplyMusicCue(musicCue=Vaults (audiocuesdynamic_assets_areabonearealibrary.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=4)
  }

  var Gate: string = ""
}
