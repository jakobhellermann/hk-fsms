fsm Unlock Behaviour {
  start Init

  state Idle {
    PlayerDataVariableTest(VariableName="PilgrimsRestDoorBroken", ExpectedValue=true, IsExpectedEvent=→"START ENTRY", IsNotExpectedEvent=(none))
    on INTERACT → Dish Toll
    on START ENTRY → Entering
  }

  state End Interaction {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Idle
  }

  state Init {
    SetAnimator(target=Self, active=false)
    CallMethodProper(gameObject=var "Transition Point", behaviour="TransitionPoint", methodName="SetIsInactive", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Sequence Animation {
    SendMessageV2(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=StartAnimationControl(), everyFrame=false)
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SetAnimator(target=Self, active=true)
    AnimatorPlayStateWait(target=Self, stateName="Drop", finishEvent=→"FINISHED")
    on FINISHED → Open
  }

  state Open {
    CheckTrackTriggerCount(target=Black Thread States Thread Only Variant/Normal World/toll door interactible/Keep Open Range, count=0, test=Equal, everyFrame=true, successEvent=→"CLOSE")
    CallMethodProper(gameObject=var "Transition Point", behaviour="TransitionPoint", methodName="SetIsInactive", parameters=[false], storeResult=(unset var), EveryFrame=false)
    on CLOSE → Raise
  }

  state Entering {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SetAnimator(target=Self, active=true)
    AnimatorPlay(gameObject=Self, stateName="Dropped", layer=0, normalizedTime=0, everyFrame=false)
    PlayerDataVariableTest(VariableName="PilgrimsRestDoorBroken", ExpectedValue=true, IsExpectedEvent=→"FINISH ENTRY", IsNotExpectedEvent=(none))
    on FINISH ENTRY → Raise
  }

  state Entered {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  state Raise {
    PlayerDataVariableTest(VariableName="PilgrimsRestDoorBroken", ExpectedValue=true, IsExpectedEvent=→"CANCEL", IsNotExpectedEvent=(none))
    AnimatorPlayStateWait(target=Self, stateName="Raise", finishEvent=→"FINISHED")
    CallMethodProper(gameObject=var "Transition Point", behaviour="TransitionPoint", methodName="SetIsInactive", parameters=[true], storeResult=(unset var), EveryFrame=false)
    on CANCEL → Inert
    on FINISHED → Entered
  }

  state Inert {
    CallMethodProper(gameObject=var "Transition Point", behaviour="TransitionPoint", methodName="SetIsInactive", parameters=[false], storeResult=(unset var), EveryFrame=false)
  }

  state Dish Toll {
    RunFSM(fsmTemplateControl=template=-8990696591063553075 in[Dish<-var "Dish", Cost Reference<-var "Cost Reference"], finishEvent=→"FINISHED", everyFrame=false)
    on CANCEL → End Interaction
    on FINISHED → Sequence Animation
  }

  var Transition Point: gameObject = door1
  var Cost Reference: object = Pilgrims Rest Door (dataassets_assets_assets/dataassets/costs.bundle)

  var Cost: int = 100
  var Activated: bool = false
  var Dish Start Position: vector3 = (0, 0, 0)
  var Dish: gameObject = Black Thread States Thread Only Variant/Normal World/toll door interactible/door/dish
  var Door: gameObject = Black Thread States Thread Only Variant/Normal World/toll door interactible/door
  var Currency Type: enum = CurrencyType(0)
}
