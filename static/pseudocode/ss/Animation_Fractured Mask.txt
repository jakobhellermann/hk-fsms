# Animation
# 2 location(s):
#   coremanagers_assets__gamecameras.bundle / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Health/Health 1/Fractured Mask
#   Menu_Title / _GameCameras/HudCamera/In-game/Anchor TL/Hud Canvas Offset/Hud Canvas/Health/Health 1/Fractured Mask

// uses template: fractured_mask_health_display
fsm Animation {
  start Init

  state Inert {
    SetMeshRenderer(gameObject=Self, active=false)
    BoolTestMulti(boolVariables=[var "Is Active", var "Is Whole"], boolStates=[true, true], trueEvent=→"ACTIVATE", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on ACTIVATE → Activate
  }

  state Activate {
    SetMeshRenderer(gameObject=Self, active=true)
    CheckHudVisible(trueEvent=(none), falseEvent=(none), storeValue=var "Hud Visible", everyFrame=false)
    ConvertBoolToFloat(boolVariable=var "Hud Visible", floatVariable=var "Volume", falseValue=0, trueValue=1, everyFrame=false)
    PlayAudioEvent(audioClip=ui_hud_fractured_mask_slide_in (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=var "Volume", audioPlayerPrefab=Audio Player HUD@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fractured Mask Appear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Break {
    CheckHudVisible(trueEvent=(none), falseEvent=(none), storeValue=var "Hud Visible", everyFrame=false)
    ConvertBoolToFloat(boolVariable=var "Hud Visible", floatVariable=var "Volume", falseValue=0, trueValue=1, everyFrame=false)
    PlayAudioEvent(audioClip=ui_hud_fractured_mask_break (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=var "Volume", audioPlayerPrefab=Audio Player HUD@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fractured Mask Break", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Break Pt
  }

  state Idle {
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"SHINE", realTime=false)
    BoolTest(boolVariable=var "Is Active", isTrue=(none), isFalse=→"DEACTIVATE", everyFrame=true)
    BoolTest(boolVariable=var "Is Whole", isTrue=(none), isFalse=→"BREAK", everyFrame=true)
    on DEACTIVATE → Disappear
    on BREAK → Break
    on SHINE → Shine
  }

  state Shine {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fractured Mask Shine", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    BoolTest(boolVariable=var "Is Active", isTrue=(none), isFalse=→"DEACTIVATE", everyFrame=true)
    BoolTest(boolVariable=var "Is Whole", isTrue=(none), isFalse=→"BREAK", everyFrame=true)
    on FINISHED → Idle
    on DEACTIVATE → Disappear
    on BREAK → Break
  }

  state Init {
    FindChild(gameObject=Self, childName="Break Pt", storeResult=var "Break Pt")
    ActivateGameObject(gameObject=var "Break Pt", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inert
  }

  state Break Pt {
    ActivateGameObject(gameObject=var "Break Pt", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Pt", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Inert
  }

  state Disappear {
    CheckHudVisible(trueEvent=(none), falseEvent=(none), storeValue=var "Hud Visible", everyFrame=false)
    ConvertBoolToFloat(boolVariable=var "Hud Visible", floatVariable=var "Volume", falseValue=0, trueValue=1, everyFrame=false)
    PlayAudioEvent(audioClip=ui_hud_fractured_mask_slide_out (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=var "Volume", audioPlayerPrefab=Audio Player HUD@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Fractured Mask Disappear", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Inert
  }

  var Volume: float = 0
  var Is Active: bool = false
  var Is Whole: bool = false
  var Hud Visible: bool = false
  var Break Pt: gameObject = <null>
}
