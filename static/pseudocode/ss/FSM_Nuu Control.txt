# FSM
# 1 location(s):
#   Greymoor_03 / Nuu Control

// uses template: nuu_halfway_home_check
fsm FSM {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Nuu")
    FindNamedChild(gameObject=Self, storeResult=var "Nuu_Scrolls")
    on FINISHED → Check
  }

  state Check {
    PlayerDataBoolTest(boolName="nuuMementoAwarded", isTrue=→"LEFT", isFalse=(none))
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"HOME", isFalse=(none))
    PlayerDataVariableTest(VariableName="JournalIsCompleted", ExpectedValue=true, IsExpectedEvent=→"HOME", IsNotExpectedEvent=(none))
    CanEndQuestV2(RequireActive=true, CannotEndEvent=(none), CanEndEvent=→"HOME", StoreValue=(unset), Quest=Journal (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    PlayerDataBoolTest(boolName="nuuIsHome", isTrue=→"HOME", isFalse=→"AWAY")
    on HOME → Home
    on AWAY → Away
    on LEFT → Left
  }

  state Home {
    ActivateGameObject(gameObject=var "Nuu", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Nuu_Scrolls", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetPlayerDataBool(boolName="nuuIsHome", value=true)
  }

  state Away {
    ActivateGameObject(gameObject=var "Nuu", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Nuu_Scrolls", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Left {
    ActivateGameObject(gameObject=var "Nuu", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Nuu_Scrolls", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Nuu: gameObject = <null>
  var Nuu_Scrolls: gameObject = <null>
}
