# ride
# 1 location(s):
#   Hang_01 / Harpoon Ring VerticalRide (3)

// uses template: harpoon_ring_ride_vertical
fsm ride {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=0, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Move Point")
    FindNamedChild(gameObject=Self, storeResult=var "Turn Cog")
    GetPosition(gameObject=var "Move Point", vector=(unset), x=(unset), y=var "Move Y", z=(unset), space=0, everyFrame=false)
    GetParent(gameObject=Self, storeResult=var "Parent")
    FloatOperator(float1=var "Move Speed", float2=-1, operation=Multiply, storeResult=var "Move Speed Down", everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Ring Parent")
    FindNamedChild(gameObject=var "Ring Parent", storeResult=var "Ring")
    SetParent(gameObject=var "Turn Cog", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Bottom
  }

  state Bottom {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Start Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on SELF RING GRAB → Rise Pause
  }

  state Rise Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rise
    on SELF RING DETACH → Bottom
  }

  state Rise {
    AccelerateToY(gameObject=Self, accelerationFactor=0.25, targetSpeed=var "Move Speed")
    CheckYPosition(gameObject=Self, compareTo=var "Move Y", compareToOffset=0, tolerance=0, equal=→"FINISHED", lessThan=(none), greaterThan=→"FINISHED", everyFrame=true, space=0, activeBool=(unset))
    CheckYPosition(gameObject=Self, compareTo=var "Start Y", compareToOffset=-0.1, tolerance=0, equal=(none), lessThan=→"CLAMP SPEED", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    AudioPlayInState(gameObject=var "Up Audio Loop", volume=1)
    GetPosition2D(gameObject=Self, vector=(unset), x=(unset), y=var "Pos Y", space=0, everyFrame=true)
    FloatClamp(floatVariable=var "Pos Y", minValue=var "Start Y", maxValue=9999, everyFrame=true)
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Pos Y", Space=0, EveryFrame=true)
    Rotate(gameObject=var "Turn Cog", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=360, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on SELF RING DETACH → Fall
    on FINISHED → Top
    on CLAMP SPEED → Clamp Rise
  }

  state Fall {
    AccelerateToY(gameObject=Self, accelerationFactor=0.25, targetSpeed=var "Move Speed Down")
    CheckYPosition(gameObject=Self, compareTo=var "Start Y", compareToOffset=0, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    CheckYPosition(gameObject=Self, compareTo=var "Move Y", compareToOffset=0.1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CLAMP SPEED", everyFrame=true, space=0, activeBool=(unset))
    AudioPlayInState(gameObject=var "Down Audio Loop", volume=1)
    Rotate(gameObject=var "Turn Cog", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=-360, space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Arrive Bottom
    on SELF RING GRAB → Rise
    on CLAMP SPEED → Clamp Fall
  }

  state Top {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetPosition(gameObject=Self, vector=(unset), x=(unset), y=var "Move Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimation(gameObject=var "Ring", animLibName="", clipName="Wobble")
    PlayAudioEventRandom(audioClips=[harpoon_ring_shake_1 (sfxstatic_assets_areabellareacogarealibraryareamoss.bundle), harpoon_ring_shake_2 (sfxstatic_assets_areabellareamoss.bundle), harpoon_ring_shake_3 (sfxstatic_assets_areamoss.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=Harpoon_Ring_VerticalRide_arrive_bottom (sfxstatic_assets_areahang.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="RING DROP IMPACT", delay=0, everyFrame=false)
    on SELF RING DETACH → Fall
  }

  state Arrive Bottom {
    Tk2dPlayAnimation(gameObject=var "Ring", animLibName="", clipName="Wobble")
    PlayAudioEventRandom(audioClips=[harpoon_ring_shake_1 (sfxstatic_assets_areabellareacogarealibraryareamoss.bundle), harpoon_ring_shake_2 (sfxstatic_assets_areabellareamoss.bundle), harpoon_ring_shake_3 (sfxstatic_assets_areamoss.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=Harpoon_Ring_VerticalRide_arrive_top (sfxstatic_assets_areahang.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Bottom
  }

  state Clamp Fall {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Move Y", Space=0, EveryFrame=false)
    on FINISHED → Fall
  }

  state Clamp Rise {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=0, everyFrame=false)
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Start Y", Space=0, EveryFrame=false)
    on FINISHED → Rise
  }

  var Move Time: float = 1.5
  var Move Speed: float = 15
  var Up Audio Loop: gameObject = Harpoon Ring VerticalRide (3)/Up Audio Loop
  var Down Audio Loop: gameObject = Harpoon Ring VerticalRide (3)/Down Audio Loop

  var Start Y: float = 0
  var Move Y: float = 0
  var Target Y: float = 0
  var Move Speed Down: float = 0
  var Pos Y: float = 0
  var Returning: bool = false
  var Move Point: gameObject = <null>
  var Parent: gameObject = <null>
  var Ring: gameObject = <null>
  var Ring Parent: gameObject = <null>
  var Turn Cog: gameObject = <null>
}
