# Wake Up
# 1 location(s):
#   Memory_Red / Memory Control/door_wakeInMemory

fsm Wake Up {
  start Pause

  state Pause {
    SetGameObjectSelf(variable=var "Self", gameObject=Self, everyFrame=false)
    on TEST → Take Control
    on START ENTRY → Set Respawn?
    on HERO RESPAWNED → Wait For Scene Enter
  }

  state Regain Control {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RegainControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StartAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    BoolTest(boolVariable=var "Return HUD", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on FINISHED → Save?
  }

  state Take Control {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT INSTANT", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetMeshRenderer(gameObject=var "Hero", active=true)
    on FINISHED → Fade Up
  }

  state Fade Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Prostrate", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ConvertBoolToFloat(boolVariable=var "Activated", floatVariable=var "Duration", falseValue=7, trueValue=4, everyFrame=false)
    ScreenFaderAlpha(EndAlpha=0, Duration=var "Duration")
    Wait(time=var "Duration", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Get Up
  }

  state Get Up {
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Prostrate Rise", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayAudioEvent(audioClip=hornet_misc_foley_move_1 (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEventDelayed(audioClip=hornet_misc_foley_move_2 (herosfxstatic_assets_all.bundle), delay=1, pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Regain Control
  }

  state Save? {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    BoolTest(boolVariable=var "Save Game", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SaveGame()
    on FINISHED → End
  }

  state End {
    SendEventByNameUpwards(Target=Self, EventName="DOOR GET UP FINISHED")
    RemoveHeroInputBlocker(Blocker=Self)
  }

  state Blank Screen {
    ScreenFader(startColour=(1, 1, 1, 1), endColour=(1, 1, 1, 1), duration=0)  // disabled
    on FINISH ENTRY → Take Control
  }

  sequence state Wait For Scene Enter {
    AddHeroInputBlocker(Blocker=Self)
    ActivateGameObject(gameObject=var "z_ActivateOnStartEntry", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    WaitForFinishedEnteringScene(sendEvent=(none))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Take Control
  }

  state Set Respawn? {
    AddHeroInputBlocker(Blocker=Self)
    ActivateGameObject(gameObject=var "z_ActivateOnStartEntry", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    BoolTest(boolVariable=var "Set Death Respawn", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetDeathRespawnMarker(Target=Memory Control/door_wakeInMemory/Death Respawn Marker)
    on FINISHED → Blank Screen
  }

  var Return HUD: bool = false
  var Save Game: bool = false
  var Set Death Respawn: bool = true
  var z_ActivateOnStartEntry: gameObject = Scenery Groups/Entry Scenery

  var Duration: float = 0
  var Activated: bool = false
  var Self: gameObject = <null>
}
