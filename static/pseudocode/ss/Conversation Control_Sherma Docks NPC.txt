# Conversation Control
# 1 location(s):
#   Bone_East_10 / Black Thread States Thread Only Variant/Normal World/Pilgrim NPC Scene/2 Brolly Revisit/Sherma Docks NPC

fsm Conversation Control {
  start Init

  state Init {
    on FINISHED → Idle Start Chiming
  }

  state Idle Start Chiming {
    SetBoolValue(boolVariable=var "Chiming", boolValue=true, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=1.2, MinReactDelay=1.4, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SetRandomAudioClipFromTable(Table=sherma_sing (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0.5)
    on INTERACT → Convo Check
    on SING → Sing
    on STARTLE → Chime Startle
  }

  state Convo Check {
    SendEvent(eventTarget=Self, sendEvent=→"TALK1", delay=0, everyFrame=false)  // disabled
    AudioStopV2(gameObject=var "Audio Loop Voice", fadeTime=0, cancelOnEarlyExit=false)
    SetPlayerDataBool(boolName="metSherma", value=true)
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="SHERMA")
    IntSwitch(intVariable=var "Value", compareTo=[0, 1, 2], sendEvent=[→"MEET", →"TALK1", →"REPEAT"], everyFrame=false)
    on MEET → Meet1
    on TALK1 → Talk1 Anim
    on REPEAT → Repeat
  }

  state Meet1 {
    IntAdd(intVariable=var "Value", add=1, everyFrame=false)
    RunDialogueV4(CustomText="", Sheet="Wanderers", Key="SHERMA_PILGRIMSREST_MEET", TextColor=(unset), NpcVoiceTableOverride=sherma_talk_sing (sfxdynamic_assets_sherma.bundle), OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on LINE_END → End Excite
  }

  state End Excite {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED END", delay=0, everyFrame=false)
    on READY → Meet2
  }

  state Meet2 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookL")
    on CONVO_END → Talk End
  }

  state Idle Watching {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    FaceObjectAnim(turningObject=Self, target=var "Hero", clipL="LookR", clipR="LookL", startFacingRight=true, everyFrame=true)  // disabled
    SetBoolValue(boolVariable=var "Chiming", boolValue=false, everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Chime Range R", storeResult=false, sendEvent="START CHIME", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Chime Range L", storeResult=false, sendEvent="START CHIME", outOfRangeEvent="", everyFrame=true)
    Wait(time=2, finishEvent=→"START CHIME", realTime=false)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=1, MinReactDelay=1, MaxReactDelay=0, None=(none), ActiveInner=→"START CHIME", ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    CheckPassedTarget(Self=Self, Target=var "Hero", IsActive=true, DefaultFacingRight=false, DistancePast=0, StoreResult=var "Chiming", PassedEvent=→"START CHIME", NotPassedEvent=(none), EveryFrame=true)
    on INTERACT → Convo Check
    on START CHIME → To Chime
    on STARTLE → Startle Idle
  }

  state Talk End {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle Watching
  }

  state To Chime {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle Start Chiming
  }

  state Turn? {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Turn", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on FINISHED → To Chime
  }

  state Talk1 Anim {
    BoolTest(boolVariable=var "Chiming", isTrue=(none), isFalse=→"READY", everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED END", delay=0, everyFrame=false)
    on READY → Talk1
  }

  state Talk1 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookL")
    IntAdd(intVariable=var "Value", add=1, everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key="SHERMA_PILGRIMSREST_TALK1", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Restart Chime
  }

  state Repeat {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="LookL")  // disabled
    IntAdd(intVariable=var "Value", add=1, everyFrame=false)  // disabled
    RunDialogueV4(CustomText="", Sheet="Wanderers", Key="SHERMA_PILGRIMSREST_REPEAT", TextColor=(unset), NpcVoiceTableOverride=sherma_talk_sing (sfxdynamic_assets_sherma.bundle), OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → Talk End To Chime
  }

  state Restart Chime {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED", delay=0, everyFrame=false)
    on FINISHED → Talk End To Chime
  }

  state Talk End To Chime {
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle Start Chiming
  }

  state Sing {
    CheckHeroPerformanceRegionV2(Target=Self, Radius=0.4, MinReactDelay=0.6, MaxReactDelay=0, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=<null>, noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on END → Idle Keep Chiming
  }

  sequence state Startle Idle {
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=sherma_alert (sfxdynamic_assets_sherma.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CheckPassedTarget(Self=Self, Target=var "Hero", IsActive=true, DefaultFacingRight=true, DistancePast=0, StoreResult=var "Hero Behind", PassedEvent=(none), NotPassedEvent=(none), EveryFrame=false)
    ConvertBoolToString(boolVariable=var "Hero Behind", stringVariable=var "Anim", falseString="Startle F", trueString="Startle B", everyFrame=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName=var "Anim", AnimationCompleteEvent=(none))
    ConvertBoolToString(boolVariable=var "Hero Behind", stringVariable=var "Anim", falseString="Idle", trueString="Idle Left", everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim")
    on FINISHED → Idle Watching
  }

  sequence state Chime Startle {
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=sherma_alert (sfxdynamic_assets_sherma.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="GO IDLE", delay=0, everyFrame=false)
    Tk2dPlayAnimationWait(Target=Self, ClipName="Chiming Startle", AnimationCompleteEvent=(none))
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED", delay=0, everyFrame=false)
    Wait(time=0.25, finishEvent=(none), realTime=false)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle Start Chiming
  }

  state Idle Keep Chiming {
    SetBoolValue(boolVariable=var "Chiming", boolValue=true, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    CheckHeroPerformanceRegionV2(Target=Self, Radius=1.2, MinReactDelay=1.4, MaxReactDelay=0, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, UseActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    SetRandomAudioClipFromTable(Table=sherma_sing (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=0.75)
    on INTERACT → Convo Check
    on SING → Sing
    on STARTLE → Chime Startle
  }

  var Value: int = 0
  var Chiming: bool = false
  var Hero Behind: bool = false
  var Anim: string = ""
  var Audio Loop Voice: gameObject = Black Thread States Thread Only Variant/Normal World/Pilgrim NPC Scene/2 Brolly Revisit/Sherma Docks NPC/Audio Loop Voice
}
