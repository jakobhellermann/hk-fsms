# Arena Setup
# 1 location(s):
#   Memory_Coral_Tower / Battle Scenes/Battle Scene Chamber 4/Wave 10 - destroy outer plats DEPRECATED

fsm Arena Setup {
  start State 2

  sequence state State 2 {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    on WAVE START → Pause
  }

  state Pause {
    SetHazardRespawn(hazardRespawnMarker=var "Respawn Marker")
    GetConstantsValue(variableName="CORAL_SPEAR_SPAWN_TIME", storeValue=var "Spawn Time")
    FloatAdd(floatVariable=var "Spawn Time", add=-0.65, everyFrame=false, perSecond=false)
    Wait(time=var "Spawn Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Break
  }

  state Break {
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Platform Set B")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Platform Break Effect B")
    ActivateGameObject(gameObject=var "Platform Set B", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Platform Break Effect B", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Respawn Marker: gameObject = Battle Scenes/Battle Scene Chamber 4/Hazard Respawn Marker Mid

  var Spawn Time: float = 0
  var Battle Scene: gameObject = <null>
  var Platform Break Effect B: gameObject = <null>
  var Platform Set B: gameObject = <null>
}
