# Cylinder States

    on CYLINDER COMPLETED → All Complete?
    on CYLINDER ALREADY COMPLETE → All Complete? 2
    on CANCEL → Has Melody
  }

  state All Complete? {
    BoolAllTrue(boolVariables=[var "Cylinder 1 Complete", var "Cylinder 2 Complete", var "Cylinder 3 Complete", var "Cylinder 4 Complete"], sendEvent=→"TRUE", storeResult=(unset), everyFrame=false)

    on TRUE → Just Complete
  }

  state All Complete? 2 {
    BoolAllTrue(boolVariables=[var "Cylinder 1 Complete", var "Cylinder 2 Complete", var "Cylinder 3 Complete", var "Cylinder 4 Complete"], sendEvent=→"TRUE", storeResult=(unset), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"FALSE", delay=0, everyFrame=false)

    on FINISHED → Pressure Plate Raise
  }

    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Prompt Control"), sendEvent="SHOW PROMPT", delay=0, everyFrame=false)
    on GET ITEM MSG COVERED → Singing End
  }

  sequence state Singing {

    SetAudioVolume(gameObject=var "Rumble Audio", volume=0.3, everyFrame=false)

    Wait(time=1.5, finishEvent=(none), realTime=false)
    SetAudioVolume(gameObject=var "Rumble Audio", volume=0.2, everyFrame=false)
    CallMethodProper(gameObject=var "Cylinder 2", behaviour="CogCylinderPuzzle", methodName="StartChoir", parameters=[], storeResult=(unset var), EveryFrame=false)
    Wait(time=1.5, finishEvent=(none), realTime=false)

    Wait(time=2, finishEvent=(none), realTime=false)
    on FINISHED → Singing End 2
  }

  state Start {
    CallMethodProper(gameObject=var "Pressure Plate", behaviour="TempPressurePlate", methodName="ActivateSilent", parameters=[], storeResult=(unset var), EveryFrame=false)
    FadeNestedFadeGroup(Target=var "BG 1 Fader", ToAlpha=1, FadeTime=0)
    FadeNestedFadeGroup(Target=var "BG 2 Fader", ToAlpha=1, FadeTime=0)

    ActivateGameObject(gameObject=var "Post Needolin Override", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

    on FINISHED → BG Choir Off
  }

    FindChild(gameObject=var "Cylinder 1", childName="Choir Audio", storeResult=var "Obj")
    ActivateGameObject(gameObject=var "Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Cylinder 2", behaviour="CogCylinderPuzzle", methodName="SetComplete", parameters=[], storeResult=(unset var), EveryFrame=false)
    FindChild(gameObject=var "Cylinder 2", childName="Choir Audio", storeResult=var "Obj")
    ActivateGameObject(gameObject=var "Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Cylinder 3", behaviour="CogCylinderPuzzle", methodName="SetComplete", parameters=[], storeResult=(unset var), EveryFrame=false)
    FindChild(gameObject=var "Cylinder 3", childName="Choir Audio", storeResult=var "Obj")
    ActivateGameObject(gameObject=var "Obj", activate=false, recursive=false, resetOnExit=false, everyFrame=false)

    FadeNestedFadeGroup(Target=var "BG 2 Fader", ToAlpha=0.8, FadeTime=0)
    ActivateGameObject(gameObject=var "Post Needolin Override", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Has Melody Idle
  }

    CallMethodProper(gameObject=var "Pressure Plate", behaviour="TempPressurePlate", methodName="Deactivate", parameters=[], storeResult=(unset var), EveryFrame=false)

  }

    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=3)
    SendEventByNameV2(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    SendEventToRegister(eventName="MELODY SEQUENCE START")
    AudioPlaySimple(gameObject=var "Rumble Audio", volume=0.35, oneShotClip=<null>)

  }

    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)

    SendEventToRegister(eventName="BG CHOIR 1 ON")
    Wait(time=3, finishEvent=(none), realTime=false)
    FadeNestedFadeGroup(Target=var "BG 2 Fader", ToAlpha=0.2, FadeTime=0.1)
    SendEventToRegister(eventName="BG CHOIR 2 ON")
    Wait(time=5, finishEvent=(none), realTime=false)

  }

    StopParticleEmitter(gameObject=var "Sing Rubble")

    FadeNestedFadeGroup(Target=var "BG 1 Fader", ToAlpha=0.6, FadeTime=0)
    SendEventToRegister(eventName="BG CHOIR 2 OFF")
    FadeNestedFadeGroup(Target=var "BG 2 Fader", ToAlpha=0.8, FadeTime=0)

  }

    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopRoarLock", parameters=[], storeResult=(unset var), EveryFrame=false)

    SetPlayerDataVariable(VariableName="disablePause", SetValue=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0, MaxReactDelay=0, None=(none), ActiveInner=(none), ActiveOuter=→"SING", IgnoreNeedolinRange=true, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)

    AudioPlaySimple(gameObject=var "All Choir Audio", volume=1, oneShotClip=<null>)

  }
