# Control
# 1 location(s):
#   enemycorpses_assets_areaweaver.bundle / Corpse Weaver Servitor Large/Chunks/Chunk

// uses template: corpse_servitor_chunk
fsm Control {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Hit")
    FindNamedChild(gameObject=Self, storeResult=var "Break Fx")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"BREAK", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=true, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"NAIL", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=var "Hit Direction", storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on BREAK → Pay Shards
    on NAIL → Nail Dir
  }

  state Break {
    PlayAudioEvent(audioClip=shell_shard_rock_break (sfxstatic_assets_areabone.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetMeshRenderer(gameObject=Self, active=false)
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    ActivateGameObject(gameObject=var "Break Fx", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
  }

  state Launch L {
    SetVelocityAsAngle(gameObject=Self, angle=150, speed=15, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.2, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Launch Frame
  }

  state Launch Frame {
    NextFrameEvent(sendEvent=→"FINISHED")
    GetVelocity2d(gameObject=var "Self", vector=(unset), x=var "Speed", y=(unset), space=0, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=15, everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=-1, everyFrame=false)
    AddTorque2d(gameObject=var "Self", forceMode=0, torque=var "Speed", everyFrame=false)
    on FINISHED → Idle
  }

  state Launch R {
    SetVelocityAsAngle(gameObject=Self, angle=30, speed=15, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.2, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Launch Frame
  }

  state Launch U {
    SetVelocityAsAngle(gameObject=Self, angle=90, speed=25, everyFrame=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.2, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Launch Frame
  }

  state Launch D {
    SetVelocityAsAngle(gameObject=Self, angle=270, speed=20, everyFrame=false)
    on FINISHED → Launch Frame
  }

  state Nail Dir {
    PlayAudioEvent(audioClip=sword_hit_reject (sfxstatic_assets_shared.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitter(gameObject=var "Pt Hit", emit=0, resetIfPlaying=false)
    SelectRandomGameObject(gameObjects=[Shell Shard 01 (globalpoolprefabs_assets_all.bundle), Shell Shard 02 (globalpoolprefabs_assets_all.bundle), Shell Shard 03 (globalpoolprefabs_assets_all.bundle)], weights=[1, 1, 1], storeGameObject=var "Shard")
    FlingObjectsFromGlobalPool(gameObject=var "Shard", spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=15, speedMax=20, angleMin=40, angleMax=140, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    IntAdd(intVariable=var "HP", add=-1, everyFrame=false)
    IntAdd(intVariable=var "Shards", add=-1, everyFrame=false)
    IntCompare(integer1=var "HP", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    PlayAudioEventRandom(audioClips=[shard_region_hit_1 (sfxstatic_assets_areabone.bundle), shard_region_hit_2 (sfxstatic_assets_areabone.bundle), shard_region_hit_3 (sfxstatic_assets_areabone.bundle), shard_region_hit_4 (sfxstatic_assets_areabone.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FloatCompare(float1=var "Hit Direction", float2=0, tolerance=30, equal=→"TINK RIGHT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Hit Direction", float2=90, tolerance=30, equal=→"TINK UP", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Hit Direction", float2=180, tolerance=30, equal=→"TINK LEFT", lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Hit Direction", float2=270, tolerance=30, equal=→"TINK DOWN", lessThan=(none), greaterThan=(none), everyFrame=false)
    SendEvent(eventTarget=Self, sendEvent=→"TINK UP", delay=0, everyFrame=false)
    on TINK DOWN → Launch D
    on TINK LEFT → Launch L
    on TINK RIGHT → Launch R
    on TINK UP → Launch U
    on BREAK → Pay Shards
  }

  state Pay Shards {
    IntCompare(integer1=var "Shards", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    SelectRandomGameObject(gameObjects=[Shell Shard 01 (globalpoolprefabs_assets_all.bundle), Shell Shard 02 (globalpoolprefabs_assets_all.bundle), Shell Shard 03 (globalpoolprefabs_assets_all.bundle)], weights=[1, 1, 1], storeGameObject=var "Shard")
    FlingObjectsFromGlobalPool(gameObject=var "Shard", spawnPoint=var "Self", position=(unset), spawnMin=1, spawnMax=1, speedMin=15, speedMax=20, angleMin=40, angleMax=140, originVariationX=0.25, originVariationY=0.25, FSM="", FSMEvent="")
    IntAdd(intVariable=var "Shards", add=-1, everyFrame=false)
    on FINISHED → Pay Shards
    on BREAK → Break
  }

  var HP: int = 3
  var Shards: int = 5

  var Speed: float = 0
  var Hit Direction: float = 0
  var Self: gameObject = <null>
  var Break Fx: gameObject = <null>
  var Pt Hit: gameObject = <null>
  var Shard: gameObject = <null>
}
