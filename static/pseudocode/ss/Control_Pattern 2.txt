// uses template: cog_dancer_dive_pattern
fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on ATTACK → Shuffle
  }

  state Shuffle {
    SetBoolValue(boolVariable=var "Reset Array", boolValue=true, everyFrame=false)
    ArrayShuffle(array=var "Attacks", startIndex=(unset), shufflingRange=(unset))
    on FINISHED → Attack
  }

  state Attack {
    ArrayGetNext(array=var "Attacks", startIndex=0, endIndex=0, loopEvent=(none), resetFlag=var "Reset Array", finishedEvent=→"ATTACK END", result=var "Attack", currentIndex=(unset))
    ActivateGameObject(gameObject=var "Attack", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.075, finishEvent=→"FINISHED", realTime=false)
    on ATTACK END → Idle
    on FINISHED → Attack
  }

  var Attacks: array = var "Attacks"

  var Reset Array: bool = false
  var Attack: gameObject = <null>
}
