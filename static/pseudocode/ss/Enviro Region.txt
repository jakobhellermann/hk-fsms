# Enviro Region
# 1 location(s):
#   Dock_03c / Enviro Region (2)

// uses template: enviro_region
fsm Enviro Region {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetPlayerDataInt(intName="environmentTypeDefault", storeValue=var "Enviro Default")
    on FINISHED → Detect
  }

  state Detect {
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter
  }

  state Enter {
    SetPlayerDataInt(intName="environmentType", value=var "Enviro Type")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=checkEnvironment())
    Trigger2dEvent(gameObject=Self, trigger=2, collideTag="", sendEvent=→"EXIT", storeCollider=var "")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ENVIRO UPDATE", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="ENVIRO UPDATE")
    on EXIT → Exit
  }

  state Exit {
    SetPlayerDataInt(intName="environmentType", value=var "Enviro Default")
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=checkEnvironment())
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    SendEventByName(eventTarget=BroadcastAll, sendEvent="ENVIRO UPDATE", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="ENVIRO UPDATE")
    on ENTER → Enter
  }

  var Enviro Type: int = 4

  var Enviro Default: int = 0
}
