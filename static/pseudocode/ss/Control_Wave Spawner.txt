# Control
# 1 location(s):
#   Cog_05 / Battle Scene/Aftermath/Wave Spawner

fsm Control {
  start Idle

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    GetRandomChild(gameObject=Self, storeResult=var "Wave")
    ActivateGameObject(gameObject=var "Wave", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on ENTER → Spawn
  }

  state Spawn {
    SendEventByName(eventTarget=GameObject(var "Wave"), sendEvent="SPAWN", delay=0, everyFrame=false)
  }

  var Wave: gameObject = <null>
}
