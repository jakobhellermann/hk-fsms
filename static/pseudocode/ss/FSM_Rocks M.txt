# FSM
# 3 location(s):
#   Bone_East_08 / Boss Scene/Lava Rocks/Rocks M
#   Bone_East_08 / Boss Scene/Lava Rocks/Rocks R
#   Bone_East_08 / Boss Scene/Lava Rocks/Rocks L

// uses template: song_golem_rockdropper
fsm FSM {
  start Init
  on GOLEM DEATH → End  // from any state
  on DROP → Set Drop  // from any state

  state Init {
    FindChild(gameObject=Self, childName="Rock Pool", storeResult=var "Rock Pool")
    FindChild(gameObject=Self, childName="Points", storeResult=var "Points")
    on FINISHED → Dormant
  }

  state Dormant {
  }

  state Drop {
    IntCompare(integer1=var "Drops", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    IntOperator(integer1=var "Drops", integer2=1, operation=Subtract, storeResult=var "Drops", everyFrame=false)
    GetRandomChild(gameObject=var "Points", storeResult=var "Next Point")
    GetRandomChild(gameObject=var "Rock Pool", storeResult=var "Next Rock")
    RandomFloat(min=-0.9, max=0.9, storeResult=var "Offset")
    SetPositionToObject(gameObject=var "Next Rock", targetObject=var "Next Point", xOffset=var "Offset", yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Next Rock"), sendEvent="DROP", delay=0, everyFrame=false)
    WaitRandom(timeMin=var "Wait Min", timeMax=var "Wait Max", finishEvent=→"FINISHED", realTime=false)
    on END → Dormant
    on FINISHED → Loop
  }

  state Loop {
    IntCompare(integer1=var "Drops", integer2=3, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    SetFloatValue(floatVariable=var "Wait Min", floatValue=1.2, everyFrame=false)
    SetFloatValue(floatVariable=var "Wait Max", floatValue=1.3, everyFrame=false)
    on FINISHED → Drop
  }

  state End {
  }

  state Set Drop {
    SetIntValue(intVariable=var "Drops", intValue=5, everyFrame=false)
    SetFloatValue(floatVariable=var "Wait Min", floatValue=0.8, everyFrame=false)
    SetFloatValue(floatVariable=var "Wait Max", floatValue=0.9, everyFrame=false)
    on FINISHED → Drop
  }

  var Offset: float = 0
  var Wait Min: float = 0.7
  var Wait Max: float = 0.9
  var Drops: int = 0
  var Next Point: gameObject = <null>
  var Next Rock: gameObject = <null>
  var Points: gameObject = <null>
  var Rock Pool: gameObject = <null>
}
