# tilt_plat
# 1 location(s):
#   Greymoor_03 / Black Thread States Thread Only Variant/Normal World/Strut Structure/Tilt Plat/terrain collider

// uses template: tilt_plat
fsm tilt_plat {
  start Init
  on READY → Idle  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    GetGrandParent(gameObject=Self, storeResult=var "Grandparent")
    FindChild(gameObject=var "Parent", childName="Plat Dust", storeResult=var "Plat Dust")
    FindChild(gameObject=var "Parent", childName="Water Drips", storeResult=var "Water Drips")
    FindChild(gameObject=var "Parent", childName="Art", storeResult=var "Art")
    FindChild(gameObject=var "Parent", childName="Art_Drop", storeResult=var "Art Drop")
    FindNamedChild(gameObject=var "Grandparent", storeResult=var "Pt Follow")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Land Audio 1")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Land Audio 2")
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    AddEventRegister(target=Self, eventName="HAZARD RESPAWNED")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Start
  }

  state Idle {
    CollisionEnter2DChildren(Target=Self, CollisionType=Enter, CollideTag="", CollideLayer=Player(9), StoreNormal=var "Normal", SendEvent=→"COLLIDE")
    on COLLIDE → Check
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink Up
  }

  state Check {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=1, everyFrame=false, lateUpdate=false)
    SetPosition(gameObject=var "Art Drop", vector=(unset), x=0, y=0, z=0, space=1, everyFrame=false, lateUpdate=false)
    GetVector2XY(vector2Variable=var "Normal", storeX=(unset), storeY=var "Normal Y", everyFrame=false)
    FloatCompare(float1=var "Normal Y", float2=-1, tolerance=0.01, equal=→"LAND", lessThan=→"FINISHED", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Idle
    on LAND → Effects?
  }

  state Down {
    SetFloatValue(floatVariable=var "Prev Tilt", floatValue=var "Tilt", everyFrame=true)
    GetPosition(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), z=(unset), space=0, everyFrame=false)
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=var "Drop Vector", time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    SetPosition(gameObject=var "Art Drop", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art Drop", shiftBy=var "Drop Vector Just", time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    CollisionEnter2DChildren(Target=Self, CollisionType=Exit, CollideTag="", CollideLayer=Player(9), StoreNormal=(unset), SendEvent=→"UNCOLLIDE")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Self X", float2=var "Hero X", operation=Subtract, storeResult=var "Tilt", everyFrame=true)
    FloatMultiply(floatVariable=var "Tilt", multiplyBy=var "Tilt Factor", everyFrame=true)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale X", yScale=var "Scale Y", zScale=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Scale X", float2=var "Scale Y", operation=Multiply, storeResult=var "Scale Sign", everyFrame=false)
    FloatSign(Value=var "Scale Sign", StoreValue=var "Scale Sign", EveryFrame=false)
    FloatOperator(float1=var "Tilt", float2=var "Scale Sign", operation=Multiply, storeResult=var "Tilt", everyFrame=true)
    SetRotationTimer(gameObject=var "Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tilt", time=0.1, space=1)
    PlayParticleEmitterInState(gameObject=var "Pt Follow")
    SetPosition(gameObject=var "Pt Follow", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatTestToBool(float1=var "Tilt", float2=var "Prev Tilt", tolerance=0.001, equalBool=var "Not Moving", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=true)
    SetBoolValueAtTime(BoolVariable=var "Creak Ready", BoolValue=true, Time=var "Creak Pause Time", SetOppositeOnStateEntry=true)
    BoolTestMulti(boolVariables=[var "Creak Ready", var "Not Moving"], boolStates=[true, false], trueEvent=→"CREAK", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on UNCOLLIDE → Set Rotation
    on HAZARD RESPAWNED → Set Rotation
    on CREAK → Creak And Stay Down
  }

  state Up {
    AnimatePositionTo(gameObject=var "Art", toValue=(0, 0, 0), localSpace=true, time=0.25, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=→"FINISHED", realTime=false)
    EaseFloat(fromValue=var "Start Rotation", toValue=0, floatVariable=var "Rotation", time=0.25, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=(none), realTime=false)
    SetRotation(gameObject=var "Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=1, everyFrame=true, lateUpdate=false)
    AnimatePositionTo(gameObject=var "Art Drop", toValue=(0, 0, 0), localSpace=true, time=0.25, speed=(unset), delay=(unset), easeType=25, reverse=false, finishEvent=→"FINISHED", realTime=false)
    CollisionEnter2DChildren(Target=Self, CollisionType=Enter, CollideTag="", CollideLayer=Player(9), StoreNormal=var "Normal", SendEvent=→"COLLIDE")
    on FINISHED → Idle
    on COLLIDE → Check
  }

  state Tink L {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(-0.1, 0, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink R {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(0.1, 0, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink Up {
    SetPosition(gameObject=var "Art", vector=(unset), x=0, y=0, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionBy(gameObject=var "Art", shiftBy=(0, 0.1, 0), time=0.05, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Tink Return
  }

  state Tink Return {
    AnimatePositionTo(gameObject=var "Art", toValue=(0, 0, 0), localSpace=true, time=0.1, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  state Effects? {
    SendEventByNameUpwards(Target=Self, EventName="TILT PLAT LANDED")
    CheckCurrentMapZoneEnum(MapZone=RED_CORAL_GORGE, trueEvent=→"CORAL", falseEvent=→"FINISHED", storeValue=(unset))
    on CORAL → Coral Effects
    on FINISHED → Reg Effects
  }

  state Reg Effects {
    PlayParticleEmitter(gameObject=var "Plat Dust", emit=0, resetIfPlaying=false)
    on FINISHED → Land Audio
  }

  state Coral Effects {
    PlayParticleEmitter(gameObject=var "Water Drips", emit=0, resetIfPlaying=false)
    on FINISHED → Land Audio
  }

  state Set Rotation {
    GetRotation(gameObject=var "Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Start Rotation", space=1, everyFrame=false)
    FloatCompare(float1=var "Start Rotation", float2=180, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false)
    FloatAdd(floatVariable=var "Start Rotation", add=-360, everyFrame=false, perSecond=false)
    on FINISHED → Up
  }

  state Land Audio {
    AudioPlaySimple(gameObject=var "Land Audio 1", volume=(unset), oneShotClip=<null>)
    AudioPlaySimple(gameObject=var "Land Audio 2", volume=(unset), oneShotClip=<null>)
    PlayRandomAudioClipTable(Table=var "Land Audio Clip Table", AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Down
  }

  state Creak And Stay Down {
    SetFloatValue(floatVariable=var "Prev Tilt", floatValue=var "Tilt", everyFrame=true)
    PlayRandomAudioClipTable(Table=var "Creak Audio Clip Table", AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    CollisionEnter2DChildren(Target=Self, CollisionType=Exit, CollideTag="", CollideLayer=Player(9), StoreNormal=(unset), SendEvent=→"UNCOLLIDE")
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true)
    FloatOperator(float1=var "Self X", float2=var "Hero X", operation=Subtract, storeResult=var "Tilt", everyFrame=true)
    FloatMultiply(floatVariable=var "Tilt", multiplyBy=var "Tilt Factor", everyFrame=true)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Scale X", yScale=var "Scale Y", zScale=(unset), space=0, everyFrame=false)
    FloatOperator(float1=var "Scale X", float2=var "Scale Y", operation=Multiply, storeResult=var "Scale Sign", everyFrame=false)
    FloatSign(Value=var "Scale Sign", StoreValue=var "Scale Sign", EveryFrame=false)
    FloatOperator(float1=var "Tilt", float2=var "Scale Sign", operation=Multiply, storeResult=var "Tilt", everyFrame=true)
    SetRotationTimer(gameObject=var "Art", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Tilt", time=0.1, space=1)
    PlayParticleEmitterInState(gameObject=var "Pt Follow")
    SetPosition(gameObject=var "Pt Follow", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=true, lateUpdate=false)
    FloatTestToBool(float1=var "Tilt", float2=var "Prev Tilt", tolerance=0.001, equalBool=var "Not Moving", lessThanBool=(unset), greaterThanBool=(unset), everyFrame=true)
    SetBoolValueAtTime(BoolVariable=var "Creak Ready", BoolValue=true, Time=var "Creak Pause Time", SetOppositeOnStateEntry=true)
    BoolTestMulti(boolVariables=[var "Creak Ready", var "Not Moving"], boolStates=[true, false], trueEvent=→"CREAK", falseEvent=(none), storeResult=(unset), everyFrame=true)
    on UNCOLLIDE → Set Rotation
    on HAZARD RESPAWNED → Set Rotation
    on CREAK → Creak And Stay Down
  }

  state Start {
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=1, everyFrame=false, lateUpdate=false)
    on FINISHED → Idle
  }

  var Tilt Factor: float = 1
  var Drop Vector: vector3 = (0, -0.1, 0)
  var Drop Vector Just: vector3 = (0, 0, 0)
  var Land Audio Clip Table: object = wood_creak (sfxdynamic_assets_areashellwood.bundle)
  var Creak Audio Clip Table: object = wood_creak (sfxdynamic_assets_areashellwood.bundle)

  var Normal Y: float = 0
  var Tilt: float = 0
  var Start Rotation: float = 0
  var Hero X: float = 0
  var Self X: float = 0
  var Rotation: float = 0
  var Scale X: float = 0
  var Scale Y: float = 0
  var Scale Sign: float = 0
  var Prev Tilt: float = 0
  var Creak Pause Time: float = 1.5
  var Creak Ready: bool = false
  var Not Moving: bool = false
  var Normal: vector2 = (0, 0)
  var Art: gameObject = <null>
  var Collider: gameObject = <null>
  var Parent: gameObject = <null>
  var Plat Dust: gameObject = <null>
  var Water Drips: gameObject = <null>
  var Pt Follow: gameObject = <null>
  var Grandparent: gameObject = <null>
  var Art Drop: gameObject = <null>
  var Land Audio 1: gameObject = <null>
  var Land Audio 2: gameObject = <null>
}
