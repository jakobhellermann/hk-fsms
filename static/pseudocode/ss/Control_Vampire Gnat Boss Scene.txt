# Control
# 1 location(s):
#   Greymoor_05_boss / Vampire Gnat Boss Scene

fsm Control {
  start Init
  on BOSS DEATH → Clear Out Scene  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Fight Start Range")
    FindNamedChild(gameObject=Self, storeResult=var "Vampire Gnat")
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Fight Start Range", trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter
  }

  state Enter {
    ActivateGameObject(gameObject=var "Fight Start Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmFloat(gameObject=var "Vampire Gnat", fsmName="Control", variableName="Entry Y", setValue=15, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Vampire Gnat"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    on FINISHED → Fighting
  }

  state Fighting {
    SendEventToRegisterDelay(EventName="GARMOND START", delay=8, ExcludeTarget=Self)
    on HORNET DEATH → Hornet Defeated
  }

  state Hornet Defeated {
    PlayerDataIntAdd(intName="vampireGnatDeaths", amount=1)
  }

  state Garmond Aid? {
    PlayerDataBoolTest(boolName="vampireGnatRequestedAid", isTrue=(none), isFalse=→"FINISHED")
    FindNamedChild(gameObject=Self, storeResult=var "Garmond Fighter")
    ActivateGameObject(gameObject=var "Garmond Fighter", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Clear Out Scene {
    SetPlayerDataBool(boolName="greymoor05_clearedOut", value=true)
  }

  var Fight Start Range: gameObject = <null>
  var Vampire Gnat: gameObject = <null>
  var Garmond Fighter: gameObject = <null>
}
