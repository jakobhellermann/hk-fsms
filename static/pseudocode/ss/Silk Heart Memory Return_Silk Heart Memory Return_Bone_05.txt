# Silk Heart Memory Return
# 1 location(s):
#   Bone_05 / Silk Heart Memory Return

fsm Silk Heart Memory Return {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Silk Heart", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on START ENTRY → Wait
  }

  state Resume Silk Heart {
    AddEventRegister(target=Self, eventName="HEART COLLECTED")
    SendEventByName(eventTarget=GameObject(var "Silk Heart"), sendEvent="CONTINUE SCENE", delay=0, everyFrame=false)
    on HEART COLLECTED → Silk Quest?
  }

  sequence state Scene Fade In {
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=true)
    HeroControllerMethods(method=SetSilkRegenBlocked, parameters=[true], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="RelinquishControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="StopAnimationControl", parameters=[], storeResult=(unset var), EveryFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=var "Hero", clipName="Prostrate", animationTriggerEvent=(none), animationCompleteEvent=(none))
    ActivateGameObject(gameObject=var "Silk Heart", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Silk Heart"), sendEvent="CONTINUE SCENE PRE", delay=0, everyFrame=false)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=3)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    AwardQueuedAchievements(delay=0)
    Wait(time=1.5, finishEvent=(none), realTime=false)
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="IN", delay=0, everyFrame=false)
    on FINISHED → Resume Silk Heart
  }

  state Save {
    HeroControllerMethods(method=SetSilkRegenBlocked, parameters=[false], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    SetPlayerDataVariable(VariableName="disableInventory", SetValue=false)
    SetDeathRespawnMarker(Target=Silk Heart Memory Return/Silk Heart Memory Respawn Marker)
    SaveGame()
  }

  state Wait {
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 1), duration=0)
    HeroControllerMethods(method=SetSilkRegenBlocked, parameters=[true], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    on FINISH ENTRY → Scene Fade In
  }

  state Silk Quest? {
    BoolTest(boolVariable=var "Start Final Silk Quest", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    BeginQuest(Quest=Citadel Ascent Silk Defeat (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    on FINISHED → Save
  }

  var Start Final Silk Quest: bool = false
  var Silk Heart: gameObject = Silk Heart
}
