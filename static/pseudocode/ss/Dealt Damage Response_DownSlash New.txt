# Dealt Damage Response
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet/Attacks/Witch/DownSlash New

// uses template: crest_downslash_reaction_witch
fsm Dealt Damage Response {
  start State 1

  state State 1 {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Interactive Object(19), sendEvent=→"HIT INTERACTIVE", storeCollider=var "Hit Object")
    on DEALT ACTUAL DAMAGE → Send Event
    on HIT INTERACTIVE → Check
    on ATTACK BLOCKED → Send Blocked
  }

  state Send Event {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="CanCustomRecoil", parameters=[], storeResult=var "Can Recoil", EveryFrame=false)
    BoolTest(boolVariable=var "Can Recoil", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Crest Attacks", variableName="Attack Target", setValue=var "Hit Object", everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Hero", fsm="Crest Attacks"), sendEvent="ATTACK LANDED", delay=0, everyFrame=false)
    on CANCEL → State 1
    on FINISHED → State 1
  }

  state Check {
    CallMethodProper(gameObject=var "Hero", behaviour="HeroController", methodName="CanCustomRecoil", parameters=[], storeResult=var "Can Recoil", EveryFrame=false)
    BoolTest(boolVariable=var "Can Recoil", isTrue=(none), isFalse=→"CANCEL", everyFrame=false)
    HasComponent(gameObject=var "Hit Object", component="NonBouncer", removeOnExit=false, trueEvent=→"CANCEL", falseEvent=→"BOUNCE TINK", store=(unset), everyFrame=false)
    on FINISHED → Send Event
    on CANCEL → State 1
    on BOUNCE TINK → Bounce Tink
  }

  state Bounce Tink {
    SendEventByName(eventTarget=GameObjectFSM(var "Hero", fsm="Crest Attacks"), sendEvent="BOUNCE TINKED", delay=0, everyFrame=false)
    SetFsmGameObject(gameObject=var "Hero", fsmName="Crest Attacks", variableName="Attack Target", setValue=var "Hit Object", everyFrame=false)
    GetPosition2d(gameObject=var "Hit Object", vector_2d=(unset), x=(unset), y=var "Hit Object Y", space=0, everyFrame=false)
    SetFsmFloat(gameObject=var "Hero", fsmName="Crest Attacks", variableName="Attack Target Y", setValue=var "Hit Object Y", everyFrame=false)
    on FINISHED → Send Event
  }

  state Send Blocked {
    SetFsmGameObject(gameObject=var "Hero", fsmName="Crest Attacks", variableName="Attack Target", setValue=var "Hit Object", everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(var "Hero", fsm="Crest Attacks"), sendEvent="BOUNCE TINKED", delay=0, everyFrame=false)
    on FINISHED → State 1
  }

  var Hit Object Y: float = 0
  var Can Recoil: bool = false
  var Hit Object: gameObject = <null>
}
