# Control
# 1 location(s):
#   Bone_East_04b / Reminder Look Down

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Wait For SF
  }

  state Idle {
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Trigger2dEvent(gameObject=Self, trigger=1, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ACTIVATE → Activated
    on ENTER → Send
  }

  state Activated {
  }

  state Send {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)  // disabled
    SendEventToRegister(eventName="REMINDER LOOK DOWN")
    CheckTargetDirection(gameObject=var "MainCamera", target=var "Hero", aboveEvent=→"END", belowEvent=(none), rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=4, reverseIfNegativeScale=false, everyFrame=true)
    GetYDistance(gameObject=var "MainCamera", target=var "Hero", storeResult=var "Y Distance", everyFrame=true, allowNegatives=false)  // disabled
    FloatCompare(float1=var "Y Distance", float2=5, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"END", everyFrame=true)  // disabled
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=→"END", rightEvent=(none), leftEvent=(none), aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=var "End if Hero Below By", reverseIfNegativeScale=false, everyFrame=true)
    on END → End Pause
    on LEAVING SCENE → End
    on END 2 → State 1
  }

  state End Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on FINISHED → End
  }

  state End {
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    SendEventToRegister(eventName="REMINDER LOOK DOWN END")
  }

  state Wait For SF {
    BoolTest(boolVariable=var "No Silkfly", isTrue=→"START ACTIVE", isFalse=(none), everyFrame=false)
    on SILKFLY LEAVE → Idle Pause
    on START ACTIVE → Idle
  }

  state Idle Pause {
    Wait(time=3.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state State 1 {
  }

  var End if Hero Below By: float = -10
  var No Silkfly: bool = true

  var Y Distance: float = 0
  var Activated: bool = false
}
