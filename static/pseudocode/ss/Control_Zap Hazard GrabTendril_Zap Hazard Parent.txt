# Control
# 2 location(s):
#   Arborium_01 / Zap Hazard Parent/Zap Hazard GrabTendril
#   Coral_29 / Zap Hazard Parent/Zap Hazard GrabTendril

fsm Control {
  start Init

  state Init {
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on DAMAGED HERO → Pause
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Play
    on HORNET DEATH → Death Effect
  }

  state Play {
    FindGameObject(objectName="", withTag="Extra Tag", store=var "Knight Zap Death")
    PlayAudioEvent(audioClip=coral_worms_damage_hornet (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D NoReverb@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayRandomAudioClipTable(Table=Wound Hornet Voice (herodynamic_assets_all.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    SetPositionToObject2D(gameObject=Self, targetObject=var "Knight Zap Death", xOffset=0, yOffset=0, everyFrame=true)
    FindNamedChild(gameObject=var "Damager", storeResult=var "Pull Centre")
    GetAngleToTarget2D(gameObject=Self, target=var "Pull Centre", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    EaseFloat(fromValue=-5, toValue=7, floatVariable=var "Speed", time=0.25, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=(none), realTime=false)
    EaseFloat(fromValue=7, toValue=0, floatVariable=var "Speed", time=0.25, speed=(unset), delay=0.25, easeType=13, reverse=false, finishEvent=(none), realTime=false)
    SetVelocityAsAngle(gameObject=var "Knight Zap Death", angle=var "Angle", speed=var "Speed", everyFrame=true)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Sprite", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, RepeatDelay=0.15)
    on HAZARD RESPAWN RESET → Reset
    on HORNET DEATH → Death Effect
  }

  state Reset {
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Death Effect {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetPositionToObject2D(gameObject=Self, targetObject=var "Hero", xOffset=0, yOffset=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt DeathZap", emit=0, resetIfPlaying=false)
  }

  var Angle: float = 0
  var Speed: float = 0
  var Damager: gameObject = <null>
  var Sprite: gameObject = Zap Hazard Parent/Zap Hazard GrabTendril/Sprite
  var Knight Zap Death: gameObject = <null>
  var Pull Centre: gameObject = <null>
  var Pt DeathZap: gameObject = Zap Hazard Parent/Zap Hazard GrabTendril/Pt DeathZap
}
