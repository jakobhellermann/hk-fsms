# Land Detector
# 3 location(s):
#   Bone_04 / Fall_platform_barrels/Landing Box
#   Under_05 / Barrel Plat Lift (2)/Fall_platform_barrel wide/Landing Box
#   Under_27 / Floor Break Set/Fall_platform_barrel/Landing Box

fsm Land Detector {
  start State 1

  state State 1 {
    GetParent(gameObject=Self, storeResult=var "Parent")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"HIT", storeCollider=var "")
    Collision2dEvent(gameObject=Self, collision=0, collideTag="Terrain", sendEvent=→"HIT", storeCollider=var "", storeForce=(unset))  // disabled
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"HIT", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerStay2D, collideTag="", collideLayer=Soft Terrain(25), sendEvent=→"HIT", storeCollider=var "")
    on HIT → Check Type
    on ACID → Check Type
  }

  state Hard Land {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="HARDLAND", delay=0, everyFrame=false)
    on ACID → Acid Land
  }

  state Acid Land {
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="ACIDLAND", delay=0, everyFrame=false)
  }

  state Check Type {
    BoolTest(boolVariable=var "Acid", isTrue=→"ACID", isFalse=→"HIT", everyFrame=false)
    on ACID → Acid Land
    on HIT → Hard Land
  }

  var Acid: bool = false

  var Parent: gameObject = <null>
}
