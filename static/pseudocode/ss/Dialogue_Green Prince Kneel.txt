# Dialogue
# 2 location(s):
#   Clover_06 / Green Prince Kneel
#   Cog_Dancers / Green Prince Kneel

fsm Dialogue {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    CheckSceneName(sceneName="Clover_06", equalEvent=→"CLOVER", notEqualEvent=(none))
    on FINISHED → State?
    on CLOVER → Set Clover
  }

  state State? {
    CheckStaticBool(variableName="GreenPrinceStayInDancers", trueEvent=→"FINISHED", falseEvent=(none), EveryFrame=false)
    PlayerDataVariableTest(VariableName="GreenPrinceLocation", ExpectedValue=enum(2), IsExpectedEvent=(none), IsNotExpectedEvent=→"CANCEL")
    on CANCEL → Gone
    on FINISHED → Stay
  }

  state Idle {
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=1.4, MaxReactDelay=1.6, None=(none), ActiveInner=→"SING", ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on INTERACT → Met?
    on SING → Sing
    on STARTLE → Startle
  }

  state Gone {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Sing {
    BoolTest(boolVariable=var "In Clover", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)  // disabled
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateTalkAnimNPC(Activate=false, StopInstantly=false, Target=Self)
    PlayRandomAudioClipTable(Table=small_movement_shell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.6, MaxReactDelay=0.7, None=→"CANCEL", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    EnemySingControl(enemyGameObject=Self, audioPlayer=var "Audio Loop Voice", singAudioTable=GP_sing (sfxdynamic_assets_greenprince.bundle), noThreadEffects=true, noPuppetString=true, randomSingStartTime=false, dontStopAudioOnExit=false, altThreadSpawnPoint=<null>)
    on CANCEL → Sing End
  }

  sequence state Sing End {
    PlayRandomAudioClipTable(Table=small_movement_shell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWait(Target=Self, ClipName="Sing End", AnimationCompleteEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle")
    ActivateTalkAnimNPC(Activate=true, StopInstantly=false, Target=Self)
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    on FINISHED → Idle
  }

  state Met? {
    BoolTest(boolVariable=var "In Clover", isTrue=→"CLOVER", isFalse=(none), everyFrame=false)
    PlayerDataVariableTest(VariableName="GreenPrinceLocation", ExpectedValue=enum(2), IsExpectedEvent=→"FALSE", IsNotExpectedEvent=→"TRUE")
    on FALSE → Meet
    on TRUE → Repeat
    on CLOVER → Clover Dialogue
  }

  state Meet {
    SetPlayerDataVariable(VariableName="GreenPrinceLocation", SetValue=enum(3))
    RunDialogue(Sheet="Wanderers", Key="G_PRINCE_DANCERS_MEET", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dlg
  }

  state End Dlg {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    on FINISHED → Idle
  }

  state Repeat {
    RunDialogue(Sheet="Wanderers", Key="G_PRINCE_DANCERS_REPEAT", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on CONVO_END → End Dlg
  }

  state Stay {
    SetStaticVariable(variableName="GreenPrinceStayInDancers", setValue=true, sceneTransitionsLimit=3)
    on FINISHED → Idle
  }

  state State 1 {
  }

  state Set Clover {
    SetBoolValue(boolVariable=var "In Clover", boolValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  state Clover Dialogue {
    ConvertBoolToString(boolVariable=var "Activated", stringVariable=var "Convo Key", falseString="GREEN_PRINCE_VERDANIA_STAG_MEET", trueString="GREEN_PRINCE_VERDANIA_STAG_REPEAT", everyFrame=false)
    RunDialogue(Sheet="Wanderers", Key=var "Convo Key", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetBoolValue(boolVariable=var "Activated", boolValue=true, everyFrame=false)
    on CONVO_END → End Dlg
  }

  sequence state Startle {
    BoolTest(boolVariable=var "In Clover", isTrue=→"NO STARTLE", isFalse=(none), everyFrame=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    ActivateTalkAnimNPC(Activate=false, StopInstantly=false, Target=Self)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=GP_startle (sfxdynamic_assets_greenprince.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayRandomAudioClipTable(Table=small_movement_shell (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Sing", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Wait(time=0.5, finishEvent=→"CANCEL", realTime=false)
    on CANCEL → Sing End
    on NO STARTLE → Idle
  }

  var In Clover: bool = false
  var Activated: bool = false
  var Convo Key: string = ""
  var Audio Loop Voice: gameObject = Green Prince Kneel/Audio Loop Voice
}
