# Control
# 1 location(s):
#   enemycorpses_assets_areasong.bundle / Corpse Song Pilgrim 03/Silk Ration Dropper

fsm Control {
  start Do Drop

  state Do Drop {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    PlayerDataBoolTest(boolName="blackThreadWorld", isTrue=→"NONE", isFalse=(none))
    RandomFloat(min=1, max=100, storeResult=var "Chance")
    FloatCompare(float1=var "Chance", float2=var "Silk Drop Chance", tolerance=0, equal=→"FULL", lessThan=→"FULL", greaterThan=→"EMPTY", everyFrame=false)
    on FULL → Full
    on EMPTY → Empty
    on NONE → State 1
  }

  state Full {
    SetGameObject(variable=var "Drop Object", gameObject=var "Silk Ration Large", everyFrame=false)
    on FINISHED → Fling
  }

  state Fling {
    RandomFloat(min=0, max=360, storeResult=var "Rotation")
    SetRotation(gameObject=var "Drop Object", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    ActivateGameObject(gameObject=var "Drop Object", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObject(flungObject=var "Silk Ration Large", speedMin=17, speedMax=20, angleMin=40, angleMax=140)
  }

  state Empty {
    SetGameObject(variable=var "Drop Object", gameObject=var "Silk Ration Large Core", everyFrame=false)
    on FINISHED → Fling
  }

  state State 1 {
  }

  var Silk Drop Chance: float = 25

  var Chance: float = 0
  var Rotation: float = 0
  var Silk Ration Large: gameObject = Corpse Song Pilgrim 03/Silk Ration Dropper/Silk Ration Citadel Large
  var Silk Ration Large Core: gameObject = Corpse Song Pilgrim 03/Silk Ration Dropper/Silk Ration Citadel Large Core
  var Drop Object: gameObject = <null>
}
