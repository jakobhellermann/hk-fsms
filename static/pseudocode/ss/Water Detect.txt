fsm Water Detect {
  start Init

  state Init {
    on FINISHED → Detect
  }

  state Detect {
    RayCast2dV2(fromGameObject=Self, fromPosition=(0, 2), direction=(0, -1), space=1, distance=4.5, minDepth=(unset), maxDepth=(unset), hitEvent=→"WATER", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "Water Surface", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Soft Terrain(25)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on WATER → Surface Water?
  }

  state In Water {
    GetColliderRange(gameObject=var "Water Surface", minX=(unset), maxX=(unset), minY=(unset), maxY=var "Y Pos", everyFrame=false)
    FloatAdd(floatVariable=var "Y Pos", add=1, everyFrame=false, perSecond=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Surface Y", setValue=var "Y Pos", everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="WATER", delay=0, everyFrame=false)
    on EXIT WATER → Detect
  }

  state Surface Water? {
    GetTag(gameObject=var "Water Surface", storeResult=var "Water Surface Tag", everyFrame=false)
    StringCompare(stringVariable=var "Water Surface Tag", compareTo="Water Surface", equalEvent=→"WATER", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Detect
    on WATER → In Water
  }

  var Y Pos: float = 0
  var Water Surface Tag: string = ""
  var Water Surface: gameObject = <null>
}
