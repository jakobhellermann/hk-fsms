# Control
# 1 location(s):
#   Under_06 / Grind Plat Control/junk_chute_ manual L

fsm Control {
  start Init

  state Init {
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=var "Wait Time", finishEvent=→"START", realTime=false)
    StopParticleEmittersInChildren(gameObject=var "junk_set")
    SetPosition(gameObject=var "Spike Collider", vector=(unset), x=(unset), y=1, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionTo(gameObject=var "Spike Collider", toValue=(-3.77, -21, 0), localSpace=true, time=0.5, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    on START → Flow
  }

  state Flow {
    Wait(time=var "Flow Time", finishEvent=→"STOP", realTime=false)
    PlayParticleEmitterChildren(gameObject=var "junk_set", resetTimeIfPlaying=false, stopOnStateExit=false)
    SetPosition(gameObject=var "Spike Collider", vector=(unset), x=(unset), y=21, z=(unset), space=1, everyFrame=false, lateUpdate=false)
    AnimatePositionTo(gameObject=var "Spike Collider", toValue=(-3.77, 1, 0), localSpace=true, time=0.5, speed=(unset), delay=0, easeType=21, reverse=false, finishEvent=(none), realTime=false)
    RandomFloat(min=0.8, max=1.2, storeResult=var "Pitch")
    SetAudioPitch(gameObject=Self, pitch=var "Pitch", everyFrame=false)
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    RandomFloat(min=0.8, max=1.2, storeResult=var "Pitch")
    SetAudioPitch(gameObject=var "Audio Loop Junk", pitch=var "Pitch", everyFrame=false)
    AudioPlayInState(gameObject=var "Audio Loop Junk", volume=1)
    StartHeroVibrationRegion(target=var "Hero Vibration Region", stopOnExit=true)
    on STOP → Pause
  }

  var Flow Time: float = 1
  var Wait Time: float = 9999

  var Pitch: float = 0
  var Spike Collider: gameObject = Grind Plat Control/junk_chute_ manual L/Spike Collider
  var junk_set: gameObject = Grind Plat Control/junk_chute_ manual L/junk_set
  var Audio Loop Junk: gameObject = Grind Plat Control/junk_chute_ manual L/Audio Loop Junk
  var Hero Vibration Region: gameObject = Grind Plat Control/junk_chute_ manual L/Hero Vibration Region junk_chut_manual
}
