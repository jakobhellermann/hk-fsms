// uses template: coral_tower_spike_floor
fsm Control {
  start Init

  sequence state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Respawns")
    FindNamedChild(gameObject=Self, storeResult=var "Spikes")
    ActivateGameObject(gameObject=var "Respawns", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="SPIKE CLEANUP")
    NextFrameEvent(sendEvent=(none))
    AddEventRegister(target=Self, eventName="SPIKE CLEANUP")
    AddEventRegister(target=Self, eventName="BATTLE END")
    ActivateGameObject(gameObject=var "Respawns", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=rock_roller_wall_disturbed (sfxstatic_assets_areabone.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=(none), realTime=false)
    PlayAudioEvent(audioClip=coral_spear_enter_1 (sfxstatic_assets_areacoralareamemory.bundle), pitchMin=1.25, pitchMax=1.125, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Deparent and Progress Wave
  }

  state Deparent and Progress Wave {
    GetGrandparent(gameObject=Self, storeResult=var "Battle scene")
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SendMessage(gameObject=var "Battle scene", delivery=SendMessage, options=0, functionCall=WaveEnd())
    on SPIKE CLEANUP → Cleanup
    on BATTLE END → Cleanup
  }

  state Cleanup {
    ActivateGameObject(gameObject=var "Respawns", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Spikes"), sendEvent="CORAL BREAK", delay=0, everyFrame=false)
  }

  var Battle scene: gameObject = <null>
  var Respawns: gameObject = <null>
  var Spikes: gameObject = <null>
}
