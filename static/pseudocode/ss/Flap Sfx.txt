# Flap Sfx
# 1 location(s):
#   Room_CrowCourt_02 / Battle Scene/Wave 6/Crawfather

fsm Flap Sfx {
  start Idle

  state Idle {
    on FLAP START → Flap Sfx
  }

  state Flap Sfx {
    PlayRandomAudioClipTable(Table=Moorwing Flap (sfxdynamic_assets_vampiregnat.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FLAP STOP → Idle
    on FINISHED → Pause
  }

  state Pause {
    Wait(time=0.4, finishEvent=(none), realTime=false)
    on FINISHED → Flap Sfx
    on FLAP STOP → Idle
  }
}
