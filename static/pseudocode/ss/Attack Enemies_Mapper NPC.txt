// uses template: mapper_battle
fsm Attack Enemies {
  start Init
  on START AWAY → Start Away Pause  // from any state
  on STUN → Stun Start  // from any state
  on ZERO HP → Defeat Start  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Body Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Enemy Range")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    FindNamedChild(gameObject=Self, storeResult=var "Ring Flash")
    FindNamedChild(gameObject=Self, storeResult=var "Ring Point 1")
    FindNamedChild(gameObject=Self, storeResult=var "Start Marker")
    FindNamedChild(gameObject=Self, storeResult=var "Tele Marker")
    FindNamedChild(gameObject=Self, storeResult=var "Tele Range")
    FindNamedChild(gameObject=Self, storeResult=var "Tele Range Hero")
    FindNamedChild(gameObject=Self, storeResult=var "Tele Range Hero Low")
    FindNamedChild(gameObject=Self, storeResult=var "Tele Range Stomp")
    FindNamedChild(gameObject=Self, storeResult=var "Stomp Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Stomp Blast")
    FindNamedChild(gameObject=Self, storeResult=var "Stand Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Stun Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Stun Fall Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Stun Stagger Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Defeat Fall Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Defeat Land Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Charge Collider")
    FindNamedChild(gameObject=Self, storeResult=var "ChargeAntic Collider")
    FindNamedChild(gameObject=Self, storeResult=var "ChargeLeave Collider")
    FindNamedChild(gameObject=Self, storeResult=var "ChargePunch Collider")
    FindNamedChild(gameObject=Self, storeResult=var "ChargeThrow Collider")
    FindNamedChild(gameObject=Self, storeResult=var "ChargeThrow Punch Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Punch Blast")
    FindNamedChild(gameObject=Self, storeResult=var "Ring Point B")
    FindNamedChild(gameObject=Self, storeResult=var "HeroLight")
    FindNamedChild(gameObject=Self, storeResult=var "Vignette Cutout")
    GetPosition(gameObject=Self, vector=var "Start Pos", x=var "Start X", y=var "Start Y", z=(unset), space=0, everyFrame=false)
    GetColliderRange(gameObject=var "Tele Range", minX=var "Tele Min X", maxX=var "Tele Max X", minY=var "Tele Min Y", maxY=var "Tele Max Y", everyFrame=false)
    SetParent(gameObject=var "Tele Marker", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Tele Range", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Tele Range Hero", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Tele Range Hero Low", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Tele Range Stomp", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetParent(gameObject=var "Start Marker", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetScale(gameObject=Self, vector=(unset), xScale=var "X Scale Orig", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Enemy Range", trigger=1, collideTag="", sendEvent=→"ENEMY", storeCollider=var "Target Enemy")
    on ENEMY → Battle Cry Start
    on BATTLE HERO → Set Fighting Hero
  }

  state Battle Cry Start {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="END ANIM", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="END CONVO", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FIGHT STARTED", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL RESET", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="STUN CONTROL START", delay=0, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="tk2dLookAnimNPC", methodName="Disable", parameters=[], storeResult=(unset var), EveryFrame=false)
    SetFsmBool(gameObject=Self, fsmName="Dialogue", variableName="Attacking", setValue=true, everyFrame=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    BoolTest(boolVariable=var "Did Battle Cry", isTrue=→"END", isFalse=(none), everyFrame=false)
    SetBoolValue(boolVariable=var "Did Battle Cry", boolValue=true, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Target Enemy", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Battle Cry 1
    on END → Target Enemy
  }

  state Battle Cry 2 {
    ActivateGameObject(gameObject=var "Ring Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=(none))
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    Wait(time=0.5, finishEvent=→"NEXT", realTime=false)
    PlayAudioEvent(audioClip=shakra_entrance_shing (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendMessage(gameObject=var "HeroLight", delivery=SendMessage, options=1, functionCall=FadeIn())
    SendMessage(gameObject=var "Vignette Cutout", delivery=SendMessage, options=1, functionCall=FadeIn())
    on NEXT → Target Enemy
  }

  state Target Enemy {
    ActivateGameObject(gameObject=var "Body Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    GameObjectCompare(gameObjectVariable=var "Target Enemy", compareTo=var "Start Marker", equalEvent=→"CANCEL", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTest(boolVariable=var "Fighting Hero", isTrue=→"BATTLE HERO", isFalse=(none), everyFrame=false)
    GameObjectIsNull(gameObject=var "Target Enemy", isNull=→"CANCEL", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Target Enemy", vector=(unset), x=var "Enemy X", y=var "Enemy Y", z=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Enemy Y", float2=var "Tele Max Y", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)  // disabled
    SendEvent(eventTarget=Self, sendEvent=→"TELE LOOK", delay=0, everyFrame=false)
    on END → 
    on CANCEL → Check For Other Enemies
    on TELE LOOK → Look Pos
    on BATTLE HERO → Target Hero
  }

  state Aim Home {
    SetFloatValue(floatVariable=var "Tele X", floatValue=var "Start X", everyFrame=false)
    SetFloatValue(floatVariable=var "Tele Y", floatValue=var "Start Y", everyFrame=false)
    SetPosition(gameObject=var "Tele Marker", vector=(unset), x=var "Tele X", y=var "Tele Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Going Home", boolValue=true, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Tele Marker", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    on FINISHED → Tele Out Antic
  }

  state Look Pos {
    FloatCompare(float1=var "Enemy X", float2=var "Tele Max X", tolerance=0, equal=(none), lessThan=(none), greaterThan=→"L", everyFrame=false)
    FloatCompare(float1=var "Enemy X", float2=var "Tele Min X", tolerance=0, equal=(none), lessThan=→"R", greaterThan=(none), everyFrame=false)
    SendRandomEvent(events=[→"L", →"R"], weights=[1, 1], delay=0)
    on L → Look L
    on R → Look R
  }

  state Look L {
    FloatOperator(float1=var "Enemy X", float2=8, operation=Subtract, storeResult=var "Range Max", everyFrame=false)
    FloatOperator(float1=var "Enemy X", float2=11, operation=Subtract, storeResult=var "Range Min", everyFrame=false)
    FloatClamp(floatVariable=var "Range Min", minValue=var "Tele Min X", maxValue=var "Tele Max X", everyFrame=false)
    FloatClamp(floatVariable=var "Range Max", minValue=var "Tele Min X", maxValue=var "Tele Max X", everyFrame=false)
    RandomFloat(min=var "Range Min", max=var "Range Max", storeResult=var "Tele X")
    on FINISHED → Do Y
  }

  state Do Y {
    FloatOperator(float1=var "Enemy Y", float2=0.5, operation=Add, storeResult=var "Range Min", everyFrame=false)
    FloatOperator(float1=var "Enemy Y", float2=6, operation=Add, storeResult=var "Range Max", everyFrame=false)
    FloatClamp(floatVariable=var "Range Min", minValue=var "Tele Min Y", maxValue=var "Tele Max Y", everyFrame=false)
    FloatClamp(floatVariable=var "Range Max", minValue=var "Tele Min Y", maxValue=var "Tele Max Y", everyFrame=false)
    RandomFloat(min=var "Range Min", max=var "Range Max", storeResult=var "Tele Y")
    SetPosition(gameObject=var "Tele Marker", vector=(unset), x=var "Tele X", y=var "Tele Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Destination Distance Check
  }

  state Look R {
    FloatOperator(float1=var "Enemy X", float2=8, operation=Add, storeResult=var "Range Min", everyFrame=false)
    FloatOperator(float1=var "Enemy X", float2=11, operation=Add, storeResult=var "Range Max", everyFrame=false)
    FloatClamp(floatVariable=var "Range Min", minValue=var "Tele Min X", maxValue=var "Tele Max X", everyFrame=false)
    FloatClamp(floatVariable=var "Range Max", minValue=var "Tele Min X", maxValue=var "Tele Max X", everyFrame=false)
    RandomFloat(min=var "Range Min", max=var "Range Max", storeResult=var "Tele X")
    on FINISHED → Do Y
  }

  state Tele Out Antic {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Shakra_movement (sfxdynamic_assets_mapper.bundle), pitchOffset=0, stopPreviousSound=false, forcePlay=false)
    SetIntValue(intVariable=var "Distance Retry", intValue=0, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Tele Marker", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    BoolTest(boolVariable=var "Grounded", isTrue=→"GROUND", isFalse=→"AIR", everyFrame=false)
    on FINISHED → Tele Out Facing
    on GROUND → Ground Antic
    on AIR → Air Antic
  }

  state Tele Out {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=var "Y Translate", z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Out", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendMessage(gameObject=var "HeroLight", delivery=SendMessage, options=1, functionCall=FadeOut())
    SendMessage(gameObject=var "Vignette Cutout", delivery=SendMessage, options=1, functionCall=FadeOut())
    on FINISHED → Charge Punch?
  }

  state Tele Out Facing {
    GetAngleToTarget2D(gameObject=Self, target=var "Tele Marker", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Tele Marker", offsetX=0, offsetY=0, storeAngle=var "Move Angle", pause=0, everyFrame=false)
    SendEventByScale(gameObject=Self, eventTarget=Self, xScale=true, positiveEvent=→"R", negativeEvent=→"L", space=0)
    on L → Tele Out L
    on R → Tele Out
  }

  state Tele Out L {
    FloatAdd(floatVariable=var "Angle", add=180, everyFrame=false, perSecond=false)
    on FINISHED → Tele Out
  }

  state Tele In {
    BoolTest(boolVariable=var "In Unalert Range", isTrue=→"UNALERT", isFalse=(none), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=true)
    PlayAudioEventRandom(audioClips=[shakra_zip_1 (sfxstatic_assets_mapper.bundle), shakra_zip_2 (sfxstatic_assets_mapper.bundle), shakra_zip_3 (sfxstatic_assets_mapper.bundle), shakra_zip_4 (sfxstatic_assets_mapper.bundle)], pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    BoolTest(boolVariable=var "Going Home", isTrue=→"GO HOME", isFalse=(none), everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Move Angle", speed=2, everyFrame=false)
    SetBoolValue(boolVariable=var "Grounded", boolValue=false, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Target Enemy", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SelectRandomString(strings=["Air Idle 1", "Air Idle 2", "Air Idle 3", "Air Idle 4"], weights=[1, 1, 1, 1], storeString=var "Anim Clip")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName=var "Anim Clip")
    WaitRandom(timeMin=var "Idle Time Min", timeMax=var "Idle Time Max", finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Body Collider", activate=var "Fighting Hero", recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "HeroLight", delivery=SendMessage, options=1, functionCall=FadeIn())
    SendMessage(gameObject=var "Vignette Cutout", delivery=SendMessage, options=1, functionCall=FadeIn())
    WaitBool(boolTest=var "Chain Stomp", time=0.1, finishEvent=→"CHAIN STOMP", realTime=false)
    on FINISHED → Next
    on GO HOME → Land Home
    on CHAIN STOMP → Do Chain Stomp
    on UNALERT → Away because Unalert
  }

  state Air Antic {
    BuildString(stringParts=[var "Anim Clip", " Tele"], separator="", addToEnd=true, storeResult=var "Tele Out Clip", everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Tele Out Clip", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetFloatValue(floatVariable=var "Y Translate", floatValue=0, everyFrame=false)
    on FINISHED → Tele Out Facing
  }

  state Ground Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Tele Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetFloatValue(floatVariable=var "Y Translate", floatValue=-2, everyFrame=false)
    ActivateGameObject(gameObject=var "Stand Collider", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Tele Out Facing
  }

  state Destination Distance Check {
    IntCompare(integer1=var "Distance Retry", integer2=200, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=false)
    BoolTest(boolVariable=var "Fighting Hero", isTrue=→"BATTLE HERO", isFalse=(none), everyFrame=false)  // disabled
    GetDistance(gameObject=Self, target=var "Tele Marker", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=5, tolerance=0, equal=(none), lessThan=→"RETRY", greaterThan=→"FINISHED", everyFrame=false)
    on FINISHED → Tele Out Antic
    on RETRY → Distance Retry
  }

  state Distance Retry {
    IntAdd(intVariable=var "Distance Retry", add=1, everyFrame=false)
    SetBoolValue(boolVariable=var "Chain Stomp", boolValue=false, everyFrame=false)
    on FINISHED → Look Pos
  }

  state Next {
    ActivateGameObject(gameObject=var "Body Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    IntAdd(intVariable=var "Moves", add=-1, everyFrame=false)
    GetDistanceV2(gameObject=Self, target=var "Target Enemy", targetOffsetX=0, targetOffsetY=0, storeResult=var "Distance", everyFrame=false)
    FloatTestToBool(float1=var "Distance", float2=15, tolerance=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Distant From Target", everyFrame=false)
    BoolAllTrue(boolVariables=[var "Fighting Hero", var "Distant From Target"], sendEvent=→"MOVE", storeResult=(unset), everyFrame=false)
    GameObjectIsNull(gameObject=var "Target Enemy", isNull=→"MOVE", isNotNull=(none), storeResult=(unset), everyFrame=false)
    IntCompare(integer1=var "Moves", integer2=0, equal=→"ATTACK", lessThan=→"ATTACK", greaterThan=→"MOVE", everyFrame=false)
    on MOVE → Target Enemy
    on ATTACK → Attack Choice
  }

  state Enemy Throw Antic {
    RandomInt(min=1, max=2, storeResult=var "Moves", inclusiveMax=true, noRepeat=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Shakra_attack_throw (sfxdynamic_assets_mapper.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    FaceObjectV2(objectA=Self, objectB=var "Target Enemy", spriteFacesRight=true, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SelectRandomString(strings=["Throw", "Throw Alt"], weights=[1, 1], storeString=var "Throw Anim")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Throw Anim", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Throw
  }

  state Throw {
    PlayAudioEvent(audioClip=shakra_throw_disc (sfxstatic_assets_mapper.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetStringValue(stringVariable=var "Anim Clip", stringValue="Air Idle 4", everyFrame=false)
    GetAngleToTarget2D(gameObject=var "Ring Point 1", target=var "Target Enemy", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Shakra Ring Friendly (localpoolprefabs_assets_mapper.bundle), spawnPoint=var "Ring Point 1", position=(unset), spawnMin=1, spawnMax=1, speedMin=var "Throw Speed", speedMax=var "Throw Speed", angleMin=var "Angle", angleMax=var "Angle", originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Target Enemy
  }

  state Check For Other Enemies {
    NextFrameEvent(sendEvent=→"FINISHED")
    Trigger2dEvent(gameObject=var "Enemy Range", trigger=1, collideTag="", sendEvent=→"ENEMY", storeCollider=var "Target Enemy")
    on ENEMY → Target Enemy
    on FINISHED → Aim Home
  }

  state Land Home {
    SetBoolValue(boolVariable=var "Grounded", boolValue=true, everyFrame=false)
    ActivateGameObject(gameObject=var "Body Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Going Home", boolValue=false, everyFrame=false)
    PlayAudioEvent(audioClip=shakra_land_normal (sfxstatic_assets_mapper.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Land To Idle", animationTriggerEvent=(none), animationCompleteEvent=→"NEXT")
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale Orig", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Trigger2dEvent(gameObject=var "Enemy Range", trigger=1, collideTag="", sendEvent=→"ENEMY", storeCollider=var "Target Enemy")
    Wait(time=1.5, finishEvent=→"NEXT", realTime=false)  // disabled
    on NEXT → Reset
    on ENEMY → Target Enemy
  }

  state Reset {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Idle Right")
    SetFsmBool(gameObject=Self, fsmName="Dialogue", variableName="Attacking", setValue=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(Self), sendEvent="MAPPER READY", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Land To Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="TalkToIdle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Start Away Pause {
    Wait(time=0.4, finishEvent=(none), realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on FINISHED → Start Away
  }

  state Start Away {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="END ANIM", delay=0, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="tk2dLookAnimNPC", methodName="Disable", parameters=[], storeResult=(unset var), EveryFrame=false)
    ActivateInteractible(Target=Self, Activate=false, AllowQueueing=false, UseChildren=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on MAPPER CALL → Call Pause
  }

  state Call Pause {
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=Self, vector=var "Start Away Translate", x=(unset), y=(unset), z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Mapper Enter
  }

  state Mapper Enter {
    SetMeshRenderer(gameObject=Self, active=true)
    GetColliderRange(gameObject=var "Tele Range", minX=var "Tele Min X", maxX=var "Tele Max X", minY=var "Tele Min Y", maxY=var "Tele Max Y", everyFrame=false)
    RandomFloat(min=var "Tele Min X", max=var "Tele Max X", storeResult=var "Tele X")
    RandomFloat(min=var "Tele Min Y", max=var "Tele Max Y", storeResult=var "Tele Y")
    SetPosition(gameObject=var "Tele Marker", vector=(unset), x=var "Tele X", y=var "Tele Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetBoolValue(boolVariable=var "Grounded", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Mapper Entering", boolValue=true, everyFrame=false)
    SetGameObject(variable=var "Target Enemy", gameObject=var "Start Marker", everyFrame=false)
    SetIntValue(intVariable=var "Moves", intValue=2, everyFrame=false)
    on FINISHED → Tele Out Facing
  }

  state Battle Cry 1 {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Ring Shout", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=Shakra_battle_alert (sfxdynamic_assets_mapper.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Start Music?
  }

  state Set Fighting Hero {
    SendMessage(gameObject=Self, delivery=SendMessage, options=1, functionCall=RefillHP())
    SetBoolValue(boolVariable=var "Fighting Hero", boolValue=true, everyFrame=false)
    SetGameObject(variable=var "Target Enemy", gameObject=var "Hero", everyFrame=false)
    GetColliderRange(gameObject=var "Tele Range Hero", minX=var "Tele Min X", maxX=var "Tele Max X", minY=var "Tele Min Y", maxY=var "Tele Max Y", everyFrame=false)
    SetFloatValue(floatVariable=var "Idle Time Min", floatValue=0.35, everyFrame=false)
    SetFloatValue(floatVariable=var "Idle Time Max", floatValue=0.5, everyFrame=false)
    SetLayer(gameObject=Self, layer=Enemies(11))
    SetBoolValue(boolVariable=var "Chain Stomp", boolValue=false, everyFrame=false)
    SetBoolValue(boolVariable=var "Can Stomp", boolValue=false, everyFrame=false)
    CallMethodProper(gameObject=Self, behaviour="NonBouncer", methodName="SetActive", parameters=[false], storeResult=(unset var), EveryFrame=false)
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0, resetOnStateExit=false)
    on FINISHED → Battle Cry Start
  }

  state Target Hero {
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
