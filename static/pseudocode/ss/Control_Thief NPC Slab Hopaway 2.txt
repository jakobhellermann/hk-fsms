# Control
# 1 location(s):
#   Slab_03 / Thief NPC Slab Hopaway 2

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    CheckIfCrestEquipped(Crest=Cloakless (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=(none), falseEvent=→"AWAY", storeValue=(unset))
    GetPlayerDataInt(intName="grindleSlabSequence", storeValue=var "Sequence")
    IntCompare(integer1=var "Sequence", integer2=1, equal=(none), lessThan=→"AWAY", greaterThan=→"AWAY", everyFrame=false)
    on AWAY → Away
    on FINISHED → Idle
  }

  state Away {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Leave Range", storeResult=false, sendEvent="LEAVE", outOfRangeEvent="", everyFrame=true)
    on ALERT → Wave Pause
    on LEAVE → Leave
  }

  state Leave {
    SetPlayerDataInt(intName="grindleSlabSequence", value=2)
    on FINISHED → Away
  }

  state Wave Pause {
    Wait(time=1.75, finishEvent=→"FINISHED", realTime=false)
    SetPlayerDataInt(intName="grindleSlabSequence", value=2)
    on FINISHED → Turn
  }

  state Turn {
    PlayAudioEvent(audioClip=Grindle_laugh_02 (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="TurnToIdle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Flip
  }

  state Leap Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Leap", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    DecelerateXY(gameObject=Self, decelerationX=0.8, decelerationY=(unset), brakeOnExit=false)
    CheckXPosition(gameObject=Self, compareTo=35, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"LEAVE", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    on FINISHED → Leap
    on LEAVE → Leave
  }

  state Leap {
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocity2d(gameObject=Self, vector=(unset), x=35, y=(unset), everyFrame=false)
    DecelerateXY(gameObject=Self, decelerationX=0.9, decelerationY=(unset), brakeOnExit=false)
    PlayAudioEvent(audioClip=bone_hopper_jump (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Land
  }

  state Flip {
    FlipScale(gameObject=Self, flipHorizontally=true, flipVertically=false, everyFrame=false, lateUpdate=false)
    on FINISHED → Leap Antic
  }

  state Land {
    PlayAudioEvent(audioClip=bone_hopper_land (sfxstatic_assets_areawilds.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Leap Antic
  }

  var Sequence: int = 0
}
