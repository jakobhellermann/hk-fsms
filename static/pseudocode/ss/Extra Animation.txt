# Extra Animation
# 1 location(s):
#   Dust_02 / Green Prince Caged/Green Prince Sit

fsm Extra Animation {
  start Idle
  on CAGE HIT → Hit React  // from any state

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Green Prince Caged/Green Prince Sit@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Rest From Left
    on RIGHT → Rest From Right
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "Should Face Left"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest From Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle Left to Rest", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Rest From Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle Right to Rest", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Rest Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest Dip", animationTriggerEvent=(none), animationCompleteEvent=(none))
    Tk2dPlayFrame(gameObject=Self, frame=0)
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rest Idle
    on CANCEL → Wake Up
  }

  state Wake Up {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="ShouldFaceLeft", parameters=[], storeResult=var "Should Face Left", EveryFrame=false)
    BoolTest(boolVariable=var "Should Face Left", isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    on LEFT → Rest to Idle
    on RIGHT → Rest to Right
  }

  state Rest to Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest to Idle Left", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Rest to Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest to Idle Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  sequence state Hit React {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=GP_startle (sfxdynamic_assets_greenprince.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    EnableBehaviour(gameObject=Self, behaviour="", component=Green Prince Caged/Green Prince Sit@HeroPerformanceSingReaction, enable=false, resetOnExit=false)
    ActivateLookAnimNPC(Activate=false, IsFacingLeft=(unset), Target=Self)
    Tk2dPlayAnimationWait(Target=Self, ClipName="Hit Reaction Right", AnimationCompleteEvent=(none))
    ActivateLookAnimNPC(Activate=true, IsFacingLeft=false, Target=Self)
    EnableBehaviour(gameObject=Self, behaviour="", component=Green Prince Caged/Green Prince Sit@HeroPerformanceSingReaction, enable=true, resetOnExit=false)
    on FINISHED → Idle
  }

  var Resting: bool = false
  var Should Face Left: bool = false
  var Audio Loop Voice: gameObject = Green Prince Caged/Green Prince Sit/Audio Loop Voice
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
