# Title Control
# 1 location(s):
#   Abyss_Cocoon / Boss Control/Boss Title

fsm Title Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Silk Swish")
    FindNamedChild(gameObject=Self, storeResult=var "Title Text")
    FindNamedChild(gameObject=Self, storeResult=var "Big_flash_title")
    on FINISHED → Idle
  }

  state Idle {
    on TITLE UP → Swish Start
  }

  state Swish Start {
    ActivateGameObject(gameObject=var "Silk Swish", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flash Effect
  }

  state Title Up {
    FadeNestedFadeGroup(Target=var "Title Text", ToAlpha=1, FadeTime=0.05)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Title Down
  }

  state Title Down {
    FadeNestedFadeGroup(Target=var "Title Text", ToAlpha=0, FadeTime=0.25)
  }

  state Flash Effect {
    ActivateGameObject(gameObject=var "Big_flash_title", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    ApplyMusicCue(musicCue=LostLace (audiocuesdynamic_assets_areaabyss.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.1)
    on FINISHED → Title Up
  }

  var Silk Swish: gameObject = <null>
  var Title Text: gameObject = <null>
  var Big_flash_title: gameObject = <null>
}
