# Control
# 1 location(s):
#   Cradle_03 / Boss Scene/Bell_Hub_Attack_Effect

// component disabled: this FSM does not run until something enables it
fsm Control {
  start Init
  on STOP → Pulse Down  // from any state

  state Init {
    on FINISHED → Dormant
  }

  state Dormant {
    StopParticleEmittersInChildren(gameObject=Self)
    on ATTACK PULSE STUN → Pulse Up
    on TEST → Test
    on NEEDOLIN START → Needolin Start
    on DEATH SEQUENCE → Death Loop Start
  }

  state Pulse Up {
    PlayAudioEvent(audioClip=cradle_hub_all_tuning_forks_on_new_burst (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitterChildren(gameObject=Self, resetTimeIfPlaying=false, stopOnStateExit=false)
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=0.25)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.5)
    on FINISHED → Pulse Down
  }

  state Pulse Down {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=1)
    AudioStop(gameObject=Self, fadeTime=3)
    on FINISHED → Dormant
  }

  state Test {
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=0.25)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=Self, resetTimeIfPlaying=false, stopOnStateExit=false)
    on TEST END → Pulse Down
  }

  state Needolin Loop 1 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=0.25)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on NEEDOLIN END → Pulse Down
    on FINISHED → Needolin Loop 2
  }

  state Needolin Loop 2 {
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on NEEDOLIN END → Pulse Down
    on FINISHED → Needolin Loop 3
  }

  state Needolin Loop 3 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0.4, FadeTime=0.5)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on NEEDOLIN END → Pulse Down
    on FINISHED → Needolin Loop 1
  }

  state Needolin Start {
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=1)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=Self, resetTimeIfPlaying=false, stopOnStateExit=false)
    on NEEDOLIN END → Pulse Down
    on FINISHED → Needolin Loop 2
  }

  state Death Loop 1 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0.35, FadeTime=0.75)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Death Loop 2
  }

  state Death Loop 2 {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0.1, FadeTime=0.75)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Death Loop 1
  }

  state Death Loop Start {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0.6, FadeTime=0.75)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterChildren(gameObject=Self, resetTimeIfPlaying=false, stopOnStateExit=false)
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=1, time=0.5)
    PlayAudioEvent(audioClip=cradle_hub_all_tuning_forks_on_new_burst (sfxstatic_assets_areacradle.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Fade Audio
  }

  state Fade Audio {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0.5, time=1)
    on FINISHED → Death Loop 2
  }
}
