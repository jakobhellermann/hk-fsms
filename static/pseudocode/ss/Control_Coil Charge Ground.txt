# Control
# 1 location(s):
#   Bellway_Centipede_Arena / Boss Scene/Coil Charge Ground

fsm Control {
  start Init

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=(unset), z=(unset), space=0, everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Pt Antic")
    FindNamedChild(gameObject=Self, storeResult=var "Coils")
    FindNamedChild(gameObject=Self, storeResult=var "Drop Bells")
    on FINISHED → Idle
  }

  state Idle {
    Wait(time=1, finishEvent=→"ATTACK", realTime=false)  // disabled
    on ATTACK → Flip
    on ATTACK NOSCALE → Attack
  }

  state Attack {
    Wait(time=3.8, finishEvent=→"FINISHED", realTime=false)
    DoCameraShakeRepeating(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=true, Delay=0, RepeatDelay=0.15)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0)  // disabled
    SetPosition(gameObject=var "Coils", vector=(unset), x=0, y=(unset), z=(unset), space=1, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=var "Coils", translateMin=(-6, 0, 0), translateMax=(6, 0, 0), space=0)
    PlayParticleEmitter(gameObject=var "Pt Antic", emit=0, resetIfPlaying=false)
    SendEventByName(eventTarget=GameObject(var "Coils"), sendEvent="ATTACK", delay=0.3, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Drop Bells"), sendEvent="DROP", delay=1.73, everyFrame=false)
    on FINISHED → Idle
  }

  state Flip {
    RandomlyFlipScale(gameObject=Self)
    on FINISHED → Attack
  }

  var Start X: float = 0
  var Pt Antic: gameObject = <null>
  var Coils: gameObject = <null>
  var Drop Bells: gameObject = <null>
}
