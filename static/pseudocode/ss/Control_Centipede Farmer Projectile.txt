# Control
# 1 location(s):
#   localpoolprefabs_assets_areagreymoor.bundle / Centipede Farmer Projectile

fsm Control {
  start Init

  state Init {
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Projectile Fly")
    SetMeshRenderer(gameObject=Self, active=true)
    SetDamageHero(Target=Self, Enabled=true)
    SetPolygonCollider(gameObject=Self, active=true, resetOnExit=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    AddForce2d(gameObject=Self, forceMode=1, atPosition=(unset), vector=(unset), x=(unset), y=5, vector3=(unset), everyFrame=false)  // disabled
    on FINISHED → Throw Effect
  }

  state Fly {
    Collision2dEvent(gameObject=Self, collision=1, collideTag="", sendEvent=→"HIT TERRAIN", storeCollider=var "", storeForce=(unset))
    Collision2dEvent(gameObject=Self, collision=0, collideTag="", sendEvent=→"HIT TERRAIN", storeCollider=var "", storeForce=(unset))
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=true)
    on HIT TERRAIN → Stick
    on PARRIED → Check
    on PROJECTILE BREAK → Break
  }

  state Stick {
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayAudioEvent(audioClip=spear_impact (sfxstatic_assets_areapeak.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetAngularVelocity2d(gameObject=Self, angularVelocity=0, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Projectile Thunk")
    ActivateGameObject(gameObject=var "Thunk Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"WAIT", realTime=false)
    on WAIT → Break
  }

  state Recycle {
    RecycleSelf()
  }

  state Break {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    PlayParticleEmitter(gameObject=var "Pt Break", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Break Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(unset), x=0, y=0, everyFrame=false)
    on FINISHED → Recycle
  }

  state Deflect {
    FloatAdd(floatVariable=var "Angle", add=var "Angle Mod", everyFrame=false, perSecond=false)
    GetSpeed2d(gameObject=Self, storeResult=var "Speed", everyFrame=false)
    FloatMultiply(floatVariable=var "Speed", multiplyBy=2, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=var "Speed", everyFrame=false)
    SetDamageHero(Target=Self, Enabled=false)
    on FINISHED → Fly
  }

  state Check {
    GetVelocityAsAngle(gameObject=Self, storeAngle=var "Angle", everyFrame=false)
    FloatCompare(float1=var "Angle", float2=0, tolerance=1, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    FloatCompare(float1=var "Angle", float2=180, tolerance=0, equal=(none), lessThan=(none), greaterThan=(none), everyFrame=false)
    on L → Deflect L
    on R → Deflect R
    on FINISHED → Deflect Other
  }

  state Deflect L {
    RandomFloat(min=-10, max=-13, storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Deflect R {
    RandomFloat(min=10, max=13, storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Deflect Other {
    RandomFloat(min=5, max=10, storeResult=var "Angle Mod")
    RandomlyFlipFloat(storeResult=var "Angle Mod")
    on FINISHED → Deflect
  }

  state Throw Effect {
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", everyFrame=false, pauseBetweenTurns=false, pauseTime=0)
    FaceAngle(gameObject=Self, angleOffset=180, angleOffsetIfMirrored=0, otherGameObject=<null>, everyFrame=false)
    ActivateGameObject(gameObject=var "NeedleThrow Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Fly
  }

  var Angle: float = 0
  var Angle Mod: float = 0
  var Speed: float = 0
  var Break Effect: gameObject = Centipede Farmer Projectile/Break Effect
  var Pt Break: gameObject = Centipede Farmer Projectile/Pt Break
  var Thunk Effect: gameObject = Centipede Farmer Projectile/Thunk Effect
  var NeedleThrow Effect: gameObject = Centipede Farmer Projectile/NeedleThrow Effect
}
