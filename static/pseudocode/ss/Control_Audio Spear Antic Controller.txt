# Control
# 1 location(s):
#   Memory_Coral_Tower / Audio Spear Antic Controller

fsm Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on SPEAR ANTIC START → Antic
  }

  state Antic {
    AudioPlaySimple(gameObject=Self, volume=0.5, oneShotClip=<null>)
    on SPEAR ANTIC END → End
  }

  state End {
    AudioStop(gameObject=Self, fadeTime=0.25)
    on FINISHED → Idle
  }
}
