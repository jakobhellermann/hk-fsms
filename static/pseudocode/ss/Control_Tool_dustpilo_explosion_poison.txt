# Control
# 1 location(s):
#   herodynamic_assets_all.bundle / Tool_dustpilo_explosion_poison

fsm Control {
  start Explode
  on CLEAR EFFECTS → Recycle  // from any state

  state Explode {
    SetTag(gameObject=Self, tag="Dust Explosion")
    PlayAudioEvent(audioClip=explosion_5 (herostatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitter(gameObject=var "Pt Smoke", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt PreWarm", emit=0, resetIfPlaying=false)
    PlayParticleEmitter(gameObject=var "Pt Chaff", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Radial Burst Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CreateNoise(From=Self, LocalOrigin=(0, 0), Radius=6)
    on FINISHED → Active
  }

  sequence state Active {
    Wait(time=0.5, finishEvent=(none), realTime=false)
    StopParticleEmitter(gameObject=var "Pt PreWarm")
    Wait(time=5.5, finishEvent=(none), realTime=false)
    on FINISHED → End
    on END → End
  }

  sequence state End {
    SetTag(gameObject=Self, tag="Untagged")
    StopParticleEmitter(gameObject=var "Pt Smoke")
    Wait(time=0.5, finishEvent=(none), realTime=false)
    StopParticleEmitter(gameObject=var "Pt Chaff")
    Wait(time=0.75, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Enemy Damager", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.75, finishEvent=(none), realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Erase Others Nearby {
    FindClosest(gameObject=Self, withTag="Dust Explosion", ignoreOwner=true, mustBeVisible=false, storeObject=var "Other Explosion", storeDistance=(unset), everyFrame=false)
    GameObjectIsNull(gameObject=var "Other Explosion", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetDistanceV2(gameObject=Self, target=var "Other Explosion", targetOffsetX=(unset), targetOffsetY=(unset), storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=4, tolerance=0, equal=(none), lessThan=→"ERASE", greaterThan=(none), everyFrame=false)
    on FINISHED → Active
    on ERASE → Do Erase
  }

  state Do Erase {
    SendEventByName(eventTarget=GameObject(var "Other Explosion"), sendEvent="END", delay=0, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Erase Others Nearby
  }

  var Distance: float = 0
  var Pt Smoke: gameObject = Tool_dustpilo_explosion_poison/Pt Smoke
  var Radial Burst Effect: gameObject = Tool_dustpilo_explosion_poison/Radial Burst Effect
  var Pt Chaff: gameObject = Tool_dustpilo_explosion_poison/Pt Chaff
  var Pt PreWarm: gameObject = Tool_dustpilo_explosion_poison/Pt PreWarm
  var Enemy Damager: gameObject = Tool_dustpilo_explosion_poison/Enemy Damager
  var Other Explosion: gameObject = <null>
}
