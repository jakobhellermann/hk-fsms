# FSM
# 1 location(s):
#   Coral_Judge_Arena / Music Fader L

fsm FSM {
  start State 1

  state State 1 {
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → State 2
  }

  state State 2 {
    GetEntryGateName(StoreValue=var "Gate Name")
    StringCompare(stringVariable=var "Gate Name", compareTo=var "Allowed Gate", equalEvent=→"ACTIVATE", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on ACTIVATE → State 3
  }

  state State 3 {
    SetCollider(gameObject=Self, active=true, resetOnExit=true)
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Fade
  }

  state Fade {
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=1)
  }

  var Allowed Gate: string = "right1"

  var Gate Name: string = ""
}
