# Detect Hit
# 6 location(s):
#   Greymoor_04 / propeller (1)/lever/Control Lever
#   Greymoor_04 / propeller (2)/lever/Control Lever
#   Greymoor_04 / propeller (3)/lever/Control Lever
#   Greymoor_04 / propeller (4)/lever/Control Lever
#   Greymoor_11 / propeller/lever/Control Lever
#   Greymoor_11 / propeller (1)/lever/Control Lever

fsm Detect Hit {
  start Init
  on ACTIVATED → Activated  // from any state

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=var "Magnitude Multiplier", firstHitOnly=true)
    on HIT → Range Check
  }

  state Send {
    FloatClamp(floatVariable=var "Magnitude Multiplier", minValue=1, maxValue=2, everyFrame=false)
    SetFsmFloat(gameObject=var "Parent", fsmName="Control", variableName="Magnitude Multiplier", setValue=var "Magnitude Multiplier", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="LEVER HIT", delay=0, everyFrame=false)
    Wait(time=0.2, finishEvent=(none), realTime=false)
    on FINISHED → Idle
  }

  state Init {
    GetGrandparent(gameObject=Self, storeResult=var "Parent")
    FindChild(gameObject=Self, childName="NoHit Range", storeResult=var "NoHit Range Obj")
    FindAlertRange(target=Self, storeResult=var "NoHit Range", childName="NoHit Range")
    on FINISHED → Idle
  }

  state Activated {
    GetNextChild(gameObject=Self, storeNextChild=var "Child", loopEvent=→"LOOP", finishedEvent=→"FINISHED", resetFlag=(unset))
    on LOOP → Set Material
    on FINISHED → Inert
  }

  state Inert {
  }

  state Set Material {
    SetMaterial(gameObject=var "Child", materialIndex=0, material=Crossroads_Material (materials_assets_shared.bundle))
    on FINISHED → Activated
  }

  state Range Check {
    GameObjectIsNull(gameObject=var "NoHit Range Obj", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    CheckAlertRange(alertRange=var "NoHit Range", storeResult=(unset), InRangeEvent=→"CANCEL", InRangeDelay=0, OutOfRangeEvent=(none), OutOfRangeDelay=0, everyFrame=false)
    on FINISHED → Send
    on CANCEL → Idle
  }

  var Magnitude Multiplier: float = 0
  var Child: gameObject = <null>
  var NoHit Range Obj: gameObject = <null>
  var Parent: gameObject = <null>
  var NoHit Range: object = <null>
}
