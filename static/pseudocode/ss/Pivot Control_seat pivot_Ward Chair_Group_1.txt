# Pivot Control
# 1 location(s):
#   Ward_07 / Group:1/Ward Chair/seat pivot

fsm Pivot Control {
  start Init

  state Init {
    SendEventByScale(gameObject=var "Parent", eventTarget=Self, xScale=true, positiveEvent=→"POSITIVE", negativeEvent=→"NEGATIVE", space=0)
    on POSITIVE → Pos
    on NEGATIVE → Pos
  }

  state Pos {
    SetFloatValue(floatVariable=var "Hit L Rotate", floatValue=5, everyFrame=false)
    SetFloatValue(floatVariable=var "Hit R Rotate", floatValue=-5, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on HIT LEFT → Hit L
    on HIT RIGHT → Hit R
  }

  state Neg {
    SetFloatValue(floatVariable=var "Hit L Rotate", floatValue=-5, everyFrame=false)
    SetFloatValue(floatVariable=var "Hit R Rotate", floatValue=5, everyFrame=false)
    on FINISHED → Idle
  }

  state Hit L {
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Hit L Rotate", worldSpace=true, time=0.15, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rotate Back L
  }

  state Rotate Back L {
    AnimateRotationToV2(gameObject=Self, fromValue=var "Hit L Rotate", toValue=0, worldSpace=true, time=0.3, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on HIT LEFT → Hit L
    on HIT RIGHT → Hit R
  }

  state Hit R {
    AnimateRotationToV2(gameObject=Self, fromValue=(unset), toValue=var "Hit R Rotate", worldSpace=true, time=0.15, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rotate Back R
  }

  state Rotate Back R {
    AnimateRotationToV2(gameObject=Self, fromValue=var "Hit R Rotate", toValue=0, worldSpace=true, time=0.3, speed=(unset), delay=(unset), easeType=14, reverse=false, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on HIT LEFT → Hit L
    on HIT RIGHT → Hit R
  }

  var Hit L Rotate: float = 0
  var Hit R Rotate: float = 0
  var Parent: gameObject = Group:1/Ward Chair
}
