// uses template: silk_blade_tink
fsm Tink {
  start Idle
  on RECOIL STOP → Recoil Stop  // from any state
  on RECOIL START → Recoil Start  // from any state

  state Idle {
    on TINK LEFT → Tink L
    on TINK RIGHT → Tink R
    on TINK UP → Tink U
    on TINK DOWN → Tink D
  }

  state Tink L {
    BoolTest(boolVariable=var "Can Recoil", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=Self, vector=(unset), x=-15, y=(unset), z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Send Event
  }

  state Tink R {
    BoolTest(boolVariable=var "Can Recoil", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=Self, vector=(unset), x=15, y=(unset), z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Send Event
  }

  state Tink U {
    BoolTest(boolVariable=var "Can Recoil", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=15, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Send Event
  }

  state Tink D {
    BoolTest(boolVariable=var "Can Recoil", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=-15, z=(unset), space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Send Event
  }

  state Recoil Stop {
    SetBoolValue(boolVariable=var "Can Recoil", boolValue=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Send Event {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="Control"), sendEvent="TINKED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Recoil Start {
    SetBoolValue(boolVariable=var "Can Recoil", boolValue=true, everyFrame=false)
    on FINISHED → Idle
  }

  var Can Recoil: bool = true
}
