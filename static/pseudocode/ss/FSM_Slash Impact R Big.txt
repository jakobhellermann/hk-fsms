# FSM
# 1 location(s):
#   globalpoolprefabs_assets_all.bundle / Slash Impact R Big

fsm FSM {
  start Vibrate
  on A RECYCLE → Reset  // from any state

  state Anim {
    SelectRandomString(strings=["Slash Impact 1 ", "Slash Impact 2"], weights=[1, 1], storeString=var "Anim")
    RandomFloatV2(min=-30, max=30, storeResult=var "Rotation", everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=1, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Impact 1", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName=var "Anim", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")  // disabled
    Tk2dPlayFrame(gameObject=Self, frame=0)
    Wait(time=0.17, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Reset {
    SetScale(gameObject=Self, vector=(unset), x=1, y=1.5, z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Dormant
  }

  state Dormant {
    on A SPAWN → Anim
  }

  state Vibrate {
    CheckSceneName(sceneName="Menu_Title", equalEvent=→"FINISHED", notEqualEvent=(none))
    VibrationPlayerPlay(target=Self)
    on FINISHED → Anim
  }

  var Rotation: float = 0
  var Chooser: bool = false
  var Anim: string = "Slash Impact 1 "
}
