# Control
# 1 location(s):
#   Halfway_01 / Music Control

fsm Control {
  start Pause

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade Up
  }

  state Fade Up {
    AudioPlaySimple(gameObject=Self, volume=0, oneShotClip=<null>)  // disabled
    FadeAudio(gameObject=Self, startVolume=0, endVolume=1, time=2)  // disabled
    SendEventToRegister(eventName="REST AREA MUSIC START")
    on LEAVING SCENE → Leave Scene
  }

  state Leave Scene {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0, time=1)  // disabled
  }
}
