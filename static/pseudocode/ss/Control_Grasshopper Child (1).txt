# Control
# 1 location(s):
#   Clover_11 / Grasshopper Child (1)

fsm Control {
  start Init
  on DISABLE → Stop  // from any state

  state Init {
    GetDefaultTk2dClipName(gameObject=Self, storeName=var "Clip")
    StringCompare(stringVariable=var "Clip", compareTo="Wall Idle", equalEvent=→"WALL", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    StringCompare(stringVariable=var "Clip", compareTo="Fly In", equalEvent=→"FLY IN", notEqualEvent=(none), storeResult=(unset), everyFrame=false)
    on FINISHED → Spawn Startle
    on WALL → Wall Idle
    on FLY IN → Fly In Ready
  }

  state Fly Aggro {
    AudioPlay(gameObject=Self, volume=1, oneShotClip=var "", WaitForEndOfClip=false, finishedEvent=(none))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly")
    WaitRandom(timeMin=0.25, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    RandomFloat(min=0, max=360, storeResult=var "Angle")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=1, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", resetFrame=true, everyFrame=true, pauseBetweenTurns=0.5)
    SetInvincible(target=Self, Invincible=var "Do Block", InvincibleFromDirection=0, resetOnStateExit=false)
    on FINISHED → Unalert?
    on BLOCKED HIT → Block
  }

  state Flit Aim {
    SetBoolValue(boolVariable=var "Did Attack", boolValue=false, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    GetScale(gameObject=Self, vector=(unset), xScale=var "Target Offset", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    FloatSignToBool(Value=var "Target Offset", StoreIsPositive=var "Facing L", EveryFrame=false)
    FloatMultiply(floatVariable=var "Target Offset", multiplyBy=6.5, everyFrame=false)
    SetPositionToObject(gameObject=var "Target", targetObject=var "Hero", xOffset=var "Target Offset", yOffset=1, zOffset=0, overrideZ=(unset), everyFrame=false)
    RandomFloat(min=-1.5, max=1.5, storeResult=var "Y Offset")
    GetAngleToTarget2D(gameObject=Self, target=var "Target", offsetX=0, offsetY=var "Y Offset", storeAngle=var "Angle", pause=0, everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Target", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Aim L
    on R → Aim R
  }

  state Aim L {
    ClampAngle(angleVariable=var "Angle", minValue=145, maxValue=215, everyFrame=false)
    SetBoolValue(boolVariable=var "Going L", boolValue=true, everyFrame=false)
    on FINISHED → Flit Dir
  }

  state Aim R {
    ClampAngle(angleVariable=var "Angle", minValue=-35, maxValue=35, everyFrame=false)
    SetBoolValue(boolVariable=var "Going L", boolValue=false, everyFrame=false)
    on FINISHED → Flit Dir
  }

  state Flit Dir {
    BoolTestMulti(boolVariables=[var "Facing L", var "Going L"], boolStates=[true, true], trueEvent=→"FLIT F", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestMulti(boolVariables=[var "Facing L", var "Going L"], boolStates=[false, false], trueEvent=→"FLIT F", falseEvent=→"FLIT B", storeResult=(unset), everyFrame=false)
    on FLIT F → Flit F
    on FLIT B → Flit B
  }

  state Flit F {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Flit F")
    on FINISHED → Flit Antic
  }

  state Flit B {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Flit B")
    on FINISHED → Flit Antic
  }

  state Flit Antic {
    PlayAudioEvent(audioClip=grasshopper_child_antic (sfxstatic_assets_areaclover.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    on FINISHED → Flit
    on BLOCKED HIT → Block
  }

  state Flit {
    PlayAudioEventRandom(audioClips=[grasshopper_child_dash_1 (sfxstatic_assets_areaclover.bundle), grasshopper_child_dash_2 (sfxstatic_assets_areaclover.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=25, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Flit Recover
    on GO UP → Flit Go Up
    on BLOCKED HIT → Block
  }

  state Flit Recover {
    DecelerateV2(gameObject=Self, deceleration=0.85, brakeOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fly Aggro
    on BLOCKED HIT → Block
  }

  state Choice {
    BoolTest(boolVariable=var "Did Attack", isTrue=→"FLIT", isFalse=(none), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Attack Range", storeResult=false, sendEvent="ATTACK", outOfRangeEvent="FLIT", everyFrame=false)
    on FLIT → Flit Aim
    on ATTACK → Attack Aim
  }

  state Attack Aim {
    SetBoolValue(boolVariable=var "Did Attack", boolValue=true, everyFrame=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Aim L 2
    on R → Aim R 2
  }

  state Aim L 2 {
    ClampAngle(angleVariable=var "Angle", minValue=145, maxValue=215, everyFrame=false)
    SetBoolValue(boolVariable=var "Going L", boolValue=true, everyFrame=false)
    on FINISHED → Attack Antic
  }

  state Aim R 2 {
    ClampAngle(angleVariable=var "Angle", minValue=-35, maxValue=35, everyFrame=false)
    SetBoolValue(boolVariable=var "Going L", boolValue=false, everyFrame=false)
    on FINISHED → Attack Antic
  }

  state Attack Antic {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=clover_grasshopper_child (sfxdynamic_assets_areaclover.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayAudioEvent(audioClip=grasshopper_child_antic (sfxstatic_assets_areaclover.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    SetVelocity2d(gameObject=Self, vector=(0, 5), x=(unset), y=(unset), everyFrame=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Slash Attack", animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    SetInvincible(target=Self, Invincible=false, InvincibleFromDirection=0, resetOnStateExit=false)
    on FINISHED → Attack Charge
  }

  state Attack Charge {
    PlayAudioEvent(audioClip=grasshopper_child_attack (sfxstatic_assets_areaclover.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Current Clip")
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=32, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=true)
    Wait(time=1, finishEvent=→"END", realTime=false)
    ActivateGameObject(gameObject=var "Slashes", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on END → Recover
    on GO UP → Atk Go Up
    on MULTI HIT CONNECT → Multislash
  }

  state Recover {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fly Aggro
  }

  state Wall Idle {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Alert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=(none), ActiveInner=→"ALERT", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ALERT → Startle Antic
    on TOOK DAMAGE → Startle Antic
  }

  state Startle Antic {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Wall Antic", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=clover_grasshopper_child (sfxdynamic_assets_areaclover.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Startle
  }

  state Startle {
    PlayAudioEvent(audioClip=grasshopper_child_getup_from_idle (sfxstatic_assets_areaclover.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetVelocityByScale(gameObject=Self, speed=-15, ySpeed=(unset), everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    on FINISHED → Fly Aggro
  }

  state Flit Go Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=1, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.95, brakeOnExit=false)
    Tk2dWatchAnimationEvents(gameObject=Self, animationTriggerEvent=→"FINISHED", animationCompleteEvent=(none))
    on FINISHED → Flit Recover
    on BLOCKED HIT → Block
  }

  state Atk Go Up {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=10, everyFrame=false)
    on FINISHED → Recover
  }

  state Spawn Startle {
    SetParent(gameObject=Self, parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Angle", space=0, everyFrame=false)
    SetVelocityAsAngle(gameObject=Self, angle=var "Angle", speed=20, everyFrame=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Startle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=clover_grasshopper_child (sfxdynamic_assets_areaclover.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Fly Aggro
  }

  state Sing {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Sing")
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.1, MaxReactDelay=0.3, None=→"END", ActiveInner=(none), ActiveOuter=(none), IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on END → Sing End
  }

  state Sing End {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Recover", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Fly Aggro
  }

  state Unalert? {
    BoolTest(boolVariable=var "Battler", isTrue=→"FINISHED", isFalse=(none), everyFrame=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Unalert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=false)
    IntAdd(intVariable=var "Unalerts", add=1, everyFrame=false)
    IntCompare(integer1=var "Unalerts", integer2=4, equal=(none), lessThan=(none), greaterThan=→"UNALERT", everyFrame=false)
    on FINISHED → Choice
    on ALERT → In Alert Range
    on UNALERT → Air Idle
  }

  state In Alert Range {
    SetIntValue(intVariable=var "Unalerts", intValue=0, everyFrame=false)
    on FINISHED → Choice
  }

  state Air Idle {
    SetIntValue(intVariable=var "Unalerts", intValue=0, everyFrame=false)
    IdleBuzzV2(gameObject=Self, waitMin=0.5, waitMax=1, speedMax=2.5, accelerationMax=15, roamingRangeX=4, roamingRangeY=4, manualStartPos=(unset))
    FaceDirection(gameObject=Self, spriteFacesRight=false, playNewAnimation=true, newAnimationClip="TurnToFly", everyFrame=true, pauseBetweenTurns=true, pauseTime=0.5)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Unalert Range", storeResult=false, sendEvent="ALERT", outOfRangeEvent="", everyFrame=true)
    CheckHeroPerformanceRegion(Target=Self, MinReactDelay=0.2, MaxReactDelay=0.3, None=(none), ActiveInner=→"ALERT", ActiveOuter=→"ALERT", IgnoreNeedolinRange=false, useActiveBool=false, ActiveBool=(unset), StoreState=None, EveryFrame=true)
    on ALERT → Startle
    on TOOK DAMAGE → Startle
  }

  state Stop {
    AudioStop(gameObject=var "Current Clip", fadeTime=0)
  }

  state Fly In Ready {
    SetMeshRenderer(gameObject=Self, active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(0, 0, 0, 1), everyframe=false)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Ambush Range", storeResult=false, sendEvent="WAKE", outOfRangeEvent="", everyFrame=true)
    on BATTLE START → Set Battler
    on WAKE → Set Battler
  }

  state Check Pos {
    GetDistance(gameObject=Self, target=var "Hero", storeResult=var "Distance", everyFrame=false)
    FloatCompare(float1=var "Distance", float2=6, tolerance=0, equal=(none), lessThan=→"SHIFT", greaterThan=(none), everyFrame=false)
    on SHIFT → Shift
    on FINISHED → Fly In
  }

  state Shift {
    SetParent(gameObject=var "Spawn Pos Alt", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetPositionToObject(gameObject=Self, targetObject=var "Spawn Pos Alt", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    on FINISHED → Fly In
  }

  state Fly In {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Fly In")
    PlayRandomAudioClipTable(Table=enemy_fly_in (sfxdynamic_assets_battle.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=clover_grasshopper_child (sfxdynamic_assets_areaclover.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Translate(gameObject=Self, vector=(unset), x=(unset), y=10, z=-15, space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    SetMeshRenderer(gameObject=Self, active=true)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -10, 15), time=0.75, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Tk2dSpriteTweenColor(gameObject=Self, color=(1, 1, 1, 1), duration=0.55)
    SetBoolValue(boolVariable=var "Startled", boolValue=true, everyFrame=false)
    on FINISHED → Activate
  }

  state Activate {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dSpriteSetColor(gameObject=Self, color=(1, 1, 1, 1), everyframe=false)
    on FINISHED → Fly Aggro
  }

  state Set Battler {
    SetBoolValue(boolVariable=var "Battler", boolValue=true, everyFrame=false)
    on FINISHED → Fly In
  }

  state Multislash {
    Wait(time=0.4, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Slashes", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Recover
  }

  state Block {
    FaceObjectV2(objectA=Self, objectB=var "Hero", spriteFacesRight=false, playNewAnimation=false, newAnimationClip="", resetFrame=false, everyFrame=false, pauseBetweenTurns=0.5)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Block")
    SetVelocityByScale(gameObject=Self, speed=30, ySpeed=5, everyFrame=false)
    DecelerateV2(gameObject=Self, deceleration=0.9, brakeOnExit=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choice
  }

  var Angle: float = 0
  var Distance: float = 0
  var Target Offset: float = 0
  var Y Offset: float = 0
  var Unalerts: int = 0
  var Going L: bool = false
  var Facing L: bool = false
  var Did Attack: bool = false
  var Battler: bool = false
  var In Battle: bool = false
  var Startled: bool = false
  var Do Block: bool = false
  var Clip: string = ""
  var Target: gameObject = Grasshopper Child (1)/Target
  var Slashes: gameObject = Grasshopper Child (1)/Slashes
  var Current Clip: gameObject = <null>
  var Spawn Pos Alt: gameObject = Grasshopper Child (1)/Spawn Pos Alt
  var Audio Loop Voice: gameObject = Grasshopper Child (1)/Audio Loop Voice
}
