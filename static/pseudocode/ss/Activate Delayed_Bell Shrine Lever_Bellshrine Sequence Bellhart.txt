# Activate Delayed
# 1 location(s):
#   Belltown_Shrine / Black Thread States Thread Only Variant/Normal World/Boss Scene/Bellshrine Sequence Bellhart/Bell Shrine Lever

fsm Activate Delayed {
  start Inert

  state Inert {
    on ACTIVATE DELAYED → Deactivate
    on ACTIVATE INSTANT → Instantly ACtive
  }

  state Deactivate {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    AnimatorPlay(gameObject=Self, stateName="Hidden", layer=(unset), normalizedTime=(unset), everyFrame=false)
    SetProperty(targetProperty=SetTriggerRandom.enabled on <null>, everyFrame=false)
    on FINISHED → Wait
  }

  state Wait {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Activate
  }

  sequence state Activate {
    SetProperty(targetProperty=SetTriggerRandom on <null>, everyFrame=false)
    PlayAudioEvent(audioClip=bellshrine_switch_appear (sfxstatic_assets_areabellshrine.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatorPlayStateWait(target=Self, stateName="Appear", finishEvent=(none))
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    on HIT LEVER → Hit Lever
  }

  state Hit Lever {
    PlayAudioEvent(audioClip=bellshrine_switch_activate (sfxstatic_assets_areabellshrine.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="FreezeMoment", parameters=[1], storeResult=(unset var), EveryFrame=false)
  }

  state Instantly ACtive {
    on HIT LEVER → Hit Lever
  }
}
