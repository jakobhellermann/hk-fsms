# Spark Effects
# 1 location(s):
#   localpoolprefabs_assets_vampiregnat.bundle / Vampire Gnat Sickle

fsm Spark Effects {
  start Init
  on EFFECT STOP → Effect Stop  // from any state

  state Init {
    ActivateGameObject(gameObject=var "Spark Light", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(0, -1), space=0, distance=1.2, minDepth=(unset), maxDepth=(unset), hitEvent=→"SPARK", noHitEvent=(none), storeDidHit=(unset), storeHitObject=var "", storeHitPoint=(unset), storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    on SPARK → Spark
  }

  state Spark {
    RayCast2dV2(fromGameObject=Self, fromPosition=(unset), direction=(0, -1), space=0, distance=1.2, minDepth=(unset), maxDepth=(unset), hitEvent=(none), noHitEvent=→"END", storeDidHit=(unset), storeHitObject=var "", storeHitPoint=var "Hit Point", storeHitNormal=(unset), storeHitDistance=(unset), storeDistance=(unset), repeatInterval=1, layerMask=[Terrain(8)], invertMask=false, ignoreTriggers=false, debugColor=(1, 0.92156863, 0.015686275, 1), debug=false)
    SetPosition2D(GameObject=var "Pt Spark", Vector=var "Hit Point", X=0, Y=0, Space=0, EveryFrame=true)
    SetPosition2D(GameObject=var "Spark Light", Vector=var "Hit Point", X=0, Y=0, Space=0, EveryFrame=true)
    PlayParticleEmitterInState(gameObject=var "Pt Spark")
    ActivateGameObject(gameObject=var "Spark Light", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    AudioPlayInState(gameObject=var "Audio Loop Grind", volume=1)  // disabled
    on END → Idle
  }

  state Effect Stop {
  }

  var Hit Point: vector2 = (0, 0)
  var Pt Spark: gameObject = Vampire Gnat Sickle/Pt Spark
  var Spark Light: gameObject = Vampire Gnat Sickle/Spark Light
  var Audio Loop Grind: gameObject = Vampire Gnat Sickle/Audio Loop Grind
}
