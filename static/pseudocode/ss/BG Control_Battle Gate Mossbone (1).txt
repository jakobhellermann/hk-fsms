# BG Control
# 1 location(s):
#   Tut_03 / Black Thread States/Normal World/Battle Scene/Gates/Battle Gate Mossbone (1)

fsm BG Control {
  start Init
  on BG QUICK OPEN → Opened  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Over", active=false)
    on BG CLOSE → Close
    on BG QUICK CLOSE → Quick Close
  }

  state Close {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gate Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=var "Over", animLibName="", clipName="Gate Close")
    SetMeshRenderer(gameObject=var "Over", active=true)
    RandomFloat(min=0, max=0.1, storeResult=var "Audio Delay")
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=vine_door_move (sfxstatic_assets_areamoss.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, delay=var "Audio Delay", storePlayer=<null>)
    on BG OPEN → Open
    on FINISHED → Closed
  }

  state Open {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    Tk2dPlayAnimationWithEvents(gameObject=var "Over", clipName="Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    PlayParticleEmitter(gameObject=var "Pt Drip", emit=0, resetIfPlaying=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=vine_door_move (sfxstatic_assets_areamoss.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, delay=0, storePlayer=<null>)
    on BG CLOSE → 
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Opened
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Gate Closed")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayAnimation(gameObject=var "Over", animLibName="", clipName="Gate Closed")
    SetMeshRenderer(gameObject=var "Over", active=true)
    on BG OPEN → Open
    on FINISHED → Closed
  }

  state Opened {
    SetMeshRenderer(gameObject=Self, active=false)
    SetMeshRenderer(gameObject=var "Over", active=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    on BG CLOSE → Close
  }

  state Closed {
    on BG OPEN → Open
    on TINKED → Tinked
  }

  state Tinked {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Gate Hit")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    on FINISHED → Closed
  }

  var Start Closed: bool = false

  var Audio Delay: float = 0
  var Over: gameObject = Black Thread States/Normal World/Battle Scene/Gates/Battle Gate Mossbone (1)/Over
  var Pt Drip: gameObject = Black Thread States/Normal World/Battle Scene/Gates/Battle Gate Mossbone (1)/Pt Drip
  var Self: gameObject = <null>
}
