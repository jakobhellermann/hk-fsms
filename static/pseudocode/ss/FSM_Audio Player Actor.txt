# FSM
# 1 location(s):
#   Library_08 / Vaultkeeper Mutter/Mutter child/Audio Player Actor

fsm FSM {
  start Idle

  state Idle {
    on VAULTKEEPER EMERGED → Emerged
  }

  state Emerged {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0, time=0.25)
    on VAULTKEEPER LEFT → Leaving
  }

  state Leaving {
    Wait(time=2, finishEvent=(none), realTime=false)
    on FINISHED → Submerged
    on VAULTKEEPER EMERGED → Emerged
  }

  state Submerged {
    FadeAudio(gameObject=Self, startVolume=(unset), endVolume=0.725, time=1)
    on VAULTKEEPER EMERGED → Emerged
  }
}
