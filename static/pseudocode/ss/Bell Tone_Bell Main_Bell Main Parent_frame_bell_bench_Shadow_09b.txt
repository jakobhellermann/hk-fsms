# Bell Tone
# 2 location(s):
#   Bellway_Aqueduct / Shadow_09b/bell_bench/frame/Bell Main Parent/Bell Main
#   Bellway_Shadow / Shadow_09b/bell_bench/frame/Bell Main Parent/Bell Main

fsm Bell Tone {
  start Init

  state Init {
    GetOwner(storeGameObject=var "Self")
    FindChild(gameObject=Self, childName="Flash", storeResult=var "Flash")
    FindChild(gameObject=Self, childName="Idle Shine", storeResult=var "Idle Shine")
    FindChild(gameObject=Self, childName="Orb Pt", storeResult=var "Orb Pt")
    on FINISHED → Wait For Hit
  }

  state Mid {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Rotation", lowerValue=var "Limit R", upperValue=85, boolVariable=(unset), trueEvent=→"R", falseEvent=(none), everyFrame=true)
    FloatInRange(floatVariable=var "Rotation", lowerValue=180, upperValue=var "Limit L", boolVariable=(unset), trueEvent=→"L", falseEvent=(none), everyFrame=true)
    on R → R
    on L → L
  }

  state R {
    AudioPlayRandomSingle(gameObject=var "Self", audioClip=Bell_deep (sfxstatic_assets_areabell.bundle), pitchMin=0.85, pitchMax=1.15)  // disabled
    BoolTest(boolVariable=var "Do Flash", isTrue=→"FLASH", isFalse=(none), everyFrame=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Rotation", lowerValue=var "Limit R", upperValue=85, boolVariable=(unset), trueEvent=(none), falseEvent=→"L", everyFrame=true)
    on L → Mid
    on FLASH → R Flash
  }

  state L {
    AudioPlayRandomSingle(gameObject=var "Self", audioClip=Bell_deep (sfxstatic_assets_areabell.bundle), pitchMin=0.85, pitchMax=1.15)  // disabled
    BoolTest(boolVariable=var "Do Flash", isTrue=→"FLASH", isFalse=(none), everyFrame=false)
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true)
    FloatCompare(float1=var "Rotation", float2=-30, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"R", everyFrame=true)  // disabled
    FloatInRange(floatVariable=var "Rotation", lowerValue=180, upperValue=var "Limit L", boolVariable=(unset), trueEvent=(none), falseEvent=→"R", everyFrame=true)
    on R → Mid
    on FLASH → L Flash
  }

  state Idle Shine {
    ActivateGameObject(gameObject=var "Idle Shine", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Wait For Hit
  }

  state Wait For Hit {
    SetBoolValue(boolVariable=var "Do Flash", boolValue=true, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="interactive_bell", variableName="Bench Bell", setValue=true, everyFrame=false)
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="Nail Attack", sendEvent=→"NAIL HIT", storeCollider=var "Hitter")
    WaitRandom(timeMin=3, timeMax=4, finishEvent=→"SHINE", realTime=false)  // disabled
    on NAIL HIT → Hit
    on SHINE → Idle Shine
    on ACTIVATED → Activated
  }

  state Hit {
    SendEventByNameV2(eventTarget=GameObjectFSM(Shadow_09b/bell_bench, fsm="Bell Bench Activation"), sendEvent="ACTIVATE", delay=0, everyFrame=false)
    on FINISHED → Mid
  }

  state R Flash {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true)
    FloatInRange(floatVariable=var "Rotation", lowerValue=var "Limit R", upperValue=85, boolVariable=(unset), trueEvent=(none), falseEvent=→"L", everyFrame=true)
    ActivateGameObject(gameObject=var "Flash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Orb Pt"), sendEvent="PLAY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Shadow_09b/bell_bench/frame, fsm="Glow Runes"), sendEvent="GLOW", delay=0, everyFrame=false)  // disabled
    on L → Mid
  }

  state L Flash {
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=true)
    FloatCompare(float1=var "Rotation", float2=-30, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"R", everyFrame=true)  // disabled
    FloatInRange(floatVariable=var "Rotation", lowerValue=180, upperValue=var "Limit L", boolVariable=(unset), trueEvent=(none), falseEvent=→"R", everyFrame=true)
    ActivateGameObject(gameObject=var "Flash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flash", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Orb Pt"), sendEvent="PLAY", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObjectFSM(Shadow_09b/bell_bench/frame, fsm="Glow Runes"), sendEvent="GLOW", delay=0, everyFrame=false)  // disabled
    on R → Mid
  }

  state Activated {
    SetBoolValue(boolVariable=var "Do Flash", boolValue=false, everyFrame=false)
    SetFsmBool(gameObject=Self, fsmName="interactive_bell", variableName="Bench Bell", setValue=false, everyFrame=false)
    on FINISHED → Mid
  }

  var Limit L: float = 345
  var Limit R: float = 15

  var Rotation: float = 0
  var Do Flash: bool = false
  var Flash: gameObject = <null>
  var Hitter: gameObject = <null>
  var Idle Shine: gameObject = <null>
  var Orb Pt: gameObject = <null>
  var Self: gameObject = <null>
}
