# Tentacle Control
# 1 location(s):
#   Library_08 / Librarian Scene/Librarian

fsm Tentacle Control {
  start Wait for Hero
  on FORCE MOVE TENTACLES → Send Move 2  // from any state

  state Idle {
    SetBoolValue(boolVariable=var "Is Paused", boolValue=false, everyFrame=false)
    WaitRandom(timeMin=2, timeMax=5, finishEvent=→"MOVE", realTime=false)
    on MOVE → Send Move
    on PAUSE TENTACLES → Paused
  }

  state Send Move {
    SendEventToRegister(eventName="LIBRARIAN TENTACLE MOVE")
    WaitRandom(timeMin=1, timeMax=1.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Send Idle
    on PAUSE TENTACLES → Paused
  }

  state Send Idle {
    SendEventToRegister(eventName="LIBRARIAN TENTACLE IDLE")
    on FINISHED → Idle
  }

  state Send Move 2 {
    SendEventToRegister(eventName="LIBRARIAN TENTACLE MOVE")
    on FORCE MOVE TENTACLES END → Send Idle 2
  }

  state Send Idle 2 {
    SendEventToRegister(eventName="LIBRARIAN TENTACLE IDLE")
    on FINISHED → Paused?
  }

  state Paused {
    SetBoolValue(boolVariable=var "Is Paused", boolValue=true, everyFrame=false)
    SendEventToRegister(eventName="LIBRARIAN TENTACLE IDLE")
    on RESUME TENTACLES → Idle
  }

  state Paused? {
    BoolTest(boolVariable=var "Is Paused", isTrue=→"TRUE", isFalse=→"FALSE", everyFrame=false)
    on FALSE → Idle
    on TRUE → Paused
  }

  state Wait for Hero {
    WaitForFinishedEnteringScene(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  var Is Paused: bool = false
}
