# Music Control
# 1 location(s):
#   Ant_09 / Battle Music (1)

// component disabled: this FSM does not run until something enables it
fsm Music Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on ALERT → Increment
  }

  state Increment {
    IntAdd(intVariable=var "Alerted", add=1, everyFrame=false)
    on FINISHED → Check
  }

  state Check {
    IntCompare(integer1=var "Alerted", integer2=0, equal=→"NORMAL", lessThan=→"NORMAL", greaterThan=→"BATTLE", everyFrame=false)
    on BATTLE → Battle
    on NORMAL → Normal
  }

  state Battle {
    TransitionToAudioSnapshot(snapshot=Action (audioobjects_assets_all.bundle), transitionTime=0.5)
    on ALERT → Increment
    on UNALERT → Decrement
  }

  state Decrement {
    IntAdd(intVariable=var "Alerted", add=-1, everyFrame=false)
    on FINISHED → Check
  }

  state Normal {
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=3)
    on ALERT → Increment
    on UNALERT → Decrement
  }

  var Alerted: int = 0
}
