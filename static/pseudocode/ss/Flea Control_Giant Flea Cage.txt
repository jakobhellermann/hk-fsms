fsm Flea Control {
  start Init

  state Init {
    FindNamedChild(gameObject=var "Parent", storeResult=var "Giant Flea")
    FindNamedChild(gameObject=Self, storeResult=var "Break Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Voice")
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Start Snore Audio
  }

  state Rescue {
    ActivateGameObject(gameObject=var "Giant Flea", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Break Effects", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=true, Delay=0)
  }

  state Idle {
    PlayerDataBoolTest(boolName="tamedGiantFlea", isTrue=→"TAMED", isFalse=(none))
    PlayerDataBoolTest(boolName="uncagedGiantFlea", isTrue=→"ACTIVATE", isFalse=(none))
    WaitRandom(timeMin=1.5, timeMax=2.75, finishEvent=→"CREAK", realTime=false)
    on ACTIVATE → Activated
    on BREAK → Rescue
    on TAMED → Tamed
    on HIT → Hit React
    on CREAK → Creak
    on RESCUE → Rescue
  }

  state Activated {
    ActivateGameObject(gameObject=var "Giant Flea", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Tamed {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Hit React {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=Giant Flea bonk  (sfxdynamic_assets_areaarborium.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on RESCUE → Rescue
    on FINISHED → Hit Creak
    on BREAK → Rescue
    on HIT → Hit React
  }

  state Creak {
    PlayRandomAudioClipTable(Table=metal_creak (sfxdynamic_assets_areasong.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayRandomAudioClipTable(Table=chain_move (sfxdynamic_assets_chain.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Idle
  }

  state Start Snore Audio {
    AudioPlaySimple(gameObject=var "Audio Loop Voice", volume=1, oneShotClip=<null>)
    on FINISHED → Idle
  }

  state Hit Creak {
    PlayRandomAudioClipTable(Table=metal_creak (sfxdynamic_assets_areasong.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    PlayRandomAudioClipTable(Table=chain_move (sfxdynamic_assets_chain.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    on FINISHED → Start Snore Audio
  }

  var Parent: gameObject = Giant Flea Scene

  var Break Effects: gameObject = <null>
  var Flea: gameObject = <null>
  var Giant Flea: gameObject = <null>
  var Audio Loop Voice: gameObject = <null>
  var Quest Tracker: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
