# Control
# 1 location(s):
#   Bonegrave / Chapel Door Control/Door DoClose

fsm Control {
  start Pause

  state Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    FindNamedChild(gameObject=Self, storeResult=var "Pt LandDust")
    FindNamedChild(gameObject=Self, storeResult=var "Art")
    ActivateGameObject(gameObject=var "Art", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Close
  }

  state Close {
    Translate(gameObject=Self, vector=(unset), x=(unset), y=7, z=(unset), space=0, perSecond=false, everyFrame=false, lateUpdate=false, fixedUpdate=false)
    AnimatePositionBy(gameObject=Self, shiftBy=(0, -7, 0), time=0.5, speed=(unset), delay=(unset), easeType=12, reverse=false, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=weaver_gate_close_start (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Art", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Slam
  }

  state Slam {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendEventToRegister(eventName="CHAPEL DOOR SLAM")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitter(gameObject=var "Pt LandDust", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=weaver_gate_close_end (sfxstatic_assets_areaweaver.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -10), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=song_gate_opened (sfxstatic_assets_areasong.bundle), pitchMin=1, pitchMax=1, volume=0.7, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, -10), SpawnedPlayerRef=var "")
    on FINISHED → Save
  }

  state Save {
    SaveGame()
  }

  var Pt LandDust: gameObject = <null>
  var Art: gameObject = <null>
}
