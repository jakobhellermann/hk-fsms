# Control
# 1 location(s):
#   Memory_Coral_Tower / Boss Scene/Air Spear

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Boss Scene")
    FindNamedChild(gameObject=var "Boss Scene", storeResult=var "Coral King")
    FindNamedChild(gameObject=Self, storeResult=var "Coral_spear_long")
    on FINISHED → Idle
  }

  state Idle {
    on AIR SPEAR → Attack
  }

  state Attack {
    SetPositionToObject(gameObject=Self, targetObject=var "Coral King", xOffset=0, yOffset=0, zOffset=0, overrideZ=(unset), everyFrame=false)
    GetScale(gameObject=var "Coral King", vector=(unset), xScale=var "X Scale", yScale=(unset), zScale=(unset), space=0, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "X Scale", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Coral_spear_long", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  var X Scale: float = 0
  var Boss Scene: gameObject = <null>
  var Coral King: gameObject = <null>
  var Coral_spear_long: gameObject = <null>
}
