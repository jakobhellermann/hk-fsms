# Control
# 1 location(s):
#   Dock_09 / Boss Scene/Lava_Waterfall M/Heat Glow

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Pt Sparks")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Sizzle")
    on FINISHED → Down
  }

  state Down {
    FadeNestedFadeGroup(Target=Self, ToAlpha=0, FadeTime=1)
    StopParticleEmitter(gameObject=var "Pt Sparks")
    FadeAudio(gameObject=var "Audio Loop Sizzle", startVolume=0.75, endVolume=0, time=1)
    on UP → Up Pause
  }

  state Up Pause {
    Wait(time=0.82, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Up
  }

  state Up {
    PlayParticleEmitter(gameObject=var "Pt Sparks", emit=0, resetIfPlaying=false)
    FadeNestedFadeGroup(Target=Self, ToAlpha=1, FadeTime=0.5)
    Wait(time=0.75, finishEvent=→"DOWN", realTime=false)
    AudioPlaySimpleV2(gameObject=var "Audio Loop Sizzle", volume=0, startTime=0, oneShotClip=<null>)
    FadeAudio(gameObject=var "Audio Loop Sizzle", startVolume=0, endVolume=0.75, time=0.5)
    on DOWN → Down Pause
  }

  state Down Pause {
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Down
  }

  var Pt Sparks: gameObject = <null>
  var Audio Loop Sizzle: gameObject = <null>
}
