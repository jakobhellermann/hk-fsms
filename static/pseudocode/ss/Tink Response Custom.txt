  start Init

  state Idle {

    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
    on TINK UP → Tink From Hero Dir
    on TINK DOWN → Tink From Hero Dir
    on LAND → Break
  }

  state Tink Left {
    PlayRandomAudioClipTable(Table=metal_creak (sfxdynamic_assets_areasong.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))

  }

    on FINISHED → Idle
  }

  state Break {
    SetRigidbodySimulated2D(gameObject=Self, isSimulated=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Active", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break", activate=true, recursive=false, resetOnExit=false, everyFrame=false)

    StopParticleEmitter(gameObject=City_spot_lamp_wheeled_interactive (1)/Trail Dust)
  }

  state Init {
    ActivateGameObject(gameObject=var "Active", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Break", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Tink From Hero Dir {
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), space=0, everyFrame=false)
    GetPosition2D(gameObject=Self, vector=(unset), x=var "Self X", y=(unset), space=0, everyFrame=false)
    FloatCompare(float1=var "Hero X", float2=var "Self X", tolerance=0, equal=→"TINK RIGHT", lessThan=→"TINK RIGHT", greaterThan=→"TINK LEFT", everyFrame=false)
    on TINK LEFT → Tink Left
    on TINK RIGHT → Tink Right
  }

  var Break Y: float = 32.59

  var Hero X: float = 0
  var Self X: float = 0
  var Active: gameObject = City_spot_lamp_wheeled_interactive (1)/Active
  var Break: gameObject = City_spot_lamp_wheeled_interactive (1)/Break
}
