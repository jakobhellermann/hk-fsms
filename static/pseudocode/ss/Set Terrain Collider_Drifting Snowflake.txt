# Set Terrain Collider
# 1 location(s):
#   Peak_02 / Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake

fsm Set Terrain Collider {
  start Stop

  state Stop {
    on START → Start
  }

  state Start {
    ActivateGameObject(gameObject=var "Terrain Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObjectDelay(gameObject=var "Terrain Collider", activate=true, resetOnExit=false, delay=3.5)
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop
  }

  var Terrain Collider: gameObject = Snowflake Scene (2)/Snowflake Spawner/Drifting Snowflake/Terrain Collider
}
