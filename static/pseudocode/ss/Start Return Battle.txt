# Start Return Battle
# 1 location(s):
#   Bone_05_boss / Boss Scene/Return Battle

fsm Start Return Battle {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Scene Setup
  }

  state Scene Setup {
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="BigShake", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gate 1"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Gate 2"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt DropBells", emit=0, resetIfPlaying=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "CamLock Battle", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "CamLock PreBattle", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Start Battle
  }

  state Rumble {
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=true, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt DropBells Long", emit=0, resetIfPlaying=false)
    on FINISHED → Start Battle
  }

  state Start Battle {
    SendEventByName(eventTarget=GameObject(var "Boss"), sendEvent="BATTLE START", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Title
  }

  state Title {
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=false, bossTitle="BELLBEAST")
    SetFsmBool(gameObject=var "CameraParent", fsmName="CameraShake", variableName="RumblingMed", setValue=false, everyFrame=false)
    ApplyMusicCue(musicCue=BellBattle (audiocuesdynamic_assets_areabell.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Normal (audioobjects_assets_all.bundle), transitionTime=0.25)
  }

  var Boss: gameObject = Boss Scene/Bone Beast
  var CamLock Battle: gameObject = Boss Scene/CamLock Battle
  var CamLock PreBattle: gameObject = Boss Scene/CamLock PreBattle
  var Gate 1: gameObject = Boss Scene/Threaded Bell Gate
  var Gate 2: gameObject = Boss Scene/Threaded Bell Gate (1)
  var Pt DropBells: gameObject = Boss Scene/Pt DropBells
  var Pt DropBells Long: gameObject = Boss Scene/Pt DropBells Long
}
