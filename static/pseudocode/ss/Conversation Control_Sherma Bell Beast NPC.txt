# Conversation Control
# 1 location(s):
#   Bone_05_boss / Sherma Bell Beast NPC

fsm Conversation Control {
  start Init

  state Init {
    on FINISHED → Check State
  }

  state Check State {
    GetPlayerDataInt(intName="shermaPos", storeValue=var "Sherma Pos")
    IntCompare(integer1=var "Sherma Pos", integer2=1, equal=(none), lessThan=→"DEACTIVATE", greaterThan=→"DEACTIVATE", everyFrame=false)
    PlayerDataBoolTest(boolName="hasNeedleThrow", isTrue=→"DEACTIVATE", isFalse=(none))
    PlayerDataBoolTest(boolName="metSherma", isTrue=(none), isFalse=→"MEET")  // disabled
    PlayerDataBoolTest(boolName="shermaConvoBellBeast", isTrue=→"REPEAT", isFalse=→"GREET")
    on MEET → Meet Idle
    on GREET → Greet Idle
    on REPEAT → Repeat Idle
    on DEACTIVATE → Deactivate
  }

  state Meet 1 {
    RunDialogue(Sheet="Wanderers", Key="SHERMA_BELLBEAST_MEET", OverrideContinue=true, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    SetPlayerDataBool(boolName="metSherma", value=true)
    SetPlayerDataBool(boolName="shermaConvoBellBeast", value=true)
    on LINE_END → Meet 2
  }

  state Meet 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="GO IDLE", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ExcitedLook")
    on CONVO_END → Meet 3
  }

  state Meet 3 {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ExcitedReturn")  // disabled
    RunDialogue(Sheet="Wanderers", Key="SHERMA_MEET_OPENED2", OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="SHERMA")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EXCITED", delay=0, everyFrame=false)
    on CONVO_END → Interact End
  }

  state Meet Idle {
    on INTERACT → Meet 1
  }

  state Interact End {
    EndDialogue(ReturnControl=true, ReturnHUD=true, Target=Self, UseChildren=false)
    HideNPCTitle(AreaTitleObject=var "AreaTitle")
    on FINISHED → Repeat Idle
  }

  state Repeat Idle {
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED", delay=0, everyFrame=false)  // disabled
    SetRandomAudioClipFromTable(Table=sherma_sing (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=1)
    on INTERACT → Repeat
    on STARTLE → Excite Startle 2
  }

  state Repeat {
    RunDialogueV4(CustomText="", Sheet="Wanderers", Key="SHERMA_BELLBEAST_REPEAT", TextColor=(unset), NpcVoiceTableOverride=sherma_talk_excited (sfxdynamic_assets_sherma.bundle), OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    DisplayNPCTitle(areaTitleObject=var "AreaTitle", displayRight=false, npcTitle="SHERMA")
    on CONVO_END → Interact End
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Greet Idle {
    ActivateInteractible(Target=Self, Activate=true, AllowQueueing=false, UseChildren=false)
    SetRandomAudioClipFromTable(Table=sherma_sing (sfxdynamic_assets_sherma.bundle), AudioSource=var "Audio Loop Voice", AutoPlay=true, delay=1)
    on INTERACT → Greet 1
    on STARTLE → Excite Startle
  }

  state Greet 1 {
    SetPlayerDataBool(boolName="shermaConvoBellBeast", value=true)
    RunDialogueV4(CustomText="", Sheet="Wanderers", Key="SHERMA_BELLBEAST", TextColor=(unset), NpcVoiceTableOverride=sherma_talk_sing (sfxdynamic_assets_sherma.bundle), OverrideContinue=false, PlayerVoiceTableOverride=<null>, PreventHeroAnimation=false, HideDecorators=false, TextAlignment=TopLeft, OffsetY=0, Target=Self)
    on LINE_END → Greet 2
  }

  state Greet 2 {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EXCITED LOOK", delay=0, everyFrame=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ExcitedLook")
    on CONVO_END → Return Anim
  }

  state Return Anim {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="ExcitedReturn")  // disabled
    SendEventByName(eventTarget=GameObject(Self), sendEvent="EXCITED", delay=0, everyFrame=false)
    on FINISHED → Interact End
  }

  state Excite Startle {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=sherma_alert (sfxdynamic_assets_sherma.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED STARTLE", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Greet Idle
  }

  state Excite Startle 2 {
    AudioPlayRandomVoiceFromTable(gameObject=var "Audio Loop Voice", audioClipTable=sherma_alert (sfxdynamic_assets_sherma.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    SendEventByName(eventTarget=GameObjectFSM(Self, fsm="sherma_anim"), sendEvent="EXCITED STARTLE", delay=0, everyFrame=false)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Repeat Idle
  }

  var Sherma Pos: int = 0
  var Hero Behind: bool = false
  var Anim: string = ""
  var Audio Loop Voice: gameObject = Sherma Bell Beast NPC/Audio Loop Voice
}
