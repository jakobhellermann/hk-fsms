# Bench Appear
# 1 location(s):
#   Bellshrine_05 / Whole Scene/Bellshrine Sequence/Bench Appear Sequence

fsm Bench Appear {
  start Wait for Hero

  state State 1 {
    ActivateGameObject(gameObject=var "Real Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BENCH APPEAR → Rune Flash
    on BENCH APPEAR QUICK → Bench Appear Quick 2
  }

  state Bench Appear Quick {
    ActivateGameObject(gameObject=var "Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rune", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Real Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventToRegister(eventName="BELLBENCH APPEARED")
    on FINISHED → Music pause
  }

  state Rune Flash {
    ActivateGameObject(gameObject=Whole Scene/Bellshrine Sequence/Bench Appear Sequence/Flash Effect, activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rune", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    CallMethodProper(gameObject=var "Rune", behaviour="NestedFadeGroupRamp", methodName="StartRamp", parameters=[], storeResult=(unset var), EveryFrame=false)  // disabled
    CallMethodProper(gameObject=var "Rune", behaviour="NestedFadeGroupRamp", methodName="get_Duration", parameters=[], storeResult=var "Rune Flash Duration", EveryFrame=false)  // disabled
    Wait(time=var "Rune Flash Duration", finishEvent=→"FINISHED", realTime=false)  // disabled
    on FINISHED → Wait 2
  }

  state Bench Appear Anim {
    ActivateGameObject(gameObject=var "Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=bellshrine_bench_appear (sfxstatic_assets_areabellshrine.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor@AudioSource (globalpoolprefabs_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AnimatorPlayStateWait(gameObject=var "Bench", layerIndex=(0B), name="Appear", nameHash=(0B), fullPathHash=(0B), shortPathHash=(0B), tagHash=(0B), isStateLooping=(0B), length=(0B), normalizedTime=(0B), loopCount=(0B), currentLoopProgress=(0B), everyFrame=false, everyFrameOption=(0B), time=(0B), finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Wait
  }

  state Wait for Hero {
    ActivateGameObject(gameObject=var "Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rune", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    WaitForHeroInPosition(sendEvent=→"FINISHED", skipIfAlreadyPositioned=true)
    on FINISHED → State 1
    on BENCH APPEAR QUICK → Bench Appear Quick 2
  }

  state Fade In Real Bench {
    ActivateGameObject(gameObject=var "Real Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Real Bench", behaviour="NestedFadeGroupRamp", methodName="StartRamp", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Real Bench", behaviour="NestedFadeGroupRamp", methodName="get_Duration", parameters=[], storeResult=var "Bench Fade Duration", EveryFrame=false)
    Wait(time=var "Bench Fade Duration", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bench Appear Quick
  }

  state Wait {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade In Real Bench
  }

  state Wait 2 {
    Wait(time=4, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Bench Appear Anim
  }

  state Music & Title {
    ShowQuestUpdatedStandalone(Quest=Grand Gate Bellshrines (dataassets_assets_assets/dataassets/questsystem/mainquests.bundle))
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="AwardQueuedAchievements", parameters=[], storeResult=(unset var), EveryFrame=false)
    BoolTest(boolVariable=var "Dont Affect Music", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetPlayerDataInt(intName="enclaveLevel", storeValue=var "Enclave Level")
    IntCompareToBool(integer1=var "Enclave Level", integer2=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Enclave Established", everyFrame=false)
    Comment(comment="If this is first shrine and enclave is established, don't play rest theme")
    BoolAllTrue(boolVariables=[var "Enclave Established", var "Show Title"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    SendEventToRegister(eventName="REST AREA MUSIC START")
    on FINISHED → Title
    on CANCEL → Title
  }

  state Music pause {
    Wait(time=0.5, finishEvent=→"NEXT", realTime=false)
    on NEXT → Music & Title
    on LEAVING SCENE → End
  }

  state Bench Appear Quick 2 {
    ActivateGameObject(gameObject=var "Bench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Rune", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Real Bench", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Music pause 2
  }

  state Music pause 2 {
    Wait(time=2, finishEvent=→"NEXT", realTime=false)
    on NEXT → Music & Title 2
    on LEAVING SCENE → End
  }

  state Music & Title 2 {
    BoolTest(boolVariable=var "Dont Affect Music", isTrue=→"CANCEL", isFalse=(none), everyFrame=false)
    GetPlayerDataInt(intName="enclaveLevel", storeValue=var "Enclave Level")
    IntCompareToBool(integer1=var "Enclave Level", integer2=0, equalBool=(unset), lessThanBool=(unset), greaterThanBool=var "Enclave Established", everyFrame=false)
    Comment(comment="If this is first shrine and enclave is established, don't play rest theme")
    BoolAllTrue(boolVariables=[var "Enclave Established", var "Show Title"], sendEvent=→"CANCEL", storeResult=(unset), everyFrame=false)
    SendEventToRegister(eventName="REST AREA MUSIC START")
    on FINISHED → Title
    on CANCEL → Title
  }

  state End {
  }

  state Title {
    BoolTest(boolVariable=var "Show Title", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=true, bossTitle=var "Shrine Title")
    on FINISHED → End
  }

  var Dont Affect Music: bool = false
  var Shrine Title: string = "BSHRINE_WILD"
  var Real Bench: gameObject = Whole Scene/Bell Bench

  var Bench Fade Duration: float = 0
  var Rune Flash Duration: float = 0
  var Enclave Level: int = 0
  var Show Title: bool = false
  var Enclave Established: bool = false
  var Audio Player Music: gameObject = <null>
  var Bench: gameObject = Whole Scene/Bellshrine Sequence/Bench Appear Sequence/bell_bench_appear
  var Rune: gameObject = Whole Scene/Bellshrine Sequence/Bench Appear Sequence/shrine_rune_giant_flash
}
