# Control
# 1 location(s):
#   Ant_20 / Bone Flyer Rage Control

fsm Control {
  start Pause

  state Init {
    PlayerDataBoolTest(boolName="defeatedBoneFlyerGiant", isTrue=→"INERT", isFalse=(none))
    RandomFloat(min=1, max=1.5, storeResult=var "Smash Timer")
    FindNamedChild(gameObject=Self, storeResult=var "Pt JumpDust")
    on FINISHED → Away
    on INERT → Inert
  }

  state Away {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Far Range", storeResult=false, sendEvent="FAR", outOfRangeEvent="", everyFrame=true)
    on FAR → Far
  }

  state Far {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Far Range", storeResult=false, sendEvent="", outOfRangeEvent="AWAY", everyFrame=true)
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Near Range", storeResult=false, sendEvent="NEAR", outOfRangeEvent="", everyFrame=true)
    FloatAdd(floatVariable=var "Smash Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Smash Timer", float2=0, tolerance=0, equal=(none), lessThan=→"SMASH", greaterThan=(none), everyFrame=true)
    on AWAY → Away
    on NEAR → Near
    on SMASH → Smash Far
  }

  state Near {
    CheckAlertRangeByName(eventTarget=Self, alertRangeName="Near Range", storeResult=false, sendEvent="", outOfRangeEvent="FAR", everyFrame=true)
    FloatAdd(floatVariable=var "Smash Timer", add=-1, everyFrame=true, perSecond=true)
    FloatCompare(float1=var "Smash Timer", float2=0, tolerance=0, equal=(none), lessThan=→"SMASH", greaterThan=(none), everyFrame=true)
    on FAR → Far
    on SMASH → Smash Near
  }

  state Smash Far {
    PlayAudioEventRandom(audioClips=[beastfly_far_away_wall_hit_1 (sfxstatic_assets_areawilds.bundle), beastfly_far_away_wall_hit_2 (sfxstatic_assets_areawilds.bundle)], pitchMin=0.85, pitchMax=1.15, volume=0.35, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    RandomFloat(min=1, max=2, storeResult=var "Smash Timer")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    PlayParticleEmitter(gameObject=var "Pt JumpDust", emit=0, resetIfPlaying=false)
    on FINISHED → Far
  }

  state Smash Near {
    RandomFloat(min=1, max=2, storeResult=var "Smash Timer")
    PlayAudioEventRandom(audioClips=[beastfly_far_away_wall_hit_1 (sfxstatic_assets_areawilds.bundle), beastfly_far_away_wall_hit_2 (sfxstatic_assets_areawilds.bundle)], pitchMin=0.85, pitchMax=1.15, volume=0.35, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SendMessage(gameObject=Self, delivery=BroadcastMessage, options=1, functionCall=StartTimedJitter())
    PlayParticleEmitter(gameObject=var "Pt JumpDust", emit=0, resetIfPlaying=false)
    on FINISHED → Near
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Inert {
  }

  var Smash Timer: float = 0
  var Pt JumpDust: gameObject = <null>
}
