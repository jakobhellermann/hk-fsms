# Control
# 1 location(s):
#   Shellwood_11b_Memory / Boss Scene/Spike Balls/Flower Queen Spike (4)

fsm Control {
  start Init
  on TINK LEFT → Tink Left  // from any state
  on TINK DOWN → Tink Down  // from any state
  on TINK UP → Tink Up  // from any state
  on TINK RIGHT → Tink Right  // from any state
  on BREAK → Break  // from any state
  on DEALT DAMAGE → Break  // from any state

  state Init {
    GetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Ground Y", z=(unset), space=0, everyFrame=false)
    on FINISHED → Dormant
  }

  state Dormant {
    SetPosition(gameObject=Self, vector=(unset), x=var "Start X", y=var "Ground Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=1, y=1, z=(unset), everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=0, space=0, everyFrame=false, lateUpdate=false)
    on ATTACK HIGH → Pos High
    on ATTACK MID → Pos Mid
    on ATTACK LOW → Pos Low
    on ATTACK TOP → Pos Top
  }

  state Position And Wait {
    CallMethodProper(gameObject=Self, behaviour="DamageEnemies", methodName="EndDamage", parameters=[], storeResult=(unset var), EveryFrame=false)
    WaitRandom(timeMin=0, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    TranslateRandom(gameObject=Self, translateMin=(-1, 0, 0), translateMax=(1, 0, 0), space=0)
    on FINISHED → Antic
  }

  state Antic {
    PlayParticleEmitterChildren(gameObject=var "Pt Emerge", resetTimeIfPlaying=false, stopOnStateExit=true)
    Wait(time=0.65, finishEvent=→"FINISHED", realTime=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=vine_ball_antic_for_emerge (sfxstatic_assets_areashellwood.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    on FINISHED → Fire
  }

  state Fire {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=swamp_muckman_shoot (sfxstatic_assets_areaswamp.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, delay=0, storePlayer=<null>)
    SetDamageEnemyAmount(target=Self, damageDealt=0)
    ResetDamageEnemiesLists(Target=Self)
    PlayAudioEventRandom(audioClips=[bloom_shooter_shoot_1 (sfxstatic_assets_areashellwood.bundle), bloom_shooter_shoot_2 (sfxstatic_assets_areashellwood.bundle), bloom_shooter_shoot_3 (sfxstatic_assets_areashellwood.bundle)], pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    RandomFloat(min=-15, max=15, storeResult=var "Random")
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Random", space=0, everyFrame=false, lateUpdate=false)
    RandomFloat(min=1, max=1.1, storeResult=var "Random")
    SetScale(gameObject=Self, vector=(unset), x=var "Random", y=var "Random", z=(unset), everyFrame=false, lateUpdate=false)
    RandomlyFlipScale(gameObject=Self)
    ActivateGameObject(gameObject=var "Emerge Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetDamageHeroAmount(target=Self, damageDealt=1)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=25, everyFrame=false)
    CheckYPosition(gameObject=Self, compareTo=var "Y Target", compareToOffset=-1, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"FINISHED", everyFrame=true, space=0, activeBool=(unset))
    SetMeshRenderer(gameObject=Self, active=true)
    SetCircleCollider(gameObject=Self, active=true)
    PlayParticleEmitterInState(gameObject=var "Pt Trail")
    RandomFloat(min=-25, max=25, storeResult=var "Rotation")
    Rotate(gameObject=Self, vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    on FINISHED → Decel
    on BREAK → Break
  }

  state Decel {
    DecelerateV2(gameObject=Self, deceleration=0.875, brakeOnExit=false)
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Y Velocity", space=0, everyFrame=true)
    FloatCompare(float1=var "Y Velocity", float2=1, tolerance=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=(none), everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer=(unset), sendEvent=→"BREAK", storeCollider=var "")
    on FINISHED → Drop
    on BREAK → Break
  }

  state Drop {
    AccelerateToY(gameObject=Self, accelerationFactor=1, targetSpeed=-30)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer=(unset), sendEvent=→"BREAK", storeCollider=var "")
    CheckYPosition(gameObject=Self, compareTo=var "Ground Y", compareToOffset=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on FINISHED → Dormant
    on BREAK → Break
  }

  state Pos High {
    RandomFloat(min=96, max=97.5, storeResult=var "Y Target")
    on FINISHED → Position And Wait
  }

  state Pos Mid {
    RandomFloat(min=93.5, max=95.5, storeResult=var "Y Target")
    on FINISHED → Position And Wait
  }

  state Pos Low {
    RandomFloat(min=90, max=92, storeResult=var "Y Target")
    on FINISHED → Position And Wait
  }

  state Tink Down {
    SetVelocity2d(gameObject=Self, vector=(0, -30), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Tinked Away
  }

  state Break {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    SetCircleCollider(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Radial Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayParticleEmitterChildren(gameObject=var "Pt Break", resetTimeIfPlaying=false, stopOnStateExit=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=moss_knight_projectile_break (sfxstatic_assets_areashellwood.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=spine_floater_explode (sfxstatic_assets_areawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=enemy_damage (herostatic_assets_all.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Dormant
  }

  state Tink Left {
    SetVelocity2d(gameObject=Self, vector=(-40, 4), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Tinked Away
  }

  state Tinked Away {
    SetDamageHeroAmount(target=Self, damageDealt=0)
    SetDamageEnemyAmount(target=Self, damageDealt=15)
    AddForce2d(gameObject=Self, forceMode=0, atPosition=(unset), vector=(unset), x=(unset), y=-0.5, vector3=(unset), everyFrame=true)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Boss", collideLayer=(unset), sendEvent=→"BREAK", storeCollider=var "")  // disabled
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"BREAK", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Recoiler", collideLayer=Attack(17), sendEvent=→"BUDDY COLLIDE", storeCollider=var "Collider")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on BREAK → Break Frame
    on FINISHED → Break
    on BUDDY COLLIDE → Buddy Collide
  }

  state Tink Right {
    SetVelocity2d(gameObject=Self, vector=(40, 4), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Tinked Away
  }

  state Tink Up {
    SetVelocity2d(gameObject=Self, vector=(0, 30), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Tinked Away
  }

  state Break Frame {
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    on FINISHED → Break
  }

  state Buddy Collide {
    SendEventByName(eventTarget=GameObject(var "Collider"), sendEvent="BREAK", delay=0, everyFrame=false)
    on FINISHED → Break
  }

  state Pos Top {
    RandomFloat(min=99, max=101, storeResult=var "Y Target")
    on FINISHED → Position And Wait
  }

  var Ground Y: float = 85
  var Random: float = 0
  var Start X: float = 0
  var Y Target: float = 0
  var Y Velocity: float = 0
  var Rotation: float = 0
  var Emerge Effect: gameObject = Boss Scene/Spike Balls/Flower Queen Spike (4)/Emerge Effect
  var Pt Break: gameObject = Boss Scene/Spike Balls/Flower Queen Spike (4)/Pt Break
  var Pt Emerge: gameObject = Boss Scene/Spike Balls/Flower Queen Spike (4)/Pt Emerge
  var Pt Trail: gameObject = Boss Scene/Spike Balls/Flower Queen Spike (4)/Pt Trail
  var Radial Burst: gameObject = Boss Scene/Spike Balls/Flower Queen Spike (4)/Radial Burst Effect
  var Self: gameObject = Boss Scene/Spike Balls/Flower Queen Spike (4)
  var Collider: gameObject = <null>
}
