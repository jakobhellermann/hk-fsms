fsm Fall Tele {
  start Start Pause
  on STOP FALL CHECK → Stop  // from any state

  state Check {
    GetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=var "Velocity Y", space=0, everyFrame=true)
    FloatCompare(float1=var "Velocity Y", float2=-0.5, tolerance=0, equal=(none), lessThan=→"FALL TELE", greaterThan=(none), everyFrame=true)
    on FALL TELE → Send Event
  }

  state Send Event {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="FALL TELE", delay=0, everyFrame=false)
    on FINISHED → Stop
  }

  state Start Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Check
  }

  state Stop {
    on START FALL CHECK → Check
  }

  var Velocity Y: float = 0
}
