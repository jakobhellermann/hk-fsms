# Control
# 1 location(s):
#   localpoolprefabs_assets_arealibrary.bundle / Delayed Pop Explosion Trobbio

fsm Control {
  start Pause

  state Pause {
    ActivateGameObject(gameObject=var "Explosion", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    WaitRandom(timeMin=1, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    PlayParticleEmitterInState(gameObject=var "Pt Antic")
    on FINISHED → Explode
  }

  state Explode {
    ActivateGameObject(gameObject=var "Explosion", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=rock_roller_trail_explode (sfxstatic_assets_areabone.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Recycle
  }

  state Recycle {
    ActivateGameObject(gameObject=var "Explosion", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    RecycleSelf()
  }

  var Explosion: gameObject = Delayed Pop Explosion Trobbio/Explosion
  var Pt Antic: gameObject = Delayed Pop Explosion Trobbio/Pt Antic
}
