# Control
# 1 location(s):
#   Dust_06 / Roachkeeper Key Control

fsm Control {
  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Collectable Item SimpleKey")
    FindNamedChild(gameObject=Self, storeResult=var "Roachkeeper")
    on FINISHED → Idle
  }

  state Idle {
    on KEYKEEPER DEFEATED → Spawn Key
  }

  state Spawn Key {
    SetPositionToObject2D(gameObject=var "Collectable Item SimpleKey", targetObject=var "Roachkeeper", xOffset=0, yOffset=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Collectable Item SimpleKey", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Collectable Item SimpleKey: gameObject = <null>
  var Roachkeeper: gameObject = <null>
}
