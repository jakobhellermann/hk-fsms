# Battle Start
# 1 location(s):
#   Room_CrowCourt_02 / Battle Scene/Battle Start

fsm Battle Start {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Battle Spotlight")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Battle Darkness")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Battle Gates")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Intro Roar Point")
    on FINISHED → Idle
  }

  state Idle {
    PlayerDataBoolTest(boolName="defeatedCrowCourt", isTrue=→"DEFEATED", isFalse=(none))
    Trigger2dEvent(gameObject=Self, trigger=0, collideTag="", sendEvent=→"ENTER", storeCollider=var "")
    on ENTER → Enter
    on DEFEATED → Deactivate
  }

  state Enter {
    GetPlayerDataBool(boolName="encounteredCrowCourt", storeValue=var "Encountered Crow Court")
    SendEventToRegister(eventName="TENSION END")
    SendEventByName(eventTarget=GameObject(var "Battle Gates"), sendEvent="BG CLOSE", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Encountered Crow Court", time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Start Wipe Audio
  }

  state Lights Up {
    PlayAudioEvent(audioClip=library_lift_large_activate (sfxstatic_assets_arealibrary.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    EaseSpriteColor(target=var "Battle Darkness", fromValue=(unset), toValue=(1, 1, 1, 0), disableSpriteRendererIfTransparent=false, time=0.3, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    ActivateGameObject(gameObject=var "Battle Spotlight", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    EaseFloat(fromValue=3.5, toValue=15, floatVariable=var "Light X Scale", time=0.3, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    SetScale(gameObject=var "Battle Spotlight", vector=(unset), x=var "Light X Scale", y=(unset), z=(unset), everyFrame=true, lateUpdate=false)
    Wait(time=2, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Encountered Crow Court", time=1, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegisterDelay(EventName="CROWD IDLE", delay=0.25, ExcludeTarget=Self)
    on FINISHED → Crowd Roar
  }

  state Crowd Roar {
    SendEventToRegister(eventName="CROWD ROAR")
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Medium Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=true, DoFreeze=false, Delay=0.2)  // disabled
    StartRoarEmitter(spawnPoint=var "Intro Roar Point", delay=0.1, stunHero=true, roarBurst=false, isSmall=false, noVisualEffect=false, forceThroughBind=false, stopOnExit=true)
    DisplayBossTitle(areaTitleObject=var "AreaTitle", displayRight=false, bossTitle="CROW_COURT")
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    WaitBool(boolTest=var "Encountered Crow Court", time=2, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Crowd Idle
  }

  state Battle Start {
    SendMessage(gameObject=var "Battle Scene", delivery=SendMessage, options=1, functionCall=StartBattle())
  }

  state Crowd Idle {
    SendEventToRegister(eventName="CROWD IDLE")
    Wait(time=1.1, finishEvent=→"FINISHED", realTime=false)
    ApplyMusicCue(musicCue=Enemy Battle Grind (audiocuesdynamic_assets_battle.bundle), delayTime=0, transitionTime=0)
    TransitionToAudioSnapshot(snapshot=Main Only (audioobjects_assets_all.bundle), transitionTime=0.1)
    on FINISHED → Battle Start
  }

  state Start Wipe Audio {
    PlayAudioEvent(audioClip=ui_screen_wipe_from_cinematic_2d (sfxstatic_assets_ui.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Lights Up
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=Battle Scene/Crowd, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Light X Scale: float = 0
  var Encountered Crow Court: bool = false
  var Battle Scene: gameObject = <null>
  var Battle Spotlight: gameObject = <null>
  var Battle Darkness: gameObject = <null>
  var Battle Gates: gameObject = <null>
  var Intro Roar Point: gameObject = <null>
}
