# Rest Animation
# 1 location(s):
#   Bone_East_10 / Black Thread States Thread Only Variant/Normal World/Bonetown Resident

fsm Rest Animation {
  start Idle

  state Idle {
    BoolTest(boolVariable=var "Resting", isTrue=→"RESTING", isFalse=(none), everyFrame=true)
    on RESTING → Get Previous State
  }

  state Get Previous State {
    GetProperty(targetProperty=tk2dLookAnimNPC.PreviousState on Black Thread States Thread Only Variant/Normal World/Bonetown Resident@tk2dLookAnimNPC, everyFrame=false)
    EnumSwitch(enumVariable=var "Look Anim State", compareTo=[LookAnimNPC+AnimState(0), LookAnimNPC+AnimState(2)], sendEvent=[→"LEFT", →"RIGHT"], everyFrame=false)
    on LEFT → Rest From Left
    on RIGHT → Rest From Right
    on FINISHED → Invalid Beginning State
  }

  state Invalid Beginning State {
    DebugLogConsole(logLevel=0, text="Started resting in invalid state.", sendToUnityLog=true)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Reset {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="EndRest", parameters=[var "Should Face Left"], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Rest From Left {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Idle to Rest", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Rest From Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Enter Rest Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Rest Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest Default", animationTriggerEvent=(none), animationCompleteEvent=(none))
    BoolTest(boolVariable=var "Resting", isTrue=(none), isFalse=→"CANCEL", everyFrame=true)
    WaitRandom(timeMin=2, timeMax=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rest Variant
    on CANCEL → Wake Up
  }

  state Rest Variant {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Rest Variant", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Rest Idle
  }

  state Wake Up {
    CallMethodProper(gameObject=Self, behaviour="LookAnimNPC", methodName="ShouldFaceLeft", parameters=[], storeResult=var "Should Face Left", EveryFrame=false)
    BoolTest(boolVariable=var "Should Face Left", isTrue=→"LEFT", isFalse=→"RIGHT", everyFrame=false)
    on LEFT → Rest to Idle
    on RIGHT → Rest to Right
  }

  state Rest to Idle {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="RestToIdle", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  state Rest to Right {
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Exit Rest Right", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    on FINISHED → Reset
  }

  var Max Turn Delay: float = 5
  var Min Turn Delay: float = 3

  var Resting: bool = false
  var Should Face Left: bool = false
  var Look Anim State: enum = LookAnimNPC+AnimState(0)
}
