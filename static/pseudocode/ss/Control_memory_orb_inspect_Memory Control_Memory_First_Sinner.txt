# Control
# 1 location(s):
#   Memory_First_Sinner / Memory Control/memory_orb_inspect

fsm Control {
  start Idle

  state Take Control {
    SendEventByName(eventTarget=GameObject(var "HUD Canvas"), sendEvent="OUT", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect Camlock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=var "Tween Pos", x=(unset), y=(unset), z=(unset), space=0, everyFrame=false)
    GetPosition(gameObject=Self, vector=(unset), x=(unset), y=(unset), z=var "Z Pos", space=0, everyFrame=false)
    SetVector3XYZ(vector3Variable=var "Tween Pos", vector3Value=(unset), x=(unset), y=(unset), z=var "Z Pos", everyFrame=false)
    HeroControllerMethods(method=RelinquishControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    HeroControllerMethods(method=StopAnimationControl, parameters=[], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    HeroControllerMethods(method=AffectedByGravity, parameters=[false], everyFrame=false, storeValue=(unused), isTrue=(none), isFalse=(none))
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FSM CANCEL", delay=0, everyFrame=false)
    SetVelocity2d(gameObject=var "Hero", vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    CheckTargetDirection(gameObject=Self, target=var "Hero", aboveEvent=(none), belowEvent=(none), rightEvent=→"R", leftEvent=→"L", aboveBool=(unset), belowBool=(unset), rightBool=(unset), leftBool=(unset), selfOffsetX=0, selfOffsetY=0, reverseIfNegativeScale=false, everyFrame=false)
    on L → Hero Face R
    on R → Hero Face L
  }

  state Suck {
    ActivateGameObject(gameObject=var "Suck Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Next Scene
  }

  state Hero Face R {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    on FINISHED → Kneel
  }

  state Hero Face L {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceLeft())
    on FINISHED → Kneel
  }

  state Idle {
    ActivateGameObject(gameObject=var "Collect Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Suck Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shatter Effect", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Active Child", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Activate On Burst", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Timeline", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on INTERACT → Take Control
  }

  state Next Scene {
    SendEventByNameUpwards(Target=Self, EventName="NEXT SCENE")
  }

  state Break {
    SendEventToRegister(eventName="MEMORY ORB COLLECT")
    PlayAudioEvent(audioClip=silk_heart_get_pt3_2nd_flash (sfxstatic_assets_silkshared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Memory Orb")
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Collect Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shatter Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Active Child", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    GetPosition2D(gameObject=Self, vector=var "Self Pos", x=(unset), y=(unset), space=0, everyFrame=false)
    AnimateRigidBody2DPositionTo(GameObject=var "Hero", ToValue=var "Self Pos", time=0.15, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=(none), realTime=false)
    on FINISHED → Suck
  }

  state Kneel {
    SetPlayerDataBool(boolName="disablePause", value=true)
    PlayRandomAudioClipTable(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=Self, SpawnPosition=(0, 0, 0))
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Collect Normal 1")
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Flash Out
  }

  state Flash Out {
    AudioStopV2(gameObject=Self, fadeTime=0.1, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=silk_heart_get_pt3_2nd_flash (sfxstatic_assets_silkshared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Activate On Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=Self, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ActivateGameObject(gameObject=var "Collect Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shatter Effect", activate=true, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    ActivateGameObject(gameObject=var "Active Child", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Inspect Camlock", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ScreenFader(startColour=(1, 1, 1, 0), endColour=(1, 1, 1, 1), duration=0.15)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    TransitionToAudioSnapshot(snapshot=Silent (audioobjects_assets_all.bundle), transitionTime=0.5)
    TransitionToAudioSnapshot(snapshot=at None (audioobjects_assets_all.bundle), transitionTime=0.5)
    on FINISHED → White Pause
  }

  state Fade To Cutscene {
    SetMeshRenderer(gameObject=var "Hero", active=false)
    ScreenFader(startColour=(0, 0, 0, 1), endColour=(0, 0, 0, 0), duration=2)
    on FINISHED → Cutscene
  }

  state Cutscene {
    GameObjectCompare(gameObjectVariable=var "Timeline", compareTo=<null>, equalEvent=(none), notEqualEvent=→"TIMELINE", storeResult=(unset), everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on TIMELINE → Play Timeline
    on FINISHED → Fade To Black Final
  }

  state Fade To Black Final {
    ScreenFader(startColour=(0, 0, 0, 0), endColour=(0, 0, 0, 1), duration=3)
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → End Memory
  }

  state End Memory {
    SendMessage(gameObject=var "Hero", delivery=SendMessage, options=1, functionCall=FaceRight())
    SendEventToRegister(eventName="MEMORY END")
  }

  state Play Timeline {
    ActivateGameObject(gameObject=var "Timeline", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    HasComponent(gameObject=var "Timeline", component="UnityEngine.Playables.PlayableDirector", removeOnExit=false, trueEvent=(none), falseEvent=→"CANCEL", store=(unset), everyFrame=false)
    PlayTimeline(Target=var "Timeline", FinishEvent=→"FINISHED")
    on FINISHED → Fade To Black Final
    on CANCEL → Play Animator
  }

  state Play Animator {
    AnimatorPlayStateWait(target=var "Timeline", stateName="", finishEvent=→"FINISHED")
    on FINISHED → Fade To Black Final
  }

  state White Pause {
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Fade To Black
  }

  state Fade To Black {
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    ScreenFader(startColour=(1, 1, 1, 1), endColour=(0, 0, 0, 1), duration=1)
    on FINISHED → Fade To Cutscene
  }

  var Inspect Camlock: gameObject = Memory Control/memory_orb_inspect/CameraLockArea Final
  var Collect Camlock: gameObject = Memory Control/memory_orb_inspect/CameraLockArea Final 2
  var Timeline: gameObject = Memory Control/First_Weaver_Memory
  var Activate On Burst: gameObject = Memory Control/Sound

  var Z Pos: float = 0
  var Self Pos: vector2 = (0, 0)
  var Tween Pos: vector3 = (0, 0, 0)
  var Collect Effect: gameObject = Memory Control/memory_orb_inspect/Collect Effect
  var Active Child: gameObject = Memory Control/memory_orb_inspect/Active
  var Suck Effect: gameObject = Memory Control/memory_orb_inspect/Suck Effects
  var Shatter Effect: gameObject = Memory Control/memory_orb_inspect/Shatter Effects
}
