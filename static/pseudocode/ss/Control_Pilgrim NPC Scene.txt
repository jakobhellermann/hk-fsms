# Control
# 1 location(s):
#   Bone_East_10 / Black Thread States Thread Only Variant/Normal World/Pilgrim NPC Scene

fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    FindNamedChild(gameObject=Self, storeResult=var "1 First Visit")
    FindNamedChild(gameObject=Self, storeResult=var "2 Brolly Revisit")
    FindNamedChild(gameObject=Self, storeResult=var "3 Final")
    GetPlayerDataBool(boolName="rhinoChurchUnlocked", storeValue=var "Rhino Church Unlocked")
    GetPlayerDataBool(boolName="churchRhinoKilled", storeValue=var "Church Rhino Killed")
    GetPlayerDataBool(boolName="rhinoRampageCompleted", storeValue=var "Rhino Rampage Completed")
    BoolTest(boolVariable=var "Rhino Church Unlocked", isTrue=→"LEFT", isFalse=(none), everyFrame=false)
    PlayerDataBoolTest(boolName="pilbyAtPilgrimsRest", isTrue=→"LEFT", isFalse=(none))
    BoolTestMulti(boolVariables=[var "Church Rhino Killed", var "Rhino Church Unlocked", var "Rhino Rampage Completed"], boolStates=[false, true, false], trueEvent=→"RAMPAGE", falseEvent=(none), storeResult=(unset), everyFrame=false)  // disabled
    PlayerDataBoolTest(boolName="hasBrolly", isTrue=(none), isFalse=→"FIRST")
    PlayerDataBoolTest(boolName="visitedGreymoor", isTrue=→"FINAL", isFalse=(none))
    PlayerDataBoolTest(boolName="visitedCoral", isTrue=→"FINAL", isFalse=(none))
    PlayerDataBoolTest(boolName="spinnerDefeated", isTrue=→"FINAL", isFalse=(none))
    PlayerDataBoolTest(boolName="visitedShellwood", isTrue=→"FINAL", isFalse=→"REVISIT")
    on FIRST → First
    on REVISIT → Revisit
    on FINAL → Final
    on LEFT → Hide From Rhino
  }

  state First {
    ActivateGameObject(gameObject=var "1 First Visit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "2 Brolly Revisit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "3 Final", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Revisit {
    ActivateGameObject(gameObject=var "1 First Visit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "2 Brolly Revisit", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "3 Final", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Final {
    ActivateGameObject(gameObject=var "1 First Visit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "2 Brolly Revisit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "3 Final", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Hide From Rhino {
    ActivateGameObject(gameObject=var "1 First Visit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "2 Brolly Revisit", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "3 Final", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Church Rhino Killed: bool = false
  var Rhino Church Unlocked: bool = false
  var Rhino Rampage Completed: bool = false
  var 1 First Visit: gameObject = <null>
  var 2 Brolly Revisit: gameObject = <null>
  var 3 Final: gameObject = <null>
}
