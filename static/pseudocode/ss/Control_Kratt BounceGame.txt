# Control
# 1 location(s):
#   Aqueduct_05_festival / Caravan_States/Flea Festival/Flea Game - Bouncing/Kratt BounceGame

fsm Control {
  start Init
  on RESET FLEA GAMES → Dormant  // from any state

  state Init {
    on FINISHED → Dormant
  }

  state Dormant {
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Tink Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    on ENTER → Enter Pause
  }

  state Choose Entry Side {
    SendRandomEventV4(events=[→"L", →"R"], weights=[0.5, 0.5], eventMax=[1, 1], missedMax=[1, 1], activeBool=(unset))
    on L → Enter L
    on R → Enter R
  }

  state Enter L {
    RandomFloat(min=223, max=226, storeResult=var "Spawn X")
    SetScale(gameObject=Self, vector=(unset), x=-1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Enter
  }

  state Enter R {
    RandomFloat(min=240, max=243, storeResult=var "Spawn X")
    SetScale(gameObject=Self, vector=(unset), x=1, y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
    on FINISHED → Enter
  }

  state Enter {
    SetPosition2D(GameObject=Self, Vector=(unset), X=var "Spawn X", Y=var "Water Y", Space=0, EveryFrame=false)
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=kratt_alert (sfxdynamic_assets_fleacaravan.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Kratt Jump")
    ActivateGameObject(gameObject=var "Tink Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=hornet_water_exit (sfxstatic_assets_water.bundle), pitchMin=0.85, pitchMax=0.85, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=Splash In Black (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0.25, 0), rotation=(unset), storeObject=var "Splash")
    Tk2dSpriteSetColor(gameObject=var "Splash", color=(1, 1, 1, 1), everyframe=false)
    SetScale(gameObject=var "Splash", vector=(unset), x=1.75, y=1.75, z=(unset), everyFrame=false, lateUpdate=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    FlingObject(flungObject=Self, speedMin=12, speedMax=13, angleMin=90, angleMax=90)
    NextFrameEvent(sendEvent=→"FINISHED")
    Translate(gameObject=Self, vector=(unset), x=(unset), y=0.001, z=(unset), space=1, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SetMeshRenderer(gameObject=Self, active=true)
    on FINISHED → Jumping
  }

  state Jumping {
    CheckYPosition(gameObject=Self, compareTo=var "Water Y", compareToOffset=0, tolerance=0, equal=→"SPLASH", lessThan=→"SPLASH", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    FadeNestedFadeGroup(Target=var "Hero Light", ToAlpha=1, FadeTime=0.2)
    on SPLASH → Splash In
    on TINK DOWN → Tink Down
    on HARPOON HOOKED → Harpoon Hooked
  }

  state Splash In {
    SetPosition2D(GameObject=Self, Vector=(unset), X=(unset), Y=var "Water Y", Space=0, EveryFrame=false)
    ActivateGameObject(gameObject=var "Tink Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=hornet_water_enter (sfxstatic_assets_water.bundle), pitchMin=0.85, pitchMax=0.85, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=Splash In Black (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", position=(0, 0.25, 0), rotation=(unset), storeObject=var "Splash")
    Tk2dSpriteSetColor(gameObject=var "Splash", color=(1, 1, 1, 1), everyframe=false)
    SetScale(gameObject=var "Splash", vector=(unset), x=1.75, y=1.75, z=(unset), everyFrame=false, lateUpdate=false)
    FadeNestedFadeGroup(Target=var "Hero Light", ToAlpha=0, FadeTime=0.2)
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Dormant
  }

  state Struck Down {
    AudioPlayRandomVoiceFromTableV2(gameObject=var "Audio Loop Voice", audioClipTable=kratt_damaged (sfxdynamic_assets_fleacaravan.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    PlayAudioEvent(audioClip=sword_hit_reject (sfxstatic_assets_shared.bundle), pitchMin=0.65, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Kratt Head Bounce")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    ActivateGameObject(gameObject=var "Tink Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)  // disabled
    CheckYPosition(gameObject=Self, compareTo=var "Water Y", compareToOffset=0, tolerance=0, equal=→"SPLASH", lessThan=→"SPLASH", greaterThan=(none), everyFrame=true, space=0, activeBool=(unset))
    on SPLASH → Splash In
    on TINK DOWN → Struck Down
  }

  state Enter Pause {
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Choose Entry Side
  }

  state Score {
    SendEventToRegister(eventName="SCORE")
    SpawnObjectFromGlobalPool(gameObject=Flea Game Score Effect (localpoolprefabs_assets_fleacaravan.bundle), spawnPoint=var "Self", position=(unset), rotation=(unset), storeObject=var "")
    on FINISHED → Struck Down
  }

  state Tink Down {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=-20, everyFrame=false)
    on FINISHED → Score
  }

  state Harpoon Hooked {
    SetVelocity2d(gameObject=Self, vector=(unset), x=(unset), y=3, everyFrame=false)
    on FINISHED → Score
  }

  var Water Y: float = 5
  var Spawn X: float = 0
  var Sprite: gameObject = Bellflea Bouncer/Sprite (localpoolprefabs_assets_fleacaravan.bundle)
  var Tink Collider: gameObject = Caravan_States/Flea Festival/Flea Game - Bouncing/Kratt BounceGame/Tink Collider
  var Splash: gameObject = <null>
  var Self: gameObject = Caravan_States/Flea Festival/Flea Game - Bouncing/Kratt BounceGame
  var Hero Light: gameObject = Caravan_States/Flea Festival/Flea Game - Bouncing/Kratt BounceGame/Hero Light
  var Audio Loop Voice: gameObject = Caravan_States/Flea Festival/Flea Game - Bouncing/Kratt BounceGame/Audio Loop Voice
}
