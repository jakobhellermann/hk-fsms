# Hero Start Line
# 1 location(s):
#   Sprintmaster_Cave / Race Group/Sprintmaster Runner

fsm Hero Start Line {
  start Init
  on END → Init  // from any state

  state Init {
    AudioStopV2(gameObject=var "Race Antic Audio", fadeTime=0, cancelOnEarlyExit=true)
    on BEGIN → Start Wait
  }

  state Start Wait {
    SetBoolValue(boolVariable=var "Holding Dash", boolValue=false, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Sprintmaster Start")
    ListenForDash(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=→"DASH", isNotPressed=(none), delayBeforeActive=0.1)
    on DASH → Dash Hold Early
  }

  state Dash Hold Early {
    SetBoolValue(boolVariable=var "Holding Dash", boolValue=true, everyFrame=false)
    Tk2dPlayAnimation(gameObject=var "Hero", animLibName="", clipName="Sprintmaster Low")
    ListenForDash(eventTarget=Self, wasPressed=(none), wasReleased=(none), isPressed=(none), isNotPressed=→"FINISHED", delayBeforeActive=0.1)
    PlayRandomAudioClipTableV3(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    PlayRandomAudioClipTableV3(Table=Hornet_race_antic (sfxdynamic_assets_sprintmaster.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0), ForcePlay=true, StoreSpawned=var "Race Antic Audio")
    on FINISHED → End
  }

  state End {
    PlayRandomAudioClipTableV3(Table=small_movement_cloak (sfxdynamic_assets_shared.bundle), AudioPlayerPrefab=var "", SpawnPoint=var "Hero", SpawnPosition=(0, 0, 0), ForcePlay=false, StoreSpawned=var "")
    StartFadeOutAudioSource(Target=var "Race Antic Audio", Duration=0.1, RecycleOnEnd=false)
    on FINISHED → Start Wait
  }

  var Holding Dash: bool = false
  var Race Antic Audio: gameObject = <null>
}
