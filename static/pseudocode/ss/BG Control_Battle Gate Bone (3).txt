# BG Control
# 1 location(s):
#   Room_CrowCourt_02 / Battle Scene/Battle Gates/Battle Gate Bone (3)

fsm BG Control {
  start Init
  on BG DESTROY → Destroy  // from any state
  on BG QUICK OPEN → Quick Open  // from any state

  state Init {
    GetOwner(storeGameObject=var "Self")
    BoolTest(boolVariable=var "Start Closed", isTrue=→"BG QUICK CLOSE", isFalse=(none), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Close 1 {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=gate_slam (sharedeffects_assets_all.bundle), pitchMin=0.8, pitchMax=1.2, volume=0.5, delay=0, storePlayer=<null>)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bone Gate Close", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Close 2
  }

  state Close 2 {
    Tk2dPlayFrame(gameObject=var "Slam Effect", frame=0)  // disabled
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Closed")
    SendEventByName(eventTarget=GameObjectFSM(var "CameraParent", fsm="CameraShake"), sendEvent="EnemyKillShake", delay=0, everyFrame=false)
    PlayParticleEmitter(gameObject=var "Dust", emit=0, resetIfPlaying=false)
    on BG OPEN → Open
    on OPEN GATES → Open
  }

  state Open {
    AudioPlayerOneShotSingle(audioPlayer=Audio Player Actor (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Self", audioClip=gate_open (sharedeffects_assets_all.bundle), pitchMin=0.85, pitchMax=1.15, volume=0.85, delay=0, storePlayer=<null>)
    PlayParticleEmitter(gameObject=var "Raise Dust", emit=0, resetIfPlaying=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Bone Gate Open", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
    on FINISHED → Opened
  }

  state Quick Close {
    SetCollider(gameObject=Self, active=true, resetOnExit=false)
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Closed")
    Wait(time=0.2, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=true)
    ActivateGameObject(gameObject=var "Wall Collider", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    on BG OPEN → Open
    on FINISHED → Double Close
  }

  state Double Close {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Closed")
    on BG OPEN → Open
  }

  state Destroy {
    DestroySelf(detachChildren=false)
  }

  state Quick Open {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="Bone Gate Opened")
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Wall Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  state Opened {
    SetMeshRenderer(gameObject=Self, active=false)
    on BG CLOSE → Close 1
    on BG QUICK CLOSE → Quick Close
  }

  var Start Closed: bool = false

  var Camera: gameObject = <null>
  var Dust: gameObject = Battle Scene/Battle Gates/Battle Gate Bone (3)/Dust
  var Raise Dust: gameObject = Battle Scene/Battle Gates/Battle Gate Bone (3)/Raise Dust
  var Self: gameObject = <null>
  var Slam Effect: gameObject = Battle Scene/Battle Gates/Battle Gate Bone (3)/Close Effect
  var Wall Collider: gameObject = Battle Scene/Battle Gates/Battle Gate Bone (3)/Wall Collider
}
