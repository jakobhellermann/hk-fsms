# Control
# 1 location(s):
#   Cog_05 / Battle Scene/Wave 1/Setup Wave 1

fsm Control {
  start Init

  state Init {
    GetParent(gameObject=Self, storeResult=var "Wave")
    GetGrandParent(gameObject=Self, storeResult=var "Battle Scene")
    FindNamedChild(gameObject=var "Battle Scene", storeResult=var "Plats")
    FindNamedChild(gameObject=var "Plats", storeResult=var "Plat Mid")
    FindNamedChild(gameObject=var "Plats", storeResult=var "Plat Mid L")
    FindNamedChild(gameObject=var "Plats", storeResult=var "Plat Mid R")
    FindNamedChild(gameObject=var "Plats", storeResult=var "Plat Far L")
    FindNamedChild(gameObject=var "Plats", storeResult=var "Plat Far R")
    FindNamedChild(gameObject=var "Plats", storeResult=var "Plat Twin L")
    FindNamedChild(gameObject=var "Plats", storeResult=var "Plat Twin R")
    on FINISHED → Idle
  }

  state Idle {
    on BATTLE START → State 3
  }

  state State 1 {
    SendEventByName(eventTarget=GameObject(var "Plat Mid L"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Plat Mid R"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → State 2
  }

  state State 2 {
    SendEventByName(eventTarget=GameObject(var "Wave"), sendEvent="SPAWN", delay=0, everyFrame=false)
    DestroySelf(detachChildren=false)
  }

  state State 3 {
    ActivateGameObject(gameObject=Battle Scene/Respawn Init, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Plat Far R"), sendEvent="DOWN", delay=0, everyFrame=false)
    Wait(time=1.5, finishEvent=→"FINISHED", realTime=false)
    SetHazardRespawn(hazardRespawnMarker=Battle Scene/Respawn Mid)
    on FINISHED → State 1
  }

  var Battle Scene: gameObject = <null>
  var Plats: gameObject = <null>
  var Plat Mid: gameObject = <null>
  var Plat Mid L: gameObject = <null>
  var Plat Mid R: gameObject = <null>
  var Plat Far L: gameObject = <null>
  var Plat Far R: gameObject = <null>
  var Plat Twin L: gameObject = <null>
  var Plat Twin R: gameObject = <null>
  var Wave: gameObject = <null>
}
