# Behaviour
# 1 location(s):
#   Belltown_Room_Spare / furnishings/Gramaphone

fsm Behaviour {
  start Idle

  state Idle {
    on START PLAYING → Start
    on ALREADY PLAYING → Playing
  }

  state Start {
    CallMethodProper(gameObject=Self, behaviour="JitterSelfForTime", methodName="StartTimedJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=furnishings/Gramaphone/Start Dust, behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    PlayAudioEvent(audioClip=librarian_gramaphone_start (sfxstatic_assets_arealibrary.bundle), pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Top", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Playing
  }

  state Stop {
    CallMethodProper(gameObject=var "Bottom", behaviour="JitterSelf", methodName="StopJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Top", behaviour="JitterSelf", methodName="StopJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=furnishings/Gramaphone/Stop Dust, behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    on FINISHED → Idle
  }

  state Playing {
    CallMethodProper(gameObject=var "Bottom", behaviour="JitterSelf", methodName="StartJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Top", behaviour="JitterSelf", methodName="StartJitter", parameters=[], storeResult=(unset var), EveryFrame=false)
    on STOP PLAYING → Stop
  }

  var Top: gameObject = furnishings/Gramaphone/gramophone_horn
  var Bottom: gameObject = furnishings/Gramaphone/gramophone_base
}
