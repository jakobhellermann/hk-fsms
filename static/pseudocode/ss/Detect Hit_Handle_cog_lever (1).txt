fsm Detect Hit {
  start Init

  state Idle {
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"HIT", storeCollider=var "")  // disabled
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"HIT", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on HIT → Send
  }

  state Send {
    CheckAlertRange(alertRange=var "Can Hit Range", storeResult=(unset), InRangeEvent=(none), InRangeDelay=0, OutOfRangeEvent=→"CANCEL", OutOfRangeDelay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Parent"), sendEvent="LEVER HIT", delay=0, everyFrame=false)
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Idle
    on CANCEL → Idle
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindAlertRange(target=var "Parent", storeResult=var "Can Hit Range", childName="Can Hit Range")
    on FINISHED → Idle
  }

  var Parent: gameObject = <null>
  var Can Hit Range: object = <null>
}
