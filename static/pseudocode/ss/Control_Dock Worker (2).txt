# Control
# 1 location(s):
#   Dock_09 / Boss Scene/Intro Minions/Dock Worker (2)

fsm Control {
  start Idle
  on TOOK DAMAGE → Dmg Response  // from any state
  on LEAVE → Leave  // from any state

  state Idle {
    GetGrandparent(gameObject=Self, storeResult=var "Boss Scene")
    on LEAVE → Leave
    on TOOK DAMAGE → Dmg Response
  }

  state Dmg Response {
    SendEventByName(eventTarget=GameObject(var "Boss Scene"), sendEvent="MINION DAMAGED", delay=0, everyFrame=false)
    on FINISHED → Leave
  }

  state Leave {
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    Tk2dPlayAnimationWithEvents(gameObject=Self, clipName="Leave", animationTriggerEvent=(none), animationCompleteEvent=→"FINISHED")
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=garmond_jump (sfxstatic_assets_garmond.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=Dock_Worker_alert_01 (sfxstatic_assets_areadocks.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Instruct Pause {
    WaitRandom(timeMin=0.5, timeMax=0.7, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → React
  }

  state React {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="StartleToIdle")
    Tk2dPlayFrame(gameObject=Self, frame=0)
    AudioPlayRandomVoiceFromTable(gameObject=Self, audioClipTable=dock_worker_instruct_react (sfxdynamic_assets_areadocks.bundle), pitchOffset=0, stopPreviousSound=true, forcePlay=false)
    on FINISHED → Idle
  }

  var Boss Scene: gameObject = <null>
}
