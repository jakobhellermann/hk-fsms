# Activation
# 1 location(s):
#   Slab_16 / Audio Loop Slab_16 flies distant

fsm Activation {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check
  }

  state Check {
    CheckIfCrestEquipped(Crest=Cloakless (dataassets_assets_assets/dataassets/tools/crestitems.bundle), trueEvent=(none), falseEvent=→"DEACTIVATE", storeValue=(unset))
    GetEntryGateName(StoreValue=var "Entry Gate")
    StringCompare(stringVariable=var "Entry Gate", compareTo="right1", equalEvent=(none), notEqualEvent=→"DEACTIVATE", storeResult=(unset), everyFrame=false)
    on DEACTIVATE → Deactivate
  }

  state Deactivate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Entry Gate: string = ""
}
