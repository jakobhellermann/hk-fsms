# Drop Shards
# 1 location(s):
#   Bone_East_14 / Vassal_Rest_Spot/Not Present/Shell Fossil Mimic

fsm Drop Shards {
  start Pause
  on ZERO HP → Die  // from any state

  state Idle {
    on TOOK DAMAGE → Fling
  }

  state Fling {
    PlayAudioEvent(audioClip=shell_shard_rock_break (sfxstatic_assets_areabone.bundle), pitchMin=0.8, pitchMax=1.2, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SelectRandomGameObject(gameObjects=[Shell Shard 01 (globalpoolprefabs_assets_all.bundle), Shell Shard 02 (globalpoolprefabs_assets_all.bundle), Shell Shard 03 (globalpoolprefabs_assets_all.bundle)], weights=[1, 1, 1], storeGameObject=var "Shard Prefab")
    FlingObjectsFromGlobalPool(gameObject=var "Shard Prefab", spawnPoint=var "Self", position=(unset), spawnMin=3, spawnMax=3, speedMin=20, speedMax=35, angleMin=40, angleMax=140, originVariationX=0, originVariationY=0, FSM="", FSMEvent="")
    IntAdd(intVariable=var "Shards", add=-3, everyFrame=false)
    SetShardDrop(target=Self, shards=var "Shards")
    SetIntValue(intVariable=var "Value", intValue=var "Shards", everyFrame=false)
    IntCompare(integer1=var "Shards", integer2=0, equal=→"END", lessThan=→"END", greaterThan=(none), everyFrame=false)
    on END → End
    on FINISHED → Idle
  }

  state Init {
    RandomInt(min=50, max=60, storeResult=var "Shards", inclusiveMax=true, noRepeat=false)  // disabled
    SetIntValue(intVariable=var "Shards", intValue=var "Value", everyFrame=false)
    IntCompare(integer1=var "Value", integer2=0, equal=→"END", lessThan=→"DEAD", greaterThan=(none), everyFrame=false)
    on FINISHED → Idle
    on DEAD → Dead
    on END → End
  }

  state End {
    SetIntValue(intVariable=var "Value", intValue=0, everyFrame=false)
    SetPersistentInt(target=Self, setValue=0)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Dead {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Die {
    ActivateGameObject(gameObject=var "Death Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetParent(gameObject=var "Death Effects", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    SetIntValue(intVariable=var "Value", intValue=-1, everyFrame=false)
    SetPersistentInt(target=Self, setValue=-1)
    CallMethodProper(gameObject=Self, behaviour="PersistentIntItem", methodName="SaveState", parameters=[], storeResult=(unset var), EveryFrame=false)
  }

  var Shards: int = 0
  var Value: int = 30
  var Self: gameObject = Vassal_Rest_Spot/Not Present/Shell Fossil Mimic
  var Shard Prefab: gameObject = <null>
  var Death Effects: gameObject = Vassal_Rest_Spot/Not Present/Shell Fossil Mimic/Death Effects
}
