# Control
# 1 location(s):
#   Abyss_08 / weaver_spire_base control

fsm Control {
  start Init
  on ABYSS TENDRIL HAZARD → Hazard Pause  // from any state

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Hazard Respawn Trigger Base")
    FindNamedChild(gameObject=Self, storeResult=var "Shrine Weaver Ability")
    FindNamedChild(gameObject=Self, storeResult=var "weaver_spire_base")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Grab")
    FindNamedChild(gameObject=Self, storeResult=var "Pt PullIn")
    FindNamedChild(gameObject=Self, storeResult=var "Sequence End Range")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Rumble")
    FindNamedChild(gameObject=Self, storeResult=var "Completed Range")
    FindNamedChild(gameObject=Self, storeResult=var "Audio Loop Weaver Atmos")
    FindNamedChild(gameObject=var "weaver_spire_base", storeResult=var "Grab Tendrils")
    GetPosition2D(gameObject=var "weaver_spire_base", vector=(unset), x=(unset), y=var "Base Init Y", space=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Completed Range", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    AudioPlaySimple(gameObject=var "Audio Loop Weaver Atmos", volume=1, oneShotClip=<null>)
    GetPlayerDataBool(boolName="hasSuperJump", storeValue=var "Has Super Jump")
    GetPlayerDataVariable(VariableName="respawnScene", StoreValue=var "Respawn Scene")
    StringCompare(stringVariable=var "Respawn Scene", compareTo="Abyss_08", equalEvent=(none), notEqualEvent=(none), storeResult=var "Death Respawn Here", everyFrame=false)
    BoolTestMulti(boolVariables=[var "Has Super Jump", var "Death Respawn Here"], boolStates=[true, false], trueEvent=→"SEQUENCE COMPLETED", falseEvent=(none), storeResult=(unset), everyFrame=false)
    BoolTestDelay(boolVariable=var "Has Super Jump", isTrue=→"HAS SUPER JUMP", isFalse=(none), delay=2)  // disabled
    ActivateGameObjectDelay(gameObject=var "Completed Range", activate=var "Has Super Jump", resetOnExit=false, delay=0.5)
    Trigger2dEvent(gameObject=var "Completed Range", trigger=0, collideTag="", sendEvent=→"SEQUENCE COMPLETED", storeCollider=var "")
    on SHRINE SEQUENCE END → Tendrils Up
    on HERO RESPAWNING HERE → Tendrils Up
    on SEQUENCE COMPLETED → Shrine Completed
  }

  state Tendrils Up {
    SetBoolValue(boolVariable=var "Grab Sequence Started", boolValue=true, everyFrame=false)
    Wait(time=0.21, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Grab Tendrils", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    PlayAudioEvent(audioClip=radiance_tentacles_whip_up (sfxstatic_assets_areaabyss.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=abyss_tendrils_emerge (sfxstatic_assets_areaabyss.bundle), pitchMin=0.8, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Grab Plat
  }

  state Grab Plat {
    AudioPlaySimple(gameObject=var "Audio Loop Rumble", volume=1, oneShotClip=<null>)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitter(gameObject=var "Pt Grab", emit=0, resetIfPlaying=false)
    AnimatePositionBy(gameObject=var "weaver_spire_base", shiftBy=(0, -0.75, 0), time=0.25, speed=(unset), delay=(unset), easeType=4, reverse=false, finishEvent=(none), realTime=false)
    SendMessage(gameObject=var "weaver_spire_base", delivery=SendMessage, options=1, functionCall=StartJitter())
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Message Up
    on SUPER JUMP LAUNCH → Drag In
  }

  state Message Up {
    SendEventToRegister(eventName="REMINDER SUPER JUMP")
    Wait(time=5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Drag In
    on SUPER JUMP LAUNCH → Drag In
    on SUPER JUMP THROW NEEDLE → Wait For Needle
  }

  state Drag In {
    AudioStop(gameObject=var "Audio Loop Weaver Atmos", fadeTime=0.5)
    CancelCameraShake(Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Small Rumble (dataassets_assets_assets/dataassets/camerashake.bundle))
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    AnimatePositionBy(gameObject=var "weaver_spire_base", shiftBy=(0, -15, 0), time=0.5, speed=(unset), delay=(unset), easeType=3, reverse=false, finishEvent=(none), realTime=false)
    AudioStop(gameObject=var "Audio Loop Rumble", fadeTime=0)
    PlayAudioEvent(audioClip=breakable_wall_death (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player Actor Distant V2@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=big splash (sfxstatic_assets_water.bundle), pitchMin=1.1, pitchMax=1.1, volume=1, audioPlayerPrefab=Audio Player Actor Distant V2@AudioSource (audioobjects_assets_all.bundle), spawnPoint=var "Hero", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayParticleEmitter(gameObject=var "Pt PullIn", emit=0, resetIfPlaying=false)
    ActivateGameObject(gameObject=var "Sequence End Range", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Trigger2dEvent(gameObject=var "Sequence End Range", trigger=0, collideTag="", sendEvent=→"SEQUENCE COMPLETED", storeCollider=var "")
    on SEQUENCE COMPLETED → Set Completed
  }

  state Hazard Pause {
    PlayerDataBoolTest(boolName="completedSuperJumpSequence", isTrue=→"SEQUENCE COMPLETED", isFalse=(none))  // disabled
    Wait(time=0.6, finishEvent=→"FINISHED", realTime=false)
    SendEventToRegister(eventName="REMINDER SUPER JUMP END")
    on FINISHED → Reset
    on SEQUENCE COMPLETED → Stay Inert
  }

  state Reset {
    BoolTest(boolVariable=var "Hornet Dead", isTrue=→"HORNET DEAD", isFalse=(none), everyFrame=false)
    GetPlayerDataVariable(VariableName="hazardRespawnLocation", StoreValue=var "Hazard Respawn Pos")
    GetVector3XYZ(vector3Variable=var "Hazard Respawn Pos", storeX=(unset), storeY=var "Hazard Respawn Y", storeZ=(unset), everyFrame=false)
    FloatCompare(float1=var "Hazard Respawn Y", float2=20, tolerance=0, equal=(none), lessThan=(none), greaterThan=→"CANCEL", everyFrame=false)
    AudioStop(gameObject=var "Audio Loop Rumble", fadeTime=0)
    SetPosition(gameObject=var "weaver_spire_base", vector=(unset), x=(unset), y=var "Base Init Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    BoolTest(boolVariable=var "Grab Sequence Started", isTrue=→"GRAB START", isFalse=→"FINISHED", everyFrame=false)
    on GRAB START → Grab Start Pause
    on FINISHED → Idle
    on HORNET DEAD → Stay Inert
    on CANCEL → Stay Inert
  }

  state Grab Start Pause {
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on HAZARD RESPAWN RESET → Tendrils Up
    on FINISHED → Tendrils Up
  }

  state Wait For Needle {
    on SUPER JUMP LAUNCH → Drag In
  }

  state Stay Inert {
  }

  state Set Completed {
    SetPlayerDataBool(boolName="completedSuperJumpSequence", value=true)  // disabled
    SetCollider(gameObject=var "Hazard Respawn Trigger Base", active=false, resetOnExit=false)
    SetDeathRespawn(gameManager=var "GameManager", respawnMarkerName="Death Respawn Marker SequenceEnd", respawnType=0, respawnFacingRight=true)  // disabled
    SaveGame()  // disabled
  }

  state Shrine Completed {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Base Init Y: float = 0
  var Hazard Respawn Y: float = 0
  var Grab Sequence Started: bool = false
  var Has Super Jump: bool = false
  var Hornet Dead: bool = false
  var Death Respawn Here: bool = false
  var Respawn Scene: string = ""
  var Hazard Respawn Pos: vector3 = (0, 0, 0)
  var Hazard Respawn Trigger Base: gameObject = <null>
  var Shrine Weaver Ability: gameObject = <null>
  var weaver_spire_base: gameObject = <null>
  var Pt Grab: gameObject = <null>
  var Grab Tendrils: gameObject = <null>
  var Pt PullIn: gameObject = <null>
  var Sequence End Range: gameObject = <null>
  var Audio Loop Rumble: gameObject = <null>
  var Audio Loop Weaver Atmos: gameObject = <null>
  var Completed Range: gameObject = <null>
}
