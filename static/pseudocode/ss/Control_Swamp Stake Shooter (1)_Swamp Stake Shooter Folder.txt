# Control
# 2 location(s):
#   Shadow_10 / Swamp Stake Shooter Folder/Swamp Stake Shooter (1)
#   Shadow_12 / Swamp Stake Shooter Folder/Swamp Stake Shooter (1)

fsm Control {
  start Init

  state Init {
    GetRotation(gameObject=Self, quaternion=(unset), vector=var "Rotation", xAngle=(unset), yAngle=(unset), zAngle=(unset), space=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Stake L", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Stake M", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Stake R", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Idle
  }

  state Idle {
    on TRAP → Fire
  }

  state Fire {
    SpawnObjectFromGlobalPool(gameObject=Swamp Stake Projectile (localpoolprefabs_assets_areaswamp.bundle), spawnPoint=var "Stake L", position=(unset), rotation=var "Rotation", storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Swamp Stake Projectile (localpoolprefabs_assets_areaswamp.bundle), spawnPoint=var "Stake M", position=(unset), rotation=var "Rotation", storeObject=var "")
    SpawnObjectFromGlobalPool(gameObject=Swamp Stake Projectile (localpoolprefabs_assets_areaswamp.bundle), spawnPoint=var "Stake R", position=(unset), rotation=var "Rotation", storeObject=var "")
  }

  var Angle: float = 0
  var Rotation: vector3 = (0, 0, 0)
  var Stake L: gameObject = Swamp Stake Shooter Folder/Swamp Stake Shooter (1)/Stake L
  var Stake M: gameObject = Swamp Stake Shooter Folder/Swamp Stake Shooter (1)/Stake M
  var Stake R: gameObject = Swamp Stake Shooter Folder/Swamp Stake Shooter (1)/Stake R
  var Self: gameObject = Swamp Stake Shooter Folder/Swamp Stake Shooter (1)
}
