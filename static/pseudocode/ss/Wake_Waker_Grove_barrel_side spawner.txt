# Wake
# 1 location(s):
#   Clover_11 / Grove_barrel_side spawner/Waker

fsm Wake {
  start Init
  on END → End  // from any state

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    FindNamedChild(gameObject=Self, storeResult=var "Antic Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Breaker")
    on FINISHED → Idle
  }

  state Idle {
    on TRIGGER STAY 2D → Wake Antic
  }

  state Wake Antic {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
    ActivateGameObject(gameObject=var "Antic Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ObjectJitterOnRender(Target=var "Parent", X=0.1, Y=0.1, Z=(unset), LimitFps=30)
    Wait(time=0.75, finishEvent=→"FINISHED", realTime=false)
    PlayAudioEvent(audioClip=coral_tree_antic_for_break_3 (sfxstatic_assets_areacoral.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Parent", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=grasshopper_fly_attack_antic_slashy (sfxstatic_assets_areaclover.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Parent", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Break
  }

  state End {
    SetCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  state Break {
    ActivateGameObject(gameObject=var "Breaker", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Parent: gameObject = <null>
  var Antic Effects: gameObject = <null>
  var Breaker: gameObject = <null>
}
