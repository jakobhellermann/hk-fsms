# Msg Control
# 1 location(s):
#   prompts_assets_all.bundle / UI Msg Get Map

fsm Msg Control {
  start Init

  state Detect {
    ListenForPromptContinue(EventTarget=Self, ContinuePressed=→"PRESS")
    on PRESS → Down
  }

  state Done {
    SendEventToRegister(eventName="GET ITEM MSG END")
    SetStaticVariable(variableName="IsUIListInputBlocked", setValue=false, sceneTransitionsLimit=0)
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Top Up {
    SendEventByName(eventTarget=GameObject(var "BG"), sendEvent="UP", delay=0, everyFrame=false)
    PlayAudioEvent(audioClip=ui_tutorial_pane_option_2, pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=Audio Player UI@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    AudioPlayerOneShotSingle(audioPlayer=var "Audio Player", spawnPoint=var "Hero", audioClip=spell_information_screen (sfxstatic_assets_menu.bundle), pitchMin=1, pitchMax=1, volume=1, delay=0.75, storePlayer=<null>)  // disabled
    SendEventByName(eventTarget=GameObject(var "Icon"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Item Name Prefix"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=2.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "Fleur", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessage(gameObject=var "Button", delivery=SendMessage, options=1, functionCall=RefreshButtonIcon())
    ActivateGameObjectDelay(gameObject=var "Icon Flash", activate=true, resetOnExit=false, delay=0.65)
    SendEventByName(eventTarget=GameObject(var "CameraParent"), sendEvent="EnemyKillShake", delay=0.65, everyFrame=false)
    on FINISHED → Bot Up
  }

  state Bot Up {
    SendEventByName(eventTarget=GameObject(var "Press"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Msg 1"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Msg 2"), sendEvent="UP", delay=0, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Button"), sendEvent="UP", delay=0, everyFrame=false)
    Wait(time=3, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Stop Up
  }

  state Stop Up {
    ActivateGameObject(gameObject=var "Stop", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Detect
  }

  state Down {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DOWN", delay=0, everyFrame=false)
    AudioPlayerOneShotSingle(audioPlayer=Audio Player UI (audioobjects_assets_all.bundle), spawnPoint=var "Hero", audioClip=button (sfxstatic_assets_shared.bundle), pitchMin=0.8, pitchMax=0.8, volume=0.3, delay=0, storePlayer=<null>)
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Done
  }

  state Set Text {
    GetLanguageString(sheetName="UI", convName="INV_NAME_MAP", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_ITEM_INTRO_GET_A", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Item Name Prefix", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="BUTTON_DESC_HOLD", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Press", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_MAP_1", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Msg 1", textString=var "Game Text")
    GetLanguageString(sheetName="Prompts", convName="GET_MAP_2", storeValue=var "Game Text")
    SetTextMeshProText(gameObject=var "Msg 2", textString=var "Game Text")
    on FINISHED → Top Up
  }

  state Init {
    ActivateGameObject(gameObject=var "Stop", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Fleur", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Icon Flash", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Disable Input
  }

  state Disable Input {
    SetStaticVariable(variableName="IsUIListInputBlocked", setValue=true, sceneTransitionsLimit=0)
    on FINISHED → Is Counter Rolling?
  }

  state Wait {
    SendEventToRegister(eventName="CLOSE SHOP WINDOW TEMP")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Set Text
  }

  sequence state Is Counter Rolling? {
    CheckIsAnyCurrencyCounterRolling(trueEvent=(none), falseEvent=→"FALSE", storeValue=(unset))
    NextFrameEvent(sendEvent=(none))
    on FINISHED → Is Counter Rolling?
    on FALSE → Wait
  }

  var Item: string = ""

  var Game Text: string = ""
  var Language: string = ""
  var Audio Player: gameObject = <null>
  var BG: gameObject = UI Msg Get Map/BG
  var Button: gameObject = UI Msg Get Map/Button
  var Fleur: gameObject = UI Msg Get Map/Fleur
  var Icon: gameObject = UI Msg Get Map/Icon
  var Icon Flash: gameObject = UI Msg Get Map/Icon Flash
  var Item Name: gameObject = UI Msg Get Map/Item Name
  var Item Name Prefix: gameObject = UI Msg Get Map/Item Name Prefix
  var Msg 1: gameObject = UI Msg Get Map/Msg 1
  var Msg 2: gameObject = UI Msg Get Map/Msg 2
  var Press: gameObject = UI Msg Get Map/Press
  var Stop: gameObject = UI Msg Get Map/Stop
}
