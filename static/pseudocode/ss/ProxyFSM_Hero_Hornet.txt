# ProxyFSM
# 1 location(s):
#   heroloading_assets_all.bundle / Hero_Hornet

fsm ProxyFSM {
  start Init

  state Idle {
    on HeroCtrl-HeroDamaged → Broadcast Damage
    on HeroCtrl-LeftGround → Left Ground
    on HeroCtrl-LeavingScene → Leaving Scene
    on HeroCtrl-Respawned → Respawn
    on HeroCtrl-Landed → Landed
    on HeroCtrl-MaxHealth → Healed Max
    on HeroCtrl-DashEnd → End Dash
    on HeroCtrl-HazardRespawned → Hazard Respawn
    on HeroCtrl-EnteringScene → Entering Scene
  }

  state Broadcast Damage {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="HERO DAMAGED", delay=0, everyFrame=false)  // disabled
    SendEventToRegister(eventName="HERO DAMAGED")
    SendEventToRegister(eventName="HEALTH UPDATE")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="HERO DAMAGED", delay=0, everyFrame=false)  // disabled
    on FINISHED → Idle
  }

  state Left Ground {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LEFT GROUND", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Leaving Scene {
    SendEventToRegister(eventName="LEAVING SCENE")
    SendEventByName(eventTarget=GameObject(Self), sendEvent="LEAVING SCENE", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Respawn {
    MissingAction(actionName="SendEventToRegister")  // disabled
    SendEventToRegister(eventName="HERO RESPAWNED")
    on FINISHED → Idle
  }

  state Landed {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="HERO LANDED", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Init {
    on FINISHED → Idle
  }

  state Healed Max {
    SendEventByName(eventTarget=BroadcastAll, sendEvent="HERO HEALED FULL", delay=0, everyFrame=false)  // disabled
    MissingAction(actionName="SendEventToRegister")  // disabled
    SendEventToRegister(eventName="HERO HEALED FULL")
    on FINISHED → Idle
  }

  state End Dash {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="DASH END", delay=0, everyFrame=false)
    on FINISHED → Idle
  }

  state Hazard Respawn {
    SendEventToRegister(eventName="HAZARD RESPAWNED")
    on FINISHED → Idle
  }

  state Entering Scene {
    SendEventToRegister(eventName="ENTERING SCENE")
    on FINISHED → Idle
  }

  var Faced Radiance: bool = false
  var Item Name: string = ""
  var Msg: gameObject = <null>
  var Msg Item: gameObject = <null>
  var Msg Text: gameObject = <null>
  var Self: gameObject = <null>
  var White_Flower_Break: gameObject = <null>
}
