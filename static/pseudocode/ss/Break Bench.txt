# Break Bench
# 1 location(s):
#   Bonetown / Black Thread States/Normal World/RestBench Control

fsm Break Bench {
  start Pause

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "RestBench")
    FindNamedChild(gameObject=Self, storeResult=var "Bench Break Effects")
    FindNamedChild(gameObject=Self, storeResult=var "Bench Skull King Detector")
    PlayerDataBoolTest(boolName="skullKingInvaded", isTrue=→"CANCEL", isFalse=(none))
    on FINISHED → Idle
    on CANCEL → Cancel
  }

  state Idle {
    Trigger2dEvent(gameObject=var "Bench Skull King Detector", trigger=0, collideTag="", sendEvent=→"BREAK", storeCollider=var "")
    on BREAK → Break
  }

  state Break {
    PlayAudioEvent(audioClip=breakable_wall_hit_1 (sfxstatic_assets_shared.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=var "Bench Break Effects", spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Bench Break Effects", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "RestBench", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    SetPlayerDataBool(boolName="atBench", value=false)
  }

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Cancel {
  }

  var RestBench: gameObject = <null>
  var Bench Skull King Detector: gameObject = <null>
  var Bench Break Effects: gameObject = <null>
}
