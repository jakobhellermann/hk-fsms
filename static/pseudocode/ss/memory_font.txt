// uses template: memory_font
fsm memory_font {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "memory_font")
    FindNamedChild(gameObject=Self, storeResult=var "Needolin Memory")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    on ACTIVATED → Completed
    on FINISHED → Idle
  }

  state Idle {
    CreateObject(gameObject=thread_memory_region (sharedeffects_assets_all.bundle), spawnPoint=<null>, position=(unset), rotation=(unset), storeObject=var "thread_memory_region")
    ActivateGameObject(gameObject=var "thread_memory_region", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Completed {
    SendEventByName(eventTarget=GameObject(var "memory_font"), sendEvent="COMPLETED", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Needolin Memory", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  var Activated: bool = false
  var Needolin Memory: gameObject = <null>
  var memory_font: gameObject = <null>
  var thread_memory_region: gameObject = <null>
}
