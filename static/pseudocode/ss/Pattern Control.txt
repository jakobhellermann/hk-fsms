# Pattern Control
# 1 location(s):
#   Slab_10b / Boss Scene/Pin Projectiles

fsm Pattern Control {
  start Init
  on BOSS STUNNED → Idle  // from any state

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on ATTACK → Single Pattern
    on ATTACK DOUBLE → Double Pattern
  }

  state Single Pattern {
    SendEvent(eventTarget=Self, sendEvent=→"CLAW", delay=0, everyFrame=false)  // disabled
    SetBoolValue(boolVariable=var "Double Ptn", boolValue=false, everyFrame=false)
    SendRandomEventV4(events=[→"RAIN", →"SWEEP", →"PINCER", →"CLAW"], weights=[0.25, 0.25, 0.25, 0.25], eventMax=[1, 1, 1, 1], missedMax=[5, 5, 5, 5], activeBool=(unset))
    on RAIN → Rain 1
    on SWEEP → Sweep Dir
    on PINCER → Pincer Check
    on CLAW → Claw Dir
  }

  state Rain 1 {
    RandomFloat(min=24.5, max=28.5, storeResult=var "Pos X")
    RandomFloat(min=19, max=19.6, storeResult=var "Pos Y")
    RandomFloat(min=-85, max=-95, storeResult=var "Rotation")
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SetIntValue(intVariable=var "Remaining", intValue=4, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Rain 2
  }

  state Rain 2 {
    FloatAdd(floatVariable=var "Pos X", add=6, everyFrame=false, perSecond=false)
    RandomFloat(min=19, max=19.6, storeResult=var "Pos Y")
    RandomFloat(min=-85, max=-95, storeResult=var "Rotation")
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=var "Pos Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    IntAdd(intVariable=var "Remaining", add=-1, everyFrame=false)
    IntCompare(integer1=var "Remaining", integer2=0, equal=→"FINISHED", lessThan=→"FINISHED", greaterThan=→"REPEAT", everyFrame=false)
    on FINISHED → Double?
    on REPEAT → Rain 2
  }

  state Sweep Dir {
    CheckXPosition(gameObject=var "Hero", compareTo=var "Scene Centre X", compareToOffset=0, tolerance=0, equal=→"R", equalBool=(unset), lessThan=→"L", lessThanBool=(unset), greaterThan=→"R", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    on R → Sweep R
    on L → Sweep L
  }

  state Sweep R {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Pos X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Pos X", add=-15, everyFrame=false, perSecond=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    RandomFloat(min=-1, max=1, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=10, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=var "Next Pin", translateMin=(-0.5, 0, 0), translateMax=(0.5, 0, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    RandomFloat(min=-1, max=1, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=17, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=var "Next Pin", translateMin=(-0.5, 0, 0), translateMax=(0.5, 0, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Double?
  }

  state Sweep L {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Pos X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Pos X", add=15, everyFrame=false, perSecond=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    RandomFloat(min=179, max=181, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=10, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=var "Next Pin", translateMin=(-0.5, 0, 0), translateMax=(0.5, 0, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    RandomFloat(min=179, max=181, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=17, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    TranslateRandom(gameObject=var "Next Pin", translateMin=(-0.5, 0, 0), translateMax=(0.5, 0, 0), space=0)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Double?
  }

  state Pincer Check {
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Hero X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatInRange(floatVariable=var "Hero X", lowerValue=var "Pincer Min X", upperValue=var "Pincer Max X", boolVariable=(unset), trueEvent=(none), falseEvent=→"CANCEL", everyFrame=false)
    on CANCEL → Single Pattern
    on FINISHED → Pincer
  }

  state Pincer {
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Pos X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Pos X", add=7, everyFrame=false, perSecond=false)
    RandomFloat(min=-133, max=-137, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=18, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    GetPosition(gameObject=var "Hero", vector=(unset), x=var "Pos X", y=(unset), z=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Pos X", add=-7, everyFrame=false, perSecond=false)
    RandomFloat(min=-43, max=-47, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=18, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Double?
  }

  state Claw Dir {
    CheckXPosition(gameObject=var "Hero", compareTo=var "Pincer Min X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"L", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    CheckXPosition(gameObject=var "Hero", compareTo=var "Pincer Max X", compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"R", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    SendRandomEvent(events=[→"L", →"R"], weights=[0.5, 0.5], delay=0)
    on R → Claw R
    on L → Claw L
  }

  state Claw R {
    RandomFloat(min=0, max=5, storeResult=var "Offset")
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    FloatOperator(float1=24.3, float2=var "Offset", operation=Add, storeResult=var "Pos X", everyFrame=false)
    RandomFloat(min=-70, max=-70, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=18.3, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    FloatOperator(float1=31.3, float2=var "Offset", operation=Add, storeResult=var "Pos X", everyFrame=false)
    RandomFloat(min=-55, max=-55, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=19.3, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    FloatOperator(float1=38.3, float2=var "Offset", operation=Add, storeResult=var "Pos X", everyFrame=false)
    RandomFloat(min=-40, max=-40, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=20.3, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Double?
  }

  state Claw L {
    RandomFloat(min=0, max=-5, storeResult=var "Offset")
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    FloatOperator(float1=52.7, float2=var "Offset", operation=Add, storeResult=var "Pos X", everyFrame=false)
    RandomFloat(min=-110, max=-110, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=18.3, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    FloatOperator(float1=44.7, float2=var "Offset", operation=Add, storeResult=var "Pos X", everyFrame=false)
    RandomFloat(min=235, max=235, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=19.3, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    GetRandomChild(gameObject=Self, storeResult=var "Next Pin")
    GameObjectIsNull(gameObject=var "Next Pin", isNull=→"FINISHED", isNotNull=(none), storeResult=(unset), everyFrame=false)
    FloatOperator(float1=36.7, float2=var "Offset", operation=Add, storeResult=var "Pos X", everyFrame=false)
    RandomFloat(min=220, max=220, storeResult=var "Rotation")
    SetPosition(gameObject=var "Next Pin", vector=(unset), x=var "Pos X", y=20.3, z=(unset), space=0, everyFrame=false, lateUpdate=false)
    SetRotation(gameObject=var "Next Pin", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotation", space=0, everyFrame=false, lateUpdate=false)
    SendEventByName(eventTarget=GameObject(var "Next Pin"), sendEvent="ATTACK", delay=0, everyFrame=false)
    on FINISHED → Double?
  }

  state Double Pattern {
    SetBoolValue(boolVariable=var "Double Ptn", boolValue=true, everyFrame=false)
    SendRandomEventV4(events=[→"SWEEP RAIN", →"SWEEP PINCER", →"SWEEP CLAW"], weights=[0.25, 0.25, 0.25], eventMax=[1, 1, 1], missedMax=[3, 3, 3], activeBool=(unset))
    on SWEEP RAIN → Set Sweep Rain
    on SWEEP PINCER → Set Sweep Pincer
    on SWEEP CLAW → Set Sweep Claw
  }

  state Set Sweep Rain {
    SetStringValue(stringVariable=var "Attack Event A", stringValue="SWEEP", everyFrame=false)
    SetStringValue(stringVariable=var "Attack Event B", stringValue="RAIN", everyFrame=false)
    on FINISHED → Set Ordering
  }

  state Set Sweep Pincer {
    SetStringValue(stringVariable=var "Attack Event A", stringValue="SWEEP", everyFrame=false)
    SetStringValue(stringVariable=var "Attack Event B", stringValue="PINCER", everyFrame=false)
    on FINISHED → Set Ordering
  }

  state Set Sweep Claw {
    SetStringValue(stringVariable=var "Attack Event A", stringValue="SWEEP", everyFrame=false)
    SetStringValue(stringVariable=var "Attack Event B", stringValue="CLAW", everyFrame=false)
    on FINISHED → Set Ordering
  }

  state Set Ordering {
    SetStringValue(stringVariable=var "Attack Event 1", stringValue=var "Attack Event A", everyFrame=false)
    SetStringValue(stringVariable=var "Attack Event 2", stringValue=var "Attack Event B", everyFrame=false)
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.5, 0.5], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    SetStringValue(stringVariable=var "Attack Event 1", stringValue=var "Attack Event B", everyFrame=false)
    SetStringValue(stringVariable=var "Attack Event 2", stringValue=var "Attack Event A", everyFrame=false)
    on FINISHED → Attack 1
  }

  state Attack 1 {
    SendEventByName(eventTarget=Self, sendEvent=var "Attack Event 1", delay=0, everyFrame=false)
    on RAIN → Rain 1
    on SWEEP → Sweep Dir
    on PINCER → Pincer Check
    on CLAW → Claw Dir
  }

  state Double? {
    BoolTest(boolVariable=var "Double Ptn", isTrue=(none), isFalse=→"FINISHED", everyFrame=false)
    SetBoolValue(boolVariable=var "Double Ptn", boolValue=false, everyFrame=false)
    Wait(time=0.5, finishEvent=→"ATTACK DOUBLE", realTime=false)
    on FINISHED → Idle
    on ATTACK DOUBLE → Attack 2
  }

  state Attack 2 {
    SendEventByName(eventTarget=Self, sendEvent=var "Attack Event 2", delay=0, everyFrame=false)
    on RAIN → Rain 1
    on SWEEP → Sweep Dir
    on PINCER → Pincer Check
    on CLAW → Claw Dir
  }

  var Pos X: float = 0
  var Pos Y: float = 0
  var Rotation: float = 0
  var Scene Centre X: float = 38.5
  var Pincer Min X: float = 31
  var Pincer Max X: float = 46
  var Hero X: float = 0
  var Offset: float = 0
  var Remaining: int = 0
  var Double Ptn: bool = false
  var Attack Event A: string = ""
  var Attack Event B: string = ""
  var Attack Event 1: string = ""
  var Attack Event 2: string = ""
  var Next Pin: gameObject = <null>
}
