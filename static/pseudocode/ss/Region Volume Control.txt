fsm Region Volume Control {
  start Normal

  state Normal {
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Audio Volume Extra", setValue=0.4, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Audio Volume", setValue=0.75, everyFrame=false)
    on GRINDER QUIET → Quiet
  }

  state Quiet {
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Audio Volume Extra", setValue=0, everyFrame=false)
    SetFsmFloat(gameObject=Self, fsmName="Control", variableName="Audio Volume", setValue=0.35, everyFrame=false)
    on GRINDER NORMAL → Normal
  }
}
