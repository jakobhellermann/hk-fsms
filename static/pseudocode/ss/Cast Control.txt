# Cast Control
# 1 location(s):
#   Clover_19 / Boss Scene/Cloverstag White Boss

// component disabled: this FSM does not run until something enables it
fsm Cast Control {
  start Init

  state Init {
    on FINISHED → Idle
  }

  state Idle {
    on CAST START → Pause
  }

  state Pause {
    WaitRandom(timeMin=0.75, timeMax=1, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Vert Choice
  }

  state Aim Choice {
    CheckXPosition(gameObject=var "Hero", compareTo=30, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=→"R", lessThanBool=(unset), greaterThan=(none), greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    CheckXPosition(gameObject=var "Hero", compareTo=45, compareToOffset=0, tolerance=0, equal=(none), equalBool=(unset), lessThan=(none), lessThanBool=(unset), greaterThan=→"L", greaterThanBool=(unset), everyFrame=false, space=0, activeBool=(unset))
    SendRandomEventV4(events=[→"L", →"R"], weights=[0.5, 0.5], eventMax=[2, 2], missedMax=[2, 2], activeBool=(unset))
    on L → L
    on R → R
  }

  state L {
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Spawn X", y=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Spawn X", add=-10, everyFrame=false, perSecond=false)
    SpawnObjectFromGlobalPool(gameObject=Cloverstag White Sickle (localpoolprefabs_assets_areaclover.bundle), spawnPoint=<null>, position=(0, 0, 0), rotation=(unset), storeObject=var "Sickle")
    SetPosition(gameObject=var "Sickle", vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state R {
    GetPosition2D(gameObject=var "Hero", vector=(unset), x=var "Spawn X", y=(unset), space=0, everyFrame=false)
    FloatAdd(floatVariable=var "Spawn X", add=10, everyFrame=false, perSecond=false)
    SpawnObjectFromGlobalPool(gameObject=Cloverstag White Sickle (localpoolprefabs_assets_areaclover.bundle), spawnPoint=<null>, position=(0, 0, 0), rotation=(unset), storeObject=var "Sickle")
    SetPosition(gameObject=var "Sickle", vector=(unset), x=var "Spawn X", y=var "Spawn Y", z=(unset), space=0, everyFrame=false, lateUpdate=false)
    on FINISHED → Pause
  }

  state Vert Choice {
    SendRandomEventV4(events=[→"MID", →"AIM"], weights=[1, 1], eventMax=[1, 1], missedMax=[1, 1], activeBool=(unset))
    on MID → Y Mid
    on AIM → Y Aim
  }

  state Y Mid {
    RandomFloat(min=17, max=19, storeResult=var "Spawn Y")
    on FINISHED → Aim Choice
  }

  state Y Aim {
    RandomFloat(min=13, max=14, storeResult=var "Spawn Y")
    CheckYPosition(gameObject=var "Hero", compareTo=18, compareToOffset=0, tolerance=0, equal=(none), lessThan=→"FINISHED", greaterThan=(none), everyFrame=false, space=0, activeBool=(unset))
    RandomFloat(min=21, max=23, storeResult=var "Spawn Y")
    on FINISHED → Aim Choice
  }

  var Spawn X: float = 0
  var Y Min: float = 11
  var Y Max: float = 23
  var Spawn Y: float = 0
  var Sickle: gameObject = <null>
}
