  start Init

  state Init {
    FindNamedChild(gameObject=Self, storeResult=var "Garmond Aid NPC")
    FindNamedChild(gameObject=Self, storeResult=var "Shakra Aid NPC")
    ActivateGameObject(gameObject=var "Garmond Aid NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Shakra Aid NPC", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Check State
  }

  state Check State {
    PlayerDataBoolTest(boolName="hang04Battle", isTrue=→"BATTLE COMPLETED", isFalse=(none))
    CheckQuestStateV2(NotTrackedEvent=(none), IncompleteEvent=(none), CompletedEvent=→"SHAKRA", Quest=Shakra Final Quest (dataassets_assets_assets/dataassets/questsystem/quests.bundle))
    PlayerDataBoolTest(boolName="garmondWillAidInForumBattle", isTrue=→"GARMOND", isFalse=(none))
    on BATTLE COMPLETED → No NPC
    on SHAKRA → Shakra
    on GARMOND → Garmond
    on FINISHED → No NPC
  }

  state Garmond {
    ActivateGameObject(gameObject=var "Garmond Aid NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Shakra {
    SetPlayerDataBool(boolName="garmondAidForumBattle", value=false)
    ActivateGameObject(gameObject=var "Shakra Aid NPC", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state No NPC {
  }

  var Garmond Aid NPC: gameObject = <null>
  var Shakra Aid NPC: gameObject = <null>
}
