fsm Splash Out Control {
  start Choose

  state Normal {
    RandomFloat(min=2.5, max=3, storeResult=var "Scaler")
    SetScale(gameObject=Self, vector=(unset), x=var "Scaler", y=var "Scaler", z=(unset), everyFrame=false, lateUpdate=false)
  }

  state Choose {
    SendRandomEvent(events=[→"NORMAL", →"MIRROR"], weights=[1, 1], delay=0)
    on NORMAL → Normal
    on MIRROR → Mirror
  }

  state Mirror {
    RandomFloat(min=3, max=2.5, storeResult=var "Scaler")
    SetScale(gameObject=Self, vector=(unset), x=var "Scaler", y=var "Scaler", z=(unset), everyFrame=false, lateUpdate=false)
    FloatMultiply(floatVariable=var "Scaler", multiplyBy=-1, everyFrame=false)
    SetScale(gameObject=Self, vector=(unset), x=var "Scaler", y=(unset), z=(unset), everyFrame=false, lateUpdate=false)
  }

  var Scaler: float = 0
}
