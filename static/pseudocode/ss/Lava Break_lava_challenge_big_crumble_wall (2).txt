# Lava Break
# 1 location(s):
#   Bone_East_LavaChallenge / lava_challenge_big_crumble_wall (2)

fsm Lava Break {
  start Idle
  on HAZARD RESPAWNED → Idle  // from any state

  state Idle {
    ActivateGameObject(gameObject=var "Active Parts", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SendMessageV2(gameObject=var "Active Parts", delivery=BroadcastMessage, options=1, functionCall=StopJitter(), everyFrame=false)
    ActivateGameObject(gameObject=var "Colliders", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Antic Effects", behaviour="PlayParticleEffects", methodName="ClearParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Break Effects", behaviour="PlayParticleEffects", methodName="ClearParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    CheckTrackTriggerCountV2(Target=var "Lava Detector 1", Count=0, Test=MoreThan, EveryFrame=true, SetBool=(unset), SuccessEvent=→"LAVA ENTERED", FailEvent=(none))
    on LAVA ENTERED → Break Antic
  }

  state Break {
    PlayAudioEvent(audioClip=explode_wall_explode (sfxstatic_assets_areawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    PlayAudioEvent(audioClip=break_wall_after_tutorial_area (sfxstatic_assets_areaantareacoralareawilds.bundle), pitchMin=0.85, pitchMax=1.15, volume=1, audioPlayerPrefab=Audio Player Actor 2D@AudioSource (audioobjects_assets_all.bundle), spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    ActivateGameObject(gameObject=var "Active Parts", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Colliders", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    CallMethodProper(gameObject=var "Antic Effects", behaviour="PlayParticleEffects", methodName="StopParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    CallMethodProper(gameObject=var "Break Effects", behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=false, Delay=0, CancelOnExit=false)
    on FINISHED → Broken
  }

  state Break Antic {
    CallMethodProper(gameObject=var "Antic Effects", behaviour="PlayParticleEffects", methodName="PlayParticleSystems", parameters=[], storeResult=(unset var), EveryFrame=false)
    SendMessageV2(gameObject=var "Active Parts", delivery=BroadcastMessage, options=1, functionCall=StartJitter(), everyFrame=false)
    CheckTrackTriggerCountV2(Target=var "Lava Detector 2", Count=0, Test=MoreThan, EveryFrame=true, SetBool=(unset), SuccessEvent=→"LAVA ENTERED", FailEvent=(none))
    on LAVA ENTERED → Break
  }

  state Broken {
  }

  var Active Parts: gameObject = lava_challenge_big_crumble_wall (2)/active parts
  var Colliders: gameObject = lava_challenge_big_crumble_wall (2)/colliders
  var Lava Detector 1: gameObject = lava_challenge_big_crumble_wall (2)/lava detector 1
  var Antic Effects: gameObject = lava_challenge_big_crumble_wall (2)/Antic Effects
  var Break Effects: gameObject = lava_challenge_big_crumble_wall (2)/Break Effects
  var Lava Detector 2: gameObject = lava_challenge_big_crumble_wall (2)/lava detector 2
}
