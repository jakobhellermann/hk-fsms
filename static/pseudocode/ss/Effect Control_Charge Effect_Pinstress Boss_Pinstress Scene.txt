fsm Effect Control {
  start Start Pause

  state Charge {
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NA Charge Effect")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hero_nail_art_charge_initiate_amplified (herosfxstatic_assets_all.bundle))
    Wait(time=0.5, finishEvent=→"FINISHED", realTime=false)
    ActivateGameObject(gameObject=var "ChargeUp", activate=true, recursive=false, resetOnExit=true, everyFrame=false)
    on FINISHED → Charged
  }

  state Charged {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    ScreenFlash(flashColour=(1, 1, 1, 0.322))
    Tk2dPlayAnimation(gameObject=Self, animLibName="", clipName="NA Charged Effect")
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=hero_nail_art_charge_complete_amplified (herosfxstatic_assets_all.bundle))
    SetMeshRenderer(gameObject=Self, active=true)
    Tk2dPlayFrame(gameObject=Self, frame=0)
    SetFsmBool(gameObject=var "Parent", fsmName="Control", variableName="Charged CrossSlash", setValue=true, everyFrame=false)
    AudioPlayInState(gameObject=Self, volume=1)
  }

  state Start Pause {
    Wait(time=0.15, finishEvent=→"FINISHED", realTime=false)
    SetMeshRenderer(gameObject=Self, active=false)
    FindNamedChild(gameObject=Self, storeResult=var "ChargeUp")
    ActivateGameObject(gameObject=var "ChargeUp", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on FINISHED → Charge
  }

  var ChargeUp: gameObject = <null>
  var Parent: gameObject = Pinstress Control/Pinstress Scene/Pinstress Boss
}
