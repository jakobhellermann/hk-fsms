fsm Control {
  start Pause

  state Pause {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Init {
    GetParent(gameObject=Self, storeResult=var "Parent")
    BoolTest(boolVariable=var "Activated", isTrue=→"ACTIVATED", isFalse=(none), everyFrame=false)
    FindNamedChild(gameObject=Self, storeResult=var "Sprite")
    FindNamedChild(gameObject=Self, storeResult=var "Phys Collider")
    FindNamedChild(gameObject=Self, storeResult=var "Centre Check")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Hit")
    FindNamedChild(gameObject=Self, storeResult=var "Pt Destroy")
    FindNamedChild(gameObject=Self, storeResult=var "Nectar Pickup")
    FindNamedChild(gameObject=var "Parent", storeResult=var "Breakable Range")
    GetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Idle Angle", space=1, everyFrame=false)
    FloatOperator(float1=var "Idle Angle", float2=-7, operation=Add, storeResult=var "Bop Angle Min", everyFrame=false)
    FloatOperator(float1=var "Idle Angle", float2=7, operation=Add, storeResult=var "Bop Angle Max", everyFrame=false)
    FloatOperator(float1=var "Idle Angle", float2=-25, operation=Add, storeResult=var "Hit Angle Min", everyFrame=false)
    FloatOperator(float1=var "Idle Angle", float2=25, operation=Add, storeResult=var "Hit Angle Max", everyFrame=false)
    on FINISHED → Deparent Phys Collider
    on ACTIVATED → Activated
  }

  state Idle {
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"STRIKE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    Collision2dEvent(gameObject=var "Phys Collider", collision=0, collideTag="", sendEvent=→"BOP", storeCollider=var "Colliding Object", storeForce=(unset))
    SetRotation(gameObject=Self, quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Idle Angle", space=1, everyFrame=false, lateUpdate=false)
    on BOP → Bop Out
    on STRIKE → Strike Check
    on HEAVY → Heavy Check
  }

  state Bop Out {
    RandomFloatEither(value1=var "Bop Angle Min", value2=var "Bop Angle Max", storeResult=var "Angle")
    AnimateRotationTo(gameObject=var "Sprite", fromValue=0, toValue=var "Angle", worldSpace=false, negativeSpace=false, time=0.075, speed=(unset), delay=(unset), easeType=21, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Wait(time=0.05, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"STRIKE", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Bop Back
    on STRIKE → Strike Check
  }

  state Bop Back {
    FloatMultiplyV2(floatVariable=var "Angle", multiplyBy=0.95, everyFrame=false, fixedUpdate=false)
    AnimateRotationTo(gameObject=var "Sprite", fromValue=var "Angle", toValue=0, worldSpace=false, negativeSpace=false, time=0.8, speed=(unset), delay=(unset), easeType=27, reverse=false, finishEvent=→"FINISHED", realTime=false)
    Collision2dEvent(gameObject=var "Phys Collider", collision=0, collideTag="", sendEvent=→"BOP", storeCollider=var "Colliding Object", storeForce=(unset))
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"STRIKE", sendEventHeavy=(none), sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Idle
    on BOP → Bop Out
    on STRIKE → Strike Check
  }

  state Strike Check {
    CheckTrackTriggerCount(target=var "Breakable Range", count=1, test=LessThan, everyFrame=false, successEvent=→"CANCEL")
    on CANCEL → Idle
    on FINISHED → Hit
  }

  state Hit {
    PlayAudioEvent(audioClip=moss_mother_small_egg_break (sfxstatic_assets_areatutorial.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Centre Check", position=(unset), rotation=(unset), storeObject=var "")
    SendEvent(eventTarget=Self, sendEvent=→"DESTROY", delay=0, everyFrame=false)  // disabled
    IntAdd(intVariable=var "Hits", add=-1, everyFrame=false)
    IntCompare(integer1=var "Hits", integer2=0, equal=→"DESTROY", lessThan=→"DESTROY", greaterThan=→"HIT", everyFrame=false)
    on HIT → Whack
    on DESTROY → Destroy
  }

  state Whack {
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitterChildren(gameObject=var "Pt Hit", resetTimeIfPlaying=false, stopOnStateExit=false)
    on FINISHED → Bop Out 2
  }

  state Bop Out 2 {
    RandomFloatEither(value1=var "Hit Angle Min", value2=var "Hit Angle Max", storeResult=var "Angle")
    AnimateRotationTo(gameObject=var "Sprite", fromValue=0, toValue=var "Angle", worldSpace=false, negativeSpace=false, time=0.075, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"STRIKE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Bop Back 2
    on STRIKE → Strike Check
    on HEAVY → Heavy Check
  }

  state Bop Back 2 {
    FloatMultiplyV2(floatVariable=var "Angle", multiplyBy=0.95, everyFrame=false, fixedUpdate=false)
    AnimateRotationTo(gameObject=var "Sprite", fromValue=var "Angle", toValue=0, worldSpace=false, negativeSpace=false, time=0.35, speed=(unset), delay=(unset), easeType=27, reverse=false, finishEvent=→"FINISHED", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=→"STRIKE", sendEventHeavy=→"HEAVY", sendEventSpikes=(none), sendEventLava=(none), sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on FINISHED → Idle
    on STRIKE → Strike Check
    on HEAVY → Heavy Check
  }

  state Destroy {
    SetCircleCollider(gameObject=Self, active=false)
    DoCameraShake(VisibleRenderer=<null>, Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Average Shake Quick (dataassets_assets_assets/dataassets/camerashake.bundle), cancelOnExit=false, DoFreeze=false, Delay=0)
    PlayParticleEmitterChildren(gameObject=var "Pt Destroy", resetTimeIfPlaying=false, stopOnStateExit=false)
    ActivateGameObject(gameObject=var "Nectar Pickup", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FireAtTarget(gameObject=var "Nectar Pickup", target=var "Centre Check", speed=10, position=(0, 0, 0), spread=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Phys Collider", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Sprite", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    StopParticleEmitter(gameObject=var "Pt IdlePetals")
  }

  state Activated {
    ActivateGameObject(gameObject=var "Parent", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Deparent Phys Collider {
    SetParent(gameObject=var "Phys Collider", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Idle
  }

  state Heavy Check {
    CheckTrackTriggerCount(target=var "Breakable Range", count=1, test=LessThan, everyFrame=false, successEvent=→"CANCEL")
    SetIntValue(intVariable=var "Hits", intValue=0, everyFrame=false)
    on CANCEL → Idle
    on FINISHED → Hit
  }

  var Angle: float = 0
  var Bop Angle Max: float = 0
  var Bop Angle Min: float = 0
  var Hit Angle Max: float = 0
  var Hit Angle Min: float = 0
  var Idle Angle: float = 0
  var Hits: int = 3
  var Activated: bool = false
  var Breakable Range: gameObject = <null>
  var Centre Check: gameObject = <null>
  var Nectar Pickup: gameObject = <null>
  var Parent: gameObject = <null>
  var Phys Collider: gameObject = <null>
  var Pt Destroy: gameObject = <null>
  var Pt Hit: gameObject = <null>
  var Sprite: gameObject = <null>
  var Pt IdlePetals: gameObject = purple_flower_set/Pt IdlePetals
  var Colliding Object: gameObject = <null>
}
