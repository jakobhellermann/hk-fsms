# Control
# 1 location(s):
#   Clover_18 / Dragonfly Large BG Variant (3)

fsm Control {
  start Init
  on DISABLE → Stop  // from any state
  on RECOIL → Recoil Wait  // from any state

  state Init {
    SetParent(gameObject=var "Nodes Parent", parent=<null>, resetLocalPosition=false, resetLocalRotation=false)
    on FINISHED → Move Wait
  }

  state Get Child {
    GetChildCount(gameObject=var "Nodes Parent", storeResult=var "Node Count")
    IntAddWrapped(Value=var "Node Index", AddValue=var "Node Index Add", Min=0, Max=var "Node Count", InclusiveMax=false, StoreResult=var "Node Index")
    GetChildNum(gameObject=var "Nodes Parent", childIndex=var "Node Index", store=var "Next Node")
    GameObjectIsActive(Target=var "Next Node", ActiveSpace=1, trueEvent=(none), falseEvent=→"LOOP", storeValue=(unset))
    on FINISHED → Move Pos
    on LOOP → Get Child
  }

  state Move Pos {
    PlayAudioEventRandom(audioClips=[dragonfly_large_dash_1 (sfxstatic_assets_areaclover.bundle), dragonfly_large_dash_2 (sfxstatic_assets_areaclover.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "Current Audio")
    GetPosition2dAs3d(SourceXY=var "Next Node", SourceZ=Self, StoreVector=var "Next Pos")
    SetFsmVector3(gameObject=var "Sprite", fsmName="Control", variableName="Start Pos", setValue=var "Next Pos", everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Sprite"), sendEvent="REFRESH", delay=0, everyFrame=false)
    iTweenMoveTo(gameObject=Self, id="", transformPosition=var "", vectorPosition=var "Next Pos", time=(unset), delay=0, speed=3, space=0, easeType=10, loopType=0, orientToPath=false, lookAtObject=var "", lookAtVector=(unset), lookTime=(unset), axis=none, moveToPath=false, lookAhead=(unset), transforms=[], vectors=[], reverse=false, startEvent=(none), finishEvent=→"FINISHED", realTime=false, stopOnExit=true, loopDontFinish=false)
    FaceObjectV3(ObjectA=var "Sprite", ObjectB=var "Next Node", SpriteFacesRight=false, NewAnimationClip="TurnToFly", ResetFrame=true, PauseBetweenTurns=0.5, StoreDuration=(unset), EveryFrame=false)
    on FINISHED → Move Wait
  }

  sequence state Antic {
    Tk2dPlayAnimationWait(Target=var "Sprite", ClipName="Antic", AnimationCompleteEvent=(none))
    Tk2dPlayAnimationWithEvents(gameObject=var "Sprite", clipName="Fly", animationTriggerEvent=(none), animationCompleteEvent=(none))
    on FINISHED → Reverse?
  }

  state Move Wait {
    WaitRandom(timeMin=0.2, timeMax=0.5, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Antic
    on TOOK DAMAGE → Antic
  }

  state Reverse? {
    SendRandomEventV4(events=[→"FINISHED", (none)], weights=[0.75, 0.25], eventMax=[5, 1], missedMax=[1, 5], activeBool=(unset))
    SendRandomEventFair(Events=[{SendEvent=→"FINISHED", Probability=0.7}, {SendEvent=(none), Probability=0.3}], TrackingArray=var "Reverse Probabilities", MissedMultiplier=2)  // disabled
    IntOperator(integer1=var "Node Index Add", integer2=-1, operation=Multiply, storeResult=var "Node Index Add", everyFrame=false)
    on FINISHED → Get Child
  }

  state Stop {
    AudioStop(gameObject=var "Current Audio", fadeTime=0)
  }

  state Recoil Wait {
    on RECOIL END → Antic
  }

  var Node Count: int = 0
  var Node Index: int = 0
  var Node Index Add: int = 1
  var Reset Node Loop: bool = false
  var Next Pos: vector3 = (0, 0, 0)
  var Nodes Parent: gameObject = Dragonfly Large BG Variant (3)/Nodes
  var Next Node: gameObject = <null>
  var Sprite: gameObject = Dragonfly Large BG Variant (3)/Sprite
  var Current Audio: gameObject = <null>
  var Audio Loop Sing: gameObject = <null>
  var Reverse Probabilities: array = var "Reverse Probabilities"
}
