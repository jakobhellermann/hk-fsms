# Text Control
# 1 location(s):
#   Cinematic_MrMushroom / ToBeContinued

fsm Text Control {
  start Wait

  state Idle {
    AwardQueuedAchievements(delay=0)
    ActivateGameObject(gameObject=var "TextMesh", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    on CINEMATIC END → Show Text
  }

  state Show Text {
    ActivateGameObject(gameObject=var "TextMesh", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    Wait(time=2, finishEvent=→"FADE OUT", realTime=false)
    on FADE OUT → Fade
  }

  state Wait {
    NextFrameEvent(sendEvent=→"BEGIN")
    on BEGIN → Idle
  }

  sequence state Fade {
    SendEventByNameV2(eventTarget=GameObject(Cutscene Blanker), sendEvent="JUST FADE", delay=0, everyFrame=false)
    Wait(time=2, finishEvent=(none), realTime=false)
    QueueAchievement(Key="HERALD_WISH")
    Wait(time=0.5, finishEvent=(none), realTime=false)
    on FINISHED → Load Next Level
  }

  state Load Next Level {
    GetConstantsValue(variableName="END_CREDITS_SCROLL", storeValue=var "Scene")
    CallMethodProper(gameObject=var "GameManager", behaviour="GameManager", methodName="LoadScene", parameters=[var "Scene"], storeResult=(unset var), EveryFrame=false)
  }

  var TextMesh: gameObject = ToBeContinued/TextMeshPro

  var Scene: string = ""
}
