# Control
# 1 location(s):
#   Dock_16 / Flea Rescue BlackSilk

fsm Control {
  start Wait

  state Init {
    CheckQuestPdSceneBool(QuestTarget=var "Quest Target", ExpectedValue=true, trueEvent=→"ACTIVATE", falseEvent=(none), storeValue=(unset))
    AnimatorPlay(gameObject=var "Thread A", stateName="Closed", layer=(unset), normalizedTime=0, everyFrame=false)
    AnimatorPlay(gameObject=var "Thread B", stateName="Closed", layer=(unset), normalizedTime=0, everyFrame=false)
    on ACTIVATE → Activate
    on FINISHED → Idle
  }

  state Activate {
    ActivateGameObject(gameObject=Self, activate=false, recursive=false, resetOnExit=false, everyFrame=false)
  }

  state Idle {
    AudioPlaySimple(gameObject=var "Audio Loop Voice", volume=1, oneShotClip=<null>)
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"CUT", storeCollider=var "")
    on CUT → Cut
  }

  state Cut Loose {
    AudioStop(gameObject=var "Audio Loop Voice", fadeTime=0)
    PlayParticleEmitter(gameObject=var "Pt Destroy", emit=0, resetIfPlaying=false)
    PlayAudioEvent(audioClip=silk_prop_kill (herosfxstatic_assets_all.bundle), pitchMin=1, pitchMax=1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    SendEventByName(eventTarget=GameObject(var "Notfify"), sendEvent="FLEA RESCUED", delay=0, everyFrame=false)
    ActivateGameObject(gameObject=var "Threads", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flea", activate=false, recursive=false, resetOnExit=false, everyFrame=false)
    ActivateGameObject(gameObject=var "Flea Rescue", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    SetFsmBool(gameObject=var "Flea Rescue", fsmName="Control", variableName="Emerge From Hero", setValue=true, everyFrame=false)
    SendEventByName(eventTarget=GameObject(var "Flea"), sendEvent="RESCUE", delay=0, everyFrame=false)  // disabled
    SetPolygonCollider(gameObject=Self, active=false, resetOnExit=false)
  }

  state Wait {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Init
  }

  state Cut {
    DoCameraShakeV2(Target=Self, MaxCameraDistance=(unset), Camera=Main Camera (dataassets_assets_assets/dataassets/camerashake.bundle), Profile=Enemy Kill (dataassets_assets_assets/dataassets/camerashake.bundle), DoFreeze=true, Delay=0, CancelOnExit=false)
    PlayAudioEventRandom(audioClips=[silk_prop_slash_1 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_2 (sfxstatic_assets_areabellshrine.bundle), silk_prop_slash_3 (sfxstatic_assets_areabellshrine.bundle)], pitchMin=0.7, pitchMax=0.8, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    IntAdd(intVariable=var "HP", add=-1, everyFrame=false)
    SpawnObjectFromGlobalPool(gameObject=Simple Strike Effect (globalpoolprefabs_assets_all.bundle), spawnPoint=var "Flea", position=(unset), rotation=(unset), storeObject=var "")
    Wait(time=0.1, finishEvent=→"FINISHED", realTime=false)
    AnimatorPlay(gameObject=var "Thread A", stateName="Wobble", layer=(unset), normalizedTime=0, everyFrame=false)
    AnimatorPlay(gameObject=var "Thread B", stateName="Wobble", layer=(unset), normalizedTime=0, everyFrame=false)
    IntCompare(integer1=var "HP", integer2=0, equal=→"BREAK", lessThan=→"BREAK", greaterThan=(none), everyFrame=false)
    PlayParticleEmitter(gameObject=var "Pt Cut", emit=0, resetIfPlaying=false)
    on FINISHED → Idle
    on BREAK → Cut Loose
  }

  var Notfify: gameObject = <null>

  var HP: int = 3
  var Flea: gameObject = Flea Rescue BlackSilk/Flea Distressed
  var Pt Cut: gameObject = Flea Rescue BlackSilk/Pt Cut
  var Pt Destroy: gameObject = Flea Rescue BlackSilk/Pt Destroy
  var Thread A: gameObject = Flea Rescue BlackSilk/Threads/Thread A
  var Thread B: gameObject = Flea Rescue BlackSilk/Threads/Thread B
  var Threads: gameObject = Flea Rescue BlackSilk/Threads
  var Flea Rescue: gameObject = Flea Rescue BlackSilk/Flea Rescue Activation
  var Audio Loop Voice: gameObject = Flea Rescue BlackSilk/Audio Loop Voice
  var Quest Target: object = FleasCollected Target (dataassets_assets_assets/dataassets/questsystem/proxies.bundle)
}
