fsm Timer {
  start Tick

  state Tick {
    SendEventByName(eventTarget=GameObject(Self), sendEvent="TICK", delay=0, everyFrame=false)
    Wait(time=var "Tick Time", finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Tick
  }

  var Tick Time: float = 2
}
