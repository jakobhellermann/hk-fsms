fsm Control {
  start Idle

  state Idle {
    on AIM → Get Target
  }

  state Get Target {
    FindClosest(gameObject=Self, withTag="Song Knight Target", ignoreOwner=true, mustBeVisible=false, storeObject=var "Current Target", storeDistance=(unset), everyFrame=false)
    GameObjectIsNull(gameObject=var "Current Target", isNull=→"NO TARGET", isNotNull=→"FOLLOW", storeResult=(unset), everyFrame=false)
    on FOLLOW → Follow Target
    on NO TARGET → No Targets Left
  }

  state Follow Target {
    SetPositionToObject2D(gameObject=Self, targetObject=var "Current Target", xOffset=0, yOffset=0, everyFrame=true)
    GameObjectIsNull(gameObject=var "Current Target", isNull=→"NO TARGET", isNotNull=(none), storeResult=(unset), everyFrame=true)
    SetBoolValue(boolVariable=var "No Valid Target", boolValue=false, everyFrame=false)
    on NO TARGET → Get Target
  }

  state No Targets Left {
    SetBoolValue(boolVariable=var "No Valid Target", boolValue=true, everyFrame=false)
  }

  var No Valid Target: bool = false
  var Current Target: gameObject = <null>
}
