# proxy_event_handler
# 1 location(s):
#   prompts_assets_all.bundle / UI Msg Get Map/Button

fsm proxy_event_handler {
  start Init

  state Init {
    on DOWN → Send Down
    on UP → Send Up
  }

  state Send Down {
    SendEventByName(eventTarget=GameObject(UI Msg Get Map/Button/Label), sendEvent="DOWN", delay=0, everyFrame=false)
    on FINISHED → Init
  }

  state Send Up {
    SendEventByName(eventTarget=GameObject(UI Msg Get Map/Button/Label), sendEvent="UP", delay=0, everyFrame=false)
    on FINISHED → Init
  }
}
