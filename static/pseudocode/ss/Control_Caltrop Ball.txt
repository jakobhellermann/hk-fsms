# Control
# 1 location(s):
#   localpoolprefabs_assets_areadust.bundle / Caltrop Ball

fsm Control {
  start Init

  state Init {
    SetAudioSource(gameObject=Self, active=true)
    SetMeshRenderer(gameObject=Self, active=true)
    SetIsKinematic2d(gameObject=Self, isKinematic=false)
    SetCircleCollider(gameObject=Self, active=true)
    on FINISHED → Air
  }

  state Air {
    Collision2dEvent(gameObject=var "Terrain Detector", collision=0, collideTag="", sendEvent=→"TERRAIN", storeCollider=var "", storeForce=(unset))
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Terrain(8), sendEvent=→"TERRAIN", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="Nail Attack", collideLayer=(unset), sendEvent=→"NAIL BREAK", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Hero Box(20), sendEvent=→"BREAK", storeCollider=var "")
    Trigger2dEventLayer(trigger=OnTriggerEnter2D, collideTag="", collideLayer=Player(9), sendEvent=→"BREAK", storeCollider=var "")
    Wait(time=4, finishEvent=→"CANCEL", realTime=false)
    ReceivedDamage(Target=Self, collideTag="", sendEvent=(none), sendEventHeavy=→"PROJECTILE BREAK", sendEventSpikes=→"PROJECTILE BREAK", sendEventLava=→"PROJECTILE BREAK", sendEventLightning=(none), storeGameObject=var "", ignoreAcid=false, ignoreLava=false, ignoreWater=false, ignoreHunterWeapon=false, ignoreTraps=false, ignoreNail=false, ignoreSpikes=false, storeDamageDealt=(unset), storeDirection=(unset), storeMagnitudeMultiplier=(unset), firstHitOnly=false)
    on BREAK → Other
    on CANCEL → Recycle
    on TERRAIN → Terrain
    on PROJECTILE BREAK → Other
    on DEALT DAMAGE → Terrain
    on TINK LEFT → Tink Break
    on TINK RIGHT → Tink Break
    on TINK UP → Tink Break
    on TINK DOWN → Tink Break
  }

  state Break {
    SetAudioSource(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Caltrop, spawnPoint=var "Burst", position=(0.2, 0, -0.001), spawnMin=1, spawnMax=1, speedMin=15, speedMax=20, angleMin=70, angleMax=70, originVariationX=0.2, originVariationY=0.2, FSM="", FSMEvent="")
    FlingObjectsFromGlobalPool(gameObject=Caltrop, spawnPoint=var "Burst", position=(-0.2, 0, -0.0011), spawnMin=1, spawnMax=1, speedMin=15, speedMax=20, angleMin=110, angleMax=110, originVariationX=0.2, originVariationY=0.2, FSM="", FSMEvent="")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Recycle {
    RecycleSelf()
  }

  state Tink Break {
    SetAudioSource(gameObject=Self, active=false)
    SetIsKinematic2d(gameObject=Self, isKinematic=true)
    SetVelocity2d(gameObject=Self, vector=(0, 0), x=(unset), y=(unset), everyFrame=false)
    SetMeshRenderer(gameObject=Self, active=false)
    ActivateGameObject(gameObject=var "Burst", activate=true, recursive=false, resetOnExit=false, everyFrame=false)
    GetAngleToTarget2D(gameObject=Self, target=var "Hero", offsetX=0, offsetY=0, storeAngle=var "Angle", pause=0, everyFrame=false)
    FloatAdd(floatVariable=var "Angle", add=180, everyFrame=false, perSecond=false)
    FloatOperator(float1=var "Angle", float2=50, operation=Subtract, storeResult=var "Angle Min", everyFrame=false)
    FloatOperator(float1=var "Angle", float2=50, operation=Add, storeResult=var "Angle Max", everyFrame=false)
    FlingObjectsFromGlobalPool(gameObject=Caltrop, spawnPoint=var "Burst", position=(unset), spawnMin=2, spawnMax=2, speedMin=20, speedMax=25, angleMin=var "Angle Min", angleMax=var "Angle Max", originVariationX=0.2, originVariationY=0.2, FSM="Control", FSMEvent="START TINK")
    Wait(time=1, finishEvent=→"FINISHED", realTime=false)
    SetCircleCollider(gameObject=Self, active=false)
    on FINISHED → Recycle
  }

  state Terrain {
    PlayAudioEventRandom(audioClips=[caltrop_ball_fast_impact_1 (sfxstatic_assets_areagreymoor.bundle), caltrop_ball_fast_impact_2 (sfxstatic_assets_areagreymoor.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Break
  }

  state Other {
    PlayAudioEventRandom(audioClips=[caltrop_ball_slow_impact_1 (sfxstatic_assets_areagreymoor.bundle), caltrop_ball_slow_impact_2 (sfxstatic_assets_areagreymoor.bundle)], pitchMin=0.95, pitchMax=1.05, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    on FINISHED → Break
  }

  var Angle: float = 0
  var Angle Max: float = 0
  var Angle Min: float = 0
  var Burst: gameObject = Caltrop Ball/Burst
  var Terrain Detector: gameObject = Caltrop Ball/Terrain Detector
}
