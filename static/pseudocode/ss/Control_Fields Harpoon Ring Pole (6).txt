# Control
# 1 location(s):
#   Bone_East_24 / Fields Harpoon Ring Pole (6)

fsm Control {
  start Init

  state Init {
    NextFrameEvent(sendEvent=→"FINISHED")
    on FINISHED → Idle
  }

  state Idle {
    SendMessageV2(gameObject=var "Spline", delivery=BroadcastMessage, options=0, functionCall=ForceFollowUpdate(), everyFrame=false)
    CallMethodProper(gameObject=var "Spline", behaviour="SplineBase", methodName="UpdateSpline", parameters=[], storeResult=(unset var), EveryFrame=false)
    on CONNECT L → Set L
    on CONNECT R → Set R
  }

  state Rotate {
    AudioPlaySimple(gameObject=Self, volume=1, oneShotClip=<null>)
    Rotate(gameObject=var "Pivot", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SendMessageV2(gameObject=var "Spline", delivery=BroadcastMessage, options=0, functionCall=ForceFollowUpdate(), everyFrame=true)
    CallMethodProper(gameObject=var "Spline", behaviour="SplineBase", methodName="UpdateSpline", parameters=[], storeResult=(unset var), EveryFrame=true)
    FloatAdd(floatVariable=var "Rotate Speed", add=var "Rotate Accel", everyFrame=true, perSecond=true)
    GetRotation(gameObject=var "Pivot", quaternion=(unset), vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Current Rotation", space=0, everyFrame=true)
    FloatCompare(float1=var "Current Rotation", float2=var "Rotate Max", tolerance=0, equal=→"MAX", lessThan=(none), greaterThan=→"MAX", everyFrame=true)
    FloatCompare(float1=var "Current Rotation", float2=var "Rotate Min", tolerance=0, equal=→"MAX", lessThan=→"MAX", greaterThan=(none), everyFrame=true)
    on DISCONNECT → Rebound
    on MAX → Max Strain
  }

  state Set L {
    SetFloatValue(floatVariable=var "Rotate Accel", floatValue=100, everyFrame=false)
    on FINISHED → Connect Pause
  }

  state Set R {
    SetFloatValue(floatVariable=var "Rotate Accel", floatValue=-100, everyFrame=false)
    on FINISHED → Connect Pause
  }

  state Rebound L {
    AnimateRotationTo(gameObject=var "Pivot", fromValue=(unset), toValue=0, worldSpace=false, negativeSpace=false, time=1, speed=(unset), delay=(unset), easeType=27, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SendMessageV2(gameObject=var "Spline", delivery=BroadcastMessage, options=0, functionCall=ForceFollowUpdate(), everyFrame=true)
    CallMethodProper(gameObject=var "Spline", behaviour="SplineBase", methodName="UpdateSpline", parameters=[], storeResult=(unset var), EveryFrame=true)
    on FINISHED → Idle
    on CONNECT R → Set R
    on CONNECT L → Set L
  }

  state Connect Pause {
    SetFloatValue(floatVariable=var "Rotate Speed", floatValue=0, everyFrame=false)
    Wait(time=0.25, finishEvent=→"FINISHED", realTime=false)
    on FINISHED → Rotate
    on DISCONNECT → Idle
  }

  state Hit Max {
    SendEventByName(eventTarget=GameObject(var "Hero"), sendEvent="FORCE DISCONNECT", delay=0, everyFrame=false)
    on FINISHED → Rebound
  }

  state Max Strain {
    SetFloatValue(floatVariable=var "Rotate Tween Start", floatValue=var "Rotate Speed", everyFrame=false)
    ObjectJitterOnRender(Target=Self, X=0.1, Y=0.1, Z=(unset), LimitFps=30)
    Wait(time=0.9, finishEvent=→"FINISHED", realTime=false)
    Rotate(gameObject=var "Pivot", vector=(unset), xAngle=(unset), yAngle=(unset), zAngle=var "Rotate Speed", space=0, perSecond=true, everyFrame=true, lateUpdate=false, fixedUpdate=false)
    SendMessageV2(gameObject=var "Spline", delivery=BroadcastMessage, options=0, functionCall=ForceFollowUpdate(), everyFrame=true)
    CallMethodProper(gameObject=var "Spline", behaviour="SplineBase", methodName="UpdateSpline", parameters=[], storeResult=(unset var), EveryFrame=true)
    EaseFloat(fromValue=var "Rotate Tween Start", toValue=0, floatVariable=var "Rotate Speed", time=0.4, speed=(unset), delay=(unset), easeType=13, reverse=false, finishEvent=(none), realTime=false)
    on DISCONNECT → Rebound
    on FINISHED → Hit Max
  }

  state Rebound R {
    AnimateRotationTo(gameObject=var "Pivot", fromValue=(unset), toValue=0, worldSpace=false, negativeSpace=true, time=1, speed=(unset), delay=(unset), easeType=27, reverse=false, finishEvent=→"FINISHED", realTime=false)
    SendMessageV2(gameObject=var "Spline", delivery=BroadcastMessage, options=0, functionCall=ForceFollowUpdate(), everyFrame=true)
    CallMethodProper(gameObject=var "Spline", behaviour="SplineBase", methodName="UpdateSpline", parameters=[], storeResult=(unset var), EveryFrame=true)
    on FINISHED → Idle
    on CONNECT R → Set R
    on CONNECT L → Set L
  }

  state Rebound {
    AudioStopV2(gameObject=Self, fadeTime=0, cancelOnEarlyExit=false)
    PlayAudioEvent(audioClip=harpoon_ring_pole_release (sfxstatic_assets_areawilds.bundle), pitchMin=0.9, pitchMax=1.1, volume=1, audioPlayerPrefab=var "", spawnPoint=Self, spawnPosition=(0, 0, 0), SpawnedPlayerRef=var "")
    FloatCompare(float1=var "Rotate Accel", float2=0, tolerance=0, equal=→"R", lessThan=→"R", greaterThan=→"L", everyFrame=false)
    on L → Rebound L
    on R → Rebound R
  }

  var Rotate Min: float = 60
  var Rotate Max: float = 120

  var Rotate Speed: float = 0
  var Rotate Accel: float = 0
  var Current Rotation: float = 0
  var Rotate Tween Start: float = 0
  var Pivot: gameObject = Fields Harpoon Ring Pole (6)/Pivot
  var Spline: gameObject = Fields Harpoon Ring Pole (6)/pole_bend
}
